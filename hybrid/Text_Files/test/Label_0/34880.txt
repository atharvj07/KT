import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class ProblemC {
	
	public static void main(String[] args) throws IOException {
		BufferedReader s = new BufferedReader(new InputStreamReader(System.in));
		PrintWriter out = new PrintWriter(System.out);
		String[] num = s.readLine().split(" ");
		long[] a = new long[4];
		for (int i = 0 ; i < 4 ; i++) {
			a[i] = Integer.valueOf(num[i]);
		}
		
		
		int tr = 2000;
		List<Integer> operation = new ArrayList<Integer>();
		for (int i = 0 ; i < 4 ; i++) {
			if (a[i] == 0) {
				a[i]++;
				a[(i+1)%4]++;
				operation.add(2);
				operation.add(i+1);
			}
		}
		
		while (--tr >= 0) {
			if (a[0] + a[1] + a[2] + a[3] == 4) {
				break;
			}

			// try divide
			int fnd = -1;
			for (int i = 0 ; i < 4 ; i++) {
				int u = i;
				int v = (i+1)%4;
				if (a[u] % 2 == 0 && a[v] % 2 == 0) {
					operation.add(1);
					operation.add((i+1));
					a[u] /= 2;
					a[v] /= 2;
					fnd = i;
					break;
				}
			}
			if (fnd == -1) {
				// try add two
				fnd = -1;
				for (int i = 0 ; i < 4 ; i++) {
					int u = i;
					int v = (i+1)%4;
					if (a[u] == 1 && a[v] == 1) {
					} else if (a[u] % 2 == 1 && a[v] % 2 == 1) {
						a[u]++;
						a[v]++;
						operation.add(2);
						operation.add(i+1);
						fnd = i;
						break;
					}
				}
			}
			if (fnd == -1) {
				for (int i = 0 ; i < 4 ; i++) {
					int u = i;
					int v = (i+1)%4;
					int w = (i-1+4)%4;
					if (a[u] % 2 == 0 && a[v] % 2 == 1 && a[w] % 2 == 1) {
						a[u] += 2;
						a[v]++;
						a[w]++;
						operation.add(2);
						operation.add((i-1+4)%4+1);
						operation.add(2);
						operation.add(i+1);
						fnd = i;
						break;
					}
				}
			}
		}
		if (tr <= 0) {
			out.println(-1);
		} else {
			int len = operation.size() / 2;
			for (int i = 0 ; i < len ; i++) {
				int op = operation.get(i*2);
				int arg = operation.get(i*2+1);
				if (op == 1) {
					out.print('/');
				} else {
					out.print('+');
				}
				out.println(arg);
			}
		}
		out.flush();
	}


	public static void debug(Object... os){
		System.err.println(Arrays.deepToString(os));
	}
}
