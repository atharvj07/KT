import java.util.Scanner;


public class B {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int x = sc.nextInt();
        int y = sc.nextInt();
        if (y % a==0) {
            System.out.println(-1);
            return;
        }
        if (y >0 && y < a && Math.abs(x) < a/2.0) {
            System.out.println(1);
            return;
        }
        int t1 = 0, t2 = 0, ad2 = 2, ad1 = 0;
        for (int i = 2; i <= 1e6+2; i += 2) {
            if (y > i*a && y < (i+1)*a) {
                t2 = i+ad2;
                break;
            }
            ad2++;
            if (y > (i-1)*a && y < i*a) {
                t1 = i+ad1;
                break;
            }
            ad1++;
        }
        if (t1 != 0) {
            if (Math.abs(x) < a/2.0) {
                System.out.println(t1);
            }
            else
                System.out.println(-1);
        }
        else {
            if (t2 != 0) {
                if (x > 0 && x < a) {
                    System.out.println(t2);
                }
                else {
                    if (x < 0 && x > -a) {
                        System.out.println(t2-1);
                    }
                    else
                        System.out.println(-1);
                }
            }
            else
                System.out.println(-1);
        }
    }
}
