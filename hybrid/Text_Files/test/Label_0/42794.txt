from sys import stdin
input = stdin.readline
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	a = list(map(int, input().split()))
	(b, c) = ([], [])
	for x in a:
		if x > 0:
			b.append(x)
		elif x < 0:
			c.append(-x)
	(nb, nc) = (len(b), len(c))
	res = 0
	b.sort()
	c.sort()
	if nb == nc == 0:
		print(0)
		continue
	if nc == 0:
		(nb, nc) = (nc, nb)
		(b, c) = (c, b)
	if (nb > 0 and nc > 0) and max(b) > max(c):
		(nb, nc) = (nc, nb)
		(b, c) = (c, b)
	if nb:
		for x in range(nb - 1, -1, -k):
			res += b[x] * 2
	for x in range(nc - 1, -1, -k):
		res += c[x] * 2
	print(res - c[-1])
