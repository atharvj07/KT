(n, m) = map(int, input().split())
le = []
ans = [1] * n
ans1 = []
flag = True
for i in range(n):
	xx = list(map(int, input().split()))
	ans1.append(xx[0])
	if flag:
		f1 = False
		for j in range(m):
			if xx[j] != xx[0]:
				f1 = True
				flag = False
				break
		if f1:
			le = [i, 0, j]
su = 0
for i in range(n):
	su ^= ans1[i]
if su != 0:
	print('TAK')
	print(*ans)
elif le:
	ans[le[0]] = le[2] + 1
	print('TAK')
	print(*ans)
else:
	print('NIE')
