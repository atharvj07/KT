def cs(a):
	for i in range(len(a) - 1):
		if a[i] > a[i + 1]:
			return False
	return True
n = int(input())
l = list(map(int, input().split()))
if cs(l) == False:
	for i in range(len(l) - 1):
		if l[i] > l[i + 1]:
			if cs(l[i + 1:]) and l[-1] <= l[0]:
				print(len(l) - i - 1)
			else:
				print(-1)
			break
else:
	print(0)
