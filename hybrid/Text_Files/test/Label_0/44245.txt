t = int(input())
for _ in range(t):
	n = int(input())
	mat = []
	for i in range(n):
		mat.append(list(input()))
	score = []
	for i in range(n):
		score.append([0] * n)
	for i in range(n):
		for j in range(n):
			if mat[i][j] != '.':
				for di in [-2, -1, 0]:
					if 0 <= i + di and i + di + 2 < n:
						trio = True
						if mat[i + di][j] != 'X' or mat[i + di + 1][j] != 'X' or mat[i + di + 2][j] != 'X':
							trio = False
						if trio:
							score[i][j] += 1
				for dj in [-2, -1, 0]:
					if 0 <= j + dj and j + dj + 2 < n:
						trio = True
						if mat[i][j + dj] != 'X' or mat[i][j + dj + 1] != 'X' or mat[i][j + dj + 2] != 'X':
							trio = False
						if trio:
							score[i][j] += 1
	oo = [0, 0, 0]
	for i in range(n):
		for j in range(n):
			if score[i][j] > 0:
				oo[(i + j) % 3] += 1
	m = 0
	if oo[1] < oo[m]:
		m = 1
	if oo[2] < oo[m]:
		m = 2
	for i in range(n):
		for j in range(n):
			if (i + j) % 3 == m:
				if score[i][j] > 0:
					mat[i][j] = 'O'
	for i in range(n):
		print(''.join(mat[i]))
