#include <iostream>
#include <algorithm>

using namespace std;

int main(){
  int T, dp[100][100], n, d[100], s, t;
  cin >> T;
  while(T--){
    cin >> n;
    for(int i=0;i<n;i++) cin >> d[i];
    fill(dp[0], dp[n], 0);
    for(int k=1;k<n;k++){
      for(int i=0;i<n;i++){
        for(int j=0;j<k;j++){
          s = (i+k)%n;
          t = (i+j+1)%n;
          dp[i][s] = max(dp[i][s], dp[i][(i+j)%n] + dp[t][s] + abs(d[i] - d[t]));
        }
      }
    }
    int ans = 0;
    for(int i=0;i<n;i++) ans = max(ans, dp[i][(i+n-1)%n]);
    cout << ans << endl;
  }
}
