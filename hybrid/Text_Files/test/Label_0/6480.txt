#include <bits/stdc++.h>
using namespace std;
const int N = 5e5 + 500;
const int M = 1e6 + 500;
const long long INF = 1e18;
const int MOD = 1e9 + 7;
const int LOG = 20;
const int ALP = 30;
const int BASE = 31337;
const double EPS = 1e-9;
const double PI = 3.1415926535;
const int OFF = (1 << 19);
long long tour[2 * OFF];
int A[N], p[N], B[N];
int n, m;
void update(int i, int a, int b, int lo, int hi, long long x) {
  if (lo <= a && b <= hi) {
    tour[i] += x;
    return;
  }
  if (a > hi || b < lo) return;
  update(2 * i, a, (a + b) / 2, lo, hi, x);
  update(2 * i + 1, (a + b) / 2 + 1, b, lo, hi, x);
}
long long query(int i) {
  long long ret = 0;
  for (i = i + OFF; i; i /= 2) ret += tour[i];
  return ret;
}
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; i++) scanf("%d", A + i);
  for (int i = 0; i < n; i++) scanf("%d", p + i);
  scanf("%d", &m);
  for (int i = 0; i < m; i++) scanf("%d", B + i);
  tour[1] = INF, tour[OFF] = -INF;
  int postao = 0;
  for (int i = 0; i < n; i++) {
    int kol = lower_bound(B, B + m, A[i]) - B;
    long long pos = query(kol);
    if (p[i] >= 0)
      update(1, 0, OFF - 1, 0, kol - (B[kol] == A[i]), p[i]);
    else
      update(1, 0, OFF - 1, 0, m, p[i]);
    if (kol != m && B[kol] == A[i]) {
      if (query(kol + 1) > pos) {
        update(1, 0, OFF - 1, kol + 1, kol + 1, pos - query(kol + 1));
      }
      if (kol + 1 == postao + 1) postao++;
    }
  }
  if (postao < m)
    printf("NO\n");
  else
    printf("YES\n%lld\n", query(m));
  return 0;
}
