#include <bits/stdc++.h>
using namespace std;
const int nmax = 1e5;
int n, t, valmax;
int v[nmax + 1], nxt[nmax + 1];
pair<int, int> d[2][nmax + 1];
void solve() {
  for (int i = 1; i <= n; ++i) {
    fscanf(stdin, "%d", &v[i]);
  }
  int ind = 1;
  for (int i = 0; i <= n; ++i) d[0][i] = {0, 1 << 29};
  d[0][0] = {n, 0};
  for (int i = 1; i <= valmax; ++i, ind = 1 - ind) {
    nxt[0] = 0;
    for (int j = 1; j <= n; ++j) {
      if (v[j] == i) {
        nxt[0] = j;
        break;
      }
    }
    nxt[n] = nxt[0];
    for (int j = n - 1; j >= 0; --j) {
      nxt[j] = nxt[j + 1];
      if (v[j + 1] == i) nxt[j] = j + 1;
    }
    d[ind][0] = {n, 0};
    if (nxt[0] > 0) {
      for (int j = 1; j <= n; ++j) {
        int x = nxt[d[1 - ind][j - 1].first];
        pair<int, int> shp = d[1 - ind][j - 1];
        if (x <= shp.first) {
          ++shp.second;
        }
        shp.first = x;
        if (1LL * n * shp.second + shp.first <
            1LL * n * d[1 - ind][j].second + d[1 - ind][j].first) {
          d[ind][j] = shp;
        } else {
          d[ind][j] = d[1 - ind][j];
        }
      }
    } else {
      for (int j = 1; j <= n; ++j) d[ind][j] = d[1 - ind][j];
    }
  }
  ind = 1 - ind;
  int ans = 0;
  for (int j = 1; j <= n; ++j) {
    if (d[ind][j].second <= t) ans = j;
  }
  fprintf(stdout, "%d\n", ans);
}
int main() {
  int nrt;
  fscanf(stdin, "%d%d%d%d", &nrt, &n, &valmax, &t);
  while (nrt--) {
    solve();
  }
  return 0;
}
