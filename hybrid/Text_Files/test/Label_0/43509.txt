for _ in range(int(input())):
	(W, H) = map(int, input().split())
	(x1, y1, x2, y2) = map(int, input().split())
	(w, h) = map(int, input().split())
	if w * h > W * H - (x2 - x1) * (y2 - y1) or (w + (x2 - x1) > W and h + (y2 - y1) > H):
		print(-1)
	else:
		maxH = max(y1, H - y2)
		maxW = max(x1, W - x2)
		if maxH == 0 and maxW == 0 or maxH >= h or maxW >= w:
			print(0)
		elif maxW == 0 or x1 + W - x2 < w:
			print(h - maxH)
		elif maxH == 0 or y1 + H - y2 < h:
			print(w - maxW)
		else:
			print(min(h - maxH, w - maxW))
