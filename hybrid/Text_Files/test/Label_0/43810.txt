import sys
import io
import math
import collections
import decimal
import itertools
import bisect
import heapq

def input():
	return sys.stdin.readline()[:-1]
INF = 10 ** 10
MOD = 1000000007
K = int(input())
N = int(input())
Info = [list() for _ in range(K)]
color_dic = {'white': 0, 'green': 1, 'red': 2, 'yellow': 3, 'blue': 4, 'orange': 5}
for _ in range(N):
	(v, col) = input().split()
	d = 0
	v1 = int(v)
	while True:
		if v1 <= 2 ** d:
			break
		v1 -= 2 ** d
		d += 1
	Info[d].append((int(v), color_dic[col]))
for d in range(K):
	Info[d].sort()
dp = [[0] * 6 for _ in range(K)]
dp_sp = [collections.defaultdict(lambda : [0] * 6) for _ in range(K)]
for c in range(6):
	dp[K - 1][c] = 1
for (v, c) in Info[K - 1]:
	T = [0] * 6
	T[c] = 1
	dp_sp[K - 1][v] = T
for k in reversed(range(K - 1)):
	for c in range(6):
		for c1_1 in range(6):
			for c1_2 in range(6):
				if (c1_1 - c) % 3 and (c1_2 - c) % 3:
					dp[k][c] = (dp[k][c] + dp[k + 1][c1_1] * dp[k + 1][c1_2]) % MOD
	for (v1, t1) in dp_sp[k + 1].items():
		if v1 % 2 == 0 and v1 + 1 in dp_sp[k + 1]:
			v2 = v1 + 1
			t2 = dp_sp[k + 1][v2]
			T = [0] * 6
			for c in range(6):
				for c1_1 in range(6):
					for c1_2 in range(6):
						if (c1_1 - c) % 3 and (c1_2 - c) % 3:
							T[c] = (T[c] + t1[c1_1] * t2[c1_2]) % MOD
			dp_sp[k][v1 // 2] = T
			continue
		if v1 % 2 == 1 and v1 - 1 in dp_sp[k + 1]:
			continue
		T = [0] * 6
		for c in range(6):
			for c1_1 in range(6):
				for c1_2 in range(6):
					if (c1_1 - c) % 3 and (c1_2 - c) % 3:
						T[c] = (T[c] + t1[c1_1] * dp[k + 1][c1_2]) % MOD
		dp_sp[k][v1 // 2] = T
	for (v, c) in Info[k]:
		if v not in dp_sp[k]:
			dp_sp[k][v] = list(dp[k])
		for c1 in range(6):
			if c1 != c:
				dp_sp[k][v][c1] = 0
if N == 0:
	ans = sum(dp[0]) % MOD
else:
	ans = sum(dp_sp[0][1]) % MOD
print(ans)
