import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.StringTokenizer;
import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 *
 * @author mvv
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskG solver = new TaskG();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskG {
        private int n;
        private int m;

        public void solve(int testNumber, InputReader in, PrintWriter out) {
            n = in.nextInt();
            m = in.nextInt();
            int[][] origA = new int[n][m];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    origA[i][j] = in.nextInt();
                }
            }
            if (n == 1) {
                out.println("YES");
                out.println(0);
                for (int j = 0; j < m; j++) {
                    out.print(origA[0][j]);
                }
                return;
            }

            if (m == 1) {
                out.println("YES");
                for (int j = 0; j < n; j++)
                    out.print(origA[j][0]);

                out.println();
                out.println(0);
                return;
            }
            int[][] a = new int[n][m];
            for (int i = 0; i <= m; i++) {
                for (int ii = 0; ii < n; ii++)
                    for (int jj = 0; jj < m; jj++)
                        a[ii][jj] = origA[ii][jj];

                int[] rows = new int[n];
                int[] columns = new int[m];
                for (int j = 0; j < m; j++) {
                    int p = (j < i) ? 0 : 1;
                    if (a[0][j] != p) {
                        reverseColumn(a, j);
                        columns[j] = 1;
                    }
                }
                boolean canTrig = (i == m) ? true : false;
                int cntChanges = 0;
                boolean isCont = false;
                for (int j = 1; j < n; j++) {
                    cntChanges = countChanges(a, j);
                    if (cntChanges > 1) {
                        isCont = true;
                        break;
                    }
                    if (cntChanges == 1 && !canTrig) {
                        isCont = true;
                        break;
                    }
                    if (cntChanges == 1) {
                        if (a[j][0] == 1) {
                            reverseRow(a, j);
                            rows[j] = 1;
                        }
                        canTrig = false;
                        continue;
                    }
                    if (canTrig && a[j][0] == 1 || !canTrig && a[j][0] == 0) {
                        reverseRow(a, j);
                        rows[j] = 1;
                    }
                }
                if (isCont) continue;

                out.println("YES");
                for (int r : rows)
                    out.print(r);
                out.println();
                for (int c : columns)
                    out.print(c);
                return;

            }
            out.print("NO");
        }

        private void reverseRow(int[][] a, int indRow) {
            for (int i = 0; i < m; i++)
                a[indRow][i] ^= 1;
        }

        private void reverseColumn(int[][] a, int indColumn) {
            for (int i = 0; i < n; i++)
                a[i][indColumn] ^= 1;
        }

        private int countChanges(int[][] a, int indRow) {
            int res = 0;
            for (int i = 1; i < m; i++)
                if (a[indRow][i - 1] != a[indRow][i]) res++;

            return res;

        }

    }

    static class InputReader {
        public BufferedReader reader;
        public StringTokenizer tokenizer;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream), 32768);
            tokenizer = null;
        }

        public String next() {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
}
