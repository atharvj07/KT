import sys
import threading
from math import *

def ri():
	return map(int, sys.stdin.readline().split())

def dfsv(u):
	global ans
	h1 = -inf
	h2 = -inf
	for i in adj[u]:
		if v[i] == 0:
			v[i] = 1
			dfsv(i)
			s[u] += s[i]
			if h1 < h[i]:
				h2 = h1
				h1 = h[i]
			elif h2 < h[i]:
				h2 = h[i]
	ans = max(ans, h1 + h2)
	h[u] = max(h1, s[u])
	return
n = int(input())
a = list(ri())
adj = [[] for i in range(n)]
v = [0 for i in range(n)]
s = [a[i] for i in range(n)]
h = [0 for i in range(n)]
for i in range(n - 1):
	(aa, bb) = ri()
	aa -= 1
	bb -= 1
	adj[aa].append(bb)
	adj[bb].append(aa)
ans = -inf

def solve():
	v[0] = 1
	dfsv(0)
	if ans == -inf:
		print('Impossible')
	else:
		print(ans)
max_recur_size = 10 ** 5 * 2 + 1000
max_stack_size = max_recur_size * 500
sys.setrecursionlimit(max_recur_size)
threading.stack_size(max_stack_size)
thread = threading.Thread(target=solve)
thread.start()
