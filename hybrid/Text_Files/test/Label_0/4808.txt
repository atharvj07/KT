#include <bits/stdc++.h>
using namespace std;
const int MOD = 1000000007;
long long POW(long long a, long long b, long long MMM = MOD) {
  long long ret = 1;
  for (; b; b >>= 1, a = (a * a) % MMM)
    if (b & 1) ret = (ret * a) % MMM;
  return ret;
}

long long ppow(long long a, int i) {
  if (i == 0) return 1;
  if (i == 1) return a;
  long long t = ppow(a, i / 2);
  if (t > 1000000000000000000 / t) return 1000000000000000000 + 1;
  t *= t;
  if (i & 1) {
    if (t > 1000000000000000000 / a) return 1000000000000000000 + 1;
    t *= a;
  }
  return t;
}
