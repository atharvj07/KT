def count_chars(s):
	res = [0] * 26
	for c in s:
		res[ord(c) - ord('a')] += 1
	return res

def inversions_parity(s):
	parity = 0
	for i in range(len(s)):
		for j in range(i + 1, len(s)):
			if s[i] > s[j]:
				parity ^= 1
	return parity
