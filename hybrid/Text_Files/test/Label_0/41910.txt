def solve(x, y, m):
	res = 0
	while max(x, y) < m:
		(x, y) = (x + y, y)
		res += 1
		if max(x, y) >= m:
			return res
		(x, y) = (x, x + y)
		res += 1
	return res

def main():
	(x, y, m) = map(int, input().split())
	if max(x, y) >= m:
		print(0)
		return
	if x <= 0 and y <= 0:
		print(-1)
		return
	if x <= 0:
		(x, y) = (y, x)
	cnt = 0
	if y < 0:
		cnt = (abs(y) + x - 1) // x
		y += cnt * x
	cnt += min(solve(x, y, m), solve(y, x, m))
	print(cnt)
main()
