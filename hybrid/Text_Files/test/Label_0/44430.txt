(n, m) = map(int, input().split())
lst1 = []
left = ''
right = ''
middle = ''

def pal(s):
	valid = True
	for i in range(len(s) // 2):
		if s[i] != s[~i]:
			valid = False
			break
	return valid
for i in range(n):
	a = input()
	lst1.append(a)
for i in range(n):
	for j in range(n):
		if lst1[i] != -1 and pal(lst1[i]):
			middle = lst1[i]
			lst1[i] = -1
		if i != j and lst1[i] != -1 and (lst1[j] != -1) and (lst1[i][::-1] == lst1[j]):
			left = left + lst1[i]
			right = lst1[j] + right
			lst1[i] = -1
			lst1[j] = -1
print(len(left + middle + right))
print(left + middle + right)
