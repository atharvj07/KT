import math
t = int(input())
for p in range(t):
	(n, m, a, b) = map(int, input().split())
	if n * a != m * b:
		print('NO')
		continue
	else:
		print('YES')
		ans = ['0' * m for i in range(n)]
		h = a % m
		k = 0
		for i in range(n):
			if k + a <= m:
				u = ans[i][:k] + '1' * a + ans[i][k + a:]
			else:
				u = '1' * (k + a - m) + ans[i][k + a - m:k] + '1' * (m - k)
			k += h
			k = k % m
			print(u)
