template<class T> constexpr T INF = ::numeric_limits<T>::max() / 32 * 15 + 208;

template <typename T>
struct edge {
    int from, to;
    T cost;

    edge(int to, T cost) : from(-1), to(to), cost(cost) {}
    edge(int from, int to, T cost) : from(from), to(to), cost(cost) {}

    explicit operator int() const {return to;}
};

class UnionFind {
    vector<int> uni;
    int n;
public:
    explicit UnionFind(int n) : uni(static_cast<u32>(n), -1) , n(n){};

    int root(int a){
        if (uni[a] < 0) return a;
        else return (uni[a] = root(uni[a]));
    }

    bool unite(int a, int b) {
        a = root(a);
        b = root(b);
        if(a == b) return false;
        if(uni[a] > uni[b]) swap(a, b);
        uni[a] += uni[b];
        uni[b] = a;
        return true;
    }
};



int main() {
    int n, m, k;
    cin >> n >> m >> k;
    vector<int> c(n);
    for (auto &&i : c) scanf("%d", &i);
    vector<int> used(k+1, -1);
    int cnt = 0;
    vector<int> num(n);
    for (int i = 0; i < n; ++i) {
        if(!c[i]) num[i] = cnt++;
        else{
            if(!~used[c[i]]) {
                used[c[i]] = cnt;
                num[i] = cnt++;
            }else {
                num[i] = used[c[i]];
            }
        }
    }
    vector<edge<int>> G;
    for (int i = 0; i < m; ++i) {
        int a, b, w;
        scanf("%d%d%d", &a, &b, &w);
        a = num[a-1]; b = num[b-1];
        G.emplace_back(a, b, w);
    }
    sort(G.begin(),G.end(), [](edge<int> a, edge<int> b){
       return a.cost < b.cost;
    });
    UnionFind uf(cnt);
    ll ans = 0, x = 0;
    for(auto &&e : G){
        if(x >= k-1) break;
        if(uf.unite(e.from, e.to)){
            ans += e.cost;
            x++;
        }
    }
    if(x == k-1){
        cout << ans << "\n";
    }else {
        puts("-1");
    }
    return 0;
}
