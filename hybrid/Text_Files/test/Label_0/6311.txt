#include <bits/stdc++.h>
const int MOD = 51123987;
const int maxn = 2e6 + 10;
struct Palindromic_Automaton {
  int n, cnte, last, nodecnt;
  int s[maxn];
  int ch[maxn];
  int len[maxn];
  int head[maxn];
  int fail[maxn];
  int deep[maxn];
  int next[maxn];
  char alpha[maxn];
  void init() {
    for (int i = 0; i <= nodecnt; i++)
      head[i] = s[i] = len[i] = deep[i] = fail[i] = 0;
    s[n = 0] = len[nodecnt = 1] = -1;
    fail[cnte = last = 0] = 1;
  }
  int getnxt(int cur, int c) {
    for (int i = head[cur]; i; i = next[i])
      if (alpha[i] == c) return ch[i];
    return 0;
  }
  int extend(int c) {
    s[++n] = c;
    int cur = last;
    while (s[n - len[cur] - 1] != s[n]) cur = fail[cur];
    if (getnxt(cur, c) == 0) {
      int np = ++nodecnt, p = fail[cur];
      len[np] = len[cur] + 2;
      while (s[n - len[p] - 1] != s[n]) p = fail[p];
      fail[np] = getnxt(p, c);
      ++cnte;
      ch[cnte] = np;
      alpha[cnte] = c;
      next[cnte] = head[cur];
      head[cur] = cnte;
      deep[np] = deep[fail[np]] + 1;
    }
    return deep[last = getnxt(cur, c)];
  }
} pam;
int n;
int b[maxn];
long long ans;
long long a[maxn];
char str[maxn];
int main() {
  std::cin >> n >> (str + 1);
  pam.init();
  for (int i = 1; i <= n; i++) ans += (a[i] = pam.extend(str[i] - 'a'));
  if (ans & 1)
    ans = 1LL * ans % MOD * (((ans - 1) >> 1) % MOD) % MOD;
  else
    ans = 1LL * (ans >> 1) % MOD * ((ans - 1) % MOD) % MOD;
  pam.init();
  for (int i = n; i >= 1; i--) b[i] = pam.extend(str[i] - 'a');
  for (int i = 1; i < n; i++) {
    a[i] = (a[i] + a[i - 1]) % MOD;
    ans = (ans - a[i] * b[i + 1] % MOD + MOD) % MOD;
  }
  std::cout << ans << '\n';
  return 0;
}
