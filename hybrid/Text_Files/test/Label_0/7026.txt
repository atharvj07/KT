#include <bits/stdc++.h>
using namespace std;
const int mxN = 1e5 + 7;
const int inf = 1e9 + 7;
const int mod = 1e9 + 7;
void solve() {
  int n;
  cin >> n;
  vector<int> a(n);
  map<int, int> mp;
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
    ++mp[a[i]];
  }
  if (mp.find(0) != mp.end() && mp[0] > 1) {
    cout << "cslnb" << '\n';
    return;
  }
  int gt_1 = 0;
  for (auto it : mp) {
    if (it.second > 1) ++gt_1;
    if (it.second > 2 || gt_1 > 1 ||
        (it.second > 1 && mp.find(it.first - 1) != mp.end())) {
      cout << "cslnb" << '\n';
      return;
    }
  }
  sort(a.begin(), a.end());
  int s = 0;
  for (int i = 0; i < n; ++i) s += a[i] - i;
  cout << (s & 1 ? "sjfnb" : "cslnb") << '\n';
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int T = 1;
  for (int tt = 1; tt <= T; ++tt) {
    solve();
  }
  return 0;
}
