#include <bits/stdc++.h>
using namespace std;
int f[520020];
int main() {
  int n, d, x, cnt, w;
  long long sum = 0;
  scanf("%d%d", &n, &d);
  f[0] = 1;
  for (int i = 1; i <= n; i++) {
    scanf("%d", &x);
    sum += x;
    for (int j = sum; j >= x; j--)
      if (f[j - x]) f[j] = 1;
  }
  int j;
  w = cnt = 0;
  while (1) {
    j = w + d;
    while (!f[j] && j > w) j--;
    if (j == w) break;
    w = j;
    cnt++;
  }
  printf("%d %d\n", w, cnt);
  return 0;
}
