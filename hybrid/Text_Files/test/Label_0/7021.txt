#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, ans = 0;
  bool good[110] = {false};
  scanf("%d%d", &n, &m);
  char s[110][110];
  for (int i = 0; i < n; i++) {
    scanf("%s", s[i]);
  }
  for (int i = 0; i < m; i++) {
    char mx = '0';
    for (int j = 0; j < n; j++) {
      if (s[j][i] > mx) mx = s[j][i];
    }
    for (int j = 0; j < n; j++) {
      if (s[j][i] == mx) good[j] = true;
    }
  }
  for (int i = 0; i < n; i++)
    if (good[i]) ans++;
  printf("%d\n", ans);
  return 0;
}
