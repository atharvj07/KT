#include <bits/stdc++.h>
using namespace std;
const long long MOD = 1000000007LL;
bool isLucky(int x) {
  for (; x > 0; x /= 10) {
    int d = x % 10;
    if (d != 4 && d != 7) return false;
  }
  return true;
}
long long pow(long long n, long long k) {
  long long a = 1;
  for (; k > 0; k >>= 1, n = (n * n) % MOD)
    if (k & 1) a = (a * n) % MOD;
  return a;
}
const int maxn = 1e5 + 10;
long long F[maxn];
long long F_inv[maxn];
long long C(int n, int k) {
  if (k > n) return 0;
  if (n == k || k == 0) return 1;
  return F[n] * F_inv[k] % MOD * F_inv[n - k] % MOD;
}
long long dp[1111][1111];
int main() {
  F[0] = 1;
  for (int _n(maxn), i(1); i < _n; i++) F[i] = (F[i - 1] * i) % MOD;
  for (int _n(maxn), i(0); i < _n; i++) F_inv[i] = pow(F[i], MOD - 2);
  int N, K;
  cin >> N >> K;
  int nL = 0;
  map<int, int> cnt;
  for (int _n(N), i(0); i < _n; i++) {
    int t;
    cin >> t;
    if (isLucky(t))
      cnt[t]++;
    else
      nL++;
  }
  vector<int> sz;
  for (map<int, int>::iterator it = cnt.begin(); it != cnt.end(); it++) {
    if (it->second >= 2)
      sz.push_back(it->second);
    else
      nL++;
  }
  memset((dp), (0), sizeof(dp));
  dp[0][0] = 1;
  for (int _n(sz.size()), i(0); i < _n; i++)
    for (int _n(sz.size()), j(0); j < _n; j++)
      if (dp[i][j]) {
        dp[i + 1][j] += dp[i][j];
        if (dp[i + 1][j] >= MOD) dp[i + 1][j] -= MOD;
        dp[i + 1][j + 1] = (dp[i + 1][j + 1] + dp[i][j] * sz[i]) % MOD;
      }
  long long ans = 0;
  for (int _n(sz.size() + 1), i(0); i < _n; i++)
    ans = (ans + dp[sz.size()][i] * C(nL, K - i)) % MOD;
  cout << ans << endl;
  return 0;
}
