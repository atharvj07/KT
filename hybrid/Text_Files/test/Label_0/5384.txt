#include <bits/stdc++.h>
using namespace std;
int primes[100000], pc = 0;
bool isPrime[1000001];
void genPrimes(int n) {
  memset(isPrime, 1, sizeof(isPrime));
  long double sqd = sqrt(n);
  int sq = (int)sqd;
  if ((sq + 1) * (sq + 1) <= n) {
    sq++;
  }
  isPrime[2] = true;
  int j;
  for (int i = 2; i <= sq; i++) {
    if (isPrime[i]) {
      j = i;
      primes[pc++] = i;
      while (j * i <= n) {
        isPrime[j * i] = false;
        j++;
      }
    }
  }
  for (int i = sq + 1; i <= n; i++) {
    if (isPrime[i]) {
      primes[pc++] = i;
    }
  }
  if (true) {
    for (int i = 0; i < pc; i++) {
      cout << primes[i] << endl;
    }
  }
}
unsigned long long res(int n) {
  if (n == 1) {
    return 1ll;
  } else if (n == 2) {
    return 2ll;
  } else if (n == 3) {
    return 6ll;
  } else if (n == 4) {
    return 12ll;
  } else if (n == 5) {
    return 60ll;
  } else if (n == 6) {
    return 60ll;
  }
  unsigned long long n1 = n, n2 = n - 1, n3 = n - 2, n4 = n - 3;
  if (n % 2 == 0) {
    return ((n1 * n2 * n3 / 2) > (n2 * n3 * n4) ? (n1 * n2 * n3 / 2)
                                                : (n2 * n3 * n4));
  }
  genPrimes(n);
  unsigned long long p1 = primes[pc - 1];
  unsigned long long p2 = primes[pc - 2];
  unsigned long long i = n;
  while (i == p1 || i == p2) {
    i--;
  }
  unsigned long long res = p1 * p2 * i;
  return res;
}
void solve() {
  FILE *in = stdin;
  FILE *out = stdout;
  int n;
  cin >> n;
  long double A[n + 1];
  for (int i = 1; i <= n; i++) {
    cin >> A[i];
  }
  long double res = 0, mult = 0;
  for (int i = 1; i <= n; i++) {
    mult = mult * A[i] + A[i];
    res = res + mult + mult - A[i];
  }
  cout << setprecision(15) << res;
  if (in != stdin) {
    fclose(in);
  }
  if (out != stdout) {
    fclose(out);
  }
}
