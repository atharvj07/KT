for _ in range(int(input())):
	(n, k) = map(int, input().split())
	s = input()
	c = 0
	f = 0
	for i in range(0, n // 2):
		if s[i] == s[n - i - 1]:
			c += 1
		else:
			break
	if c > k:
		print('YES')
	elif c == k:
		if n % 2 != 0:
			print('YES')
		elif s == s[::-1]:
			print('NO')
		else:
			print('YES')
	else:
		print('NO')
