#include <bits/stdc++.h>
using namespace std;
const int maxx = 1e5 + 10;
int n, m;
int a[maxx];
int main() {
  while (scanf("%d%d", &n, &m) != EOF) {
    int cnt = 0;
    int ans = 0;
    while (n--) {
      if ((m - ans) % 2)
        cout << (m + ans + 1) / 2 << endl;
      else
        cout << (m - ans) / 2 << endl;
      ++cnt;
      ans = cnt % m;
    }
  }
}
