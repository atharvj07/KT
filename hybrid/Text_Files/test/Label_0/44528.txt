import sys
input = sys.stdin.readline
t = int(input())
for tests in range(t):
	(n, k) = map(int, input().split())
	A = list(map(int, input().split()))
	B = [1] * n
	for i in range(n):
		if A[i] < k:
			B[i] = -1
	if n == 1 and A[0] == k:
		print('yes')
		continue
	if not k in A:
		print('no')
		continue
	SUM = 0
	ok = 0
	flag = 0
	for i in range(n):
		SUM += B[i]
		if SUM >= 0 and flag:
			ok = 1
			print('yes')
			break
		if B[i] == 1:
			flag = 1
			SUM = 0
	if ok:
		continue
	SUM = 0
	flag = 0
	for i in range(n - 1, -1, -1):
		SUM += B[i]
		if SUM >= 0 and flag:
			ok = 1
			print('yes')
			break
		if B[i] == 1:
			flag = 1
			SUM = 0
	if ok:
		continue
	else:
		print('no')
