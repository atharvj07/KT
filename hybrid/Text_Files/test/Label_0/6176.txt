#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
char book[1000000]= {'2','0','2','0'};
char book2[1000000];
char a[1000000];
int main() {
	int T;
	scanf("%d",&T);
	while(T--) {
		int n;
		scanf("%d",&n);
		book2[n-1]='0';
		book2[n-2]='2';
		book2[n-3]='0';
		book2[n-4]='2';
		int flag=0;

		scanf("%s",&a[0]);
		if(n<4) {
			printf("NO\n");
		} else {
			if(a[0]=='2'&&a[1]=='0'&&a[2]=='2'&&a[3]=='0') {
				flag=1;
			} else if(a[n-4]=='2'&&a[n-3]=='0'&&a[n-2]=='2'&&a[n-1]=='0') {
				flag=1;
			} else {
				for(int i=0;i<=2;i++)
				{
					for(int j=0;j<=i;j++)
					{
						if(a[j]==book[j])
						{
							flag=1;
						}else
						{
							flag=0;
							break;
						}
					}
					if(flag)
					{
						for(int j=n-3+i;j<n;j++)
						{
							if(a[j]==book2[j])
							flag=1;
							else
							{
								flag=0;
								break;
							}
						}
					}
					if(flag)
					break;
				}
			}
			if(flag) {
				printf("YES\n");
			} else {
				printf("NO\n");
			}
		}

	}


	return 0;
}
