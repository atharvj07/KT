#include <bits/stdc++.h>
using namespace std;
const int NMAX = 200002;
int poz[NMAX], v[NMAX];
int main() {
  cin.tie(0)->sync_with_stdio(0);
  cin.tie(NULL);
  cout.tie(NULL);
  int T;
  cin >> T;
  while (T--) {
    long long n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; i++) poz[i] = 0;
    bool par = 0;
    if (k % 2 == 0) par = 1;
    int jum = k / 2;
    bool zero = 0;
    for (int i = 1; i <= n; i++) {
      cin >> v[i];
      if (v[i] > k)
        poz[i] = 1;
      else if (v[i] < jum)
        poz[i] = 0;
      else if (v[i] > jum)
        poz[i] = 1;
      else if (v[i] == jum && par == 0)
        poz[i] = 0;
      else {
        if (zero == 0)
          poz[i] = 0;
        else
          poz[i] = 1;
        zero = 1 - zero;
      }
    }
    for (int i = 1; i <= n; i++) cout << poz[i] << ' ';
    cout << '\n';
  }
  return 0;
}
