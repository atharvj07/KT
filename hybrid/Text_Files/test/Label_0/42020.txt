from sys import stdin, stdout
from collections import defaultdict
from collections import deque
import math
import copy
N = int(input())
arr = [int(x) for x in stdin.readline().split()]
freq = {}
res = 9999
prefix = [0] * N
for i in range(N):
	num = arr[i]
	if num not in freq:
		freq[num] = 0
	else:
		break
	prefix[i] = 1
freq = {}
left = -1
right = 0
for i in range(N - 1, -1, -1):
	num = arr[i]
	if num not in freq:
		freq[num] = 1
	else:
		right = i + 1
		break
r = right - (left + 1)
res = min(res, r)
if r == 0:
	print(0)
	quit()
for i in range(N):
	left = i
	if prefix[i] == 0:
		break
	num = arr[i]
	if num not in freq:
		freq[num] = 1
	elif freq[num] == 1:
		target = num
		while right < N:
			r_num = arr[right]
			if arr[right] == target:
				freq[r_num] -= 1
				right += 1
				break
			freq[r_num] -= 1
			right += 1
	r = right - (left + 1)
	res = min(res, r)
print(res)
