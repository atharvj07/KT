#include <bits/stdc++.h>
using namespace std;
string mirror = "AHIMOTUVWXY";
int main() {
  long long n, m, i, j, k, x, y, t;
  string s;
  cin >> s;
  map<char, long long> mp;
  for (i = 0; i < mirror.size(); i++) {
    mp[mirror[i]] = 1;
  }
  n = s.size();
  for (i = 0; i < n; i++) {
    if (s[i] != s[n - i - 1]) {
      cout << "NO" << endl;
      return 0;
    }
    if (mp[s[i]] == 0) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
}
