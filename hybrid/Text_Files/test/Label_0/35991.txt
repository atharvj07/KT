public class B {
    private static boolean comp(String first, String second) {
        return first.length() < second.length();
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<Pair<String, Integer>> str = new ArrayList<>(2 * n - 2);
        for (int i = 0; i < 2 * n - 2; i++) {
            str.add(i, new Pair(sc.next(), i));
        }
        str.sort(Comparator.comparingInt(o -> o.fi.length()));
        String[] variants = {str.get(0).fi + str.get(2 * n - 3).fi, str.get(0).fi + str.get(2 * n - 4).fi,
                str.get(1).fi + str.get(2 * n - 3).fi, str.get(1).fi + str.get(2 * n - 4).fi};
        for (String s : variants) {
            boolean[] used = new boolean[2 * n - 2];
            int cnt = 0;
            StringBuilder current = new StringBuilder();
            char[] answer = new char[2 * n - 2];
            for (int i = 0; i < 2 * n - 2; i++) {
                current.append(s.charAt(i / 2));
                if (current.toString().equals(str.get(i).fi)) {
                    used[i] = true;
                    answer[str.get(i).se] = 'P';
                    cnt++;
                } else {
                    current.deleteCharAt(current.length() - 1);
                }
            }
//            for (int i = 0; i < 2 * n - 2; i++) {
//                if (used[i]) {
//                    System.out.println(str.get(i).fi);
//                }
//            }
            current = new StringBuilder();
            for (int i = 0; i < 2 * n - 2; i++) {
                if (used[i]) {
                    continue;
                }
                current.insert(0, s.charAt(n - i / 2 - 1));
                if (current.toString().equals(str.get(i).fi)) {
                    used[i] = true;
                    answer[str.get(i).se] = 'S';
//                    System.out.println(str.get(i).fi);
                    cnt++;
                } else {
                    current.deleteCharAt(current.length() - 1);
                }
            }
            if (cnt == 2 * n - 2) {
                for (int i = 0; i < 2 * n - 2; i++) {
                    System.out.print(answer[i]);
                }
                System.out.println();
                break;
            }
        }
    }

    public static class Pair<T, V> {
        T fi;
        V se;

        public Pair(T fi, V se) {
            this.fi = fi;
            this.se = se;
        }
    }
}
