#include <bits/stdc++.h>
using namespace std;
int n, m;
int a[2005];
int second[2005];
int c[2005 * 2];
int dp[2005][5 + (1 << 13)];
int mdp[2005];
int sm[2005 * 2];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cin >> n >> m;
  for (long long i = (1); i <= (signed)(n); i++) {
    cin >> a[i];
  }
  for (long long i = (1); i <= (signed)(n); i++) {
    cin >> second[i];
  }
  for (long long i = (1); i <= (signed)(n + m); i++) {
    cin >> c[i];
    sm[i] = sm[i - 1] + c[i];
  }
  memset(dp, -0x3f, sizeof dp);
  int bst = 0;
  for (int i = (n); i >= 1; i--) {
    int bon = c[a[i]] - second[i];
    for (long long k = 0; k < (signed)(13); k++) {
      int loc = a[i] - k;
      if (loc <= 0) continue;
      for (int j = ((1 << 13) - 1); j >= 0; j--) {
        if (dp[loc][j] <= -0x3f3f3f3f / 2) continue;
        int ns = j >> k;
        int nw = ++ns;
        int zs = __builtin_ctz(nw);
        int eb = sm[a[i] + zs] - sm[a[i]];
        dp[a[i]][nw] = max(dp[a[i]][nw], dp[loc][j] + bon + eb);
        mdp[a[i]] = max(mdp[a[i]], dp[a[i]][nw]);
      }
    }
    dp[a[i]][0] = max(dp[a[i]][0], 0);
    dp[a[i]][1] = max(dp[a[i]][1], bon);
    mdp[a[i]] = max(mdp[a[i]], bon);
    if (a[i] - 13 >= 0) {
      for (long long j = (0); j <= (signed)(a[i] - 13); j++) {
        dp[a[i]][1] = max(dp[a[i]][1], mdp[j] + bon);
        mdp[a[i]] = max(mdp[a[i]], dp[a[i]][1]);
      }
    }
  }
  int ans = 0;
  for (long long i = (1); i <= (signed)(m); i++) ans = max(ans, mdp[i]);
  cout << ans << "\n";
  return 0;
}
