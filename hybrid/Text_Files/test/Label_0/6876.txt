#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k, l, i, j, x;
  double a[110], b[110];
  bool f[110];
  memset(f, 0, sizeof(f));
  scanf("%d%d", &n, &k);
  l = n / k;
  for (i = 1; i <= n; ++i) scanf("%lf", a + i);
  int m;
  double minn = 100000000000.0;
  double maxn = -10000000000.0;
  double tmp = 0.0;
  scanf("%d", &m);
  for (i = 0; i < m; ++i) {
    tmp = 0.0;
    for (j = 0; j < l; ++j) {
      scanf("%d", &x);
      f[x] = 1;
      tmp += a[x];
    }
    tmp /= l;
    minn = min(tmp, minn);
    maxn = max(tmp, maxn);
  }
  int nn = 0;
  for (i = 1; i <= n; ++i) {
    if (!f[i]) b[nn++] = a[i];
  }
  if (nn >= l && (n - nn) / l < k) {
    sort(b, b + nn);
    tmp = 0.0;
    for (i = 0; i < l; ++i) {
      tmp += b[i];
    }
    tmp /= l;
    minn = min(tmp, minn);
    tmp = 0.0;
    for (i = nn - 1; i >= nn - l; --i) {
      tmp += b[i];
    }
    tmp /= l;
    maxn = max(tmp, maxn);
  }
  printf("%.10lf %.10lf\n", minn, maxn);
  return 0;
}
