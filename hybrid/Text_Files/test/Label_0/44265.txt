t = int(input())
for i in range(t):
	n = int(input())
	b = list(map(int, input().split()))
	if n == 1:
		print('YES')
		print(b[0])
		continue
	S = sum(b)
	N = n * (n + 1)
	if 2 * S % N != 0:
		print('NO')
		continue
	s = 2 * S // N
	a = []
	flag = True
	for i in range(n):
		x = s - b[i] + b[i - 1]
		if x % n != 0 or x <= 0:
			flag = False
			print('NO')
			break
		a.append(str(x // n))
	if flag:
		print('YES')
		print(' '.join(a))
