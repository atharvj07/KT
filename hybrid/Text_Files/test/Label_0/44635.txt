n = int(input())
b = []
b = [int(el) for el in input().split()]
n = len(b)
i = n - 1
while b[i - 1] >= b[i] and i > 0:
	i = i - 1
if i == 0 and b[0] == b[n - 1]:
	if b[0] != 0:
		print('NO')
		raise SystemExit()
	else:
		print('YES')
		print(' '.join((str(1) for i in range(n))))
		raise SystemExit()
else:
	k = i + 1
b.insert(0, b[n - 1])
a = [None for i in range(n + 1)]
a[k] = b[k]
for i in range(k - 1, 0, -1):
	a[i] = b[i] + a[i + 1]
	while a[i] <= b[i - 1]:
		a[i] = a[i] + a[i + 1]
i = b.pop(0)
i = a.pop(0)
if k == n:
	print('YES')
	print(' '.join((str(a[i]) for i in range(n))))
	raise SystemExit()
a[n - 1] = a[0] + b[n - 1]
while a[n - 1] <= b[n - 2]:
	a[n - 1] = a[n - 1] + a[0]
i = n - 2
while a[i] == None:
	a[i] = b[i] + a[i + 1]
	while a[i] <= b[i - 1]:
		a[i] = a[i] + a[i + 1]
	i = i - 1
if k <= n - 2 and a[k] <= a[k + 1]:
	a[k] = a[k] + a[k + 1]
if k == n - 1 and a[k] <= a[0]:
	a[k] = a[k] + a[0]
print('YES')
print(' '.join((str(a[i]) for i in range(n))))
