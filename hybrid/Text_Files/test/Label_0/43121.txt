import sys
input = sys.stdin.readline
t = int(input())
for _ in range(t):
	n = int(input())
	ans = []
	for _ in range(n):
		a = int(input())
		if not ans:
			ans.append(a)
		elif a == 1:
			ans.append(a)
		else:
			while not ans[-1] + 1 == a:
				ans.pop()
			ans[-1] += 1
		print('.'.join(map(str, ans)))
