n = int(input())
even_arr = []
uneven_arr = []
i = 0
if n % 2 == 0:
	first_index = n // 2
	while i < first_index:
		uneven_arr.append(2 * i + 1)
		i += 1
	i = 0
	while i < first_index:
		i += 1
		even_arr.append(2 * i)
else:
	first_index = (n + 1) // 2
	second_index = (n + 1) // 2 - 1
	i = 0
	while i < first_index:
		uneven_arr.append(2 * i + 1)
		i += 1
	i = 0
	while i < second_index:
		i += 1
		even_arr.append(2 * i)
if n == 4:
	print(4)
	even_arr.extend(uneven_arr)
	print(' '.join([str(i) for i in even_arr]))
elif n == 3:
	print(2)
	print(' '.join([str(i) for i in uneven_arr]))
elif n == 2 or n == 1:
	print(1)
	print(' '.join([str(i) for i in uneven_arr]))
else:
	print(n)
	uneven_arr.extend(even_arr)
	print(' '.join([str(i) for i in uneven_arr]))
