import java.util.Scanner;

public class D1027 {
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		
		int n = in.nextInt(), i, result = 0, curr, cost;
		int[] c = new int[n], a = new int[n];
		int[] df = new int[n];
		
		for(i = 0; i < n; ++i)
			c[i] = in.nextInt();
		for(i = 0; i < n; ++i)
			a[i] = in.nextInt() - 1;
		
		for(i = 0; i < n; ++i) {
			curr = i;
			while(df[curr] == 0) {
				df[curr] = 1;
				curr = a[curr];
			}
			if(df[curr] == 1) {
				cost = Integer.MAX_VALUE;
				while(df[curr] == 1) {
					df[curr] = -1;
					cost = Math.min(cost, c[curr]);
					curr = a[curr];
				}
				if(cost < Integer.MAX_VALUE)
					result += cost;
			}
			
			for(curr = i; df[curr] == 1; curr = a[curr])
				df[curr] = -1;
		}
		System.out.println(result);
	}
}
