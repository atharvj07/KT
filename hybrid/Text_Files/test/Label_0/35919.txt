        public FastReader()

        String next()
        {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                }
                catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

    void bubbleSort(int[] x, int[] y)
    {
        int n = x.length;
        for (int i = 0; i < n-1; i++)
            for (int j = 0; j < n-i-1; j++)
                if (x[j] > x[j+1])
                {
                    // swap arr[j+1] and arr[j]
                    int temp = x[j];
                    x[j] = x[j+1];
                    x[j+1] = temp;
                    temp = y[j];
                    y[j] = y[j+1];
                    y[j+1] = temp;
                }
    }

    static int GCD(int a, int b)
    {
        if (a == 0 || b == 0)
            return 0;

        if (a == b)
            return a;

        if (a > b)
            return GCD(a-b, b);
        return GCD(a, b-a);
    }

    static int LCM(int a, int b)
    {
        return (a*b)/GCD(a, b);
    }
