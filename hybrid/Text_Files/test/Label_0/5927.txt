#include <bits/stdc++.h>
using namespace std;
const int INF = 0x3f3f3f3f;
const long long mod = 1e9 + 7;
const long long maxn = 4000 + 5;
long long quick_power(long long a, long long b, long long c) {
  long long ans = 1;
  a = a % c;
  while (b) {
    if (b & 1) ans = (ans * a) % c;
    b = b >> 1;
    a = (a * a) % c;
  }
  return ans;
}
bool isprime(int x) {
  int flag = 1;
  for (int i = 2; i * i <= x; i++) {
    if (x % i == 0) {
      flag = 0;
      break;
    }
  }
  if (flag)
    return 1;
  else
    return 0;
}
long long dp[maxn];
long long a[maxn];
long long num[maxn];
int cnt;
int main() {
  int t;
  cin >> t;
  while (t--) {
    memset(dp, 0, sizeof(dp));
    memset(num, 0, sizeof(num));
    cnt = 1;
    int n;
    cin >> n;
    for (int i = 1; i <= 2 * n; i++) cin >> a[i];
    int tem = a[1];
    num[1] = 1;
    for (int i = 2; i <= 2 * n; i++) {
      if (a[i] < tem)
        num[cnt]++;
      else if (a[i] > tem) {
        cnt++;
        num[cnt] = 1;
        tem = a[i];
      }
    }
    sort(num + 1, num + 1 + cnt);
    for (int i = 1; i <= cnt; i++) {
      for (int j = n; j >= num[i]; j--)
        dp[j] = max(dp[j], dp[j - num[i]] + num[i]);
    }
    if (dp[n] - n == 0)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
