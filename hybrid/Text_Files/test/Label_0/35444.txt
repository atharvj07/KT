function main() {

    var n = Number(readline());

    var str = readline();

    var numL = 0;
    var numR = 0;

    for ( var i = 0;i < n;i++){
        if(str[i] === "<"){
            numL ++;
        }else {
            break;
        }
    }

    for (var j = n-1;j>=0;j--){
        if(str[j] === ">"){
            numR ++;
        }else {
            break;
        }
    }

    print(numL+numR);


}

main();
