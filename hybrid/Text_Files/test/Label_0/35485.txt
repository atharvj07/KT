import java.io.*;
import java.util.*;
public class _920G {
    static int x, p, k;
    static int[] f;
    static HashSet<Integer> s, sp;
    static void prime(){
        int N = 1000000;
        f = new int[N + 10];
        for(int i = 2; i * i <= N; i++){
            if(f[i] > 0) continue;
            for(int j = 0; i * i + i * j <= N; j++){
                f[i * i + i * j] = i;
            }
        }
        for(int i = 2; i <= N; i++){
            if(f[i] == 0) f[i] = i;
        }
    }
    static int calc(int l, int r, int c){
        return (r / c) - ((l - 1) / c);
    }
    static void preSolve(){
        while(p > 1){
            int nx = f[p];
            s.add(nx);
//            out.println(nx);
            p /= nx;
        }
        int len = s.size();
        for(int i = 1; i < (1 << len); i++){
            int it = 0;
            int c = 1;
            int cnt = 0;
            for(int v : s){
                if((i & (1 << it)) > 0){
                    cnt++;
                    c = c * v;
                }
                it++;
            }
            if(cnt % 2 == 0) sp.add(-c);
            else sp.add(c);
        }
    }
    static boolean check(int num){
        int res = 0;
        for(int v : sp){
            if(v > 0) res += calc(x + 1, num, v);
            else res -= calc(x + 1, num, -v);
        }
        if(num - (x + 1) + 1 - res < k) return false;
        return true;
    }
    static void run(){
        x = in.nextInt();
        p = in.nextInt();
        k = in.nextInt();
        s = new HashSet<>();
        sp = new HashSet<>();
        preSolve();
        int l = x, r = 10000000 + 1, mid;
        while(r - l > 1){
            mid = (l + r) / 2;
            if(check(mid)) r = mid;
            else l = mid;
        }
        out.println(r);
    }
    static MyScanner in;
    static PrintWriter out;
    public static void main(String[] args) throws IOException {
        boolean stdStreams = true;
        Locale.setDefault(Locale.US);
        BufferedReader br;
        if (stdStreams) {
            br = new BufferedReader(new InputStreamReader(System.in));
            out = new PrintWriter(System.out);
        } else {
            br = new BufferedReader(new FileReader("C:\\Users\\Admin\\IdeaProjects\\untitled1\\src\\_920G.in"));
            out = new PrintWriter(System.out);
        }
        in = new MyScanner(br);
        int tests = in.nextInt();
        prime();
        for (int test = 0; test < tests; test++) {
            new _920G().run();
        }
        br.close();
        out.close();
    }

    static class MyScanner {
        BufferedReader br;
        StringTokenizer st;

        MyScanner(BufferedReader br) {
            this.br = br;
        }

        void findToken() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
        }

        String next() {
            findToken();
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }
    }
}
