#include <bits/stdc++.h>
using namespace std;
long long int mx = 1e10;
void Find_All_lucky_number(vector<long long int> &v, long long int val) {
  long long int luc_num1 = val * 10 + 4;
  long long int luc_num2 = val * 10 + 7;
  if (luc_num1 <= mx) {
    v.push_back(luc_num1);
    Find_All_lucky_number(v, luc_num1);
  }
  if (luc_num2 <= mx) {
    v.push_back(luc_num2);
    Find_All_lucky_number(v, luc_num2);
  }
}
signed main() {
  vector<long long int> v;
  Find_All_lucky_number(v, 0);
  sort(v.begin(), v.end());
  long long int l, r;
  cin >> l >> r;
  long long int ans = 0;
  long long int mn = lower_bound(v.begin(), v.end(), l) - v.begin();
  long long int mx = lower_bound(v.begin(), v.end(), r) - v.begin();
  for (long long int i = mn; i <= mx; i++) {
    if (l <= r) {
      ans += (min(r, v[i]) - l + 1) * v[i];
      l = v[i] + 1;
    } else {
      break;
    }
  }
  cout << ans << endl;
}
