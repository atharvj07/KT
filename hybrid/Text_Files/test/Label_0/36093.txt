import java.util.Scanner;


public class Opposites {
	public static void main(String[] args)
	{
		Scanner br=new Scanner(System.in);
		int[] nums=new int[21];
		int n=br.nextInt();
		for(int i=0;i<n;i++)
		{
			nums[br.nextInt()+10]++;
		}
		long res=0;
		for(int i=0;i<10;i++)
			res+=(long)nums[i]*(long)nums[-(i-10)+10];
		if(nums[10]>=2)
			res+=(long)nums[10]*(long)(nums[10]-1)/2;
		System.out.println(res);
	}
}
