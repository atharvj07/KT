#include <bits/stdc++.h>
using namespace std;
const long long INF = (long long)1e18;
struct point {
  long long x, y;
};
struct box {
  point l, r;
};
box intersection(const vector<box>& v) {
  box ans = {{0, 0}, {INF, INF}};
  for (auto& a : v) {
    ans.l.x = max(ans.l.x, a.l.x);
    ans.r.x = min(ans.r.x, a.r.x);
    ans.l.y = max(ans.l.y, a.l.y);
    ans.r.y = min(ans.r.y, a.r.y);
  }
  return ans;
}
long long area(const box& a) {
  if (a.l.x >= a.r.x || a.l.y >= a.r.y) return 0;
  return (a.r.x - a.l.x) * (a.r.y - a.l.y);
}
int main() {
  cin.sync_with_stdio(false);
  cin.tie(0);
  box a, b, c;
  cin >> a.l.x >> a.l.y >> a.r.x >> a.r.y;
  cin >> b.l.x >> b.l.y >> b.r.x >> b.r.y;
  cin >> c.l.x >> c.l.y >> c.r.x >> c.r.y;
  if (area(a) == area(intersection({a, b})) + area(intersection({a, c})) -
                     area(intersection({a, b, c}))) {
    cout << "NO\n";
  } else {
    cout << "YES\n";
  }
  return 0;
}
