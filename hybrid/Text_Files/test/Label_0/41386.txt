from sys import stdin
s = stdin.readline().strip()
s1 = stdin.readline().strip()
ones = [0 for i in range(len(s1) + 10)]
zeros = [0 for i in range(len(s1) + 10)]
for i in range(len(s1)):
	if s1[i] == '1':
		ones[i] = 1
	else:
		zeros[i] = 1
	if i > 0:
		ones[i] += ones[i - 1]
		zeros[i] += zeros[i - 1]
r = len(s1) - len(s)
ans = 0
for i in range(len(s)):
	if s[i] == '0':
		ans += ones[r] - ones[i - 1]
	else:
		ans += zeros[r] - zeros[i - 1]
	r += 1
print(ans)
