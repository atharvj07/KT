#include <bits/stdc++.h>
using namespace std;
bool cmp(const pair<int, int> &a, const pair<int, int> &b) {
  return a.first < b.first;
}
int main() {
  int N;
  cin >> N;
  vector<pair<int, int>> vec;
  int a, b;
  while (N--) {
    cin >> a >> b;
    vec.push_back(pair<int, int>(a, b));
  }
  sort(vec.begin(), vec.end(), cmp);
  int count = 0;
  int maxa = (*vec.begin()).second;
  for (vector<pair<int, int>>::iterator iter = vec.begin(); iter != vec.end();
       ++iter) {
    if ((*iter).second < maxa) {
      count++;
    } else {
      maxa = (*iter).second;
    }
  }
  cout << count << endl;
  return 0;
}
