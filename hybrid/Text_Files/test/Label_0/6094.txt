#include <bits/stdc++.h>
using namespace std;
int main() {
  vector<pair<int, int> > a;
  int i, j, k, d, n;
  scanf("%d", &n);
  for (i = 0; i < n; i++) {
    scanf("%d %d", &k, &d);
    a.push_back(make_pair(k, k + d));
  }
  for (i = 0; i < n; i++) {
    for (j = 0; j < n; j++)
      if (a[j].first == a[i].second && a[j].second == a[i].first) {
        cout << "YES" << endl;
        return 0;
      }
  }
  cout << "NO" << endl;
  return 0;
}
