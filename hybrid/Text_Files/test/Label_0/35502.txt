import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.*;

/**
 * @author Don Li
 */
public class AnnBooks {
    
    static int B;
    
    void solve() {
        int n = in.nextInt(), k = in.nextInt();
        int[] t = new int[n];
        for (int i = 0; i < n; i++) t[i] = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            if (t[i] == 2) a[i] = -a[i];
        }
        
        long[] sum = new long[n + 1];
        for (int i = 0; i < n; i++) sum[i + 1] = sum[i] + a[i];
        
        Set<Long> set = new HashSet<>();
        for (int i = 0; i < n + 1; i++) {
            set.add(sum[i] - k);
            set.add(sum[i]);
            set.add(sum[i] + k);
        }
        List<Long> list = new ArrayList<>(set);
        Collections.sort(list);
        Map<Long, Integer> ids = new HashMap<>(list.size());
        for (int i = 0, p = 0; i < list.size(); i++) ids.put(list.get(i), p++);
        int[] V = new int[n + 1], L = new int[n + 1], R = new int[n + 1];
        for (int i = 0; i < n + 1; i++) {
            L[i] = ids.get(sum[i] - k);
            V[i] = ids.get(sum[i]);
            R[i] = ids.get(sum[i] + k);
        }
        
        B = (int) Math.sqrt(n);
        int m = in.nextInt();
        Query[] queries = new Query[m];
        for (int i = 0; i < m; i++) {
            queries[i] = new Query(i, in.nextInt() - 1, in.nextInt() - 1);
        }
        Arrays.sort(queries);
        
        long ans = 0;
        long[] res = new long[m];
        int[] map = new int[ids.size()];
        for (int i = 0, l = 0, r = -1; i < m; i++) {
            Query q = queries[i];
            while (r < q.r) {
                int cnt = map[L[r + 2]];
                if (sum[r + 2] - sum[l] == k) cnt++;
                ans += cnt;
                map[V[r + 2]]++;
                r++;
            }
            while (r > q.r) {
                map[V[r + 1]]--;
                int cnt = map[L[r + 1]];
                if (sum[r + 1] - sum[l] == k) cnt++;
                ans -= cnt;
                r--;
            }
            while (l < q.l) {
                int cnt = map[R[l]];
                ans -= cnt;
                map[V[l + 1]]--;
                l++;
            }
            while (l > q.l) {
                map[V[l]]++;
                int cnt = map[R[l - 1]];
                ans += cnt;
                l--;
            }
            res[q.id] = ans;
        }
        
        for (int i = 0; i < m; i++) out.println(res[i]);
    }
    
    static class Query implements Comparable<Query> {
        int id;
        int l, r;
        
        public Query(int id, int l, int r) {
            this.id = id;
            this.l = l;
            this.r = r;
        }
        
        @Override
        public int compareTo(Query q) {
            if (l / B != q.l / B) return Integer.compare(l / B, q.l / B);
            return Integer.compare(r, q.r);
        }
    }
    
    public static void main(String[] args) {
        in = new FastScanner(new BufferedReader(new InputStreamReader(System.in)));
        out = new PrintWriter(System.out);
        new AnnBooks().solve();
        out.close();
    }
    
    static FastScanner in;
    static PrintWriter out;
    
    static class FastScanner {
        BufferedReader in;
        StringTokenizer st;
        
        public FastScanner(BufferedReader in) {
            this.in = in;
        }
        
        public String nextToken() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(in.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
        
        public int nextInt() {
            return Integer.parseInt(nextToken());
        }
        
        public long nextLong() {
            return Long.parseLong(nextToken());
        }
        
        public double nextDouble() {
            return Double.parseDouble(nextToken());
        }
    }
}
