#include <bits/stdc++.h>
using namespace std;
vector<int> w[410], g[410];
int a[410], f[410][410], n, m;
bool v[410];
double mx[410], p[410];
double get(int d) {
  for (int i = 1; i <= n; i++) p[i] = 0, v[i] = 0;
  int t = 0;
  for (int i = 0; i < w[d].size(); i++) {
    int k = w[d][i];
    for (int j = 0; j < g[k].size(); j++) {
      int y = g[k][j];
      p[y] += 1.0 / n / g[k].size();
      if (!v[y]) {
        a[++t] = y;
        v[y] = 1;
      }
    }
  }
  double ans = 0;
  for (int y = 1; y <= n; y++) {
    for (int i = 1; i <= t; i++) mx[f[y][a[i]]] = max(mx[f[y][a[i]]], p[a[i]]);
    double tmp = 0;
    for (int i = 1; i <= t; i++) {
      int d2 = f[y][a[i]];
      tmp += mx[d2];
      mx[d2] = 0;
    }
    ans = max(ans, tmp);
  }
  return ans;
}
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) f[i][j] = (i == j) ? 0 : n + 1;
  for (int i = 1; i <= m; i++) {
    int x, y;
    scanf("%d%d", &x, &y);
    g[x].push_back(y);
    g[y].push_back(x);
    f[x][y] = f[y][x] = 1;
  }
  for (int k = 1; k <= n; k++)
    for (int i = 1; i <= n; i++)
      for (int j = 1; j <= n; j++) f[i][j] = min(f[i][j], f[i][k] + f[k][j]);
  double ans = 0;
  for (int x = 1; x <= n; x++) {
    for (int i = 0; w[i].size(); i++) w[i].clear();
    for (int i = 1; i <= n; i++) w[f[x][i]].push_back(i);
    double tmp = 0;
    for (int d = 0; d <= n; d++) {
      if (!w[d].size()) break;
      tmp += max(1.0 / n, get(d));
    }
    ans = max(ans, tmp);
  }
  printf("%.7lf\n", ans);
}
