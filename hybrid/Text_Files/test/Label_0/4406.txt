#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n, k;
  cin >> n >> k;
  int r[n + 1], x = 1;
  for (int i = 1; i <= n; i++) r[i] = x++, r[++i] = n - x + 2;
  for (int i = 1; i <= k; i++) cout << r[i] << " ";
  if (r[k] > n / 2)
    for (int i = k; i < n; i++) cout << --r[k] << " ";
  else
    for (int i = k; i < n; i++) cout << ++r[k] << " ";
  return 0;
}
