t = int(input())
for i in range(t):
	n = int(input())
	students = input()
	first_angry = -1
	for j in range(n):
		if students[j] == 'A':
			first_angry = j
			break
	if first_angry == -1:
		print(0)
	else:
		pointer = 0
		arr_of_lens = [0 for _ in range(n)]
		for j in range(first_angry + 1, n):
			if students[j] == 'P' or students[j] == 'p':
				arr_of_lens[pointer] += 1
			else:
				pointer += 1
		print(max(arr_of_lens))
