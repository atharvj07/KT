t = int(input())

def good(s):
	for i in range(1, len(s)):
		if abs(ord(s[i]) - ord(s[i - 1])) == 1:
			return False
	return True
for i in range(t):
	s = input()
	count = {}
	for c in s:
		if c not in count:
			count[c] = 0
		count[c] += 1
	out = ''
	alpha = 'abcdefghijklmnopqrstuvwxyz'
	a1 = alpha[::2] + alpha[1::2]
	a2 = alpha[1::2] + alpha[::2]
	s1 = ''
	s2 = ''
	for c in a1:
		if c in count:
			s1 += count[c] * c
	for c in a2:
		if c in count:
			s2 += count[c] * c
	if good(s1):
		print(s1)
	elif good(s2):
		print(s2)
	else:
		print('No answer')
