import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;

public class Main {

    public static void main(String[] args) throws IOException {
        StreamTokenizer streamTokenizer = new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));

        streamTokenizer.nextToken();
        int t = (int) streamTokenizer.nval;

        int row, column, length;
        char[] letter;
        int[] count = new int[26];
        boolean[] f = new boolean[30001];
        int i, j, k, max, tempInt, left;
        int result;

        while (t > 0) {
            t--;
            result = Integer.MAX_VALUE;
            streamTokenizer.nextToken();
            row = (int) streamTokenizer.nval;
            streamTokenizer.nextToken();
            column = (int) streamTokenizer.nval;
            streamTokenizer.nextToken();
            length = (int) streamTokenizer.nval;
            streamTokenizer.nextToken();
            letter = streamTokenizer.sval.toCharArray();

            for (i = 0; i < 26; i++)
                count[i] = 0;
            for (i = 0; i < length; i++)
                count[letter[i] - 'A']++;

            left = length - row - column;
            // suppose letter i is the middle letter
            for (i = 0; i < 26; i++) {
                //set f to false
                max = 0;
                for (j = 0; j <= row; j++)
                    f[j] = false;
                f[0] = true;
                // except letter i, calculate other condition
                for (j = 0; j < 26; j++) {
                    if (j == i)
                        continue;
                    max = Math.min(row, max + count[j]);
                    tempInt = count[j];
                    for (k = max; k >= tempInt; k--)
                        if (f[k - tempInt])
                            f[k] = true;
                }

                for (j = 0; j <= count[i]; j++)
                    if (row - j >= 0 && f[row - j]) {
                        tempInt = Math.max(count[i] - j - left, 0);
                        result = Math.min(result, j * tempInt);
                    }

            }
            System.out.println(result);
        }
    }

}
