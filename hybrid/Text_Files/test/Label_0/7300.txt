#include <iostream>
#include <vector>
#include <queue>
#define REP(i, a, n) for(int i = (a); i < (n); i++)
#define INF 1000000000
using namespace std;
typedef long long ll;

struct edge { ll n, d, e; };
struct state {
  ll n, l, s;
  bool operator<(const state &state) const { return s > state.s; }
};

int main(void) {
  int N, M, L;
  while(cin >> N >> M >> L, N || M || L) {
    vector<edge> G[100];

    REP(i, 0, M) {
      int A, B, D, E;
      cin >> A >> B >> D >> E;
      G[A - 1].push_back((edge) { B - 1, D, E });
      G[B - 1].push_back((edge) { A - 1, D, E });
    }

    ll dp[100][101];
    REP(i, 0, N) REP(j, 0, L + 1) dp[i][j] = INF;
    dp[0][0] = 1;

    priority_queue<state> q;
    q.push((state) { 0, 0, 0 });
    while(q.size()) {
      state c = q.top();
      q.pop();

      if(c.n == N - 1) {
        cout << c.s << endl;
        break;
      }

      if(dp[c.n][c.l] <= c.s) continue;
      dp[c.n][c.l] = c.s;

      REP(i, 0, G[c.n].size()) {
        q.push((state) { G[c.n][i].n, c.l, c.s + G[c.n][i].e });
        if(c.l + G[c.n][i].d <= L) {
          q.push((state) { G[c.n][i].n, c.l + G[c.n][i].d, c.s });
        }
      }
    }
  }

  return 0;
}
