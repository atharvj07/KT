(n, m, k) = map(int, input().split())
ans = []
append = ans.append
(x, y) = (1, 1)
r_cnt = (m - 1) // 4
if r_cnt:
	ans.append((r_cnt, 'RRRR'))
x += r_cnt * 4
if x < m:
	ans.append((1, 'R' * (m - x) + 'L' * min(m - 1, 4 - (m - x))))
	x = m - min(m - 1, 4 - (m - x))
l_cnt = (x - 1) // 4
if l_cnt:
	ans.append((l_cnt, 'LLLL'))
x -= l_cnt * 4
if x > 1:
	ans.append((1, 'L' * (x - 1)))
if n > 1:
	ans.append((1, 'D'))
(x, y) = (1, 2)
for y in range(2, n + 1):
	if m > 1:
		ans.append((m - 1, 'RUD'))
	if m > 4:
		ans.append(((m - 1) // 4, 'LLLL'))
	if y < n:
		ans.append((1, 'L' * ((m - 1) % 4) + 'D'))
	elif (m - 1) % 4:
		ans.append((1, 'L' * ((m - 1) % 4)))
if n > 4:
	ans.append(((n - 1) // 4, 'UUUU'))
if (n - 1) % 4:
	ans.append((1, 'U' * ((n - 1) % 4)))
steps = sum((x[0] * len(x[1]) for x in ans))
ans2 = []
if steps < k:
	print('NO')
else:
	print('YES')
	for (cnt, s) in ans:
		if len(s) * cnt <= k:
			ans2.append('{} {}'.format(cnt, s))
			k -= len(s) * cnt
			if k == 0:
				break
		else:
			c = k // len(s)
			if c:
				ans2.append('{} {}'.format(c, s))
			if k % len(s):
				ans2.append('{} {}'.format(1, s[:k % len(s)]))
			break
	print(len(ans2))
	print(*ans2, sep='\n')
