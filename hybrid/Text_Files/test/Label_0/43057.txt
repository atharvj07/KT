n = int(input())
exams = []
max_date = 0
for i in range(n):
	(a, b) = map(int, input().split())
	if a > max_date:
		max_date = a
	exams.append([a, b])
exams.sort()
i = 0
last_day = 0
now = exams[i][1]
while i < n - 1:
	if now > exams[i + 1][1]:
		now = exams[i + 1][0]
	else:
		now = exams[i + 1][1]
	i += 1
print(now)
