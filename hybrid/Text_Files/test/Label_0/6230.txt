#include <bits/stdc++.h>
using namespace std;
int isprime(long long int n) {
  for (long long int i = 2; i * i <= n; i++) {
    if (n % i == 0) return 0;
  }
  return 1;
}
int main() {
  long long int n;
  cin >> n;
  if (isprime(n))
    cout << '1';
  else if ((n & 1) != 1)
    cout << '2';
  else if (isprime(n - 2))
    cout << '2';
  else
    cout << '3';
  cout << "\n";
  return 0;
}
