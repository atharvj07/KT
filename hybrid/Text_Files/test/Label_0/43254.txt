n = int(input())
s = input()
a = int(s)
for k in range(-1, 2):
	if n // 2 + k >= 0 and n // 2 + k < n:
		for i in range(max(1, n // 2 + k), n):
			if s[i] != '0':
				a = min(a, int(s[i:]) + int(s[:i]))
				break
		for i in range(n // 2 + k, 0, -1):
			if s[i] != '0':
				a = min(a, int(s[i:]) + int(s[:i]))
				break
print(a)
