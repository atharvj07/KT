na = int(input())
a = list(map(int, input().split()))
nb = int(input())
b = list(map(int, input().split()))

def judge(na, a, nb, b):
	ans = []
	if sum(a) != sum(b):
		return (False, None)
	cur = 0
	ind = -1
	j = 0
	for (i, x) in enumerate(a):
		cur += x
		if cur < b[j]:
			pass
		elif cur == b[j]:
			begin = ind + 1
			end = i
			interval = a[begin:end + 1]
			if len(interval) > 1 and all((val == a[begin] for val in interval)):
				return (False, None)
			else:
				ans.append(interval)
			ind = i
			cur = 0
			j += 1
		else:
			return (False, None)
		if j >= len(b):
			break
	return (True, ans)

def PrintAns(ans):
	for ii in range(len(ans)):
		num = ans[ii]
		begin = end = 0
		maxn = max(num)
		lenn = len(num)
		for i in range(lenn):
			if num[i] == maxn and (i > 0 and num[i] > num[i - 1] or (i < lenn - 1 and num[i] > num[i + 1])):
				begin = end = i
				break
		while begin > 0 or end < lenn - 1:
			if begin > 0 and num[begin] > num[begin - 1]:
				PrintOne(begin, ii, 'L')
				begin -= 1
				num[begin] += num[begin + 1]
				num[end] = num[begin]
			elif end < len(num) - 1 and num[end] > num[end + 1]:
				PrintOne(begin, ii, 'R')
				end += 1
				num[end] += num[end - 1]
				num[begin] = num[end]

def PrintOne(ind, offset, op):
	print(ind + offset + 1, op)
(ret, ans) = judge(na, a, nb, b)
if ret:
	print('YES')
	PrintAns(ans)
else:
	print('NO')
