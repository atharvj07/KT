from sys import stdin
from math import gcd

def lcm(a, b):
	return a * b // gcd(a, b)

def exgcd(a, b) -> (int, int, int):
	if b == 0:
		return (a, 1, 0)
	(d, x, y) = exgcd(b, a % b)
	t = x
	x = y
	y = t - a // b * y
	return (d, x, y)

def cal(v: list) -> int:
	now = v[0]
	for i in range(1, len(v)):
		a = now[0]
		b = -v[i][0]
		c = v[i][1] - now[1]
		(g, x, y) = exgcd(a, b)
		x *= c // g
		y *= c // g
		lc = lcm(a, b)
		now = (lc, (a * x + now[1]) % lc)
	return now[1]

def no():
	print('NO')
	exit(0)

def yes():
	print('YES')
	exit(0)
(n, m, k) = list(map(int, stdin.readline().split()))
a = list(map(int, stdin.readline().split()))
ll = a[0]
for x in a:
	ll = lcm(ll, x)
	if ll > n:
		no()
v = [(x, x - i % x) for (i, x) in enumerate(a)]
ans = cal(v)
ans = ans % ll
if ans <= 0:
	ans += ll
if ans + k - 1 > m:
	no()
for (i, x) in enumerate(a):
	if gcd(ll, ans + i) != x:
		no()
yes()
