public class SerejaBrackets {

	public static void main(String[] args) throws IOException {
		Scanner sc = new Scanner(System.in);
		String br = sc.nextLine();
		int n = br.length();
		int N = (int) Math.pow(2, Math.ceil(Math.log(n)/Math.log(2)));
		Vertex a[] = new Vertex[N+1];
		for(int i=0; i<n; i++)
		{
			char b = br.charAt(i);
			if(b == '(')
			{
				a[i+1] = new Vertex(1,0,0);
			}
			else
			{
				a[i+1] = new Vertex(0,1,0);
			}
		}
		for(int i = n+1; i<=N; i++)
		{
			a[i] = new Vertex(0,0,0);
		}
		
		SegmentTree tree = new SegmentTree(a);
		int m = sc.nextInt();
		PrintWriter out = new PrintWriter(System.out);
		for(int i=0; i<m; i++)
		{
			int l = sc.nextInt();
			int r = sc.nextInt();
			out.println(tree.query(l,r).max);
		}
		out.flush();

	}
	static class Vertex
	{
		int o;
		int c;
		int max;
		
		public Vertex(int x, int y, int z)
		{
			o = x;
			c = y;
			max = z;
		}
	}
	static class SegmentTree
	{
		int N; 			//the number of elements in the array as a multipls of 2 (i.e. after padding)
		Vertex[] array, sTree, lazy;
		
		SegmentTree(Vertex[] in)		
		{
			array = in; N = in.length - 1;
			sTree = new Vertex[N<<1];		//no. of nodes = 2*N - 1, we add one to cross out index zero
			build(1,1,N);
		}
		
		void build(int node, int b, int e)	// O(n)
		{
			if(b == e)					
				sTree[node] = array[b];
			else						
			{
				build(node<<1,b,(b+e)/2);
				build((node<<1)+1,(b+e)/2+1,e);
				int op = sTree[node<<1].o;
				int cl = sTree[(node<<1)+1].c;
				int lo = sTree[node<<1].max + sTree[(node<<1)+1].max;
				sTree[node] = new Vertex(sTree[(node<<1)+1].o + op - Math.min(op, cl), sTree[(node<<1)].c + cl - Math.min(op, cl),Math.min(op, cl)*2 + lo);
			}
		}
		
		
		
		
		
		
		
		
		Vertex query(int i, int j)
		{
			return query(1,1,N,i,j);
		}
		
		Vertex query(int node, int b, int e, int i, int j)	// O(log n)
		{
			if(i>e || j <b)
				return new Vertex(0,0,0);		
			if(b>= i && e <= j)
				return sTree[node];
			
			Vertex left = query(node<<1,b,(b+e)/2,i,j);
			Vertex right = query((node<<1)+1,(b+e)/2+1,e,i,j);
			
			int op = left.o;
			int cl = right.c;
			int lo = left.max + right.max;
			
			return new Vertex(op + right.o - Math.min(op, cl), left.c + cl - Math.min(op, cl),Math.min(op, cl)*2 + lo);
				
					
		}

	static class Scanner {
		BufferedReader br;
		StringTokenizer st;

		Scanner(FileReader f) {
			br = new BufferedReader(f);
		}

		public boolean ready() throws IOException {
			return br.ready();
		}

		Scanner(InputStream s) {
			br = new BufferedReader(new InputStreamReader(s));
		}

		String next() throws IOException {
			while (st == null || !st.hasMoreTokens())
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		String nextLine() throws IOException {
			return br.readLine();
		}

		int nextInt() throws NumberFormatException, IOException {
			return Integer.parseInt(next());
		}

		long nextLong() throws NumberFormatException, IOException {
			return Long.parseLong(next());
		}
	}
