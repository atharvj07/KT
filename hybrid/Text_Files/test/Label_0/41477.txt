NUMBER = 10 ** 9 + 7

def factorial(n):
	M = NUMBER
	f = 1
	for i in range(1, n + 1):
		f = f * i % M
	return f

def mult(a, b):
	return a * b % NUMBER

def minus(a, b):
	return (a - b) % NUMBER

def plus(a, b):
	return (a + b) % NUMBER

def egcd(a, b):
	if a == 0:
		return (b, 0, 1)
	else:
		(g, y, x) = egcd(b % a, a)
		return (g, x - b // a * y, y)

def modinv(a):
	m = NUMBER
	(g, x, y) = egcd(a, m)
	if g != 1:
		raise Exception('modular inverse does not exist')
	else:
		return x % m

def bfs(V, E):
	s = set()
	print('\n'.join(map(str, E.items())))
	cnt = 0
	for v in V:
		if v in s:
			continue
		q = deque([])
		s.add(v)
		q.append(v)
		while len(q) > 0:
			x = q.popleft()
			for e in E[x]:
				if e not in s:
					s.add(e)
					q.append(e)
			print(q)
		cnt += 1
	return cnt

def regular_bs(L, x, l=None, r=None):
	l = 0 if l == None else l
	r = len(L) if r == None else r
	while r - l > 1:
		m = (r + l) // 2
		l = m if L[m] <= x else l
		r = m if L[m] > x else r
	return l if L[l] == x else -1
from typing import List, Tuple

class TreeNode:

	def __init__(self, x):
		self.val = x
		self.left = None
		self.right = None

	def __repr__(self):
		return f'({self.val}, {self.left.__repr__()}, {self.right.__repr__()})'

def helper(preorder: List[int], index, max_value=2 ** 300) -> Tuple[TreeNode, int]:
	if index >= len(preorder) or preorder[index] >= max_value:
		return (None, index)
	root_val = preorder[index]
	next_index = index + 1
	node = TreeNode(preorder[index])
	(node.left, next_index) = helper(preorder, next_index, root_val)
	(node.right, next_index) = helper(preorder, next_index, max_value)
	return (node, next_index)

def solution(L, n):
	pos = True if L[0] < 0 else False
	total = 0
	curr_opt = 0
	for x in L:
		if pos:
			if x > 0:
				curr_opt = max(curr_opt, x)
			else:
				total += curr_opt
				curr_opt = x
				pos = False
		elif x < 0:
			curr_opt = max(curr_opt, x)
		else:
			total += curr_opt
			curr_opt = x
			pos = True
	total += curr_opt
	return total
