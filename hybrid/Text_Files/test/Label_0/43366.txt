import sys
(m, n) = [int(i) for i in input().split()]
data = [0] * (m + 1)
for i in range((m + 1) // 2):
	data[2 * i] = i + 1
	data[2 * i + 1] = m - i
for x in range(n // 2):
	for i in range(m):
		sys.stdout.write(str(i + 1) + ' ' + str(x + 1) + '\n')
		sys.stdout.write(str(m - i) + ' ' + str(n - x) + '\n')
if n % 2 == 1:
	q = (n + 1) // 2
	s = ' ' + str(q) + '\n'
	for i in range(m):
		sys.stdout.write(str(data[i]) + s)
