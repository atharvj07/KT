#include <bits/stdc++.h>
using namespace std;
char save[55][55];
char s[55 * 55];
int n, w[55 * 55][55 * 55], len, siz[55], dp[55 * 55];
void preprocess() {
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < len; j++) {
      int now = j;
      for (int k = 0; k < siz[i]; k++) {
        if (s[now] == save[i][k]) {
          now++;
          w[j][now] = 1;
        }
      }
    }
  }
}
int DP(int pos) {
  if (pos == 0) return 0;
  if (dp[pos]) return dp[pos];
  int k = 2000000005;
  for (int i = 0; i < pos; i++) {
    if (w[i][pos]) k = min(k, DP(i));
  }
  return dp[pos] = k + 1;
}
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    scanf("%s", save[i]);
    siz[i] = strlen(save[i]);
  }
  scanf("%s", s);
  len = strlen(s);
  preprocess();
  int ans = DP(len);
  if (ans >= 2000000005)
    printf("-1");
  else
    printf("%d", ans);
  return 0;
}
