import sys, math, itertools, collections, copy
input = sys.stdin.readline

def solve():
	(N, M) = map(int, input().split())
	table = []
	mn = 101
	(flag, neg) = (False, 0)
	for i in range(N):
		arr = list(map(int, input().split()))
		table.append(arr)
		for j in range(M):
			mn = min(mn, abs(table[i][j]))
			if table[i][j] < 0:
				neg += 1
			elif table[i][j] == 0:
				flag = True
	if neg & 1 and flag:
		s = 0
		for i in range(N):
			for j in range(M):
				s += abs(table[i][j])
		print(s)
	elif neg & 1 and (not flag):
		s = 0
		for i in range(N):
			for j in range(M):
				s += abs(table[i][j])
		s -= 2 * abs(mn)
		print(s)
	else:
		s = 0
		for i in range(N):
			for j in range(M):
				s += abs(table[i][j])
		print(s)
for _ in range(int(input())):
	solve()
