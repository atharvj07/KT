for _ in range(int(input())):
	n = int(input())
	b = [int(x) for x in input().split()]
	(j, count) = (0, 0)
	(bad_left, bad_right) = (0, 0)
	for i in range(1, 2 * n + 1):
		if j < n and b[j] == i:
			j += 1
			if count == 0:
				bad_left += 1
			else:
				count -= 1
		else:
			count += 1
	(j, count) = (n - 1, 0)
	for i in range(2 * n, 0, -1):
		if j > -1 and b[j] == i:
			j -= 1
			if count == 0:
				bad_right += 1
			else:
				count -= 1
		else:
			count += 1
	print(n + 1 - bad_left - bad_right)
