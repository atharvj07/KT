def factorize(x):
	(dictionary, i) = ({}, 2)
	while i * i <= x:
		exp = 0
		while x % i == 0:
			(exp, x) = (exp + 1, x // i)
		if exp > 0:
			dictionary[i] = exp
		i = i + 1
	if x > 1:
		dictionary[x] = 1
	return dictionary

def get_exponential_from_factorial(n, prime):
	ans = 0
	while n > 0:
		ans += n // prime
		n //= prime
	return ans
(n, b) = map(int, input().split())
b_prime_factors = factorize(b)
trailing = float('inf')
for prime in b_prime_factors:
	trailing = min(trailing, get_exponential_from_factorial(n, prime) // b_prime_factors[prime])
print(trailing)
