(h, w) = map(int, input().split())
mat = [list(input()) for _ in range(h)]
rows = [[0 for _ in range(w + 1)] for i in range(h + 1)]
cols = [[0 for _ in range(w + 1)] for i in range(h + 1)]
for i in range(h):
	for j in range(1, w):
		rows[i + 1][j + 1] = rows[i + 1][j]
		if mat[i][j] == '.' and mat[i][j - 1] == '.':
			rows[i + 1][j + 1] += 1
for j in range(w):
	for i in range(1, h):
		cols[i + 1][j + 1] = cols[i][j + 1]
		if mat[i][j] == '.' and mat[i - 1][j] == '.':
			cols[i + 1][j + 1] += 1
q = int(input())
for _ in range(q):
	(r1, c1, r2, c2) = map(int, input().split())
	ans = 0
	for i in range(r1, r2 + 1):
		ans += rows[i][c2] - rows[i][c1]
	for i in range(c1, c2 + 1):
		ans += cols[r2][i] - cols[r1][i]
	print(ans)
