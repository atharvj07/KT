n = int(input())
MOD = 10 ** 9 + 7
p = [int(i) for i in input().split()]
p.sort()
a = []
index = 0
now = p[0]
cnt_now = 0
while index < n:
	while index < n and p[index] == now:
		index += 1
		cnt_now += 1
	a.append([now, cnt_now])
	if index < n:
		now = p[index]
		cnt_now = 0
tau = 1
ans = 1
for i in range(len(a)):
	tau = tau * (a[i][1] + 1)
for i in range(len(a)):
	deg = tau * a[i][1] // 2
	ans = ans * pow(a[i][0], deg % (MOD - 1), MOD) % MOD
print(ans)
