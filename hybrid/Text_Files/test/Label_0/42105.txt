import sys
input = sys.stdin.buffer.readline
t = int(input())
ans = []
for _ in range(t):
	n = int(input())
	(*a,) = map(int, input().split())
	a.sort()
	l = [0] * n
	if n % 2:
		l[0] = a[n // 2]
		i = 1
		for (x, y) in zip(a[:n // 2][::-1], a[n // 2 + 1:]):
			l[i] = x
			i += 1
			l[i] = y
			i += 1
	else:
		i = 0
		for (x, y) in zip(a[:n // 2][::-1], a[n // 2:]):
			l[i] = x
			i += 1
			l[i] = y
			i += 1
	ans.append(' '.join(map(str, l)))
print('\n'.join(ans))
