#include <bits/stdc++.h>
using namespace std;
int toi(string a) {
  int ans;
  sscanf(a.c_str(), "%d", &ans);
  return ans;
}
string tos(int a) {
  ostringstream st;
  st << a;
  string ans = st.str();
  return ans;
}
int d1[100005], d2[100005], f[100005];
vector<vector<int> > adj;
void dfs1(int padre, int x, int d) {
  d1[x] = d;
  for (int j = (0); j < (adj[x].size()); j++) {
    int where = adj[x][j];
    if (padre == where) continue;
    dfs1(x, where, d + 1);
  }
}
void dfs2(int padre, int x, int d) {
  d2[x] = d;
  for (int j = (0); j < (adj[x].size()); j++) {
    int where = adj[x][j];
    if (padre == where) continue;
    dfs2(x, where, d + 1);
  }
}
int main() {
  int n, m, d, x, y;
  cin >> n >> m >> d;
  for (int i = (0); i < (m); i++) scanf("%d", &f[i]), f[i]--;
  adj.assign(n, vector<int>());
  for (int i = (0); i < (n - 1); i++) {
    scanf("%d %d", &x, &y);
    adj[x - 1].push_back(y - 1);
    adj[y - 1].push_back(x - 1);
  }
  int c = 0;
  dfs1(-1, 0, 0);
  for (int i = (0); i < (m); i++)
    if (d1[f[i]] > d1[c]) c = f[i];
  dfs1(-1, c, 0);
  for (int i = (0); i < (m); i++)
    if (d1[f[i]] > d1[c]) c = f[i];
  dfs2(-1, c, 0);
  int ans = 0;
  for (int i = (0); i < (n); i++)
    if (d1[i] <= d && d2[i] <= d) ++ans;
  cout << ans << endl;
  return 0;
}
