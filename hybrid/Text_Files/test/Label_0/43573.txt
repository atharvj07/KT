def solve():
	(i, pre) = (0, 0)
	for (j, v) in enumerate(b):
		while v > 0:
			v -= a[i]
			i += 1
		if v:
			return 0
		(l, pre) = (a[pre:i], i)
		m = max(l)
		l.append(inf)
		for (n, val) in enumerate(l):
			if val == m:
				merge(l, n, j)
			if len(l) == 2:
				continue
		if len(l) > 2:
			return 0
	if a[i] < inf:
		return 0
	else:
		return 1

def merge(l, i, j):
	if l[i] > l[i - 1]:
		ans.append('%d L' % (i + j + 1))
		i -= 1
	elif l[i] > l[i + 1]:
		ans.append('%d R' % (i + j + 1))
	else:
		return
	l[i] += l.pop(i + 1)
	merge(l, i, j)
ans = ['YES']
inf = 1 << 30
(_, a) = (input(), [*map(int, input().split()), inf])
(_, b) = (input(), [int(x) for x in input().split()])
if solve():
	print('\n'.join(ans))
else:
	print('NO\n')
