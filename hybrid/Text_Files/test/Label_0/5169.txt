#include <iostream>
using namespace std;
int coaster[1001][1001];
int main(void){
    int c,n,x,y,d,temp;
    char a;
    int cut=0;
    int ans=0;
    cin>>c>>n;
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            cin>>a;
            if(a=='1'){
                coaster[i][j]=1;
            }
            else{
                coaster[i][j]=0;
            }
        }
    }
    for(int i=1;i<=n/2;i++){
        for(int j=1;j<=n/2;j++){
            if(coaster[i][j]!=coaster[n-i+1][j]||coaster[i][j]!=coaster[i][n-j+1]||coaster[i][j]!=coaster[n-i+1][n-j+1]){
                cut++;
            }
        }
    }
    if(cut==0){
        ans++;
    }
    for(int i=2;i<=c;i++){
        cin>>d;
        for(int j=0;j<d;j++){
            cin>>y>>x;
            if(coaster[y][x]==0){
                coaster[y][x]=1;
                temp=0;
            }
            else{
                coaster[y][x]=0;
                temp=1;
            }
            if((temp==coaster[n-y+1][x]&&temp==coaster[y][n-x+1]&&temp==coaster[n-y+1][n-x+1])&&(coaster[y][x]!=coaster[n-y+1][x]||coaster[y][x]!=coaster[y][n-x+1]||coaster[y][x]!=coaster[n-y+1][n-x+1])){
                cut++;
            }
            else if((temp!=coaster[n-y+1][x]||temp!=coaster[y][n-x+1]||temp!=coaster[n-y+1][n-x+1])&&(coaster[y][x]==coaster[n-y+1][x]&&coaster[y][x]==coaster[y][n-x+1]&&coaster[y][x]==coaster[n-y+1][n-x+1])){
                cut--;
            }
        }
        if(cut==0){
            ans++;
        }
    }
    
    cout<<ans<<endl;
}
