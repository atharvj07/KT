#include <bits/stdc++.h>
using namespace std;
const int N = 11111111;
const int M = 666666;
int n, m, i, j, x, Max, p[M], mi[N];
long long q[N], s[N], tmp, l, r, mid, ans;
bool check(long long x) {
  for (int i = 0; i < m; i++) {
    long long t = 0;
    for (long long o = p[i]; o <= x; o *= p[i]) t += x / o;
    if (t < q[i]) return 0;
  }
  return 1;
}
int main() {
  scanf("%d", &n);
  for (; n--;) {
    scanf("%d", &x);
    s[x]++;
    Max = max(Max, x);
  }
  for (i = 2; i <= Max; i++) {
    if (!mi[i]) p[m++] = i;
    for (j = 0; j < m && i * p[j] <= Max; j++) {
      mi[i * p[j]] = p[j];
      if (i % p[j] == 0) break;
    }
  }
  for (i = Max; i >= 2; i--) {
    tmp += s[i];
    if (mi[i]) {
      s[mi[i]] += tmp;
      s[i / mi[i]] += tmp;
      s[mi[i] - 1] -= tmp;
      s[i / mi[i] - 1] -= tmp;
    } else
      q[lower_bound(p, p + m, i) - p] = tmp;
  }
  l = 1;
  r = ans = (long long)1 << (long long)60;
  while (l <= r) {
    mid = (l + r) / 2;
    if (check(mid)) {
      ans = mid;
      r = mid - 1;
    } else
      l = mid + 1;
  }
  printf("%I64d\n", ans);
}
