	static class KMP {
		/**
		 * Generates the table used by KMP. Not really useful on its own :( O(m)
		 * [m=length of needle]
		 */
		public int[] KMPTableGen(String needle) {
			int[] T = new int[needle.length()];
			int pos = 2;
			int cnd = 0;
			T[0] = -1;
			T[1] = 0;
			while (pos < needle.length()) {
				if (needle.charAt(pos - 1) == needle.charAt(cnd)) {
					cnd++;
					T[pos] = cnd;
					pos++;
				} else if (cnd > 0)
					cnd = T[cnd];
				else {
					T[pos] = 0;
					pos++;
				}
			}
			return T;
		}

		/**
		 * KMP - String Searching
		 * 
		 * O(n+m) [n=length of haystack, m=length of needle] Searches for the string
		 * needle in the string haystack Requires: KMPTableGen(String)
		 */
		public boolean KMPSearch(String needle, String haystack) {
			int m = 0;
			int i = 0;
			int[] T = KMPTableGen(needle);
			while (m + i < haystack.length()) {
				if (needle.charAt(i) == haystack.charAt(m + i)) {
//					if (i == needle.length() - 1)
//						return m;
					i++;
				} else {
					m = m + i - T[i];
					if (T[i] > -1)
						i = T[i];
					else
						i = 0;
				}
			}
			//return haystack.length();
			return  i ==0;
		}
	}
