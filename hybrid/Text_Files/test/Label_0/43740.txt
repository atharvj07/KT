t = int(input())
for __ in range(t):
	x = int(input())
	s = [c for c in input()]
	l = 0
	total_length = len(s)
	while l != x:
		l += 1
		count = int(s[l - 1]) - 1
		if total_length <= x:
			for k in range(count):
				for c in s[l:total_length]:
					if len(s) > x:
						break
					s.append(c)
		total_length += (total_length - l) * count
		total_length %= int(1000000000.0) + 7
	print(total_length)
