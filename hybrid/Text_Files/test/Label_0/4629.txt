#include<bits/stdc++.h>
using namespace std;

int main(){
	int n,k;
	string s;
	cin>>n>>k>>s;

	int i = 0;
	int j = 0;
	int res=0;

	while(j<n){
		while(j<n && s[j]=='1') 
			++j;
		res=max(res,j-i);
		while(j<n && s[j]=='0') 
			++j;
		--k;
		if(k<0){
			while(i<n && s[i]=='1') 
				++i;
			while(i<n && s[i]=='0') 
				++i; 	
		}
		res = max(res,j-i);
	}
	cout<<res;
}
