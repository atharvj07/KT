#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2010;
int N, M;
char ar[MAXN][MAXN];
int cnt[MAXN][MAXN];
bool vis[MAXN][MAXN];
bool valid(int x, int y) { return x >= 0 && x < N && y >= 0 && y < M; }
int get(int x, int y) { return ar[x][y] == '*'; }
void add(int x, int y, queue<pair<int, int> >& q) {
  if (!valid(x, y) || vis[x][y] || ar[x][y] == '.') return;
  for (int i = -1; i <= 0; ++i) {
    for (int k = -1; k <= 0; ++k) {
      if (valid(x + i, y + k) && cnt[x + i][y + k] == 1) {
        q.push(pair<int, int>(x, y));
        vis[x][y] = true;
        return;
      }
    }
  }
}
void upd(int x, int y) {
  ar[x][y] = '.';
  for (int i = -1; i <= 0; ++i) {
    for (int k = -1; k <= 0; ++k) {
      if (valid(x + i, y + k)) {
        --cnt[x + i][y + k];
      }
    }
  }
  vis[x][y] = false;
}
int main() {
  cin >> N >> M;
  for (int i = 0; i < N; ++i) {
    cin >> ar[i];
  }
  for (int i = 0; i < N - 1; ++i) {
    for (int k = 0; k < M - 1; ++k) {
      for (int ni = i; ni <= i + 1; ++ni) {
        for (int nk = k; nk <= k + 1; ++nk) {
          cnt[i][k] += get(ni, nk);
        }
      }
    }
  }
  queue<pair<int, int> > q;
  for (int i = 0; i < N; ++i) {
    for (int k = 0; k < M; ++k) {
      add(i, k, q);
    }
  }
  while (!q.empty()) {
    int cx = q.front().first;
    int cy = q.front().second;
    q.pop();
    upd(cx, cy);
    for (int nx = cx - 1; nx <= cx + 1; ++nx) {
      for (int ny = cy - 1; ny <= cy + 1; ++ny) {
        if (!(nx == cx && ny == cy)) {
          add(nx, ny, q);
        }
      }
    }
  }
  for (int i = 0; i < N; ++i) {
    cout << ar[i] << '\n';
  }
  return 0;
}
