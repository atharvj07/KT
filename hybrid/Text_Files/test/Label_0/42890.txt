a = list(map(int, input().split()))
a.sort()
if a[0] == 0:
	print(a[1] // 3 + a[2] // 3)
	exit()
b = a.copy()
c = a.copy()
ans = min(a)
a[1] -= ans
a[2] -= ans
ans += a[1] // 3
ans += a[2] // 3
ans1 = min(b) - 1
b[1] -= ans1
b[2] -= ans1
ans1 += b[1] // 3
ans1 += b[2] // 3
ans2 = min(c) - 2
c[1] -= ans2
c[2] -= ans2
ans2 += c[1] // 3
ans2 += c[2] // 3
print(max(ans2, ans1, ans))
