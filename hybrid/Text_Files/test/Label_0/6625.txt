#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
int n, mod, phi, Q, pnum;
int a[N], P[11], C[11];
namespace IO {
int read() {
  int ret = 0;
  char c = getchar();
  while (!isdigit(c)) c = getchar();
  while (isdigit(c)) ret = ret * 10 + (c ^ 48), c = getchar();
  return ret;
}
void write(int x) {
  if (x > 9) write(x / 10);
  putchar(x % 10 ^ 48);
}
void writeln(int x) {
  write(x);
  putchar('\n');
}
}  // namespace IO
using namespace IO;
namespace Math {
int upm(int x) { return x >= mod ? x - mod : (x < 0 ? x + mod : x); }
void up(int &x, int y) { x = upm(x + y); }
int mul(int x, int y) { return 1ll * x * y % mod; }
int getphi(int n) {
  int res = n;
  for (int i = 2; 1ll * i * i <= n; ++i) {
    if (!(n % i)) res = res / i * (i - 1);
    while (!(n % i)) n /= i;
  }
  if (n > 1) res = res / n * (n - 1);
  return res;
}
int qpow(int x, int y) {
  int res = 1;
  for (; y; y >>= 1, x = mul(x, x))
    if (y & 1) res = mul(res, x);
  return res;
}
int inv(int x) { return qpow(x, phi - 1); }
void divide(int x) {
  int t = x;
  for (int i = 2; t > 1 && 1ll * i * i <= x; ++i) {
    if (!(t % i)) P[++pnum] = i;
    while (!(t % i)) t /= i, ++C[pnum];
  }
  if (t > 1) P[++pnum] = t;
  C[pnum] = 1;
}
void calc(int x, int &res, int *p) {
  for (int i = 1; i <= pnum; ++i)
    while (!(x % P[i])) x /= P[i], p[i]++;
  res = x % mod;
}
int calcres(int x, int *p) {
  for (int i = 1; i <= pnum; ++i) x = mul(x, qpow(P[i], p[i]));
  return x;
}
}  // namespace Math
using namespace Math;
namespace Data_Structure {
struct node {
  int tar, itar, val, p[12];
};
struct Seg {
  node t[N << 2];
  void pushdown(int x) {
    for (int i = 1; i <= pnum; ++i) {
      t[(x << 1)].p[i] += t[x].p[i];
      t[(x << 1 | 1)].p[i] += t[x].p[i];
      t[x].p[i] = 0;
    }
    t[(x << 1)].val = mul(t[(x << 1)].val, t[x].tar);
    t[(x << 1 | 1)].val = mul(t[(x << 1 | 1)].val, t[x].tar);
    t[(x << 1)].tar = mul(t[(x << 1)].tar, t[x].tar);
    t[(x << 1 | 1)].tar = mul(t[(x << 1 | 1)].tar, t[x].tar);
    t[(x << 1)].itar = mul(t[(x << 1)].itar, t[x].itar);
    t[(x << 1 | 1)].itar = mul(t[(x << 1 | 1)].itar, t[x].itar);
    t[x].tar = t[x].itar = 1;
  }
  void pushup(int x) { t[x].val = upm(t[(x << 1)].val + t[(x << 1 | 1)].val); }
  void build(int x, int l, int r) {
    t[x].tar = t[x].itar = 1;
    memset(t[x].p, 0, sizeof(t[x].p));
    if (l == r) {
      t[x].val = a[l] % mod;
      calc(a[l], t[x].itar, t[x].p);
      return;
    }
    int mid = (l + r) >> 1;
    build((x << 1), l, mid);
    build((x << 1 | 1), mid + 1, r);
    pushup(x);
  }
  void update1(int x, int l, int r, int L, int R, int v, int iv, int *p) {
    if (L <= l && r <= R) {
      for (int i = 1; i <= pnum; ++i) t[x].p[i] += p[i];
      t[x].val = mul(t[x].val, v);
      t[x].tar = mul(t[x].tar, v);
      t[x].itar = mul(t[x].itar, iv);
      return;
    }
    pushdown(x);
    int mid = (l + r) >> 1;
    if (L <= mid) update1((x << 1), l, mid, L, R, v, iv, p);
    if (R > mid) update1((x << 1 | 1), mid + 1, r, L, R, v, iv, p);
    pushup(x);
  }
  void update2(int x, int l, int r, int pos, int v, int *p) {
    if (l == r) {
      for (int i = 1; i <= pnum; ++i) t[x].p[i] -= p[i];
      t[x].itar = mul(t[x].itar, inv(v));
      t[x].val = calcres(t[x].itar, t[x].p);
      return;
    }
    pushdown(x);
    int mid = (l + r) >> 1;
    if (pos <= mid)
      update2((x << 1), l, mid, pos, v, p);
    else
      update2((x << 1 | 1), mid + 1, r, pos, v, p);
    pushup(x);
  }
  int query(int x, int l, int r, int L, int R) {
    if (L <= l && r <= R) return t[x].val;
    pushdown(x);
    int mid = (l + r) >> 1, res = 0;
    if (L <= mid) up(res, query((x << 1), l, mid, L, R));
    if (R > mid) up(res, query((x << 1 | 1), mid + 1, r, L, R));
    return res;
  }
} tr;
}  // namespace Data_Structure
using namespace Data_Structure;
namespace DreamLolita {
void solution() {
  n = read();
  mod = read();
  phi = getphi(mod);
  divide(mod);
  for (int i = 1; i <= n; ++i) a[i] = read();
  tr.build(1, 1, n);
  Q = read();
  while (Q--) {
    static int ps[11];
    int op = read(), a = read(), b = read(), x, v;
    memset(ps, 0, sizeof(ps));
    if (op == 1)
      x = read(), calc(x, v, ps), tr.update1(1, 1, n, a, b, x, v, ps);
    else if (op == 2)
      calc(b, v, ps), tr.update2(1, 1, n, a, v, ps);
    else
      writeln(tr.query(1, 1, n, a, b));
  }
}
}  // namespace DreamLolita
int main() {
  DreamLolita::solution();
  return 0;
}
