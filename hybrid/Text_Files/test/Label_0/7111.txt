#include <bits/stdc++.h>
using namespace std;
template <typename T>
void read(T &x) {
  x = 0;
  char ch = getchar();
  int fh = 1;
  while (ch < '0' || ch > '9') {
    if (ch == '-') fh = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') x = x * 10 + ch - '0', ch = getchar();
  x *= fh;
}
template <typename T>
void write(T x) {
  if (x < 0) x = -x, putchar('-');
  if (x > 9) write(x / 10);
  putchar(x % 10 + '0');
}
template <typename T>
void writeln(T x) {
  write(x);
  puts("");
}
int n, q;
int a[100005];
int pre[100005];
vector<int> g[100005];
int depth[100005];
int st[2000005];
int ans[100005];
void dfs(int x, int fa) {
  pre[x] = fa;
  ans[x] = -1;
  for (int i = 1; i * i <= a[x]; ++i)
    if (a[x] % i == 0) {
      if (i != 1 && st[i] != -1) {
        if (ans[x] == -1)
          ans[x] = st[i];
        else if (depth[ans[x]] < depth[st[i]])
          ans[x] = st[i];
      }
      if (a[x] / i != 1 && st[a[x] / i] != -1) {
        if (ans[x] == -1)
          ans[x] = st[a[x] / i];
        else if (depth[ans[x]] < depth[st[a[x] / i]])
          ans[x] = st[a[x] / i];
      }
    }
  vector<pair<int, int>> chg;
  for (int i = 1; i * i <= a[x]; ++i)
    if (a[x] % i == 0) {
      chg.push_back(make_pair(i, st[i]));
      st[i] = x;
      chg.push_back(make_pair(a[x] / i, st[a[x] / i]));
      st[a[x] / i] = x;
    }
  for (int i = 0; i < g[x].size(); ++i)
    if (g[x][i] != fa) {
      depth[g[x][i]] = depth[x] + 1;
      dfs(g[x][i], x);
    }
  for (int i = chg.size() - 1; i >= 0; --i) st[chg[i].first] = chg[i].second;
}
signed main() {
  cin >> n >> q;
  for (int i = 1; i <= n; ++i) cin >> a[i];
  for (int i = 1; i < n; ++i) {
    int x, y;
    cin >> x >> y;
    g[x].push_back(y);
    g[y].push_back(x);
  }
  memset(st, -1, sizeof(st));
  dfs(1, -1);
  while (q--) {
    int op;
    cin >> op;
    if (op == 1) {
      int v;
      cin >> v;
      cout << ans[v] << endl;
    } else {
      int v, w;
      cin >> v >> w;
      a[v] = w;
      vector<int> path;
      while (v != -1) {
        path.push_back(v);
        v = pre[v];
      }
      memset(st, -1, sizeof(st));
      for (int i = path.size() - 1; i >= 1; --i) {
        int x = path[i];
        ans[x] = -1;
        for (int j = 1; j * j <= a[x]; ++j)
          if (a[x] % j == 0) {
            if (j != 1 && st[j] != -1) {
              if (ans[x] == -1)
                ans[x] = st[j];
              else if (depth[ans[x]] < depth[st[j]])
                ans[x] = st[j];
            }
            if (a[x] / j != 1 && st[a[x] / j] != -1) {
              if (ans[x] == -1)
                ans[x] = st[a[x] / j];
              else if (depth[ans[x]] < depth[st[a[x] / j]])
                ans[x] = st[a[x] / j];
            }
          }
        for (int j = 1; j * j <= a[x]; ++j)
          if (a[x] % j == 0) {
            st[j] = x;
            st[a[x] / j] = x;
          }
      }
      dfs(path[0], (path.size() == 1 ? -1 : path[1]));
    }
  }
  return 0;
}
