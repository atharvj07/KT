#include <bits/stdc++.h>
using namespace std;
const double pi = acos(-1.);
int gcd(int a, int b) {
  if (b == 0) {
    return a;
  }
  return gcd(b, a % b);
}
int main() {
  int x, y, n;
  cin >> x >> y >> n;
  if (y <= n) {
    cout << x / gcd(x, y) << "/" << y / gcd(x, y);
    return 0;
  }
  double goal = (double)x / (double)y;
  double top = 1;
  double bottom = 1;
  double rtop = 1;
  double rbottom = 0;
  double ltop = 0;
  double lbottom = 1;
  double res = top / bottom;
  double bestdiff = abs(0 - goal);
  int besttop = 0;
  int bestbottom = 1;
  while (true) {
    double diff = abs(res - goal);
    if (diff < bestdiff) {
      bestdiff = abs(res - goal);
      besttop = top;
      bestbottom = bottom;
    }
    if (res > goal) {
      res = (top + ltop) / (bottom + lbottom);
      rtop = top;
      rbottom = bottom;
      bottom = bottom + lbottom;
      top = top + ltop;
    } else {
      res = (top + rtop) / (bottom + rbottom);
      ltop = top;
      lbottom = bottom;
      bottom = bottom + rbottom;
      top = top + rtop;
    }
    if (bottom > n) {
      break;
    }
  }
  if (bestbottom == 0) {
    bestbottom = 1;
  }
  cout << besttop << "/" << bestbottom;
  return 0;
}
