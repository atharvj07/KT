#include <bits/stdc++.h>
using namespace std;
const int maxN = 1005, INF = 1e9;
int n, len;
int w[10], f[maxN + 1][maxN + 1];
int id[maxN + 1], l[maxN + 1];
char A[maxN + 1], B[maxN + 1][maxN + 1];
vector<int> num[10];
int main() {
  scanf("%s", A + 1);
  l[0] = strlen(A + 1);
  len = max(len, l[0]);
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%s", B[i] + 1);
    l[i] = strlen(B[i] + 1);
    len = max(len, l[i]);
  }
  for (int i = 0; i < 10; i++) scanf("%d", &w[i]);
  reverse(A + 1, A + l[0] + 1);
  for (int i = len; i > l[0]; i--) A[i] = '0';
  for (int i = 1; i <= n; i++) {
    reverse(B[i] + 1, B[i] + l[i] + 1);
    for (int j = len; j > l[i]; j--) B[i][j] = '0';
  }
  for (int i = 1; i <= n; i++) id[i] = i;
  for (int i = 0; i <= len; i++)
    for (int j = 0; j <= n; j++) f[i][j] = -INF;
  f[0][0] = 0;
  for (int i = 1; i <= len; i++) {
    for (int j = 0; j < 10; j++) {
      if (A[i] != '?' && A[i] - '0' != j) continue;
      if (i == l[0] && j == 0) continue;
      int s = 0, res = 0;
      for (int k = 1; k <= n; k++) {
        if (i > l[0] && i > l[k]) continue;
        int t = B[k][i] - '0' + j;
        if (t >= 10) s++;
        res += w[t % 10];
      }
      if (f[i - 1][0] != -INF) f[i][s] = max(f[i][s], f[i - 1][0] + res);
      for (int k = 1; k <= n; k++) {
        int t = B[id[k]][i] - '0' + j;
        if (i > l[0] && i > l[id[k]])
          res += w[(t + 1) % 10];
        else
          res += w[(t + 1) % 10] - w[t % 10];
        if (t + 1 == 10) s++;
        if (f[i - 1][k] != -INF) f[i][s] = max(f[i][s], f[i - 1][k] + res);
      }
    }
    for (int j = 0; j < 10; j++) num[j].clear();
    for (int j = 1; j <= n; j++) {
      int t = B[id[j]][i] - '0';
      num[t].push_back(id[j]);
    }
    int c = 0;
    for (int j = 9; j >= 0; j--)
      for (int k = 0; k < num[j].size(); k++) id[++c] = num[j][k];
  }
  int ans = -INF;
  for (int i = 0; i <= n; i++)
    if (f[len][i] != -INF) ans = max(ans, f[len][i] + i * w[1]);
  printf("%d\n", ans);
  return 0;
}
