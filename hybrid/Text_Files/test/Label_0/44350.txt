t = int(input())
for j in range(t):
	(n, x) = list(map(int, input().split()))
	dmax = 0
	smax = 0
	for i in range(n):
		(d, h) = list(map(int, input().split()))
		dmax = max(dmax, d)
		smax = max(smax, d - h)
	if dmax >= x:
		print(1)
	elif smax == 0:
		print(-1)
	else:
		ct = (x - dmax) // smax
		if (x - dmax) % smax != 0:
			ct += 1
		print(ct + 1)
