import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.io.StreamTokenizer;

public class B176 {

    static StreamTokenizer in = new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));
    static PrintWriter out = new PrintWriter(System.out);
    
    static int nextInt() throws Exception {
        in.nextToken();
        return (int)in.nval;
    }
    
    static String nextString() throws Exception {
        in.nextToken();
        return in.sval;
    }
    
    public static void main(String[] args) throws Exception {
        final int MD = 1000000007;
        
        String a = nextString();
        String b = nextString();
        String c = a + a;
        int k = nextInt();
        
        int n = a.length();
        int cnt = 0;
        boolean good = false;
        for (int i = 0; i < n; i++) {
            if (a.equals(c.substring(i, i+n))) cnt++;
            good |= b.equals(c.substring(i, i+n));
        }
        int q = n/cnt - 1;
        
        long init = 1, other = 0;
        for (int i = 0; i < k; i++) {
            long newInit = (other*q*cnt + init*(cnt-1))%MD;
            long newOther = (other*(q-1)*cnt + other*(cnt-1) + init*cnt)%MD;
            init = newInit;
            other = newOther;
        }
        
        if (!good) out.println(0);
        else if (a.equals(b)) out.println(init);
        else out.println(other);
        
        out.flush();
    }

}
