a = [int(i) for i in list(input())]
b = [int(i) for i in list(input())]
aa = len(a)
bb = len(b)
l = t = 0
t = bb - aa + 1
ans = 0
for i in range(bb - aa + 1):
	l += int(b[i])
for i in range(aa):
	if a[i] == 1:
		ans += t - l
	else:
		ans += l
	if i == len(a) - 1:
		break
	l += int(b[t + i])
	l -= int(b[i])
print(ans)
