for _ in range(0, int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	a.sort()
	ans = [a[(n - 1) // 2]] if n % 2 else []
	p = n // 2 - 1 if n % 2 == 0 else (n - 1) // 2 - 1
	q = n // 2 if n % 2 == 0 else (n - 1) // 2 + 1
	for k in range(n // 2):
		ans.append(a[p])
		ans.append(a[q])
		p -= 1
		q += 1
	print(*ans)
