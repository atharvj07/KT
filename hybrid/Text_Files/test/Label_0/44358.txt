import math
x = list(map(int, input().split()))
num_types = x[0]
pocket_size = x[1]
pebble_list = list(map(int, input().split()))
num_pockets_filled = 0
for i in range(num_types):
	curr = pebble_list[i]
	num_pockets_filled += math.ceil(curr / pocket_size)
print(num_pockets_filled // 2 + num_pockets_filled % 2)
