import io, os
input = io.StringIO(os.read(0, os.fstat(0).st_size).decode()).readline
ii = lambda : int(input())
kk = lambda : map(int, input().split())
ll = lambda : list(kk())

def bs(ls):
	c = 0
	for _ in range(len(ls)):
		for i in range(len(ls) - 1):
			if ls[i] > ls[i + 1]:
				c += 1
				(ls[i + 1], ls[i]) = (ls[i], ls[i + 1])
	return c & 1
for _ in range(ii()):
	(l, a, b) = (ii(), list(input()), list(input()))
	(da, db) = ({}, {})
	for c in a:
		if c not in da:
			da[c] = 0
		da[c] += 1
	for c in b:
		if c not in db:
			db[c] = 0
		db[c] += 1
	ok = False
	for k in da:
		if k not in db or db[k] != da[k]:
			print('NO')
			break
		if da[k] > 1:
			ok = True
	else:
		if ok:
			print('YES')
		else:
			print('YNEOS'[bs(a) != bs(b)::2])
