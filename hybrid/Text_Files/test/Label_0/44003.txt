import math

class Pair:

	def __init__(self, x=0, y=0):
		self.x = x
		self.y = y
		self.neighbors = []
		self.visited = False

def isMedian(pair1, pair2):
	if pair1.x > pair2.x and pair1.x < pair2.y:
		return True
	elif pair1.y > pair2.x and pair1.y < pair2.y:
		return True
	else:
		return False

def dfs(startPair, targPair):
	stack = []
	stack.append(startPair)
	while len(stack) > 0:
		currPair = stack.pop()
		if currPair.visited == True:
			continue
		currPair.visited = True
		if currPair.x == targPair.x and currPair.y == targPair.y:
			return True
		for n in currPair.neighbors:
			stack.append(n)
	return False
numQueries = input()
numQueries = int(numQueries)
count = 0
pairs = {}
results = []
for x in range(0, numQueries):
	nums = list(map(int, input().split()))
	queryType = nums[0]
	first = nums[1]
	second = nums[2]
	if queryType == 1:
		newPair = Pair(first, second)
		for key in pairs:
			if isMedian(newPair, pairs[key]) == True:
				newPair.neighbors.append(pairs[key])
			if isMedian(pairs[key], newPair) == True:
				pairs[key].neighbors.append(newPair)
		pairs[count] = newPair
		count += 1
	else:
		firstPair = pairs[first - 1]
		secondPair = pairs[second - 1]
		for tempKey in pairs:
			pairs[tempKey].visited = False
		if dfs(firstPair, secondPair) == True:
			results.append('YES')
		else:
			results.append('NO')
for result in results:
	print(result)
