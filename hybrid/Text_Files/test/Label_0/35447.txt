import java.util.Scanner;


public class Task1 {
	static Scanner sc = new Scanner(System.in);
	
	public static void main(String[] args) {
		long h1 = sc.nextInt();
		long h2 = sc.nextInt();
		long a = sc.nextInt();
		long b = sc.nextInt();
		
		if(b>=a) {
			if(h2-h1 <= 8*a) {
				System.out.println(0);
			} else {
				System.out.println(-1);
			}
			return;
		}
		
		long days = -1;
		long h = h1-4*a;
		while(true) {
			days++;
			h += 12*a;
			
			if(h>=h2) {
				break;
			}
			
			h -= 12*b;
		}
		System.out.println(days);
	}
	
}
