#include <bits/stdc++.h>
using namespace std;
int t[100000], first[100000], second[100000];
int ccw(pair<double, double> a, pair<double, double> b,
        pair<double, double> c) {
  return (b.first - a.first) * (c.second - a.second) >
         (b.second - a.second) * (c.first - a.first);
}
bool comp(pair<double, double> a, pair<double, double> b) {
  if (abs(a.first - b.first) < 1e-9)
    return a.second < b.second;
  else
    return a.first < b.first;
}
bool comp2(pair<double, double> a, pair<double, double> b) {
  return (abs(a.first - b.first) < 1e-9) && (abs(a.second - b.second) < 1e-9);
}
vector<pair<double, double> > convexHull(vector<pair<double, double> >& vv) {
  int i;
  vector<pair<double, double> > hull;
  sort(vv.begin(), vv.end(), comp);
  vv.resize(unique(vv.begin(), vv.end(), comp2) - vv.begin());
  for (i = 0; i < vv.size(); i++) {
    while ((hull.size() >= 2) && ccw(hull[hull.size() - 2], hull.back(), vv[i]))
      hull.pop_back();
    hull.push_back(vv[i]);
  }
  int l = hull.size();
  for (i = vv.size() - 2; i >= 0; i--) {
    while ((hull.size() > l) && ccw(hull[hull.size() - 2], hull.back(), vv[i]))
      hull.pop_back();
    hull.push_back(vv[i]);
  }
  return hull;
}
int main() {
  int i;
  int n, w, h, a, b;
  scanf("%d %d %d %d %d", &n, &w, &h, &a, &b);
  for (i = 0; i < n; i++) scanf("%d %d %d", &t[i], &first[i], &second[i]);
  int j;
  double l = 0, r = 1001;
  while (abs(l - r) > 1e-9) {
    double m = (l + r) / 2;
    pair<double, double> p1, p2;
    p1 = p2 = make_pair(a, b);
    for (i = 0; i < n; i++) {
      int tt = (i == 0) ? t[i] : t[i] - t[i - 1];
      if (tt == 0) {
        double mm = (double)(second[i] - h) / second[i],
               bb = (double)h * first[i] / second[i];
        if ((abs(p1.first - p2.first) < 1e-9) &&
            (abs(p1.second - p2.second) < 1e-9)) {
          if (abs(mm * p1.first + bb - p1.second) > 1e-9) break;
        } else if (!((abs(mm * p1.first + bb - p1.second) < 1e-9) &&
                     (abs(mm * p2.first + bb - p2.second) < 1e-9))) {
          double mm2 = (p1.second - p2.second) / (p1.first - p2.first),
                 bb2 = p1.second - mm2 * p1.first;
          if (abs(mm - mm2) < 1e-9)
            break;
          else {
            double a2 = (bb2 - bb) / (mm - mm2);
            double b2 = mm * a2 + bb;
            if ((a2 < min(p1.first, p2.first) - 1e-9) ||
                (a2 > max(p1.first, p2.first) + 1e-9))
              break;
            else
              p1 = p2 = make_pair(a2, b2);
          }
        }
      } else {
        vector<pair<double, double> > vv;
        vv.push_back(make_pair(p1.first - m * tt, p1.second - m * tt));
        vv.push_back(make_pair(p1.first - m * tt, p1.second + m * tt));
        vv.push_back(make_pair(p1.first + m * tt, p1.second - m * tt));
        vv.push_back(make_pair(p1.first + m * tt, p1.second + m * tt));
        vv.push_back(make_pair(p2.first - m * tt, p2.second - m * tt));
        vv.push_back(make_pair(p2.first - m * tt, p2.second + m * tt));
        vv.push_back(make_pair(p2.first + m * tt, p2.second - m * tt));
        vv.push_back(make_pair(p2.first + m * tt, p2.second + m * tt));
        vv = convexHull(vv);
        double mm = (double)(second[i] - h) / second[i],
               bb = (double)h * first[i] / second[i];
        pair<double, double> q1, q2;
        q1 = q2 = make_pair(1e99, 1e99);
        for (j = 1; j < vv.size(); j++) {
          if (abs(vv[j - 1].first - vv[j].first) < 1e-9) {
            double b2 = mm * vv[j].first + bb;
            if ((b2 > min(vv[j - 1].second, vv[j].second) - 1e-9) &&
                (b2 < max(vv[j - 1].second, vv[j].second) + 1e-9)) {
              pair<double, double> q = make_pair(vv[j].first, b2);
              if (q1 == make_pair(1e99, 1e99))
                q1 = q;
              else if ((abs(q.first - q1.first) > 1e-9) ||
                       (abs(q.second - q1.second) > 1e-9))
                q2 = q;
            }
          } else {
            double mm2 = (vv[j - 1].second - vv[j].second) /
                         (vv[j - 1].first - vv[j].first),
                   bb2 = vv[j].second - mm2 * vv[j].first;
            if (abs(mm - mm2) > 1e-9) {
              double a2 = (bb2 - bb) / (mm - mm2);
              double b2 = mm * a2 + bb;
              if ((a2 > min(vv[j - 1].first, vv[j].first) - 1e-9) &&
                  (a2 < max(vv[j - 1].first, vv[j].first) + 1e-9)) {
                pair<double, double> q = make_pair(a2, b2);
                if (q1 == make_pair(1e99, 1e99))
                  q1 = q;
                else if ((abs(q.first - q1.first) > 1e-9) ||
                         (abs(q.second - q1.second) > 1e-9))
                  q2 = q;
              }
            }
          }
        }
        if (q2 == make_pair(1e99, 1e99)) {
          if ((q1.first > -1e-9) && (q1.first < w + 1e-9) &&
              (q1.second > -1e-9) && (q1.second < w + 1e-9))
            p1 = p2 = q1;
          else
            break;
        } else {
          vv.clear();
          vv.push_back(make_pair(0, 0)), vv.push_back(make_pair(w, 0)),
              vv.push_back(make_pair(w, w)), vv.push_back(make_pair(0, w)),
              vv.push_back(make_pair(0, 0));
          for (j = 1; j < vv.size(); j++) {
            if (abs(vv[j - 1].first - vv[j].first) < 1e-9) {
              double b2 = mm * vv[j].first + bb;
              if ((b2 > min(vv[j - 1].second, vv[j].second) - 1e-9) &&
                  (b2 < max(vv[j - 1].second, vv[j].second) + 1e-9)) {
                pair<double, double> q = make_pair(vv[j].first, b2);
                if (q1.first > q2.first) swap(q1, q2);
                if (j == 2) {
                  if (q.first < q2.first)
                    q2 = make_pair(q.first, mm * q.first + bb);
                } else {
                  if (q.first > q1.first)
                    q1 = make_pair(q.first, mm * q.first + bb);
                }
                if (q1.first > q2.first + 1e-9) break;
              }
            } else {
              double mm2 = (vv[j - 1].second - vv[j].second) /
                           (vv[j - 1].first - vv[j].first),
                     bb2 = vv[j].second - mm2 * vv[j].first;
              if (abs(mm - mm2) > 1e-9) {
                double a2 = (bb2 - bb) / (mm - mm2);
                double b2 = mm * a2 + bb;
                if ((a2 > min(vv[j - 1].first, vv[j].first) - 1e-9) &&
                    (a2 < max(vv[j - 1].first, vv[j].first) + 1e-9)) {
                  pair<double, double> q = make_pair(a2, b2);
                  if (q1.second > q2.second) swap(q1, q2);
                  if (j == 1) {
                    if (q.second > q1.second)
                      q1 = make_pair((q.second - bb) / mm, q.second);
                  } else {
                    if (q.second < q2.second)
                      q2 = make_pair((q.second - bb) / mm, q.second);
                  }
                  if (q1.second > q2.second + 1e-9) break;
                }
              }
            }
          }
          if (j < vv.size())
            break;
          else
            p1 = q1, p2 = q2;
        }
      }
    }
    if (i == n)
      r = m;
    else
      l = m;
  }
  if (l > 1000)
    printf("-1\n");
  else
    printf("%.10lf\n", l);
  return 0;
}
