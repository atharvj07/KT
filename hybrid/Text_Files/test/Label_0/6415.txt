#include <bits/stdc++.h>
using namespace std;
int mcd(int a, int b) {
  if (a > b) swap(a, b);
  if (b % a == 0)
    return a;
  else
    return mcd(a, b % a);
}
int mcm(int a, int b) { return a * b / mcd(a, b); }
int main() {
  int n, cr[110], vis[110];
  scanf("%d", &n);
  memset(vis, 0, sizeof vis);
  for (int i = 1; i <= n; i++) {
    scanf("%d", cr + i);
    vis[cr[i]] = 1;
  }
  for (int i = 1; i <= n; i++) {
    if (!vis[i]) {
      printf("-1\n");
      return 0;
    }
  }
  memset(vis, 0, sizeof vis);
  int nc = 0, c[110];
  for (int i = 1; i <= n; i++) {
    if (!vis[i]) {
      int t = 0;
      while (!vis[i]) {
        vis[i] = 1;
        t++;
        i = cr[i];
      }
      c[nc] = !(t % 2) ? t / 2 : t;
      nc++;
    }
  }
  int m = c[0];
  for (int i = 0; i < nc; i++) {
    m = mcm(m, c[i]);
  }
  printf("%d\n", m);
}
