import sys
from math import *
from math import *
from functools import *
for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split())) + [0]
	b = [-1] * (n + 1)
	a.sort()
	for i in a:
		if i <= n:
			b[i] += 1
	c = 0
	j = 1
	i = 1
	while j <= n:
		if b[j] < 0:
			if a[i] > n or b[a[i]]:
				if a[i] <= n:
					b[a[i]] -= 1
				if a[i] > 2 * j:
					c += 1
					j += 1
					i += 1
				else:
					print(-1)
					break
			elif b[a[i]] == 0:
				i += 1
		else:
			j += 1
	else:
		print(c)
