#include <bits/stdc++.h>
using namespace std;
int a[100010], b[100010];
int main() {
  int n, sum, k;
  while (~scanf("%d%d", &n, &k)) {
    sum = 0;
    for (int i = 0; i < n; i++) {
      scanf("%d", &a[i]);
    }
    sort(a, a + n);
    for (int i = 0; i < n && k; i++) {
      for (int j = n - 1 - i; j >= 0 && k; j--) {
        cout << i + 1 << " ";
        for (int q = n - 1; q > n - 1 - i; q--) {
          cout << a[q] << " ";
        }
        cout << a[j];
        cout << endl;
        k--;
      }
    }
  }
  return 0;
}
