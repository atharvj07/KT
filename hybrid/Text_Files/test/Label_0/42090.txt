(n, m, k) = map(int, input().split())
a = [list(map(int, input().split())) for i in range(n)]
out = 0
mxout = 0
x = 0
for j in range(m):
	kol = 0
	mx = -1
	for i in range(n):
		if a[i][j] == 1:
			t = 0
			for l in range(i, min(i + k, n)):
				if a[l][j] == 1:
					t += 1
			if mx < t:
				mx = t
				x = kol
			kol += 1
	out += x
	mxout += max(mx, 0)
	kol = 0
print(mxout, out)
