import java.io.StreamTokenizer;
import java.util.Scanner;

public class Sh2D {

    static Scanner in;
    static StreamTokenizer st;
    static int n;
    static Kubik[] a;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        a = new Kubik[n];
        for (int i = 0; i < n; ++i) {
            a[i] = new Kubik(in.nextDouble(), in.nextDouble(), in.nextDouble(),
                    in.nextDouble());
            if (i > 0 && !check(i)) {
                System.out.println(i);
                return;
            }
        }
        System.out.println(n);
    }

    private static boolean check(int k) {
        double curx = 0, cury = 0, cm = 0;
        for (int i = k; i >= 0; --i) {
            if (i == k) {
                curx = a[i].cx;
                cury = a[i].cy;
                cm = a[i].m;
            } else {
                double x = find(a[i].m, a[i].cx, cm, curx);
                double y = find(a[i].m, a[i].cy, cm, cury);
                curx = x;
                cury = y;
                cm += a[i].m;
            }
            if (curx > a[i].x2 || curx < a[i].x1)
                return false;
            if (cury > a[i].y2 || cury < a[i].y1)
                return false;
            if (i >= 1) {
                if (curx > a[i - 1].x2 || curx < a[i - 1].x1)
                    return false;
                if (cury > a[i - 1].y2 || cury < a[i - 1].y1)
                    return false;
            }
        }
        return true;
    }

    private static double find(double m, double cx, double cm, double curx) {
        return (m * cx + cm * curx) / (m + cm);
    }

    static class Kubik {

        double cx, cy;
        double m;
        double x1, y1, x2, y2;

        public Kubik(double xx, double yy, double x, double y) {
            x1 = Math.min(x, xx);
            x2 = x + xx - x1;
            y1 = Math.min(y, yy);
            y2 = y + yy - y1;
            m = (x2 - x1) * (x2 - x1) * (x2 - x1);
            cx = (x1 + x2) / 2.0;
            cy = (y1 + y2) / 2.0;
        }
    }
}
