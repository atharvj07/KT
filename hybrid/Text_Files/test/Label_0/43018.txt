read_integer = lambda : int(input().strip())
read_list = lambda : list(map(int, input().strip().split()))

def operations(a, b):
	if (a + b) % 2 == 1:
		return -1
	if a == b:
		if a == 0:
			return 0
		return 1
	else:
		return 2
t = read_integer()
ans = []
for _ in range(t):
	(ca, cb) = read_list()
	ans.append(operations(ca, cb))
print('\n'.join(list(map(str, ans))))
