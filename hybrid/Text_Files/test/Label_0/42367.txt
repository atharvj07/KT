def f(a):
	bad = []
	n = len(a)
	vis = [-1] * (len(a) + 1)
	nt = []
	for i in a:
		if i > n or vis[i] != -1:
			bad.append(i)
		else:
			vis[i] = 1
	for i in range(1, n + 1):
		if vis[i] == -1:
			nt.append(i)
	nt.sort(reverse=True)
	bad.sort(reverse=True)
	ans = len(bad)
	for val in bad:
		id = None
		for i in range(len(nt)):
			if val % (val - nt[i]) == nt[i]:
				id = i
				break
		if id == None:
			return -1
		nt.pop(id)
	return ans
for _ in range(int(input())):
	a = input()
	l = list(map(int, input().strip().split()))
	print(f(l))
