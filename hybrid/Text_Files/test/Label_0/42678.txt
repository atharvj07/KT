s = h = 0
number = int(input())
for x in range(number):
	lin = map(str, input().split())
	if 'soft' in lin:
		s += 1
	else:
		h += 1
if s == 0 and h == 1 or (s == 1 and h == 0):
	print(1)
	exit()
z = 1
while 1:
	l = z * z
	if l % 2 == 0:
		if l / 2 >= h and l / 2 >= s:
			print(z)
			exit()
	elif l / 2 + 1 >= h and l / 2 >= s or (l / 2 >= h and l / 2 + 1 >= s):
		print(z)
		exit()
	z += 1
