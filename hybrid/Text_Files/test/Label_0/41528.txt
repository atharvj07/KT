for t in range(int(input())):
	l = list(map(int, input().split()))
	if max(l) == min(l):
		print(0)
	else:
		p = sorted(l)
		if p[0] == p[1] or p[1] == p[2]:
			if abs(max(l) - min(l)) >= 2:
				print((abs(max(l) - min(l)) - 2) * 2)
			else:
				print((abs(max(l) - min(l)) - 1) * 2)
		else:
			p[0] += 1
			p[2] -= 1
			c = abs(p[2] - p[1]) + abs(p[2] - p[0]) + abs(p[1] - p[0])
			print(c)
