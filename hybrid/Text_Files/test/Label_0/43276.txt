import sys

def input():
	return sys.stdin.readline().rstrip('\r\n')

def maps():
	return [int(i) for i in input().split()]
from collections import defaultdict
from math import ceil

def func(n, m, a):
	A = sorted([(j, i) for (i, j) in enumerate(a)], key=lambda x: (x[0], -x[1]))
	s = [0] * m
	for i in range(m):
		s[A[i][1]] = i
	vis = [0] * m
	ans = 0
	for i in range(m):
		vis[s[i]] = 1
		ans += sum(vis[:s[i]])
	return ans
for i in range(*maps()):
	(n, m) = maps()
	a = sorted(([j, i] for (i, j) in enumerate(maps())))
	for i in range(n * m):
		a[i][1] = -a[i][1]
	ans = 0
	for i in range(n):
		x = a[i * m:i * m + m]
		x.sort()
		for j in range(m):
			ans += sum((1 if x[j][1] < x[k][1] else 0 for k in range(j)))
	print(ans)
