import sys, math, heapq, collections, itertools, bisect
sys.setrecursionlimit(101000)
n = int(input())
a = list(map(int, input().split()))
f = [0] * (10 ** 5 + 5)
f2 = [0]
f4 = [0]
f6 = [0]
f8 = [0]

def _set(s, x):
	if s == '+':
		f[x] += 1
		if f[x] == 2:
			f2[0] += 1
		elif f[x] == 4:
			f4[0] += 1
		elif f[x] == 6:
			f6[0] += 1
		elif f[x] == 8:
			f8[0] += 1
	else:
		f[x] -= 1
		if f[x] == 7:
			f8[0] -= 1
		elif f[x] == 5:
			f6[0] -= 1
		elif f[x] == 3:
			f4[0] -= 1
		elif f[x] == 1:
			f2[0] -= 1

def _can():
	if f8[0] > 0:
		return True
	if f6[0] > 1 or f4[0] > 1:
		return True
	if f6[0] == 1 and f2[0] > 1:
		return True
	if f4[0] == 1 and f2[0] > 2:
		return True
	return False
for x in a:
	_set('+', x)
q = int(input())
for _ in range(q):
	(s, x) = input().split()
	_set(s, int(x))
	print('YES' if _can() else 'NO')
