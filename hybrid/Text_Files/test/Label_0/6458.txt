#include <bits/stdc++.h>

using namespace std;

int main() {
  int Tc;
  double arr[6] = {}, x;
  cin >> Tc;
  while(Tc--) {
    cin >> x;
    if(x < 165.0) arr[0] ++;
    else if(x < 170.0) arr[1] ++;
    else if(x < 175.0) arr[2] ++;
    else if(x < 180.0) arr[3] ++;
    else if(x < 185.0) arr[4] ++;
    else arr[5] ++;
  }
  
  for(int i=0; i<6; i++) {
    cout << i+1 << ":";
    for(int j=0; j<arr[i]; j++) {
      cout << '*';
    }
    cout << endl;
  }
  
  return 0;
}
