import math

class SegmentTree:

	def __init__(self, nums) -> None:
		self.l = len(nums)
		self.tree = [0] * self.l + nums
		for i in range(self.l - 1, 0, -1):
			self.tree[i] = math.gcd(self.tree[2 * i], self.tree[2 * i + 1])

	def sumRange(self, i, j):
		m = self.l + i
		n = self.l + j
		res = self.tree[m]
		while m <= n:
			if m & 1:
				res = math.gcd(res, self.tree[m])
				m += 1
			m >>= 1
			if n & 1 == 0:
				res = math.gcd(res, self.tree[n])
				n -= 1
			n >>= 1
		return res
