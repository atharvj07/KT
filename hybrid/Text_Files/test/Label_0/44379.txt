s = input()
k = int(input())
first = []
second = []
i = 0
while i < len(s):
	if s[i] == '?':
		first.append(i - 1)
		s = s[:i] + s[i + 1:]
	elif s[i] == '*':
		second.append(i - 1)
		s = s[:i] + s[i + 1:]
	i += 1
if len(s) == k:
	print(s)
	exit()
if len(second) > 0:
	if len(s) < k:
		print(s[:second[0]] + s[second[0]] * (k - len(s) + 1) + s[second[0] + 1:])
	else:
		for i in range(len(first)):
			s = s[:first[i]] + s[first[i] + 1:]
			for j in range(len(second)):
				if second[j] > first[i]:
					second[j] -= 1
			for j in range(len(first)):
				if first[j] > first[i]:
					first[j] -= 1
			if len(s) == k:
				print(s)
				exit()
		for i in range(len(second)):
			s = s[:second[i]] + s[second[i] + 1:]
			for j in range(len(second)):
				if second[j] > second[i]:
					second[j] -= 1
			if len(s) == k:
				print(s)
				exit()
		print('Impossible')
		exit()
elif len(s) < k:
	print('Impossible')
	exit()
else:
	for i in range(len(first)):
		s = s[:first[i]] + s[first[i] + 1:]
		for j in range(len(first)):
			if first[j] > first[i]:
				first[j] -= 1
		if len(s) == k:
			print(s)
			exit()
	print('Impossible')
	exit()
