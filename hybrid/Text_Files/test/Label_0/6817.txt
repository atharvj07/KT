#include <bits/stdc++.h>
using namespace std;
signed main() {
  ios::sync_with_stdio(false);
  cin.tie(0), cout.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> b(n);
    map<int, int> ma;
    bool bad = 0;
    for (int i = 0; i < n; ++i) {
      cin >> b[i];
      if (!bad && ma[b[i]]++ == 1) {
        bad = 1;
      }
    }
    if (bad) {
      cout << "YES\n";
    } else {
      cout << "NO\n";
    }
  }
  return 0;
}
