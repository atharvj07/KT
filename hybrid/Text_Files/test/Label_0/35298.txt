import java.util.*;
public class C {
	Scanner sc = new Scanner(System.in);
	
	class House {
		int in_h = 0, out_h = 0, in_d = Integer.MAX_VALUE, out_d = Integer.MAX_VALUE;
	}
	class Ans {
		int from, to, d;
		Ans(int from, int to, int d) { this.from = from; this.to = to; this.d = d; }
	}
	void doIt()
	{
		int n = sc.nextInt(), p = sc.nextInt();
		House[] h = new House[n+1];
		for(int i = 1; i <= n; i++) {
			h[i] = new House();
			//System.out.println(h[i].in_d + " " + h[i].out_d);
		}
		for(int i = 0; i < p; i++) {
			int a = sc.nextInt(), b = sc.nextInt(), d = sc.nextInt();
			h[a].out_h = b;
			h[a].out_d = d;
			h[b].in_h = a;
			h[b].in_d = d;
		}
		ArrayList<Ans> ans = new ArrayList<Ans>();
		for(int i = 1; i <= n; i++) {
			if(h[i].in_h == 0) {
				int pos = i, min_d = Integer.MAX_VALUE;
				while(h[pos].out_h != 0) {
					min_d = Math.min(min_d, h[pos].out_d);
					pos = h[pos].out_h;
				}
				if(i != pos) ans.add(new Ans(i, pos, min_d));
			}
		}
		System.out.println(ans.size());
		for(Ans a: ans) {
			System.out.println(a.from + " " + a.to + " " + a.d);
		}
	}
	public static void main(String[] args) {
		new C().doIt();
	}

}
