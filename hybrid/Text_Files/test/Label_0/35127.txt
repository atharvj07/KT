
import java.io.BufferedReader;

public class D {

    static int n=1000000;
    
    static void solve(int[] x) {
        boolean[] 
                bx=new boolean[n], 
                by=new boolean[n];
        for (int i=0; i<x.length; i++) {
            bx[x[i]]=true;
        }
        
        int free_cnt=0, pending_cnt=0, cnt=0;
        int[] free=new int[n];

        for (int i=0; i<n>>1; i++) {
            if(bx[i]||bx[n-1-i]) {
                if(bx[i]&&bx[n-1-i]) {
                    if(free_cnt>0) {
                        by[free[free_cnt-1]]=by[n-1-free[free_cnt-1]]=true;
                        cnt+=2;
                        free_cnt--;
                    } else {
                        pending_cnt++;
                    }
                } else {
                    if(bx[i]) {
                        by[n-1-i]=true;
                        cnt++;
                    } else {
                        by[i]=true;
                        cnt++;
                    }
                }
            } else {
                if(pending_cnt>0) {
                    by[i]=by[n-1-i]=true;
                    cnt+=2;
                    pending_cnt--;
                } else {
                    free[free_cnt++]=i;
                }
            }
        }
        
        sb.setLength(0);
        sb.append(cnt+"\n");
        for (int i=0; i<n; i++) {
            if(by[i]) {
                sb.append((i+1)+" ");
            }
        }
        sb.setLength(sb.length()-1);
        
        System.out.println(sb.toString());
    }
    
    static StringBuilder sb=new StringBuilder();
    static void parse(String l, int[] here) {
        sb.setLength(0);
        int len=l.length(), j=0;
        for (int i=0; i<len; i++) {
            char c=l.charAt(i);
            if(c==' ') {
                here[j++]=Integer.parseInt(sb.toString())-1;
                sb.setLength(0);
            } else {
                sb.append(c);
            }
        }
        if(sb.length()>0) {
            here[j++]=Integer.parseInt(sb.toString())-1;
        }
    }
    
    static void run_stream(InputStream ins) throws IOException {
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        int n=Integer.parseInt(br.readLine());
        int[] x=new int[n];
        parse(br.readLine(), x);
        solve(x);
    }
