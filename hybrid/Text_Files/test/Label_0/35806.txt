import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.Scanner;
import java.util.StringTokenizer;
import java.util.Iterator;
import java.util.PriorityQueue;

public class MonopoleMagnets {
	public static void main(String args[]) throws IOException{
		
		BufferedReader in=new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st;
		
		
			
			String first1=in.readLine();
			st=new StringTokenizer(first1);
			int rows=Integer.parseInt(st.nextToken());
			int columns=Integer.parseInt(st.nextToken());
			
			char[][] squares=new char[rows][columns];
			
			HashSet<Pair> Hashtag=new HashSet<Pair>();
			
			for(int i=0; i<rows; i++) {
				String first2=in.readLine();
				for(int j=0; j<columns; j++) {
					squares[i][j]=first2.charAt(j);
					if(squares[i][j]=='#') {
						Hashtag.add(new Pair(i,j));
					}
					
				}
			}
			
			boolean death=false;
			boolean emptyC=false;
			boolean emptyR=false;
				
			for(int i=0; i<rows; i++) {
				int yes=0;
				for(int j=0; j<columns; j++) {
					if(squares[i][j]=='#') {
						if(yes==0) {
							yes=1;
						}
						else if(yes==2) {
							death=true;
							break;
						}
					}
					else {
						if(yes==1) {
							yes=2;
						}
					}
				}
				if(death) {
					break;
				}
				if(yes==0) {
					emptyR=true;
				}
			}
			
			if(!death) {
				for(int i=0; i<columns; i++) {
					int yes=0;
					for(int j=0; j<rows; j++) {
						if(squares[j][i]=='#') {
							if(yes==0) {
								yes=1;
							}
							else if(yes==2) {
								death=true;
								break;
							}
						}
						else {
							if(yes==1) {
								yes=2;
							}
						}
					}
					if(death) {
						break;
					}
					if(yes==0) {
						emptyC=true;
					}
				}
			}
			
			
			int count=0;
			
			if(!(emptyC^emptyR)&&!death) {
				while(Hashtag.size()>0) {
					HashSet<Pair> visited=new HashSet<Pair>();
					count++;
					Iterator<Pair> iterator = Hashtag.iterator();
					Pair hi=iterator.next();
					floodfill(Hashtag,rows,columns,hi,visited);
					visited.clear();
				}
				System.out.println(count);
			}
			else {
				System.out.println(-1);
			}
			
		}
	
	public static void floodfill(HashSet<Pair> Hashtag, int rows, int columns, Pair current, HashSet<Pair> visited) {

		visited.add(new Pair(current.x,current.y));
		
		ArrayList<Pair> queue=new ArrayList<Pair>();
		queue.add(current);
		
		while(!queue.isEmpty()) {
			
			current=queue.remove(0);
			Hashtag.remove(current);
			
			if(current.x>0) {
				Pair temp=new Pair(current.x-1,current.y);
				if(!visited.contains(new Pair(current.x-1,current.y))&&
						Hashtag.contains(temp)) {
					visited.add(temp);
					queue.add(temp);
				}
			}
			if(current.y>0) {
				Pair temp=new Pair(current.x,current.y-1);
				if(!visited.contains(new Pair(current.x,current.y-1))&&
						Hashtag.contains(temp)) {
					visited.add(temp);
					queue.add(temp);
				}
			}
			if(current.x<rows-1) {
				Pair temp=new Pair(current.x+1,current.y);
				if(!visited.contains(new Pair(current.x+1,current.y))&&
						Hashtag.contains(temp)) {
					visited.add(temp);
					queue.add(temp);
				}
			}
			if(current.y<columns-1) {
				Pair temp=new Pair(current.x,current.y+1);
				if(!visited.contains(new Pair(current.x,current.y+1))&&
						Hashtag.contains(temp)) {
					visited.add(temp);
					queue.add(temp);
				}
			}
		}
	}
}
class Pair{
	int x;
	int y;
	
	public Pair(int x, int y) {
		this.x=x;
		this.y=y;
	}
	
	@Override
	public boolean equals(Object hi) {
		
		Pair yes=(Pair) hi;
		
		if(yes.x==this.x&&yes.y==this.y) {
			return true;
		}
		return false;
	}
	
	@Override
	public int hashCode() {
		return 1001*this.x+this.y;
	}
	
}
