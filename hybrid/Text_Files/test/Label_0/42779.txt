n = int(input())
l = list(map(int, input().split()))
if len(set(l)) == 1:
	print(n)
	for i in l:
		print(i, end=' ')
	print()
	exit(0)
(temp, m) = ([0], [])
for i in range(len(l)):
	if l[i] > temp[-1]:
		temp.append(l[i])
	else:
		m.append(temp[1:])
		temp.clear()
		temp.append(0)
		temp.append(l[i])
if len(temp) > 1:
	m.append(temp[1:])
print(len(m))
for i in m:
	print(len(i), end=' ')
print()
