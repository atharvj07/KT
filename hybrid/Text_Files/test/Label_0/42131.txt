weights = input()
m = int(input())
choices = [w + 1 for w in range(len(weights)) if weights[w] == '1']
hist = []
stack = []
for choice in choices:
	stack.append(choice)
	stack.append(choice)
while stack and len(hist) < m:
	state = stack.pop()
	weight = stack.pop()
	stack.append(-1)
	for choice in choices:
		dif = choice - state
		if choice != weight and dif > 0:
			stack.append(choice)
			stack.append(dif)
	if stack[-1] == -1 and len(hist) < m - 1:
		del stack[-1]
		while stack and stack[-1] == -1:
			del stack[-1]
			del hist[-1]
	else:
		hist.append(weight)
if len(hist) == m:
	print('YES')
	print(*hist)
else:
	print('NO')
