    final long mod = (long)1e9 + 7;
    int n, m, k;
    int[][] grid, org;
    int cntOrgCol;
    boolean[] orgCol;
    long cntIter = 0;
    long ans = 0;
    long[] fact = new long[20];{
        fact[0] = 1;
        for( int i = 1; i < fact.length; ++i ){
            fact[i] = i * fact[i-1];
            fact[i] %= mod;
        }
    }

    void show(){
        for( int i = 0; i<n; ++i ){
            for( int j = 0; j < m; ++j )
                out.printf( "%2d", grid[i][j] );
            out.println();
        }
        out.println();
    }

    boolean haveSame( int ie, int je, int col ){
        for( int i = 0; i <= ie; ++i )
        for( int j = 0; j <= je; ++j ){
            if( i==ie && j==je ) return false;
            if( grid[i][j]==col ) return true;
        }
        MLE();
        return false;
    }

    boolean OK(){
        int[] newCol = new int[k+1];
        boolean[] curUsedCol = new boolean[k+1];
        for( int i = 0; i < n; ++i )
        for( int j = 0; j < m; ++j ){
            curUsedCol[grid[i][j]] = true;
            if( 0 < org[i][j] ){
                if( newCol[org[i][j]] == 0 )
                    newCol[org[i][j]] = grid[i][j];
                else if( newCol[org[i][j]] != grid[i][j] )
                    return false;
            }
        }
        for( int a =   1; a <= k; ++a )
        for( int b = a+1; b <= k; ++b )
            if( 0<newCol[a] && 0<newCol[b] && newCol[a]==newCol[b] )
                return false;
        for( int c = 1; c <= k; ++c )
            if( orgCol[c] && !curUsedCol[c] )
                return false;
        return true;
    }

    int[] cntUsed;
    void f( int i, int j ){
        if( j == m ){
            j = 0;
            ++i;
        }


        if( i == n ){
            if( OK() )
            {
                ++cntIter;

                int cntUsedCol = 0;
                for( int c = 1; c <= k; ++c )
                    if( 0 < cntUsed[c] )
                        ++cntUsedCol;
                ans += fact[cntUsedCol - cntOrgCol];
                ans %= mod;
//                out.println(cntIter);
//                out.println("ans: " + ans);
//                show();
            }
        }
        else{
            for( int c = 1; c <= k; ++c ){
                //
                boolean OK = true;
                if( cntUsed[c]==0 ){
                    for( int pr = 1; pr <= k; ++pr )
                        if( c < pr && 0 < cntUsed[pr] )
                            OK = false;
                }
                if( !OK ) continue;
                /**/
                if( haveSame(i,j,c) ) continue;

                ++cntUsed[c];
                grid[i][j] = c;

                f( i, j+1 );

                --cntUsed[c];
                grid[i][j] = 0;
            }
        }
    }


    void solve(){
        n = nextInt();
        m = nextInt();
        k = nextInt();
        if( k < m+n-1 ){
            out.println(0);
            return;
        }

        orgCol = new boolean[k+1];
        org = new int[n][m];
        for( int i = 0; i < n; ++i )
        for( int j = 0; j < m; ++j )
            orgCol[ org[i][j] = nextInt() ] = true;
        cntOrgCol = 0;
         for( int c = 1; c <= k; ++c )
             if( orgCol[c] )
                 ++cntOrgCol;

        grid = new int[n][m];
        cntUsed = new int[k+1];
        f( 0, 0 );
//        out.println( "cntIter: " + cntIter );
        out.println( ans );
    }
