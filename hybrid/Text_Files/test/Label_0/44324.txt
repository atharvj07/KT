(n, k) = map(int, input().split())
n2 = n * n
shift = n - k + 1
kth_column = [n2 + 1 - i * shift for i in range(1, n + 1)]
table = [[0] * n for i in range(n)]
left_part_cell = 1
for i in range(n):
	for j in range(n):
		if j < k - 1:
			table[i][j] = left_part_cell
			left_part_cell += 1
		elif j == k - 1:
			table[i][j] = kth_column[i]
		else:
			table[i][j] = table[i][j - 1] + 1
print(sum(kth_column))
[print(' '.join(map(str, row))) for row in table]
