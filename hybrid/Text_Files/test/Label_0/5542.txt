#include <bits/stdc++.h>
using namespace std;
long long int n, ar[100000], i, lp = 2, cnt = 1, x, y, ans;
priority_queue<int> pq;
string s;
int main() {
  cin >> n >> s;
  for (i = 0; i < n; i++) {
    if (s[i] == lp)
      cnt++;
    else {
      ans++;
      pq.push(cnt);
      cnt = 1;
    }
    lp = s[i];
  }
  pq.push(cnt);
  if (pq.size() > 1) {
    x = pq.top();
    pq.pop();
    if (x > 2) {
      cout << ans + 2;
      return 0;
    }
    if (x == 1) {
      cout << ans;
      return 0;
    }
    if (pq.top() == 2) {
      cout << ans + 2;
      return 0;
    }
    cout << ans + 1;
  } else {
    if (pq.top() == 1)
      cout << ans;
    else if (pq.top() == 2)
      cout << ans + 1;
    else
      cout << ans + 2;
  }
}
