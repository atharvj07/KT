import java.io.*;
import java.util.*;


public class E894{
	
	int n;
	int N;
	int componentCount;
	int[] stack;
	boolean[] stacked;
	int[] low;
	int[] comp;
	int[] d;
	int[] p;
	int[] idx;
	int adj[][];
	
	int[] from,to,size;
	long[] w;
	
	long[] componentMushrooms;
	
	int[][] graph;
	int[] inDegree;

	void dfs(){
		N = 1;
		stack = new int[n];
		SP = 0;
		int n = adj.length;
		stacked = new boolean[n];
		d = new int[n];
		idx = new int[n];
		p = new int[n];
		low = new int[n];
		comp = new int[n];
		componentCount = 0;
		
		for(int i = 0; i < n; i++) if(d[i] == 0) visit(i);
	}
	
	int SP;

	void visit(int s){
		int u = s;
		stack[SP++] = u;
		stacked[u] = true;
		d[u] = N++;
		low[u] = d[u];
		while(true){
			while(idx[u] < size[u]){
				int e = adj[u][idx[u]++];
				int v = to[e];
				if(d[v] == 0){
					p[v] = u;
					u = v;
					stack[SP++] = u;
					stacked[u] = true;
					d[u] = N++;
					low[u] = d[u];
				}else if(stacked[v]) low[u] = Math.min(low[u], d[v]);
			}
			if(low[u] == d[u]) {
				while(true){
					int v = stack[--SP];
					comp[v] = componentCount;
					stacked[v] = false;
					if(v == u) break;
				}
				componentCount++;
			}			
			if(u == s && idx[s] == size[s]) break;
			
			int v = u;
			u = p[u];
			low[u] = Math.min(low[u], low[v]);
		}
	}
	
	
	
	
	long totalMushrooms(long n){
		long left = 0;
		long right = n;
		while(left < right-1){
			long mid = (left+right)/2;
			if((mid*(mid+1))/2 <= n) left = mid;
			else right = mid;
		}
		return (left+1)*n - (left*(left+1)*(left+2))/6;
	}
	
	long totalMush(long n){
		long k = (long)Math.floor((Math.sqrt(1+8*n)-1)/2);
		return (k+1)*n - (k*(k+1)*(k+2))/6;
	}

	private void solve() throws Exception {
		n = nextInt();
		int m = nextInt();
	
		adj = new int[n][];
		from = new int[m];
		to = new int[m];
		size = new int[n];
		w = new long[m];
		for(int i = 0; i < m; i++) {
			from[i] = nextInt()-1;
			to[i] = nextInt()-1;
			w[i] = nextInt();
			size[from[i]]++;
		}
		
		for(int i = 0; i < n; i++) {
			adj[i] = new int[size[i]];
			size[i] = 0;
		}
		for(int i = 0; i < m; i++) adj[from[i]][size[from[i]]++] = i;
		
		int s = nextInt()-1;
		dfs();
		
		componentMushrooms = new long[componentCount];
		graph = new int[componentCount][];
		inDegree = new int[componentCount];

		
		Arrays.fill(size, 0);
		
		for(int e = 0; e < m; e++){
			int u = from[e];
			int v = to[e];
			if(comp[v] == comp[u]) componentMushrooms[comp[u]] += totalMush(w[e]);
			else{
				size[comp[u]]++;
				inDegree[comp[v]]++;
			}
		}
		
		for(int i = 0; i < componentCount; i++){
			graph[i] = new int[size[i]];
			size[i] = 0;
		}
		
		for(int e = 0; e < m; e++){
			int u = from[e];
			int v = to[e];
			if(comp[v] != comp[u]) graph[comp[u]][size[comp[u]]++] = e;
		}
		out.println(longestPath(comp[s]));
	}
	
	
	
	long longestPath(int s){
		long[] pd = new long[componentCount];
		Arrays.fill(pd, -(1L<<60));
		pd[s] = componentMushrooms[s];
		LinkedList<Integer> queue = new LinkedList<Integer>();		
		for(int i = 0; i < componentCount; i++)	if(inDegree[i] == 0) queue.addLast(i);		
		long res = pd[s];
		while(!queue.isEmpty()){
			int u = queue.removeFirst();
			res = Math.max(res, pd[u]);
			for(int e : graph[u]){
				int v = comp[to[e]];
				if(pd[u] >= 0) pd[v] = Math.max(pd[v], pd[u] + w[e] + componentMushrooms[v]);
				if(--inDegree[v] == 0) queue.addLast(v);
			}
		}
		return res;
	}

	public static void main(String[] args) {
		(new E894()).run();
	}

	private BufferedReader in;
	private PrintWriter out;
	private StringTokenizer tokenizer;

	public void run() {
		try {
			in = new BufferedReader(new InputStreamReader(System.in));
			tokenizer = null;
			out = new PrintWriter(System.out);
			solve();
			in.close();
			out.close();
		} catch (Exception e) {
			e.printStackTrace();
			System.exit(1);
		}
	}

	private int nextInt() throws IOException {
		return Integer.parseInt(nextToken());
	}

	private long nextLong() throws IOException {
		return Long.parseLong(nextToken());
	}

	private float nextFloat() throws IOException {
		return Float.parseFloat(nextToken());
	}

	private String nextLine() throws IOException {
		return new String(in.readLine());
	}

	private String nextToken() throws IOException {
		while (tokenizer == null || !tokenizer.hasMoreTokens()) {
			tokenizer = new StringTokenizer(in.readLine());
		}
		return tokenizer.nextToken();
	}
	
	class Edge{
		int v;
		long w;
		
		public Edge(int v, long w){
			this.v = v;
			this.w = w;
		}
	}


}
