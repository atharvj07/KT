t = int(input())
for T in range(t):
	n = int(input())
	s1 = input()
	s2 = input()
	i = 0
	ans = 0
	while i < n:
		if s1[i] == '0' and s2[i] == '1':
			ans += 2
			i += 1
		elif s1[i] == '1' and s2[i] == '0':
			ans += 2
			i += 1
		elif s1[i] == '0' and s2[i] == '0':
			if i + 1 < n and s1[i + 1] == '1' and (s2[i + 1] == '1'):
				ans += 2
				i += 2
			else:
				ans += 1
				i += 1
		else:
			temp = 0
			for j in range(i + 1, n, 1):
				if s1[i] == '1' and (s1[j] == '0' or s2[j] == '0'):
					ans += 2
					temp = j
					break
				else:
					break
			if temp != 0:
				i = temp + 1
			else:
				i += 1
	print(ans)
