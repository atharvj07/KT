(x1, y1) = map(int, input().split())
(x2, y2) = map(int, input().split())
n = int(input())
s = input()
px = [0]
py = [0]
x2 -= x1
y2 -= y1
for c in s:
	if c == 'L':
		px.append(px[-1] - 1)
	elif c == 'R':
		px.append(px[-1] + 1)
	else:
		px.append(px[-1])
	if c == 'D':
		py.append(py[-1] - 1)
	elif c == 'U':
		py.append(py[-1] + 1)
	else:
		py.append(py[-1])

def get_pos_by_wind(day):
	(full, partial) = divmod(day, n)
	x = px[-1] * full + px[partial]
	y = py[-1] * full + py[partial]
	return (x, y)

def get_d_to_pos(day):
	(x, y) = get_pos_by_wind(day)
	return abs(x - x2) + abs(y - y2)

def solve():
	day = 1
	for i in range(100):
		v = get_d_to_pos(day)
		if v <= day:
			break
		day *= 2
	else:
		return -1
	l = 1
	r = day * 2
	mid = (l + r) // 2
	va = r
	for i in range(day.bit_length() + 2):
		v = get_d_to_pos(mid)
		if v <= mid:
			va = mid
			r = mid
		else:
			l = mid
		mid = (l + r + 1) // 2
	return va
print(solve())
