#include <bits/stdc++.h>
using namespace std;

struct dice{
  int x[6];
  void roll_x(){roll(0,4,5,1);}
  void roll_y(){roll(0,2,5,3);}
  void roll_z(){roll(1,3,4,2);}

  vector<dice> all_rolls(){
    vector<dice> res;
    for(int k=0; k<6; (k&1 ? roll_y() : roll_x()), k++){
      for(int i=0; i<4; roll_z(),i++){
        res.push_back(*this);
      }
    }
    return res;
  }

  void roll(int a,int b,int c,int d){
    int tmp = x[a];
    x[a] = x[b];
    x[b] = x[c];
    x[c] = x[d];
    x[d] = tmp;
  }
};

vector<dice> a[4];
int n,ans;

void solve(vector<int> z) {
  if(z.size()==n) {
    int sum=0;
    dice v[n];
    for(int i=0; i<n; i++) v[i]=a[i][z[i]];
    for(int i=0; i<6; i++) {
      int c[30];
      memset(c,0,sizeof(c));
      for(int j=0; j<n; j++) c[v[j].x[i]]++;
      int ma=0;
      for(int i=0; i<30; i++) ma=max(ma,c[i]);
      sum+=n-ma;
    }
    ans=min(ans,sum);
    return;
  }
  for(int i=0; i<a[z.size()].size(); i++) {
    vector<int> y=z;
    y.push_back(i);
    solve(y);
  }
}

int main() {
  while(cin >> n && n) {
    for(int i=0; i<n; i++) a[i].clear();
    ans=1000000007;

    map<string, int> m;m.clear();
    for(int i=0; i<n; i++) {
      dice b;
      for(int j=0; j<6; j++) {
	string s;
	cin >> s;
	if(!m[s]) {
	  int q=m.size()+1;
	  m[s]=q;
	}
	b.x[j]=m[s];
      }
      a[i]=b.all_rolls();
    }
    vector<int> y;
    solve(y);
    cout << ans << endl;
  }
  return 0;
}
