import sys
import math
inf = float('inf')

def lcm(a, b):
	return int(a / math.gcd(a, b) * b)
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	perm = [0] * k
	diff = n - k + 1
	if diff == 0:
		ans = [i for i in range(1, k + 1)]
		print(*ans)
	else:
		z = k
		for i in range(k - diff, k):
			perm[i] = z
			z -= 1
		for i in range(k - diff):
			perm[i] = i + 1
		print(*perm)
