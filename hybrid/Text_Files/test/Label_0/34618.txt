    static BigDecimal k(int l, int u) {
        BigDecimal ret = new BigDecimal(l);
        for (int i = l + 1; i < u + 1; i++) {
            ret = ret.multiply(new BigDecimal(i));
        }
        return ret;
    }

    // int c(int a,int b){
    // return k(a-b+1,a)/k(1,b);
    // }

    static BigDecimal c(int a, int b) {
        return k(a - b + 1, a).divide(k(1, b));
    }
