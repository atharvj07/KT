#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  int valores[101];
  int resultado[101];
  int qtd = 0;
  int valor = 1;
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    scanf("%d", &valores[i]);
  }
  for (int i = 0; i < n; i++) {
    if (valores[i] == valor) {
      resultado[qtd] = 2000 + i + 1;
      qtd++;
      valor++;
    }
  }
  if (qtd == 0)
    printf("0");
  else {
    printf("%d\n", qtd);
    for (int i = 0; i < qtd; i++) {
      printf("%d", resultado[i]);
      if (i != n - 1) printf(" ");
    }
  }
  return 0;
}
