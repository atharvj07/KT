#include <bits/stdc++.h>
using namespace std;
const double pi = acos(-1.0);
const double eps = 1e-8;
const int INF = 1E9;
const int MAXN = 100500;
long long n, m, k, ans;
long long a, b, h, w;
int main() {
  cin >> n >> m >> k;
  if ((n - 1) + (m - 1) < k) {
    cout << -1;
    return 0;
  }
  ans = 0;
  a = max(0ll, k - m + 1);
  for (int i = 0; i < (int)(1111); i++) {
    b = k - a;
    if (a < 0 || a >= n || b < 0 || b >= m) continue;
    h = n / (a + 1);
    w = m / (b + 1);
    ans = max(ans, h * w);
    a++;
  }
  b = max(0ll, k - n + 1);
  for (int i = 0; i < (int)(1111); i++) {
    a = k - b;
    if (a < 0 || a >= n || b < 0 || b >= m) continue;
    h = n / (a + 1);
    w = m / (b + 1);
    ans = max(ans, h * w);
    b++;
  }
  cout << ans;
  return 0;
}
