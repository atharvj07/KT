#include <bits/stdc++.h>
using namespace std;
int Set(int N, int pos) { return N = N | (1 << pos); }
int reset(int N, int pos) { return N = N & ~(1 << pos); }
bool check(int N, int pos) { return (bool)(N & (1 << pos)); }
bool Notprime[4000005];
int main() {
  Notprime[1] = true;
  for (int i = 2; (i * i) < 4000005; i++) {
    if (Notprime[i] == true) continue;
    for (int j = i * 2; j < 4000005; j += i) Notprime[j] = true;
  }
  int n;
  scanf("%d", &n);
  int a[n + 1];
  int on = 0;
  for (int i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
  }
  vector<int> ans;
  bool take = true;
  for (int i = 1; i <= n; i++) {
    if (a[i] == 1)
      ans.push_back(1);
    else if (take && Notprime[a[i] + 1] == false) {
      ans.push_back(a[i]);
      take = false;
    }
  }
  if (ans.size() == 0) ans.push_back(a[1]);
  if (ans.size() < 2) {
    for (int i = 1; i <= n; i++) {
      for (int j = i + 1; j <= n; j++) {
        if (Notprime[a[i] + a[j]] == false) {
          ans.clear();
          ans.push_back(a[i]);
          ans.push_back(a[j]);
          n = -1;
          break;
        }
      }
    }
  }
  cout << ans.size() << endl;
  for (int i : ans) cout << i << " ";
  return 0;
}
