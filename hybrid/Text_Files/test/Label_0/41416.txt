t = int(input())

def is_sorted(a, i, j):
	if i == j or i + 1 == j:
		return True
	if a[i] > a[i + 1]:
		return False
	for ii in range(i + 1, j):
		if a[ii - 1] > a[ii]:
			return False
	return True
while t:
	(n, x) = list(map(int, input().split()))
	arr = list(map(int, input().split()))
	if is_sorted(arr, 0, len(arr)):
		print(0)
	else:
		ans = 0
		for i in range(len(arr)):
			if arr[i] > x:
				tmp = arr[i]
				arr[i] = x
				x = tmp
				ans += 1
			if is_sorted(arr, i, len(arr)):
				break
		if is_sorted(arr, 0, len(arr)):
			print(ans)
		else:
			print(-1)
	t -= 1
