from collections import defaultdict
import sys
input = sys.stdin.readline
get_int = lambda : int(input().rstrip())
get_arr = lambda : [int(w) for w in input().split()]
t = get_int()

class DSU:

	def __init__(self, size, arr):
		self.par = list(range(size))
		self.hs = [0] * size
		self.node2min = [arr[i][0] for i in range(size)]

	def find(self, i):
		if self.par[i] != i:
			self.par[i] = self.find(self.par[i])
		return self.par[i]

	def union(self, i, j):
		(pi, pj) = (self.find(i), self.find(j))
		if pi == pj:
			return
		if self.hs[pi] < self.hs[pj]:
			(pi, pj) = (pj, pi)
		self.par[pj] = pi
		self.node2min[pi] = min(self.node2min[pi], self.node2min[pj])
		self.hs[pi] += 1 if self.hs[pi] == self.hs[pj] else 0

def can(t, times):
	return times[len(times) - (t + 2)] <= t
for _ in range(t):
	input()
	(n, k) = get_arr()
	arr = []
	x2i = defaultdict(list)
	y2i = defaultdict(list)
	for i in range(n):
		(x, y, s) = get_arr()
		arr.append((s, x, y))
		x2i[x].append((y, i))
		y2i[y].append((x, i))
	dsu = DSU(n, arr)
	for x in x2i.keys():
		x2i[x].sort()
		for j in range(len(x2i[x]) - 1):
			(y1, i1) = x2i[x][j]
			(y2, i2) = x2i[x][j + 1]
			if y2 - y1 <= k:
				dsu.union(i1, i2)
	for y in y2i.keys():
		y2i[y].sort()
		for j in range(len(y2i[y]) - 1):
			(x1, i1) = y2i[y][j]
			(x2, i2) = y2i[y][j + 1]
			if x2 - x1 <= k:
				dsu.union(i1, i2)
	comps = {dsu.find(i) for i in range(n)}
	times = sorted([dsu.node2min[c] for c in comps])
	l = -1
	r = len(times) - 1
	while r - l >= 2:
		m = (l + r) // 2
		if can(m, times):
			r = m
		else:
			l = m
	print(r)
