import java.util.*;

public class Replacement {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        int a[] = new int[n];
        for (int k = 0; k < n; k++) {
            a[k] = cin.nextInt();
        }
        int max = a[0];
        int index = 0;
        for (int i = 1; i < n; i++) {
            if (a[i] > max) {
                max = a[i];
                index = i;
            }
        }
        if (a[index] != 1) {
            a[index] = 1;
        } else {
            a[index] = 2;
        }
        Arrays.sort(a);
        for (int i = 0; i < n; i++) {
            System.out.print(a[i] + " ");
        }
        cin.close();
    }
}
