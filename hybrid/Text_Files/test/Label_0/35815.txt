import java.math.BigInteger;
import java.util.Scanner;

public class Quiz {
	static int x = 1000000009;
	static long ModExp(long base,long power) 
	{
		if (power == 0) 
			return 1;
		if (power == 1) 
			return base%x;

		if (power%2==1) {
			return (base*ModExp(base, power-1))%x;
		} 
		else 
		{
			long intermediate = ModExp(base, power/2)%x;
			return (intermediate*intermediate)%x;
		}
	}
	public static void main(String args[])
	{
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int m = sc.nextInt();
		int k = sc.nextInt();
		int rightSeq = k-1;
		int a = Math.min(m/rightSeq, n-m);
		int score = Math.min(m,a*(k-1));
		m -= score;
		if (m%k!=0)
		{
			score += m%k;
		}
		System.out.println((x+(ModExp(2, m/k+1)-2)*k+score)%x);
	}
}
