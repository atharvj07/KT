n = int(input())
li = list(map(int, input().split()))
d = {}
c = 0
c4 = 0
for i in li:
	if i not in d:
		d[i] = 1
	else:
		d[i] += 1
	if d[i] % 2 == 0:
		c += 1
	if d[i] % 4 == 0:
		c4 += 1
q = int(input())
for i in range(q):
	(a, b) = map(str, input().split())
	b = int(b)
	if a == '+':
		if b not in d:
			d[b] = 0
		if d[b] % 2 != 0:
			c += 1
		if d[b] % 4 == 3:
			c4 += 1
		if b not in d:
			d[b] = 1
		else:
			d[b] += 1
	else:
		if d[b] % 2 == 0:
			c -= 1
		if d[b] % 4 == 0:
			c4 -= 1
		d[b] -= 1
	if c4 > 1 or (c >= 4 and c4 == 1):
		print('YES')
	else:
		print('NO')
