n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))
ans = 0
if sum(a) != sum(b):
	print(-1)
else:
	sum_i = a[0]
	sum_j = b[0]
	i = 0
	j = 0
	while i < n and j < m:
		while sum_i < sum_j and i + 1 < n:
			i += 1
			sum_i += a[i]
		while sum_i > sum_j and j + 1 < m:
			j += 1
			sum_j += b[j]
		if sum_i == sum_j:
			ans += 1
			if i + 1 < n and j + 1 < m:
				i += 1
				j += 1
				sum_i += a[i]
				sum_j += b[j]
			else:
				break
	print(ans)
