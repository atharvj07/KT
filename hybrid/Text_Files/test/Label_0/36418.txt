import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.math.BigInteger;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.PriorityQueue;
import java.util.Scanner;
import java.util.StringTokenizer;

public class d {
	public static void main(String[] args) throws IOException {
		FastScanner in = new FastScanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		int k = in.nextInt(), n = in.nextInt(), m = in.nextInt();
		long[] vs = new long[k];
		boolean[] usedSet = new boolean[k];
		
		for(int i=0;i<k;i++) {
			vs[i] = in.nextInt();
		}
		ArrayList<Improvement> ans = new ArrayList<>();
		ArrayList<Improvement> choices = new ArrayList<>();
		for(int i=0;i<n;i++) {
			int tj = in.nextInt(), ij = in.nextInt()-1, bj = in.nextInt();
			if(tj == 1) {
				bj -= vs[ij];
			}
			choices.add(new Improvement(tj, ij, bj, i));
		}
		Collections.sort(choices);
		for(Improvement i : choices) {
			if(i.t == 1) {
				if(usedSet[i.i]) {
					i.scale = new Fraction(0, 1);
					continue;
				}
				usedSet[i.i] = true;
			}
			if(i.t == 1 || i.t == 2) {
				i.scale = new Fraction(vs[i.i]+i.b, vs[i.i]);
				vs[i.i] += i.b;
			} else {
				i.scale = new Fraction(i.b, 1);
			}
		}
		Collections.sort(choices, (a,b) -> {
			return -a.scale.compareTo(b.scale);
		});
		for(int i=0;i<m;i++) {
			if(choices.get(i).scale.compareTo(Fraction.ONE) < 0) break;
			ans.add(choices.get(i));
		}
		Collections.sort(ans, (a,b) -> {
			return Integer.compare(a.t, b.t);
		});
		out.println(ans.size());
		for(int i=0;i<ans.size();i++) {
			if(i != 0)
				out.print(' ');
			out.print(ans.get(i).id+1);
		}
		out.println();
		out.close();
	}
	static class Improvement implements Comparable<Improvement>{
		int t, i, b;
		int id;
		Fraction scale;
		public Improvement(int t, int i, int b, int id) {
			this.t = t;
			this.i = i;
			this.b = b;
			this.id = id;
		}
		@Override
		public int compareTo(Improvement o) {
			return -Integer.compare(b, o.b);
		}
	}
	static class Fraction implements Comparable<Fraction> {
		public BigInteger n, d;
		public static Fraction ONE = new Fraction(1, 1);
		public Fraction(long nn, long dd) {
			this(BigInteger.valueOf(nn), BigInteger.valueOf(dd));
		}
		public Fraction(BigInteger nn, BigInteger dd) {
			n = nn;
			d = dd;
			if(d.compareTo(BigInteger.ZERO) < 0) {
				n = n.negate();
				d = d.negate();
			}
			if(n.compareTo(BigInteger.ZERO) == 0) {
				d = BigInteger.ONE;
			}
		}
		public Fraction sub(Fraction f) {
			return new Fraction(f.d.multiply(n).subtract(d.multiply(f.n)), d.multiply(f.d));
		}
		public int compareTo(Fraction f) {
			Fraction a = sub(f);
			return a.n.compareTo(BigInteger.ZERO);
		}
	}
	static class FastScanner {
	    BufferedReader br;
	    StringTokenizer st;
		
	    public FastScanner(InputStream i) {
	        br = new BufferedReader(new InputStreamReader(i));
	        st = new StringTokenizer("");
	    }
				
	    public String next() throws IOException {
	        if(st.hasMoreTokens())
	            return st.nextToken();
	        else
	            st = new StringTokenizer(br.readLine());
	        return next();
	    }

	    public int nextInt() throws IOException {
	        return Integer.parseInt(next());
	    }
	    public long nextLong() throws IOException {
	        return Long.parseLong(next());
	    }
	    public double nextDouble() throws IOException {
	        return Double.parseDouble(next());
	    }
	}
}
