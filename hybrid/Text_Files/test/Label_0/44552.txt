(n, x, y) = map(int, input().split())
num = list(input())
l = len(num) - 1
x10 = 10 ** x
y10 = 10 ** y
count = 0
ans = 1
while ans % x10 != y10:
	ans = ans * 10
ans_str = list('0' * (x - y - 1) + str(ans))
smalll = len(ans_str) - 1
while smalll > -1:
	if ans_str[smalll] != num[l]:
		count = count + 1
	smalll = smalll - 1
	l = l - 1
print(count)
