#include <bits/stdc++.h>
using namespace std;
int main() {
  int la, lb, k, m, i, j;
  long long arra[100010], arrb[100010];
  scanf("%d%d%d%d", &la, &lb, &k, &m);
  for (i = 1; i <= la; i++) scanf("%I64d", &arra[i]);
  for (i = 1; i <= lb; i++) scanf("%I64d", &arrb[i]);
  if (arra[k] < arrb[lb - m + 1])
    printf("YES");
  else
    printf("NO");
  return 0;
}
