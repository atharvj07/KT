[n, p, q, r] = list(map(int, input().split()))
a = list(map(int, input().split()))
Inf = int(1000000000.0 + 7)
pref_min = [Inf for i in range(n)]
pref_max = [-Inf for i in range(n)]
suf_min = [Inf for i in range(n)]
suf_max = [-Inf for i in range(n)]
pref_min[0] = a[0]
pref_max[0] = a[0]
for i in range(1, n):
	pref_min[i] = min(pref_min[i - 1], a[i])
	pref_max[i] = max(pref_max[i - 1], a[i])
suf_min[n - 1] = a[n - 1]
suf_max[n - 1] = a[n - 1]
for i in range(n - 2, -1, -1):
	suf_min[i] = min(suf_min[i + 1], a[i])
	suf_max[i] = max(suf_max[i + 1], a[i])
res = -int(3e+18 + 7)
for j in range(n):
	x = q * a[j]
	x += max(p * pref_max[j], p * pref_min[j])
	x += max(r * suf_max[j], r * suf_min[j])
	res = max(res, x)
print(res)
