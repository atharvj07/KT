n = int(input())
s = input()
if s.count('(') != s.count(')'):
	print(-1)
	exit()
ans = 0
ls = [0]
for i in range(n):
	if s[i] == '(':
		ls.append(ls[-1] + 1)
	else:
		ls.append(ls[-1] - 1)
ans = 0
cnt = 0
for i in range(1, n + 1):
	if ls[i] == 0:
		if cnt:
			ans += cnt + 1
		cnt = 0
	elif ls[i] < 0:
		cnt += 1
print(ans)
