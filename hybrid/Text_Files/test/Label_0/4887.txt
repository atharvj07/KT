#include <bits/stdc++.h>
using namespace std;
struct myp {
  int x;
  int y;
};
bool compare(const myp &left, const myp &right) {
  if (left.y > right.y)
    return true;
  else if (left.y == right.y && left.x > right.x)
    return true;
  else
    return false;
}
int arr[1000][1000][3];
vector<myp> arr1;
int main() {
  int n, m, t;
  cin >> n;
  cin >> m;
  cin >> t;
  string str;
  for (int i = 0; i < n; i++) {
    cin >> str;
    for (int j = 0; j < m; j++) {
      for (int k = 0; k < 3; k++) {
        cin >> arr[i][j][k];
      }
    }
  }
  int S = 0, max = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (i != j) {
        int t1 = t;
        for (int k = 0; k < m; k++) {
          myp tmp;
          tmp.x = arr[i][k][2];
          tmp.y = t * arr[j][k][1] - t * arr[i][k][0];
          arr1.insert(arr1.begin() + k, tmp);
        }
        sort(arr1.begin(), arr1.end(), compare);
        bool tr = false;
        for (int k = 0; k < m; k++) {
          if (arr1[k].y > 0) {
            if (arr1[k].x >= t1) {
              arr1[k].x = t1;
              tr = true;
            }
            S += arr1[k].y / t * arr1[k].x;
            t1 -= arr1[k].x;
          }
          if (tr) k = m;
        }
      }
      if (max < S) max = S;
      S = 0;
      arr1.clear();
    }
  }
  cout << max;
  return 0;
}
