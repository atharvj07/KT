a = int(input())
b = input().split()
factors = []
for x in range(1, int(b[0]) + 1):
	if int(b[0]) % x == 0:
		factors.append(str(x))
hc = 1
for y in range(0, len(factors)):
	c = 0
	for z in range(0, a):
		if int(b[z]) % int(factors[y]) == 0:
			c = c + 1
		if c == a:
			hc = int(factors[y])
d = 0
for p in range(0, a):
	if int(b[p]) == int(b[0]):
		d = d + 1
if d == a:
	print(str(int(b[0]) * a))
else:
	print(str(hc * a))
