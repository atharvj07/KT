import sys
from collections import deque
input = sys.stdin.readline

def solve():
	(n, k) = map(int, input().split())
	arr = [0] + list(map(int, input().split()))
	graph = [[] for _ in range(n + 1)]
	ind = [0] * (n + 1)
	val = [0] * (n + 1)
	for _ in range(n - 1):
		(u, v) = map(int, input().split())
		graph[u].append(v)
		graph[v].append(u)
		ind[u] += 1
		ind[v] += 1
	qu = deque()
	V = 0
	for i in range(1, n + 1):
		V ^= arr[i]
		if ind[i] == 1:
			qu.append(i)
	if V == 0:
		return 'YES'
	if k < 3:
		return 'NO'
	nums = 0
	while qu:
		curr = qu.popleft()
		val[curr] ^= arr[curr]
		if val[curr] == V:
			nums += 1
		for nei in graph[curr]:
			ind[nei] -= 1
			if val[curr] != V:
				val[nei] ^= val[curr]
			if ind[nei] == 1:
				qu.append(nei)
	return 'YES' if nums >= 2 else 'NO'
for _ in range(int(input())):
	print(solve())
