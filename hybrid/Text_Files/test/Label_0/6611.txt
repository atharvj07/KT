#include <cstdio>
#include <vector>
using namespace std;
#define pb push_back
#define mp make_pair
#define fs first
#define sc second

const int N = (int) 1e6  +10;
int a[N], n;

bool check(int k) {
	vector< pair<int, int> > s;
	s.pb(mp(0, 0));
	for (int i = 1; i < n; ++i) {
		if (a[i] > a[i - 1]) {
			continue;
		}
		if (k == 1) {
			return false;
		}
		int j = a[i] - 1;
		while (!s.empty() && (s.back().fs > j)) {
			s.pop_back();
		}
		while (!s.empty() && (s.back() == mp(j, k - 1))) {
			s.pop_back();
			--j;
		}
		if (s.empty()) {
			return false;
		}
		if (s.back().fs != j) {
       		s.pb(mp(j, 1));
       		continue;
       	}
       	s.back().sc++;
	}
	return true;
}

int main() {
	scanf("%d", &n);
	for (int i = 0; i < n; ++i) {
		scanf("%d", &a[i]);
	}
	int l = 0, r = n;
	while (l + 1 < r) {
		int q = (l + r) / 2;
		(check(q) ? r : l) = q;
	}
	printf("%d\n", r);
	return 0;
}
