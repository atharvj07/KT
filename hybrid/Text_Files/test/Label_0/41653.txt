def solve():
	n = int(input())
	arr = []
	start = 0
	while True:
		for a in range(1, 10 ** 9):
			arr.append(a)
			for i in range(1, len(arr)):
				if arr[i] % arr[i - 1] == 0:
					start = max(arr)
					arr.clear()
					break
			if len(arr) == n:
				return arr

def main():
	test_cases = int(input())
	for _ in range(test_cases):
		print(' '.join(map(str, solve())))
main()
