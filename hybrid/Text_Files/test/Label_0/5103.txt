#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n;
  cin >> n;
  vector<long long int> v;
  while (n) {
    long long int z = 0, x = n, y = 1;
    while (x) {
      z += (x % 10 ? y : 0);
      x /= 10;
      y *= 10;
    }
    n -= z;
    v.push_back(z);
  }
  cout << v.size() << "\n";
  for (auto a : v) cout << a << " ";
  return 0;
}
