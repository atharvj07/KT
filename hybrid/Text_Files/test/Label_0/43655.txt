for _ in range(int(input())):
	s = input()
	x = int(input())
	w1 = '1' * len(s)
	w = list(w1)
	f = 0
	for i in range(len(s)):
		if s[i] == '0':
			if i - x >= 0 and i + x <= len(s) - 1:
				w[i - x] = '0'
				w[i + x] = '0'
			elif i - x < 0 and i + x <= len(s) - 1:
				w[i + x] = '0'
			elif i - x >= 0 and i + x > len(s) - 1:
				w[i - x] = '0'
	for i in range(len(s)):
		if s[i] == '1':
			if i - x >= 0 and i + x <= len(s) - 1:
				if w[i - x] == '0' and w[i + x] == '0':
					f = 1
			elif i - x < 0 and i + x <= len(s) - 1:
				if w[i + x] == '0':
					f = 1
			elif i - x >= 0 and i + x > len(s) - 1:
				if w[i - x] == '0':
					f = 1
			else:
				f = 1
	if f == 1:
		print(-1)
	else:
		print(''.join(w))
