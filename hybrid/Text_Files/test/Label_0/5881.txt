#include <bits/stdc++.h>
int main() {
  char s[11];
  char ABC[] = {'a', 'b', 'c', 'd', 'e', 'f', 'g', 'h', 'i',
                'j', 'k', 'l', 'm', 'n', 'o', 'p', 'q', 'r',
                's', 't', 'u', 'v', 'w', 'x', 'y', 'z', '\0'};
  scanf("%s", s);
  int i, j, b = 1;
  int v[27] = {0};
  for (i = 0; s[i] != '\0'; i++) {
    for (j = 0; ABC[j] != '\0'; j++) {
      if (s[i] == ABC[j]) {
        v[j]++;
      }
    }
  }
  for (i = 26; i >= 0; i--) {
    if (v[i] > 0) {
      for (j = 0; j < v[i]; j++) {
        printf("%c", ABC[i]);
      }
      i = 0;
    }
  }
  return 0;
}
