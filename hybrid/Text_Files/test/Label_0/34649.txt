import java.io.*;
import java.util.*;

public class B {

	public B () throws IOException {
		int m = sc.nextInt();
		int n = sc.nextInt();
		String [] rows = new String[m];
		for (int i = 0; i < m; ++i)
			rows[i] = sc.next();
		solve(m, n, rows);
	}
	
	Set<String> cache = new HashSet<String>();
	List<String> steps = new ArrayList<String>();
	List<Integer> colors = new ArrayList<Integer>();
	
	public void solve (int h, int n, String [] rows) {
		t = millis();
		
		int w = rows[0].length();
		int [][] P = new int [h][w];
		
		for (int i = 0; i < h; ++i)
			for (int j = 0; j < w; ++j)
				P[i][j] = rows[i].charAt(j) - '0';
		
		int x = 0, y = 0;
		int [] DP = { 0, 1 }, CP = { -1, 0 };
		boolean R = true;
		int col = -1;
		for (int i = 0; i <= n; ++i) {
			col = P[x][y];
			int [] xy = advance(P, x, y, col, DP, CP); x = xy[0]; y = xy[1];
			int dx = DP[0], dy = DP[1];
			if (x+dx < h && y+dy < w && x+dx >= 0 && y+dy >= 0 && P[x+dx][y+dy] != 0) {
				x += dx;
				y += dy;
			} else {
				if (R) {
					CP = minus(CP);
					R = false;
				} else {
					DP = cw(DP);
					CP = minus(cw(CP));
					R = true;
				}
			}
			String key = Arrays.toString(new int [] { x,y, CP[0], CP[1], DP[0], DP[1] });
			if (cache.contains(key)) {
				for (int j = 0; j < steps.size(); ++j)
					if (steps.get(j).equals(key)) {
						int period = i - j;
						//System.err.println("CACHE! PERIOD IS " + period);
						print(colors.get(j + ((n-j)%period)));
					}
			} else {
				cache.add(key);
				steps.add(key);
				colors.add(col);
			}
		}
		print(col);
	}
	
	int [] minus (int [] X) {
		return new int [] { -X[0], -X[1] };
	}
	
	int [] cw (int [] X) {
		int [][] W = { {1,0}, {0,-1}, {-1,0}, {0,1} };
		int x = X[0], y = X[1];
		for (int i = 0; i < 4; ++i)
			if (x == W[i][0] && y == W[i][1]) {
				int j = (i+1) % 4;
				return new int [] { W[j][0], W[j][1] };
			}
		throw new Error ("NO!");
	}
	
	int [] advance (int [][] P, int x, int y, int col, int [] DP, int [] CP) {
		int h = P.length, w = P[0].length;
		
		int dx = DP[0], dy = DP[1];
		while (x+dx < h && y+dy < w && x+dx >= 0 && y+dy >= 0 && P[x+dx][y+dy] == col) {
			x += dx;
			y += dy;
		}
		
		dx = CP[0]; dy = CP[1];
		while (x+dx < h && y+dy < w && x+dx >= 0 && y+dy >= 0 && P[x+dx][y+dy] == col) {
			x += dx;
			y += dy;
		}		
		
		return new int [] { x, y };
	}
	
	////////////////////////////////////////////////////////////////////////////////////

	static Scanner sc;
	static long t;
	
	static void print2 (Object o) {
		System.out.println(o);
	}
	
	static void print (Object o) {
		print2(o);
		//print2((millis() - t) / 1000.0);
		System.exit(0);
	}
	
	static void run () throws IOException {
		sc = new Scanner(System.in);
		new B();
	}
	
	public static void main(String[] args) throws IOException {
		run();
	}
	
	static long millis() {
		return System.currentTimeMillis();
	}
}
