long long ceill(long long x, long long y) {
  if (x % y == 0) {
    return x / y;
  }
  return (x / y) + 1;
}
long long prime(long long n) {
  vector<long long> v;
  while (n % 2 == 0) {
    v.push_back(2);
    n /= 2;
  }
  for (long long i = 3; i < sqrt(n); i += 1) {
    while (n % i == 0) {
      v.push_back(i);
      n /= i;
    }
  }
  if (n > 2) {
    v.push_back(n);
  }
  return v.size();
}
int min_sum(int n) {
  if (n == 0)
    return 0;
  else if (n == 1)
    return 2;
  return (n + 1 + min_sum(n / 2) + min_sum(n - n / 2 - 1));
}
void printDivisors(long long n) {
  vector<long long> v;
  for (long long i = 1; i <= sqrt(n); i++) {
    if (n % i == 0) {
      if (n / i == i) {
        v.push_back(i);
      } else {
        v.push_back(i);
        v.push_back(n / i);
      }
    }
  }
  cout << v.size() << "\n";
}
bool numis(string s) {
  for (int i = 0; i < s.length(); i++)
    if (isdigit(s[i]) == false) return false;
  return true;
}
int fact(int a[], int n) {
  if (a[0] == a[1] && a[0] != a[2]) {
    return 2;
  }
  if (a[0] != a[1] && a[1] == a[2]) {
    return 0;
  }
  if (a[0] == a[2] && a[1] != a[0]) {
    return 1;
  }
  for (long long i = 3; i < n; i += 1) {
    if (a[i] != a[i - 1]) {
      return i;
    }
  }
}
long long b_Search(long long *a, long long l, long long r, long long x) {
  if (l > r) {
    return -1;
  }
  int m = ((l + r) / 2);
  if (x >= a[m] && x < a[m + 1]) {
    return m;
  }
  if (x > a[m]) {
    return b_Search(a, m + 1, r, x);
  }
  return b_Search(a, l, m - 1, x);
}
int sub(string s, string s2) {
  int l1 = s.size();
  int l2 = s2.size();
  int l = l1 - l2;
  for (int i = 0; i <= l; i++) {
    int j;
    for (j = 0; j < l2; j++) {
      if (s[i + j] != s2[j]) {
        break;
      }
    }
    if (j == l2) {
      s.erase(i, 1);
      return sub(s, s2);
    }
  }
  return s.size();
}
pair<string, int> findEntryWithLargestValue(map<string, int> m) {
  pair<string, int> mm = make_pair("", 0);
  map<string, int>::iterator itr;
  for (itr = m.begin(); itr != m.end(); ++itr) {
    if (itr->second > mm.second) {
      mm = make_pair(itr->first, itr->second);
    }
  }
  return mm;
}
