#include <bits/stdc++.h>
using namespace std;
const long long MOD = 998244353;
long long mod_exp(long long x, long long y, long long mod) {
  long long res = 1;
  while (y > 0) {
    if (y & 1) res = res * x % mod;
    y >>= 1;
    x = x * x % mod;
  }
  return res;
}
long long choose2(long long k) { return (k * (k - 1) / 2) % MOD; }
long long contribute(vector<pair<int, int> >& multi) {
  int m = multi.size();
  long long card = 0;
  for (int i = 0; i < m; i++) {
    card += multi[i].second;
  }
  if (card <= 1) return 0;
  long long subsets2 = mod_exp(2, card - 2, MOD);
  long long subsets3;
  if (card >= 3) subsets3 = mod_exp(2, card - 3, MOD);
  long long ans = 0;
  long long prefix = 0;
  for (int i = 0; i < m; i++) {
    long long curr = multi[i].first;
    long long numpairs = choose2(multi[i].second);
    long long fixedcta = (2 * subsets2) % MOD;
    if (card >= 3) {
      long long inc = (2 * (card - 2)) % MOD;
      inc = (inc * subsets3) % MOD;
      fixedcta = (fixedcta + inc) % MOD;
    }
    long long inc = (numpairs * fixedcta) % MOD;
    inc = (inc * curr) % MOD;
    inc = (inc * curr) % MOD;
    ans = (ans + inc) % MOD;
    numpairs = multi[i].second;
    long long fixedct = (((card - 1) % MOD) * subsets2) % MOD;
    inc = (numpairs * fixedct) % MOD;
    inc = (inc * curr) % MOD;
    inc = (inc * curr) % MOD;
    ans = (ans + inc) % MOD;
    inc = (prefix * numpairs) % MOD;
    inc = (inc * curr) % MOD;
    inc = (inc * fixedcta) % MOD;
    ans = (ans + inc) % MOD;
    prefix = (prefix + 1ll * multi[i].first * multi[i].second) % MOD;
  }
  return ans;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int fq[(int)1e5 + 69];
  int xtra[(int)1e5 + 69];
  memset(fq, 0, sizeof fq);
  memset(xtra, 0, sizeof xtra);
  int m;
  cin >> m;
  vector<pair<int, int> > multi(m);
  for (int i = 0; i < m; i++) {
    cin >> multi[i].first >> multi[i].second;
    fq[multi[i].first] = multi[i].second;
  }
  long long ans = contribute(multi);
  long long totOvercount = 0;
  for (int i = 1e5; i >= 2; i--) {
    long long sub = 0;
    vector<pair<int, int> > tmp;
    for (int j = i; j <= 1e5; j += i) {
      sub = (sub + xtra[j]) % MOD;
      if (fq[j] > 0) tmp.push_back(make_pair(j, fq[j]));
    }
    long long overcount = contribute(tmp);
    overcount = (overcount - sub) % MOD;
    overcount = (overcount + MOD) % MOD;
    xtra[i] = overcount;
    totOvercount = (totOvercount + overcount) % MOD;
  }
  ans = (ans - totOvercount) % MOD;
  ans = (ans + MOD) % MOD;
  cout << ans << '\n';
}
