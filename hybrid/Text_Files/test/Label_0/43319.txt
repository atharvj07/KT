import sys
try:
	sys.stdin = open('input.txt', 'r')
	sys.stdout = open('output.txt', 'w')
except:
	pass
input = sys.stdin.readline
(n, s) = map(int, input().split())
a = list(map(int, input().split()))
(l, r) = (0, n)
new_s = 0
ans = 0
while l < r:
	mid = (l + r + 1) // 2
	new_a = sorted([a[i] + (i + 1) * mid for i in range(n)])
	new_s = sum(new_a[:mid])
	if new_s > s:
		r = mid - 1
	else:
		ans = new_s
		l = mid
print(l, ans)
