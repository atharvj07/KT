n = int(input())
lst = list(map(int, input().split()))
if len(lst) < 3:
	maxlen = lenght = len(lst)
else:
	maxlen = lenght = 2
for i in range(2, len(lst)):
	if lst[i] == lst[i - 1] + lst[i - 2]:
		lenght += 1
		if lenght > maxlen:
			maxlen = lenght
	else:
		if lenght > maxlen:
			maxlen = lenght
		lenght = 2
print(maxlen)
