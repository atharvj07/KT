import java.util.Scanner;
public class gnome
{
    public static void main(String args[])
    {
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        long ans=1;
        for(int a=1;a<=3*n;a++)ans=(ans*3)%1000000007;
        long e=1;
        for(int a=1;a<=n;a++)
        {
            e=(e*7)%1000000007;
        }
        ans-=e;
        System.out.println((ans+1000000007)%1000000007);
    }
}
