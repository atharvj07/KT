from sys import *
(n, k) = map(int, stdin.readline().split())
a = list(map(int, input().split()))
dp1 = [0] * n
dp2 = [0] * n
for i in range(1, n):
	if a[i - 1] + dp1[i - 1] + a[i] < k:
		dp1[i] = k - (a[i - 1] + dp1[i - 1] + a[i])
for i in range(n - 2, -1, -1):
	if a[i + 1] + dp2[i + 1] + a[i] < k:
		dp2[i] = k - (a[i + 1] + dp2[i + 1] + a[i])
if sum(dp1) <= sum(dp2):
	ans = sum(dp1)
	b = [dp1[i] + a[i] for i in range(n)]
else:
	ans = sum(dp2)
	b = [dp2[i] + a[i] for i in range(n)]
print(ans)
print(*b)
