for t in range(int(input())):
	(n, p) = map(int, input().split())
	a = list(map(int, input().split()))
	res = [p] * (n - 1)
	c = a[0]
	for i in range(1, n):
		if a[i] % c == 0:
			res[i - 1] = min(c, res[i - 1])
		else:
			c = a[i]
	c = a[n - 1]
	for i in range(n - 2, -1, -1):
		if a[i] % c == 0:
			res[i] = min(c, res[i])
		else:
			c = a[i]
	print(sum(res))
