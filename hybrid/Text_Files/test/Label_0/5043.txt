#include <bits/stdc++.h>
using namespace std;
mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());
const long long maxn = 150000 * 20, INF = 1e18;
long long expo, n = 1, k, a[maxn], dp[maxn][2][2], pos[maxn], cnt;
bool fan[maxn];
vector<long long> adj[maxn];
map<pair<long long, long long>, long long> p;
void build(long long l, long long r) {
  pair<long long, long long> pt = {l, r};
  long long tmp, mid = (l + r) >> 1;
  tmp = p[pt] = ++cnt;
  if (l == r) {
    pos[tmp] = l;
    return;
  }
  build(l, mid);
  build(mid + 1, r);
  adj[tmp].push_back(p[{l, mid}]);
  adj[tmp].push_back(p[{mid + 1, r}]);
}
void solve(long long u) {
  long long c1 = adj[u][0], c2 = adj[u][1];
  if (!(long long)adj[c1].size()) {
    if (fan[pos[c1]] and fan[pos[c2]])
      dp[u][1][1] = 1;
    else if (fan[pos[c1]] or fan[pos[c2]])
      dp[u][1][0] = dp[u][0][1] = 1;
    else
      dp[u][0][0] = 0;
    return;
  }
  solve(c1);
  solve(c2);
  long long mx1 = -INF, mx2 = -INF, mx3 = -INF;
  for (long long i = 0; i < 2; i++)
    for (long long j = 0; j < 2; j++)
      if (dp[c1][1][i] != -INF and dp[c2][0][j] != -INF)
        mx1 =
            max(mx1, dp[c1][1][i] + dp[c2][0][j] + 1 + 2 * (i == 1 or j == 1));
  for (long long i = 0; i < 2; i++)
    for (long long j = 0; j < 2; j++)
      if (dp[c1][0][i] != -INF and dp[c2][1][j] != -INF)
        mx2 =
            max(mx2, dp[c1][0][i] + dp[c2][1][j] + 1 + 2 * (i == 1 or j == 1));
  for (long long i = 0; i < 2; i++)
    for (long long j = 0; j < 2; j++)
      if ((i != 1 or j != 1) and
          (dp[c1][1][i] != -INF and dp[c2][1][j] != -INF))
        mx3 = max(mx3, dp[c1][1][i] + dp[c2][1][j] + 2 + (i == 1 or j == 1));
  dp[u][1][0] = max({dp[u][1][0], mx1, mx2, mx3});
  long long mx4 = -INF, mx5 = -INF, mx6 = -INF;
  for (long long i = 0; i < 2; i++)
    for (long long j = 0; j < 2; j++)
      if ((i == 1 or j == 1) and
          (dp[c1][1][i] != -INF and dp[c2][0][j] != -INF))
        mx4 = max(mx4, dp[c1][1][i] + dp[c2][0][j] + 3);
  for (long long i = 0; i < 2; i++)
    for (long long j = 0; j < 2; j++)
      if ((i == 1 or j == 1) and
          (dp[c1][0][i] != -INF and dp[c2][1][j] != -INF))
        mx5 = max(mx5, dp[c1][0][i] + dp[c2][1][j] + 3);
  for (long long i = 0; i < 2; i++)
    for (long long j = 0; j < 2; j++)
      if (dp[c1][1][i] != -INF and dp[c2][1][j] != -INF)
        mx6 = max(mx6, dp[c1][1][i] + dp[c2][1][j] + 2 + (i == 1 or j == 1));
  dp[u][1][1] = max({dp[u][1][1], mx4, mx5, mx6});
  long long mx7 = -INF, mx8 = -INF, mx9 = -INF;
  for (long long i = 0; i < 2; i++)
    for (long long j = 0; j < 2; j++)
      if ((!i or !j) and (dp[c1][0][i] != -INF and dp[c2][1][j] != -INF))
        mx7 = max(mx7, dp[c1][0][i] + dp[c2][1][j] + 2 + (i == 1 or j == 1));
  for (long long i = 0; i < 2; i++)
    for (long long j = 0; j < 2; j++)
      if ((!i or !j) and (dp[c1][1][i] != -INF and dp[c2][0][j] != -INF))
        mx8 = max(mx8, dp[c1][1][i] + dp[c2][0][j] + 2 + (i == 1 or j == 1));
  for (long long i = 0; i < 2; i++)
    for (long long j = 0; j < 2; j++)
      if (dp[c1][0][i] != -INF and dp[c2][0][j] != -INF)
        mx9 = max(mx9, dp[c1][0][i] + dp[c2][0][j] + 2 * (i == 1 or j == 1));
  dp[u][0][0] = max({dp[u][1][0], mx7, mx8, mx9});
  long long mx10 = -INF, mx11 = -INF, mx12 = -INF;
  for (long long i = 0; i < 2; i++)
    for (long long j = 0; j < 2; j++)
      if (dp[c1][1][i] != -INF and dp[c2][0][j] != -INF)
        mx10 =
            max(mx10, dp[c1][1][i] + dp[c2][0][j] + 2 + (c1 == 1 or c2 == 1));
  for (long long i = 0; i < 2; i++)
    for (long long j = 0; j < 2; j++)
      if (dp[c1][0][i] != -INF and dp[c2][1][j] != -INF)
        mx11 =
            max(mx11, dp[c1][0][i] + dp[c2][1][j] + 2 + (c1 == 1 or c2 == 1));
  for (long long i = 0; i < 2; i++)
    for (long long j = 0; j < 2; j++)
      if ((i == 1 or j == 1) and
          (dp[c1][0][i] != -INF and dp[c2][0][j] != -INF))
        mx12 = max(mx12, dp[c1][0][i] + dp[c2][0][j] + 2);
  dp[u][0][1] = max({dp[u][0][1], mx10, mx11, mx12});
}
signed main() {
  cin >> expo >> k;
  for (long long i = 1, x; i <= k; i++) cin >> x, fan[x] = 1;
  for (long long i = 1; i <= expo; i++) n <<= 1;
  build(1, n);
  for (long long i = 1; i <= cnt; i++)
    for (long long j = 0; j < 2; j++)
      for (long long t = 0; t < 2; t++) dp[i][j][t] = -INF;
  solve(p[{1, n}]);
  long long ans = 0;
  for (long long i = 0; i < 2; i++)
    for (long long j = 0; j < 2; j++)
      ans = max(ans, dp[p[{1, n}]][i][j] + (i or j));
  cout << ans;
}
