for _ in [0] * int(input()):
	n = int(input())
	arr = list(map(int, input().split()))
	c = 0
	bOne = [0] * (n + 2)
	for j in range(1, n - 2, 1):
		bOne[arr[j - 1]] += 1
		bSec = [0] * (n + 2)
		added = 0
		for l in range(j + 2, n, 1):
			v = arr[l - 1]
			added -= bSec[v] * bOne[v]
			bSec[v] += 1
			added += bSec[v] * bOne[v]
			if arr[j] == arr[l]:
				c += added
	print(c)
