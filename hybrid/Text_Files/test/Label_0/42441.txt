import sys
import math

def read_int():
	return int(input().strip())

def read_int_list():
	return list(map(int, input().strip().split()))

def read_string():
	return input().strip()

def read_string_list(delim=' '):
	return input().strip().split(delim)
n = read_int()
inp = list(map(lambda x: int(x), list(read_string())))
out = [-1 for _ in range(n)]
counter = [0, 0, 0]
for i in inp:
	counter[i] += 1
for i in range(3):
	counter[i] -= n // 3
start = 0
while start < n and counter[2] > 0:
	if inp[start] == 2:
		if counter[0] < 0:
			counter[2] -= 1
			out[start] = 0
			counter[0] += 1
			start += 1
			continue
		if counter[1] < 0:
			counter[2] -= 1
			out[start] = 1
			counter[1] += 1
	start += 1
start = 0
while start < n and counter[1] > 0:
	if inp[start] == 1:
		if counter[0] < 0:
			counter[1] -= 1
			out[start] = 0
			counter[0] += 1
	start += 1
end = n - 1
while end > -1 and counter[0] > 0:
	if inp[end] == 0:
		if counter[2] < 0:
			counter[0] -= 1
			out[end] = 2
			counter[2] += 1
			end -= 1
			continue
		if counter[1] < 0:
			counter[0] -= 1
			out[end] = 1
			counter[1] += 1
	end -= 1
end = n - 1
while end > -1 and counter[1] > 0:
	if inp[end] == 1:
		if counter[2] < 0:
			out[end] = 2
			counter[2] += 1
			counter[1] -= 1
	end -= 1
for i in range(n):
	if out[i] == -1:
		out[i] = inp[i]
print(''.join(list(map(lambda x: str(x), out))))
