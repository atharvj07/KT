import java.io.*;
import java.util.*;
import java.math.*;
import java.text.*;
public class Main
{
public static void main(String[] args)throws java.lang.Exception
{
BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
PrintWriter out=new PrintWriter(System.out);
int n,i,j,res,min,pos,p1,temp;


n=Integer.parseInt(in.readLine());

int[] a=new int[n];
int[][] ans=new int[n][2];


StringTokenizer st1=new StringTokenizer(in.readLine());

for(i=0;i<n;i++)
{
a[i]=Integer.parseInt(st1.nextToken());
}




res=0;

p1=0;

for(i=0;i<(n-1);i++)
{
min=a[i];
pos=i;
for(j=i+1;j<n;j++)
	{
	if(a[j]<min)
		{
		min=a[j];
		pos=j;
		}
	}

if(pos!=i)
	{
	++res;

	temp=a[i];
	a[i]=a[pos];
	a[pos]=temp;

	ans[p1][0]=i;
	ans[p1][1]=pos;

	++p1;

	}
}

out.println(res);

for(i=0;i<p1;i++)
{
out.println(ans[i][0]+" "+ans[i][1]);
}


out.flush();
out.close();
}
}
