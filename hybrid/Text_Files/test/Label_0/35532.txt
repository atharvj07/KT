import java.io.*;
import java.util.*;

public class D {

	class Ramp {
		Ramp (int X, int D, int T, int P) {
			S = X - P;
			Y = X + D;
			TT = P + T;
		}
		public final int S, Y, TT;
	}
	
	public D () throws IOException {
		String [] S = r.readLine().split(" ");
		int N = Integer.parseInt(S[0]);
		int L = Integer.parseInt(S[1]);
		Ramp [] R = new Ramp [N];
		for (int i = 0; i < N; ++i) {
			S = r.readLine().split(" ");
			int X = Integer.parseInt(S[0]);
			int D = Integer.parseInt(S[1]);
			int T = Integer.parseInt(S[2]);
			int P = Integer.parseInt(S[3]);
			R[i] = new Ramp(X,D,T,P);
			
			if (R[i].S < 0 || R[i].TT >= R[i].Y - R[i].S)
				R[i] = null;
		}
		solve(N, L, R);
	}
	
	long M = 1000000001;
	long key(int x, int y) {
		return M * x + y;
	}
	
	int [] dekey (long z) {
		return new int [] { (int)(z/M), (int)(z%M) };
	}
	
	void addEdge(int x, int y, int c) {
		if (!E.containsKey(x))
			E.put(x, new AbstractList<Integer>() {

				int cnt = 0, C = 1;
				int [] L = new int[C];
				
				@Override
				public int size() {
					return cnt;
				}
				
				@Override
				public Integer get(int index) {
					return L[index];
				}
				
				@Override 
				public boolean add(Integer x) {
					if (cnt == C)
						L = Arrays.copyOf(L, ++C);
					L[cnt++] = x;
					return true;
				}
			});
			
		E.get(x).add(y);
		G.put(key(x,y), c);		
	}
	
	public void solve (int N, int L, Ramp[] R) {
		t = millis();

		Set<Integer> points = new HashSet<Integer>();
		for (Ramp r : R)
			if (r != null) {
				points.add(r.S);
				points.add(r.Y);
			}
		points.add(0);
		points.add(L);
		
		Integer [] ps = points.toArray(new Integer[0]);
		Arrays.sort(ps);
		
		for (int i = 1; i < ps.length; ++i) {
			int x = ps[i-1], y = ps[i], d = Math.abs(x-y);
			addEdge(x,y,d);
			addEdge(y,x,d);
		}
		
		Map<Long, Integer> RS = new HashMap<Long, Integer>();
		for (int i = 0; i < R.length; ++i) {
			if (R[i] != null) {
				Ramp r = R[i];
				long k = key(r.S, r.Y);
				if (G.containsKey(k) && G.get(k) <= r.TT)
					continue;
				else {
					addEdge(r.S, r.Y, r.TT);
					RS.put(key(r.S,r.Y), i+1);
				}
			}
		}
		
		Map<Integer, Integer> vis = new HashMap<Integer, Integer>();		
		Queue<Long> Q = new PriorityQueue<Long>(1000000, new EC());
		D.put(0, 0);
		for (int v : E.get(0))
			Q.add(key(0,v));
		vis.put(0, -1);
		
		for (;;) {
			Long e = Q.poll();
			int [] xy = dekey(e);
			int x = xy[0], y = xy[1];
			if (!vis.containsKey(y)) {
				D.put(y, D.get(x) + G.get(e));
				vis.put(y, x);
				if (y == L)
					break;
				for (int v : E.get(y))
					if (!vis.containsKey(v))
						Q.add(key(y,v));
			}
		}		

		int x = L;
		List<Integer> Z = new LinkedList<Integer>();
		while (x != -1) {
			Z.add(0, x);
			x = vis.get(x);
		}

		StringBuffer B = new StringBuffer();
		int cnt = 0;
		while (Z.size () > 1) {
			long k = key(Z.get(0), Z.get(1));
			if (RS.containsKey(k)) {
				++cnt;
				B.append(" " + RS.get(k));
			}
			Z.remove(0);
		}
		
		print2(D.get(L));
		print2(cnt);
		if (cnt > 0)
			print(B.toString().trim());
	}
	
	Map<Integer, Integer> D = new HashMap<Integer, Integer>();
	Map<Integer, List<Integer>> E = new HashMap<Integer, List<Integer>>();	
	Map<Long, Integer> G = new HashMap<Long, Integer>();
	
	class EC implements Comparator<Long> {

		@Override
		public int compare(Long e1, Long e2) {
			return z(e1) - z(e2);
		}
		
		private int z(long e) {
			try {
				return D.get(dekey(e)[0]) + G.get(e);
			} catch (Exception ex) {
				throw new RuntimeException(Arrays.toString(dekey(e)));
			}
		}
	}
	
	////////////////////////////////////////////////////////////////////////////////////

	static BufferedReader r;
	static long t;
	
	static void print2 (Object o) {
		System.out.println(o);
	}
	
	static void print (Object o) {
		print2(o);
		//print2((millis() - t) / 1000.0);
		System.exit(0);
	}
	
	static void run () throws IOException {
		r = new BufferedReader(new InputStreamReader(System.in));
		new D();
	}
	
	public static void main(String[] args) throws IOException {
		run();
	}
	
	static long millis() {
		return System.currentTimeMillis();
	}
}
