#include <iostream>

#define rep2(i, b, n) for(int i=b; i < n; ++i)
#define rep(i, n) rep2(i, 0, n)

using namespace std;

int main( void )
{
	int n;
	int fd[5][5], fd2[5][5];
	cin >> n;
	while(n--) {
		rep(y, 5) rep(x, 5) { cin >> fd[y][x]; fd2[y][x] = fd[y][x]; }
		rep(y, 5) rep2(x, 1, 5) {
			if(fd[y][x]) fd[y][x] += fd[y][x-1];
			if(fd2[x][y]) fd2[x][y] += fd2[x-1][y];
		}
		int Max = 0;
		rep(x, 5) {
			rep(s, 4) {
				int val = 0, count = 0;
				rep2(y, s, 5) {
					if(!fd[y][x]) {
						count = 0;
						val = 0;
					}
					else {
						if(val == 0) val = fd[y][x];
						else val = min(val, fd[y][x]);
						++count;
						Max = max(Max, count * val);
					}
				}
			}
		}
		rep(y, 5) {
			rep(s, 4) {
				int val = 0, count = 0;
				rep2(x, s, 5) {
					if(!fd2[y][x]) {
						count = 0;
						val = 0;
					}
					else {
						if(val == 0) val = fd2[y][x];
						else val = min(val, fd2[y][x]);
						++count;
						Max = max(Max, count * val);
					}
				}
			}
		}
		//rep(y, 5) {
		//	rep(x, 5) cout << fd2[y][x] << ' ';
		//	cout << endl;
		//}
		cout << Max << endl;
	}
	return 0;
}
