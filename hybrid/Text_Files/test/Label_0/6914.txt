struct UnionFind {
	vector<int> v;
	UnionFind(int n) : v(n) { for (int i = 0; i < n; i++) v[i] = i; }
	int  find(int x) { return v[x] == x ? x : v[x] = find(v[x]); }
	void unite(int x, int y) { v[find(x)] = find(y); }
};

vi ask(vi num) {
	vi res;
	int a = -1;
	REP(i, num.size()) {
		if (num[i]!=a) {
			res.push_back(num[i]);
		}
		a = num[i];
	}
	
	while (!(res.size() == 1 && res[0] == 0) && res[res.size() - 1] == 0)
		res.erase(res.begin() + res.size() - 1);
	return res;
}

struct st{
	vi num;
	int cnt;
};

int main() {
	int n;
	while (cin >> n, n) {
		string a, b;
		cin >> a >> b;
		vi sub;
		REP(i, n) {
			sub.push_back((a[i] - b[i]+10) % 10);
		}
		sub = ask(sub);
		map<vi, int> G;
		G[sub] = 0;
		queue<st> Q;
		Q.push({sub,0});
		int ans = 0;
		while (!Q.empty()) {
			st q = Q.front();Q.pop();
			if (q.num.size()==1&&q.num[0]==0) {
				ans = q.cnt;
				break;
			}
			int num2= q.num[0];
			REP(i, q.num.size()){
				vi vq = q.num;
				REP(j, i + 1) {
					vq[j] = (vq[j] - num2 + 10) % 10;
					vi vq2 = ask(vq);
					if (G.find(vq2) == G.end() || G[vq2]>q.cnt + 1) {
						G[vq2] = q.cnt + 1;
						Q.push({ vq2,q.cnt + 1 });
					}
				}
			}
			num2 = q.num[q.num.size() - 1];
			REP(i, q.num.size()) {
				vi vq = q.num;
				REP(j, i + 1) {
					vq[vq.size() - 1 - j] = (vq[vq.size() - 1 - j] - num2 + 10) % 10;
					vi vq2 = ask(vq);
					if (G.find(vq2) == G.end() || G[vq2]>q.cnt + 1) {
						G[vq2] = q.cnt + 1;
						Q.push({ vq2,q.cnt + 1 });
					}
				}
			}
		}
		cout << ans << endl;
	}
}
