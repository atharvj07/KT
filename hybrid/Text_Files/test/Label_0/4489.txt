int main(){
	int H,W,q;
	while(cin >> H >> W >> q && H){
		int ans = 0;
		rep(i,q) cin >> a[i] >> b[i] >> o[i];
		set<int> g[2];
		rev(i,q){
			if(a[i]==0){
				if(!g[0].count(b[i]) && o[i]) ans += W - g[1].size();
				g[0].insert(b[i]);
			}else{
				if(!g[1].count(b[i]) && o[i]) ans += H - g[0].size();
				g[1].insert(b[i]);
			}
		}
		cout << ans << endl;
	}
