#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, z, ans, i, j, count = 0, arr[100000], brr[100000], k = 0, l = 0;
  cin >> n >> m >> z;
  int x = n;
  i = 1;
  while (1) {
    x = n * i;
    if (x > z)
      break;
    else {
      arr[k] = x;
      k++;
    }
    i++;
  }
  i = 1;
  int y = m;
  while (1) {
    y = m * i;
    if (y > z)
      break;
    else {
      brr[l] = y;
      l++;
    }
    i++;
  }
  for (i = 0; i < k; i++) {
    for (j = 0; j < l; j++) {
      if (arr[i] == brr[j]) count++;
    }
  }
  cout << count;
  return 0;
}
