	public void submit(){
		int k = readInt(), n = readInt();
		int[] a = new int[k], b = new int[n];
		for(int i=0;i<k;i++) a[i] = readInt();
		for(int i=0;i<n;i++) b[i] = readInt();
		bufpl(marks(k,n,a,b));
	}
	
	public int marks1(int k, int n, int[] a, int[] b){
		int[] p = new int[k];
		p[0] = 0;
		for(int i=1;i<k;i++) p[i] = p[i-1] + a[i];
		Arrays.sort(p);
		Arrays.sort(b);
		int m = 0;
		for(int i=0;i<k-1;i++) if(p[i+1]>p[i]) m++;
		
		int[] seq = new int[m];
		int[] pattern = new int[n-1];
		int j=0;
		for(int i=0;i<k-1;i++) if(p[i+1]>p[i]) seq[j++] = p[i+1] - p[i];
		for(int i=0;i<n-1;i++) pattern[i] = b[i+1] - b[i];
		return search(seq, pattern);
	}
	
	public int marks(int k, int n, int[] a, int[] b){
		for(int i=1;i<k;i++) a[i] += a[i-1];
		Arrays.sort(a);
		Arrays.sort(b);
		int m = 1;
		for(int i=0;i<k-1;i++) if(a[i+1]>a[i]) m++;
		
		int[] seq = new int[m];
		int j=1;
		seq[0] = a[0];
		for(int i=0;i<k-1;i++) if(a[i+1]>a[i]) seq[j++] = a[i+1];
		return search(seq, b);
	}
	
	public int search(int[] seq, int[] pattern){
		//return the number of x, so that
		//pattern[] + x is a subseq (not necessarily continuous) of seq[]
		//both seq[] and pattern are supposed to be strictly increasing
		int n = seq.length;
		int m = pattern.length;
		int ret = 0;
		HashMap<Integer, Integer> map = new HashMap();
		for(int i=0;i<n;i++) map.put(seq[i], i);
		boolean[][] match = new boolean[n][m];
		for(int i=n-1;i>=0;i--) for(int j=m-1;j>=0;j--){
			if(j==m-1) match[i][j] = true;
			else if(i==n-1) match[i][j] = false;
			else{
				Integer k = map.get(seq[i]-pattern[j]+pattern[j+1]);
				if(k==null || match[k][j+1]==false) match[i][j] = false;
				else match[i][j] = true;
			}
		}
		
		for(int i=0;i<n;i++) if(match[i][0]) ret += 1;
		return ret;
	}
