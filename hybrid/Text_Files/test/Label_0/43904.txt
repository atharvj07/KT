(h, w) = map(int, input().split())
r = [*map(int, input().split())]
c = [*map(int, input().split())]
m = []
for i in range(h):
	m.append([1] * r[i] + [0] * (w - r[i]))
for i in range(w):
	for j in range(c[i]):
		m[j][i] = 1
ans = 0
for i in range(h):
	t = 0
	for j in range(w):
		if m[i][j] == 1:
			t += 1
		else:
			break
	if r[i] != t:
		ans = -1
for i in range(w):
	t = 0
	for j in range(h):
		if m[j][i] == 1:
			t += 1
		else:
			break
	if c[i] != t:
		ans = -1
if ans == -1:
	print(0)
	exit()
for i in range(h):
	t = True
	for j in range(w):
		if t and m[i][j] == 0:
			t = False
		m[i][j] = [m[i][j], t]
for i in range(w):
	t = True
	for j in range(h):
		if t and m[j][i][0] == 0:
			t = False
		m[j][i].append(t)
for i in range(1, h):
	for j in range(1, w):
		if m[i - 1][j][2] == m[i][j - 1][1] == False:
			ans += 1
print(pow(2, ans, int(1000000000.0 + 7)))
