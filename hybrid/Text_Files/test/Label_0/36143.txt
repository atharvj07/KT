import java.util.HashMap;
import java.util.Scanner;

public class Main {
	
	public static void main(String[] args)
	{
		Scanner s=new Scanner(System.in);
		
		int n=s.nextInt();
		
		long[] arr=new long[n+1];
		arr[0]=0;
		
		for(int i=1;i<=n;i++)
		{
			arr[i]=s.nextInt();
		}
		
		long[] pre=new long[n+1];
		
		pre[0]=arr[0];
		
		for(int i=1;i<=n;i++)
		{
			pre[i]=pre[i-1]+arr[i];
		}
		
		long count=0;
		
		HashMap<Long,Integer> map=new HashMap<>();
		
		int end=0;
		
		for(int i=0;i<n;i++)
		{
			while(end<n+1)
			{
				if(map.containsKey(pre[end]))
				{
					break;
				}
				else
				{
					map.put(pre[end], 1);
				}
				
				end++;
			}
			
			long l=i+1;
			long r=end-1;
			
			count=count+r-l+1;
			
			map.remove(pre[i]);
			
		}
		
		System.out.println(count);
		
	}
	
}
