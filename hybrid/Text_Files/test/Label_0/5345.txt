using namespace std;

typedef long long ll;
typedef pair<int, int> P;
typedef vector<int> vi;

int main(){
	int s, d, m;
	while(cin >> s >> d >> m){
		int num = 0;
		vector<int> w;
		vector<int> p;
		vector<P> idx;
		rep(i, s){
			int k;
			cin >> k;
			idx.push_back(MP(num, num+k-1));
			num += k;
			rep(j, k){
				int we, pr;
				cin >> we >> pr;
				w.push_back(we);
				p.push_back(pr);
			}
		}

		vector<int> ww;
		vector<int> pp;
		rep(i, d){
			int f;
			cin >> f;
			for(int j = idx[f].first; j <= idx[f].second; j++){
				ww.push_back(w[j]);
				pp.push_back(p[j]);
			}
		}

		int n = ww.size();
		vector<int> dp(m+1, 0);
		rep(i, n){
			for(int j = m; j >= pp[i]; j--){
				dp[j] = max(dp[j], dp[j-pp[i]] + ww[i]);
			}
		}
		int answ = 0;
		int ansp = 0;
		rep(i, m+1){
			if(answ < dp[i]){
				answ = dp[i];
				ansp = i;
			}
		}
		cout << answ << ' ' << ansp << endl;
	}

	return 0;
}
