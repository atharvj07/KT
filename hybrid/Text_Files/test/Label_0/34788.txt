import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Door {
    public static void main(String[] args) throws IOException {
        BufferedReader f = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(f.readLine());
        PrintWriter out = new PrintWriter(System.out);
        for (int t1 = 0; t1 < t; t1++) {
            StringTokenizer tokenizer = new StringTokenizer(f.readLine());
            int n = Integer.parseInt(tokenizer.nextToken());
            int l = Integer.parseInt(tokenizer.nextToken()) - 1;
            tokenizer = new StringTokenizer(f.readLine());
            int[] heights = new int[n];
            for (int i = 0; i < n; i++) {
                int num = Integer.parseInt(tokenizer.nextToken());
                heights[i] = num;
            }
            boolean[] isPeak = new boolean[n];
            for (int i = 1; i < n - 1; i++) {
                if (heights[i] > heights[i - 1] && heights[i] > heights[i + 1]) {
                    isPeak[i] = true;
                }
            }

            int[] prefPeaks = new int[n + 1];
            for (int i = 1; i <= n; i++) {
                if (isPeak[i - 1]) {
                    prefPeaks[i]++;
                }
                prefPeaks[i] += prefPeaks[i - 1];
            }

            if (prefPeaks[n] == 0) {
                out.println(1 + " " + 1);
                continue;
            }

            int bestNumOfPeaks = 0;
            int index = -1;
            for (int i = 0; i + l < n; i++) {
                if (prefPeaks[i + l] - prefPeaks[i + 1] > bestNumOfPeaks) {
                    bestNumOfPeaks = prefPeaks[i + l] - prefPeaks[i + 1];
                    index = i;
                }
            }


            out.println((bestNumOfPeaks  + 1) + " " + (index + 1));
        }

        out.close();
    }
}
