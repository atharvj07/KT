public class d {
    static int MOD = 998244353;
    public static void main(String[] args) {
        FS in = new FS(System.in);
        PrintWriter out = new PrintWriter(System.out);
        int n = in.nextInt();
        int[] wanted = new int[2000000];
        int[][] wants = new int[n][];
        for(int i = 0; i < n; i++) {
            int k = in.nextInt();
            wants[i] = new int[k];
            for(int j = 0; j < k; j++) wanted[wants[i][j] = in.nextInt()]++;
        }
        
        long res = 0;
        for(int i = 0; i < n; i++) {
            long tot = 0;
            for(int j = 0; j < wants[i].length; j++) {
                tot += wanted[wants[i][j]];
            }
            
            res = (res+comp(tot, (long)wants[i].length*n*n))%MOD;
        }
        System.out.println(res);
    }
    
    static long comp(long x, long y) {
        long gcd = gcd(x, y);
        x /= gcd;
        y /= gcd;
        long modInv = (modInv(y, MOD)+MOD)%MOD;
        return (x*modInv)%MOD;
    }

    static long gcd(long a, long b) {
        return b == 0 ? a : gcd(b, a%b);
    }
    
    // Computes the modular inverse of x
    // Returns 0 if the GCD of x and mod is not 1
    // O(log n)  :  Can be converted to use BigIntegers
    static long modInv(long x, long mod) {
        if (x == 1) {
            return 1;
        }
        if (gcd(x, mod) != 1) {
            return 0;
        }
        long r = mod % x;
        return modInv(r, x % r, 0, 1, mod / x, x / r);
    }

    static long modInv(long a, long b, long y0, long y1, long q0, long q1) {
        long y2 = y0 - y1 * q0;
        return b == 0 ? y2 : modInv(b, a % b, y1, y2, q1, a / b);
    }

    static class FS {

        BufferedReader in;
        StringTokenizer token;

        public FS(InputStream st) {
            in = new BufferedReader(new InputStreamReader(st));
        }

        public String next() {
            if (token == null || !token.hasMoreElements()) {
                try {
                    token = new StringTokenizer(in.readLine());
                } catch (Exception e) {
                }
                return next();
            }
            return token.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }
    }
}
