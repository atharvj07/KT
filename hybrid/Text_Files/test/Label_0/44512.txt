st = input()
ans = ''
k = 0
while len(st) > k and st[k] == 'a':
	k += 1
	ans += '0 '
while len(st) > k and st[k] == 'b':
	k += 1
	ans += '0 '
for i in range(k, len(st)):
	if st[i - 1] != st[i]:
		ans += '1 '
	else:
		ans += '0 '
if st[-1] == 'a':
	ans += '1'
else:
	ans += '0'
ans = ans[2:]
print(ans)
