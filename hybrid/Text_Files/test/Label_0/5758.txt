#include <iostream>
#include <algorithm>
#include <vector>
#include <queue>

using namespace std;

class Solver {
public:
	static void solve() {
		int n;
		cin >> n;

		int ts[55] = {};
		int sum = 0;
		for (int i = 0; i < n; i++) {
			cin >> ts[i];
			sum += ts[i];
		}

		bool*** dp = new bool** [55];
		for (int i = 0; i < 55; i++) {
			dp[i] = new bool* [1000];
			for (int j = 0; j < 1000; j++) {
				dp[i][j] = new bool[1000];
				for (int k = 0; k < 1000; k++) {
					dp[i][j][k] = false;
				}
			}
		}

		dp[0][ts[0]][0] = true;
		dp[0][0][ts[0]] = true;
		dp[0][0][0] = true;
		for (int i = 0; i + 1 < n; i++) {
			for (int j = 0; j < 1000; j++) {
				for (int k = 0; k < 1000; k++) {
					if (!dp[i][j][k])continue;

					if (j + ts[i+1] < 1000)dp[i + 1][j + ts[i+1]][k] = true;
					if (k + ts[i+1] < 1000)dp[i + 1][j][k + ts[i+1]] = true;
					dp[i + 1][j][k] = true;
				}
			}
		}

		int res = 1e9;
		for (int i = 0; i < 1000; i++) {
			for (int j = 0; j < 1000; j++) {
				if (!dp[n - 1][i][j])continue;

				int tmp = max(max(i, j), sum - i - j);
				res = min(res, tmp);
				//cout << i << ' ' << j << ' ' << sum - i - j << endl;
			}
		}

		cout << res << endl;
	}
};

int main() {
	Solver::solve();
	return 0;
}
