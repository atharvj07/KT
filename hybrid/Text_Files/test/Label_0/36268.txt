
import java.io.IOException;
import java.io.InputStream;
import java.io.OutputStream;
import java.io.PrintWriter;
import java.util.Arrays;
import java.io.InputStreamReader;
import java.util.StringTokenizer;
import java.io.BufferedReader;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 *
 * @author neuivn
 */
public class fancy_number {
    public static void main(String[] args) {
        InputReader in = new InputReader(System.in);
        PrintWriter out = new PrintWriter(System.out);
        TaskC solver = new TaskC();
        solver.solve(in, out);
        out.close(); // Remember else data not saved
    }

    /*

     */
    static class TaskC {
        public void solve(InputReader in, PrintWriter out) {

            int N = in.nextInt();
            int K = in.nextInt();
            String line = in.nextLine();
            int[] num = new int[line.length()];
            for (int i = 0; i < line.length(); ++i) num[i] = line.charAt(i) - '0'; // Store number into array

            int[] cnt = new int[10];
            boolean exists = false;

            // Count each digit
            for (int j : num) {
                cnt[j]++;
                exists = (K <= cnt[j]); // At least there are k i.
            }

            if (exists) {
                out.println(0);
                out.println(line);
                return;
            }

            int[] ans = new int[num.length];
            int min = Integer.MAX_VALUE;
            for (int dig = 0; dig <= 9; ++dig) {
                TaskC.Pair[] checker = new TaskC.Pair[num.length];
                for (int i = 0; i < line.length(); ++i) {
                    int number = num[i];
                    if (number > dig) {
                        checker[i] = new TaskC.Pair(i, number - dig);
                    } else {
                        checker[i] = new TaskC.Pair(2 * line.length() - i, -(number - dig));
                    }
                }
                Arrays.sort(checker);

                int[] mul = new int[line.length()];
                System.arraycopy(num, 0, mul, 0, num.length);
                int cost = 0;
                for (int i = 0; i < K; ++i) {
                    cost += checker[i].val;
                    if (checker[i].pos < line.length()) {
                        mul[checker[i].pos] = dig;
                    } else {
                        mul[2 * line.length() - checker[i].pos] = dig;
                    }
                }
                if (cost < min) {
                    min = cost;
                    System.arraycopy(mul, 0, ans, 0, mul.length);
                } else if (cost == min) {
                    if (isLess(mul, ans)) {
                        System.arraycopy(mul, 0, ans, 0, mul.length);
                    }
                }

            }
            out.println(min);
            StringBuilder q = new StringBuilder();
            for (int an : ans) {
                q.append(an);
            }
            out.println(q);
        }

        private boolean isLess(int[] a, int[] b) {
            // Lexicographical order
            for (int i = 0; i < a.length; ++i) {
                if (a[i] < b[i]) return true;
                if (a[i] > b[i]) return false;
            }
            return false;
        }

        static class Pair implements Comparable<TaskC.Pair> {
            int pos;
            int val;

            public Pair(int _pos, int _val) {
                pos = _pos;
                val = _val;

            }


            public int compareTo(TaskC.Pair o) {
                if (this.val != o.val) return this.val - o.val;

                return this.pos - o.pos;
            }

        }

    }

    static class InputReader {
        public BufferedReader reader;
        public StringTokenizer tokenizer;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream), 32768);
            tokenizer = null;
        }

        public String next() {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

        public String nextLine() {
            try {
                return reader.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return null;
        }

    }
}
