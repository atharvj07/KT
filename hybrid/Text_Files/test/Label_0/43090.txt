from functools import reduce
s1 = input().split()
k = int(s1[0])
n = int(s1[1])
suck = list(map(int, input().split()))
maper = [0 for i in range(0, k)]
fild = 0
answer = []
for prob in suck:
	prob -= 1
	if maper[prob] == 0:
		fild += 1
	maper[prob] += 1
	if fild == k:
		maper = list(map(lambda i: i - 1, maper))
		fild = reduce(lambda res, i: res if i == 0 else res + 1, [0] + maper)
		answer.append('1')
	else:
		answer.append('0')
print(''.join(answer))
