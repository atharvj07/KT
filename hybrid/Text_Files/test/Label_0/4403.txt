#include <bits/stdc++.h>
using namespace std;
const int INF = 0x3f3f3f3f;
const long long LINF = 0x3f3f3f3f3f3f3f3fll;
const int MAX = 1e4 + 5;
int dp[101][MAX];
vector<vector<int>> it;
int n, m;
int solve(int second, int k) {
  if (second == n) return 0;
  if (k == 0) return 0;
  if (dp[second][k] != -1) return dp[second][k];
  int maxi = 0;
  for (int i = 0; i < it[second].size() and i <= k; i++) {
    maxi = max(solve(second + 1, k - i) + it[second][i], maxi);
  }
  return dp[second][k] = maxi;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> n >> m;
  it = vector<vector<int>>(n);
  for (int second = 0; second < n; second++) {
    int mm;
    cin >> mm;
    vector<int> a(mm);
    for (int &i : a) cin >> i;
    vector<int> p(mm + 1);
    for (int i = 1; i <= mm; i++) p[i] = p[i - 1] + a[i - 1];
    it[second].push_back(0);
    for (int c = 1; c <= mm; c++) {
      int ans = 0;
      int l = c, r = 0;
      while (l >= 0) {
        int act = p[l] + (p[mm] - p[mm - r]);
        ans = max(ans, act);
        l--;
        r++;
      }
      it[second].push_back(ans);
    }
  }
  memset(dp, -1, sizeof dp);
  cout << solve(0, m) << '\n';
  exit(0);
}
