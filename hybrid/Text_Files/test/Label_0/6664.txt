#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<long long> pre(n, 0);
  map<long long, long long> pos;
  for (int i = 0; i < n; i++) {
    cin >> pre[i];
    if (i) pre[i] += pre[i - 1];
  }
  long long first = -1, ans = 0;
  pos[0] = -1;
  for (long long i = 0; i < n; i++) {
    if (pos.find(pre[i]) == pos.end())
      pos[pre[i]] = i;
    else {
      first = max(first, pos[pre[i]] + 1);
      pos[pre[i]] = i;
    }
    ans += i - first;
  }
  cout << ans;
}
