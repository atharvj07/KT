#include <bits/stdc++.h>
using namespace std;
const int maxn = (int)2e5 + 9;
const double eps = 1e-6;
int n;
char s[maxn];
void init() {
  scanf("%d", &n);
  scanf("%s", s);
}
int ans[maxn];
int mx[10];
int mn[10];
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    init();
    memset(ans, 0, sizeof(ans));
    for (int i = 0; i < 10; i++) {
      mx[i] = -1;
      mn[i] = maxn;
    }
    for (int i = 0; i < n; i++) {
      s[i] -= '0';
      mx[s[i]] = max(mx[s[i]], i);
      mn[s[i]] = min(mn[s[i]], i);
    }
    int pos = -1;
    int f = -1;
    int fl = 0;
    int pp;
    for (int i = 0; i < 10; i++) {
      if (mx[i] == -1) continue;
      if (mn[i] > pos) {
        pos = mx[i];
        for (int j = mn[i]; j <= mx[i]; j++) {
          if (s[j] == i) {
            ans[j] = '1';
          }
        }
      } else {
        for (int j = pos; j < n; j++) {
          if (s[j] == i) {
            ans[j] = '1';
          }
        }
        f = i;
        break;
      }
    }
    if (f != -1) {
      for (int i = pos; i >= 0; i--) {
        if (s[i] == f) {
          mx[f] = i;
          break;
        }
      }
      pos = mn[f] - 1;
    }
    for (int i = f; f != -1 && i < 10; i++) {
      if (mx[i] == -1) continue;
      if (mn[i] > pos) {
        pos = mx[i];
        for (int j = mn[i]; j <= mx[i]; j++) {
          if (s[j] == i) {
            ans[j] = '2';
          }
        }
      } else {
        for (int j = pos; j < n; j++) {
          if (s[j] == i) {
            ans[j] = '2';
          }
        }
        fl = 1;
        break;
      }
    }
    if (!fl) {
      for (int i = 0; i < n; i++) {
        printf("%c", ans[i]);
      }
      printf("\n");
    } else {
      printf("-\n");
    }
  }
  return 0;
}
