#include <bits/stdc++.h>
using namespace std;
vector<pair<int, int> > v;
int main() {
  int n, i, x, y, flag = 0;
  cin >> n;
  for (i = 0; i < n; i++) {
    cin >> x >> y;
    v.push_back(make_pair(x, y));
  }
  sort(v.begin(), v.end());
  for (i = 0; i < n - 1; i++) {
    if (v[i].second > v[i + 1].second) flag = 1;
  }
  if (flag == 1) {
    cout << "Happy Alex\n";
  } else {
    cout << "Poor Alex\n";
  }
}
