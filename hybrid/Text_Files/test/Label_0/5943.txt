#include <bits/stdc++.h>
using namespace std;
int n, k, i, s, a[100005], c, xb, b[100005];
int main() {
  cin >> n >> k;
  for (i = 1; i <= n; ++i) {
    cin >> a[i];
    s += a[i];
  }
  if (s % k != 0) {
    cout << "No\n";
    return 0;
  }
  s /= k;
  int x = 0;
  for (i = 1; i <= n; ++i) {
    ++x;
    c += a[i];
    if (c > s) {
      cout << "No\n";
      return 0;
    }
    if (c == s) {
      ++xb;
      b[xb] = x;
      c = 0;
      x = 0;
    }
  }
  cout << "Yes\n";
  for (i = 1; i <= xb; ++i) cout << b[i] << " ";
  return 0;
}
