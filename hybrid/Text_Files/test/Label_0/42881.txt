def solve(n, arr):
	if sum(arr) < n:
		return -1
	m = len(arr)
	bits = [0] * 61
	for num in arr:
		bits[num.bit_length() - 1] += 1
	i = 0
	l = n.bit_length()
	res = 0
	while i < l:
		if n >> i & 1:
			if bits[i]:
				bits[i] -= 1
			else:
				while bits[i] == 0:
					i += 1
					res += 1
				bits[i] -= 1
				continue
		bits[i + 1] += bits[i] >> 1
		i += 1
	return res
tests = int(input())
for _ in range(tests):
	(n, m) = list(map(int, input().split()))
	arr = list(map(int, input().split()))
	print(solve(n, arr))
