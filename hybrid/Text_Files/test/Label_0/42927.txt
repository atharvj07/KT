s = input()
summ = 0

def tocost(s):
	if len(s) >= 4 and s[-3] == '.':
		return int(s.replace('.', ''))
	else:
		return int(s.replace('.', '')) * 100
i = 0
s += 'b'
while i < len(s):
	if s[i].isdigit():
		for j in range(i, len(s)):
			if not s[j].isdigit() and s[j] != '.':
				summ += tocost(s[i:j])
				i = j
				break
	i += 1
summ = str(summ)
if summ == '0':
	print(0)
elif len(summ) == 1:
	print('0.0' + summ)
elif len(summ) == 2:
	print('0.' + summ)
elif summ[-1] == summ[-2] == '0':
	ans = ''
	summ = summ[:-2]
	for i in range(0, len(summ)):
		if (len(summ) - i) % 3 == 0 and i != 0:
			ans += '.'
		ans += summ[i]
	print(ans)
else:
	ans = ''
	for i in range(0, len(summ)):
		if (len(summ) - 2 - i) % 3 == 0 and i != 0:
			ans += '.'
		ans += summ[i]
	print(ans)
