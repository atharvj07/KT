#include <bits/stdc++.h>
const int N = 110;
using namespace std;
inline int read() {
  int x = 0, f = 1;
  int ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') {
      f = -1;
    }
    ch = getchar();
  }
  while (ch <= '9' && ch >= '0') {
    x = (x << 1) + (x << 3) + ch - '0';
    ch = getchar();
  }
  return x * f;
}
int n, a[N];
int main() {
  int q = read();
  while (q--) {
    bool flg = 0;
    n = read();
    for (int i = 1; i <= n; i++) {
      a[i] = read();
    }
    sort(a + 1, a + 1 + n);
    for (int i = 2; i <= n; i++) {
      if (a[i] == a[i - 1] + 1) {
        cout << 2 << endl;
        flg = 1;
        break;
      }
    }
    if (!flg) cout << 1 << endl;
  }
  return 0;
}
