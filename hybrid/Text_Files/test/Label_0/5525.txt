#include <bits/stdc++.h>
using namespace std;
const double pi = acos(-1.0);
const int INF = 0x3f3f3f3f;
const int MOD = 1e9 + 7;
const double EPS = 1e-9;
int n, s, m, k, a[1515], c[1515], cnt_m;
struct Line {
  int l, r;
  void input() { scanf("%d%d", &l, &r); }
  bool operator<(const Line &h) const { return r < h.r; }
} line[1515];
int dp[2][1515];
bool cmp1(Line a, Line b) { return a.l == b.l ? a.r > b.r : a.l < b.l; }
bool DP(int x) {
  for (int i = 1; i <= n; ++i) {
    c[i] = c[i - 1] + (a[i] <= x);
  }
  memset(dp[0], 0, sizeof(dp[0]));
  for (int j = 1; j <= s; ++j) {
    int f = j & 1, ma = 0;
    for (int i = 1, k = 0; i <= cnt_m; ++i) {
      while (k < cnt_m && line[k + 1].r < line[i].l) {
        ++k;
        ma = max(ma, dp[f ^ 1][k]);
      }
      dp[f][i] = max(ma + c[line[i].r] - c[line[i].l - 1],
                     dp[f ^ 1][k + 1] + c[line[i].r] - c[line[k + 1].r]);
    }
  }
  int ans = 0;
  for (int i = 1; i <= cnt_m; ++i) ans = max(ans, dp[s & 1][i]);
  return ans >= k;
}
int main() {
  scanf("%d%d%d%d", &n, &m, &s, &k);
  for (int i = 1; i <= n; ++i) {
    scanf("%d", &a[i]);
  }
  line[0].l = line[0].r = 0;
  for (int i = 1; i <= m; ++i) {
    line[i].input();
  }
  sort(line + 1, line + m + 1, cmp1);
  for (int i = 1, mr = 0; i <= m; ++i) {
    if (line[i].r > mr) {
      line[++cnt_m] = line[i];
      mr = line[i].r;
    }
  }
  int s = 1, e = 1e9, mi, ans = -1;
  while (s <= e) {
    mi = s + e >> 1;
    if (DP(mi))
      e = (ans = mi) - 1;
    else
      s = mi + 1;
  }
  printf("%d\n", ans);
  return 0;
}
