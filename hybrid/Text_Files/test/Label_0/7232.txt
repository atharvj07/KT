#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int t; std::cin >> t;
    for (int j = 0; j < t; j++) {
        int n; std::cin >> n;
        std::vector<long long> d(n);
        for (int i = 0; i < n; i++) std::cin >> d[i];
        sort(d.begin(), d.end());
        long long ans = d[n - 1];
        std::vector<long long> a(n + 5);
        a[0] = 0;
        a[1] = 0;
        a[2] = d[2] - d[1];
        for (int i = 1; i < n; i++) {
            ans += -d[i];
            //std::cout << ans << ' ' << i << std::endl;
            if (i > 2) {
                a[i] = a[i - 1] + (d[i] - d[i - 1]) * (i - 1);
                ans -= a[i];
            }
            else if (i == 2) ans -= d[i] - d[i - 1];
            //std::cout << ans << ' ' << 1 << std::endl;
        }
        std::cout << ans << std::endl;
    }
}
