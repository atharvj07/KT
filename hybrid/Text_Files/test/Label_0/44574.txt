from math import sqrt, ceil, floor
n = int(input())
cnt = 0
for i in range(0, n):
	(a, b) = input().split()
	if b == 'soft':
		cnt += 1
cnt = max(cnt, n - cnt)
for i in range(0, 100):
	t = i * i
	(a, b) = (int(ceil(t / 2)), int(floor(t / 2)))
	if a >= cnt and b >= n - cnt or (b >= cnt and a >= n - cnt):
		print(i)
		break
