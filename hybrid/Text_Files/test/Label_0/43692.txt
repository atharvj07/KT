t = int(input())
for i in range(t):
	(n, m, k) = list(map(int, input().strip().split()))
	a = list(map(int, input().strip().split()))
	if k >= m:
		k = m - 1
	z = m - 1 - k
	sol = []
	for j in range(k + 1):
		big = max(a[j], a[j - k - z - 1])
		for o in range(z + 1):
			if max(a[j + o], a[o + j - k - z - 1]) < big:
				big = max(a[j + o], a[o + j - k - z - 1])
		sol.append(big)
	print(max(sol))
