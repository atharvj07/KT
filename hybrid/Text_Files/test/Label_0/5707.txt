#include<bits/stdc++.h>
#define inf 0x3f3f3f3f
using namespace std;
typedef long long ll;
typedef double dd;
const int N = 1e5+5;
const dd eps = 1e-8;
const int mod = 998244353;

int n,m,k,num[N],fa[N],siz[N];

inline int find(int x){
  return fa[x] == x ? x : fa[x] = find(fa[x]);
}

void unite(int u,int v){
  int x = find(u),y = find(v);
  if(x != y){
    if(siz[x] > siz[y]){
      fa[y] = x;
      siz[x] += siz[y];      
    }
    else{
      fa[x] = y;
      siz[y] += siz[x];
    }
  }
}

int u,v;

int main(){
  // freopen("input.txt","r",stdin);
  ios::sync_with_stdio(false);
  cin>>n>>m>>k;
  for(int i = 1;i <= n;i++){
    siz[i] = 1;
    fa[i] = i;
  }
  for(int i = 1;i <= m;i++){
    cin>>u>>v;
    num[u]++;
    num[v]++;
    unite(u,v);    
  }
  for(int i = 1;i <= k;i++){
    cin>>u>>v;
    if(find(u) == find(v)){
      num[u]++,num[v]++;
    }
  }
  for(int i = 1;i <= n;i++){
    cout<<siz[find(i)]-num[i]-1<<" ";
  }
  cout<<endl;
  return 0;
}
