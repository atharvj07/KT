#include <bits/stdc++.h>
using namespace std;
const int maxd = 1000000007, N = 100000;
const double pi = acos(-1.0);
map<string, int> mp;
int n, m, sq[50][50], siz[50], tot = 0, ans = 0, group[50];
bool vis[50];
int read() {
  int x = 0, f = 1;
  char ch = getchar();
  while ((ch < '0') || (ch > '9')) {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while ((ch >= '0') && (ch <= '9')) {
    x = x * 10 + (ch - '0');
    ch = getchar();
  }
  return x * f;
}
void init() {
  m = read();
  n = read();
  int i, j;
  vector<int> tmp;
  for (i = 1; i <= m + 1; i++) {
    int op;
    if (i <= m)
      op = read();
    else
      op = 1;
    if (op == 1) {
      memset(vis, 0, sizeof(vis));
      int j, k, len = tmp.size();
      for (j = 0; j < len; j++)
        for (k = j + 1; k < len; k++)
          sq[tmp[j]][tmp[k]] = sq[tmp[k]][tmp[j]] = 1;
      tmp.clear();
    } else {
      string s;
      int now;
      cin >> s;
      if (mp[s])
        now = mp[s];
      else {
        mp[s] = (++tot);
        now = mp[s];
      }
      if (!vis[now]) {
        vis[now] = 1;
        tmp.push_back(now);
      }
    }
  }
  for (i = 1; i <= n; i++)
    for (j = 1; j <= n; j++) sq[i][j] ^= 1;
}
bool dfs(int u, int pos) {
  int i, j;
  for (i = u + 1; i <= n; i++) {
    if (siz[i] + pos <= ans) return 0;
    if (!sq[u][i]) continue;
    bool ok = 1;
    for (j = 1; j <= pos; j++)
      if (!sq[i][group[j]]) {
        ok = 0;
        break;
      }
    if (ok) {
      group[pos + 1] = i;
      if (dfs(i, pos + 1)) return 1;
    }
  }
  if (pos > ans) {
    ans = pos;
    return 1;
  }
  return 0;
}
void work() {
  int i;
  for (i = n; i >= 1; i--) {
    group[1] = i;
    dfs(i, 1);
    siz[i] = ans;
  }
  printf("%d", ans);
}
int main() {
  init();
  work();
  return 0;
}
