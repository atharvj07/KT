#include<bits/stdc++.h>

using namespace std;

int main(){
    int N, K;
    cin >> N >> K;
    int n = N;
    int sum = 0;
    int ans = 0;
    while (n >= 0) {
        int row = (sum + K - 1) / K;
        row = max(row, 1);
        // cout << n << " " << sum << " " << ans << " " << row << endl;
        if (row > n)
            break;
        else {
            ans++;
            n -= row;
            sum += row;
        }
    }
    cout << ans << endl;
    return 0;
}
