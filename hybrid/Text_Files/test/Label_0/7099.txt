#include <bits/stdc++.h>
using namespace std;
const int MAX = 110;
int N;
char S[MAX][MAX];
void out() {
  printf("Impossible\n");
  exit(0);
}
vector<int> V[2 * MAX];
bool bio[2 * MAX];
vector<int> topsort;
void dfs(int curr) {
  if (bio[curr]) return;
  bio[curr] = 1;
  for (int nxt : V[curr]) dfs(nxt);
  topsort.push_back(curr);
}
int pos[MAX * 2];
bool ok(const char *a, const char *b) {
  int i;
  for (i = 0; a[i] && b[i] && a[i] == b[i]; ++i)
    ;
  return pos[a[i]] <= pos[b[i]];
}
int main(void) {
  scanf("%d", &N);
  for (int i = 0; i < N; ++i) scanf("%s", S[i]);
  for (int i = 0; i < N; ++i)
    for (int j = 0; j < i; ++j) {
      int it = 0;
      while (S[i][it] == S[j][it] && S[i][it] && S[j][it]) ++it;
      if (S[i][it] && S[j][it] && S[i][it] != S[j][it])
        V[S[i][it]].push_back(S[j][it]);
      if (!S[i][it] && S[j][it]) out();
    }
  for (int i = 'a'; i <= 'z'; ++i)
    if (!bio[i]) dfs(i);
  for (int i = 0; i < 26; ++i) pos[topsort[i]] = i;
  for (int i = 0; i < N; ++i)
    for (int j = 0; j < i; ++j)
      if (!ok(S[j], S[i])) out();
  for (int c : topsort) printf("%c", c);
  printf("\n");
  return 0;
}
