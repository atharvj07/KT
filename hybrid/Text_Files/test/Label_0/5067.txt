typedef long long LL;


LL max_popcnt(LL L, LL R) {
    assert(L <= R);
    LL popcnt = 0;
    for(LL bit = 29; bit >= 0; bit--) {
        if( ((L>>bit)&1) == ((R>>bit)&1) ) {
            popcnt += ((L>>bit)&1);
            continue;
        }
        else {
            bool allOne = true;
            for(LL nxtBit = bit; nxtBit >= 0; nxtBit--) {
                if((R>>nxtBit)&1) {}
                else allOne = false;
            }
            if(allOne) {
                popcnt += bit+1;
            }
            else {
                popcnt += bit;
            }
            break;
        }
    }
    return popcnt;
}

LL solve(vector<int> cnt, int x, LL K) {
    int n = cnt.size();
    LL at_most = 0;
    for(int i = 0; i < n; i++) {
        at_most += 1ll * (1LL<<i) * cnt[i];
    }
    if(at_most < K) return -1;
    LL low = 0, ans = 0, cur = 0;
    for(int i = 0; i <= x; i++) {
        low += 1ll * (1LL<<i) * cnt[i];
        low -= cnt[i];
        cur += cnt[i];
    }
    if(cur >= K) return 0;
//    debug(), dbg(cnt);
//    debug(), dbg(cur);
    for(int i = x+1; i < n; i++) {
        LL sz = 1LL<<(i-x);
        LL req = (K-cur+sz-1)/sz;
        if(req > cnt[i]) {
            cur += sz*cnt[i];
            ans += (sz-1) * cnt[i];
            low += 1ll*((1LL<<x)-1)*cnt[i]*sz;
        }
        else {
            LL full = (K-cur)/sz;
            LL extra = (K-cur)%sz;
//            debug(), dbg(ans), dbg(full), dbg(extra);
            cur += full*sz;
            low += 1ll*((1LL<<x)-1)*full*sz;
            ans += full*(sz-1);

            if(extra <= low) ans += extra;
            else {
                LL lo = 1, hi = extra;
                while(hi > lo) {
                    LL mid = (lo+hi)/2;
                    LL got = mid+low+mid*((1LL<<x)-1);
                    if(got >= extra) hi = mid;
                    else lo = mid+1;
                }
                LL l = lo-1, r = extra-1;
                ans += extra + (i-x-1) - max_popcnt(l, r);
            }
            return ans;
        }
    }
    LL required = K-cur;
//    debug(), dbg(required), dbg(low), dbg(ans);
    assert(required <= low);
    ans += required;
    return ans;
}
