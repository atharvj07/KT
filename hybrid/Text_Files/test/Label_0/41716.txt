t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	a = sorted(a)
	arr = []
	i = 0
	j = n - 1
	if n % 2 == 0:
		while i <= j:
			arr.append(a[i])
			arr.append(a[j])
			i += 1
			j -= 1
	else:
		while i < j:
			arr.append(a[i])
			arr.append(a[j])
			i += 1
			j -= 1
		arr.append(a[i])
	arr = list(reversed(arr))
	print(*arr)
