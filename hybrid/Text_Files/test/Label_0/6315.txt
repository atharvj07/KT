#include <bits/stdc++.h>
using namespace std;
#define rep(i,a,b) for(int i=a;i<b;++i)
#define debug(x) cerr<<#x<<": "<<x<<endl
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, k;
    cin >> n >> k;
    vector<int> a(n), b(n);
    rep(i,0,n) {
        cin >> a[i];
        --a[i];
        b[a[i]] = i;
    }
    vector<int> next(n+1,n), prev(n+1,n);
    next[n] = 0;
    prev[n] = 0;
    set<int> last{a[0]};
    rep(i,1,n) {
        if(i-k>=0) last.erase(a[i-k]);
        auto it = last.lower_bound(a[i]);
        int j = n;
        if(it!=last.end()) {
            j = b[*it];
        }
        next[i] = j;
        prev[i] = prev[j];
        next[prev[j]] = i;
        prev[j] = i;
        last.insert(a[i]);
    }
    vector<int> ans(n);
    int j = n;
    rep(i,0,n) {
        j = next[j];
        ans[j] = i;
    }
    rep(i,0,n) cout << ans[i]+1<<"\n";
}
