l1 = [int(x) for x in input().split()]
(n, d) = (l1[0], l1[1])
l2 = [int(x) for x in input().split()]
l2.sort()
l3 = [l2[0] - d, l2[-1] + d]
if len(l2) == 1:
	print(2)
else:
	if abs(l2[1] - (l2[0] + d)) >= d:
		l3.append(l2[0] + d)
	if abs(l2[-2] - (l2[-1] - d)) >= d:
		l3.append(l2[-1] - d)
	for y in range(1, len(l2) - 1):
		x = l2[y]
		if abs(l2[y + 1] - (l2[y] + d)) >= d:
			l3.append(x + d)
		if abs(l2[y - 1] - (l2[y] - d)) >= d:
			l3.append(x - d)
	print(len(set(l3)))
