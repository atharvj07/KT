#include <bits/stdc++.h>
using namespace std;
long long n, f[4005], mx, my, ans, l[200005], p[200005];
void get_C() {
  p[0] = l[0] = p[1] = 1;
  for (long long i = 2; i <= mx + my; i++)
    p[i] = (1000000007 - 1000000007 / i) * p[1000000007 % i] % 1000000007;
  for (long long i = 1; i <= mx + my; i++)
    l[i] = l[i - 1] * i % 1000000007, p[i] = p[i] * p[i - 1] % 1000000007;
}
long long C(long long x, long long y) {
  return l[x] * p[y] % 1000000007 * p[x - y] % 1000000007;
}
struct ss {
  long long x, y;
  bool operator<(const ss v) const {
    if (x == v.x) return y < v.y;
    return x < v.x;
  }
} a[4005];
signed main() {
  scanf("%lld%lld%lld", &mx, &my, &n);
  get_C();
  mx--;
  my--;
  for (long long i = 1; i <= n; i++) {
    scanf("%lld%lld", &a[i].x, &a[i].y);
    a[i].x--;
    a[i].y--;
  }
  a[++n] = (ss){mx, my};
  sort(a + 1, a + n + 1);
  for (long long i = 1; i <= n; i++) {
    f[i] = C(a[i].x + a[i].y, a[i].x);
  }
  for (long long i = 2; i <= n; i++) {
    for (long long j = 1; j < i; j++) {
      if (a[i].y >= a[j].y) {
        f[i] = (f[i] + 1000000007 -
                f[j] * C(a[i].x - a[j].x + a[i].y - a[j].y, a[i].x - a[j].x) %
                    1000000007) %
               1000000007;
      }
    }
  }
  cout << f[n];
}
