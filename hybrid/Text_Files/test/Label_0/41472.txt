n = int(input())
l = [int(i) for i in input().split()]
ans = 0
for i in range(1, n):
	l1 = l[i - 1]
	l2 = l[i]
	if l1 > l2:
		(l1, l2) = (l2, l1)
	if l1 == 1 and l2 == 2:
		ans += 3
	elif l1 == 1 and l2 == 3:
		ans += 4
	else:
		print('Infinite')
		exit()
	if i > 1:
		if l[i] == 2 and l[i - 1] == 1 and (l[i - 2] == 3):
			ans -= 1
print('Finite')
print(ans)
