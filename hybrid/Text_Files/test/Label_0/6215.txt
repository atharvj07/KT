#include <bits/stdc++.h>
using namespace std;
long long n, m, res1, res2;
long long getpairs(long long n) {
  if (n % 2 == 0) return ((n / 2) * (n - 1));
  return (((n - 1) / 2) * n);
}
int main() {
  ios::sync_with_stdio(false);
  cin >> n >> m;
  if (m == 1) {
    res2 = getpairs(n);
    res1 = res2;
  } else {
    long long a, b;
    res2 = getpairs((n - m) + 1);
    a = n / m;
    b = n % m;
    res1 = ((b)*getpairs(a + 1)) + ((m - b) * getpairs(a));
  }
  if (res1 <= res2)
    cout << res1 << " " << res2;
  else
    cout << res2 << " " << res1;
  return 0;
}
