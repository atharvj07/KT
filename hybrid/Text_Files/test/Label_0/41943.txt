n = int(input())
s = input()
p2 = (0, 0)
p1 = (0, 0)
r = 0
for c in s:
	if c == 'U':
		if p2[0] == p1[0] and p1[0] == p1[1] and (p1[0] != 0):
			r += 1
		p2 = p1
		p1 = (p1[0], p1[1] + 1)
	else:
		if p2[1] == p1[1] and p1[0] == p1[1] and (p1[0] != 0):
			r += 1
		p2 = p1
		p1 = (p1[0] + 1, p1[1])
print(r)
