#include <bits/stdc++.h>
using namespace std;
const int maxn = 200004;
int n, a1[100004], a2[100004], t, ch[200006], res;
int* check = &ch[100003];
int main() {
  ios::sync_with_stdio(NULL);
  cin.tie(NULL);
  cin >> t;
  while (t--) {
    cin >> n;
    res = 0;
    for (int i = -n - 1; i <= n + 1; i++) check[i] = -1;
    check[0] = 0;
    for (int i = 0; i <= n + 2; i++) a1[i] = a2[i] = 0;
    for (int i = 1; i <= n; i++) {
      cin >> a1[i];
      a1[i] = a1[i] == 1 ? 1 : -1;
      a1[i] += a1[i - 1];
      check[a1[i]] = i;
    }
    for (int i = 1; i <= n; i++) cin >> a2[i], a2[i] = a2[i] == 1 ? -1 : 1;
    for (int i = n + 1; i >= 1; i--) {
      a2[i] += a2[i + 1];
      if (check[a2[i]] != -1) res = max(res, check[a2[i]] + n - i + 1);
    }
    cout << 2 * n - res << "\n";
  }
  return 0;
}
