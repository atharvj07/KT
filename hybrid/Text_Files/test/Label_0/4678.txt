#include <bits/stdc++.h>
using namespace std;

int main() {
  int N, M;
  cin >> N >> M;

  vector<int> X(N);
  for (auto &i : X) cin >> i;

  sort(X.begin(), X.end());

  vector<vector<int>> a(M);
  for (auto &i : X) a[i % M].push_back(i);

  int ans = a[0].size() / 2;
  if (M % 2 == 0) {
    ans += a[M / 2].size() / 2;
  }

  for (int i = 1; i <= (M - 1) / 2; ++i) {
    int s = i, t = M - i;
    if (a[s].size() > a[t].size()) swap(s, t);

    ans += a[s].size();

    int cnt = 0;
    for (int j = 1; j < a[t].size(); ++j) {
      if (a[t][j] == a[t][j - 1]) {
        cnt++;
        j++;
      }
    }

    ans += min((int)(a[t].size() - a[s].size()) / 2, cnt);
  }

  cout << ans << endl;
}
