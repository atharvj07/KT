def check_infinite(N, P, A, B):
	if sum(A) <= P:
		return True
	return False

def f(N, P, A, B):
	if check_infinite(N, P, A, B) == True:
		return -1
	AB = [(i, A[i], B[i]) for i in range(N)]
	AB.sort(key=lambda x: -(x[2] / x[1]))
	Ord = [x[0] for x in AB]
	sa = sum(A) - P
	sb = sum(B)
	for i in Ord:
		if sa - A[i] <= 0.0:
			continue
		if (sb - B[i]) / (sa - A[i]) <= B[i] / A[i]:
			sb -= B[i]
			sa -= A[i]
	return sb / sa
(N, P) = input().split(' ')
N = int(N)
P = float(P)
A = []
B = []
for _ in range(N):
	(a, b) = [float(x) for x in input().split(' ')]
	A.append(a)
	B.append(b)
print(f(N, P, A, B))
