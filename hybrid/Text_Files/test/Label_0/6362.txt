#include <bits/stdc++.h>
using namespace std;
long long power(long long x, long long y, long long p) {
  long long res = 1;
  x = x % p;
  while (y > 0) {
    if (y & 1) res = (res * x) % p;
    y = y >> 1;
    x = (x * x) % p;
  }
  return res;
}
const long long M = 1e18 + 7;
const long long MOD = 1e9 + 7;
const int MX = 2e5 + 5;

void solve() {
  int n, c, cz, h;
  cin >> n >> c >> cz >> h;
  string s;
  cin >> s;
  int o = 0, z = 0;
  for (long long i = 0; i < (n); ++i) {
    if (s[i] == '1')
      o++;
    else
      z++;
  }
  int ans = min(min(h * o + c * n, h * z + cz * n), c * z + cz * o);
  cout << ans << endl;
}
int main() {
  ios_base::sync_with_stdio(false);

  int t;
  cin >> t;
  while (t--) {
    solve();
  }

}
