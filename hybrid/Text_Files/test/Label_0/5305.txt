#include <bits/stdc++.h>
#define MOD 1000000007LL
using namespace std;
typedef long long ll;
typedef pair<int,int> P;

int n,c;

struct BITree{
	ll bit[100005];
	BITree(){}
	void update(int k,ll a){
		while(k<=100000){
			bit[k]+=a;
			k+=(k&(-k));
		}
	}

	ll sum(int k){
		ll s=0;
		while(k>0){
			s+=bit[k];
			k-=(k&(-k));
		}
		return s;
	}

	ll query(int a,int b){
		return sum(b)-sum(a);
	}
};

BITree seg,seg2;

int b[100005],w[100005];
P vec_l[100005],vec_r[100005];
P vec[100005];

ll solve(int l,int r){
	int n=r-l;
	if(n<=1)return 0;
	ll ans=0;
	ans+=solve(l,(l+r)/2);
	ans+=solve((l+r)/2,r);
	int li=0,ri=0;
	int size_vl=(l+r)/2-l;
	int size_vr=r-(l+r)/2;
	for(int i=0;i<size_vl;i++){
		vec_l[i]=vec[l+i];
	}
	for(int i=0;i<size_vr;i++){
		vec_r[i]=vec[(l+r)/2+i];
	}
	for(int i=0;i<size_vl;i++){
		seg.update(vec_l[i].second,1);
		seg2.update(vec_l[i].second,vec_l[i].second);
	}
	for(int i=0;i<n;i++){
		if(li<size_vl && (ri==size_vr || vec_l[li].first<vec_r[ri].first)){
			seg.update(vec_l[li].second,-1);
			seg2.update(vec_l[li].second,-vec_l[li].second);
			vec[l+i]=vec_l[li++];
		}else{
			if(ri<size_vr){
				ans+=(ll)seg2.query(0,vec_r[ri].second)*c+seg.query(0,vec_r[ri].second)*vec_r[ri].second;
				ans+=(ll)seg2.query(vec_r[ri].second,100000)+seg.query(vec_r[ri].second,100000)*vec_r[ri].second*c;
			}
			vec[l+i]=vec_r[ri++];
		}
	}
	return ans; 
}

int main(void){
	scanf("%d%d",&n,&c);
	for(int i=0;i<n;i++){
		scanf("%d%d",&b[i],&w[i]);
		b[i]--;
		vec[i]=P(b[i],w[i]);
	}
	printf("%lld\n",solve(0,n));
	return 0;
}
