(n, k) = map(int, input().split())
A = list(map(int, input().split()))
B = list(map(int, input().split()))
B.sort()
flag = True
if A[0] == 0:
	A[0] = B[-1]
	prev = A[0]
	B.remove(B[-1])
else:
	prev = A[0]
for i in range(1, len(A)):
	if A[i] == 0:
		A[i] = B[-1]
		B.remove(B[-1])
	if prev >= A[i]:
		flag = False
		break
	prev = A[i]
if flag == False:
	print('Yes')
else:
	print('No')
