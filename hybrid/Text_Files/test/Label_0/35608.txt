import java.util.Scanner;
import java.util.Stack;
import java.util.Arrays;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.HashMap;
public class C{
	public static boolean canJump(ArrayList<Integer> a, int d){
		for(int i =0 ;i<a.size()-1;++i)
			if(a.get(i) + d < a.get(i+1))
				return false;
		return true;
	}		
	public static void main(String[] args){
			Scanner in = new Scanner(System.in);
			int t = in.nextInt();
			StringBuilder resstr = new StringBuilder();
			for(int j = 0;j<t;++j){
				String str=  in.next();
				int n = str.length();
				ArrayList<Integer> a = new ArrayList<>();
				a.add(0);
				for(int i = 0;i<n;++i){
					if(str.charAt(i) == 'R') a.add(i+1);
				}
				a.add(n+1);
				int l = 1;
				int r = n+2;
				int res = 0;
				while(l+1<r){
					int mid = (l+r)/2;
					if(canJump(a,mid)){
						res = mid;
						r = mid;
					} else
						l = mid+1;
				}
				if((res > l || res == 0) && canJump(a,l)) res = l;
				resstr.append(res + "\n");
			}
			System.out.println(resstr);
	}
}
