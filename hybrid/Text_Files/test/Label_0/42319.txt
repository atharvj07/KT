(n, m) = map(int, input().split())
a = []
for _ in range(n):
	x = list(input().strip())
	a.append(x)
hor = [[0 for _ in range(m)] for _ in range(n)]
ver = [[0 for _ in range(m)] for _ in range(n)]
for i in range(n):
	for j in range(m):
		if i - 1 >= 0 and j - 1 >= 0:
			hor[i][j] -= hor[i - 1][j - 1]
			ver[i][j] -= ver[i - 1][j - 1]
		if i - 1 >= 0:
			hor[i][j] += hor[i - 1][j]
			ver[i][j] += ver[i - 1][j]
		if j - 1 >= 0:
			hor[i][j] += hor[i][j - 1]
			ver[i][j] += ver[i][j - 1]
		if j - 1 >= 0 and a[i][j - 1] == '.' and (a[i][j] == '.'):
			hor[i][j] += 1
		if i - 1 >= 0 and a[i - 1][j] == '.' and (a[i][j] == '.'):
			ver[i][j] += 1
q = int(input())
for _ in range(q):
	(r1, c1, r2, c2) = map(int, input().split())
	r1 -= 1
	c1 -= 1
	r2 -= 1
	c2 -= 1
	val = hor[r2][c2] + ver[r2][c2]
	val -= hor[r2][c1]
	if r1 - 1 >= 0:
		val += hor[r1 - 1][c1]
		val -= hor[r1 - 1][c2]
	val -= ver[r1][c2]
	if c1 - 1 >= 0:
		val += ver[r1][c1 - 1]
		val -= ver[r2][c1 - 1]
	print(val)
