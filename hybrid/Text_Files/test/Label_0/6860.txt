#include <bits/stdc++.h>
using namespace std;
int a[31][4] = {
    {128, 0, 0, 0},       {192, 0, 0, 0},       {224, 0, 0, 0},
    {240, 0, 0, 0},       {248, 0, 0, 0},       {252, 0, 0, 0},
    {254, 0, 0, 0},       {255, 0, 0, 0},       {255, 128, 0, 0},
    {255, 192, 0, 0},     {255, 224, 0, 0},     {255, 240, 0, 0},
    {255, 248, 0, 0},     {255, 252, 0, 0},     {255, 254, 0, 0},
    {255, 255, 0, 0},     {255, 255, 128, 0},   {255, 255, 192, 0},
    {255, 255, 224, 0},   {255, 255, 240, 0},   {255, 255, 248, 0},
    {255, 255, 252, 0},   {255, 255, 254, 0},   {255, 255, 255, 0},
    {255, 255, 255, 128}, {255, 255, 255, 192}, {255, 255, 255, 224},
    {255, 255, 255, 240}, {255, 255, 255, 248}, {255, 255, 255, 252},
    {255, 255, 255, 254}};
const int MAX_N = 100 * 1000 + 10;
int st[MAX_N][4];
int and_q(int a, int b) { return a & b; }
int main() {
  ios_base::sync_with_stdio(false);
  int n, k;
  cin >> n >> k;
  char t;
  for (register int i = (0); i < (int)(n); ++(i))
    cin >> st[i][0] >> t >> st[i][1] >> t >> st[i][2] >> t >> st[i][3];
  for (register int i = (0); i < (int)(31); ++(i)) {
    set<long long> s;
    for (register int j = (0); j < (int)(n); ++(j)) {
      long long x;
      int Q1, Q2, Q3, Q4;
      Q1 = and_q(st[j][0], a[i][0]);
      Q2 = and_q(st[j][1], a[i][1]);
      Q3 = and_q(st[j][2], a[i][2]);
      Q4 = and_q(st[j][3], a[i][3]);
      x = Q4 | (Q3 << 8) | (Q2 << 16) | (Q1 << 24);
      s.insert(x);
    }
    if (s.size() == k) {
      cout << a[i][0] << "." << a[i][1] << "." << a[i][2] << "." << a[i][3]
           << endl;
      return 0;
    }
  }
  puts("-1");
}
