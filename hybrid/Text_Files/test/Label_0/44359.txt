from sys import stdin
import bisect
input = stdin.readline
(n, q) = map(int, input().split())
t = [[] for i in range(n)]
p = list(map(int, input().split()))
for i in range(n - 1):
	t[p[i] - 1].append(i + 1)
	t[i + 1].append(p[i] - 1)
for i in range(n):
	t[i].sort(reverse=True)
vis = []
que = [(-1, 0)]
cnt = 0
p = [-1] + p
for i in range(1, n):
	p[i] -= 1
while que:
	(par, ver) = que.pop()
	vis.append(ver)
	for to in t[ver]:
		if par != to:
			que.append((ver, to))
size = [1] * n
for i in vis[1:][::-1]:
	size[p[i]] += size[i]
idx = [-1] * n
for i in range(n):
	idx[vis[i]] = i
for _ in range(q):
	(u, k) = map(int, input().split())
	if size[u - 1] < k:
		print(-1)
	else:
		print(vis[idx[u - 1] + k - 1] + 1)
