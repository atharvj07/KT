#include <bits/stdc++.h>
using namespace std;
string s[55];
int D, U, L, R;
int ok(int i, int j) { return D <= i && i <= U && L <= j && j <= R; }
int get(int i, int j) {
  return ok(i, j) + ok(i + 1, j) + ok(i, j + 1) + ok(i + 1, j + 1);
}
int main() {
  ios::sync_with_stdio(false);
  int n;
  cin >> n;
  set<int> RR, C;
  for (int i = 0; i < n; i++) {
    cin >> s[i];
    for (int j = 0; j < n; j++)
      if (s[i][j] != '0') {
        RR.insert(i);
        C.insert(j);
      }
  }
  if (RR.empty()) {
    cout << "No" << endl;
    return 0;
  }
  int d = *RR.begin();
  int u = *(--RR.end());
  int l = *C.begin();
  int r = *(--C.end());
  for (D = 0; D <= n + 2; D++) {
    if (abs(D - d) > 2) continue;
    for (U = D + 1; U <= n + 2; U++) {
      if (abs(U - u) > 2) continue;
      for (L = 0; L <= n + 2; L++) {
        if (abs(L - l) > 2) continue;
        for (R = L + 1; R <= n + 2; R++) {
          if (abs(R - r) > 2) continue;
          for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
              if (get(i, j) != s[i][j] - '0') goto done;
          cout << "Yes" << endl;
          return 0;
        done:;
        }
      }
    }
  }
  cout << "No" << endl;
  return 0;
}
