#include <bits/stdc++.h>
using namespace std;
vector<long long> adjac[350000];
long long flag, x, y, visited[350000], mod = 998244353;
long long modular_exponentiation(long long a, long long b, long long c) {
  long long res = 1, exp = a % c;
  if (b == 0) return 1;
  while (b != 0) {
    if (b % 2) res = ((res) % c * (exp) % c) % c;
    exp = ((exp) % c * (exp) % c) % c;
    b = b / 2;
  }
  return res % c;
}
void dfs(vector<long long> adjac[], long long a, long long b) {
  long long i;
  visited[a] = b;
  if (b == 0)
    x++;
  else
    y++;
  for (i = 0; i < adjac[a].size(); i++) {
    if (visited[adjac[a][i]] == -1) dfs(adjac, adjac[a][i], 1 - b);
    if (visited[a] == visited[adjac[a][i]]) flag = 1;
  }
}
int main() {
  long long t, n, m, i, a, b, s, ans;
  cin >> t;
  while (t--) {
    ans = 1;
    flag = 0;
    cin >> n >> m;
    for (i = 0; i < n; i++) {
      visited[i] = -1;
      adjac[i].clear();
    }
    while (m--) {
      cin >> a >> b;
      --a;
      --b;
      adjac[a].push_back(b);
      adjac[b].push_back(a);
    }
    for (i = 0; i < n; i++) {
      x = y = 0;
      s = 1;
      if (visited[i] == -1) {
        dfs(adjac, i, 0);
        s = (modular_exponentiation(2, x, mod) +
             (modular_exponentiation(2, y, mod))) %
            mod;
      }
      if (flag == 1) break;
      ans = ((ans % mod) * (s % mod)) % mod;
    }
    if (flag == 1)
      cout << 0 << endl;
    else
      cout << ans << endl;
  }
  return 0;
}
