def fly(n, elem):
	lst = ['f', 'e', 'd', 'a', 'b', 'c']
	if n == 1 or n == 3:
		return 1 + lst.index(elem)
	return 8 + lst.index(elem)

def eat(t):
	(n, elem) = (int(t[:-1]), t[-1])
	z = n // 4 * 16
	if n % 4 == 0:
		z -= 16
		z += fly(n, elem)
	else:
		z += fly(n % 4, elem)
	return z
print(eat(input()))
