#include <bits/stdc++.h>
using namespace std;
vector<int> v1, v2, v3;
int main() {
  ios_base::sync_with_stdio(0);
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    int x;
    cin >> x;
    if (x == 1) {
      v1.push_back(i);
    } else if (x == 2) {
      v2.push_back(i);
    } else {
      v3.push_back(i);
    }
  }
  cout << min(v1.size(), min(v2.size(), v3.size())) << endl;
  for (int i = 0; i < min(v1.size(), min(v2.size(), v3.size())); i++) {
    cout << v1[i] << " " << v2[i] << " " << v3[i] << endl;
  }
  return 0;
}
