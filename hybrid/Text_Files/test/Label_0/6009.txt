#include <bits/stdc++.h>
using namespace std;
int main() {
  long long t;
  cin >> t;
  while (t--) {
    long long x, y, n, k, ans = 0, sum = 0, val = INT_MAX, res = 0, rem = 0,
                          count = 0, bch = 0;
    cin >> x >> y >> k;
    ans = y * k;
    sum = ans + k;
    rem = (sum - x) / (x - 1) + 1;
    res = x + (rem - 1) * (x - 1);
    if (res < sum) {
      count = rem + 1;
    } else {
      count = rem;
    }
    cout << count + k << "\n";
  }
  return 0;
}
