n = int(input())
k = int(input())
A = int(input())
B = int(input())
kapa = 0
if k == 1:
	print((n - 1) * A)
else:
	while n != 1:
		if n < k:
			kapa += (n - 1) * A
			n = 1
		elif n % k == 0:
			kapa += min(B, (n - n // k) * A)
			n //= k
		else:
			kapa += A * (n - n // k * k)
			n = n // k * k
	if n == 0:
		kapa += 1
	print(kapa)
