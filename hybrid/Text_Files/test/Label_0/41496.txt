n = int(input())
import heapq
tot = []
for i in range(2 * n):
	l = input()
	tot.append(l)
cur = []
heapq.heapify(cur)
can = True
for i in range(2 * n - 1, -1, -1):
	if tot[i][0] == '-':
		heapq.heappush(cur, int(tot[i][2:]))
	else:
		if len(cur) == 0:
			can = False
			break
		tot[i] = heapq.heappop(cur)
cur = []
if can:
	for i in tot:
		if isinstance(i, int) == False:
			if len(cur) == 0:
				can = False
				break
			k = int(i[2:])
			j = heapq.heappop(cur)
			if j != k:
				can = False
				break
		else:
			heapq.heappush(cur, i)
if can:
	print('YES')
	for t in tot:
		if isinstance(t, int):
			print(t, end=' ')
else:
	print('NO')
