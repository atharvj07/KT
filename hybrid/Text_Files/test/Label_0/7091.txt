#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

const int MaxN = 100010;
const int MaxLabel = 20;

int n, ans, f[MaxN]; vector<int> edge[MaxN];

void dfs(int x, int fa) {
  int all = 0;
  for (int y : edge[x]) {
    if (y == fa)
      continue;
    dfs(y, x);
    all |= (f[x] & f[y]);
    f[x] |= f[y];
  }
  
  int t = 0;
  while ((1 << t) <= all || f[x] >> t & 1)
    ++t;
  ans = max(ans, t);
  f[x] >>= t;
  f[x] <<= t;
  f[x] |= (1 << t);
}

int main() {
  scanf("%d", &n);
  for (int i = 1, u, v; i < n; i++) {
    scanf("%d%d", &u, &v);
    edge[u].push_back(v);
    edge[v].push_back(u);
  }
  
  dfs(1, 0);
  cout << ans << "\n";
  return 0;
}
