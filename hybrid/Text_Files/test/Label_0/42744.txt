for _ in range(int(input())):
	(n, m, a, b) = map(int, input().split())
	l = [[0 for j in range(m)] for i in range(n)]
	if n / m == b / a:
		(i, j) = (0, 0)
		for k in range(0, n):
			for z in range(0, a):
				l[i][j] = 1
				j = (j + 1) % m
			i = (i + 1) % n
		print('YES')
		for i in range(0, n):
			for j in range(0, m):
				print(l[i][j], end='')
			print()
	else:
		print('NO')
