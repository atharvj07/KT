import sys
input = sys.stdin.buffer.readline
T = int(input())
for _ in range(T):
	(n, k) = map(int, input().split())
	s = list(input())
	al = s.count(49)
	res = al
	p = [0] * n
	ctr = 0
	for i in range(n):
		if i - k >= 0 and s[i - k] == 49:
			ctr -= 1
		p[i] = ctr
		if s[i] == 49:
			ctr += 1
	dp = [0] * n
	c = 0
	for i in range(n):
		if s[i] == 49:
			if i >= k:
				tmp = min(c, dp[i - k] + p[i])
			else:
				tmp = c
			c += 1
		else:
			tmp = 1
			if i >= k:
				tmp += min(c, dp[i - k] + p[i])
			else:
				tmp += c
		dp[i] = tmp
		res = min(res, dp[i] + (al - c))
	print(res)
