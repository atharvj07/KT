#include <bits/stdc++.h>
using namespace std;
long long int count_zift(map<long long int, long long int> x) {
  long long int ans = 0;
  map<long long int, long long int>::iterator i;
  for (i = x.begin(); i != x.end(); i++) {
    if (!i->first) {
      ans += i->second * (i->second - 1);
    } else {
      ans += i->second * x[-i->first];
    }
  }
  return ans / 2;
}
int main() {
  long long int n;
  map<long long int, long long int> a;
  long long int x;
  cin >> n;
  while (n--) {
    cin >> x;
    a[x]++;
  }
  long long int ans = count_zift(a);
  cout << ans << endl;
}
