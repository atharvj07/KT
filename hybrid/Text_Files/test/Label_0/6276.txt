#include <bits/stdc++.h>
using namespace std;
int n, res, temp, crnt;
string S;
vector<int> V;
int main() {
  cin >> n;
  for (int i = 0; i < n * 2; i++) {
    cin >> S;
    if (S == "add") {
      cin >> temp;
      V.push_back(temp);
    } else {
      crnt++;
      if (!V.size()) continue;
      if (crnt == V[V.size() - 1])
        V.pop_back();
      else {
        res++;
        V.clear();
      }
    }
  }
  cout << res;
}
