from collections import deque
n = int(input())
arr = list(map(int, input().split()))
ind = list(map(int, input().split()))
for i in range(n):
	ind[i] -= 1
indeg = [0 for i in range(n)]
for i in range(n):
	if ind[i] != -2:
		indeg[ind[i]] += 1
q = deque([])
for i in range(n):
	if indeg[i] == 0:
		q.append(i)
(ans, rev) = ([], [])
cnt = 0
while len(q) != 0:
	node = q.popleft()
	nei = ind[node]
	if arr[node] >= 0:
		ans.append(node)
		cnt += arr[node]
		if nei != -2:
			arr[nei] += arr[node]
			indeg[nei] -= 1
			if indeg[nei] == 0:
				q.append(nei)
	else:
		cnt += arr[node]
		rev.append(node)
		if nei != -2:
			indeg[nei] -= 1
			if indeg[nei] == 0:
				q.append(nei)
ans = ans + rev[-1::-1]
for i in range(len(ans)):
	ans[i] += 1
print(cnt)
print(*ans)
