import collections
_n = input()
n = int(_n)
int_list = list()
first = (0, 0)
int_list.append(first)
graph = {}
for x in range(n):
	q = input().split()
	a = int(q[1])
	b = int(q[2])
	interval = (a, b)
	if q[0] == '1':
		int_list.append(interval)
		val = list()
		for k in graph.keys():
			it = graph[k]
			c = k[0]
			d = k[1]
			if a > c and a < d:
				val.append((c, d))
			elif b > c and b < d:
				val.append((c, d))
			if c > a and c < b:
				it.append(interval)
			elif d > a and d < b:
				it.append(interval)
			graph[k] = it
		graph[interval] = val
	elif q[0] == '2':
		beg = a
		end = b
		beg_int = int_list[beg]
		end_int = int_list[end]
		visited = set()
		que = collections.deque()
		que.append(beg_int)
		found = False
		while que:
			i = que.pop()
			adj_list = graph[i]
			if i[0] == end_int[0] and i[1] == end_int[1]:
				found = True
				break
			visited.add(i)
			for a_int in adj_list:
				if a_int not in visited:
					que.append(a_int)
		if found:
			print('YES')
		else:
			print('NO')
