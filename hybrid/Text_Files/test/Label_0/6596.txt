#include <bits/stdc++.h>
using namespace std;
const int INF = 1000000000;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<char> ar(n - 1);
    ar.reserve(n);
    for (auto &i : ar) cin >> i;
    ar.emplace_back('>');
    vector<int> ans1;
    ans1.reserve(n);
    int mx = n;
    for (int i = 0, l = 0; i < n; ++i) {
      if (ar[i] == '>') {
        for (int j = l; j <= i; ++j) {
          ans1.emplace_back(mx - (i - l) + 2 * (j - l));
          mx--;
        }
        l = i + 1;
      }
    }
    for (auto i : ans1) cout << i << ' ';
    cout << endl;
    ar[n - 1] = '<';
    vector<int> ans2;
    ans2.resize(n);
    int mn = 1;
    for (int i = 0, l = 0; i < n; ++i) {
      if (ar[i] == '<') {
        for (int j = i; j >= l; --j) {
          ans2[j] = mn;
          mn++;
        }
        l = i + 1;
      }
    }
    for (auto i : ans2) cout << i << ' ';
    cout << endl;
  }
  return 0;
}
