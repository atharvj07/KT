import java.io.*;
import java.util.*;
import java.math.*;

public class B {
	static BufferedReader in;
	static PrintWriter out;
	static StringTokenizer st;
	static Random rnd;

	String getVar(int a, int b) {
		StringBuilder res = new StringBuilder();

		while (a > 1 || b > 1) {
			if (a == b)
				return null;

			if (a > b) {
				a -= b;
				res.append('T');
			} else {
				b -= a;
				res.append('B');
			}
		}

		res.append('T');

		res.reverse();

		// out.println("DEBUG RES: " + a + " " + b);

		return res.toString();
	}

	int getErrors(String s) {
		int res = 0;
		for (int i = 0; i < s.length() - 1; i++) {
			if (s.charAt(i) == s.charAt(i + 1))
				++res;
		}
		return res;
	}

	int gcd(int a, int b) {
		if (b == 0)
			return a;
		return gcd(b, a% b);
	}

	void solve() throws IOException {
		int n = nextInt();
		int r = nextInt();

		String best = null;
		int bestErrors = Integer.MAX_VALUE;

		for (int i = 1; i <= 1; i++) {
			String[] vars = { getVar(r, i), getVar(i, r) };

			for (String var : vars) {
				if (var != null && var.length() == n) {
					int currentErrors = getErrors(var);
					if (best == null || currentErrors < bestErrors) {
						best = var;
						bestErrors = currentErrors;
					}
				}
			}
		}

		for (int i = (r / 2); i <= r; i++) {
			if (gcd(i, r) != 1)
				continue;

			String[] vars = { getVar(r, i), getVar(i, r) };

			for (String var : vars) {
				if (var != null && var.length() == n) {
					int currentErrors = getErrors(var);
					if (best == null || currentErrors < bestErrors) {
						best = var;
						bestErrors = currentErrors;
					}
				}
			}
		}

		if (best == null) {
			out.println("IMPOSSIBLE");
		} else {
			out.println(bestErrors);
			out.println(best);
		}
	}

	public static void main(String[] args) {
		new B().run();
	}

	public void run() {
		try {
			in = new BufferedReader(new InputStreamReader(System.in));
			out = new PrintWriter(System.out);

			rnd = new Random();

			solve();

			out.close();
		} catch (IOException e) {
			e.printStackTrace();
			System.exit(42);
		}
	}

	String nextToken() throws IOException {
		while (st == null || !st.hasMoreTokens()) {
			String line = in.readLine();

			if (line == null)
				return null;

			st = new StringTokenizer(line);
		}

		return st.nextToken();
	}

	int nextInt() throws IOException {
		return Integer.parseInt(nextToken());
	}

	long nextLong() throws IOException {
		return Long.parseLong(nextToken());
	}

	double nextDouble() throws IOException {
		return Double.parseDouble(nextToken());
	}
}
