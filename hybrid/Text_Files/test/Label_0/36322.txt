import java.util.ArrayList;
import java.util.HashMap;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        HashMap<Integer, Boolean> map = new HashMap<>();
        HashMap<Integer, Integer> last = new HashMap<>();
        ArrayList<Integer> arr = new ArrayList<>();
        int n = scanner.nextInt(), cont = 0, suma = 0;
        int days = 0;
        while (n-- > 0)
        {
            int e = scanner.nextInt();
            suma += e;
            if(days != -1){
                if(e > 0){
                    if(map.containsKey(e) && map.get(e) || (last.containsKey(e) && last.get(e) == days)){
                        days = -1;
                    } else{
                        map.put(e, true);
                        last.put(e, days);
                    }
                } else{
                    if(!map.containsKey(e*-1) || !map.get(e*-1)){
                        days = -1;
                    } else{
                        if(!map.containsKey(e*-1)) days = -1;
                        else map.put(e*-1, false);
                    }
                }
                if(suma == 0){
                    days++;
                    arr.add(map.size()*2);
                    map.clear(); last.clear();
                    suma = 0;
                }
            }
        }
        if(suma != 0) days = -1;
        System.out.println(days);
        if(days != -1){
            for(int num : arr){
                System.out.print(num + " ");
            }
        }
    }
}
