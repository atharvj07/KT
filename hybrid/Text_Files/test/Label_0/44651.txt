import math

def divisors(n):
	divs = [1, n]
	for i in range(2, int(math.sqrt(n)) + 1):
		if n % i == 0:
			divs.extend([i, n // i])
	return set(divs)
(l, r, x, y) = map(int, input().split())
cnt = 0
if y % x != 0:
	print(0)
	exit(0)
vv = y // x
for d in divisors(vv):
	c = vv // d
	if l <= c * x <= r and l <= d * x <= r and (math.gcd(c, d) == 1):
		cnt += 1
print(cnt)
