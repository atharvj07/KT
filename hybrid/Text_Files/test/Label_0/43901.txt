from sys import stdin, stdout
from operator import itemgetter
import sys

class UnionFind:

	def __init__(self, n):
		self.parent = list(range(n))

	def find(self, a):
		acopy = a
		while a != self.parent[a]:
			a = self.parent[a]
		while acopy != a:
			(self.parent[acopy], acopy) = (a, self.parent[acopy])
		return a

	def union(self, a, b):
		self.parent[self.find(b)] = self.find(a)

def main():
	input = sys.stdin.buffer.readline
	t = int(input())
	for _ in range(t):
		(n, m, k) = [int(x) for x in input().split()]
		uf = UnionFind(n + 1)
		stack = 0
		v = []
		for _ in range(m):
			v.append([int(x) for x in input().split()])
		v.sort(key=itemgetter(2))
		min_speed = 10000000000
		for (a, b, s) in v:
			if uf.find(a) != uf.find(b):
				uf.union(a, b)
				stack += max(0, s - k)
		if stack != 0:
			stdout.write(str(stack) + '\n')
		else:
			for (a, b, s) in v:
				min_speed = min(min_speed, abs(s - k))
			stdout.write(str(min_speed) + '\n')
main()
