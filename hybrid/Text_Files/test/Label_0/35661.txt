import java.util.*;

public class Main
{
	public static void main (String [] args)
	{
		
		Scanner sc = new Scanner(System.in);
		int total = sc.nextInt();
		int counter = 0;

		for(int i = 1; i < total; i++)
		{
			if((total - i)%i == 0)
			{
				counter++;
			}
		}
		System.out.println(counter);
	}	
}
