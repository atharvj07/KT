import sys
input = sys.stdin.buffer.readline
sys.setrecursionlimit(10 ** 7)
from collections import defaultdict
con = 10 ** 9 + 7
INF = float('inf')

def getlist():
	return list(map(int, input().split()))

class Graph(object):

	def __init__(self):
		self.graph = defaultdict(list)

	def __len__(self):
		return len(self.graph)

	def add_edge(self, a, b):
		self.graph[a].append(b)

def DFS(G, W, P, Q, node):
	for i in G.graph[node]:
		DFS(G, W, P, Q, i)
		if W[i] > 0:
			W[node] += W[i]
			P.append(i + 1)
		else:
			Q.append(i + 1)

def main():
	N = int(input())
	W = getlist()
	B = getlist()
	G = Graph()
	for i in range(N):
		(u, v) = (B[i], i)
		if u != -1:
			u -= 1
			G.add_edge(u, v)
	P = []
	Q = []
	for i in range(N):
		if B[i] == -1:
			DFS(G, W, P, Q, i)
			P.append(i + 1)
	P += Q[::-1]
	weight = sum(W)
	print(weight)
	print(*P)
import threading
threading.stack_size(1024 * 100000)
thread = threading.Thread(target=main)
thread.start()
thread.join()
