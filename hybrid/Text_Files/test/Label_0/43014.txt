n = int(input())
s = input()
b = list(map(int, input().split()))
h = [0] * n
ha = [0] * n
har = [0] * n
hard = [0] * n
if s[0] == 'h':
	h[0] = b[0]
for j in range(1, n):
	if s[j] == 'h':
		h[j] = h[j - 1] + b[j]
		ha[j] = ha[j - 1]
		har[j] = har[j - 1]
		hard[j] = hard[j - 1]
	elif s[j] == 'a':
		ha[j] = min(ha[j - 1] + b[j], h[j - 1])
		h[j] = h[j - 1]
		har[j] = har[j - 1]
		hard[j] = hard[j - 1]
	elif s[j] == 'r':
		har[j] = min(har[j - 1] + b[j], h[j - 1], ha[j - 1])
		h[j] = h[j - 1]
		ha[j] = ha[j - 1]
		hard[j] = hard[j - 1]
	elif s[j] == 'd':
		hard[j] = min(hard[j - 1] + b[j], har[j - 1], ha[j - 1], h[j - 1])
		h[j] = h[j - 1]
		ha[j] = ha[j - 1]
		har[j] = har[j - 1]
	else:
		h[j] = h[j - 1]
		ha[j] = ha[j - 1]
		har[j] = har[j - 1]
		hard[j] = hard[j - 1]
print(hard[-1])
