#include<bits/stdc++.h>
typedef long long int ll;
typedef unsigned long long int ull;
#define BIG_NUM 2000000000
#define HUGE_NUM 99999999999999999
#define MOD 1000000007
#define EPS 0.000000001
using namespace std;


int H,W;
vector<ll> A,B;

int main(){

	scanf("%d %d",&H,&W);

	ll tmp;

	for(int loop = 0; loop < H; loop++){
		scanf("%lld",&tmp);
		A.push_back(tmp);
	}

	for(int loop = 0; loop < W; loop++){
		scanf("%lld",&tmp);
		B.push_back(tmp);
	}

	sort(A.begin(),A.end());
	sort(B.begin(),B.end());

	ll ans = 0;

	int left,right,mid;
	int min_index;

	for(int i = 0; i < A.size(); i++){

		left = 0,right = B.size()-1,mid = (left+right)/2;
		min_index = B.size();

		while(left <= right){

			if(A[i] <= B[mid]){

				min_index = mid;
				right = mid-1;

			}else{

				left = mid+1;
			}
			mid = (left+right)/2;
		}
		ans += A[i]*(B.size()-min_index);
	}

	for(int i = 0; i < B.size(); i++){

		left = 0,right = A.size()-1,mid = (left+right)/2;
		min_index = A.size();

		while(left <= right){

			if(B[i] < A[mid]){

				min_index = mid;
				right = mid-1;

			}else{

				left = mid+1;
			}
			mid = (left+right)/2;
		}
		ans += B[i]*(A.size()-min_index);
	}

	printf("%lld\n",ans);

	return 0;
}
