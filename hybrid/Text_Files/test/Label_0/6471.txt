#include <bits/stdc++.h>

using namespace std;

int main(){
  long n,p;
  cin >> n>>p;
  
  int a,even=0;
  long pow=1;
  for(int i=0;i<n;i++){
    cin >> a;
     pow*=2;
    if(a%2==0) even++;
  }
  
  if(even==n&&p==0) cout << pow << endl;
  else if(even==n&&p==1) cout << 0 << endl;
  else cout << pow/2 << endl;

}
