(n, m, k) = [*map(int, input().split())]
a = [*map(int, input().split())]
a.sort()
a = a[::-1]
ans = 0
if k == 1:
	if m % 2 == 0:
		print(m // 2 * a[0] + m // 2 * a[1])
	else:
		print((m // 2 + 1) * a[0] + m // 2 * a[1])
elif k == 2:
	su = m // (k + 1)
	count = m - su
	ans += count * a[0]
	ans += su * a[1]
	print(ans)
elif m % k == 0:
	su = m // k - 1
	count = m - su
	ans += count * a[0]
	ans += su * a[1]
	print(ans)
else:
	su = m // (k + 1)
	count = m - su
	ans += count * a[0]
	ans += su * a[1]
	print(ans)
