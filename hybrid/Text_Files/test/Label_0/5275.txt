#include <bits/stdc++.h>
using namespace std;
const int N = 5e5 + 10;
const int inf = 0x3f3f3f3f;
const long long mod = 998244353;
long long p[N];
long long f[N];
long long qmi(long long a, long long b) {
  long long res = 1;
  while (b) {
    if (b & 1) {
      res = res * a % mod;
    }
    a = a * a % mod;
    b >>= 1;
  }
  return res;
}
int main() {
  ios::sync_with_stdio(false);
  int n;
  int i;
  cin >> n;
  for (i = 1; i <= n; i++) {
    cin >> p[i];
  }
  for (i = 1; i <= n; i++) {
    f[i] = (f[i - 1] + 1) * 100 % mod * qmi(p[i], mod - 2) % mod;
  }
  cout << f[n] << endl;
  return 0;
}
