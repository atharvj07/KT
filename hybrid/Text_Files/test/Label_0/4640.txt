#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t = 1;
  while (t--) {
    double n, k;
    cin >> n >> k;
    vector<double> v(n + 1);
    for (double i = 1; i <= n; i++) cin >> v[i];
    vector<double> pref(n + 1, 0);
    pref[1] = v[1];
    for (double i = 2; i <= n; i++) pref[i] += (pref[i - 1] + v[i]);
    double start = 1, end = k;
    double mx = 0;
    while (end <= n) {
      int i = start, j = end;
      while (j <= n) {
        mx = max(mx, ((pref[j] - pref[i - 1]) / (j - i + 1)));
        i++, j++;
      }
      end++;
    }
    cout << setprecision(15) << fixed << mx << endl;
  }
  return 0;
}
