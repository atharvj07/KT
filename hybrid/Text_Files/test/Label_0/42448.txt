(n, m, k) = map(int, input().split())
instanteLinhas = [0] * n
instanteColunas = [0] * m
cores = [0] * (k + 1)
for i in range(k):
	(op, lc, cor) = input().split()
	lc = int(lc)
	cor = int(cor)
	cores[i + 1] = cor
	if op == '1':
		instanteLinhas[lc - 1] = i + 1
	elif op == '2':
		instanteColunas[lc - 1] = i + 1
resp = ''
for i in range(n):
	for j in range(m):
		resp += str(cores[max(instanteLinhas[i], instanteColunas[j])])
		if j != m - 1:
			resp += ' '
	if i != n - 1:
		resp += '\n'
print(resp)
