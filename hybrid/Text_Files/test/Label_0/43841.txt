n = int(input())
blocks = []
for block in reversed(range(1, n + 1)):
	for count in range(n - block + 1):
		blocks.append([block, count + 1, False])
iteration = used = 0
while used < len(blocks):
	block_len = n
	shift = 0
	for index in range(len(blocks)):
		if not blocks[index][2] and blocks[index][0] <= block_len and (blocks[index][1] >= shift):
			block_len -= blocks[index][0]
			blocks[index][2] = True
			shift = blocks[index][0] + blocks[index][1]
			used += 1
	iteration += 1
print(iteration)
