void solve() {
    int n, k1, k2;
    cin >> n >> k1 >> k2;
    int w, b;
    cin >> w >> b;
    int x = min(k1, k2);
    int x1 = n - k1;
    int x2 = n - k2;
    k1 -= x;
    k2 -= x;
    w -= min(w, x);
    w -= min(w, k1 / 2);
    w -= min(w, k2 / 2);
    x = min(x1, x2);
    x1 -= x;
    x2 -= x;
    b -= min(b, x);
    b -= min(b, x1 / 2);
    b -= min(b, x2 / 2);
    if(w == 0 && b == 0) cout << "YES\n";
    else cout << "NO\n";
}
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    cin >> t;
    while (t--) solve();
    return 0;
}
