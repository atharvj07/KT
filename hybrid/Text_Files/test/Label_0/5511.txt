#include <bits/stdc++.h>
using namespace std;
int main() {
  long long t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    char g[n][m];
    int c = 0;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        cin >> g[i][j];
        if (g[i][j] == 'A') c++;
      }
    }
    if (!c) {
      cout << "MORTAL"
           << "\n";
      continue;
    }
    if (c == (n * m)) {
      cout << 0 << "\n";
      continue;
    }
    int f = 0;
    for (int j = 0; j < m; j++) {
      if (g[0][j] != 'A') {
        f = 1;
        break;
      }
    }
    if (!f) {
      cout << 1 << "\n";
      continue;
    }
    f = 0;
    for (int j = 0; j < m; j++) {
      if (g[n - 1][j] != 'A') {
        f = 1;
        break;
      }
    }
    if (!f) {
      cout << 1 << "\n";
      continue;
    }
    f = 0;
    for (int i = 0; i < n; i++) {
      if (g[i][0] != 'A') {
        f = 1;
        break;
      }
    }
    if (!f) {
      cout << 1 << "\n";
      continue;
    }
    f = 0;
    for (int i = 0; i < n; i++) {
      if (g[i][m - 1] != 'A') {
        f = 1;
        break;
      }
    }
    if (!f) {
      cout << 1 << "\n";
      continue;
    }
    if (g[0][0] == 'A' || g[0][m - 1] == 'A' || g[n - 1][0] == 'A' ||
        g[n - 1][m - 1] == 'A') {
      cout << 2 << "\n";
      continue;
    }
    int l = 0;
    for (int j = 0; j < m; j++) {
      int ch = 0;
      for (int i = 0; i < n; i++) {
        if (g[i][j] != 'A') ch++;
      }
      if (ch == 0) {
        l = 1;
        break;
      }
    }
    if (l) {
      cout << 2 << "\n";
      continue;
    }
    for (int i = 0; i < n; i++) {
      int ch = 0;
      for (int j = 0; j < m; j++) {
        if (g[i][j] != 'A') ch++;
      }
      if (ch == 0) {
        l = 1;
        break;
      }
    }
    if (l) {
      cout << 2 << "\n";
      continue;
    }
    int s = 0;
    for (int j = 0; j < m; j++) {
      if (g[0][j] == 'A') {
        s = 1;
        break;
      }
    }
    if (s) {
      cout << 3 << "\n";
      continue;
    }
    for (int j = 0; j < m; j++) {
      if (g[n - 1][j] == 'A') {
        s = 1;
        break;
      }
    }
    if (s) {
      cout << 3 << "\n";
      continue;
    }
    for (int i = 0; i < n; i++) {
      if (g[i][0] == 'A') {
        s = 1;
        break;
      }
    }
    if (s) {
      cout << 3 << "\n";
      continue;
    }
    for (int i = 0; i < n; i++) {
      if (g[i][m - 1] == 'A') {
        s = 1;
        break;
      }
    }
    if (s) {
      cout << 3 << "\n";
      continue;
    }
    cout << 4 << "\n";
  }
}
