from sys import stdin
n = int(input())
val = [None] + list(map(int, input().split()))
L = stdin.read().splitlines()
tree = [None] * (n + 1)
for i in range(1, n + 1):
	tree[i] = []
for (i, k) in enumerate(L):
	(p, c) = map(int, k.split())
	tree[p].append((i + 2, c))
val[1] = 0
S = [None] * n
S[0] = 1
size = 1
pos = 0
while pos < size:
	vert = S[pos]
	for k in tree[vert]:
		if val[vert] + k[1] <= val[k[0]]:
			val[k[0]] = max(0, val[vert] + k[1])
			S[size] = k[0]
			size += 1
	pos += 1
print(n - size)
