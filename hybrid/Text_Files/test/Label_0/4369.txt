#include <bits/stdc++.h>
using namespace std;
struct node {
  int a, b, c, d, e, f;
};
int main() {
  int t, l, n, m, s, c, i, j, k, h, r, a, b, d;
  cin >> t;
  while (t--) {
    cin >> n >> m;
    string ar[n];
    for (i = 0; i < n; i++) cin >> ar[i];
    int ap[n + 1][m + 1];
    for (i = 1; i <= n; i++) {
      for (j = 1; j <= m; j++)
        if (ar[i - 1][j - 1] == '1')
          ap[i][j] = 1;
        else
          ap[i][j] = 0;
    }
    vector<node> v;
    if (n & 1) {
      for (i = 1; i <= m; i++) {
        if (ap[n][i]) {
          if (i == 1) {
            ap[n - 1][i + 1] ^= 1;
            ap[n - 1][i] ^= 1;
            v.push_back({n, i, n - 1, i, n - 1, i + 1});
          } else {
            ap[n - 1][i - 1] ^= 1;
            ap[n - 1][i] ^= 1;
            v.push_back({n, i, n - 1, i, n - 1, i - 1});
          }
        }
      }
      n--;
    }
    if (m & 1) {
      for (i = 1; i <= n; i++) {
        if (ap[i][m]) {
          if (i == 1) {
            ap[i + 1][m - 1] ^= 1;
            ap[i][m - 1] ^= 1;
            v.push_back({i, m, i + 1, m - 1, i, m - 1});
          } else {
            ap[i - 1][m - 1] ^= 1;
            ap[i][m - 1] ^= 1;
            v.push_back({i, m, i - 1, m - 1, i, m - 1});
          }
        }
      }
      m--;
    }
    for (i = 1; i <= n; i += 2) {
      for (j = 1; j <= m; j += 2) {
        a = 0, b = 0, c = 0, d = 0;
        if (ap[i][j]) b ^= 1, c ^= 1, d ^= 1;
        if (ap[i + 1][j]) a ^= 1, c ^= 1, d ^= 1;
        if (ap[i][j + 1]) a ^= 1, b ^= 1, d ^= 1;
        if (ap[i + 1][j + 1]) a ^= 1, b ^= 1, c ^= 1;
        if (a) v.push_back({i + 1, j, i, j + 1, i + 1, j + 1});
        if (b) v.push_back({i, j, i, j + 1, i + 1, j + 1});
        if (c) v.push_back({i + 1, j, i, j, i + 1, j + 1});
        if (d) v.push_back({i + 1, j, i, j + 1, i, j});
      }
    }
    cout << v.size() << "\n";
    for (auto it : v) {
      cout << it.a << " " << it.b << " " << it.c << " " << it.d << " " << it.e
           << " " << it.f << "\n";
    }
  }
  return 0;
}
