#include <bits/stdc++.h>
using namespace std;
long long a[100005];
bool cmp(const long long a, const long long b) {
  return (10 - a % 10) < (10 - b % 10);
}
int main() {
  long long n, k;
  cin >> n >> k;
  for (int i = 0; i < n; i++) scanf("%I64d", &a[i]);
  sort(a, a + n, cmp);
  long long ans = 0, p = 0;
  for (int i = 0; i < n; i++) {
    long long t = 10 - a[i] % 10;
    if (k >= t && a[i] < 100) {
      k -= t;
      ans += (a[i] + t) / 10;
      p += (100 - a[i]) / 10;
      if (a[i] % 10 == 0) p--;
    } else
      ans += a[i] / 10;
  }
  ans += min(k / 10, p);
  cout << ans << endl;
  return 0;
}
