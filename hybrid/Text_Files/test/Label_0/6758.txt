#include <bits/stdc++.h>
using namespace std;
const int M = 1e9 + 7;
long long power(long long a, long long n) {
  long long res = 1;
  while (n) {
    if (n % 2) {
      res = (res * a) % M;
    }
    n >>= 1;
    a = (a * a) % M;
  }
  return res;
}
const int N = 1e5 + 5;
int f(int x) {
  int r = 0;
  int a = 1;
  while (x) {
    int d = x % 10;
    x /= 10;
    if (d) {
      r += d * a;
      a *= 10;
    }
  }
  return r;
}
void solve() {
  int a, b;
  cin >> a >> b;
  int c = a + b;
  a = f(a);
  b = f(b);
  c = f(c);
  if (a + b == c)
    cout << "YES";
  else
    cout << "NO";
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int tc = 1;
  while (tc--) {
    solve();
  }
  return 0;
}
