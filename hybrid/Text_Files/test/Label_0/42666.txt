import math
t = int(input())
arr = list(map(int, input().split(' ')))
arr = sorted(arr)
if len(arr) % 2 != 0:
	val = int(len(arr) / 2)
	print(val)
	final_str = ''
	new_arr = arr[val:len(arr) + 1]
	k = 1
	for i in range(val):
		new_arr.insert(k, arr[i])
		k += 2
	for i in range(len(new_arr)):
		final_str += str(new_arr[i])
		if i != len(new_arr) - 1:
			final_str += ' '
	print(final_str)
else:
	val = int(len(arr) / 2) - 1
	print(val)
	final_str = ''
	new_arr = arr[val:len(arr) + 1]
	k = 1
	for i in range(val):
		new_arr.insert(k, arr[i])
		k += 2
	for i in range(len(new_arr)):
		final_str += str(new_arr[i])
		if i != len(new_arr) - 1:
			final_str += ' '
	print(final_str)
