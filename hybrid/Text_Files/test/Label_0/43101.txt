from collections import deque
n = int(input())
arr = list(map(int, input().split()))
ans = [-1] * n
queue = deque()
maximum = 0
j = 0
for i in range(3 * n):
	while queue and arr[i % n] * 2 < queue[0][0]:
		[val, index] = queue.popleft()
		while j <= index and j < n:
			ans[j] = i - j
			j += 1
	while queue and arr[i % n] >= queue[-1][0]:
		queue.pop()
	queue.append([arr[i % n], i])
ans = ' '.join(map(str, ans))
print(ans)
