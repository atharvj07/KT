import sys
import math
input = sys.stdin.readline
(a, b, c) = (int(i) for i in input().split())
(x, y, z) = (int(i) for i in input().split())
req = 0
hv = 0
if a < x:
	req += x - a
else:
	hv += (a - x) // 2
if b < y:
	req += y - b
else:
	hv += (b - y) // 2
if c < z:
	req += z - c
else:
	hv += (c - z) // 2
if hv >= req:
	print('Yes')
else:
	print('No')
