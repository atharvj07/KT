#include<bits/stdc++.h>
using namespace std;
const int N=3000005;
int flag[N],ne[N][3],cnt,n,fac[N],v[N],ans[N],ss;
char s[N];
void build(){
int last1=0,last2=0;
for (int i=0;i<n;i++){
for (int j=last1;j<=last2;j++){
ne[j][0]=++cnt;
v[cnt]=v[j];
ne[j][1]=++cnt;
v[cnt]=v[j]+fac[i];
ne[j][2]=++cnt;
v[cnt]=v[j]+fac[i]*2;
}
last2+=fac[i+1];last1+=fac[i];
}
}
void down(int x){
if (flag[x]){
swap(ne[x][1],ne[x][2]);
if (ne[x][0])flag[ne[x][0]]^=1,flag[ne[x][1]]^=1,flag[ne[x][2]]^=1;
flag[x]=0;
}
}
void insert1(int x){
down(x);
swap(ne[x][0],ne[x][1]);
swap(ne[x][0],ne[x][2]);
if (ne[x][0])x=ne[x][0],insert1(x);
}
void solve(int x,int t,int d){
down(x);
if (ne[x][0])solve(ne[x][0],t,d+1),solve(ne[x][1],t+fac[d],d+1),solve(ne[x][2],t+2*fac[d],d+1);
else ans[v[x]]=t;
}
int main(){
scanf("%d",&n);
scanf("%s",s+1);
fac[0]=1;ss=0;
for (int i=1;i<=n;i++)fac[i]=fac[i-1]*3;
for (int i=0;i<n;i++)ss+=fac[i];
build();
for (int i=1;s[i];i++){
if (s[i]=='R')insert1(0);
else flag[0]^=1;
}
solve(0,0,0);
for (int i=0;i<fac[n];i++)printf("%d ",ans[i]);
}
