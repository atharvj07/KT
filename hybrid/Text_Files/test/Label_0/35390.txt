import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;
public class Main {
	public static int gcd(int a,int b) {
		if(a<b) return gcd(b,a);
		if(a%b==0) 
			return b;
		return gcd(b,a%b);
	}
	public static void main(String[] args) {
		Scanner input=new Scanner(System.in);
		int n=input.nextInt();
		int []a=new int[n];
		int []p=new int[n];
		int []b=new int[n];
		for(int i=0;i<n;i++){
			a[i]=input.nextInt();
			p[i]=a[i];
		}
		Arrays.sort(p);
		for(int i=0;i<n;i++) {
			for(int j=0;j<n;j++) {
				if(a[i]==p[j]) {
					b[i]=p[(j+1)%n];
					break;
				}
			}
		}
		for(int i=0;i<n;i++) {
			System.out.printf("%d%c",b[i],i==n-1?'\n':' ');
		}

	}
}
