#include <bits/stdc++.h>
using namespace std;
const int maxn = 300005;
int dp[maxn][30], d[maxn];
char a[maxn];
vector<int> E[maxn];
queue<int> q;
int main() {
  int n, m, ans = 0, x, y;
  cin >> n >> m;
  scanf("%s", a + 1);
  memset(d, 0, sizeof d);
  for (int i = 0; i < m; i++) {
    scanf("%d%d", &x, &y);
    E[x].push_back(y);
    d[y]++;
  }
  for (int i = 1; i <= n; i++) {
    if (!d[i]) {
      q.push(i);
      dp[i][a[i] - 'a'] = 1;
    }
  }
  while (!q.empty()) {
    int fr = q.front();
    q.pop();
    for (int i = 0; i < E[fr].size(); i++) {
      int now = E[fr][i];
      for (char j = 'a'; j <= 'z'; j++) {
        dp[now][j - 'a'] =
            max(dp[now][j - 'a'], dp[fr][j - 'a'] + (j == a[now]));
        ans = max(dp[now][j - 'a'], ans);
      }
      d[now]--;
      if (!d[now]) q.push(now);
    }
  }
  for (int i = 1; i <= n; i++) {
    if (d[i]) {
      cout << "-1";
      return 0;
    }
  }
  cout << ans;
}
