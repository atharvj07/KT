def calc(a):
	cost = 0
	index = dd(list)
	n = len(a)
	for i in range(n):
		index[a[i]].append(i)
	n //= 2
	for i in range(1, n + 1):
		d = index[i][1] - index[i][0]
		cost += (n - i) * abs(d + i - n)
	return cost

def solve():
	n = geta()
	diff = n - 1
	ans = [0] * (2 * n)
	index = 0
	for i in range(1, n, 2):
		ans[index] = i
		ans[index + diff] = i
		diff -= 2
		while index < 2 * n and ans[index]:
			index += 1
		if index == 2 * n:
			break
	index = n
	diff = n - 2
	for i in range(2, n, 2):
		ans[index] = i
		ans[index + diff] = i
		diff -= 2
		while index < 2 * n and ans[index]:
			index += 1
	for i in range(2 * n):
		if not ans[i]:
			ans[i] = n
	print(*ans)
solve()
