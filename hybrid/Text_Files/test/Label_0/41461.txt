import math
for p in range(int(input())):
	num = int(input())
	b = []
	a = num
	for i in range(2, int(math.sqrt(num)) + 1):
		if len(b) < 2:
			if a % i == 0:
				b.append(int(i))
				a /= i
		else:
			break
	if a not in b and len(b) == 2:
		print('YES')
		print(str(b[0]) + ' ' + str(b[1]) + ' ' + str(int(a)))
	else:
		print('NO')
