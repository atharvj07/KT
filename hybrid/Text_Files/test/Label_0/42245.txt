from functools import reduce
from itertools import islice, product
from operator import mul

def prod(iterable, start=1):
	return reduce(mul, iterable, start)

def isqrt(n):
	(x, px, nx) = (0, 0, n)
	while nx not in (x, px):
		(px, x) = (x, nx)
		nx = (x + n // x) // 2
	return min(x, nx)

def relevant_factors(n):
	for i in range(2, 2000000):
		if i ** 2 > n:
			break
		c = 0
		(q, r) = divmod(n, i)
		while not r:
			c += 1
			n = q
			(q, r) = divmod(n, i)
		if c:
			yield (i, c)
	if n > 1:
		sn = isqrt(n)
		if sn * sn == n:
			yield (sn, 2)
		elif sn * (sn + 1) == n:
			yield ((sn, 1), (sn + 1, 1))
		else:
			yield (n, 1)

def divisors(fp):
	return sorted((prod(t) for t in product(*[powers(f, e + 1) for (f, e) in fp])))

def powers(b, e):
	k = 1
	for _ in range(e):
		yield k
		k *= b

def tables(n):
	if n == 1:
		return [(1, 1)]
	t = set()
	n6 = 6 * n
	for a in divisors(relevant_factors(n6)):
		(q, r) = divmod(n6, a * (a + 1))
		if r:
			continue
		(b, r) = divmod(q + a - 1, 3)
		if r:
			continue
		if b < a:
			continue
		t.add((a, b))
		t.add((b, a))
	return sorted(t)
n = int(input())
ts = tables(n)
print(len(ts))
for t in ts:
	print('{} {}'.format(*t))
