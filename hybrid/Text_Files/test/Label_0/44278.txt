from collections import defaultdict
mod = 10 ** 9 + 7
for _ in range(int(input())):
	n = int(input())
	B = list(map(int, input().split()))
	dp = defaultdict(int)
	(cur, ans) = (0, 0)
	for (i, b) in enumerate(B):
		v = 1 if not i else (ans - dp[cur]) % mod
		ans = (ans + v) % mod
		dp[cur] = (dp[cur] + v) % mod
		cur += b
	print(ans)
