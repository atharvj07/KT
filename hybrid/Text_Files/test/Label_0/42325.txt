(n, m) = map(int, input().split())
L1 = []
L2 = []
L3 = []
L4 = []
for i in range(1, m + 1):
	if i <= 2 * n:
		if i % 2 == 1:
			L1.append(i)
		else:
			L4.append(i)
	elif i % 2 == 1:
		L2.append(i)
	else:
		L3.append(i)
L1.reverse()
L2.reverse()
L3.reverse()
L4.reverse()
ans = []
while L1 or L2 or L3 or L4:
	if L2:
		ans.append(L2.pop())
	if L1:
		ans.append(L1.pop())
	if L3:
		ans.append(L3.pop())
	if L4:
		ans.append(L4.pop())
print(*ans)
