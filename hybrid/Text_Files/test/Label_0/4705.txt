#include <bits/stdc++.h>
using namespace std;
signed long long int d[100000 + 5];
int main() {
  int n, f = 0, ans = 0;
  signed long long int k;
  cin >> n >> k;
  for (int i = 1; i <= n; ++i) {
    cin >> d[i];
  }
  sort(d + 1, d + n + 1);
  int s = 1, t = 1;
  signed long long int used = 0;
  while (t <= n && s <= n) {
    if (used <= k) {
      if (f < t - s + 1) {
        f = t - s + 1, ans = t;
      }
      t++;
      used += (t - s) * (d[t] - d[t - 1]);
    } else {
      used -= d[t] - d[s];
      ++s;
    }
  }
  cout << f << ' ' << d[ans];
}
