import sys
input = sys.stdin.readline
(n, m, k) = map(int, input().split())
if k > 4 * (n * m) - 2 * n - 2 * m:
	print('NO')
else:
	count = 0
	ans = []
	if m == 1:
		if k > 0:
			ans.append((min(n - 1, k), 'D'))
			k -= min(n - 1, k)
			count += 1
		if k > 0:
			ans.append((min(n - 1, k), 'U'))
			k -= min(n - 1, k)
			count += 1
	if n == 1:
		if k > 0:
			ans.append((min(m - 1, k), 'R'))
			k -= min(m - 1, k)
			count += 1
		if k > 0:
			ans.append((min(m - 1, k), 'L'))
			k -= min(m - 1, k)
			count += 1
	else:
		while n > 1:
			if k > 0:
				ans.append((min(m - 1, k), 'R'))
				k -= min(m - 1, k)
				count += 1
			if k > 2:
				if k >= (m - 1) * 3:
					ans.append((m - 1, 'DUL'))
					k -= (m - 1) * 3
					count += 1
				else:
					ans.append((k // 3, 'DUL'))
					k -= k // 3 * 3
					count += 1
					if k > 0:
						if k == 1:
							ans.append((1, 'D'))
							count += 1
							k -= 1
						elif k == 2:
							ans.append((1, 'DU'))
							count += 1
							k -= 2
			elif k == 1:
				ans.append((1, 'D'))
				count += 1
				k -= 1
			elif k == 2:
				ans.append((1, 'DU'))
				count += 1
				k -= 2
			if k > 0:
				ans.append((1, 'D'))
				k -= 1
				count += 1
			n -= 1
		if k > 0:
			ans.append((min(m - 1, k), 'R'))
			k -= min(m - 1, k)
			count += 1
		if k > 0:
			ans.append((min(m - 1, k), 'L'))
			k -= min(m - 1, k)
			count += 1
		if k > 0:
			ans.append((k, 'U'))
			k -= k
			count += 1
	print('YES')
	print(count)
	for (j, jk) in ans:
		print(j, jk)
