import sys

def solve(n, string):
	if int(string[n - 1]) % 2 == 0:
		print(0)
		return
	elif int(string[0]) % 2 == 0:
		print(1)
		return
	else:
		for i in range(1, n - 1):
			if int(string[i]) % 2 == 0:
				print(2)
				return
		print(-1)
		return

def main():
	test_case = int(sys.stdin.readline())
	for tc in range(test_case):
		string = sys.stdin.readline().strip('\n')
		n = len(string)
		solve(n, string)
main()
