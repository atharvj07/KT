    class Pair implements Comparable<Pair> {
        int x;
        int y;

        Pair(int i, int j) {
            x = i;
            y = j;
        }

        @Override
        public boolean equals(Object o) {
            if (this == o)
                return true;
            if (o == null || getClass() != o.getClass())
                return false;
            Pair p = (Pair) o;
            return (x == p.x && y == p.y);
        }

        @Override
        public int compareTo(Pair that) {
            if (x == that.x) {
                return Double.compare(y, that.y);
            }
            return Double.compare(x, that.x);
        }
    }




    class quemin {
        int[] stack1;
        int[] min1;
        int sz1;
        int stack2[];
        int min2[];
        int sz2;

        quemin(int n) {
            stack1 = new int[n];
            min1 = new int[n];
            stack2 = new int[n];
            min2 = new int[n];
            sz1 = sz2 = 0;
        }

        void add(int x) {
            stack1[sz1] = x;
            min1[sz1] = sz1 > 0 ? min(x, min1[sz1 - 1]) : x;
            sz1++;
        }

        void pop() {
            if (sz2 > 0) {
                sz2--;
            } else {
                while (sz1 > 0) {
                    stack2[sz2] = stack1[sz1 - 1];
                    min2[sz2] = sz2 > 0 ? min(stack2[sz2], min2[sz2 - 1]) : stack2[sz2];
                    sz2++;
                    sz1--;
                }
                sz2--;
            }
        }

        int getmin1() {
            if (sz1 > 0) {
                return min1[sz1 - 1];
            }
            return Integer.MAX_VALUE;
        }

        int getmin2() {
            if (sz2 > 0) {
                return min2[sz2 - 1];
            }
            return Integer.MAX_VALUE;
        }

        int getmin() {
            return min(getmin1(), getmin2());
        }
    }

    private void solveL() throws IOException {
        int n = in.nextInt(), m = in.nextInt();
        int[][] tab = new int[n][m];
        boolean[][] used = new boolean[n][m];
        long[][] ans = new long[n][m];
        for(int i = 0; i < n; i++){
            for(int j = 0; j < m; j++){
                tab[i][j] = in.nextInt();
            }
        }
        ArrayDeque<Pair> bf = new ArrayDeque<>();
        bf.add(new Pair(0, 0));
        boolean f = false;
        while(!bf.isEmpty() &&!f){
            Pair cur = bf.poll();
            for(int i = cur.x + 1; i < n &&!f; i++){
                if(tab[i][cur.y] == 2){
                    out.print(ans[cur.x][cur.y] + 1);
                    f = true;
                    break;
                }
                if(i == n - 1 || tab[i + 1][cur.y] == 1){
                    if(!used[i][cur.y]){
                        ans[i][cur.y] = ans[cur.x][cur.y] + 1;
                        bf.add(new Pair(i, cur.y));
                    }
                    break;
                }
            }
            for(int i = cur.x - 1; i >= 0 && !f; i--){
                if(tab[i][cur.y] == 2){
                    out.print(ans[cur.x][cur.y] + 1);
                    f = true;
                    break;
                }
                if(i == 0 || tab[i - 1][cur.y] == 1){
                    if(!used[i][cur.y]){
                        ans[i][cur.y] = ans[cur.x][cur.y] + 1;
                        bf.add(new Pair(i, cur.y));
                    }
                    break;
                }
            }
            for(int j = cur.y + 1; j < n && !f; j++){
                if(tab[cur.x][j] == 2){
                    out.print(ans[cur.x][j] + 1);
                    f = true;
                    break;
                }
                if( j == n - 1 || tab[cur.x][j + 1] == 1){
                    if(!used[cur.x][j]){
                        ans[cur.x][j] = ans[cur.x][cur.y] + 1;
                        bf.add(new Pair(cur.x, j));
                    }
                    break;
                }
            }
            for(int j = cur.y - 1; j >= 0 && !f; j--){
                if(tab[cur.x][j] == 2){
                    out.print(ans[cur.x][j] + 1);
                    f = true;
                    break;
                }
                if( j == 0 || tab[cur.x][j - 1] == 1){
                    if(!used[cur.x][j]){
                        ans[cur.x][j] = ans[cur.x][cur.y] + 1;
                        bf.add(new Pair(cur.x, j));
                    }
                    break;
                }
            }
            used[cur.x][cur.y] = true;
        }

    }



    class FastScanner {
        StringTokenizer st;
        BufferedReader br;

        FastScanner(InputStream s){
            br = new BufferedReader(new InputStreamReader(s));
        }

        String next() throws IOException {
            while(st == null || !st.hasMoreTokens()){
                st = new StringTokenizer(br.readLine());
            }
            return st.nextToken();
        }
        boolean hasNext() throws IOException {
            return br.ready() || (st != null && st.hasMoreTokens());
        }

        int nextInt() throws IOException {
            return Integer.parseInt(next());
        }

        long nextLong() throws IOException {
            return Long.parseLong(next());
        }

        double nextDouble() throws IOException {
            return Double.parseDouble(next().replace(',', '.'));
        }

        boolean hasNextLine() throws IOException {
            return br.ready();
        }

        String nextLine() throws IOException {
            return br.readLine();
        }

    }
