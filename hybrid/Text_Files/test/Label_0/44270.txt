import math

def calc(k):
	if k == 0:
		return 0
	result = 0
	mod = 1
	lower_odd = 1
	lower_even = 2
	length = 1
	while k > 0:
		if mod == 1:
			upper_odd = lower_odd + 2 * (length - 1)
			result += (upper_odd + lower_odd) * length // 2
			lower_odd = upper_odd + 2
		else:
			upper_even = lower_even + 2 * (length - 1)
			result += (upper_even + lower_even) * length // 2
			lower_even = upper_even + 2
		mod ^= 1
		k -= length
		length *= 2
		length = min(length, k)
	return result
(l, r) = map(int, input().split())
result = (calc(r) - calc(l - 1)) % (10 ** 9 + 7)
print(result)
