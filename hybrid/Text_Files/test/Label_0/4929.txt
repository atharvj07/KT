#include <bits/stdc++.h>
using namespace std;
const int MAX = 1000010;
const int MOD = 1000000007;
const int INF = 1e8;
const long long LLINF = 0x3f3f3f3f3f3f3f3f;
const long double EPS = 1e-7;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  ;
  int n, k;
  cin >> n >> k;
  vector<bool> foi(n, false);
  vector<int> res(n, -1), b(n);
  for (int i = 0; i < (int)n; i++) cin >> b[i];
  for (int i = 0; i < (int)n; i++) {
    int val = n;
    for (int j = 0; j < (int)n; j++) {
      if (b[j] == 0 and !foi[j]) {
        val = j;
        break;
      }
    }
    foi[val] = true;
    res[i] = val;
    for (int j = 0; j <= val - k; j++) {
      b[j]--;
    }
  }
  for (auto r : res) cout << r + 1 << " ";
  cout << '\n';
  return 0;
}
