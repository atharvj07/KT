#include <bits/stdc++.h>
using namespace std;
int n, m, x, tot[500], size[500], dp[500][10000], y, t, maxn, sum[2000], ans;
vector<int> mp[500];
bool cmp(int x, int y) { return x > y; }
int dfs(int x, int y) {
  if (y >= sum[x]) return dp[x][y] = tot[x];
  if (y == 0) return dp[x][y] = 0;
  if (dp[x][y] != -1) return dp[x][y];
  dp[x][y] = 0;
  if (x == maxn) {
    int ll = mp[x].size();
    int r = min(y, ll);
    for (int i = 0; i < r; i++) {
      dp[x][y] += mp[x][i];
    }
    return dp[x][y];
  } else {
    int res = 0;
    dp[x][y] = dfs(x + 1, y * 2);
    int ll = mp[x].size();
    int r = min(y, ll);
    for (int i = 0; i < r; i++) {
      res += mp[x][i];
      dp[x][y] = max(dp[x][y], res + dfs(x + 1, (y - i - 1) * 2));
    }
    return dp[x][y];
  }
}
int main() {
  memset(dp, -1, sizeof(dp));
  cin >> n >> t;
  for (int i = 1; i <= n; i++) {
    scanf("%d%d", &x, &y);
    int tt = t - x;
    maxn = max(maxn, tt);
    if (tt < 0) continue;
    m++;
    mp[tt].push_back(y);
    size[tt]++;
    tot[tt] += y;
  }
  for (int i = 0; i <= maxn; i++) sort(mp[i].begin(), mp[i].end(), cmp);
  for (int i = maxn; i >= 0; i--) {
    sum[i] = sum[i + 1] + size[i], tot[i] += tot[i + 1];
  }
  ans = dfs(0, 1);
  cout << ans;
}
