import sys
t = 'NO'
r = []
for i in range(4):
	r.append(list(map(int, sys.stdin.readline().split())))
for i in range(4):
	for j in range(3):
		if r[i][j] == 1:
			if j == 0:
				if r[i][3] == 1:
					t = 'YES'
					break
				if r[i - 1][3] == 1:
					t = 'YES'
					break
			elif j == 1:
				if r[i][3] == 1:
					t = 'YES'
					break
				if r[(i + 2) % 4][3] == 1:
					t = 'YES'
					break
			else:
				if r[(i + 1) % 4][3] == 1:
					t = 'YES'
					break
				if r[i][3] == 1:
					t = 'YES'
					break
sys.stdout.write(t)
