#include <bits/stdc++.h>
using namespace std;
int n, m, k, ans;
int dp[2010][2010], mp[15][15];
bool check(int sta1, int sta2, int node1, int node2) {
  if (!mp[node1][node2]) {
    return 0;
  }
  if (!(sta1 & (1 << (node1)))) {
    return 0;
  }
  if (sta1 & (1 << (node2))) {
    return 0;
  }
  if (((sta2 & (~(1 << (node1)))) | (1 << (node2))) >> (node2 + 1)) {
    return 0;
  }
  return 1;
}
int cal(int x) {
  int ret = 0;
  while (x) {
    x -= x & -x;
    ret++;
  }
  return ret;
}
int main() {
  scanf("%d%d%d", &n, &m, &k);
  for (int i = 1; i <= m; i++) {
    int x, y;
    scanf("%d%d", &x, &y);
    x--, y--;
    mp[x][y] = mp[y][x] = 1;
    dp[(1 << (x)) | (1 << (y))][(1 << (x)) | (1 << (y))] = 1;
  }
  for (int i = 0; i <= (1 << (n)) - 1; i++) {
    for (int j = 0; j <= (1 << (n)) - 1; j++) {
      if ((i & j) != j || dp[i][j] == 0) {
        continue;
      }
      for (int u = 0; u < n; u++) {
        for (int v = 0; v < n; v++) {
          if (check(i, j, u, v)) {
            dp[i | (1 << (v))][(j & (~(1 << (u))) | (1 << (v)))] += dp[i][j];
          }
        }
      }
    }
  }
  int ans = 0;
  for (int i = 0; i <= (1 << (n)) - 1; i++) {
    if (cal(i) == k) {
      ans += dp[(1 << (n)) - 1][i];
    }
  }
  printf("%d\n", ans);
  return 0;
}
