import sys
input = sys.stdin.buffer.readline

def solve(a, b, c, m):
	n = a + b + c
	maxpos = a - 1 + (b - 1) + (c - 1)
	maxi = max(a, b, c)
	mini = min(a, b, c)
	medi = n - maxi - mini
	minpos = maxi - mini - medi - 1
	if m > n - 1 or m > maxpos or minpos > m:
		return 'NO'
	else:
		return 'YES'
T = int(input())
for i in range(T):
	(a, b, c, m) = map(int, input().split())
	print(solve(a, b, c, m))
