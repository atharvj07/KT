#include <bits/stdc++.h>
using namespace std;
const int MAXN = 100010;
int n, m, x, sol, r, s;
int c[MAXN];
vector<int> p;
int main() {
  cin >> n >> m;
  p.push_back(0);
  for (int i = 0; i < n; i++) {
    cin >> x;
    p.push_back(x);
  }
  p.push_back(m);
  c[0] = 0;
  for (int i = 1; i < p.size(); i++) {
    if (i % 2)
      c[i] = c[i - 1] + p[i] - p[i - 1];
    else
      c[i] = c[i - 1];
  }
  sol = c[p.size() - 1];
  for (int i = 1; i < p.size(); i += 2) {
    r = -1;
    s = -1;
    if (p[i] - p[i - 1] != 1)
      r = c[i] - 1 + m - p[i] - (c[p.size() - 1] - c[i - 1]);
    if (i + 1 < p.size() && p[i + 1] - p[i] != 1)
      s = c[i] + p[i + 1] - p[i] - 1 + m - p[i + 1] + -(c[p.size() - 1] - c[i]);
    sol = max(sol, max(r, s));
  }
  cout << sol << "\n";
  return 0;
}
