#include <bits/stdc++.h>
using namespace std;
int main(int argc, char *argv[]) {
  static char a[1024][1024];
  int n, m;
  cin >> n >> m;
  int ans = -1;
  for (int k = 0; k < m; k++) {
    int x, y;
    cin >> x >> y;
    a[x][y] = 1;
    if (ans < 0) {
      bool found = false;
      for (int i = max(1, x - 2); i <= min(x, n - 2); i++) {
        for (int j = max(1, y - 2); j <= min(y, n - 2); j++) {
          int ok = true;
          for (int di = 0; di < 3 && ok; di++)
            for (int dj = 0; dj < 3; dj++)
              if (a[i + di][j + dj] == 0) {
                ok = false;
                break;
              }
          if (ok) {
            found = true;
            break;
          }
        }
        if (found) break;
      }
      if (found) ans = k + 1;
    }
  }
  cout << ans << endl;
  return 0;
}
