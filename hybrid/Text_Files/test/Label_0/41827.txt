n = int(input())
a = list(map(int, input().split()))
(x, y) = (int(1000000000.0), -1)
for i in range(n - 1):
	d = abs(a[i] - a[i + 1])
	if d == 0:
		print('NO')
		exit()
	if d > 1:
		if y == -1:
			y = d
		if y != -1 and y != d:
			print('NO')
			exit()
if y == -1:
	y = int(1000000000.0)
for i in range(n - 1):
	if abs(a[i] - a[i + 1]) == 1:
		if a[i] % y == 0 and a[i + 1] % y == 1 and (a[i + 1] > a[i]) or (a[i] % y == 1 and a[i + 1] % y == 0 and (a[i + 1] < a[i])):
			print('NO')
			exit()
print('YES')
print(x, y)
