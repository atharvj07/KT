#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
  cin.tie(0);
  ios::sync_with_stdio(false);
  int n;
  cin >> n;
  ll ans = 0;
  vector<ll> a(n);
  for (int i = 0; i < (n); i++) {
    cin >> a[i];
    a[i] = abs(a[i]);
  }
  sort(a.begin(), a.end());
  int j = 0;
  for (int i = 0; i < (n - 1); i++) {
    ll x = a[i];
    j = max(j, i);
    if (j != n - 1)
      while (a[j + 1] <= 2 * x) {
        j++;
        if (j == n - 1) break;
      }
    ans += (ll)(j - i);
  }
  cout << ans << endl;
  return 0;
}
