import java.io.*;
import java.util.*;

public class CodeForces {
    static boolean ONLINE_JUDGE = System.getProperty("ONLINE_JUDGE") != null;

    void runCase(int caseNum) throws IOException {
        int n = nextInt();
        int m = nextInt();

        int[][] table = new int[n][m];

        for (int i = 0; i < n; ++i)
            for (int j = 0; j < m; ++j)
                table[i][j] = nextInt();
        boolean[] rows = new boolean[n];
        boolean[] cols = new boolean[m];

        while (true) {
            for (int i = 0; i < n; ++i) {
                int sum = 0;
                for (int j = 0; j < m; ++j) {
                    sum += table[i][j];
                }
                if (sum < 0) {
                    rows[i] = rows[i] ? false : true;
                    for (int j = 0; j < m; ++j)
                        table[i][j] = -table[i][j];
                }
            }
            boolean OK = true;

            for (int i = 0; i < m; ++i) {
                int sum = 0;
                for (int j = 0; j < n; ++j)
                    sum += table[j][i];
                if (sum < 0) {
                    OK = false;
                    cols[i] = cols[i] ? false : true;
                    for (int j = 0; j < n; ++j)
                        table[j][i] = -table[j][i];
                }
            }

            if (OK)
                break;
        }
        int cnt = 0;
        for (int i = 0; i < n; ++i)
            if (rows[i])
                ++cnt;
        out.print(cnt);
        for (int i = 0; i < n; ++i)
            if (rows[i])
                out.print(" " + (i + 1));
        out.println();
        cnt = 0;
        for (int i = 0; i < m; ++i)
            if (cols[i])
                ++cnt;
        out.print(cnt);
        for (int i = 0; i < m; ++i)
            if (cols[i])
                out.print(" " + (i + 1));
        out.println();
    }

    public static void main(String[] args) throws IOException {
        if (ONLINE_JUDGE){
            in = new BufferedReader(new InputStreamReader(System.in));
            out = new PrintWriter(System.out);
        }else{
            in = new BufferedReader(new FileReader("input.txt"));
            out = new PrintWriter("output.txt");
        }
        new CodeForces().runIt();
        out.flush();
        out.close();
        return;
    }

    static BufferedReader in;
    private StringTokenizer st;
    static PrintWriter out;

    String next() throws IOException {
        while (!st.hasMoreTokens()) {
            String line = in.readLine();
            if (line == null) {
                return null;
            }
            st = new StringTokenizer(line, " ");
        }
        return st.nextToken();
    }

    int nextInt() throws IOException {
        return Integer.parseInt(next());
    }

    double nextDouble() throws IOException {
        return Double.parseDouble(next());
    }

    long nextLong() throws IOException {
        return Long.parseLong(next());
    }


    void runIt() throws IOException {
        st = new StringTokenizer("");

//        int N = nextInt();
//        for (int i = 0; i < N; i++) {
//            runCase(i + 1);
//        }
        runCase(0);

        out.flush();
    }

}
