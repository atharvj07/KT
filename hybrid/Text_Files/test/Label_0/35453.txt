    void dfs(int u, int fa){
        for(int i = 2; i <= n; i++)dp[u][i] = 0;
        dp[u][0] = 1;
        siz[u] = 1;
        for(int i = head[u]; i!=-1; i = edge[i].nex){
            int v = edge[i].to; if(v == fa)continue;
            dfs(v, u);
            siz[u] += siz[v];
            for(int j = n-1; j >= 0; j--)
                if(dp[u][j]>0)
                    dp[u][j+siz[v]] = 1;
        }
        int s = n-siz[u];
        for(int i = n-1; i >= 0; i--){
            if(dp[u][i]>0)dp[u][i+s] = 1;
        }
        for(int i = 1; i < n-1; i++)
            if(dp[u][i]>0)ans[i] = 1;
    }

    int upper_bound(int[] A, int l, int r, int val) {// upper_bound(A+l,A+r,val)-A;
        int pos = r;
        r--;
        while (l <= r) {
            int mid = (l + r) >> 1;
            if (A[mid] <= val) {
                l = mid + 1;
            } else {
                pos = mid;
                r = mid - 1;
            }
        }
        return pos;
    }

    int lower_bound(int[] A, int l, int r, int val) {// upper_bound(A+l,A+r,val)-A;
        int pos = r;
        r--;

            if (A[mid] < val) {

            } else {
                pos = mid;

            }
        }
        return pos;
    }

    int Pow(int x, int y) {
        int ans = 1;
        while (y > 0) {
            if ((y & 1) > 0)
                ans *= x;
            y >>= 1;
            x = x * x;
        }
        return ans;
    }

    int gcd(int x, int y) {
        if (x > y) {
            int tmp = x;
            x = y;
            y = tmp;
        }
        while (x > 0) {
            y %= x;
            int tmp = x;
            x = y;
            y = tmp;
        }
        return y;
    }
