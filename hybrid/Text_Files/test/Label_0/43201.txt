def main():
	t = int(input())
	for i in range(t):
		(n, k) = map(int, input().split())
		c = input().split()
		dic = dict()
		for c_i in c:
			if dic.get(c_i) is None:
				dic[c_i] = 1
			else:
				dic[c_i] += 1
		ans = []
		for color in dic.keys():
			temp = color
			cont = 0
			j = 0
			while j < n:
				if c[j] != temp:
					cont += 1
					j = j + k - 1
				j += 1
			ans.append(cont)
		print(min(ans))
main()
