#include <bits/stdc++.h>
using namespace std;
long long int n, x, y, a, b, c;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n;
  long long int arr[n];
  for (long long int i = 0; i < n; i++) cin >> arr[i];
  sort(arr, arr + n);
  a = arr[n - 2] - arr[0];
  b = arr[n - 1] - arr[1];
  if (n == 2)
    cout << 0;
  else
    cout << min(a, b);
  return 0;
}
