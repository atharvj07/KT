#include <bits/stdc++.h>
using namespace std;
string s, s1, s2, s3;
int main() {
  cin >> s;
  int k = s.size(), a = 0;
  for (int i = a; i < k; i++) {
    if (s[i] == '|')
      s1.push_back(s[i]);
    else {
      a = i;
      break;
    }
  }
  for (int i = a + 1; i < k; i++) {
    if (s[i] == '|')
      s2.push_back(s[i]);
    else {
      a = i;
      break;
    }
  }
  for (int i = a + 1; i < k; i++) s3.push_back(s[i]);
  int k1, k2, k3;
  k1 = s1.size();
  k2 = s2.size();
  k3 = s3.size();
  if ((k1 + k2) == k3) {
    cout << s;
    return 0;
  }
  if ((k1 + k2) > k3) {
    if (k1 > k2)
      k1--;
    else
      k2--;
    k3++;
  } else {
    k3--;
    k1++;
  }
  if ((k1 + k2) != k3) {
    cout << "Impossible";
    return 0;
  }
  for (int i = 0; i < k1; i++) cout << "|";
  cout << "+";
  for (int i = 0; i < k2; i++) cout << "|";
  cout << "=";
  for (int i = 0; i < k3; i++) cout << "|";
  return 0;
}
