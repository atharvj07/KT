#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 105;
const double eps = 1e-5;
int s,w[maxn],t[maxn];
double r;
bool vis[maxn];
double ans;
double li[maxn],ri[maxn],val[maxn];
void init() {
	memset(vis,false,sizeof(vis));
	memset(w,0,sizeof(w));
	memset(t,0,sizeof(t));
}
void cal(int num) {
	memset(li,0,sizeof(li));
	memset(ri,0,sizeof(ri));
	memset(val,0,sizeof(val));
	for(int i = num - 1;i >= 1;i--) {
		if(t[i] == -1) {
			val[i] = val[2*i] + val[2*i+1];
			double L = val[2*i+1]/val[i],R = val[2*i]/val[i];
			li[i] = min(li[2*i]-L,li[2*i+1]+R);
			ri[i] = max(ri[2*i+1]+R,ri[2*i]-L);
		}
		else if(t[i]) val[i] = w[t[i]];
	}
	double temp = ri[1] - li[1];
	if(temp - r < eps && temp > ans) ans = temp;
}
void dfs(int cur,int cnt,int now) {
	if(cnt == s) {
		cal(cur);
		return;	
	}
	if(t[cur/2] != -1) dfs(cur+1,cnt,now);
	else {
		if(s - cnt > now) {
			t[cur] = -1;
			dfs(cur+1,cnt,now+1);
			t[cur] = 0;
		}
		if(now == 1 && s - cnt > now) return; 
		for(int i = 1;i <= s;i++) {
			if(!vis[i]) {
				vis[i] = true;
				t[cur] = i;
				dfs(cur+1,cnt+1,now-1);
				vis[i] = false;
				t[cur] = 0;
			}
		}
	}
}
int main() {
	ios::sync_with_stdio(false);
	cin.tie(0);
	int T;
	scanf("%d",&T);
	while(T--) {
		scanf("%lf %d",&r,&s);
		init();
		for(int i = 1;i <= s;i++) scanf("%d",&w[i]);
		if(s == 1) {printf("0.00\n");continue;}
		ans = -1;
		t[1] = -1;
		dfs(2,0,2);
		if(ans == -1) printf("-1\n");
		else printf("%.10lf\n",ans);
	}
	return 0;
}
