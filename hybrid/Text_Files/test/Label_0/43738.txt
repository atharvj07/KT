n = int(input())
numbers = map(int, input().split())
indexed_numbers = zip(numbers, range(n * 2))
indexed_numbers = list(sorted(indexed_numbers, key=lambda x: x[0], reverse=True))
cur_a = 0
cur_b = 0
total = 0
while len(indexed_numbers) > 0:
	next_a = indexed_numbers.pop(-1)[1]
	next_b = indexed_numbers.pop(-1)[1]
	total += min(abs(cur_a - next_a) + abs(cur_b - next_b), abs(cur_a - next_b) + abs(cur_b - next_a))
	cur_a = next_a
	cur_b = next_b
print(total)
