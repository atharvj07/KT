from sys import stdin
input = stdin.readline
t = int(input())
for _ in range(t):
	n = int(input())
	arr = input().rstrip()
	ch = []
	for i in range(len(arr)):
		if arr[i] == '*':
			ch.append(i)
	if ch:
		mid = ch[len(ch) // 2]
		res = 0
		check = 0
		for i in range(mid + 1, n):
			if arr[i] == '*':
				check += 1
				res += i - mid - check
		check = 0
		for i in range(mid - 1, -1, -1):
			if arr[i] == '*':
				check += 1
				res += mid - i - check
		print(res)
	else:
		print(0)
