(n, m) = map(int, input().split())
a = []
for i in range(n):
	a.append(list(map(int, input().split())))
x = 0
c = [1] * (n - 1)
for i in range(n - 1):
	x = x ^ a[i][0]
f = 0
for j in range(m):
	if a[n - 1][j] != x:
		c.append(j + 1)
		f = 1
		break
if f == 1:
	print('TAK')
	print(*c)
else:
	r = 0
	for i in range(n - 1):
		for j in range(1, m):
			if a[i][j] != a[i][0]:
				f = 1
				r = i
				c[i] = j + 1
				break
		if f == 1:
			break
	if f == 1:
		x = 0
		for i in range(n - 1):
			if i == r:
				x = x ^ a[i][c[i] - 1]
			else:
				x = x ^ a[i][0]
		for j in range(m):
			if a[n - 1][j] != x:
				c.append(j + 1)
				break
		print('TAK')
		print(*c)
	else:
		print('NIE')
