import java.util.*;

public class A {
	
	public static void main(String[]args) {
		
		Scanner scanner = new Scanner(System.in);
		
		int t = scanner.nextInt();
		
		while(t>0) {
			
			int[] arr = new int[3];
			for(int i = 0; i < 3; i++) {
				arr[i] = scanner.nextInt();
			}
			Arrays.sort(arr);
			
			System.out.println(solve(arr));
			t--;
			
			
		}
		
		scanner.close();
	}
	
	public static long solve(int[] arr) {
		
		int k = arr[2] - arr[0];
		
		double ans = Math.pow(k, 2)+ Math.pow(arr[1], 2);
		ans = Math.sqrt(ans) +1 ;
		ans = Math.round(ans);
		long d = (long)ans;
		return d;
	}

}
