#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long int ll;

int dp[1010][1010];

int main(){
	while(1){
		int n,m; cin >> n >> m;
		if(n==0&&m==0)break;
		vector<int> a(n),b(n),c(n+1);
		for(int i=0;i<n;i++){
			cin >> a[i];
		}
		for(int i=0;i<n;i++){
			cin >> b[i];
		}
		c[0]=0;
		for(int i=0;i<n;i++){
			if(a[i]==b[i])c[i+1]=0;
			else if(a[i]<b[i])c[i+1]=b[i]-a[i];
			else c[i+1]=b[i]+m-a[i];
		}
		for(int i=0;i<1010;i++){
			for(int j=0;j<1010;j++){
				dp[i][j]=1e9;
			}
		}
		dp[0][0]=0;
		for(int i=0;i<n;i++){
			for(int j=0;j<n;j++){
				if(dp[i][j]==1e9)continue;
				if(j>0){
					dp[i+1][j-1]=min(dp[i+1][j-1],dp[i][j]+(c[i]+j*m<c[i+1]+(j-1)*m?c[i+1]+(j-1)*m-c[i]-j*m:0));
				}
				dp[i+1][j]=min(dp[i+1][j],dp[i][j]+(c[i]+j*m<c[i+1]+j*m?c[i+1]+j*m-c[i]-j*m:0));
				dp[i+1][j+1]=min(dp[i+1][j+1],dp[i][j]+(c[i]+j*m<c[i+1]+(j+1)*m?c[i+1]+(j+1)*m-c[i]-j*m:0));
			}
		}
		int ans=1e9;
		for(int i=0;i<10;i++){
			ans=min(ans,dp[n][i]);
		}
		cout << ans << endl;
	}
}
