import math
T = int(input())

def process(tab, nb_piece, width):
	tab = sorted(tab)
	data = [0] * (int(math.log(tab[-1], 2)) + 1)
	power = 1
	index = 0
	for elem in tab:
		while elem > power:
			power *= 2
			index += 1
		if elem == power:
			data[index] += 1
	stage = 1
	trous = width
	while nb_piece > 0:
		for index in range(len(data) - 1, -1, -1):
			elem = 2 ** index
			while data[index] > 0 and elem <= trous:
				data[index] -= 1
				trous -= elem
				nb_piece -= 1
		stage += 1
		trous = width
	if trous == width:
		stage -= 1
	return stage
for _ in range(T):
	data = input().split()
	(n, width) = (int(data[0]), int(data[1]))
	data = input().split()
	tab = [int(elem) for elem in data]
	res = process(tab, n, width)
	print(res)
