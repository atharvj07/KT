import java.util.*;
import java.io.*;
public class Main3 {
    public static void main(String args[] ) throws Exception {
        PrintWriter pr=new PrintWriter(System.out);
        Scanner scan=new Scanner(System.in);
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        int n=scan.nextInt();
        int m=scan.nextInt();
        int ta=scan.nextInt();
        int tb=scan.nextInt();
        int k=scan.nextInt();
        int[] a=new int[n];
        int[] b=new int[m+1];
        for(int i=0;i<n;i++)
            a[i]=scan.nextInt();
        for(int i=0;i<m;i++)
            b[i]=scan.nextInt();
        b[m]=2000000000;
        int itr=Math.min(k,n);
        int j=-1;
        if(k>=n || k>=m){
            System.out.println(-1);
            return;
        }
        int maxT=0;
        for(int i=0;i<=itr;i++){
            int t=a[i]+ta;
            while(j<m && b[++j]<t);
            int l=k-i;
            if(l>=m-j){
                System.out.println(-1);
                return;
            }
            if(maxT<b[j]+tb){
                maxT=b[j]+tb;
            }

        }
        System.out.println(maxT);
    }
}
