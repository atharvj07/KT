import java.io.*;

public class Cats
{
    public static void main(String[] klein) throws IOException
    {
        final IntIO in = new IntIO(System.in);

        final int n = in.getInt(), m = in.getInt(), p = in.getInt();

        final int[] d = new int[n];
        for(int i = 1; i<n; i++) d[i] = d[i-1] + in.getInt();

        final long[] t = new long[m];
        for(int i = 0; i<m; i++) t[i] = -d[in.getInt()-1] + in.getInt();
        java.util.Arrays.sort(t);

        final long[] s = new long[m]; s[0] = t[0];
        for(int i = 1; i<m; i++) s[i] = s[i-1] + t[i];

        //dp[p][i] = min(j<i) dp[p-1][j] + (i-j)*t[i] - s[i] + s[j];
        //--> i*t[i]-s[i] + dp[j]-j*t[i]+s[j] = -j*t[i] + (dp[j]+s[j])

        final long[] prv = new long[m], nxt = new long[m];
        for(int i = 0; i<m; i++) prv[i] = (i+1)*t[i] - s[i];
        final int[] q = new int[m];
        for(int k = 1; k<p; k++)
        {
            q[0] = 0; int front = 0, back = 0;
            for(int i = 1; i<m; i++)
            {
                while(front<back && (prv[q[back]]+s[q[back]]-prv[q[back-1]]-s[q[back-1]])/(double)(-q[back-1] - -q[back])>=(prv[i]+s[i]-prv[q[back]]-s[q[back]])/(double)(-q[back] - -i)) --back;
                q[++back] = i;

                while(front<back && -q[front]*t[i] + prv[q[front]]+s[q[front]] >= -q[front+1]*t[i] + prv[q[front+1]]+s[q[front+1]]) ++front;
                nxt[i] = i*t[i]-s[i] - q[front]*t[i] + prv[q[front]]+s[q[front]];
            }
            System.arraycopy(nxt,0,prv,0,m);
        }

        System.out.println(prv[m-1]);
    }

    private static class IntIO extends BufferedInputStream
    {
        public IntIO(final InputStream in)
        {
            super(in);
        }

        public IntIO(final InputStream in, final int buf_size)
        {
            super(in, buf_size);
        }

        public int getInt() throws IOException
        {
            int chr = read();
            while(chr<=' ') chr = read();
            int sum = chr-'0';
            while((chr = read())>' ') sum = (sum<<3)+sum+sum + chr-'0';
            return sum;
        }
    }
}
