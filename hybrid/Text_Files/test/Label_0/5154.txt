#include <bits/stdc++.h>
using namespace std;
inline int read() {
  char ch = getchar();
  int x = 0, op = 1;
  while (ch < '0' || '9' < ch) {
    if (ch == '-') op = -1;
    ch = getchar();
  }
  while ('0' <= ch && ch <= '9') {
    x = x * 10 + ch - '0';
    ch = getchar();
  }
  return op * x;
}
const int maxdep = 16, maxn = 30009;
const int base = 10000019, MOD = 1000000007;
int power[maxn], v[maxn], w[maxn], L[maxn], R[maxn], flag[maxn];
vector<pair<int, int> > T[maxn << 2];
long long dp[maxdep][1009];
int n, K, Q, m, t, a, b, ans;
void update(int o, int l, int r, int x, int y, int v, int w) {
  if (l == x && y == r) {
    T[o].push_back(make_pair(v, w));
    return;
  }
  if (x <= ((l + r) >> 1))
    update((o << 1), l, ((l + r) >> 1), x, min(y, ((l + r) >> 1)), v, w);
  if (((l + r) >> 1) + 1 <= y)
    update((o << 1 | 1), ((l + r) >> 1) + 1, r, max(((l + r) >> 1) + 1, x), y,
           v, w);
}
void solve(int o, int dep, int l, int r) {
  memset(dp[dep], 0, sizeof dp[dep]);
  if (dep)
    for (int i = (1); i <= (K); i++) dp[dep][i] = dp[dep - 1][i];
  if (!T[o].empty())
    for (int i = (0); i <= (T[o].size() - 1); i++)
      for (int j = (K); j >= (T[o][i].second); j--)
        dp[dep][j] =
            max(dp[dep][j], dp[dep][j - T[o][i].second] + T[o][i].first);
  if (l == r) {
    if (flag[l]) {
      ans = 0;
      for (int i = (1); i <= (K); i++)
        ans = (ans + 1LL * dp[dep][i] % MOD * power[i - 1]) % MOD;
      printf("%d\n", ans);
    }
    return;
  }
  solve((o << 1), dep + 1, l, ((l + r) >> 1));
  solve((o << 1 | 1), dep + 1, ((l + r) >> 1) + 1, r);
}
int main() {
  m = n = read();
  K = read();
  power[0] = 1;
  for (int i = (1); i <= (K); i++) power[i] = 1LL * power[i - 1] * base % MOD;
  for (int i = (1); i <= (n); i++) {
    v[i] = read();
    w[i] = read();
  }
  Q = read();
  for (int i = (1); i <= (n); i++) {
    L[i] = 1;
    R[i] = Q;
  }
  for (int i = (1); i <= (Q); i++) {
    t = read();
    if (t == 1) {
      v[++m] = read();
      w[m] = read();
      L[m] = i;
      R[m] = Q;
    } else if (t == 2)
      R[read()] = i;
    else
      flag[i] = 1;
  }
  for (int i = (1); i <= (m); i++) update(1, 1, Q, L[i], R[i], v[i], w[i]);
  solve(1, 0, 1, Q);
  return 0;
}
