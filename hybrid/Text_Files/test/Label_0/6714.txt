#include <bits/stdc++.h>

#define ll long long
#define llong long long
#define REP(i,n) for(auto i=0;i<n;++i)
#define REPI(itr,v) for(auto itr=v.begin();itr!=v.end();++itr)
#define FOR(i,a,b) for(auto i=a;i<b;++i)
#define ALL(v) v.begin(),v.end()
#define pb push_back

using namespace std;

//約数が現在の範囲で何個あるか
int divisor[100001];

//その値が範囲内に2個以上あるか
bool is_used[100001] = {};

//現在の範囲で2個以上あるものの総数
int num;


//値xの約数について、範囲に追加
void plus_divisor(int x){
  
  divisor[x]++;
  
  for(int i = 1; i * i <= x; ++i){
    if(x % i == 0){
      divisor[i]++; 
      if(i * i != x) divisor[x / i]++;
      if(divisor[i] >= 2 && !is_used[i]){
        num++;
        is_used[i] = true;
      }
      if(divisor[x / i] >= 2 && !is_used[x / i]){
        num++;
        is_used[x / i] = true;
      }
    }
  }
}

//値xの約数について、範囲から削除
void minus_divisor(int x){

  
  divisor[x]--;
  
  for(int i = 1; i * i <= x; ++i){
    if(x % i == 0){
      divisor[i]--; 
      if(i * i != x) divisor[x / i]--;
      if(divisor[i] < 2 && is_used[i]){
        num--;
        is_used[i] = false;
      }
      if(divisor[x / i] < 2 && is_used[x / i]){
        num--;
        is_used[x / i] = false;
      }
    }
  }
}

int main(){

  int n, w;
  cin >> n >> w;
  
  for(int i = 1; i <= w; ++i){
    plus_divisor(i);
  }

  for(int i = 1; i <= n - w + 1; ++i){
    cout << num;
    if(i != n - w + 1){
      minus_divisor(i); plus_divisor(i+w);
      cout << " ";
    }else{
      cout << "\n";
    }
  }

  return 0;
}
