#include <bits/stdc++.h>
#define ll long long
#define fast ios::sync_with_stdio(false), cin.tie(0), cout.tie(0)
using namespace std;
int nxt[5010];
int fnxt(int u)
{
    if (nxt[u] == u)
        return u;
    return nxt[u] = (fnxt(nxt[u]));
}
int main()
{
    fast;
    int t;
    cin >> t;
    int s[5010];
    queue<int> q;
    while (t--)
    {
        int n;
        cin >> n;
        for (int i = 1; i <= n; i++)
        {
            cin >> s[i];
        }
        for (int i = 1; i <= n + 1; i++)
        {
            nxt[i] = i + (s[i] == 1);
        }
        ll ans = 0;
        for (int i = 1; i <= n; i++)
        {
            if (s[i] + i > n)
            {
                if (i == n)
                {
                    ans += s[i] - 1;
                    s[i] = 1;
                }
                else
                {
                    ans += s[i] - (n - i);
                    s[i] = n - i;
                }
            }
            while (s[i] != 1)
            {
                s[i]--;
                ans++;
                int t = s[i] + i + 1;
                while (t <= n)
                {
                    if (s[t] == 1)
                        t = fnxt(t);
                    else
                    {
                        s[t]--;
                        nxt[t] = t + (s[t] == 1);
                        t = t + s[t] + 1;
                    }
                }
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
