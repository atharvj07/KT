def gcd(a, b):
	if b == 0:
		return a
	else:
		return gcd(b, a % b)

def leftRotate(arr, d, n):
	for i in range(gcd(d, n)):
		temp = arr[i]
		j = i
		while 1:
			k = j + d
			if k >= n:
				k = k - n
			if k == i:
				break
			arr[j] = arr[k]
			j = k
		arr[j] = temp

def prefix(a):
	b = [0] * n
	b[0] = a[0]
	for i in range(1, len(a)):
		b[i] = b[i - 1] + a[i]
	return b
for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	a.sort()
	b = [0] * n
	b[0] = a[0]
	for i in range(1, n):
		b[i] = b[i - 1] + a[i]
	if 0 not in b:
		print('YES')
		print(*a)
	else:
		cycle = 0
		while 0 in b:
			cycle += 1
			leftRotate(a, 1, n)
			b = prefix(a)
			if cycle == n + 1:
				break
		if 0 in b:
			print('NO')
		else:
			print('YES')
			print(*a)
