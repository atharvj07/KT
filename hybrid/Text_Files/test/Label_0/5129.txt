#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  scanf("%d", &n);
  long long mini = INT_MAX, miniNode;
  int arr[n];
  for (int i = 1; i <= n; i++) scanf("%d", arr + i);
  sort(arr + 1, arr + n + 1);
  int mn = arr[1], mx = arr[n];
  for (int i = mn; i <= mx; i++) {
    long long sum = 0;
    for (int j = 1; j <= n; j++) {
      if (arr[j] > i) {
        sum += arr[j] - (i + 1);
      } else if (arr[j] < i) {
        sum += abs((i - 1) - arr[j]);
      }
    }
    if (sum < mini) {
      mini = sum;
      miniNode = i;
    }
  }
  printf("%lld %lld\n", miniNode, mini);
  return 0;
}
