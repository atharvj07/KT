import java.util.Scanner;

/**
 * @see https://codeforces.com/contest/1143/problem/B
 * @author https://codeforces.com/profile/maht
 */
public class ProblemB {

    static long nirvana(long n) {
        if(n < 1) {
            return n + 1;
        } else {
            long b = n % 10;
            long a = n / 10;
            return Math.max(
                nirvana(a)*b,
                nirvana(a-1)*9
                );
        }
    }

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        long n = scan.nextLong(); // 1 < n < 2e9
        System.out.println("" + nirvana(n));
        scan.close();
    }
}
