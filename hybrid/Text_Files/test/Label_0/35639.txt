import java.lang.*;
import java.util.*;
import java.io.*;

public class ArrayCancellation{

    public long getMinCost(int n, Integer[]arr){
	long cost = 0L;
	int currNeg = -1;
	for(int i=0;i<n;i++){
	    int el = arr[i];
	    if(el>0){
		while(el>0){
		    currNeg = getCurrNeg(i,currNeg,n,arr,cost);
		    int minQty = Math.min(el,-arr[currNeg]);
		    el -= minQty;
		    arr[currNeg] += minQty;
		    if(currNeg<i)cost += (long)minQty;
		    //currNeg = getCurrNeg(i,currNeg,n,arr,cost);
		    //		    System.out.format("getMinCust, i %d, el %d, cost %d, currNeg %d\n",i,el,cost,currNeg);
		}
	    }
	}
	return cost;
    }

    int getCurrNeg(int ind, int cn, int n, Integer[]arr,long cost){
	//	System.out.format("getCurrNeg, ind %d, cost %d, cn %d\n",ind,cost,cn);
	if(cn<0){
	    cn = (ind+1)%n;
	}else if((cn<ind)&&(cost==0L)){
	    cn = (ind+1)%n;
	}
	while((((cn)%n)!=ind)&&(arr[cn]>=0)){
	    cn++;
	    cn %=n;
	}

	//	System.out.format("getcurrneg returns %d\n",cn);
	return cn;
    }
    
    static Integer[]strToIntArr(String s){
	//	s = s.substring(1,s.length()-1);
	//	String[] sArr = s.split(",");
	String[] sArr = s.split(" ");	
	int len = sArr.length;
	Integer[] retArr = new Integer[len];
	for(int i=0;i<len;i++){
	    retArr[i] = Integer.parseInt(sArr[i]);
	}
	return retArr;
    }

    public static void main(String[]args)throws Exception{
	BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
	BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
	ArrayCancellation ac = new ArrayCancellation();
	String tLn = br.readLine();
	int t = Integer.parseInt(tLn);
	for(int i=0;i<t;i++){
	    String nLn = br.readLine();
	    int n = Integer.parseInt(nLn);
	    String aLn = br.readLine();
	    Integer[]arr = strToIntArr(aLn);
	    long mc = ac.getMinCost(n,arr);
	    bw.write(Long.toString(mc));
	    bw.newLine();
	    bw.flush();
	}
    }
}
