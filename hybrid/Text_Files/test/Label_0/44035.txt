import sys
import math

def check(d):
	w = 0
	for i in d:
		if d[i] % 2 != 0:
			w = w + 1
	return w

def fn(n):
	b = []
	for i in n:
		b.append(n.count(i))
	d = dict(zip(n, b))
	w = check(d)
	if len(n) % 2 == 0 and w == 0:
		print('First')
	if len(n) % 2 == 0 and w != 0:
		print('Second')
	if len(n) % 2 != 0 and w % 2 != 0:
		print('First')
	if len(n) % 2 != 0 and w % 2 == 0:
		print('Second')
input = sys.stdin.read()
data = list(map(str, input.split()))
n = data[0]
fn(n)
