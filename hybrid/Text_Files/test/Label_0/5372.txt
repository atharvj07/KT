#include <bits/stdc++.h>
int main(int argc, char *argv[]) {
  int tc;
  scanf("%d", &tc);
  while (tc--) {
    char a[100005], b[100005];
    int n, i, j = 0, ans[200005], det = 0, p, q;
    scanf("%d%s%s", &n, a, b);
    p = 1, q = n - 1;
    for (i = n - 1; i >= 1; i--) {
      det++;
      if (a[0] == b[i]) ans[j++] = 1;
      ans[j++] = i + 1;
      if (det % 2 == 0) {
        a[0] = a[p];
        p++;
      } else if (det % 2) {
        if (a[q] == '0')
          a[0] = '1';
        else
          a[0] = '0';
        q--;
      }
    }
    if (a[0] != b[0]) ans[j++] = 1;
    printf("%d ", j);
    for (i = 0; i < j; i++) printf("%d ", ans[i]);
    printf("\n");
  }
}
