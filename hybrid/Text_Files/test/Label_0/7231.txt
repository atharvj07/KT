#include <bits/stdc++.h>
using namespace std;
void fio() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
}
int main() {
  fio();
  int n;
  cin >> n;
  int* a = new int[n];
  int i;
  for (i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a, a + n);
  int c = 0;
  for (i = 0; i < n; i++) {
    if (a[i] == 0) continue;
    int j = i + 1;
    while (j < n) {
      if (a[j] % a[i] == 0) a[j] = 0;
      j++;
    }
    c++;
  }
  cout << c << endl;
}
