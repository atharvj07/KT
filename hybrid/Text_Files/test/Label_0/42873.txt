def pal(string):
	reversed_string = string[::-1]
	return string == reversed_string
(n, m) = map(int, input().split())
a = []
s = ''
s1 = ''
for _ in range(n):
	a.append(str(input()))
for i in range(len(a) - 1):
	for j in range(i + 1, len(a)):
		if a[i] == a[j][::-1]:
			s += a[i]
			s1 = str(a[j]) + s1
for i in range(len(a)):
	if pal(a[i]):
		if pal(s + a[i] + s1):
			s += a[i]
s2 = s + s1
print(len(s2))
print(s2)
