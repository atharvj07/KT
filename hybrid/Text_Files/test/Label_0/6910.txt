#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n, m, i, j, ans = 0;
  cin >> n >> m;
  long long int cnt[n];
  long long int a[n][m];
  for (i = 0; i < n; i++)
    for (j = 0; j < m; j++) cin >> a[i][j];
  for (i = 0; i < m; i++) {
    for (j = 0; j < n; j++) cnt[j] = 0;
    for (j = 0; j < n; j++) {
      if (a[j][i] % m == (i + 1) % m && a[j][i] <= (n - 1) * m + i + 1) {
        long long int r = 0;
        if (i == m - 1) r++;
        long long int x = j - a[j][i] / m + r;
        x = (x + n) % n;
        cnt[x]++;
      }
    }
    long long int x = n - cnt[0];
    for (j = 0; j < n; j++) x = min(x, n - cnt[j] + j);
    ans += x;
  }
  cout << ans << endl;
  return 0;
}
