n = int(input())
a = list(map(int, input().split()))
occur = [0] * (n + 1)
for i in range(n):
	occur[a[i]] += 1
need = []
change = []
for i in range(1, n + 1):
	if occur[i] == 0:
		need.append(i)
print(len(need))
j = 0
done = [0] * n
for i in range(n):
	if occur[a[i]] >= 1:
		if j < len(need) and need[j] < a[i] and (occur[a[i]] > 1):
			occur[a[i]] -= 1
			a[i] = need[j]
			done[need[j] - 1] = 1
			j += 1
		elif j < len(need) and done[a[i] - 1] == 1:
			occur[a[i]] -= 1
			a[i] = need[j]
			done[need[j] - 1] = 1
			j += 1
		else:
			occur[a[i]] -= 1
			done[a[i] - 1] = 1
	else:
		done[a[i] - 1] = 1
		occur[a[i]] -= 1
print(*a)
