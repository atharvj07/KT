(n, k) = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
dp = [0]
x = [0]
for i in a:
	x.append(x[-1] + i)
for i in range(n):
	dp.append(dp[-1] + a[i] * b[i])
ans = 0
for i in range(1, n + 1):
	c = i
	y = i - k
	y = max(y, 0)
	ans = max(ans, x[c] - x[y] + dp[y] + dp[n] - dp[i])
print(ans)
