n = int(input())
l1 = list(map(int, input().split()))
l1.sort()
(l2, l3) = ([], [])
x = n // 2
if n % 2 != 0:
	print(x)
	l2 = l1[0:x]
	l3 = l1[x:n]
else:
	print(x - 1)
	l2 = l1[0:x - 1]
	l3 = l1[x - 1:n]
(x, y) = (0, 0)
for i in range(n):
	if i % 2 == 0:
		print(l3[x], end=' ')
		x += 1
	else:
		if y == len(l2):
			break
		print(l2[y], end=' ')
		y += 1
for i in range(x, len(l3)):
	print(l3[i], end=' ')
