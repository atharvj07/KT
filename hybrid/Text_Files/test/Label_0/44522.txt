L = int(input())
D = input()
s = 0
b = 42069666
for c in range(1, L):
	if D[c] != '0':
		if abs(L / 2 - c) < b:
			b = abs(L / 2 - c)
			s = c

def c(s):
	if s == '':
		return 0
	return int(s)
res = c(D[:s]) + c(D[s:])
if L % 2:
	r = L // 2
	if D[r] != '0':
		res = min(c(D[:r]) + c(D[r:]), res)
	if D[r + 1] != '0':
		res = min(c(D[:r + 1]) + c(D[r + 1:]), res)
print(res)
