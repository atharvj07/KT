#include <bits/stdc++.h>
using namespace std;
int a[200001], Min, x, n, Max;
bitset<200001> ok;
int main() {
  scanf("%d", &n);
  for (int i = 1; i < n; i++) {
    scanf("%d", &x);
    a[i] = a[i - 1] + x;
    Min = min(Min, a[i]);
    Max = max(Max, a[i]);
    if ((a[i] - Min + 1 > n) || (Max - a[i] > n)) {
      printf("%d ", -1);
      exit(0);
    }
  }
  for (int i = 0; i < n; i++) ok[a[i] - Min + 1] = 1;
  for (int i = 1; i <= n; i++)
    if (!ok[i]) {
      printf("%d ", -1);
      exit(0);
    }
  for (int i = 0; i < n; i++) printf("%d ", a[i] - Min + 1);
  return 0;
}
