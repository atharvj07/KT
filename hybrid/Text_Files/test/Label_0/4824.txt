#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main(){
    cin.tie(0);
    ios::sync_with_stdio(false);
    int N,Q; cin >> N >> Q;
    vector<ll> A(N),left((N+1)/2+1,0),right((N+1)/2+1,0);
    for (int i=0;i<N;++i) cin >> A[i];
    for (int i=0;i<(N+1)/2;++i){
        left[i+1]=left[i]+(N&1?A[i*2]:A[i*2+1]);
        right[i+1]=right[i]+A[N-1-i];
    }
    vector<ll> border;
    for (int i=1;i<=(N+1)/2;++i) border.emplace_back(A[N-i]+A[N-2*i+1]);
    reverse(border.begin(),border.end());
    for (;Q--;){
        ll X; cin >> X;
        int r=border.end()-lower_bound(border.begin(),border.end(),2*X);
        if (!r) ++r;
        cout << left[(N+1)/2-r]+right[r] << '\n';
    }
}
