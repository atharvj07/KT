import sys
input = sys.stdin.readline

def read_list():
	return map(int, input().split())

def print_list(l):
	print(' '.join(map(str, l)))
from collections import defaultdict as dc
N = int(input())
for _ in range(N):
	(n, m) = read_list()
	res = []
	gress = [0] * (n + 1)
	dn = dc(list)
	for _ in range(m):
		(d, x, y) = read_list()
		if d == 1:
			dn[x].append(y)
			gress[y] += 1
		res.append([d, x, y])
	zp = [i for i in range(1, n + 1) if gress[i] == 0]
	count = 0
	p = [0] * (n + 1)
	while zp:
		u = zp.pop()
		count += 1
		p[u] = count
		for v in dn[u]:
			gress[v] -= 1
			if gress[v] == 0:
				zp.append(v)
	if count == n:
		print('YES')
		for i in range(m):
			if res[i][0] == 1:
				print(res[i][1], res[i][2])
			elif p[res[i][1]] < p[res[i][2]]:
				print(res[i][1], res[i][2])
			else:
				print(res[i][2], res[i][1])
	else:
		print('NO')
