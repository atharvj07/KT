def main():
	from sys import stdin, stdout
	n = int(stdin.readline())
	tup = tuple(map(int, stdin.readline().split()))
	minim = min(tup)
	maxim = max(tup)
	maximindex = tup.index(maxim)
	flag = 1
	i = 0
	pv = tup[-1]
	while i < n:
		if tup[i] < pv:
			if tup[i] == minim and pv == maxim:
				l = i + 1
				while l < max(maximindex, n):
					if tup[l] < tup[l - 1]:
						flag = 0
						break
					l += 1
				break
			else:
				flag = 0
				break
		pv = tup[i]
		i += 1
	if flag:
		if i:
			stdout.write(str(n - i))
		else:
			stdout.write('0')
	else:
		stdout.write('-1')
main()
