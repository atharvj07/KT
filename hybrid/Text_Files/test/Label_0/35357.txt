import java.util.Scanner;

public class A {
    public static void main(String args[]) {
        Scanner in = new Scanner(System.in);
        int res = -1;
        int n = in.nextInt();
        for (int i = 0; i < n; i++) {
            int s = in.nextInt();
            int d = in.nextInt();

            while (s <= res) {
                s += d;
            }
            res = Math.max(res, s);

        }
        System.out.println(res);

    }
}
