import sys

def dijkstra_t(adj: list, start: int, inf: int=10 ** 18) -> list:
	from heapq import heappush, heappop
	v_count = len(adj)
	dists = [inf] * v_count
	dists[start] = 0
	(hq, rem) = ([(0, start)], v_count - 1)
	while hq and rem:
		(cost, src) = heappop(hq)
		if dists[src] < cost:
			continue
		rem -= 1
		for (dest, _cost) in adj[src]:
			newcost = cost + _cost
			if dists[dest] > newcost:
				dists[dest] = newcost
				heappush(hq, (newcost, dest))
	return dists
(n, m, s, t) = map(int, sys.stdin.buffer.readline().decode('utf-8').split())
s -= 1
t -= 1
adj = [[] for _ in range(n)]
matrix = [[0] * n for _ in range(n)]
for (u, v) in (map(int, line.decode('utf-8').split()) for line in sys.stdin.buffer):
	adj[u - 1].append((v - 1, 1))
	adj[v - 1].append((u - 1, 1))
	matrix[u - 1][v - 1] = matrix[v - 1][u - 1] = 1
dist1 = dijkstra_t(adj, s, 10 ** 9)
dist2 = dijkstra_t(adj, t, 10 ** 9)
ans = 0
for i in range(n):
	for j in range(i + 1, n):
		if matrix[i][j]:
			continue
		if dist1[i] + 1 + dist2[j] >= dist1[t] and dist1[j] + 1 + dist2[i] >= dist1[t]:
			ans += 1
print(ans)
