from sys import stdin, stdout
from collections import defaultdict
import math
from itertools import groupby

def get_list():
	return list(map(int, stdin.readline().strip().split()))

def get_int():
	return int(stdin.readline())

def get_ints():
	return map(int, stdin.readline().strip().split())

def get_string():
	return stdin.readline().strip()

def printn(n):
	stdout.write(str(n) + '\n')

def printlis(a):
	for x in a:
		stdout.write(str(x) + ' ')
	printn('')

def solve():
	n = get_int()
	a = get_string()
	b = get_string()
	ans = 0
	i = 0
	x = -1
	while i < n:
		if a[i] != b[i]:
			ans += 2
		elif a[i] == b[i] and a[i] == '1':
			if i > 0 and a[i - 1] == b[i - 1] and (a[i - 1] == '0') and (x != i - 1):
				ans += 1
			elif i < n - 1 and a[i + 1] == b[i + 1] and (a[i + 1] == '0'):
				ans += 1
				x = i + 1
		else:
			ans += 1
		i += 1
	printn(ans)
	return
t = get_int()
while t:
	t -= 1
	solve()
