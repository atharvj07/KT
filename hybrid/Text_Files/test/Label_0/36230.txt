import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.Arrays;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.StringTokenizer;
import java.io.BufferedReader;
import java.util.Comparator;
import java.util.ArrayDeque;
import java.util.Collections;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        EdoAndMagnets solver = new EdoAndMagnets();
        solver.solve(1, in, out);
        out.close();
    }

    static class EdoAndMagnets {
        ArrayList<int[]> orders = new ArrayList<>();
        ArrayList<Magnet>[] sorted = new ArrayList[4];
        int[] sentinel = new int[100001];

        public void solve(int testNumber, InputReader in, PrintWriter out) {
            int N = in.nextInt();
            int K = in.nextInt();
            ArrayList<Magnet> magnets = new ArrayList<>();
            for (int i = 0; i < N; i++) {
                magnets.add(new Magnet(i, in.nextInt(), in.nextInt(), in.nextInt(), in.nextInt()));
            }
            Collections.sort(magnets, new Comparator<Magnet>() {

                public int compare(Magnet magnet, Magnet t1) {
                    return Long.compare(magnet.x, t1.x);
                }
            });
            add(magnets, 0);
            Collections.reverse(magnets);
            add(magnets, 1);
            Collections.sort(magnets, new Comparator<Magnet>() {

                public int compare(Magnet magnet, Magnet t1) {
                    return Long.compare(magnet.y, t1.y);
                }
            });
            add(magnets, 2);
            Collections.reverse(magnets);
            add(magnets, 3);
            genOrder(K);
            Arrays.fill(sentinel, -1);
            long res = getArea(new int[]{0, 0, 0, 0});
            for (int i = 0; i < orders.size(); i++) {
                long sol = solveOrder(orders.get(i), i);
                if (sol > 0) {
                    res = Math.min(res, sol);
                }
                if (res == 0) {
                    solveOrder(orders.get(i), i + 1);
                }
            }
            out.println(res / 4);
        }

        long getArea(int[] l) {
            long res = (Math.max(2, Math.abs(sorted[1].get(l[1]).x - sorted[0].get(l[0]).x))) * (Math.max(2, sorted[3].get(l[3]).y - sorted[2].get(l[2]).y));
            assert (res > 0) : "Not positive";
            return res;
        }

        long solveOrder(int[] a, int i) {
            int[] l = new int[4];
            for (int iter = 0; iter < a.length; iter++) {
                while (sentinel[sorted[a[iter]].get(l[a[iter]]).i] == i) {
                    l[a[iter]]++;
                }
                sentinel[sorted[a[iter]].get(l[a[iter]]).i] = i;
                l[a[iter]]++;
            }
            for (int j = 0; j < 4; j++) {
                while (sentinel[sorted[j].get(l[j]).i] == i) {
                    l[j]++;
                }
            }
            return getArea(l);
        }

        void genOrder(int k) {
            if (k == 0) {
                return;
            }
            ArrayDeque<int[]> deque = new ArrayDeque<>();
            deque.add(new int[]{});
            while (deque.size() > 0) {
                int[] next = deque.remove();
                if (next.length == k) {
                    orders.add(next);
                } else {
                    int[] cloned = new int[next.length + 1];
                    for (int i = 0; i < next.length; i++) {
                        cloned[i] = next[i];
                    }
                    for (int i = 0; i < 4; i++) {
                        cloned[next.length] = i;
                        deque.add(cloned.clone());
                    }
                }
            }
        }

        void add(ArrayList<Magnet> magnets, int i) {
            sorted[i] = new ArrayList<>();
            for (int j = 0; j < magnets.size(); j++) {
                sorted[i].add(magnets.get(j));
            }
        }

        class Magnet {
            int i;
            long x;
            long y;

            Magnet(int i, int x1, int y1, int x2, int y2) {
                this.x = (x1 + x2);
                this.y = (y1 + y2);
                this.i = i;
            }

        }

    }

    static class InputReader {
        public BufferedReader reader;
        public StringTokenizer tokenizer;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream), 32768);
            tokenizer = null;
        }

        public String next() {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
}
