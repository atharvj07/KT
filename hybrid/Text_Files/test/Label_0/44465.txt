import collections, itertools, fractions, functools, heapq, math, queue, copy

def other(a, b):
	for c in 'abc':
		if c != a and c != b:
			return c

def solve():
	(n, t) = map(int, input().split())
	s1 = list(input())
	s2 = list(input())
	dif = sum((1 for (a, b) in zip(s1, s2) if a != b))
	if dif > 2 * t:
		return -1
	s = []
	if dif <= t:
		for (a, b) in zip(s1, s2):
			if a == b:
				if dif < t:
					dif += 1
					s.append(other(a, b))
				else:
					s.append(a)
			else:
				s.append(other(a, b))
	else:
		diff1 = dif
		diff2 = dif
		for (a, b) in zip(s1, s2):
			if a == b:
				s.append(a)
				continue
			if diff1 > t:
				s.append(b)
				diff1 -= 1
				continue
			if diff2 > t:
				s.append(a)
				diff2 -= 1
				continue
			s.append(other(a, b))
	return str.join('', s)
print(solve())
