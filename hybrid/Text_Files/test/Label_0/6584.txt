#include <bits/stdc++.h>
using namespace std;
string str;
int len;
int dp[200005][4];
int fun(int pos, int mod) {
  mod %= 3;
  if (pos == len) {
    return mod = 0;
  }
  if (dp[pos][mod] != -1) return dp[pos][mod];
  int ret1 = 0, ret2 = 0;
  int now = str[pos] - '0';
  if (now % 3 == 0) {
    ret1 = 1 + fun(pos + 1, 0);
  } else {
    ret1 = fun(pos + 1, 0);
    if ((mod + now) % 3 == 0)
      ret2 = 1 + fun(pos + 1, 0);
    else
      ret2 = fun(pos + 1, mod + now);
  }
  return dp[pos][mod] = max(ret1, ret2);
}
int main() {
  memset(dp, -1, sizeof dp);
  cin >> str;
  len = str.size();
  cout << fun(0, 0) << endl;
}
