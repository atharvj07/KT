import sys
(mod, MOD) = (1000000007, 998244353)

def get_array():
	return list(map(int, sys.stdin.readline().strip().split()))

def get_ints():
	return map(int, sys.stdin.readline().strip().split())

def input():
	return sys.stdin.readline().strip()
(a, b) = get_ints()
c = d = e = 0
for i in range(1, 7):
	if abs(a - i) < abs(b - i):
		c += 1
	elif abs(a - i) > abs(b - i):
		e += 1
	else:
		d += 1
print(c, d, e)
