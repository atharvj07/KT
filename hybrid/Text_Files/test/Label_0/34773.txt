import java.util.*;
public class bob {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int r = in.nextInt();
        int c = in.nextInt();
        for(int i = 0; i < r; i++) {
            for(int k = 0; k < c; k++) {
                String inp = in.next();
                if(!inp.equals("W") && !inp.equals("B")&& !inp.equals("G")) {
                    System.out.println("#Color");
                    return;
                }
            }
        }
        System.out.println("#Black&White");
    }
}
