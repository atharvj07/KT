def solve():
	cnt = [1, 0]
	ans = 0
	for i in range(1, n + 1):

		ans += cnt[1 ^ a[i]]

n = int(input())
s = input()
a = list(map(int, ('0 ' + s).split()))
for i in range(n + 1):
	if a[i] >= 0:
		a[i] = 0
	else:
		a[i] = 1
	if i > 0:
		a[i] ^= a[i - 1]
ans = solve()
print(ans, n * (n + 1) // 2 - ans)
