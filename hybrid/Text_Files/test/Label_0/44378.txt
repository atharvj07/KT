M = pow(10, 6) + 3

def wall(n, c):
	fact = factorial(n + c)
	(_, inv_n, _) = extended_euclidean_alg_1(fact[n], M)
	(_, inv_c, _) = extended_euclidean_alg_1(fact[c], M)
	return (fact[n + c] * (inv_n % M) % M * (inv_c % M) % M - 1) % M

def factorial(m):
	facts = [1 for _ in range(m + 1)]
	for n in range(1, m + 1):
		facts[n] = facts[n - 1] * n % M
	return facts

def extended_euclidean_alg_1(a, b):
	if not b:
		return (0, 1, 0)
	(u0, u1) = (1, 0)
	(v0, v1) = (0, 1)
	while b:
		q = a // b
		r = a - b * q
		u = u0 - q * u1
		v = v0 - q * v1
		a = b
		b = r
		u0 = u1
		u1 = u
		v0 = v1
		v1 = v
	return (a, u0, v0)

def main():
	(n, c) = [int(i) for i in input().split()]
	count = wall(n, c)
	print(int(count))
main()
