for _ in range(int(input())):
	(n, m) = list(map(int, input().split()))
	a = []
	for _ in range(n):
		a.append(list(map(int, input().split())))
	c = 0
	s = 0
	mini = abs(a[0][0])
	for i in range(n):
		for j in range(m):
			s += abs(a[i][j])
			if a[i][j] < 0:
				c += 1
			if abs(a[i][j]) < mini:
				mini = abs(a[i][j])
	if c % 2 == 0:
		print(s)
	else:
		print(s - 2 * abs(mini))
