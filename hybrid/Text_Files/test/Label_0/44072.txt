n = int(input())
a = [int(i) for i in input().split()]
c = 1
for i in range(1, n):
	if abs(a[i] - a[i - 1]) > 1:
		c = abs(a[i] - a[i - 1])
x3 = 10 ** 9
r = 1
for i in range(n - 1):
	x1 = max(0, a[i] - 1) // c
	y1 = a[i] % c
	if y1 == 0:
		y1 = c
	x2 = max(0, a[i + 1] - 1) // c
	y2 = a[i + 1] % c
	if y2 == 0:
		y2 = c
	if x2 == x1 + 1 and y2 == y1 and (x1 < x3):
		continue
	elif x2 == x1 and y2 == y1 + 1 and (y1 < c):
		continue
	elif x2 == x1 - 1 and y2 == y1 and (x1 > 0):
		continue
	elif x2 == x1 and y2 == y1 - 1 and (y1 > 0):
		continue
	else:
		r = 0
if r == 1:
	print('YES')
	print(x3, c)
else:
	print('NO')
