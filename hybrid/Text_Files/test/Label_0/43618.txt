(m, n) = map(int, input().split())
a = list(map(int, input().split()))
l = 0
r = 10 ** 18
mn = r
s = 0
for i in a:
	s += i
	mn = min(s, mn)
if abs(mn) < m and s >= 0:
	print(-1)
	exit(0)

def f(k):
	x = k // n * s
	if -1 * x >= m:
		return 1
	if abs(x - s + mn) >= m:
		return 1
	for i in range(k % n):
		x += a[i]
		if -1 * x >= m:
			return 1
	return 0
i = 0
t = 0
while i < n:
	t += a[i]
	if m + t <= 0:
		print(i + 1)
		exit(0)
	i += 1
while r - l > 1:
	mid = (l + r) // 2
	if f(mid):
		r = mid
	else:
		l = mid
if f(r):
	print(r)
else:
	print(-1)
