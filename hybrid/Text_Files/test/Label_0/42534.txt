(n, k, a, b) = map(int, input().split())
m1 = max(a, b)
m2 = min(a, b)
if m1 - k * m2 > k:
	print('NO')
elif a == b:
	while a != 0:
		print('G', end='')
		print('B', end='')
		a -= 1
elif a > b:
	while b != 0 or a != 0:
		for i in range(k):
			if a == 0:
				break
			print('G', end='')
			a -= 1
			while a == b and a != 0:
				print('B', end='')
				print('G', end='')
				a -= 1
				b -= 1
		if b != 0:
			print('B', end='')
			b -= 1
			while a == b and a != 0:
				print('G', end='')
				print('B', end='')
				a -= 1
				b -= 1
elif a < b:
	while b != 0 or a != 0:
		for i in range(k):
			if b == 0:
				break
			print('B', end='')
			b -= 1
			while a == b and a != 0:
				print('G', end='')
				print('B', end='')
				a -= 1
				b -= 1
		if a != 0:
			print('G', end='')
			a -= 1
			while a == b and a != 0:
				print('B', end='')
				print('G', end='')
				a -= 1
				b -= 1
