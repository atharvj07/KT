(n, m) = list(map(int, input().split()))
s = input()
t = input()
equals = []
for i in range(m - n + 1):
	(j, equal) = (0, 0)
	while j < n:
		if t[i + j] == s[j]:
			equal += 1
		j += 1
	equals.append(equal)
index = equals.index(max(equals))
k = n - equals[index]
print(k)
if k > 0:
	j = 0
	answer = ''
	for i in range(index, index + n):
		if s[j] != t[i]:
			answer += str(j + 1) + ' '
		j += 1
	print(answer.strip())
