a = list(input())
b = list(set(a))
b.sort()
c = []
p = 96
for k in range(len(a)):
	if ord(a[k]) < p:
		p = 0
		break
	if a[k] not in c:
		c.append(a[k])
		p += 1
if len(set(a)) <= 2 or p == 0:
	print('NO')
elif b == c and (a.count('a') == a.count('c') or a.count('b') == a.count('c')):
	print('YES')
else:
	print('NO')
