(n, m) = map(int, input('').split())
t = list(map(int, input('').split()))
k = []
for i in range(n):
	sum = 0
	for j in range(i + 1):
		sum += t[j]
	if sum <= m:
		k.append(0)
	else:
		q = []
		for h in range(i):
			q.append(t[h])
		q.sort()
		s = len(q) - 1
		c = 0
		p = 0
		count = 0
		z = 0
		while c == 0:
			z = 0
			for e in range(p + 1):
				z = z + q[s - e]
			count += 1
			if sum - z <= m:
				k.append(count)
				break
			p += 1
for a in k:
	print(a, end=' ')
