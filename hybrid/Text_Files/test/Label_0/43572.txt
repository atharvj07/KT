import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
import sys
n = int(input())
SUM = sum(map(int, input().split()))
OK = SUM // n - n // 2 - 10
NG = SUM // n - n // 2 + 10

def tousa(s, n):
	return (s + (s + n - 1)) * n // 2
while NG - OK > 1:
	mid = (OK + NG) // 2
	if tousa(mid, n) > SUM:
		NG = mid
	else:
		OK = mid
rest = SUM - tousa(OK, n)
for i in range(n):
	if i < rest:
		sys.stdout.write(str(OK + i + 1) + ' ')
	else:
		sys.stdout.write(str(OK + i) + ' ')
