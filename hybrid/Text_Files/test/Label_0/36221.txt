public class C356B {

	static StringTokenizer st;
	static BufferedReader br;
	static PrintWriter pw;
	static int maxx = 1000001;
	static long[] a = new long [maxx];
	//static long[] s = new long [maxx];
	static long[] k = new long [maxx];
	static long[] p = new long [100002];
	
	public static void main(String[] args) throws IOException {
		br = new BufferedReader(new InputStreamReader(System.in));
		pw = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
		for (long i = 0; i < 100002; ++i)
			p[(int)i] = i * i * i;
		int curi = 1;
		for (int i = 1; i < maxx; ++i) {
			if (p[curi + 1] <= i) {
				++curi;
			}
			k[i] = k[(int) (i - p[curi])] + 1;
		}
		int count = 0;
		for (int i = 1; i < maxx; ++i) {
			if (k[i - 1] > k[i]) {
				k[i] = k[i - 1];
				a[i] = a[i - 1];
			} else {
				//System.out.println(i + " " + k[i]);
				++count;
				a[i] = i;
			}
		}
		long[] l, r;
		l = new long [maxx];
		r = new long [maxx];
		l[12] = 330;
		int kk = 12;
		for (long i = 1; i <= 100100; ++i) {
			if ((i + 1) * (i + 1) * (i + 1) - i * i * i >= l[kk]) {
				//System.out.println(i);
				r[kk] = l[kk] + i * i * i - 1;
				++kk;
				l[kk] = r[kk - 1] + 1;
			}
		}
		r[kk] = (long)1e18;
		long m = nextLong();
		if (m < maxx) {
			pw.println(k[(int) m] + " " + a[(int) m]);
		} else {
			int kol = 12;
			for (int i = 12; i < 100; ++i) {
				if (l[i] <= m && r[i] >= m) {
					kol = i;
					break;
				}
			}
			pw.print(kol + " ");
			long ans = 0;
			while (m >= maxx) {
				int maxs = find(m);
				if (m - p[maxs] >= l[kol - 1]) {
					m -= p[maxs];
					ans += p[maxs];
					--kol;
				} else {
					
					m = p[maxs] - p[maxs - 1] - 1;
					ans += p[maxs - 1];
					--kol;
					//pw.println(ans + l[kol]);
					//return;
				}
			}
			pw.println(ans + a[(int) m]);
		}
		pw.close();
	}
	
	private static int find(long m) {
		int l = 0, r = 100001;
		int mid = (l + r) / 2;
		while (r - l > 1) {
			mid = (l + r) / 2;
			if (p[mid] <= m)
				l = mid;
			else
				r = mid;
		}
		return l;
	}

	private static int sumf(int[] fen, int id) {
		int summ = 0;
		for (; id >= 0; id = (id & (id + 1)) - 1) 
 			summ += fen[id];
		return summ;
	}

	private static void addf(int[] fen, int id) {
		for (; id < fen.length; id |= id + 1) 
			fen[id]++;
	}

	private static int nextInt() throws IOException {
		return Integer.parseInt(next());
	}
	private static long nextLong() throws IOException {
		return Long.parseLong(next());
	}
	private static double nextDouble() throws IOException {
		return Double.parseDouble(next());
	}
	private static String next() throws IOException {
		while (st==null || !st.hasMoreTokens())
			st = new StringTokenizer(br.readLine());
		return st.nextToken();
	}
}
