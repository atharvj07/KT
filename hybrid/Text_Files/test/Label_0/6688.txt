#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  int a[n], b[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];
  priority_queue<int> q;
  for (int i = 0; i < n; i++) q.push(abs(a[i] - b[i]));
  int op = m + k;
  while (op) {
    int x = q.top();
    q.pop();
    if (x == 0) {
      if (op % 2 == 1) {
        cout << "1\n";
        return 0;
      } else {
        cout << "0\n";
        return 0;
      }
    } else {
      x--;
      q.push(x);
      op--;
    }
  }
  long long ans = 0;
  while (!q.empty()) {
    int x = q.top();
    q.pop();
    ans += (long long)x * x;
  }
  cout << ans << endl;
}
