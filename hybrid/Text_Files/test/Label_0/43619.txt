def main():
	t = int(input())
	for _ in range(t):
		n = int(input())
		(last_p, last_c) = [int(i) for i in input().split()]
		flag = True
		if last_p < last_c:
			flag = False
		for q in range(n - 1):
			(p, c) = [int(i) for i in input().split()]
			(diff_p, diff_c) = (p - last_p, c - last_c)
			if p < c or p < last_p or c < last_c or (diff_p < diff_c):
				flag = False
			(last_p, last_c) = (p, c)
		print('YES' if flag else 'NO')
main()
