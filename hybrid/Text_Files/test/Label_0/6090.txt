#include <bits/stdc++.h>
int N, M;
char st[1000001];
void init();
void doit();
int main() {
  init();
  doit();
  return 0;
}
void init() { gets(st); }
void doit() {
  int y, ans = 1, x = 1, L = strlen(st);
  while ((y = x) < L) {
    while (st[x] == st[y]) y++;
    ans += (st[x] == '1' || y < L);
    x = y + 1;
  }
  printf("%d\n", ans);
}
