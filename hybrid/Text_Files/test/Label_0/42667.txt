import math
t = int(input())
for _ in range(t):
	n = int(input())
	x = list(map(int, input().split()))
	s = 0
	if x[0] > 0:
		f = 1
	else:
		f = 0
	mx = -1000000000000.0
	mn = -1000000000000.0
	for i in range(n):
		if x[i] < 0 and f == 1 and (i > 0):
			f = 0
			s = s + mx
			mn = -1000000000000.0
		if x[i] > 0 and f == 0 and (i > 0):
			f = 1
			s = s + mn
			mx = -1000000000000.0
		if f == 1:
			mx = max(mx, x[i])
		if f == 0:
			mn = max(mn, x[i])
	if x[-1] > 0:
		s = s + max(mx, x[-1])
	else:
		s = s + max(mn, x[-1])
	print(s)
