#include <bits/stdc++.h>
using namespace std;
int n;
char a[200002];
char b[200002];
int c1[26];
int c2[26];
inline int get_inverse(char* a) {
  int res = 0;
  for (int i = 0; i < n; ++i)
    for (int j = 0; j < i; ++j) res += a[j] > a[i];
  return res;
}
inline void solve() {
  scanf("%d", &n);
  scanf("%s", a);
  scanf("%s", b);
  memset(c1, 0, sizeof(c1));
  memset(c2, 0, sizeof(c2));
  for (int i = 0; i < n; ++i) {
    ++c1[a[i] - 'a'];
    ++c2[b[i] - 'a'];
  }
  int ok = 0;
  for (int i = 0; i < 26; ++i) {
    if (c1[i] != c2[i]) {
      puts("NO");
      return;
    }
    if (c1[i] > 1) ok = 1;
  }
  if (ok) {
    puts("YES");
    return;
  }
  int res1 = get_inverse(a);
  int res2 = get_inverse(b);
  puts((res1 & 1) == (res2 & 1) ? "YES" : "NO");
}
int main() {
  int tt;
  scanf("%d", &tt);
  for (int t = 0; t < tt; ++t) solve();
  return 0;
}
