import sys
from collections import deque

def solve(n, edges):
	vis = [-1] * (n + 1)
	g = [set() for _ in range(n + 1)]
	for (u, v) in edges:
		g[u].add(v)
		g[v].add(u)
	subs = [[], []]
	for i in range(1, n + 1):
		if vis[i] < 0:
			q = deque([i])
			vis[i] = 0
			subs[0].append(i)
			while q:
				u = q.popleft()
				for v in g[u]:
					if vis[v] < 0:
						g[v].remove(u)
						vis[v] = 1 - vis[u]
						q.append(v)
						subs[vis[v]].append(v)
	subs.sort(key=lambda it: len(it))
	m = len(subs[0])
	val = []
	while m:
		b = m & -m
		val.extend(list(range(b, b << 1)))
		m -= b
	st = set(val)
	for i in subs[0]:
		vis[i] = val.pop()
	k = 1
	for i in subs[1]:
		while k in st:
			k += 1
		vis[i] = k
		k += 1
	return ' '.join(map(str, vis[1:]))
t = int(sys.stdin.readline())
ans = []
for _ in range(t):
	n = int(sys.stdin.readline())
	edges = [tuple(map(int, sys.stdin.readline().split())) for _ in range(n - 1)]
	ans.append(solve(n, edges))
print('\n'.join(ans))
