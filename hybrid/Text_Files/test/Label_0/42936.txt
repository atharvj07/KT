n = int(input())
d = 1000000000.0
x = list(map(int, input().split()))
x1 = x[:]
for i in range(n + 1):
	for j in range(len(x) - 1):
		try:
			if x[j + 1] - x[j] > 0:
				d = min(d, x[j + 1] - x[j])
				x.append(x[j + 1] - x[j])
			elif x[j] - x[j + 1] > 0:
				d = min(d, x[j] - x[j + 1])
				x.append(x[j] - x[j + 1])
			x = sorted(set(x))
		except IndexError:
			break
if d == 1000000000.0:
	print(sum(x1))
else:
	d = min(d, min(x))
	print(n * d)
