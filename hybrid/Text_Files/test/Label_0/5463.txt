bool solve(void){
	int i,h,w,n;cin>>h>>w>>n;
	vector<pair<int,int>>ami(n);
	for(i=0;i<n;i++){cin>>ami[i].fir>>ami[i].sec;ami[i].fir--;ami[i].sec--;}
	SO(ami);REV(ami);
	vector<int>go(w);
	for(i=0;i<w;i++){
		int L;
		if(i%2==0){L=i+h;}
		else{L=i-h;}
		if(L<0){L=abs(L)-1;}L%=w*2;if(L>=w){L=w+w-1-L;}
		go[i]=L;
		
	}
	for(auto it:ami){
		int x,y;
		tie(y,x)=it;
		int L=x-y;if(L<0){L=abs(L)-1;}L%=w*2;if(L>=w){L=w+w-1-L;}
		int R=x+1+y;R%=w*2;if(R>=w){R=w+w-1-R;}
		swap(go[L],go[R]);
	}
	for(i=0;i<w;i++){cout<<go[i]+1<<endl;}
	return false;
}
int main(void){
	cout<<fixed<<setprecision(20);
