from sys import stdin, stdout
input = stdin.readline
n = int(input())
lst = []
for i in range(n):
	lst.append(list(map(int, input().split())))
col = [0] * n
row = [0] * n
anti = 0
dig = 0
ri = -1
ci = -1
dd = 0
ad = 0
if n == 1:
	print(1)
	exit(0)
for i in range(n):
	for j in range(n):
		if lst[i][j] == 0:
			ri = i
			ci = j
			if i == j:
				dd = 1
			if i + j == n - 1:
				ad = 1
		row[i] += lst[i][j]
		col[j] += lst[i][j]
		if i == j:
			dig += lst[i][j]
		if i + j == n - 1:
			anti += lst[i][j]
if len(set(row)) != 2 or len(set(col)) != 2 or len(set(col + row)) != 2:
	print(-1)
elif dd == 1:
	if ad == 1:
		if max(row) - row[ri] == max(col) - col[ci] == max(row) - dig == max(row) - anti:
			print(max(row) - row[ri])
		else:
			print(-1)
	elif anti == max(row) and max(row) - row[ri] == max(col) - col[ci] == max(row) - dig:
		print(max(row) - row[ri])
	else:
		print(-1)
elif ad == 1:
	if dig == max(row) and max(row) - row[ri] == max(col) - col[ci] == max(row) - anti:
		print(max(row) - row[ri])
	else:
		print(-1)
elif dig == max(row) == anti and max(row) - row[ri] == max(col) - col[ci]:
	print(max(row) - row[ri])
else:
	print(-1)
