#include <bits/stdc++.h>
using namespace std;
const int MAX = 1000005;
const int MOD = 1e9 + 7;
int main() {
  int n;
  cin >> n;
  int mx1, mx2;
  long long wsum = 0;
  std::vector<int> w;
  std::vector<int> h;
  w.resize(n + 1);
  h.resize(n + 1);
  for (int i = 0; i < n; ++i) {
    cin >> w[i];
    cin >> h[i];
    if (i == 0)
      mx1 = i;
    else if (i == 1) {
      mx2 = i;
      if (h[mx1] < h[mx2]) swap(mx1, mx2);
    } else if (h[i] > h[mx1]) {
      mx2 = mx1;
      mx1 = i;
    } else if (h[i] > h[mx2]) {
      mx2 = i;
    }
    wsum += w[i];
  }
  for (int i = 0; i < n; ++i) {
    long long ans;
    if (i == mx1) {
      ans = h[mx2] * (wsum - w[i]);
    } else {
      ans = h[mx1] * (wsum - w[i]);
    }
    cout << ans << " ";
  }
}
