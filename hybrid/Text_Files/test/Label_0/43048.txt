n = int(input())
a = [0] + list(map(int, input().split()))
s = [0] * (n + 2)
for i in range(1, n + 1):
	s[i] = s[i - 1] + a[i]
s[-1] = s[-2] + 100
m = int(input())
q = list(map(int, input().split()))
for t in q:
	(l, r) = (0, n + 1)
	for i in range(300):
		mid = (l + r) // 2
		if s[mid - 1] < t and t <= s[mid]:
			print(mid)
			break
		elif s[mid] < t:
			l = mid + 1
		else:
			r = mid - 1
