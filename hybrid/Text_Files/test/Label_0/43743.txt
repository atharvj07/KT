from collections import defaultdict
prefix = defaultdict(int)
for i in range(1, 10000):
	prefix[i ** 3] = i
cache = defaultdict(str)
for i in range(int(input())):
	n = int(input())
	if n in cache:
		print(cache[n])
		continue
	for i in prefix:
		if n - i in prefix:
			print('YES')
			cache[n] = 'YES'
			break
	else:
		cache[n] = 'NO'
		print('NO')
