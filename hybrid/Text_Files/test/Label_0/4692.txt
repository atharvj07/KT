#include <bits/stdc++.h>
using namespace std;
template <class T>
void pr1(T a) {
  cout << a;
  cout << '\n';
}
template <class T, class T2>
void pr2(T a, T2 b) {
  cout << a << ' ' << b;
  cout << '\n';
}
template <class T, class T2, class T3>
void pr3(T a, T2 b, T3 c) {
  cout << a << ' ' << b << ' ' << c;
  cout << '\n';
}
template <class T>
void PR(T a, int n) {
  for (int i = (int)(0); i < (int)(n); i++) {
    if (i) cout << ' ';
    cout << a[i];
  }
  cout << '\n';
}
bool check(int n, int m, int x, int y) {
  return x >= 0 && x < n && y >= 0 && y < m;
}
const long long MAX = 1000000007, MAXL = 1LL << 60, dx[4] = {-1, 0, 1, 0},
                dy[4] = {0, 1, 0, -1};
void Main() {
  int n;
  cin >> n;
  int x1, x2, x3, y1, y2, y3;
  cin >> x1 >> y1 >> x2 >> y2 >> x3 >> y3;
  n -= x1 + x2 + x3;
  int z1 = x1, z2 = x2, z3 = x3;
  z1 += min(n, y1 - x1);
  n -= min(n, y1 - x1);
  z2 += min(n, y2 - x2);
  n -= min(n, y2 - x2);
  z3 += min(n, y3 - x3);
  pr3(z1, z2, z3);
}
int main() {
