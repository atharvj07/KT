import java.util.List;

import java.util.stream.Collectors;
import java.util.stream.IntStream;

public class D542 {

    private static Integer modAdd(final Integer value, final Integer N) {
        if(value >= 0) {
            return value;
        }
        return N + value;
    }

    private static Integer computeTravelTime(final Integer start, final Integer source,
                                             final Integer destination, final Integer N) {
//        if(source <= destination && destination <= N) {
            return modAdd(destination - source, N) + modAdd(source - start, N);
//        }
//        return N + destination;
    }

    private static Integer positionCount(final List<Integer> position, final Integer N) {
        if(position.isEmpty()) {
            return 0;
        }
        return (position.size() - 1) * N;
    }

    private static Integer getMinimumPath(final Integer start,
                                          final List<List<Integer>> positions,
                                          final Integer N) {
        return IntStream.range(0, N).map(source -> {
            final Stream<Integer> maximumDistances = positions.get(source).stream()
                    .map(destination -> computeTravelTime(start, source, destination, N));
            return positionCount(positions.get(source), N) +
                    maximumDistances.mapToInt(d -> d).sorted().findFirst().orElse(0);
        }).max().orElse(0);
    }

    public static void main(String[] args) throws FileNotFoundException {
//        final File inFile = new File("resources/D542.in");
        final Scanner scanner = new Scanner(System.in);

        final Integer N = scanner.nextInt();
        final Integer M = scanner.nextInt();
        final List<List<Integer>> positions =
                IntStream.range(0, N).mapToObj(i -> new LinkedList<Integer>())
                        .collect(Collectors.toList());


        IntStream.range(0, M).forEach(i -> {
            final Integer location = scanner.nextInt();
            final Integer desired = scanner.nextInt();
            positions.get(location - 1).add(desired - 1);
        });


        IntStream.range(0, N).forEach(i -> {
            System.out.println(getMinimumPath(i, positions, N));
        });
    }
}
