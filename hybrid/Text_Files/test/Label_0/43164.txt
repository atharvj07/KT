import sys
import math
from collections import defaultdict, deque
import heapq
fact = [1]
mod = 10 ** 9 + 7
for i in range(1, 1005):
	fact.append(i * fact[-1] % mod)

def comb(n, k, fact):
	x = fact[n]
	y = pow(fact[k], mod - 2, mod)
	z = pow(fact[n - k], mod - 2, mod)
	return x * (y * z) % mod

def getset(num):
	cnt = 0
	while num > 0:
		cnt += num % 2
		num = num >> 1
	return cnt

def get(s, bits):
	n = len(s)
	if n != 0 and bits == 0:
		return 1
	if n == 0 and bits == 0:
		return 1
	if n == 0:
		return 0
	if s[0] == '1':
		ans = 0
		if n - 1 >= bits:
			ans += comb(n - 1, bits, fact)
		x = get(s[1:], bits - 1)
		ans += x
		ans %= mod
		return ans
	x = get(s[1:], bits)
	x %= mod
	return x
num = sys.stdin.readline()[:-1]
m = len(num)
n = 0
for i in range(m):
	if num[i] == '1':
		n = (n << 1) + 1
	else:
		n = n << 1
k = int(sys.stdin.readline()[:-1])
if k == 0:
	print(1)
	return
dic = defaultdict(list)
for i in range(2, 1001):
	x = getset(i)
	cnt = 0
	while x != 1:
		y = getset(x)
		cnt += 1
		x = y
	dic[cnt + 1].append(i)
ans = 0
s = num
dic[0] = [1]
for j in dic[k - 1]:
	if j <= n:
		x = get(s, j)
		ans += x
		ans %= mod
if s == '1' and k == 1:
	print(0)
	return
if k == 1:
	ans -= 1
print(ans)
