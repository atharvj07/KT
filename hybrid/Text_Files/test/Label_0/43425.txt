def chan(y, lrc):
	h = []
	for x in y:
		h.append(x)
	l = int(lrc[0]) - 1
	r = int(lrc[1]) - 1
	for x in range(r - l + 1):
		if h[l] == lrc[2]:
			h[l] = lrc[3]
		l += 1
	return ''.join(h)

def run():
	indicaciones = input().strip()
	ind = indicaciones.split(' ')
	cadena = input()
	new = cadena
	for x in range(int(ind[1])):
		tmp = input().strip()
		lrc = tmp.split(' ')
		new = chan(new, lrc)
	print(new)
run()
