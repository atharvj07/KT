#include<iostream>
#include<cmath>
using namespace std;
const double eps = 1.0e-9;
int main()
{
  while(true){
    long long int n,r,ans=0;
    scanf("%lld",&n);
    if( n == 0 )break;
    long long int tmp = n>>1;
    long long int rn = (long long int)ceil(sqrt(tmp));
    for(int i = 2; i <=rn; ++i){
      int _2i1 = (i-1)<<1;
      if(n%_2i1==0)tmp+=n/_2i1;
      else tmp+=n/_2i1+1;
    }
    tmp <<= 1;
    tmp -= rn*rn;
    ans = (n+tmp)<<3;
    printf("%lld\n",ans);
  }
  return 0;
}
