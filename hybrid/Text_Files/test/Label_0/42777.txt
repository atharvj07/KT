import math
(n, p) = map(int, input().split())
items = []
for i in range(n):
	(ai, bi) = map(int, input().split())
	items += [(ai, bi)]
items.sort(key=lambda item: item[1] / item[0])
surv_time = math.inf
cur_saved = 0
cur_power = 0
for (i, item) in enumerate(items):
	(ai, bi) = item
	current_death_time = bi / ai
	if surv_time < current_death_time:
		break
	cur_saved += bi
	cur_power += ai
	if cur_power <= p:
		surv_time = math.inf
	else:
		surv_time = cur_saved / (cur_power - p)
if surv_time == math.inf:
	print(-1)
else:
	print(surv_time)
