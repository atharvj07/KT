from collections import defaultdict
n = int(input())
ar = list(map(int, input().split()))
ar.sort()
chk = defaultdict(lambda : 0)
ans = []
for i in range(0, n):
	if not chk[i]:
		ans.append(ar[i])
		cur = ar[i]
		chk[i] = 1
		for j in range(i + 1, n):
			if not chk[j] and ar[j] > cur:
				ans.append(ar[j])
				chk[j] = 1
				cur = ar[j]
(cur, anss) = (ar[0], 0)
for i in range(1, n):
	if ans[i] > cur:
		anss += 1
		cur = ans[i]
	else:
		cur = ans[i]
print(anss)
