#include <bits/stdc++.h>
using namespace std;
int main() {
  vector<int> vec;
  vector<int> res;
  int n;
  cin >> n;
  while (n--) {
    int a;
    cin >> a;
    vec.push_back(a);
  }
  int max = 0;
  res.resize(vec.size());
  for (int i = vec.size() - 1; i >= 0; i--) {
    if (vec[i] > max) {
      res[i] = 0;
      max = vec[i];
    } else {
      res[i] = max - vec[i] + 1;
    }
  }
  for (int i = 0; i < res.size(); i++) {
    cout << res[i];
    if (i != vec.size() - 1) cout << " ";
  }
  return 0;
}
