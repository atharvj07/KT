from math import gcd
(n, m) = list(map(int, input().split()))
edges = []
p = m
for i in range(1, n + 1):
	for j in range(i + 1, n + 1):
		if gcd(i, j) == 1:
			m -= 1
			edges.append([i, j])
			if m == 0:
				break
	if m == 0:
		break
if len(edges) >= p and p >= n - 1:
	print('Possible')
	for i in range(len(edges)):
		print(*edges[i])
else:
	print('Impossible')
