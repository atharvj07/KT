import java.util.*;
import java.io.*;
import java.math.*;

   public class x1004D
   {
      public static void main(String hi[]) throws Exception
      {
         BufferedReader infile = new BufferedReader(new InputStreamReader(System.in));
         StringTokenizer st = new StringTokenizer(infile.readLine());
         int N = Integer.parseInt(st.nextToken());
         int[] arr = new int[N];
         st = new StringTokenizer(infile.readLine());
         for(int i=0; i < N; i++)
            arr[i] = Integer.parseInt(st.nextToken());
         int max = arr[0];
         int[] freq = new int[N];
         for(int x: arr)
         {
            max = Math.max(max, x);
            freq[x]++;
         }
         if(freq[0] != 1)
         {
            System.out.println(-1);
            return;
         }
         int x = 0;
         for(int i=1; i < N; i++)
            if(4*i != freq[i])
            {
               x = i-1;
               break;
            }
         ArrayList<Integer> divs = findDiv(N);
         for(int R: divs)
         {
            if(x >= R)
               continue;
            int C = N/R;
            int y = R+C-x-2-max;
            if(y >= 0 && y < C)
            {
               int[] point = bash(x, y, R, C, freq);
               if(point != null)
               {
                  System.out.println(R+" "+C);
                  System.out.println(point[0]+" "+point[1]);
                  return;
               }
            }
         }
         System.out.println(-1);
      }
      public static int[] bash(int r, int c, int R, int C, int[] freq)
      {
         int[] used = new int[freq.length];
         for(int a=0; a < R; a++)
            for(int b=0; b < C; b++)
            {
               int md = Math.abs(r-a)+Math.abs(c-b);
               used[md]++;
               if(used[md] > freq[md])
                  return null;
            }
         int[] res = new int[2];
         res[0] = r+1;  res[1] = c+1;
         return res;
      }
      public static ArrayList<Integer> findDiv(int N)
      {
         //gens all divisors of N
         ArrayList<Integer> ls1 = new ArrayList<Integer>();
         ArrayList<Integer> ls2 = new ArrayList<Integer>();
         for(int i=1; i <= (int)(Math.sqrt(N)+0.00000001); i++)
            if(N%i == 0)
            {
               ls1.add(i);
               ls2.add(N/i);
            }
         Collections.reverse(ls2);
         for(int b: ls2)
            if(b != ls1.get(ls1.size()-1))
               ls1.add(b);
         return ls1;
      }
   }
