import sys
import math
get_string = lambda : sys.stdin.readline().strip()
get_int_list = lambda : list(map(int, sys.stdin.readline().strip().split()))
get_int = lambda : int(sys.stdin.readline())
(n, m, a, b) = get_int_list()
if n % m == 0:
	print(0)
else:
	res1 = n % m
	res2 = m - res1
	if res1 * b < res2 * a:
		print(res1 * b)
	else:
		print(res2 * a)
