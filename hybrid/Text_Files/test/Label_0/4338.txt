#include <bits/stdc++.h>
using namespace std;
long long w[100010];
map<long long, long long> mp;
long long Mod(long long a, long long b) { return a < b ? a : a % b + b; }
long long Exponente(long long x, long long n, long long m) {
  long long res = 1;
  while (n) {
    if (n & 1) {
      res = Mod(res * x, m);
      n--;
    }
    x = Mod(x * x, m);
    n >>= 1;
  }
  return res;
}
long long Euler(long long m) {
  long long s = m;
  if (mp.count(m)) return mp[m];
  for (long long i = 2; i * i <= m; i++) {
    if (m % i == 0) s = s / i * (i - 1);
    while (m % i == 0) m /= i;
  }
  if (m > 1) s = s / m * (m - 1);
  mp[m] = s;
  return s;
}
long long PowerTower(long long l, long long r, long long m) {
  if (l == r || m == 1) return Mod(w[l], m);
  return Exponente(w[l], PowerTower(l + 1, r, Euler(m)), m);
}
int main() {
  long long n, m, q;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) cin >> w[i];
  cin >> q;
  while (q > 0) {
    int l, r;
    cin >> l >> r;
    cout << PowerTower(l, r, m) % m << endl;
    q--;
  }
  return 0;
}
