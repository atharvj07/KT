#include<iostream>
#include<algorithm>
#include<cstring>
#include<cstdio>
#include<cmath>
using namespace std;

const int N=100000+10;
const int B=15;
const int S=(1<<B)-1;

int _r;char _c;

int read()
{
	while(!isdigit(_c=getchar()));
	_r=_c-48;
	while(isdigit(_c=getchar()))
		_r=_r*10-48+_c;
	return _r;
}

int n,b[N],ans,fl[B+1];
int sum[S+1],dp[S+1];

int count(int x)
{
	int cnt(1);
	while(!(x&1))
	{
		++cnt;
		x>>=1;
	}
	return cnt;
}

int main()
{
	ans=n=read();
	int i,j,k,x,y,a;
	for(i=1;i<n;++i)
	{
		x=read()+1;y=read()+1;
		a=read();
		b[x]^=a;
		b[y]^=a;
	}
	for(i=1;i<=n;++i)
		++fl[b[i]];
	ans-=fl[0];
	for(i=1;i<=B;++i)
	{
		ans-=fl[i]>>1;
		fl[i]&=1;
	}
	for(i=1;i<=S;++i)
	{
		k=count(j=i&(-i));
		sum[i]=sum[i^j]^k;
	}
	for(i=0;i<=S;++i)
	{
		k=S^i;
		for(j=k;j;(j-=1)&=k)if(!sum[j])
			dp[i^j]=max(dp[i^j],dp[i]+1);
	}
	for(i=1,j=0;i<=B;++i)if(fl[i])
		j^=1<<(i-1);
	printf("%d",ans-dp[j]);
	return 0;
}
