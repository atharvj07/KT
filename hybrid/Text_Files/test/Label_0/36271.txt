    final double eps = 1e-7;

    long sq( long x ){
        return x*x;
    }

    class Pt{
        long x, y;
        Pt( long _x, long _y ){
            x = _x;
            y = _y;
        }
        double dist( Pt p ){
            return sqrt( pow(x-p.x,2) + pow(y-p.y,2) );
        }
        long dist2( Pt p ){
            return sq(x-p.x) + sq(y-p.y);
        }
    }

    Comparator<Pt> cmpX = (a,b) -> Long.compare(a.x,b.x);
    Comparator<Pt> cmpY = (a,b) -> Long.compare(a.y,b.y);

    double f( Pt[] arr ){
        double ans = 1e300;
        if( arr.length <= 3 ){
            for (int i = 0; i < arr.length; i++) {
                for (int j = i + 1; j < arr.length; j++) {
                    ans = min( ans, arr[i].dist(arr[j]) );
                }
            }
        }
        else{
            Pt[] arr0 = Arrays.copyOfRange( arr, 0, arr.length/2 );
            Pt[] arr1 = Arrays.copyOfRange( arr, arr.length/2, arr.length );
            ans = min( f(arr0), f(arr1) );
            long xMid = arr[arr.length/2].x;
            ArrayList<Pt> cur = new ArrayList<>();
            for (Pt p : arr) {
                if( abs(p.x - xMid) <= ans + eps ){
                    cur.add( p );
                }
            }
            Collections.sort( cur, cmpY.thenComparing(cmpX::compare) );
            long azaza = Long.MAX_VALUE;
            for (int i = 0; i < cur.size(); i++) {
                for (int j = max(0,i-10); j < i; j++) {
                    azaza = min( azaza, cur.get(i).dist2( cur.get(j) ) );
                }
            }
            ans = min( ans, sqrt(azaza) );
        }
        return ans;
    }


    void solve(){
        int n;
        Pt[] arr;

        n = sc.nextInt();
//        n = 100000;
        arr = new Pt[n];
        for (int i = 0; i < n; i++) {
            arr[i] = new Pt(i,sc.nextInt());
//            arr[i] = new Pt(i,i);
            if( i != 0 )
                arr[i].y += arr[i-1].y;
        }

        sort( arr, cmpX.thenComparing(cmpY::compare) );
        out.println( (long)( pow(f(arr),2) + eps ) );
    }
