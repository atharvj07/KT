import java.io.*;
import java.util.*;

public class Main {

    public static void main(String[] args) {
        Reader in = new Reader();
        PrintWriter out = new PrintWriter(System.out);

        int t = in.nextInt();

        while (t-- > 0) {

            int n = in.nextInt();
            int m = in.nextInt();

            int[] arr = new int[n];

            for (int i = 0; i < n; i++) {
                arr[i] = in.nextInt();
            }


            LinkedList<Integer> stack = new LinkedList<>();
            HashSet<Integer> set = new HashSet<>();
            HashSet<Integer> set2 = new HashSet<>();

            for (int i = 0; i < m; i++) {
                int pres = in.nextInt();
                set.add(pres);
                stack.addLast(pres);
            }

            int sent = 0;
            long time = 0;

            for (int i = 0; i < n; i++) {

                if (stack.isEmpty())
                    break;

                if (stack.peek() == arr[i]) {
                    stack.pop();

                    time += (i-sent)*2 + 1;
                    sent++;

                    while (set2.contains(stack.peek())) {
                        sent++;
                        set2.remove(stack.pop());
                    }

                } else if (set.contains(arr[i])) {
                    time += 1;
                    set2.add(arr[i]);
                }

            }

            out.println(time);
        }

        out.close();
    }

    private static class Reader {
        private StringTokenizer tk;
        private BufferedReader in;

        Reader () {
            in = new BufferedReader(new InputStreamReader(System.in));
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        String next() {
            if (tk == null || !tk.hasMoreElements()) {
                try {
                    tk = new StringTokenizer(in.readLine());
                } catch (IOException ignored) {
                }
            }
            return tk.nextToken();
        }
    }
}
