#include <bits/stdc++.h>
using namespace std;
long long int n, m, res, type[14];
vector<pair<long long int, long long int> > domino;
string base[33];
char ans[33][33];
long long int mp[33][33];
int main() {
  cin >> n >> m;
  for (long long int i = 0; i < ((long long int)(n)); i++) cin >> base[i];
  for (long long int i = 0; i < ((long long int)(n)); i++)
    for (long long int j = 0; j < ((long long int)(m)); j++)
      ans[i][j] = base[i][j];
  for (long long int i = 0; i < ((long long int)(n)); i++)
    for (long long int j = 0; j < ((long long int)(m)); j++)
      for (long long int x = 0; x < ((long long int)(n)); x++)
        for (long long int y = 0; y < ((long long int)(m)); y++)
          if (i * m + j < x * m + y) {
            if (base[i][j] == base[x][y] && base[i][j] != '.') {
              domino.push_back(make_pair(i * m + j, x * m + y));
            }
          }
  long long int cnt = 0;
  memset(mp, -1, sizeof(mp));
  for (long long int i = 0; i < ((long long int)(n)); i++)
    for (long long int j = 0; j < ((long long int)(m)); j++)
      if (base[i][j] != '.' && mp[i][j] == -1) {
        for (long long int x = 0; x < ((long long int)(2)); x++)
          for (long long int y = 0; y < ((long long int)(2)); y++)
            mp[i + x][j + y] = cnt;
        cnt++;
      }
  vector<long long int> already, yet;
  for (long long int i = 0;
       i < ((long long int)(((long long int)domino.size()))); i++) {
    long long int t0 = mp[domino[i].first / m][domino[i].first % m];
    long long int t1 = mp[domino[i].second / m][domino[i].second % m];
    if (t0 == t1) {
      type[t0] = ((long long int)already.size());
      already.push_back(t0);
    }
  }
  for (long long int i = 0; i < ((long long int)(14)); i++)
    if (find(already.begin(), already.end(), i) == already.end()) {
      yet.push_back(i);
    }
  vector<long long int> color;
  for (long long int i = 0; i < ((long long int)(((long long int)yet.size())));
       i++)
    color.push_back((13 - i) % 7);
  sort(color.begin(), color.end());
  do {
    for (long long int i = 0;
         i < ((long long int)(((long long int)yet.size()))); i++)
      type[yet[i]] = color[i];
    bool ok = true;
    long long int used[7][7];
    memset(used, 0, sizeof(used));
    for (long long int i = 0; i < ((long long int)(28)); i++) {
      long long int t0 = mp[domino[i].first / m][domino[i].first % m];
      long long int t1 = mp[domino[i].second / m][domino[i].second % m];
      if (used[type[t0]][type[t1]]) {
        ok = false;
        break;
      }
      used[type[t0]][type[t1]] = used[type[t1]][type[t0]] = true;
    }
    if (ok) {
      res++;
      for (long long int i = 0; i < ((long long int)(n)); i++)
        for (long long int j = 0; j < ((long long int)(m)); j++)
          if (ans[i][j] != '.') ans[i][j] = '0' + type[mp[i][j]];
    }
  } while (next_permutation(color.begin(), color.end()));
  for (long long int i = 0;
       i < ((long long int)(((long long int)already.size()))); i++)
    res *= 7 - i;
  cout << res << endl;
  for (long long int i = 0; i < ((long long int)(n)); i++) {
    for (long long int j = 0; j < ((long long int)(m)); j++) cout << ans[i][j];
    cout << endl;
  }
}
