(h, n) = [int(i) for i in input().split(' ')]
a = [int(i) for i in input().split(' ')]
f = -1
su = h
for i in range(n):
	su = su + a[i]
	if su <= 0:
		f = i + 1
		break
if f != -1:
	print(f)
else:
	s = sum(a)
	if s >= 0:
		print(-1)
	else:
		p = [a[0]]
		low = [a[0], 0]
		for i in range(1, n):
			p.append(a[i] + p[-1])
			if low[0] > p[-1]:
				low = [p[-1], i]
		s = abs(s)
		L = abs(low[0])
		if (h - L) % s == 0:
			j = (h - L) // s
		else:
			j = (h - L) // s + 1
		ans = j * n
		su = h - j * s
		f = 0
		for i in range(n):
			su = su + a[i]
			if su <= 0:
				f = i + 1
				break
		ans += f
		print(ans)
