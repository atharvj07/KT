bool check(ll arr[],int n,int i,int j)
{
    for(ll ii=i;ii<=j;ii++)
    {
        for(ll jj=ii+1;jj<=j;jj++)
        {
            for(ll kk=jj+1;kk<=j;kk++)
            {
                if(arr[ii]<=arr[jj] && arr[jj]<=arr[kk]) 
                {
                    return false;
                }
                
                if(arr[ii]>=arr[jj] && arr[jj]>=arr[kk]) 
                {
                    return false;
                }
            }
        }
    }
    return true;
}
int main()
{
    int t;cin >> t;
    while(t--)
    {
        ll n;cin >> n;
        ll arr[n];
        for(ll i=0;i<n;i++) cin >> arr[i];
        if(n<=2)
        {
            cout << n + n - 1 << endl;
            continue;
        }
        
        
        ll ans = n + (n-1);
        for(ll k=3;k<=min(n,(ll)20);k++)
        {
            for(ll i=0;i+k-1<n;i++) 
            {
                if(check(arr,n,i,i+k-1)) ans++; 
            }
        }

        cout << ans << endl;
        
    }
