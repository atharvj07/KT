#include <bits/stdc++.h>
using namespace std;
long long fac[1000000], inv[1000000], a, MOD, m, ans[1000000], p;
int n;
long long pw(long long a, long long b, long long c) {
  long long re = 1;
  while (b) {
    if (b & 1) re = re * a % c;
    a = a * a % c;
    b >>= 1;
  }
  return re;
}
long long C(long long n, long long m, long long p) {
  return fac[n] * inv[m] % p * inv[n - m] % p;
}
int main() {
  scanf("%d%lld%lld%lld", &n, &m, &a, &MOD);
  long long p;
  for (int i = 1; i <= MOD; i++)
    if (pw(a, i, MOD) == 1) {
      p = i;
      break;
    }
  fac[0] = 1;
  for (int i = 1; i <= m; i++) fac[i] = fac[i - 1] * i % p;
  inv[0] = inv[1] = 1;
  for (int i = 2; i <= m; i++) inv[i] = (p - p / i) * inv[p % i] % p;
  for (int i = 1; i <= m; i++) inv[i] = inv[i - 1] * inv[i] % p;
  for (int i = 0; i <= m; i++) ans[i] = C(m, i, p);
  for (int i = 1; i <= m; i++) ans[i] = (ans[i - 1] + ans[i]) % p;
  for (int i = 0; i < n; i++)
    printf("%lld ", pw(a, ans[min(n - i - 1, (int)m)], MOD));
}
