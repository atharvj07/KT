from math import sqrt
for _ in range(int(input())):
	n = int(input())
	f = 0
	for i in range(2, int(sqrt(n)) + 1):
		if n % i == 0:
			if n // i != i:
				t = n // i
				for j in range(2, int(sqrt(t)) + 1):
					if t % j == 0:
						if t // j != j:
							if i != j and i != t // j:
								print('YES')
								print(i, j, t // j)
								f = 1
								break
		if f:
			break
	if not f:
		print('NO')
