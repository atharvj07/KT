#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1010;
const int MAXM = 5010;
const int MAXE = MAXM << 1;
int N, M, u[MAXM], v[MAXM];
int e[MAXE], w[MAXE], enxt[MAXE], head[MAXN], ec;
void regi(int u, int v, int w) {
  ec++;
  e[ec] = v;
  ::w[ec] = w;
  enxt[ec] = head[u];
  head[u] = ec;
}
bool nchk[MAXN], tmp[MAXN];
int dist[MAXN], process[MAXN];
bool inq[MAXN];
queue<int> Q;
void spfa() {
  int i;
  for (i = 1; i <= N; i++) {
    dist[i] = 0x23333333;
    process[i] = 0;
    inq[i] = false;
  }
  while (Q.size()) Q.pop();
  Q.push(1);
  dist[1] = 0;
  inq[1] = true;
  while (Q.size()) {
    int u = Q.front();
    Q.pop();
    inq[u] = false;
    if (!nchk[u]) continue;
    if ((process[u]++) > N) {
      printf("No\n");
      exit(0);
    }
    for (int E = head[u]; E; E = enxt[E])
      if (dist[u] + w[E] < dist[e[E]]) {
        dist[e[E]] = dist[u] + w[E];
        if (!inq[e[E]]) {
          Q.push(e[E]);
          inq[e[E]] = true;
        }
      }
  }
}
int main() {
  scanf("%d%d", &N, &M);
  int i;
  for (i = 1; i <= M; i++) scanf("%d%d", &u[i], &v[i]);
  for (i = 1; i <= N; i++) head[i] = 0;
  ec = 0;
  for (i = 1; i <= M; i++) {
    regi(u[i], v[i], 2);
    regi(v[i], u[i], -1);
  }
  for (i = 1; i <= N; i++) nchk[i] = false;
  nchk[1] = true;
  while (Q.size()) Q.pop();
  Q.push(1);
  while (Q.size()) {
    int u = Q.front();
    Q.pop();
    for (int E = head[u]; E; E = enxt[E])
      if ((w[E] == 2) && (!nchk[e[E]])) {
        nchk[e[E]] = true;
        Q.push(e[E]);
      }
  }
  for (i = 1; i <= N; i++) tmp[i] = false;
  tmp[N] = true;
  while (Q.size()) Q.pop();
  Q.push(N);
  while (Q.size()) {
    int u = Q.front();
    Q.pop();
    for (int E = head[u]; E; E = enxt[E])
      if ((w[E] == -1) && (!tmp[e[E]])) {
        tmp[e[E]] = true;
        Q.push(e[E]);
      }
  }
  for (i = 1; i <= N; i++) nchk[i] &= tmp[i];
  spfa();
  printf("Yes\n");
  for (i = 1; i <= M; i++)
    if ((nchk[u[i]]) && (nchk[v[i]]))
      printf("%d\n", dist[v[i]] - dist[u[i]]);
    else
      printf("1\n");
}
