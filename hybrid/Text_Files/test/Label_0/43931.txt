from math import sqrt
(a, b) = map(int, input().split())
(a, b) = (min(a, b), max(a, b))
rec_size = []
for i in range(1, int(sqrt(b)) + 1):
	if i <= int(sqrt(a)):
		if a % i == 0:
			rec_size.append([i, a // i])
		elif b % i == 0:
			rec_size.append([i, b // i])
	elif b % i == 0:
		rec_size.append([i, b // i])
ok = False
for i in range(int(sqrt(a + b)), 0, -1):
	if (a + b) % i == 0:
		for rec in rec_size:
			if i >= rec[0] and (a + b) // i >= rec[1]:
				ok = True
				break
	if ok:
		break
print(2 * (i + (a + b) // i))
