#include <bits/stdc++.h>
using namespace std;
int ReadInt() {
  char c;
  for (c = getchar(); c < '0' || c > '9'; c = getchar())
    ;
  int ans = c - '0';
  for (c = getchar(); c >= '0' && c <= '9'; c = getchar())
    ans = ans * 10 + c - '0';
  return ans;
}
void WriteInt(int x) {
  if (x > 9) WriteInt(x / 10);
  putchar(x % 10 + '0');
}
struct TNode {
  TNode *Left, *Right;
  int val;
  TNode(TNode *L, TNode *R, int val) : Left(L), Right(R), val(val) {}
};
const int maxn = 1e5 + 5;
TNode *tree[maxn], *curTree;
int n, root, nPos, nChain;
vector<int> adj[maxn];
int nChild[maxn], ChainID[maxn], PosInHLD[maxn], h[maxn], cHead[maxn],
    par[maxn], PosID[maxn];
int f[maxn][17], mark[maxn];
void Input() {
  n = ReadInt();
  for (int i = (1); i <= (n); ++i) {
    int tmp = ReadInt();
    if (tmp == 0)
      root = i;
    else
      adj[tmp].push_back(i);
  }
}
void DFS(int u) {
  nChild[u] = 1;
  for (int v : adj[u]) {
    par[v] = u;
    f[v][0] = u;
    for (int j = (1); j <= (16); ++j) f[v][j] = f[f[v][j - 1]][j - 1];
    h[v] = h[u] + 1;
    DFS(v);
    nChild[u] += nChild[v];
  }
}
int LCA(int u, int v) {
  if (h[u] > h[v]) swap(u, v);
  for (int j = (16); j >= (0); --j)
    if (h[v] - (1 << j) >= h[u]) v = f[v][j];
  if (u == v) return u;
  for (int j = (16); j >= (0); --j)
    if (f[u][j] != f[v][j]) u = f[u][j], v = f[v][j];
  return f[u][0];
}
void HLD(int u) {
  PosInHLD[u] = ++nPos;
  PosID[nPos] = u;
  ChainID[u] = nChain;
  if (cHead[nChain] == 0) cHead[nChain] = u;
  int spec = -1;
  for (int v : adj[u]) {
    if (spec == -1 || nChild[v] > nChild[spec]) spec = v;
  }
  if (spec != -1) HLD(spec);
  for (int v : adj[u])
    if (v != spec) {
      ++nChain;
      HLD(v);
    }
}
TNode *build(int l, int r) {
  if (l == r) return new TNode(NULL, NULL, 0);
  int mid = (l + r) >> 1;
  return new TNode(build(l, mid), build(mid + 1, r), 0);
}
TNode *update(TNode *root, int l, int r, int u) {
  if (l > u || r < u) return root;
  if (l == r) return new TNode(NULL, NULL, 1);
  int mid = (l + r) >> 1;
  TNode *L = update(root->Left, l, mid, u),
        *R = update(root->Right, mid + 1, r, u);
  return new TNode(L, R, L->val + R->val);
}
int Get(TNode *R, TNode *L, int l, int r, int u, int v) {
  if (l > v || r < u) return 0;
  if (u <= l && r <= v) return R->val - L->val;
  int mid = (l + r) >> 1;
  return Get(R->Left, L->Left, l, mid, u, v) +
         Get(R->Right, L->Right, mid + 1, r, u, v);
}
int GetPos(TNode *R, TNode *L, int u, int v, int k) {
  int l = u, r = v, res = -1;
  while (l <= r) {
    int mid = (l + r) >> 1;
    if (mid - u + 1 - Get(R, L, 1, n, u, mid) >= k)
      res = mid, r = mid - 1;
    else
      l = mid + 1;
  }
  assert(res != -1);
  return PosID[res];
}
int GetCnt(int u, int a, int &k, int year, int typ = 0) {
  int achain = ChainID[a], uchain = ChainID[u];
  while (true) {
    if (uchain == achain) {
      int tmp = Get(curTree, tree[year], 1, n, PosInHLD[a], PosInHLD[u]);
      tmp = PosInHLD[u] - PosInHLD[a] + 1 - tmp;
      if (typ)
        k += tmp;
      else {
        if (k <= tmp)
          return GetPos(curTree, tree[year], PosInHLD[a], PosInHLD[u],
                        tmp - k + 1);
        else {
          k -= tmp;
          return -1;
        }
      }
      return 0;
    } else {
      int tmp =
          Get(curTree, tree[year], 1, n, PosInHLD[cHead[uchain]], PosInHLD[u]);
      tmp = PosInHLD[u] - PosInHLD[cHead[uchain]] + 1 - tmp;
      if (typ)
        k += tmp;
      else {
        if (k <= tmp)
          return GetPos(curTree, tree[year], PosInHLD[cHead[uchain]],
                        PosInHLD[u], tmp - k + 1);
        else
          k -= tmp;
      }
      u = par[cHead[uchain]];
      uchain = ChainID[u];
    }
  }
  return k;
}
void Solve() {
  DFS(root);
  HLD(root);
  int m = ReadInt();
  tree[0] = build(1, n);
  for (int i = (1); i <= (m); ++i) {
    int typ = ReadInt();
    tree[i] = tree[i - 1];
    if (typ == 1) {
      int u = ReadInt();
      mark[u] = i;
      tree[i] = update(tree[i], 1, n, PosInHLD[u]);
    } else {
      curTree = tree[i];
      int u = ReadInt(), v = ReadInt(), k = ReadInt(), year = ReadInt();
      if (mark[u] <= year) ++k;
      int lca = LCA(u, v);
      int tmp = GetCnt(u, lca, k, year);
      if (tmp != -1 && tmp != u && tmp != v) {
        WriteInt(tmp);
        putchar('\n');
        continue;
      }
      if (mark[lca] <= year) ++k;
      int cnt = 0;
      GetCnt(v, lca, cnt, year, 1);
      if (cnt < k) {
        puts("-1");
        continue;
      }
      cnt = cnt - k + 1;
      tmp = GetCnt(v, lca, cnt, year);
      if (tmp == v)
        puts("-1");
      else {
        WriteInt(tmp);
        putchar('\n');
      }
    }
  }
}
int main() {
  iostream::sync_with_stdio(false);
  cin.tie(0);
  Input();
  Solve();
  return 0;
}
