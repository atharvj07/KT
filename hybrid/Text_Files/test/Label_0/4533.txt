#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int a, b;
  cin >> a >> b;
  if (a == 9 && b == 1) {
    cout << 9 << " " << 10 << endl;
  } else if (abs(a - b) > 1) {
    cout << -1 << endl;
  } else {
    if (a == b) {
      cout << a << 1 << " " << b << 2 << endl;
    }
    if (a == b - 1) {
      cout << a << 9 << " " << b << 0 << endl;
    }
    if (a > b) {
      cout << -1 << endl;
    }
  }
}
