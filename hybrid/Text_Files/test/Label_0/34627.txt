import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {

    public static void main(String[] args) throws IOException {
        new Main().run();
    }

    private void run() throws IOException {

        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(reader.readLine());

        /*
         * Читаем какие элементы есть вообще в руке
         * */
        String[] line = reader.readLine().split("\\s");
        final boolean[] handPlaces = new boolean[n+1];

        for (int i = 0; i < n; i++) {
            handPlaces[Integer.parseInt(line[i])] = true;
        }


        /*
         * Читаем все элементы из колоды, сохраняя их места, а также пробуем найти под конец последовательность,
         * чтобы можно было попробовать сразу собрать последовательность если все доступны
         * */
        line = reader.readLine().split("\\s");
        boolean seq = false;
        int index = 1;
        final int[] deckPlaces = new int[n+1];

        for (int i = 0; i < n; i++) {
            int val = Integer.parseInt(line[i]);
            deckPlaces[val] = i;

            if(val == 1){
                index++;
                seq = true;
            }else if(seq && val == index){
                index++;
            }else if(seq){
                seq = false;
            }
        }

        /*
         * Если в конце есть последовательность и все элементы подтянутся,
         * то выводим сколько времени потребуется на ее создание
         * */
        if(seq && available(handPlaces, deckPlaces, index)){
            System.out.println(n - (index - 1));
            return;
        }

        /*
         * Иначе бежим по всем элементам от 1 до n и смотрим сколько надо будет подождать элемент,
         * которого нет в колоде, если начать собирать последовательность прямо сейчас
         * В итоге в переменной delay будет храниться максимальное время задержки, после которого можно начинать
         * собирать последовательность и все элементы к тому времени подтянутся
         * */
        int delay = 0;
        for (int i = 1; i <= n; i++) {
            if(!handPlaces[i] && deckPlaces[i] >= (i - 1 + delay)){
                delay = deckPlaces[i] - (i - 2);
            }
        }

        /*
         * Осталось вывести время на постройку последовательности (n) и макс задержку на ожидание элементов
         * */
        System.out.println(n + delay);

    }

    /*
     * Копия внутренней процедуры, чтобы если пришлось ждать какой-то элемент, то это значит не получится
     * эффективно использовать последовательность на конце колоды
     *
     * Причём время отсчитываем не с 1, а с i - start, т.к. start элементов уже поставлены
     * */
    private boolean available(final boolean[] handPlaces, final int[] deckPlaces, final int start){

        for (int i = start; i < handPlaces.length; i++) {
            if(!handPlaces[i] && deckPlaces[i] >= (i - start)){
                return false;
            }
        }

        return true;
    }



}
