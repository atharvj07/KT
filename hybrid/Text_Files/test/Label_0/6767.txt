#include <bits/stdc++.h>
using namespace std;
void go(string& a, long long idx, long long n) {
  for (long long i = 0; i <= idx; i++) {
    if (a[i] == '0') {
      a[i] = '1';
    } else {
      a[i] = '0';
    }
  }
  long long i = 0;
  long long j = idx;
  while (i <= j) {
    if (a[i] == a[j]) {
      i++;
      j--;
      continue;
    }
    if (a[i] == '0') {
      a[i] = '1';
    } else {
      a[i] = '0';
    }
    if (a[j] == '0') {
      a[j] = '1';
    } else {
      a[j] = '0';
    }
    i++;
    j--;
  }
}
int32_t main() {
  long long t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    string a, b;
    cin >> a >> b;
    long long j = n - 1;
    vector<long long> ans;
    while (j >= 0) {
      if (a[j] == b[j]) {
        j--;
      } else {
        if (b[j] == a[0]) {
          ans.push_back(1);
          ans.push_back(j + 1);
          if (a[0] == '0') {
            a[0] = '1';
          } else {
            a[0] = '0';
          }
        } else {
          ans.push_back(j + 1);
        }
        go(a, j, n);
        j--;
      }
    }
    cout << ans.size() << " ";
    for (long long i = 0; i < ans.size(); i++) {
      cout << ans[i] << " ";
    }
    cout << "\n";
  }
}
