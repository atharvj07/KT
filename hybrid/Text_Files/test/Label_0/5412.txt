#include<algorithm>
#include<iostream>
#include<string>
#include<cstdlib>
#include<cstring>
#include<cmath>
using namespace std;
int main(){
  int i,j;
  int a[1001],b[1002001],m,n;
  int mx;
  a[0]=0;
  while(cin>>n>>m&&n+m){
    for(i=1;i<n+1;i++)
      cin>>a[i];
    for(i=0;i<n+1;i++)
      for(j=0;j<n+1;j++)
	b[(n+1)*i+j]=a[i]+a[j];
    n=(n+1)*(n+1);
    sort(b,b+n);
    mx=0;
    j=n-1;
    for(i=0;i<n&&b[i]<=m;i++){
      for(;b[i]+b[j]>m;j--);
      mx=max(mx,b[i]+b[j]);
    }
    cout<<mx<<endl;
  }
  return 0;
}
