n = int(input())
s = input()
nuq1 = 0
nuq2 = 0
sum1 = 0
sum2 = 0
for i in range(n):
	if s[i] == '?' and i < n // 2:
		nuq1 += 1
	elif s[i] == '?' and i >= n // 2:
		nuq2 += 1
	elif i < n // 2:
		sum1 += int(s[i])
	else:
		sum2 += int(s[i])
if sum1 - sum2 == 9 * ((nuq2 - nuq1) // 2):
	print('Bicarp')
else:
	print('Monocarp')
