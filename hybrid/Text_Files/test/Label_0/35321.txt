import java.util.Scanner;

public class NewYearCandies {

    public static void main(String[] argv) {
        Scanner sc = new Scanner(System.in);
        int sum = sc.nextInt();
        if(sum==2){
            System.out.println(1);
            System.out.println(2);
            return;
        }
        int upper_n = (int)Math.ceil((Math.sqrt(1+8*sum)-1)/2);
        int upperSum = upper_n*(upper_n+1)/2;

        if(upperSum == sum) {
            System.out.println(upper_n);
            for(int i=1; i<=upper_n; i++) System.out.print(i+" ");
        }else{
            int delta = upperSum - sum;
            System.out.println(upper_n-1);
            for(int i=1; i<=upper_n;i++){
                if(i != delta) System.out.print(i+" ");
            }
        }
    }
}
