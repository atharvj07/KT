from collections import defaultdict
import sys

def find_cum(numbers):
	cumsum = [0]
	sums = 0
	for n in numbers:
		sums += n
		cumsum.append(sums)
	return cumsum

def solve(cumsum, query_l, query_r):
	l = query_l - 1
	r = query_r
	ans = cumsum[r] - cumsum[l]
	print(ans)
trash = sys.stdin.readline()
numbs = [int(x) for x in sys.stdin.readline().strip().split(' ')]
sorted_numbers = sorted(numbs)
numbs = find_cum(numbs)
num_questions = int(sys.stdin.readline().strip())
sorted_numbers = find_cum(sorted_numbers)
for i in range(0, num_questions):
	question = [int(x) for x in sys.stdin.readline().strip().split(' ')]
	inp = []
	if question[0] == 1:
		inp = numbs
	else:
		inp = sorted_numbers
	solve(inp, question[1], question[2])
