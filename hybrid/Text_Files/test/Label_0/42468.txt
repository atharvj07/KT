def Getprime(goal):
	ans = []
	if goal % 2 == 0:
		cnt = 0
		while goal % 2 == 0:
			goal //= 2
			cnt += 1
		ans.append([2, cnt])
	i = 3
	while i * i <= goal:
		if goal % i == 0:
			cnt = 0
			while goal % i == 0:
				goal //= i
				cnt += 1
				ans.append([i, cnt])
		i += 2
	if goal != 1:
		ans.append([goal, 1])
	return ans

def find(g, p):
	ans = 0
	t = p
	while t <= g:
		ans += g // t
		t *= p
	return ans
ans = input().split()
n = int(ans[0])
b = int(ans[1])
temp = Getprime(b)
sz = len(temp)
ans = find(n, temp[0][0]) // temp[0][1]
i = 1
while i < sz:
	now = find(n, temp[i][0]) // temp[i][1]
	if now < ans:
		ans = now
	i += 1
print(ans)
