#include <bits/stdc++.h>
using namespace std;
int n, a, b, c, d, deg[5005];
vector<int> g[5005];
int head[5005], tail[5005], cnt[5005], slx;
int fa, fb, fc, fd;
int f[5005], slf;
int PN;
int dp[2][5005];
void dfs(int u) {
  head[u] = ++slx;
  for (int i = 0; i < deg[u]; ++i) {
    int v = g[u][i];
    dfs(v);
    cnt[u] += cnt[v];
  }
  if (deg[u] == 0) cnt[u]++;
  tail[u] = slx;
}
bool contain(int u, int v) {
  if (head[u] <= head[v] && tail[v] <= tail[u]) return 1;
  return 0;
}
bool solve(int a, int b, int other_a, int other_b) {
  memset(dp, 0, sizeof(dp));
  dp[0][0] = 1;
  for (int i = 1; i <= slf; ++i) {
    for (int j = 0; j <= cnt[1]; ++j) dp[i % 2][j] = 0;
    for (int j = 0; j <= cnt[1]; ++j)
      if (dp[1 - i % 2][j]) {
        dp[i % 2][j] = 1;
        dp[i % 2][j + f[i]] = 1;
      }
  }
  for (int j = cnt[1]; j >= 0; --j)
    if (dp[slf % 2][j]) {
      dp[slf % 2][j + other_a] = 2;
      dp[slf % 2][j + other_b] = 2;
    }
  for (int j = 0; j <= cnt[1]; ++j)
    if (dp[slf % 2][j] == 2)
      if (j >= PN - (a - 1) - (b - 1) && j <= PN) return 1;
  return 0;
}
int main() {
  cin.tie(0);
  cout.tie(0);
  ios::sync_with_stdio(0);
  cin >> n;
  cin >> a >> b >> c >> d;
  for (int i = 2; i <= n; ++i) {
    int u;
    cin >> u;
    deg[u]++;
    g[u].push_back(i);
  }
  dfs(1);
  for (int i = 0; i < deg[1]; ++i) {
    int u = g[1][i];
    if (contain(u, a))
      fa = cnt[u];
    else if (contain(u, b))
      fb = cnt[u];
    else if (contain(u, c))
      fc = cnt[u];
    else if (contain(u, d))
      fd = cnt[u];
    else
      f[++slf] = cnt[u];
  }
  if (cnt[1] % 2) {
    cout << "NO";
    return 0;
  }
  PN = cnt[1] / 2 - 1;
  if (solve(fa, fc, fb, fd) && solve(fb, fd, fa, fc))
    cout << "YES";
  else
    cout << "NO";
}
