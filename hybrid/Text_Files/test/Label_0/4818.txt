const long long N = 100009;
struct node {
  int l, r, val;
  node() { l = r = val = 0; }
  node(int _l, int _r, int _val) { l = _l, r = _r, val = _val; }
} t[20 * N];
int root[N], a[N], cnt, k;
void build(int cur, int b, int e) {
  if (b == e) {
    t[cur] = node(0, 0, 0);
    return;
  }
  int left, right, mid = (b + e) / 2;
  t[cur].l = left = ++cnt;
  t[cur].r = right = ++cnt;
  build(left, b, mid);
  build(right, mid + 1, e);
  t[cur].val = t[left].val + t[right].val;
}
void upd(int pre, int cur, int b, int e, int i, int v) {
  if (i < b || i > e) return;
  if (b == e) {
    t[cur].val += v;
    return;
  }
  int left, right, mid = (b + e) / 2;
  if (i <= mid) {
    t[cur].r = right = t[pre].r;
    t[cur].l = left = ++cnt;
    upd(t[pre].l, t[cur].l, b, mid, i, v);
  } else {
    t[cur].l = left = t[pre].l;
    t[cur].r = right = ++cnt;
    upd(t[pre].r, t[cur].r, mid + 1, e, i, v);
  }
  t[cur].val = t[left].val + t[right].val;
}
int query(int pre, int cur, int b, int e, int x, int y, int k) {
  if (b > y || e < x) {
    return 0;
  }
  if (x <= b && y >= e) return t[cur].val;
  int cnt = t[t[cur].l].val - t[t[pre].l].val;
  int mid = (b + e) / 2;
  return query(t[pre].l, t[cur].l, b, mid, x, y, k) +
         query(t[pre].r, t[cur].r, mid + 1, e, x, y, k);
}

queue<int> Q[100009];
int main() {
  ios_base::sync_with_stdio(0);

  int i, j, n, m, q, x, l, r, c = 0;

  cnt = 0;
  root[0] = ++cnt;
  build(root[0], 1, n);
  for (i = 1; i <= n; i++) {
    root[i] = ++cnt;
    int xx;
    cin >> xx;
    Q[xx].push(i);
    if (Q[xx].size() > k) {
      upd(root[i - 1], root[i], 1, n, Q[xx].front(), 1);
      Q[xx].pop();
    } else {
      root[i] = root[i - 1];
    }
  }
  cin >> q;
  int xx = 0, yy = 0;
  int last = 0;
  while (q--) {
    cin >> l >> r;
    xx = ((l + last) % n) + 1;
    yy = ((r + last) % n) + 1;
    if (xx > yy) {
      swap(xx, yy);
    }
    l = query(root[yy - 1], root[yy], 1, n, xx, yy, 0);
    long long ans = t[root[yy]].val - t[root[xx - 1]].val;
    last = ((yy - xx) + 1) - l;
    cout << last << endl;
  }
}
