#include <bits/stdc++.h>
using namespace std;
using LL = long long;
int main() {
  int n;
  scanf("%d", &n);
  LL A = 0, B = 0;
  for (int i = 1; i <= n; ++i) {
    LL x;
    scanf("%lld", &x);
    if (i & 1) {
      A += ((x >> 1));
      B += ((x >> 1) + (x & 1));
    } else {
      A += ((x >> 1) + (x & 1));
      B += ((x >> 1));
    }
  }
  printf("%lld\n", min(A, B));
  return (0);
}
