import sys
readline = sys.stdin.readline

def solve(K):
	Ans = []
	if K > M - 1:
		K -= M - 1
		Ans.append(f'{M - 1} R')
	else:
		Ans.append(f'{K} R')
		return Ans
	if K > M - 1:
		K -= M - 1
		Ans.append(f'{M - 1} L')
	else:
		Ans.append(f'{K} L')
		return Ans
	cycle = 4 * M - 3
	if K > cycle * (N - 1):
		K -= cycle * (N - 1)
		for _ in range(N - 1):
			Ans.append(f'{M - 1} DRU')
			Ans.append('1 D')
			Ans.append(f'{M - 1} L')
	else:
		rot = K // cycle
		K -= rot * cycle
		for _ in range(rot):
			Ans.append(f'{M - 1} DRU')
			Ans.append('1 D')
			Ans.append(f'{M - 1} L')
		if K:
			if K < 3 * (M - 1):
				tc = K // 3
				if tc:
					Ans.append(f'{tc} DRU')
				K -= tc * 3
				if K:
					Ans.append('1 ' + 'DRU'[:K])
				return Ans
			else:
				K -= 3 * (M - 1)
				Ans.append(f'{M - 1} DRU')
			if not K:
				return Ans
			Ans.append('1 D')
			K -= 1
			if K:
				Ans.append(f'{K} L')
		return Ans
	if K:
		Ans.append(f'{K} U')
	return Ans

def solve2(K):
	Ans = []
	if K > N - 1:
		K -= N - 1
		Ans.append(f'{N - 1} D')
	else:
		Ans.append(f'{K} D')
		return Ans
	if K:
		Ans.append(f'{K} U')
	return Ans
(N, M, K) = map(int, readline().split())
if 4 * N * M - 2 * N - 2 * M < K:
	sys.stdout.write('NO\n')
else:
	sys.stdout.write('YES\n')
	if M != 1:
		Ans = solve(K)
	else:
		Ans = solve2(K)
	print(len(Ans))
	print('\n'.join(Ans))
