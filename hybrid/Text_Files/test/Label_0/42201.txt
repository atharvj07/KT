for tc in range(int(input())):
	n = int(input())
	lisT = []
	ans = []
	leN = 0
	for i in range(n):
		a = int(input())
		while 1:
			if a == 1:
				lisT.append(1)
				ans.append('.'.join((str(k) for k in lisT)))
				break
			if a == lisT[-1] + 1:
				lisT[-1] += 1
				ans.append('.'.join((str(k) for k in lisT)))
				break
			lisT.pop()
	for i in ans:
		print(i)
