#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 10;
int n, nxt[MAXN], sol[MAXN], m;
set<pair<pair<int, int>, int> > tr[MAXN];
queue<int> Q;
map<pair<pair<int, int>, int>, int> id;
void solve() {
  assert(!Q.empty());
  int i = Q.front();
  Q.pop();
  pair<pair<int, int>, int> T = *tr[i].begin();
  sol[m++] = id[T];
  int a = T.first.first, b = T.first.second, c = T.second;
  for (auto y : {a, b, c}) {
    tr[y].erase(T);
    if (tr[y].size() == 1) Q.push(y);
  }
  if (m == n - 2) {
    nxt[a] = b;
    nxt[b] = c;
    nxt[c] = a;
  } else {
    solve();
    if (b == i) swap(a, b);
    if (c == i) swap(a, c);
    if (nxt[b] == c) {
      nxt[b] = a;
      nxt[a] = c;
    } else {
      nxt[c] = a;
      nxt[a] = b;
    }
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 0; i < n - 2; i++) {
      int a, b, c;
      cin >> a >> b >> c;
      if (a > b) swap(a, b);
      if (b > c) swap(b, c);
      if (a > b) swap(a, b);
      assert(a <= b && b <= c);
      pair<pair<int, int>, int> x = {{a, b}, c};
      for (auto y : {a, b, c}) tr[y].insert(x);
      id[x] = i + 1;
    }
    for (int i = 1; i <= n; i++)
      if (tr[i].size() == 1) Q.push(i);
    m = 0;
    solve();
    int x = 1;
    for (int i = 0; i < n; i++) {
      cout << x << " ";
      x = nxt[x];
    }
    cout << "\n";
    for (int i = 0; i < m; i++) cout << sol[i] << " ";
    cout << "\n";
    while (!Q.empty()) Q.pop();
    for (int i = 1; i <= n; i++) tr[i].clear();
    id.clear();
  }
  return 0;
}
