#include <bits/stdc++.h>
using namespace std;
const long long NR = 1e6 + 10;
void Min(long long &x, long long y) { x = min(x, y); }
void Max(long long &x, long long y) { x = max(x, y); }
long long n, m, len;
char s[NR];
long long dep[NR];
long long tr[NR][5];
long long tot = 1;
struct Nd {
  long long t[5][5];
  void init() { memset(t, 0x3f, sizeof(t)); }
} w;
Nd operator*(Nd a, Nd b) {
  Nd x;
  x.init();
  for (long long i = 1; i <= 4; i++)
    for (long long j = 1; j <= 4; j++)
      for (long long k = 1; k <= 4; k++) Min(x.t[i][j], a.t[i][k] + b.t[k][j]);
  return x;
}
Nd ksm(Nd a, long long b) {
  b--;
  Nd res = a;
  while (b) {
    if (b & 1) res = res * a;
    a = a * a;
    b >>= 1;
  }
  return res;
}
void ins(long long l, long long r) {
  long long x = 1;
  for (long long i = l; i <= r; i++) {
    if (!tr[x][s[i] - 'A' + 1]) tr[x][s[i] - 'A' + 1] = ++tot;
    dep[tr[x][s[i] - 'A' + 1]] = dep[x] + 1, x = tr[x][s[i] - 'A' + 1];
  }
}
void query(long long l, long long r) {
  long long x = 1;
  for (long long i = l; i <= r; i++) {
    long long p = s[i] - 'A' + 1;
    x = tr[x][p];
    for (long long j = 1; j <= 4; j++)
      if (!tr[x][j]) Min(w.t[s[l] - 'A' + 1][j], dep[x]);
  }
}
long long read() {
  long long x = 0, f = 1;
  char ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') {
    x = (x << 3) + (x << 1) + (ch ^ 48);
    ch = getchar();
  }
  return x * f;
}
signed main() {
  n = read();
  scanf("%s", s + 1);
  w.init();
  if (n == 1) {
    puts("1");
    return 0;
  }
  len = m = strlen(s + 1);
  len = min(len, 10ll);
  for (long long i = 1; i <= m; i++) ins(i, min(i + len - 1, m));
  for (long long i = 1; i <= m; i++) query(i, min(i + len - 1, m));
  long long l = 1, r = n, ans = -1;
  while (l <= r) {
    long long mid = (l + r >> 1);
    Nd p = ksm(w, mid);
    long long mn = 1e18;
    for (long long i = 1; i <= 4; i++)
      for (long long j = 1; j <= 4; j++) Min(mn, p.t[i][j]);
    if (mn < n)
      ans = mid, l = mid + 1;
    else
      r = mid - 1;
  }
  printf("%lld\n", ans + 1);
  return 0;
}
