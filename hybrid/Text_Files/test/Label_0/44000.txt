import io
import os
import sys
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline

def printd(*args, **kwargs):
	print(*args, **kwargs)
	pass

def get_str():
	return input().decode().strip()

def rint():
	return map(int, input().split())

def oint():
	return int(input())
t = oint()
for _ in range(t):
	n = oint()
	c = dict()
	for i in range(n):
		s = get_str()
		for cc in s:
			if cc in c:
				c[cc] += 1
			else:
				c[cc] = 1
	for cc in c.values():
		if cc % n:
			print('NO')
			break
	else:
		print('YES')
