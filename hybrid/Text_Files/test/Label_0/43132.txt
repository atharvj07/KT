t = int(input())
while t > 0:
	line = input()
	n = int(line.split(' ')[0])
	k = int(line.split(' ')[1])
	i = 1
	p = i
	while k > p:
		i += 1
		p += i
	if k == p:
		print('a' * (n - i - 1) + 'bb' + 'a' * (i - 1))
	else:
		diff = p % k
		print('a' * (n - i - 1) + 'b' + 'a' * diff + 'b' + 'a' * (i - 1 - diff))
	t -= 1
