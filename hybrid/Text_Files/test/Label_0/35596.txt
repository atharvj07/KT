import java.util.*;
import java.io.*;
import java.math.BigInteger;

public class E {

	static BigInteger power(int x, int y) {
		if (y == 0)
			return new BigInteger("1");
		else if (y % 2 == 0)
			return power(x, y / 2).multiply(power(x, y / 2));
		else
			return new BigInteger(x + "").multiply(power(x, y / 2).multiply(power(x, y / 2)));
	}

	static BigInteger help(char[] a) {
		BigInteger ans = new BigInteger("0");
		int j = 0;
		for (int i = a.length - 1; i >= 0; i--) {
			if (a[i] != '0')
				ans = ans.add(power(2, j));
			j++;
		}
		return ans;
	}

	public static void main(String[] args) throws NumberFormatException, IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		PrintWriter pw = new PrintWriter(System.out);
		Scanner sc = new Scanner(System.in);

		char[] s = br.readLine().toCharArray();
		int n = s.length;
		Stack<Integer> idxofone = new Stack<>();
		for (int i = 0; i < n; i++) {
			if (s[i] == '1')
				idxofone.push(i);
			else if (!idxofone.isEmpty())
				s[idxofone.pop()] = '1';
			s[i] = '0';

		}
		pw.println(s);
		pw.flush();
		pw.close();
	}

	static class pair {
		int x;
		int y;

		public pair(int d, int u) {
			x = d;
			y = u;
		}

	}
}
