#include <bits/stdc++.h>
int main() {
  unsigned long t;
  std::cin >> t;
  std::vector<unsigned long> countForLevel;
  for (; t > 0; t--) {
    unsigned long n;
    long d;
    std::cin >> n >> d;
    countForLevel.clear();
    countForLevel.push_back(1);
    n--;
    while (n > 0) {
      unsigned long newDepth = countForLevel.size();
      unsigned long max = 2 * countForLevel.at(newDepth - 1);
      unsigned long delta = (n <= max ? n : max);
      countForLevel.push_back(delta);
      d -= newDepth * delta;
      n -= delta;
    }
    if (d < 0) {
      std::cout << "NO\n";
    } else {
      unsigned long ud = (unsigned long)d;
      for (unsigned long row = countForLevel.size() - 1; ud > 0 && row > 0;
           row--) {
        while (countForLevel.at(row) > 1 && ud > 0) {
          unsigned long M = countForLevel.size();
          countForLevel.at(row)--;
          if (ud + row >= M) {
            countForLevel.push_back(1);
            ud -= (M - row);
          } else {
            countForLevel.at(row + ud)++;
            ud = 0;
          }
        }
      }
      if (ud > 0) {
        std::cout << "NO\n";
      } else {
        std::cout << "YES\n";
        long currParent = 1;
        long firstChild = 2;
        std::string prefix = "";
        for (unsigned long i = 1; i < countForLevel.size(); i++) {
          unsigned long N = countForLevel.at(i);
          long childCount = 0;
          long nextChild = firstChild;
          for (unsigned long z = 0; z < N; z++) {
            std::cout << prefix << currParent;
            prefix = " ";
            nextChild++;
            childCount++;
            if (childCount == 2) {
              childCount = 0;
              currParent++;
            }
          }
          currParent = firstChild;
          firstChild = nextChild;
        }
        std::cout << "\n";
      }
    }
  }
}
