#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
const int mod = 998244353;
long long bigmod(long long a, long long b) {
  if (b == 0) {
    return 1;
  }
  long long tmp = bigmod(a, b / 2);
  long long ans = tmp * tmp % mod;
  if (b & 1) {
    ans = ans * a % mod;
  }
  return ans;
}
long long f[N], v[N];
void init() {
  v[0] = v[1] = f[0] = f[1] = 1;
  for (int i = 2; i < N; i++)
    f[i] = 1LL * f[i - 1] * i % mod,
    v[i] = 1LL * v[mod % i] * (mod - mod / i) % mod;
  for (int i = 2; i < N; i++) v[i] = 1LL * v[i - 1] * v[i] % mod;
}
long long C(long long n, long long m) {
  if (m < 0 || m > n) return 0;
  return 1LL * f[n] * v[m] % mod * v[n - m] % mod;
}
long long add(long long a, long long b) {
  a = (a % mod + mod) % mod;
  b = (b % mod + mod) % mod;
  return (a + b) % mod;
}
long long mut(long long a, long long b) { return (a % mod * b % mod) % mod; }
long long s2(long long p, long long k) {
  long long ans = 0;
  for (int i = 0; i <= k; i++) {
    long long tmp = mut(C(k, i), bigmod(k - i, p));
    if (i & 1) {
      ans = add(ans, -tmp);
    } else {
      ans = add(ans, tmp);
    }
  }
  return ans;
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  init();
  int n, k;
  long long ans;
  cin >> n >> k;
  if (k >= n)
    ans = 0;
  else if (k == 0)
    ans = f[n];
  else
    ans = mut(s2(n, n - k), C(n, n - k)) * 2 % mod;
  cout << ans;
}
