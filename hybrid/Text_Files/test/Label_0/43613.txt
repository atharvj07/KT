k = int(input())
s = input()
(n, s1) = (len(s), s[:k])
p = n // k
if k > n:
	print('1' + (k - 1) * '0')
else:
	if n % k:
		p += 1
	s2 = s1 * p
	if len(s2) > len(s):
		s1 = '1' + (k - 1) * '0'
		print(p * s1)
	elif s2 > s:
		print(s2)
	else:
		f = 0
		for i in range(k - 1, -1, -1):
			if s1[i] != '9':
				f = 1
				ch = s1[i]
				s1 = s1[:i] + chr(ord(ch) + 1) + (k - 1 - i) * '0'
				break
		s2 = s1 * p
		if f:
			print(s2)
		else:
			s1 = '1' + (k - 1) * '0'
			if p * k == n:
				p += 1
			print(p * s1)
