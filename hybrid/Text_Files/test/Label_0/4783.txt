#include <bits/stdc++.h>
using namespace std;
int main() {
  int f, n, t;
  cin >> f >> n >> t;
  vector<int> cnt(n, 0);
  int ans = 0;
  for (int i = 0; i < f; i++) {
    string str;
    cin >> str;
    for (int j = 0; j < n; j++) {
      if (str[j] == 'Y') cnt[j]++;
    }
  }
  ans = count_if(cnt.begin(), cnt.end(), [&t](const int i) { return i >= t; });
  cout << ans << endl;
  return 0;
}
