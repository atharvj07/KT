(n, m) = map(int, input().split())
a = [int(i) for i in input().split()]
b = [int(i) for i in input().split()]
a_even = 0
a_odd = 0
b_even = 0
b_odd = 0
for i in range(n):
	if int(a[i] % 2) == 0:
		a_even += 1
	else:
		a_odd += 1
for i in range(m):
	if int(b[i] % 2) == 0:
		b_even += 1
	else:
		b_odd += 1
answer = min(a_even, b_odd) + min(a_odd, b_even)
print(answer)
