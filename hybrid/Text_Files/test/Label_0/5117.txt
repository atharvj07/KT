#include <bits/stdc++.h>
using namespace std;
const int N = 2000 + 5;
int main() {
  int n, nn, a[N];
  scanf("%d", &n);
  nn = 2 * n;
  for (int i = 1; i <= nn; ++i) scanf("%d", &a[i]);
  sort(a + 1, a + 1 + nn);
  bool flag = true;
  for (int i = 1; i <= n; ++i)
    if (a[i] != a[nn - i + 1]) {
      flag = false;
      break;
    }
  if (flag)
    printf("-1");
  else
    for (int i = 1; i <= nn; ++i) printf("%d ", a[i]);
}
