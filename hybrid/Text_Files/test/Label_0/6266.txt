#include <bits/stdc++.h>
using namespace std;
vector<pair<pair<int, int>, int> > v;
int main() {
  int i, j, k, a, x, y, m, p, b, n = 0, ans = 0;
  scanf("%d", &n);
  for (i = 1; i <= n; i++) {
    scanf("%d%d%d%d%d", &k, &a, &x, &y, &m);
    p = 0;
    b = 0;
    for (j = 1; j <= k; j++) {
      if (a < b) p++;
      if (v.size() <= 200000) v.push_back(make_pair(make_pair(p, a), i));
      b = a;
      a = ((long long)a * x + y) % m;
    }
    if (p > ans) ans = p;
  }
  printf("%d\n", ans);
  if (v.size() <= 200000) {
    sort(v.begin(), v.end());
    for (i = 0; i < v.size(); i++)
      printf("%d %d\n", v[i].first.second, v[i].second);
  }
  return 0;
}
