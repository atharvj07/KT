def draw(n):
	MOD = 10 ** 9 + 7
	INV = 47619048
	ADD = [6, -30, -18]

	def fpow(x, n):
		r = 1
		while n > 1:
			if n & 1:
				r = r * x % MOD
			x = x * x % MOD
			n >>= 1
		return x * r % MOD
	return (fpow(2, n + 3) + 14 * (2 * (n & 1) - 1) + ADD[n % 3]) * INV % MOD
t = int(input())
for i in range(t):
	n = int(input())
	print(draw(n))
