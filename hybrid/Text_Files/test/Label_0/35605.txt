
import java.util.*;

public class E1189D {
    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        pair2 [] p = new pair2[n];
        ArrayList<Integer> [] a = new ArrayList[n];
        
        for (int i=0; i< n ; i++)
            a[i] = new ArrayList();

        for (int i = 0; i < n -1 ; i++) {
            int s1= sc.nextInt() -1;
            int e1 =sc.nextInt() -1;
            p[i] = new pair2(s1, e1);
            a[s1].add(e1);
            a[e1].add(s1);
        }
        
        for (int i=0; i< n ; i++) {
            if (a[i].size() == 2) {
                System.out.println("NO");
                System.exit(0);
            }
        }
        System.out.println("YES");
    }
 
}

class pair2 {
    int s;
    int e;
    public pair2(int s1, int e1) {
        s = s1;
        e = e1;
    }
}
