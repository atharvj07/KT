#include<iostream>
#include<string>

const char c[5]={'M','A','R','C','H'};

using namespace std;

int main(){
  int n;
  cin >> n;
  
  string s;
  long cnt[5]={0};
  for(int i=0;i<n;i++){
   cin >> s;
   for(int j=0;j<5;j++){
    if(s[0]==c[j]) cnt[j]++;
   } 
  }
  
  long long res=0;
  
  for(int i=0;i<5;i++){
   for(int j=i+1;j<5;j++){
    for(int k=j+1;k<5;k++){
     res=res+cnt[i]*cnt[j]*cnt[k];
    }
   }
  }

 cout << res << endl;

}
