import java.util.*;
public class Main{
	public static void main(String[] args) {
		Scanner input = new Scanner(System.in);
		int n=input.nextInt();
		int a[]=new int[100008];
		int b[]=new int[100008];
		for(int i=1;i<=n;i++) {
			a[i]=input.nextInt();
			b[i]=input.nextInt();
		}
		int ans=0;
		for(int i=1;i<=n;i++) {
			int mx=Math.max(a[i-1], b[i-1]);
			int mn=Math.min(a[i], b[i]);
			if(mn<mx) {ans+=0;}
			else {
				ans+=Math.max(0,mn-mx);
				if(a[i-1]!=b[i-1]) {ans++;}
			}
		}
		System.out.println(ans+1);
	}
}
