#include <bits/stdc++.h>
using namespace std;
bool is_palindrome(vector<int> &str) {
  int n = str.size();
  for (int i = 0, j = n - 1; i <= j; i++, j--) {
    if (str[i] != str[j]) return false;
  }
  return true;
}
void next_min(vector<int> &str) {
  int n = str.size();
  if (str[n - 1] == 9) {
    str[n - 1] = 0;
    if (str[n - 2] == 5) {
      str[n - 2] = 0;
      if (str[n - 4] == 0 && str[n - 3] != 9)
        str[n - 3]++;
      else if (str[n - 4] == 0 && str[n - 3] == 9) {
        str[n - 4] = 1;
        str[n - 3] = 0;
      } else if (str[n - 4] == 1 && str[n - 3] != 9)
        str[n - 3]++;
      else if (str[n - 4] == 1 && str[n - 3] == 9) {
        str[n - 4] = 2;
        str[n - 3] = 0;
      } else if (str[n - 4] == 2 && str[n - 3] != 3)
        str[n - 3]++;
      else {
        str[n - 4] = 0;
        str[n - 3] = 0;
      }
    } else
      str[n - 2]++;
  } else
    str[n - 1]++;
}
int main(void) {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  string temp;
  cin >> temp;
  vector<int> str;
  str.push_back(temp[0] - '0');
  str.push_back(temp[1] - '0');
  str.push_back(temp[3] - '0');
  str.push_back(temp[4] - '0');
  long min = 0;
  while (true) {
    if (is_palindrome(str) == true) {
      cout << min;
      return 0;
    }
    next_min(str);
    min++;
  }
  return 0;
}
