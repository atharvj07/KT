import sys, os, io
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
for _ in range(int(input())):
	n = int(input())
	ans = []
	a = [[] for i in range(n)]
	cnt = 0
	s = []
	for i in range(n):
		c = int(input())
		if c == 1:
			a[i].append(1)
		else:
			while c != a[i][-1] + 1:
				a[i].pop()
			a[i][-1] = c
		if i != n - 1:
			a[i + 1] = a[i][:]
	for i in a:
		for j in range(len(i) - 1):
			print(i[j], end='.')
		print(i[-1])
