import java.io.*;
import java.util.*;
import java.math.BigInteger;

public class D {
	
	// gcd(a_1, a_2, ...) = x
	// so all a_i values are multiples of x: 
	//    a_i = b_i * x for some b_i >= 1
	// But there can't be a common factor in b_i ! So gcd(b_i) must be 1.
	// sum a_i = y
	//    sum a_i = x * sum b_i = y
	//    sum b_i = y/x (if x divides y)
	// So, given y/x = c. 
	// Take sequences of length 1, 2, ...:
	// length 1: c  WRONG if c > 1, because gcd(b_i) = c > 1
	// length 2: 
	//     1, c-1
	//     2, c-2   Only ok if gcd(2,c-2) = 1
	//     ...
	//     c/2, c-c/2
	// length 3:
	//     1, 1, c-2
	//     1, 2, c-3
	//     ...
	//     1, (c-1)/2, c-1-(c-1)/2
	//     2, 2, c-4
	//     2, 3, c-5
	//     ...
	//     2, (c-2)/2, c-2-(c-2)/2
	// If ok, add number of permutations 
	// Let's look at it differently:
	//   total number of distributions of y = t(y) = 2^(y-1)
	//   now number of distributions of y with gcd 1 = d(y)
	//       total number of distributions - distributions with gcd > 1
	// distributions with gcd > 1   = SUM_g distributions of y with gcd g = SUM_g d(y/g)   where g divides y, g > 1
	//       suppose y=5, gcd=3, 3 does not divide 5, that could never work.
	// distributions of y with gcd g = distributions of y/g with gcd 1 = d(y/g)
	// so d(y) = 2^(y-1) - SUM_g|y d(y/g)
	// d(1) = 2^0 - 0 = 1                          -> 1
	// d(2) = 2^1 - [d(1)] = 2 - 1 = 1             -> 1+1
	// d(3) = 2^2 - [d(1)] = 4 - 1 = 3             -> 1+2, 2+1, 1+1+1
	// d(4) = 2^3 - [d(2)+d(1)] = 8 - 1 - 1 = 6    -> 1+3, 3+1, 1+1+2, 1+2+1, 2+1+1, 1+1+1+1
		
	public static void main(String[] args) throws Exception {
		Scanner sc = new Scanner(new BufferedReader(new InputStreamReader(System.in)));
		
		int x = sc.nextInt();
		int y = sc.nextInt();

		if (y%x != 0) {
			System.out.println("0");
		} else {
			y = y/x;
			int result = getD(y);
			System.out.println(result+"");
		}
	}
	
	private static List<Integer> getDivisors(int a) {
		int sqrt = (int)Math.floor(Math.sqrt(a));
		ArrayList<Integer> result = new ArrayList<>();
		for (int i=1; i <= sqrt; i++) {
			if (a%i==0) {
				result.add(i);
				if (i != (a/i)) result.add(a/i);
			}
		}
		Collections.sort(result);
		//System.out.println(String.format("%d -> %s", a, result));
		return result;
	}
	
	private static HashMap<Integer, Integer> d = new HashMap<>();
	private static int m = 1_000_000_007;
	private static BigInteger M = new BigInteger(m+"");
	private static BigInteger TWO = new BigInteger(2+"");
	
	private static int getD(int a) {
		if (d.containsKey(a)) return d.get(a);
		
		int result = pow(a-1);
		for (int g : getDivisors(a)) {
			if (g > 1) {
				result = result - getD(a/g);
				result = ((result%m)+m)%m;
			}
		}
		
		d.put(a, result);
		//System.out.println(String.format("%d -> %d", a, result));
		return result;
	}
	
	private static int pow(int a) {
		int result = TWO.modPow(new BigInteger(a+""), M).intValue();
		//System.out.println(String.format("%d -> %d", a, result));
		return result;
	}

}
