#include <bits/stdc++.h>
int main() {
  int testnum = 0;
  std::cin >> testnum;
  int tests[testnum][4];
  for (int t = 0; t < testnum; ++t) {
    std::cin >> tests[t][0] >> tests[t][1] >> tests[t][2] >> tests[t][3];
  }
  for (int t = 0; t < testnum; ++t) {
    int n = tests[t][0];
    int x = tests[t][1];
    int a = tests[t][2];
    int b = tests[t][3];
    if (a > b) {
      std::swap(a, b);
    }
    if (a + n - b - 1 < x) {
      std::cout << n - 1 << std::endl;
    } else {
      std::cout << b + x - a << std::endl;
    }
  }
  return 0;
}
