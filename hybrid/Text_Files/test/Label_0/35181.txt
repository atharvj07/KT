import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.StringTokenizer;

/**
 * @author Don Li
 */
public class Timetable {
    
    long INF = (long) 3e18 + 1;
    
    void solve() {
        int n = in.nextInt(); long t = in.nextLong();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = in.nextLong();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) x[i] = in.nextInt() - 1;
        
        for (int i = 0; i < n; i++) {
            if (x[i] < i) {
                out.println("No");
                return;
            }
        }
        
        for (int i = n - 1; i > 0; i--) {
            if (x[i] < x[i - 1]) {
                out.println("No");
                return;
            }
        }
        
        long[] l = new long[n], r = new long[n];
        for (int i = 0; i < n; i++) {
            l[i] = 0;
            r[i] = INF;
        }
        
        for (int i = 0; i < n; i++) {
            l[i] = Math.max(l[i], a[i] + t);
        }
        
        int[] cnt = new int[n];
        for (int i = n - 1; i >= 0; i--) {
            l[x[i]] = Math.max(l[x[i]], a[i] + t);
            cnt[i]++; cnt[x[i]]--;
            if (x[i] + 1 < n) r[x[i]] = Math.min(r[x[i]], a[x[i] + 1] + t);
        }
        for (int i = 1; i < n; i++) cnt[i] += cnt[i - 1];
        for (int i = 0; i < n; i++) {
            if (cnt[i] > 0 && i + 1 < n) l[i] = Math.max(l[i], a[i + 1] + t);
        }
        
        long[] b = new long[n];
        for (int i = n - 1; i >= 0; i--) {
            if (l[i] >= r[i]) {
                out.println("No");
                return;
            }
            b[i] = r[i] - 1;
            if (i - 1 >= 0) r[i - 1] = Math.min(r[i - 1], b[i]);
        }
        
        out.println("Yes");
        for (int i = 0; i < n; i++) out.print(b[i] + " ");
    }
    
    public static void main(String[] args) {
        in = new FastScanner(new BufferedReader(new InputStreamReader(System.in)));
        out = new PrintWriter(System.out);
        new Timetable().solve();
        out.close();
    }
    
    static FastScanner in;
    static PrintWriter out;
    
    static class FastScanner {
        BufferedReader in;
        StringTokenizer st;
        
        public FastScanner(BufferedReader in) {
            this.in = in;
        }
        
        public String nextToken() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(in.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
        
        public int nextInt() {
            return Integer.parseInt(nextToken());
        }
        
        public long nextLong() {
            return Long.parseLong(nextToken());
        }
        
        public double nextDouble() {
            return Double.parseDouble(nextToken());
        }
    }
}
