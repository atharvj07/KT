#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N=2e5+10;
int po[N];
int cnt[N];
signed main()
{
	long long ans=0;
	int n,p;
	string a;
	cin>>n>>p>>a;
	int now=0;
	if(p%2&&p%5){
		po[0]=1;
		for(int i=1;i<=n;i++){
			po[i]=10*po[i-1]%p;
		}
		cnt[0]=1;
		for(int i=n-1;i>=0;i--){
			now=now+po[n-1-i]*(a[i]-'0')%p;
			now%=p;
			ans+=cnt[now];
			cnt[now]++;
		}
		cout<<ans;
	}
	else {
		for(int i=0;i<n;i++){
			if((a[i]-'0')%p==0){
				ans+=i+1;
			}
		}
		cout<<ans;
	}
}
