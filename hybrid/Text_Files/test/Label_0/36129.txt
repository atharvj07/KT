class TaskE {
	private int f(int c3, int c4, int c5, int k3, int k4, int k5) {
		return Math.abs(c3 * k3 - c4 * k4) + Math.abs(c4 * k4 - c5 * k5);
	}
	private Pair<Integer, Integer> calc(int c3, int c4, int c5, int k4, int s) {
		int d = NumberTheory.gcd(c3, c5);
		int a = c3 / d, b = c5 / d, c = s / d;
		long y = (long)NumberTheory.getPower(b, NumberTheory.getPhi(a) - 1, a) * c % a;
		long x = (c - b * y) / a;
		long head = (long)Math.ceil(-(double)x / b);
		long tail = (long)Math.min(Math.floor((double)(k4 - x) / b), Math.floor((double)(y - k4) / a)) - 1;
		if (head > tail + 1) return null;
		while (head <= tail) {
			long mid = (head + tail) / 2;
			int t1 = f(c3, c4, c5, (int)(x + mid * b), k4, (int)(y - mid * a));
			int t2 = f(c3, c4, c5, (int)(x + (mid + 1) * b), k4, (int)(y - (mid + 1) * a));
			if (t1 < t2) tail = mid - 1; else head = mid + 1;
		}
		return new Pair<Integer, Integer>((int)(x + (tail + 1) * b), (int)(y - (tail + 1) * a));
	}
	public void solve(int testNumber, InputReader in, OutputWriter out) {
		int n = in.readInt();
		int s = in.readInt();
		int c3 = 0, c4 = 0, c5 = 0;
		for (int i = 0; i < n; i ++) {
			int x = in.readInt();
			if (x == 3) c3 ++;
			if (x == 4) c4 ++;
			if (x == 5) c5 ++;
		}
		int minValue = Integer.MAX_VALUE;
		int k3 = 0, k4 = 0, k5 = 0;
		for (int i = 0; i <= s; i += c4) {
			int t = NumberTheory.gcd(c3, c5);
			if ((s - i) % t != 0) continue;
			Pair<Integer, Integer> res = calc(c3, c4, c5, i / c4, s - i);
			if (res != null && f(c3, c4, c5, res.first, i / c4, res.second) < minValue) {
				minValue = f(c3, c4, c5, res.first, i / c4, res.second);
				k3 = res.first;
				k4 = i / c4;
				k5 = res.second;
			}
		}
		if (minValue == Integer.MAX_VALUE) {
			out.printLine(-1);
		} else {
			out.printLine(k3, k4, k5);
		}
	}
}

class Pair<T1 extends Comparable<T1>, T2 extends Comparable<T2>> implements Comparable<Pair<T1, T2>> {
	public T1 first;
	public T2 second;

	public boolean equals(Object o) {
		if (this == o) return true;
		if (o == null || getClass() != o.getClass()) return false;

		Pair pair = (Pair) o;

		if (!first.equals(pair.first)) return false;
		if (!second.equals(pair.second)) return false;

		return true;
	}

	public int hashCode() {
		int result = first.hashCode();
		result = 31 * result + second.hashCode();
		return result;
	}

	public Pair(T1 first, T2 second) {
		this.first = first;
		this.second = second;
	}

	public int compareTo(Pair<T1, T2> o) {
		if (first.compareTo(o.first) != 0) return first.compareTo(o.first);
		return second.compareTo(o.second);
	}
}

class NumberTheory {
	public static int gcd(int m, int n) {
		if (m == 0) return n;
		while (n > 0) {
			int r = m % n;
			m = n;
			n = r;
		}
		return m;
	}

	public static int getPower(int a, int p, int modulo) {
		if (p == 0) return 1;
		int res = getPower(a, p / 2, modulo);
		res = (int)((long)res * res % modulo);
		if (p % 2 == 1) res = (int)((long)res * a % modulo);
		return res;
	}

	public static int getPhi(int n) {
		int res = 1;
		for (int i = 2; i * i <= n; i ++)
			if (n % i == 0) {
				n /= i;
				res *= i - 1;
				while (n % i == 0) {
					res *= i;
					n /= i;
				}
			}
		if (n > 1) res *= n - 1;
		return res;
	}

	}

class InputReader {
	private InputStream stream;
	private byte[] buf = new byte[1000];
	private int curChar;
	private int numChars;

	public InputReader(InputStream stream) {
		this.stream = stream;
	}

	private int read() {
		if (numChars == -1)
			throw new InputMismatchException();
		if (curChar >= numChars) {
			curChar = 0;
			try {
				numChars = stream.read(buf);
			} catch (IOException e) {
				throw new InputMismatchException();
			}
			if (numChars <= 0)
				return -1;
		}
		return buf[curChar++];
	}

	public int readInt() {
		int c = read();
		while (isSpaceChar(c))
			c = read();
		int sgn = 1;
		if (c == '-') {
			sgn = -1;
			c = read();
		}
		int res = 0;
		do {
			if (c < '0' || c > '9')
				throw new InputMismatchException();
			res *= 10;
			res += c - '0';
			c = read();
		} while (!isSpaceChar(c));
		return res * sgn;
	}

	private boolean isSpaceChar(int c) {
		return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
	}
}

class OutputWriter {
	private final PrintWriter writer;

	public OutputWriter(OutputStream outputStream) {
		writer = new PrintWriter(new BufferedWriter(new OutputStreamWriter(outputStream)));
	}

	public OutputWriter(Writer writer) {
		this.writer = new PrintWriter(writer);
	}

	public void print(Object...objects) {
		for (int i = 0; i < objects.length; i++) {
			if (i != 0)
				writer.print(' ');
			writer.print(objects[i]);
		}
	}

	public void printLine(Object...objects) {
		print(objects);
		writer.println();
	}

	public void close() {
		writer.close();
	}
}
