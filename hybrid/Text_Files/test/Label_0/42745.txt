def solver():
	n = int(input())
	num = list(map(int, input().split()))
	a = num + num + num
	ans = [0] * (3 * n)
	maxn = []
	minn = []
	for i in range(len(a) - 1, -1, -1):
		while len(maxn) > 0 and a[maxn[-1]] < a[i]:
			maxn.pop()
		while len(minn) > 0 and a[minn[-1]] > a[i]:
			minn.pop()
		(left, right) = (0, len(minn))
		while left < right:
			mid = (left + right) // 2
			if a[minn[mid]] * 2 < a[i]:
				left = mid + 1
			else:
				right = mid
		ret = 3 * n
		if left > 0:
			ret = min(ret, minn[left - 1])
		if len(maxn) > 0:
			ret = min(ret, maxn[-1])
		if ret < 3 * n and a[ret] >= a[i]:
			ans[i] = ans[ret]
		else:
			ans[i] = ret
		maxn.append(i)
		minn.append(i)
	for i in range(n):
		print(-1 if ans[i] == 3 * n else ans[i] - i)

def main():
	while 1:
		try:
			solver()
		except EOFError:
			break
main()
