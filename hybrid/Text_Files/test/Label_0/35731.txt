import java.util.*;
public class CapsLock {
	public static void main(String[] args) {
		
		Scanner in = new Scanner(System.in);
		String s = in.nextLine();
		String sCap = s.substring(0,1).toUpperCase() + s.substring(1, s.length()).toLowerCase();
		String sAnti = s.substring(0,1).toLowerCase() + s.substring(1, s.length()).toUpperCase();
		
		if(s == s.toUpperCase())
			System.out.println(s.toLowerCase());
		else if(s.compareTo(sAnti) == 0)
			System.out.println(sCap);
		else if(s == s.toLowerCase())
			System.out.println(s);
		else
			System.out.println(s);
		
	}
}
