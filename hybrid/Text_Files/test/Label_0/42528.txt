def main():
	n = int(input())
	arr = list(map(int, input().split()))
	b = []
	b.append(1)
	for i in range(1, n):
		if arr[i] >= b[i - 1]:
			b.append(arr[i] + 1)
		else:
			b.append(b[i - 1])
	for i in range(n - 2, 0, -1):
		if b[i] < b[i + 1] - 1:
			b[i] = b[i + 1] - 1
	ans = 0
	for i in range(0, n):
		ans += b[i] - arr[i] - 1
	print(ans)
main()
