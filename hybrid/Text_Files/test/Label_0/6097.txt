#include <bits/stdc++.h>
int main() {
  int h1, m1, h2, m2;
  scanf("%d:%d", &h1, &m1);
  scanf("%d:%d", &h2, &m2);
  int t1 = 60 * h1 + m1;
  int t2 = 60 * h2 + m2;
  int t = t1 + (t2 - t1) / 2;
  if (t >= 720) {
    if (((t1 / 60) > 12) && ((t2 / 60) < 12)) {
      t -= 720;
    }
  }
  if ((t / 60) < 10) printf("%d", 0);
  printf("%d:", t / 60);
  if ((t % 60) < 10) printf("%d", 0);
  printf("%d", t % 60);
  return 0;
}
