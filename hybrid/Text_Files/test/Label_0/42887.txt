from collections import deque
for _ in range(int(input())):
	N = int(input())
	I = [[] for _ in range(N)]
	for _ in range(N - 1):
		(u, v) = map(int, input().split())
		u -= 1
		v -= 1
		I[u].append(v)
		I[v].append(u)
	state = [-1] * N
	s = 0
	state[s] = 0
	task = deque([s])
	while task:
		p = task.pop()
		for q in I[p]:
			if state[q] != -1:
				continue
			state[q] = 1 ^ state[p]
			task.append(q)
	g0 = []
	g1 = []
	for (i, s) in enumerate(state):
		if s:
			g1.append(i)
		else:
			g0.append(i)
	if len(g1) > len(g0):
		(g1, g0) = (g0, g1)
	p1 = []
	t1 = len(g1)
	ans = [-1] * N
	idx0 = 0
	idx1 = 0
	now = 1
	while t1 > 0:
		if t1 & 1:
			for j in range(now, 2 * now):
				ans[g1[idx1]] = j
				idx1 += 1
		else:
			for j in range(now, 2 * now):
				ans[g0[idx0]] = j
				idx0 += 1
		t1 >>= 1
		now *= 2
	for j in range(now, N + 1):
		ans[g0[idx0]] = j
		idx0 += 1
	print(*ans)
