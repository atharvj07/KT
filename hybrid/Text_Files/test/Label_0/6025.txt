#include <bits/stdc++.h>
using namespace std;
int main() {
  int T;
  string s;
  cin >> T;
  for (int i = 1; i <= T; i++) {
    cin >> s;
    int a = 0, b = 0, c;
    for (int j = 0; j < s.length(); j++) {
      if (s[j] == '0') {
        a++;
      } else {
        b++;
      }
    }
    c = min(a, b);
    if (c % 2 == 1) {
      cout << "DA" << endl;
    } else {
      cout << "NET" << endl;
    }
  }
}
