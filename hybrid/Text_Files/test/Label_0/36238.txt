import java.util.HashMap;
import java.util.Scanner;

public class Solution {
    private static int MAX_RACERS = 50;
    private static int[] pts = {25, 18, 15, 12, 10, 8, 6, 4, 2, 1};

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = Integer.parseInt(in.nextLine());

        HashMap<String, Integer[]> stat = new HashMap<String, Integer[]>();
        for (int i = 0; i < n; i++) {
            int d = Integer.parseInt(in.nextLine());
            for (int j = 1; j <= d; j++) {
                String racerName = in.nextLine();
                Integer[] statistic;
                if (stat.containsKey(racerName)) {
                    statistic = stat.get(racerName);
                } else {
                    statistic = new Integer[MAX_RACERS + 1];
                    for (int t = 0; t < statistic.length; t++) {
                        statistic[t] = 0;
                    }
                }

                if (j <= 10) {
                    statistic[0] += pts[j - 1];
                }
                statistic[j]++;

                stat.put(racerName, statistic);
            }

        }

        String winnerClassic = (String) stat.keySet().toArray()[0];
        String winnerModern = winnerClassic;
        for (String key : stat.keySet()) {
            Integer[] statistic = stat.get(key);
            Integer[] championClassic = stat.get(winnerClassic);
            if (statistic[0] >= championClassic[0]) {
                if (statistic[0].equals(championClassic[0])) {
                    for (int i = 1; i < MAX_RACERS; i++) {
                        if (statistic[i] < championClassic[i]) {
                            break;
                        } else if (statistic[i] > championClassic[i]) {
                            winnerClassic = key;
                            break;
                        }
                    }
                } else {
                    winnerClassic = key;
                }
            }

            Integer[] championModern = stat.get(winnerModern);
            if (statistic[1] >= championModern[1]) {
                if (statistic[1].equals(championModern[1])) {
                    if (statistic[0] >= championModern[0]) {
                        if (statistic[0].equals(championModern[0])) {
                            for (int i = 2; i < MAX_RACERS + 1; i++) {
                                if (statistic[i] < championModern[i]) {
                                    break;
                                } else if (statistic[i] > championModern[i]) {
                                    winnerModern = key;
                                    break;
                                }
                            }
                        } else {
                            winnerModern = key;
                        }
                    }
                } else {
                    winnerModern = key;
                }
            }
        }

        System.out.println(winnerClassic + "\n" + winnerModern);
    }
}
