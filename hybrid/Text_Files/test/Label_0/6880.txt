#include <bits/stdc++.h>
using namespace std;
int main() {
  unordered_set<char> store;
  for (int i = 0; i < 26; i++) {
    store.insert('a' + i);
  }
  int test, cnt = 0;
  cin >> test;
  string op, word;
  for (int i = 0; i < test; i++) {
    cin >> op >> word;
    if (store.size() == 1) {
      if (op == "!")
        cnt++;
      else if (op == "?" and !store.count(word[0]))
        cnt++;
      continue;
    }
    if (op == "!") {
      unordered_set<char> mark;
      for (auto j : word) mark.insert(j);
      for (auto j = store.begin(); j != store.end();) {
        if (mark.count(*j))
          j++;
        else {
          j = store.erase(j);
        }
      }
    } else if (op == ".") {
      for (int j = 0; j < word.size(); j++) {
        store.erase(word[j]);
      }
    } else if (op == "?")
      store.erase(word[0]);
  }
  cout << cnt << endl;
}
