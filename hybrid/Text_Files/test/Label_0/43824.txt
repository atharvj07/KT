n = int(input())
a = [int(x) for x in input().split()]
dic = {}
i = 1
for item in a:
	if item not in dic:
		dic[item] = [i]
	else:
		dic[item].append(i)
	i += 1
counter = 0
i = j = 1
for item in range(1, n + 1):
	if abs(dic[item][0] - i) + abs(dic[item][1] - j) >= abs(dic[item][1] - i) + abs(dic[item][0] - j):
		counter += abs(dic[item][1] - i) + abs(dic[item][0] - j)
		i = dic[item][1]
		j = dic[item][0]
	else:
		counter += abs(dic[item][0] - i) + abs(dic[item][1] - j)
		i = dic[item][0]
		j = dic[item][1]
print(counter)
