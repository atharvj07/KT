import sys
from array import array
import typing as Tp

def input():
	return sys.stdin.buffer.readline().decode('utf-8')

def output(*args):
	sys.stdout.buffer.write(('\n'.join(map(str, args)) + '\n').encode('utf-8'))

def main():
	t = int(input())
	ans_a = []
	mod = 10 ** 9 + 7
	for ti in range(t):
		n = int(input())
		adj = [[] for _ in range(n)]
		edge_i = [[] for _ in range(n)]
		deg = [0] * n
		for (ei, (u, v)) in enumerate((map(int, input().split()) for _ in range(n - 1))):
			adj[u - 1].append(v - 1)
			adj[v - 1].append(u - 1)
			edge_i[u - 1].append(ei)
			edge_i[v - 1].append(ei)
			deg[u - 1] += 1
			deg[v - 1] += 1
		m = int(input())
		p = sorted(map(int, input().split()))
		path_cnt = []
		visited_edge = [0] * (n - 1)
		stack = [i for i in range(n) if deg[i] == 1]
		size = [1] * n
		while stack:
			v = stack.pop()
			for (dest, ei) in zip(adj[v], edge_i[v]):
				if visited_edge[ei]:
					continue
				visited_edge[ei] = 1
				path_cnt.append(size[v] * (n - size[v]))
				if deg[dest] == 0:
					continue
				size[dest] += size[v]
				deg[dest] -= 1
				if deg[dest] == 1:
					stack.append(dest)
		path_cnt.sort()
		while len(p) > n - 1:
			p[-1] = p.pop() * p[-1] % mod
		ans = 0
		while p:
			ans += path_cnt.pop() * p.pop() % mod
		ans = (ans + sum(path_cnt)) % mod
		ans_a.append(ans)
	output(*ans_a)
main()
