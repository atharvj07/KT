import java.util.*;
public class Main {

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    for ( int i = 1; i <= n; ++i )
      a[i-1] = i;
    f( a, n );
    for ( int i = 0; i < n-1; ++i )
      System.out.print( a[i] + " " );
    System.out.println( a[n-1] );
  }
  public static void f( int[]a, int n ){
    if ( n != 1 ){
      int t = a[n-1];
      a[n-1] = a[n-2];
      a[n-2] = t;
      f( a, n-1 );      
    }
  }
  
}
