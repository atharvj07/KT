#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, i, min1, min2;
  cin >> n;
  int a[n + 1];
  for (i = 1; i <= n; i++) {
    cin >> a[i];
  }
  min1 = a[1];
  min2 = 101;
  for (i = 2; i <= n; i++) {
    if (a[i] < min1) min1 = a[i];
  }
  for (i = 1; i <= n; i++) {
    if (a[i] < min2 && a[i] != min1) min2 = a[i];
  }
  if (min2 == 101)
    cout << "NO";
  else
    cout << min2;
  return 0;
}
