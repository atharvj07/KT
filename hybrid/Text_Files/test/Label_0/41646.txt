sumC = sum(list(map(int, input().split())))
sumM = sum(list(map(int, input().split())))
s = int(input())

def qu(a, b):
	if a == 0:
		return 0
	elif a / b <= 1:
		return 1
	elif a % b == 0:
		return int(a / b)
	else:
		return int(a / b) + 1
if qu(sumC, 5) + qu(sumM, 10) <= s:
	print('YES')
else:
	print('NO')
