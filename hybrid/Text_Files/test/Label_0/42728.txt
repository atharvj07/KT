import sys
input = sys.stdin.readline
from collections import Counter
q = int(input())
for _ in range(q):
	n = int(input())
	a = list(map(int, input().split()))
	cnt = list(Counter(a).items())
	cnt.sort(key=lambda k: k[1], reverse=True)
	prev = n + 1
	ans = 0
	for (_, c) in cnt:
		if prev > c:
			ans += c
			prev = c
		else:
			ans += prev - 1
			prev -= 1
		if prev == 0:
			break
	print(ans)
