vl f(int v){
	vl dp(S);
	for(int i=0;i<S;i++) if(i&1<<v){
		int T=(i-1)&i;
		dp[i]=a[i];
		while(T){
			if(T&1<<v) dp[i]=(dp[i]-dp[T]*a[i-T]%mod+mod)%mod;
			T=(T-1)&i;
		}
	}
	return dp;
}

int main(){
	cin>>n>>m;
	g=vvi(n,vi(n));
	for(int i=0;i<m;i++){
		int u,v;

		u--;v--;
		g[u][v]++;g[v][u]++;
	}
	S=1<<n;
	a=vl(S,1);
	for(int i=0;i<S;i++) for(int j=0;j<n;j++) for(int k=j+1;k<n;k++) if(i&1<<j&&i&1<<k&&g[j][k]) (a[i]*=2)%=mod;
	vl dp1=f(0),dp2=f(1);
	ll res=a[S-1];
	for(int i=1;i<S;i+=2) if(!(i&2)){
		int T=S-i-1;
		for(int j=0;j<n;j++) for(int k=0;k<n;k++) if(i&1<<j&&T&1<<k&&g[j][k]) T-=1<<k;
		int T_=T;
		while(T){
			if(T&2) res=(res-dp1[i]*dp2[T]%mod*a[S-i-T-1]%mod+mod)%mod;
			T=(T-1)&T_;
		}
	}
	cout<<res<<endl;
}
