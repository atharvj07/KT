import math
t = int(input())
for _ in range(t):
	n = int(input())
	arr = list(map(int, input().strip().split()))
	r = arr[0]
	p = arr[1]
	s = arr[2]
	l = []
	win = 0
	string = input()
	for i in string:
		if i == 'R' and p != 0:
			win += 1
			p -= 1
			l.append('P')
		elif i == 'P' and s != 0:
			win += 1
			s -= 1
			l.append('S')
		elif i == 'S' and r != 0:
			win += 1
			r -= 1
			l.append('R')
		else:
			l.append('N')
	if win >= math.ceil(n / 2):
		print('YES')
		for i in l:
			if i == 'R' or i == 'P' or i == 'S':
				print(i, end='')
			elif r != 0:
				print('R', end='')
				r -= 1
			elif p != 0:
				print('P', end='')
				p -= 1
			elif s != 0:
				print('S', end='')
				s -= 1
		print()
	else:
		print('NO')
