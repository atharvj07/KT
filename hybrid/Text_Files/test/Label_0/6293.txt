#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  string s, s1, s2;
  cin >> s;
  int i, j;
  cin >> s1;
  string s3 = "6789TJQKA";
  cin >> s2;
  if (s1[1] == s[0] && s2[1] != s[0])
    cout << "YES";
  else if (s1[1] == s[0] && s2[1] == s[0]) {
    for (i = 0; i < s3.size(); i++) {
      if (s1[0] == s3[i]) break;
    }
    for (j = 0; j < s3.size(); j++) {
      if (s2[0] == s3[j]) break;
    }
    if (i > j)
      cout << "YES";
    else
      cout << "NO";
  } else if (s1[1] != s[0] && s2[1] == s[0]) {
    cout << "NO";
  } else {
    if (s1[1] == s2[1]) {
      for (i = 0; i < s3.size(); i++) {
        if (s1[0] == s3[i]) break;
      }
      for (j = 0; j < s3.size(); j++) {
        if (s2[0] == s3[j]) break;
      }
      if (i > j)
        cout << "YES";
      else
        cout << "NO";
    } else
      cout << "NO";
  }
  return 0;
}
