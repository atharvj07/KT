from collections import defaultdict
from math import gcd
inf = 10 ** 20
t = 1
for _ in range(t):
	n = int(input())
	arr = []
	for i in range(n):
		(x, y) = map(int, input().split())
		arr += [[x, y]]
	d = defaultdict(set)
	infslope = defaultdict(int)
	for i in range(n):
		for j in range(i + 1, n):
			dy = arr[i][1] - arr[j][1]
			dx = arr[i][0] - arr[j][0]
			g = gcd(abs(dx), abs(dy))
			dx //= g
			dy //= g
			if dx == 0:
				infslope[arr[i][0]] += 1
				continue
			if dy < 0:
				dy *= -1
				dx *= -1
			elif dy == 0:
				dx = 1
			m = (dy, dx)
			one = m[1]
			two = m[1] * arr[i][1] - m[0] * arr[i][0]
			g = gcd(abs(one), abs(two))
			one //= g
			two //= g
			if one < 0:
				one *= -1
				two *= -1
			d[dy, dx].add((one, two))
	curr = 0
	ans = 0
	for x in d:
		ans += len(d[x]) * curr
		curr += len(d[x])
	ans += len(infslope) * curr
	print(ans)
