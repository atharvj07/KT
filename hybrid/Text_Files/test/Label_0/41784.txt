t = int(input())
sol = []
while t:
	t -= 1
	n = int(input())
	a = [int(i) for i in input().split()]
	d = {}
	for i in a:
		if not d.get(i):
			d[i] = 1
		else:
			d[i] += 1
	a = []
	for i in d.keys():
		a.append(d[i])
	a.sort(reverse=True)
	d = set()
	for i in a:
		if not i in d:
			d.add(i)
		else:
			j = i
			while j and j in d:
				j -= 1
			d.add(j)
	a = d
	sol.append(sum(a))
print(*sol, sep='\n')
