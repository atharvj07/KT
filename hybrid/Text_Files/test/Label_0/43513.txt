import sys
input = sys.stdin.readline
for _ in range(int(input())):
	(n, m) = map(int, input().strip().split())
	arr = [[int(x)] for x in input().strip().split()]
	for x in range(n * m):
		arr[x].append(x + 1)
	arr.sort()
	cnt = 0
	for r in range(0, n * m, m):
		for i in range(r, r + m):
			for j in range(r, i):
				if arr[j][0] < arr[i][0] and arr[j][1] < arr[i][1]:
					cnt += 1
	print(cnt)
