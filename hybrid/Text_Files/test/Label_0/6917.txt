#include <bits/stdc++.h>
using namespace std;
long long int visited[100005];
vector<long long int> ami[100005];
long long int colour[100005];
int main() {
  long long int i, j, k, m, a, x, s, p, q, z, v, n, b;
  for (int i = 0; i < 100005; i++) {
    colour[i] = -1;
    visited[i] = -1;
  }
  cin >> v;
  queue<long long int> ss;
  i = 1;
  while (i < v) {
    cin >> a >> b;
    ami[a].push_back(b);
    ami[b].push_back(a);
    i++;
  }
  ss.push(a);
  colour[a] = 0;
  while (!ss.empty()) {
    x = ss.front();
    ss.pop();
    vector<long long int>::iterator it;
    it = ami[x].begin();
    while (it != ami[x].end()) {
      if (visited[*it] == -1) {
        if (colour[x] == 0) {
          colour[*it] = 1;
        } else {
          colour[*it] = 0;
        }
        ss.push(*it);
        visited[*it] = 1;
      }
      it++;
    }
  }
  p = 0;
  q = 0;
  for (int i = 1; i <= v; i++) {
    if (colour[i] == 0) {
      p++;
    } else if (colour[i] == 1) {
      q++;
    }
  }
  p = p * q;
  cout << p - v + 1 << endl;
  for (int i = 0; i < v; i++) {
    ami[i].clear();
  }
}
