#include <bits/stdc++.h>
int a[6];
int main() {
  for (int i = 0; i < 6; i++) {
    scanf("%d", &a[i]);
  }
  int h = a[1] + a[2];
  int ans = 0, w = a[0], nw;
  for (int i = 0; i < h; i++) {
    if ((i < a[1]) && (i < a[5]))
      nw = w + 1;
    else if ((i < a[1]) ^ (i < a[5]))
      nw = w;
    else
      nw = w - 1;
    ans += w + nw;
    w = nw;
  }
  printf("%d\n", ans);
}
