import java.io.*;
import java.math.BigInteger;
import java.util.*;

public class Main extends Thread {
    public Main(int what) {
        try {
            if (what == 0) {
                this.input = new BufferedReader(new InputStreamReader(System.in));
                this.output = new PrintWriter(System.out);
            } else {

            }
        } catch (Throwable e) {
            e.printStackTrace();
            System.exit(666);
        }
    }

    static final int get(int i, int j, char[][] board) {
        boolean ok = board[i][j] == '1';
        ok &= board[i][j - 1] == '1';
        ok &= board[i][j + 1] == '1';
        ok &= board[i - 1][j] == '1';
        ok &= board[i + 1][j] == '1';
        return ok ? 1 : 0;
    }

    static final int count(int lX, int lY, int rX, int rY, int [][]sums) {
        if (lX > rX || lY > rY) {
            return 0;
        }
        int result = sums[rX][rY] - sums[rX][lY - 1] - sums[lX - 1][rY] + sums[lX - 1][lY - 1];
        return result;
    }


    private void solve() throws Throwable {
        int n = nextInt(), m = nextInt(), k = nextInt();
        char[][] grid = new char[n][];
        for (int i = 0; i < n; ++i) {
            grid[i] = nextToken().toCharArray();
        }
        int has[][] = new int[n][m];
        for (int i = 1; i + 1 < n; ++i) {
            for (int j = 1; j + 1 < m; ++j) {
                has[i][j] = get(i, j, grid);
            }
        }
        int [][]howMany = new int[n][m];
        for (int i = 1; i < n; ++i) {
            for (int j = 1; j < m; ++j) {
                howMany[i][j] = howMany[i - 1][j] + howMany[i][j - 1] - howMany[i - 1][j - 1] + has[i][j];
            }
        }
        long result = 0;
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < m; ++j) {
                for (int l = 0, from = j; l <= i && from >= 0; ++l) {
                    while (from >= 0 && count(l + 1, from + 1, i - 1, j - 1, howMany) < k) {
                        --from;
                    }
                    result += from + 1;
                }
            }
        }
        output.println(result);


    }

    public void run() {
        try {
            solve();
        } catch (Throwable e) {
            e.printStackTrace();
            System.exit(666);
        } finally {
            output.flush();
            output.close();
        }
    }


    public static void main(String... args) {
        new Main(0).start();
    }

    private String nextToken() throws IOException {
        while (tokens == null || !tokens.hasMoreTokens()) {
            tokens = new StringTokenizer(input.readLine());
        }
        return tokens.nextToken();
    }

    private int nextInt() throws IOException {
        return Integer.parseInt(nextToken());
    }

    private double nextDouble() throws IOException {
        return Double.parseDouble(nextToken());
    }

    private long nextLong() throws IOException {
        return Long.parseLong(nextToken());
    }

    private BufferedReader input;
    private PrintWriter output;
    private StringTokenizer tokens = null;
}
