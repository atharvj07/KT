#include <bits/stdc++.h>
using namespace std;
const int maxn = 3e6 + 100;
vector<int> ans;
int main() {
  int n;
  scanf("%d", &n);
  if (n % 2) {
    for (int i = 1; i <= n; i++) {
      if (i % 2)
        ans.push_back(i * 2 - 1);
      else if (i % 2 == 0)
        ans.push_back(i * 2);
    }
    for (int i = 1; i <= n; i++) {
      if (i % 2)
        ans.push_back(i * 2);
      else if (i % 2 == 0)
        ans.push_back(i * 2 - 1);
    }
    printf("YES\n");
    for (int i = 0; i < 2 * n; i++) {
      printf("%d", ans[i]);
      if (i == 2 * n - 1)
        printf("\n");
      else
        printf(" ");
    }
  } else
    printf("NO\n");
  return 0;
}
