#include <bits/stdc++.h>
using namespace std;
int main() {
  int qAmount;
  cin >> qAmount;
  int recruits = 0;
  int u = 0;
  for (int i = 0; i < qAmount; i++) {
    int x;
    cin >> x;
    if (x == -1) {
      if (recruits > 0)
        recruits--;
      else
        u++;
    } else {
      recruits += x;
    }
  }
  cout << u << endl;
  return 0;
}
