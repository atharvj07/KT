(n, f) = map(int, input().split())
tally = []
for i in range(n):
	(a, b) = map(int, input().split())
	tally.append([a, b])
diff = []
total = 0
for i in range(n):
	total += min(tally[i][1], tally[i][0])
	diff.append(min(2 * tally[i][0], tally[i][1]) - min(tally[i][1], tally[i][0]))
diff.sort()
for i in range(f):
	total += diff[n - 1 - i]
print(total)
