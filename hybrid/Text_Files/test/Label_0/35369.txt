import java.util.*;
import java.lang.*;
import java.io.*;

/* Name of the class has to be "Main" only if the class is public. */
public class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		// your code goes here
		try{
		    Scanner k=new Scanner(System.in);
		    int t=k.nextInt();
		    for(int q1=0;q1<t;q1++)
		    {
		        long n=k.nextInt();
		        long c=0;
		        for(long i=0;;i++)
		        {
		            for(long j=1;j<n;j++)
		            {
		                long temp=(j*(j+1)*3/2)-j;
		                if(temp==n)
		                {
		                c++;
		                n=n-temp;
		                j=0;
		                }
		                else if(temp>n&&j!=1)
		                {
		                    c++;
		                    n-=((j-1)*j*3/2)-(j-1);
		                    j=0;
		                }
		            }
		            if(n==2)
		            c++;
		            if(n<=2)
		            break;
		        }
		        System.out.println(c);
		    }
		    }
		    catch(Exception e)
		    {
		        return;
		    }
	}
}
