import sys, math
n = int(input())
a = [0 for _ in range(n)]
b = [1 for _ in range(n)]
k = -1
for i in range(n):
	(p, c) = map(int, sys.stdin.readline().split())
	if p == -1:
		k = i
	elif c == 0:
		a[p - 1] = 1
	else:
		b[i] = 0
ans = []
for i in range(n):
	if a[i] == 0 and b[i] == 0 and (i != k):
		ans.append(i + 1)
if len(ans) == 0:
	print(-1)
else:
	print(*ans)
