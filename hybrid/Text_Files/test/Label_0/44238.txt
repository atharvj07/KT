(n, m, k) = list(map(int, input().split()))
arr = []
c = 0
for i in range(n):
	x = input().replace('.', 'X')
	c += x.count('#')
	arr.append(list(map(str, x)))
k = n * m - k - c
(i, p) = (0, [])
while k:
	if 'X' in arr[i]:
		j = arr[i].index('X')
		(arr[i][j], p) = ('.', [(i, j)])
		k -= 1
		break
	i += 1
while k:
	(x, y) = p.pop()
	for (i, j) in ((x, y - 1), (x, y + 1), (x - 1, y), (x + 1, y)):
		if i < 0 or j < 0:
			continue
		if i < n and j < m and (arr[i][j] == 'X'):
			arr[i][j] = '.'
			p.append((i, j))
			k -= 1
			if k == 0:
				break
for i in arr:
	for j in i:
		print(j, end='')
	print()
