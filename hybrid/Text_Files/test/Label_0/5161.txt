template <typename T>
void _(const char* s, T h) {
  cerr << s << " = " << h << "\n";
}
template <typename T, typename... Ts>
void _(const char* s, T h, Ts... t) {
  int b = 0;
  while (((b += *s == '(') -= *s == ')') != 0 || *s != ',') cerr << *s++;
  cerr << " = " << h << ",";
  _(s + 1, t...);
}

  if (x < y) x = y;
}

  if (x > y) x = y;

const ll N = 1010, OO = 1e18 + 123456789;
void add(ll& x, ll y) {
  if ((x += y) > OO) x = OO;
}
ll n, m, k;
ll dp[N][N], lc[N][N];
pair<string, ll> su[N];
string second;
bool ch(ll l, ll r) {
  ;
  ll le = r - l + 1;
  memset(dp, 0, sizeof dp);
  dp[n][0] = 1;
  ll res = 0;
  for (ll i = n - 1; i >= 0; --i) {
    dp[i][0] = 1;
    if (lc[l][i] < le) {
      ll ke = lc[l][i] + 1;
      char ci = second[i + ke - 1], cl = second[l + ke - 1];
      if (ci < cl) {
        for (ll j = m; j >= 1; --j) {
          dp[i][j] = dp[i + 1][j];
          ;
        }
        continue;
      } else {
        for (ll j = m; j >= 1; --j) {
          dp[i][j] = dp[i + 1][j];
          add(dp[i][j], dp[i + ke][j - 1]);
          if (j == m && i == 0) res = dp[i + ke][j - 1];
          ;
        }
      }
    } else {
      ll ke = le;
      for (ll j = m; j >= 1; --j) {
        dp[i][j] = dp[i + 1][j];
        add(dp[i][j], dp[i + ke][j - 1]);
        if (j == m && i == 0) res = dp[i + ke][j - 1];
        ;
      }
    }
  };
  ;
  return res >= k;
}
int32_t main() {
  cin >> n >> m >> k >> second;
  second += "$";
  for (ll i = 0; i < (n); ++i) su[i] = {second.substr(i), i};
  sort(su, su + n);
  for (ll i = 0; i < (n - 1); ++i) {
    ll x = su[i].second, y = su[i + 1].second;
    for (ll j = 0; j < (n); ++j) {
      if (su[i].first[j] != su[i + 1].first[j]) {
        lc[x][y] = lc[y][x] = j;
        break;
      }
    }
  }
  for (ll i = 0; i < (n); ++i) lc[i][i] = n - i;
  for (ll i = 0; i < (n); ++i) {
    ll x = su[i].second;
    for (ll j = i + 1; j < n; ++j) {
      ll y = su[j].second;
      ll z = su[j - 1].second;
      lc[y][x] = lc[x][y] = min(lc[x][z], lc[z][y]);
    }
  }
  vector<pll> po;
  ll la = 0;
  for (ll i = 0; i < (n); ++i) {
    ll x = su[i].second;
    for (ll j = la; j + x < n; ++j) po.emplace_back(x, x + j);
    if (i != n - 1) la = lc[x][su[i + 1].second];
  }
  ll l = 0, r = po.size();
  while (r - l > 1) {
    ll mi = (l + r) / 2;
    if (ch(po[mi].first, po[mi].second))
      l = mi;
    else
      r = mi;
  }
  cout << second.substr(po[l].first, po[l].second - po[l].first + 1) << '\n';
  return 0;
}
