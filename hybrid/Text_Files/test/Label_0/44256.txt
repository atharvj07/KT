(n, ls) = (int(input()), list(map(int, input().split())))
maxi = max(ls)
mini = min(ls)
if maxi - mini < 3:
	print(mini + 1, 0)
else:
	(bs, v) = (100000, -1)
	for i in range(mini + 1, maxi):
		score = 0
		for l in ls:
			if abs(l - i) > 1:
				score += abs(l - i) - 1
		if score < bs:
			(bs, v) = (score, i)
	print(v, bs)
