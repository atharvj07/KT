#include <bits/stdc++.h>
using namespace std;
const int maxn = 2005, maxm = 205;
const int seed1 = 23333, seed2 = 18869;
const int MOD[2] = {998244353, 1000000009};
int a[maxn][maxm][2], b[maxm][maxn][2];
char s[maxn][maxm], t[maxm][maxn];
int pow1[maxn][2], pow2[maxn][2];
int n, m;
bool check(int r, int c) {
  for (int k = (0); k <= int(1); k++) {
    int aa =
        (a[r + m][m][k] - 1LL * a[r][m][k] * pow1[m][k] % MOD[k] + MOD[k]) %
        MOD[k];
    int bb =
        (b[m][c + m][k] - 1LL * b[m][c][k] * pow2[m][k] % MOD[k] + MOD[k]) %
        MOD[k];
    if (aa != bb) return 0;
  }
  return 1;
}
int main() {
  scanf("%d%d", &n, &m);
  for (int i = (1); i <= int(n); i++) scanf("%s", s[i] + 1);
  for (int i = (1); i <= int(m); i++) scanf("%s", t[i] + 1);
  for (int k = (0); k <= int(1); k++) {
    pow1[0][k] = pow2[0][k] = 1;
    for (int i = (1); i <= int(n); i++) {
      pow1[i][k] = 1LL * pow1[i - 1][k] * seed1 % MOD[k];
      pow2[i][k] = 1LL * pow2[i - 1][k] * seed2 % MOD[k];
    }
  }
  for (int i = (1); i <= int(n); i++)
    for (int j = (1); j <= int(m); j++)
      for (int k = (0); k <= int(1); k++)
        a[i][j][k] =
            (1LL * seed1 * a[i - 1][j][k] + 1LL * seed2 * a[i][j - 1][k] -
             1LL * seed1 * seed2 % MOD[k] * a[i - 1][j - 1][k] % MOD[k] +
             MOD[k] + s[i][j]) %
            MOD[k];
  for (int i = (1); i <= int(m); i++)
    for (int j = (1); j <= int(n); j++)
      for (int k = (0); k <= int(1); k++)
        b[i][j][k] =
            (1LL * seed1 * b[i - 1][j][k] + 1LL * seed2 * b[i][j - 1][k] -
             1LL * seed1 * seed2 % MOD[k] * b[i - 1][j - 1][k] % MOD[k] +
             MOD[k] + t[i][j]) %
            MOD[k];
  for (int i = (0); i <= int(n - m); i++)
    for (int j = (0); j <= int(n - m); j++)
      if (check(i, j)) return printf("%d %d", i + 1, j + 1), 0;
  return 0;
}
