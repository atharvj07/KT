import math
from collections import defaultdict
from sys import stdin
T = int(input())
for _ in range(T):
	(n, op) = map(int, input().split())
	s = input()
	if op == 0:
		print('YES')
		continue
	(i, j) = (0, n - 1)
	f = 1
	while op:
		if s[i] != s[j]:
			f = 0
			break
		op -= 1
		i += 1
		j -= 1
	if f:
		if i > j:
			print('NO')
		else:
			print('YES')
	else:
		print('NO')
