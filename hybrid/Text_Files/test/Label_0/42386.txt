def nodds(n):
	return n ** 2

def nevens(n):
	return n ** 2 + n

def neach(n):
	i = 1
	mode = True
	reach = 0
	odds = 0
	evens = 0
	while reach + i < n:
		if mode:
			odds += i
			mode = False
		else:
			evens += i
			mode = True
		reach += i
		i *= 2
	if mode:
		odds += n - reach
	else:
		evens += n - reach
	return (odds, evens)

def sm(n):
	x = neach(n)
	return nevens(x[1]) + nodds(x[0])
(l, r) = input().split()
(l, r) = (int(l), int(r))
print((sm(r) - sm(l - 1)) % 1000000007)
