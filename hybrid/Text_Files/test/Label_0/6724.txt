#include <bits/stdc++.h>
using namespace std;
inline long double min(long double a, long double b) { return a > b ? b : a; }
inline long double tim(long double dis, long double vel) {
  long double tm = dis / vel;
  return tm;
}
inline long double unidist(long double x, long double y, long double xi) {
  long double dist = sqrt((x - xi) * (x - xi) + y * y);
  return dist;
}
int main(void) {
  long double vb, vs;
  long double X[106] = {};
  long double T1[106] = {};
  long long n;
  cin >> n >> vb >> vs;
  long long place = 0;
  long double dist;
  long double mdist = 100000000;
  long double mintime = 100000000;
  for (int i = 1; i <= n; i++) {
    cin >> X[i];
    T1[i] = tim(X[i], vb);
  }
  long long x, y;
  cin >> x >> y;
  long double T2[106];
  long double T[106];
  for (int i = 2; i <= n; i++) {
    T2[i] = tim((dist = unidist(x, y, X[i])), vs);
    T[i] = T1[i] + T2[i];
    if (mintime > T[i]) {
      mintime = T[i];
      place = i;
    } else if (mintime == T[i] && dist < mdist) {
      place = i;
      mdist = dist;
    }
  }
  cout << place << endl;
  return 0;
}
