#include <bits/stdc++.h>
using namespace std;
int a[1010], b[1010], c[1010], s[10010];
int main() {
  int n;
  while (scanf("%d", &n) != EOF) {
    memset(a, 0, sizeof(a));
    memset(b, 0, sizeof(b));
    memset(c, 0, sizeof(c));
    memset(s, 0, sizeof(s));
    for (int i = 1; i <= n - 1; i++) scanf("%d", &a[i]);
    for (int i = 1; i <= n - 1; i++) scanf("%d", &b[i]);
    for (int i = 1; i <= n; i++) scanf("%d", &c[i]);
    for (int i = 1; i <= n; i++) {
      int k = 0;
      for (int j = 1; j <= i; j++) {
        k += a[j - 1];
      }
      k += c[i];
      for (int j = i; j <= n - 1; j++) {
        k += b[j];
      }
      s[i - 1] = k;
    }
    sort(s, s + n);
    printf("%d\n", s[0] + s[1]);
  }
  return 0;
}
