#include <bits/stdc++.h>
using namespace std;
const int Maxn = 200010;
const int inf = 2147483647;
long long read() {
  long long x = 0, f = 1;
  char ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9')
    x = (x << 3) + (x << 1) + (ch ^ 48), ch = getchar();
  return x * f;
}
map<long long, long long> f;
map<long long, long long>::iterator it;
long long n, a[Maxn];
int main() {
  n = read();
  for (int i = 1; i <= n; i++) a[i] = read();
  f[a[1] - 1] = 0;
  for (int i = 2; i <= n; i++) {
    while (f.begin() != f.end()) {
      it = f.end(), it--;
      long long x = (*it).first, k = (*it).second;
      if (x < a[i]) break;
      f[a[i] - 1] = max(f[a[i] - 1], k + (i - 1) * (x - x % a[i] - a[i]));
      f[x % a[i]] = max(f[x % a[i]], k + (i - 1) * (x - x % a[i]));
      f.erase(it);
    }
  }
  long long ans = 0;
  for (it = f.begin(); it != f.end(); it++)
    ans = max(ans, n * ((*it).first) + (*it).second);
  printf("%lld", ans);
}
