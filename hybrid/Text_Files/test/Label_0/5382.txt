#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:102400000,102400000")
using namespace std;
const long long inf = 0x3f3f3f3f3f3f3f3f;
const double pi = acos(-1.0);
const int maxn = 1e5;
int ans = 0;
struct node {
  long long n, d;
  bool train;
  node(long long nn, long long dd) : n(nn), d(dd) {}
  node(long long nn, long long dd, bool t) : n(nn), d(dd), train(t) {}
  bool friend operator<(node a, node b) { return a.d > b.d; }
};
vector<node> vec[maxn + 5];
int n;
long long dist[maxn + 5], dists[maxn + 5];
bool vis[maxn + 5];
int num[maxn + 5];
void djs(int v0) {
  memset(vis, 0, sizeof(vis));
  memset(dist, 0x3f, sizeof(dist));
  dist[v0] = 0;
  priority_queue<node> q;
  q.push(node(v0, 0));
  int cnt = 0;
  while (!q.empty()) {
    if (vis[q.top().n]) {
      q.pop();
      continue;
    }
    node temp = q.top();
    q.pop();
    vis[temp.n] = 1;
    for (int i = 0; i < vec[temp.n].size(); i++) {
      if (vis[vec[temp.n][i].n] &&
          vec[temp.n][i].d + dist[vec[temp.n][i].n] == dist[temp.n])
        num[temp.n]++;
      if (vis[vec[temp.n][i].n] &&
          dist[vec[temp.n][i].n] == temp.d + vec[temp.n][i].d) {
        num[vec[temp.n][i].n]++;
      }
      if (!vis[vec[temp.n][i].n] &&
          dist[vec[temp.n][i].n] > temp.d + vec[temp.n][i].d) {
        dist[vec[temp.n][i].n] = temp.d + vec[temp.n][i].d;
        q.push(node(vec[temp.n][i].n, dist[vec[temp.n][i].n]));
      }
    }
  }
  return;
}
long long train[maxn + 5];
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int m, k;
  cin >> n >> m >> k;
  for (int i = 1; i <= m; i++) {
    int u, v, d;
    cin >> u >> v >> d;
    vec[u].push_back(node(v, d, 0));
    vec[v].push_back(node(u, d, 0));
  }
  memset(train, 0x3f, sizeof(train));
  for (int i = 1; i <= k; i++) {
    long long v, d;
    cin >> v >> d;
    if (train[v] != inf) ans++;
    train[v] = min(train[v], d);
  }
  for (int i = 1; i <= n; i++) {
    if (train[i] != inf) {
      vec[1].push_back(node(i, train[i], 1));
      vec[i].push_back(node(1, train[i], 1));
    }
  }
  djs(1);
  for (int i = 2; i <= n; i++) {
    if (train[i] != inf) {
      if (train[i] > dist[i])
        ans++;
      else if (train[i] == dist[i] && num[i] >= 2)
        ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
