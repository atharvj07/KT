int n, m;
char ok[501][501];
int minX, minY, maxX, maxY;
bool isRect() {
  int xcount = 0;
  minX = MOD;
  minY = MOD;
  maxX = 0;
  maxY = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (ok[i][j] == 'X') {
        minX = min(minX, i);
        minY = min(minY, j);
        maxX = max(maxX, i);
        maxY = max(maxY, j);
        xcount++;
      }
    }
  }
  return ((maxX - minX + 1) * (maxY - minY + 1) == xcount);
}
int main() {
  std::ios::sync_with_stdio(false);
  cin.tie(0);
  ;
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> ok[i][j];
    }
  }
  op(isRect() ? "YES" : "NO");
  return 0;
}
