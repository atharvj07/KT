#include <bits/stdc++.h>
using namespace std;
int ti, td, tr, te, na, nb, f[4050][4050], pa[4050][30], pb[4050][30];
char a[4050], b[4050];
int main() {
  scanf("%d%d%d%d%s%s", &ti, &td, &tr, &te, a + 1, b + 1);
  na = strlen(a + 1);
  nb = strlen(b + 1);
  for (int i = 1; i < na + 1; i++) {
    f[i][0] = i * td;
    for (int j = 0; j < 26; j++) pa[i][j] = pa[i - 1][j];
    pa[i][a[i] - 'a'] = i;
  }
  for (int i = 1; i < nb + 1; i++) {
    f[0][i] = i * ti;
    for (int j = 0; j < 26; j++) pb[i][j] = pb[i - 1][j];
    pb[i][b[i] - 'a'] = i;
  }
  for (int i = 1; i < na + 1; i++)
    for (int j = 1; j < nb + 1; j++) {
      f[i][j] = min(f[i][j - 1] + ti, f[i - 1][j] + td);
      f[i][j] = min(f[i][j], f[i - 1][j - 1] + tr * (a[i] != b[j]));
      int c = pa[i - 1][b[j] - 'a'], d = pb[j - 1][a[i] - 'a'];
      if (c && d)
        f[i][j] = min(f[i][j], te + f[c - 1][d - 1] + (i - c - 1) * td +
                                   (j - d - 1) * ti);
    }
  printf("%d\n", f[na][nb]);
}
