import math
elem = int(input())
for i in range(elem):
	num = int(input())
	ar = []
	for i in input().split(' '):
		ar.append(int(i))
	curr = 0
	for i in range(len(ar) - 1):
		if ar[i] > ar[i + 1]:
			temp = ar[i] - ar[i + 1]
			ar[i + 1] = ar[i]
			curr = max(curr, temp)
	print(math.ceil(math.log(curr + 1, 2)))
