for aww in range(int(input())):
	b = []
	c = int(input())
	d = []
	for i in range(3):
		a = [*map(int, input().split())]
		b.append(a)
	d.append(b[0][0])
	for i in range(1, c):
		if b[0][i] == d[i - 1] and i != c - 1:
			d.append(b[1][i])
		elif i == c - 1:
			if b[0][i] == d[0]:
				if b[1][i] == d[i - 1]:
					d.append(b[2][i])
				else:
					d.append(b[1][i])
			elif b[0][i] == d[i - 1]:
				if b[1][i] == d[0]:
					d.append(b[2][i])
				else:
					d.append(b[1][i])
			else:
				d.append(b[0][i])
		else:
			d.append(b[0][i])
	print(' '.join(map(str, d)))
