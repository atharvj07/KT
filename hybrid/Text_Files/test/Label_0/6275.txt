#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> v(n);
  vector<pair<int, int>> p(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
    p[i] = make_pair(v[i], i);
  }
  sort(p.begin(), p.end());
  reverse(p.begin(), p.end());
  int ans = 0;
  for (int i = 0; i < k; i++) ans += p[i].first;
  cout << ans << endl;
  ans = 0;
  for (int i = 0; i < k; i++) {
    v[p[i].second] = 0;
  }
  int r = 1;
  for (int i = 0; i < n; i++) {
    if (v[i])
      ans++;
    else {
      if (r < k)
        cout << ans + 1 << ' ';
      else
        cout << ans + 1 + n - 1 - i << endl;
      ans = 0;
      r++;
    }
  }
}
