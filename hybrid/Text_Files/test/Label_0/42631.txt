t = int(input())
for _ in range(t):
	n = int(input())
	(pa, pb) = (-1, -1)
	flag = True
	for i in range(n):
		(a, b) = map(int, input().split())
		if a >= pa and b >= pb and (a >= b) and (a - pa >= b - pb):
			(pa, pb) = (a, b)
			continue
		else:
			flag = False
	if flag:
		print('YES')
	else:
		print('NO')
