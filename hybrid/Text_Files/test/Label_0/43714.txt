n = int(input())
arr = list(map(int, input().split()))
scoreSum = 0
average = 0
subject = 0
arr.sort()
for i in range(n):
	scoreSum += arr[i]
average = scoreSum / n
i = 0
if average >= 4.5:
	print(subject)
else:
	while average < 4.5:
		scoreSum += 5 - arr[i]
		average = scoreSum / n
		subject += 1
		i += 1
	print(subject)
