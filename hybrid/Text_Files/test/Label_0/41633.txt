result = 0
mod = 10 ** 6 + 3
(n, C) = map(int, input().split())

def fact(n):
	fact = 1
	for i in range(1, n + 1):
		fact = fact * i % mod
	return fact

def pow(a, b):
	exp = 1
	x = a % mod
	b = b % (mod - 1)
	while b > 0:
		if b % 2 == 1:
			exp = exp * x % mod
		x = x * x % mod
		b = b // 2
	return exp
result = fact(n + C) * pow(fact(n), mod - 2) * pow(fact(C), mod - 2) - 1
print(int(result % mod))
