def f(x):
	return 1 - x
n = int(input())
q = list(map(int, input().split()))
if n == 1:
	print(1 - q[0])
else:
	s = []
	for i in range(n - 1):
		for j in range(i, n):
			a = list(map(f, q))
			t = 0
			for k in range(i, j + 1):
				t += a[k]
			for m in range(0, i):
				t += q[m]
			for h in range(j + 1, n):
				t += q[h]
			s.append(t)
	print(max(s))
