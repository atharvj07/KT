import math
import bisect

def sieve(n):
	primes = [-1] * n
	primes[0] = -1
	primes[1] = -1
	i = 2
	has_comp = False
	sqrt_n = int(n ** 0.5 + 1)
	while i <= sqrt_n:
		if primes[i] == -1:
			primes[i] = 1
			for j in range(i * i, n, i):
				has_comp = True
				primes[j] = 2
		i += 1
	for i in range(sqrt_n, n):
		if primes[i] == -1:
			primes[i] = 1
	return (has_comp, primes)
n = int(input())
(has_comp, res) = sieve(n + 2)
if has_comp:
	print(2)
else:
	print(1)
res_r = ''
for i in res[2:]:
	res_r = res_r + str(i) + ' '
print(res_r)
