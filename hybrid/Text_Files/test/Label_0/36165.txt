import java.util.*;
import java.io.*;
public class RockPaperScissors 
{
	public static void main(String[] args)
	{
		Scanner in = new Scanner(System.in);
		
		int n = in.nextInt();
		
		char[] nike = in.next().toCharArray();
		char[] poly = in.next().toCharArray();
		
		int m = nike.length;
		int k = poly.length;
		
		int lcm = m * (k / gcd(m, k));
		
		int nPinches = 0;
		int pPinches = 0;
		
		for(int x = 0; x < lcm; x++)
		{
			int winner = play(nike[x % nike.length], poly[x % poly.length]);
			
			if(winner == 1)
			{
				nPinches += n / lcm;
				
				if(x < n % lcm)
				{
					nPinches++;
				}
			}
			else if(winner == 0)
			{
				pPinches += n / lcm;
				
				if(x < n % lcm)
				{
					pPinches++;
				}
			}
		}
		
		System.out.println(nPinches + " " + pPinches);
	}
	
	public static int gcd(int a, int b)
	{
		if(b == 0)
		{
			return a;
		}
		else
		{
			return gcd(b, a % b);
		}
	}
	
	public static int play(char n, char p)
	{
		if(n == p)
		{
			return -1;
		}
		else if(n == 'R')
		{
			if(p == 'P')
			{
				return 1;
			}
			else
			{
				return 0;
			}
		}
		else if(n == 'P')
		{
			if(p == 'S')
			{
				return 1;
			}
			else
			{
				return 0;
			}
		}
		else
		{
			if(p == 'R')
			{
				return 1;
			}
			else
			{
				return 0;
			}
		}
	}
}
