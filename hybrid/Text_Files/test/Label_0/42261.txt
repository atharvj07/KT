(l, s) = (int(input()), input())
ans = int(s)

def check(a, b):
	b = b[::-1]
	(b, a) = (list(b), list(a))
	while len(b) and b[-1] == '0':
		b.pop()
		a.append('0')
	if not len(b):
		return
	b = b[::-1]
	b = ''.join(b)
	a = ''.join(a)
	global ans
	ans = min(ans, int(a) + int(b))

def check2(a, b):
	b = b[::-1]
	a = list(a)
	b = list(b)
	while len(a) and a[-1] == '0':
		a.pop()
		b.append('0')
	if len(a) <= 1:
		return
	b.append(a.pop())
	a = ''.join(a)
	b = ''.join(b)
	b = b[::-1]
	global ans
	ans = min(ans, int(a) + int(b))
check(s[:l // 2], s[l // 2:])
check2(s[:l // 2], s[l // 2:])
if l >= 3:
	check(s[:l // 2 + 1], s[l // 2 + 1:])
	check2(s[:l // 2 + 1], s[l // 2 + 1:])
print(ans)
