def f1(a, b, c):
	d1 = abs(a - b)
	d2 = min(c, d1)
	if a < b:
		a += d2
	else:
		b += d2
	c = (c - d2) // 2
	a += c
	b += c
	return min(a, b)

def f2(s):
	(a, b, c) = map(int, s.split())
	return max(f1(a, b, c), f1(b, c, a), f1(c, a, b))
q = int(input())
lines = [''] * q
for i in range(q):
	lines[i] = input()
for s in lines:
	print(f2(s))
