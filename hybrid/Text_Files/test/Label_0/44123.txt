import sys
input = sys.stdin.readline
I = lambda : list(map(int, input().split()))
s = input().strip()
(m,) = I()
l = [i + 1 for i in range(10) if s[i] == '1']
n = len(l)
q = [[0, 0, 0, []]]
while q:
	(x, cb, mv, p) = q.pop()
	if mv == m:
		print('YES')
		print(*p)
		exit()
	for i in range(n):
		if l[i] != x and l[i] > cb:
			q.append([l[i], l[i] - cb, mv + 1, p + [l[i]]])
print('NO')
