    void run() {
        int tc=1;
        while(tc-->0) {
            int n=ni(),a=ni(),b=ni(),k=ni(),f=ni(),id=0;
            Map<String, Integer> uid = new HashMap<>();
            int[][] trips = new int[n][2], tab = new int[0x25F][0x25F];
			for(int i=0;i<n;i++) {
				String u=ns(),v=ns();
				uid.put(u,uid.containsKey(u)?uid.get(u):++id);
				uid.put(v,uid.containsKey(v)?uid.get(v):++id);
				trips[i][0]=uid.get(u); // from
				trips[i][1]=uid.get(v); // to
			}
			long tot_amt=0;
			PriorityQueue<int[]> pq = new PriorityQueue<>((x,y)->tab[x[0]][x[1]]-tab[y[0]][y[1]]);
			Map<Integer,Set<Integer>> map = new HashMap<>(); // map contains all travel cards from u to v
			for(int i=0;i<n;i++) {
				int cost = i==0?a:trips[i-1][1]==trips[i][0]?b:a;
					// from       	// to
				int u=trips[i][0], v=trips[i][1];
				if(u>v) u^=v^(v=u);
				if(map.containsKey(u) && map.get(u).contains(v)) {
					tab[u][v]+=cost;
					tab[v][u]+=cost;
				}
				else if(tab[u][v]+cost<=f || k==0) {
					tab[u][v]+=cost;
					tab[v][u]+=cost;
					tot_amt+=cost;
				}
				else {
					if(pq.size()<k) {
						map.put(u,map.getOrDefault(u,new HashSet<>()));
						map.get(u).add(v);
						tot_amt+=f-tab[u][v];
						tab[u][v]+=cost;
						tab[v][u]+=cost;
						pq.add(new int[]{u,v});
					}
                    else {
                        PriorityQueue<int[]> tmp = new PriorityQueue<>((x,y)->tab[x[0]][x[1]]-tab[y[0]][y[1]]);
                        tmp.addAll(pq); pq.clear(); pq.addAll(tmp);
                        
                        if(tab[pq.peek()[0]][pq.peek()[1]] >= tab[u][v]+cost) {
    						tab[u][v] += cost;
    						tab[v][u]+=cost;
    						tot_amt += cost;
    					}
    					else {
    						int x=pq.peek()[0],y=pq.peek()[1];
    						tot_amt += tab[x][y] - tab[u][v];
    						pq.poll();
    						map.get(x).remove(y);
    						tab[u][v] += cost;
    						tab[v][u] += cost;
                            map.put(u,map.getOrDefault(u,new HashSet<>()));
    						map.get(u).add(v);
    						pq.add(new int[]{u,v});
    					}
                    }
				}
			}
			out.println(tot_amt);
        }
        out.flush(); 
        out.close();
    }

    static class FastReader {
        private InputStream stream;
        private byte[] buf = new byte[1024];
        private int curChar;
        private int numChars;
        private FastReader.SpaceCharFilter filter;
        public FastReader(InputStream stream) {
            this.stream = stream;
        }

        public int read() {
            if (numChars == -1) throw new InputMismatchException();
            if (curChar >= numChars) {
                curChar = 0;
                try {
                    numChars = stream.read(buf);
                } catch (IOException e) {
                    throw new InputMismatchException();
                }
                if (numChars <= 0) return -1;
            }
            return buf[curChar++];
        }

        public int nextInt() {
            int c = read();
            while (isSpaceChar(c)) c = read();
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = read();
            }
            int res = 0;
            do {
                if (c < '0' || c > '9') throw new InputMismatchException();
                res *= 10;
                res += c - '0';
                c = read();
            }
            while (!isSpaceChar(c));
            return res * sgn;
        }
        
        public long nextLong() {
            int c = read();
            while (isSpaceChar(c)) c = read();
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = read();
            }
            long res = 0;
            do {
                if (c < '0' || c > '9') throw new InputMismatchException();
                res = res*1L*10;
                res += c - '0';
                c = read();
            }
            while (!isSpaceChar(c));
            return res *1L* sgn;
        }
        
        public String next() {
            int c = read();
            while (isSpaceChar(c)) c = read();
            StringBuilder res = new StringBuilder();
            do {
                res.appendCodePoint(c);
                c = read();
            } while (!isSpaceChar(c));
            return res.toString();
        }

        public boolean isSpaceChar(int c) {
            if (filter != null) return filter.isSpaceChar(c);
            return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
        }

        public interface SpaceCharFilter {
            public boolean isSpaceChar(int ch);

        }
    }
