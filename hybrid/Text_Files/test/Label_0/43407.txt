for _ in range(int(input())):
	(n, k) = map(int, input().split())
	a = list(map(int, input().split()))
	np = [0] * n
	vis = [0] * n
	sm = 0
	for i in range(1, n - 1):
		if a[i] > a[i - 1] and a[i] > a[i + 1]:
			np[i] += 1
			vis[i] = 1
		sm += np[i]
		np[i] = sm
	np[n - 1] = sm
	(t, l) = (0, 0)
	for i in range(n - k + 1):
		p = np[i + k - 1] - np[max(i - 1, 0)]
		if vis[i + k - 1]:
			p -= 1
		if vis[i]:
			p -= 1
		if p > t:
			l = i
			t = p
	print(t + 1, l + 1)
