#include <bits/stdc++.h>
using namespace std;
class seg_mx {
 public:
  static const int N = 1 << 17;
  int dp[1 << 18];
  seg_mx() { memset(dp, -1, sizeof(dp)); }
  void update(int k, int v) {
    k += N - 1;
    dp[k] = v;
    while (k > 0) {
      k = (k - 1) / 2;
      dp[k] = max(dp[k * 2 + 1], dp[k * 2 + 2]);
    }
  }
  int query(int a, int b, int k = 0, int l = 0, int r = N) {
    if (b <= l || r <= a) return -1;
    if (a <= l && r <= b) return dp[k];
    int mid = (l + r) / 2;
    int vl = query(a, b, k * 2 + 1, l, mid);
    int vr = query(a, b, k * 2 + 2, mid, r);
    return max(vl, vr);
  }
};
class seg_mn {
 public:
  static const int N = 1 << 17;
  int dp[1 << 18];
  seg_mn() {
    for (int i = 0; i < N * 2; i++) {
      dp[i] = 1145141;
    }
  }
  void update(int k, int v) {
    k += N - 1;
    dp[k] = v;
    while (k > 0) {
      k = (k - 1) / 2;
      dp[k] = min(dp[k * 2 + 1], dp[k * 2 + 2]);
    }
  }
  int query(int a, int b, int k = 0, int l = 0, int r = N) {
    if (b <= l || r <= a) return 1145141;
    if (a <= l && r <= b) return dp[k];
    int mid = (l + r) / 2;
    int vl = query(a, b, k * 2 + 1, l, mid);
    int vr = query(a, b, k * 2 + 2, mid, r);
    return min(vl, vr);
  }
};
seg_mx s_mx[2];
seg_mn s_mn[2];
int n, k;
string s;
int col[100005];
int sum[100005];
int main(void) {
  scanf("%d%d", &n, &k);
  cin >> s;
  for (int i = 0; i < n; i++) {
    if (s[i] == '0') {
      col[i] = 0;
    } else {
      col[i] = 1;
    }
    sum[i + 1] += sum[i];
    sum[i + 1] += col[i];
  }
  for (int i = 0; i + k <= n; i++) {
    int val = sum[i] + sum[n] - sum[i + k];
    if (val == 0 || val == n - k) {
      printf("tokitsukaze\n");
      return 0;
    }
  }
  for (int i = 0; i < n; i++) {
    s_mx[col[i]].update(i, i);
    s_mn[col[i]].update(i, i);
    s_mx[1 - col[i]].update(i, -1);
    s_mn[1 - col[i]].update(i, 1145141);
  }
  bool flag = false;
  for (int i = 0; i + k <= n; i++) {
    for (int side = 0; side < 2; side++) {
      bool flag2 = true;
      for (int j = 0; j < 2; j++) {
        int mm = max(s_mx[j].query(0, i), s_mx[j].query(i + k, n));
        int mi = min(s_mn[j].query(0, i), s_mn[j].query(i + k, n));
        if (side == j) {
          mm = max(mm, i + k - 1);
          mi = min(mi, i);
        }
        if ((mm - mi) <= (k - 1)) flag2 = false;
      }
      if (flag2) flag = true;
    }
  }
  printf("%s\n", flag ? "once again" : "quailty");
  return 0;
}
