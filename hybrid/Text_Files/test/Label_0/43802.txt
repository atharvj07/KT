n = int(input())
arr = list(map(int, input().split()))
flag = 0
counter = [0, 0, 0]
for i in range(len(arr)):
	if arr[i] == 50:
		if counter[0] < 1:
			print('NO')
			flag = 1
			break
		counter[1] += 1
		counter[0] -= 1
	elif arr[i] == 100:
		if counter[0] > 0 and counter[1] > 0:
			counter[2] += 1
			counter[0] -= 1
			counter[1] -= 1
		elif counter[1] < 1 and counter[0] > 2:
			counter[2] += 1
			counter[0] -= 3
		else:
			print('NO')
			flag = 1
			break
	elif arr[i] == 25:
		counter[0] += 1
if flag == 0:
	print('YES')
