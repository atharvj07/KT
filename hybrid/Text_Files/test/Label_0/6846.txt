#include <bits/stdc++.h>
using namespace std;
const int N = 1500 + 10;
long long n, x[N], y[N], ans[N], subtree_sz[N];
vector<int> tmp_v[N], v[N];
pair<double, long long> vo[N];
vector<bool> visit(N, false);
void to_directed(int now_v) {
  if (visit[now_v]) return;
  visit[now_v] = true;
  for (auto child : tmp_v[now_v]) {
    if (visit[child]) continue;
    v[now_v].push_back(child);
    to_directed(child);
  }
}
void cal_sz(int now_v) {
  if (visit[now_v]) return;
  visit[now_v] = true;
  for (auto child : v[now_v]) cal_sz(child);
  subtree_sz[now_v] = 1;
  for (auto child : v[now_v]) subtree_sz[now_v] += subtree_sz[child];
}
void dfs(int now_v, vector<int>& nodes) {
  vector<pair<int, int> > find_first;
  for (int i = 0; i < nodes.size(); ++i)
    find_first.push_back(make_pair(x[nodes[i]] + y[nodes[i]], nodes[i]));
  sort(find_first.begin(), find_first.end());
  ans[find_first.back().second] = now_v;
  long long now_x = x[find_first.back().second];
  long long now_y = y[find_first.back().second];
  find_first.pop_back();
  nodes.clear();
  for (int i = 0; i < find_first.size(); ++i)
    nodes.push_back(find_first[i].second);
  vector<pair<double, int> > cal_nxt;
  for (int i = 0; i < nodes.size(); ++i)
    cal_nxt.push_back(
        make_pair(atan2(now_y - y[nodes[i]], now_x - x[nodes[i]]), nodes[i]));
  sort(cal_nxt.begin(), cal_nxt.end());
  int idx = 0;
  for (auto child : v[now_v]) {
    vector<int> tmp;
    for (int i = idx; i < idx + subtree_sz[child]; ++i)
      tmp.push_back(cal_nxt[i].second);
    dfs(child, tmp);
    idx += subtree_sz[child];
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.setf(ios::fixed);
  cout.precision(20);
  cin >> n;
  for (int i = 0; i < n - 1; ++i) {
    int foo, bar;
    cin >> foo >> bar;
    tmp_v[foo].push_back(bar);
    tmp_v[bar].push_back(foo);
  }
  for (int i = 1; i <= n; ++i) cin >> x[i] >> y[i];
  to_directed(1);
  fill(visit.begin(), visit.end(), false);
  cal_sz(1);
  vector<int> init_nodes;
  for (int i = 1; i <= n; ++i) init_nodes.push_back(i);
  dfs(1, init_nodes);
  for (int i = 1; i <= n; ++i) cout << ans[i] << " \n"[i == n];
}
