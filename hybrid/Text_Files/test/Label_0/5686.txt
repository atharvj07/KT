#include <bits/stdc++.h>
using namespace std;
int a, b[26], cnt, ch;
int c[24] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31,
             31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
int d[24] = {31, 29, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31,
             31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
int main() {
  scanf("%d", &a);
  for (int i = 0; i < a; i++) {
    scanf("%d", &b[i]);
  }
  for (int i = 0; i < 24; i++) {
    int z = i;
    int x = a;
    cnt = 0;
    int y = 0;
    while (x--) {
      if (b[y] != c[z]) break;
      y = (y + 1) % a;
      z = (z + 1) % 24;
      cnt++;
    }
    if (cnt == a) {
      ch = 1;
      break;
    }
    z = i;
    y = 0;
    cnt = 0;
    x = a;
    while (x--) {
      if (b[y] != d[z]) break;
      y = (y + 1) % a;
      z = (z + 1) % 24;
      cnt++;
    }
    if (cnt == a) {
      ch = 1;
      break;
    }
  }
  if (ch)
    printf("Yes");
  else
    printf("No");
}
