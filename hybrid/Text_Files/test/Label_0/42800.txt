a = [int(i) for i in str(input()).split()]
n = a[0]
x = a[1]
(li, ls) = ([], [])
for i in range(n):
	h = [int(i) for i in str(input()).split()]
	li.append(h[0])
	ls.append(h[1])
minutei = 1
watchedMin = 0
best_i = 0
while minutei <= ls[-1] and best_i < n:
	if (li[best_i] - minutei) // x > 0:
		minutei += x * ((li[best_i] - minutei) // x)
	else:
		watchedMin += ls[best_i] - minutei + 1
		minutei = ls[best_i] + 1
		best_i += 1
print(watchedMin)
