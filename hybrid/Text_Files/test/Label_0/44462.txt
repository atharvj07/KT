a = int(input())
z = list(map(int, input().split()))
dp = [0]
for i in range(len(z)):
	dp.append(dp[-1] + z[i])
r = sorted(z)
rp = [0]
for i in range(len(z)):
	rp.append(rp[-1] + r[i])
l = int(input())
for i in range(l):
	(x, y, p) = map(int, input().split())
	if x == 1:
		print(dp[p] - dp[y] + z[y - 1])
	else:
		print(rp[p] - rp[y] + r[y - 1])
