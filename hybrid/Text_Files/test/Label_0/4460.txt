#include <bits/stdc++.h>
int main() {
  unsigned n;
  std::cin >> n;
  std::string s = "\n";
  unsigned l = 0, k = 0;
  for (unsigned i = 0; i < n; ++i) {
    int x;
    std::cin >> x;
    if (x == 1 && i != 0) {
      ++k;
      s += std::to_string(l);
      s += " ";
      l = 1;
    } else {
      ++l;
    }
  }
  ++k;
  s += std::to_string(l);
  std::cout << k << s << std::endl;
  return 0;
}
