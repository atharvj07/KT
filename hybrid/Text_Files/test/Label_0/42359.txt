(n, m, k) = map(int, input().split())
a = list(map(int, input().split()))
b = sorted(enumerate(a), key=lambda x: (x[1], x[0]), reverse=True)
b = b[:m * k]
b = sorted(b)
rec = []
s = 0
tmp = 0
for i in range(len(b)):
	tmp += 1
	if tmp == m:
		rec.append(b[i][0] + 1)
		tmp = 0
	s += b[i][1]
print(s)
print(' '.join(map(str, rec[:-1])))
