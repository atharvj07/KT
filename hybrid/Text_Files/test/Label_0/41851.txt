def distance(list):
	a = list[0]
	b = list[1]
	c = list[2]
	return abs(a - b) + abs(a - c) + abs(b - c)
n = int(input())
for i in range(n):
	instring = input()
	inlist = instring.split()
	inlist = [int(i) for i in inlist]
	inlist.sort()
	if inlist[0] < inlist[1] < inlist[2]:
		inlist[0] = inlist[0] + 1
		inlist[2] = inlist[2] - 1
	elif inlist[0] == inlist[1] == inlist[2]:
		print(0)
		continue
	elif inlist[0] == inlist[1]:
		if inlist[1] == inlist[2] - 1:
			print(0)
			continue
		inlist[2] = inlist[2] - 1
		inlist[1] = inlist[1] + 1
		inlist[0] = inlist[0] + 1
	elif inlist[1] == inlist[2]:
		if inlist[0] == inlist[2] - 1:
			print(0)
			continue
		inlist[2] = inlist[2] - 1
		inlist[1] = inlist[1] - 1
		inlist[0] = inlist[0] + 1
	print(distance(inlist))
