import sys
import math

def isPrime(p):
	for x in range(2, int(math.sqrt(p)) + 1):
		if p % x == 0:
			return False
	return True

def nextPrime(n):
	while not isPrime(n):
		n += 1
	return n
for line in sys.stdin:
	n = int(line)
	if n == 8:
		print(13)
		print('1 2')
		print('1 3')
		print('2 3')
		print('1 4')
		print('2 4')
		print('1 5')
		print('2 5')
		print('1 6')
		print('2 6')
		print('1 7')
		print('1 8')
		print('5 8')
		print('7 8')
		break
	p = nextPrime(n)
	print(p)
	for i in range(1, n + 1):
		print('%d %d' % (i, i % n + 1))
	for i in range(1, p - n + 1):
		print('%d %d' % (3 * i - 2, 3 * i))
	break
