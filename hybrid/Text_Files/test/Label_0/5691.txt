#include <bits/stdc++.h>
using namespace std;
static const long long LLMAX = 0x7fffffffffffffffLL;
static const long long LLMIN = 0x8000000000000000LL;
static const int IMAX = 0x7fffffff;
static const int IMIN = 0x80000000;
long long m, l, r, k;
template <class T>
struct Matrix {
  T data[2][2];
  Matrix& operator=(const Matrix& rhs) {
    for (int i = 0; i < 2; ++i) {
      for (int j = 0; j < 2; ++j) {
        this->data[i][j] = rhs.data[i][j];
      }
    }
    return *this;
  }
  Matrix& operator*=(const Matrix& rhs) {
    Matrix temp;
    for (int i = 0; i < 2; ++i) {
      for (int j = 0; j < 2; ++j) {
        temp.data[i][j] = 0;
        for (int k = 0; k < 2; ++k) {
          temp.data[i][j] += (this->data[i][k] * rhs.data[k][j]) % m;
          temp.data[i][j] %= m;
        }
      }
    }
    *this = temp;
    return *this;
  }
};
template <class T>
Matrix<T> quickpow(long long n) {
  Matrix<T> p;
  p.data[0][0] = 1;
  p.data[0][1] = 1;
  p.data[1][0] = 1;
  p.data[1][1] = 0;
  Matrix<T> I;
  I.data[0][0] = 1;
  I.data[0][1] = 0;
  I.data[1][1] = 1;
  I.data[1][0] = 0;
  Matrix<T> r = I;
  while (n) {
    if (n & 1) r *= p;
    n >>= 1;
    p *= p;
  }
  return r;
}
int main() {
  while (~scanf("%I64d%I64d%I64d%I64d", &m, &l, &r, &k)) {
    long long t = (r - l) / (k - 1);
    while (t > 1) {
      if (t * (k - 1) + 1 <= r - ((l - 1) / t + 1) * t + 1) break;
      long long n = 1 + (r / t);
      t -= ((n * t - r) + (n - 1)) / n;
    }
    Matrix<long long> ans = quickpow<long long>(t);
    printf("%I64d\n", ans.data[1][0]);
  }
  return 0;
}
