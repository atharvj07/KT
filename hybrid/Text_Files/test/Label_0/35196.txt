
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class D {
    
    static int 
        MAXK=50;
    static int[] BMASK=new int[4];
    
    static {
        BMASK[0]=1; 
        for (int i=1; i<BMASK.length; i++) {
            BMASK[i]=BMASK[i-1]<<1;
        }
    }
    
    static long[] f=new long[MAXK];
    
    
    static boolean bit(int i, int j) {
        return (i&BMASK[j])>0;
    }

    static int bit(int i, int j, boolean b) {
        if(b) {
            i|=BMASK[j];
        } else {
            i&=~BMASK[j];
        }
        return i;
    }
    
    static boolean compute(int k, int x, int f0, int f1, int mask) {
        f[0]=f0; f[1]=f1;
        for (int i=2; i<k; i++) {
            f[i]=f[i-1]+f[i-2];
            if(bit(mask,1)&&bit(mask,2)) {
                f[i]++;
            }
            boolean tmps=bit(mask,0), tmpe=bit(mask,3);
            mask=mask>>2;
            mask=bit(mask,2,tmps);
            mask=bit(mask,3,tmpe);
            
            if(f[i]>x) {
                return false;
            }
        }
        
        return f[k-1]==x;
    }

    static StringBuilder sb=new StringBuilder(), sb2=new StringBuilder();
    
    static void print_string(int i, int l, boolean sc, boolean ea) {
        sb2.setLength(0);
        for (int j=0; j<i; j++) {
            sb2.append("AC");
        }
        
        sb.setLength(0);
        if(sc) {
            sb.append('C');
        }
        
        sb.append(sb2.toString());
        
        while (sb.length()<l-1) {
            sb.append('Z');
        }

        if(ea) {
            sb.append('A');
        } else {
            if(sb.length()<l) {
                sb.append('Z');
            }
        }
        
        System.out.println(sb.toString());
    }
    
    static void compute(int k, int x, int n, int m) {
        int n2=n>>1, m2=m>>1;
        for (int i=0; i<=n2; i++) {
            for (int j=0; j<=m2; j++) {
                
                int r1=n-(i<<1), r2=m-(j<<1);
                
                for (int l=0; l<16; l++) {
                    int s1=(l&1)+((l>>1)&1);
                    int s2=((l>>2)&1)+((l>>3)&1);
                    
                    if(r1>=s1&&r2>=s2) {
                        if(compute(k, x, i, j, l)) {
                            print_string(i, n, bit(l,0), bit(l,1));
                            print_string(j, m, bit(l,2), bit(l,3));
                            
                            return;
                        }
                    }
                }
            }
        }
        
        System.out.println("Happy new year!");
    }
    
    
    
    public static void main(String[] args) throws IOException {
        BufferedReader r=new BufferedReader(new InputStreamReader(System.in));
        String l=r.readLine();
        String[] sl=l.split(" ");
        int k=Integer.parseInt(sl[0]);
        int x=Integer.parseInt(sl[1]);
        int n=Integer.parseInt(sl[2]);
        int m=Integer.parseInt(sl[3]);
        
        compute(k,x,n,m);
    }


}
