def is_all_zero(l):
	for i in l:
		if i != 0:
			return False
	return True

def main():
	n = int(input())
	for _ in range(n):
		l = [0] * 26
		s = input()
		for i in s:
			l[ord(i) - ord('a')] += 1
		pr = []
		l1 = l[::2]
		l2 = l[1::2]
		if is_all_zero(l1):
			for i in range(13):
				for j in range(l2[i]):
					pr.append(chr(ord('a') + 2 * i + 1))
			print(''.join(pr))
			continue
		if is_all_zero(l2):
			for i in range(13):
				for j in range(l1[i]):
					pr.append(chr(ord('a') + 2 * i))
			print(''.join(pr))
			continue
		res = (-1, -1)
		for i in range(13):
			for j in range(13):
				if not (i - j in (0, 1) or l[2 * i] * l[2 * j + 1] == 0):
					res = (i, j)
		if res == (-1, -1):
			print('No answer')
		else:
			for i in range(13):
				if i != res[0]:
					for j in range(l1[i]):
						pr.append(chr(ord('a') + 2 * i))
			for j in range(l1[res[0]]):
				pr.append(chr(ord('a') + 2 * res[0]))
			for j in range(l2[res[1]]):
				pr.append(chr(ord('a') + 2 * res[1] + 1))
			for i in range(13):
				if i != res[1]:
					for j in range(l2[i]):
						pr.append(chr(ord('a') + 2 * i + 1))
			print(''.join(pr))
	return
main()
