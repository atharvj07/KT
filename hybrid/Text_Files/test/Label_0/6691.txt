#include<bits/stdc++.h>
using namespace std;
const int N=1e5+10;
int n,rd[N],a[N],f[N];
vector<int> g[N];
queue<int> q;
inline bool cmp(int x,int y){return f[x]<f[y];}
int main(){
	scanf("%d",&n);
	for (int i=2;i<=n;++i){
		scanf("%d",a+i);
		g[a[i]].push_back(i);
		++rd[a[i]];
	}
	for (int i=1;i<=n;++i) if (!rd[i]) q.push(i);
	while (!q.empty()){
		int u=q.front();q.pop();
		sort(g[u].begin(),g[u].end(),cmp);
		for (vector<int>::iterator it=g[u].begin();it!=g[u].end();++it) f[u]=max(f[u],f[*it])+1;
		if (--rd[a[u]]==0) q.push(a[u]);
	}
	printf("%d",f[1]);
	return 0;
}
