import collections

def DCJ(n, heights):
	path = [{i + 1} for i in range(n)]

	def monotonic(x):
		temp = []
		for (i, h) in enumerate(x):
			while temp and temp[-1][1] < h:
				temp.pop()
			if temp:
				path[temp[-1][0]].add(i)
			temp.append((i, h))
		temp = []
		for (i, h) in enumerate(x[::-1]):
			while temp and temp[-1][1] < h:
				temp.pop()
			if temp:
				path[n - 1 - i].add(temp[-1][0])
			temp.append((n - 1 - i, h))
	monotonic(heights)
	monotonic([-x for x in heights])
	visited = [1] + [0] * (n - 1)
	queue = collections.deque()
	queue.append((0, 1))
	while queue:
		(i, jump) = queue.popleft()
		for j in path[i]:
			if j == n - 1:
				return jump
			if not visited[j]:
				visited[j] = 1
				queue.append((j, jump + 1))
n = int(input())
heights = list(map(int, input().split(' ')))
print(DCJ(n, heights))
