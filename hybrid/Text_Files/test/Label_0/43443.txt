import math
result = []

def check(x, shops, friends):
	if min(map(max, friends)) < x:
		return False
	for shop in shops:
		status = ''
		for value in shop:
			if value >= x:
				status += 'x'
		if len(status) >= 2:
			return True
	return False

def binary_search(func, l, r, *args, **kwargs):
	if l == r:
		return l
	c = math.ceil((l + r) / 2)
	if check(c, *args, **kwargs):
		return binary_search(func, c, r, *args, **kwargs)
	else:
		return binary_search(func, l, c - 1, *args, **kwargs)
for _ in range(int(input())):
	input()
	(m, n) = list(map(int, input().split()))
	shops = [list(map(int, input().split())) for i in range(m)]
	friends = [[shops[i][j] for i in range(m)] for j in range(n)]
	answer = binary_search(check, 0, 10 ** 9, shops, friends)
	result.append(answer)
print('\n'.join(map(str, result)))
