#include <bits/stdc++.h>
using namespace std;
const int INF = 1500000000;
const int MOD = 1000000007;
const int HASH = 137;
bool Go(int x, int y) {
  double gold = (1.0 + sqrt(5.0)) / 2.0;
  if (x > y) swap(x, y);
  int k = y - x;
  if (x == (int)floor(k * gold))
    return false;
  else
    return true;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int N;
  cin >> N;
  if (N == 1) {
    int x;
    cin >> x;
    if (x > 0)
      cout << "BitLGM" << endl;
    else
      cout << "BitAryo" << endl;
  } else if (N == 2) {
    int x, y;
    cin >> x >> y;
    if (Go(x, y))
      cout << "BitLGM" << endl;
    else
      cout << "BitAryo" << endl;
  } else {
    int x, y, z;
    cin >> x >> y >> z;
    if (x ^ y ^ z)
      cout << "BitLGM" << endl;
    else
      cout << "BitAryo" << endl;
  }
  return 0;
}
