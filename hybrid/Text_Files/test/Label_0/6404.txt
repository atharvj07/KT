#include<iostream>
#include<algorithm>
#include<vector>
#include<queue>
#include<string>
using namespace std;

vector<int> ans;
int pre[40];
int in[40];
int N;

struct Node {
	int left;
	int right;
};

void solve(int begin, int end, int& j) {
	int mid;
	for (int i = begin; i < end; i++) {
		if (in[i] == pre[j]) {
			mid = i;
		}
	}

	if(begin != mid)solve(begin, mid, ++j);
	if(mid + 1 != end)solve(mid + 1, end, ++j);
	ans.push_back(in[mid]);
}

int main() {

	cin >> N;
	for (int i = 0; i < N; i++) {
		cin >> pre[i];
	}
	for (int i = 0; i < N; i++) {
		cin >> in[i];
	}

	int j = 0;
	solve(0, N, j);
	
	for (int i = 0; i < N; i++) {
		cout << ans[i];
		if (i < N - 1)cout << " ";
	}
	cout << endl;
	
	return 0;
}
