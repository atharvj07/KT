import java.util.*;
import java.io.*;
public class Main
{
    static BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
    static PrintWriter pw=new PrintWriter(System.out);
    static ArrayList<Integer> edge[];
    static String st;
    static HashMap<Integer,Integer> par=new HashMap<>();
    static HashMap<Integer,Integer> cnt=new HashMap<>();
    public static void dfs(int v,TreeSet<Integer> ts,boolean vis[])
    {
        vis[v]=true;
        boolean b=true;
        for(int i=0;i<edge[v].size();i++)
        {
            int tp=edge[v].get(i);
            if(!vis[tp])
            dfs(tp,ts,vis);
            if(ts.contains(tp))
            b=false;
        }
        if(b)
        ts.add(v);
    }
    public static boolean dfs2(int v,int p,boolean vis[],int cntr,int k)
    {
        vis[v]=true;
        par.put(v,p);
        cnt.put(v,cntr);
        //boolean b=true;
        for(int i=0;i<edge[v].size();i++)
        {
            int tp=edge[v].get(i);
            if(tp!=p&&par.containsKey(tp)&&cntr-cnt.get(tp)+1<=k)
            {
                //ts.add(tp);
                st=v+" "+tp;
                return true;
            }
            else if(!vis[tp])
            {
                if(dfs2(tp,v,vis,cntr+1,k))
                return true;
            }
        }
        return false;
    }
    public static void solve()throws Exception
    {
        String str[]=br.readLine().split(" ");
        int n=Integer.parseInt(str[0]);
        int m=Integer.parseInt(str[1]);
        int k=Integer.parseInt(str[2]);
        edge=new ArrayList[n+1];
        for(int i=1;i<=n;i++)
        edge[i]=new ArrayList<>();
        for(int i=0;i<m;i++)
        {
            str=br.readLine().split(" ");
            int u=Integer.parseInt(str[0]);
            int v=Integer.parseInt(str[1]);
            edge[u].add(v);
            edge[v].add(u);
        }
        if(n==3)
        {
            if(m==3)
            {
                pw.println(2);
                pw.println(3);
                pw.println("1 2 3");
            }
            else
            {
                pw.println(1);
                if(edge[1].size()==2)
                pw.println("2 3");
                else if(edge[2].size()==2)
                pw.println("1 3");
                else
                pw.println("1 2");
            }
            return;   
        }
        TreeSet<Integer> ts=new TreeSet<>();
        //if(m<=n)
        //{
            ts.add(1);
            boolean vis[]=new boolean[n+1];
            dfs(1,ts,vis);
            for(int i=1;i<=n;i++)
            {
                if(!vis[i])
                dfs(i,ts,vis);
            }
            //for(int i=0;i<(k+1)/2;i++)
            //pw.print(ts.pollFirst()+" ");
            int cntr=0;
            if(ts.size()>=(k+1)/2)
            {
                pw.println(1);
                for(int i:ts)
                {
                    if(cntr==(k+1)/2)
                    break;
                    pw.print(i+" ");
                    cntr++;
                }
            }
        //}
        else
        {
            vis=new boolean[n+1];
            //ts.clear();
            dfs2(1,-1,new boolean[n+1],1,k);
            if(str==null)
            {
                for(int i=1;i<=n;i++)
                {
                    if(!vis[i])
                    dfs2(2,-1,new boolean[n+1],1,k);
                }
            }
            pw.println(2);
            int node=Integer.parseInt(st.split(" ")[0]);
            int last=Integer.parseInt(st.split(" ")[1]);
            ArrayList<Integer> vert=new ArrayList<>();
            vert.add(node);
            //ts.add(last);
            //System.out.println(node);
            while(par.get(node)!=last)
            {
                vert.add(par.get(node));
                node=par.get(node);
            }
            vert.add(last);
            pw.println(vert.size());
            for(int i=vert.size()-1;i>=0;i--)
            pw.print(vert.get(i)+" ");
        }
    }
    public static void main(String args[])throws Exception
    {
        //int t=Integer.parseInt(br.readLine());
        //for(int p=0;p<t;p++)
        solve();
        pw.flush();
        pw.close();
    }
}
