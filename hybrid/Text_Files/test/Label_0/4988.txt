#include <bits/stdc++.h>
using namespace std;
int main() {
  set<long long> S;
  long long n, x, a;
  cin >> n >> x;
  for (long long i = 0; i < n; i++) {
    cin >> a;
    S.insert(a);
  }
  long long s = 0;
  for (long long i = 0; i < x; i++) {
    if (S.count(i) == 0) s++;
  }
  if (S.count(x) > 0) s++;
  cout << s;
}
