#include <bits/stdc++.h>
using namespace std;
int main() {
  int N, K;
  cin >> N >> K;
  vector<int> seq(2 * N);
  for (int i = 0; i < 2 * N; i += 1) seq[i] = i + 1;
  for (int i = 0; i < K; i += 1) swap(seq[2 * i], seq[2 * i + 1]);
  for (int i = 0; i < 2 * N; i += 1) cout << seq[i] << " ";
  cout << endl;
  return 0;
}
