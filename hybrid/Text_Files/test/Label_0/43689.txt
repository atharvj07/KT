for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	b = c = 0
	for i in a:
		if i == 1:
			b += 1
		else:
			c += 1
	if c >= b:
		flag = True
	else:
		flag = False
	if flag:
		print(n // 2)
		for i in range(n // 2):
			print(0, end=' ')
		print()
	else:
		temp = n // 2 + n // 2 % 2
		print(temp)
		for i in range(temp):
			print(1, end=' ')
		print()
