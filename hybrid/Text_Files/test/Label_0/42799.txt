import sys
input = sys.stdin.readline

def inp():
	return int(input())

def inara():
	return list(map(int, input().split()))

def insr():
	s = input()
	return list(s[:len(s) - 1])

def invr():
	return map(int, input().split())
(n, k) = invr()
if 2 * k + 1 >= n:
	print(1)
	print(1 + (n - 1) // 2)
	exit(0)
l = 1 + (n - 1) // (2 * k + 1)
pos = []
for i in range(k + 1, 2 * k + 2):
	flag = False
	for j in range(k + 1, 2 * k + 2):
		temp = i + j + (l - 2) * (2 * k + 1)
		if temp == n:
			pos.append(i - k)
			pos.append(n - j + k + 1)
			for x in range(pos[0] + 2 * k + 1, pos[1] - k, 2 * k + 1):
				pos.append(x)
			flag = True
			break
	if flag:
		break
pos.sort()
print(l)
print(*pos)
