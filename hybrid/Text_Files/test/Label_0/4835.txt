#include <bits/stdc++.h>
using namespace std;
int main() {
  int a, b, c, d;
  scanf("%d:%d", &a, &b);
  c = a;
  d = b + 1;
  int i, j;
  for (i = c; i <= 23; i++) {
    for (j = d; j <= 59; j++) {
      if (i / 10 == j % 10 && i % 10 == j / 10) {
        printf("%02d:%02d", i, j);
        return 0;
      }
    }
    d = 0;
    if (c == 23) i = -1;
  }
}
