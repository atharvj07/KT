long long power(long long x,long long p)
{
    if(p==0){
        return 1;
    }
    if(p%2==1){
        return (x*power(x,p-1))%mod;
    }
    long long z=power(x,p/2);
    z*=z;
    z%=mod;
    return z;
}
vector<long long>fact,inv;
void inverse(long x)
{
    fact.clear();
    inv.clear();
    fact.resize(x+1);
    inv.resize(x+1);
    fact[0]=1;
    for(long long i=1;i<=x;i++){
        fact[i]=(fact[i-1]*i)%mod;
    }
    inv[x]=power(fact[x],mod-2);
    for(long long i=x-1;i>=0;i--){
        inv[i]=(inv[i+1]*(i+1))%mod;
    }
}
long long nck(long long n,long long k)
{
    if(n<0||k<0||n<k){
        return 0;
    }
    long long z=(fact[n]*inv[k])%mod;
    z=(z*inv[n-k])%mod;
    return z;
}
