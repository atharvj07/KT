#include <bits/stdc++.h>
const int MAXN = 205;
const int MAXK = 1005;
const int MOD = 1e9 + 7;
using namespace std;
int N, K;
int F[MAXN], idx[MAXN];
long long dp[2][MAXN][MAXK];
inline void f(long long &x, long long y) { x = (x + y) % MOD; }
int main() {
  scanf("%d%d", &N, &K);
  for (int i = 1; i <= N; i++) scanf("%d", &F[i]);
  sort(F + 1, F + 1 + N);
  for (int i = 1; i <= N + 1; i++) idx[i] = i % 2;
  for (int k = 0; k <= K; k++) dp[idx[N + 1]][0][k] = 1LL;
  for (int i = N; i > 0; i--) {
    for (int g = 0; g <= N; g++)
      for (int k = 0; k <= K; k++) {
        dp[idx[i]][g][k] = 0LL;
        int cost = k + (F[i] - F[i - 1]) * g;
        if (cost > K) continue;
        if (g > 0) f(dp[idx[i]][g][k], dp[idx[i + 1]][g - 1][cost] * g);
        f(dp[idx[i]][g][k], dp[idx[i + 1]][g][cost] * (g + 1));
        f(dp[idx[i]][g][k], dp[idx[i + 1]][g + 1][cost]);
      }
  }
  printf("%lld\n", dp[idx[1]][0][0]);
}
