#include <iostream>

using namespace std;

int main() {
	short t;
	cin >> t;
	while(t--) {
		short n;
		cin >> n;
		if(n % 2 == 0) {
			for(int i = 1; i <= n; i++) {
				if(i % 2 == 1) {
					printf("%d ", i + 1);
				}
				else {
					printf("%d ", i - 1);
				}
			}
		}
		else {
			printf("3 1 2 ");
			for(int i = 4; i <= n; i++) {
				if(i % 2 == 0) {
					printf("%d ", i + 1);
				}
				else {
					printf("%d ", i - 1);
				}
			}	
		}
		cout << endl;
	}
	return 0;
}
