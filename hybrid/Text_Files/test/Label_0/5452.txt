#include <bits/stdc++.h>
using namespace std;
int sum;
int a[200010];
map<int, int> mi;
map<int, int> mj;
int main() {
  int n, l, f = 0;
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> l;
    sum = 0;
    for (int j = 0; j < l; j++) {
      cin >> a[j];
      sum += a[j];
    }
    if (!f) {
      for (int j = 0; j < l; j++) {
        if (mi[sum - a[j]] != i + 1 && mj[sum - a[j]] != 0) {
          cout << "YES" << endl;
          cout << i + 1 << " " << j + 1 << endl;
          cout << mi[sum - a[j]] << " " << mj[sum - a[j]] << endl;
          return 0;
        } else {
          mi[sum - a[j]] = i + 1;
          mj[sum - a[j]] = j + 1;
        }
      }
    }
  }
  if (!f) cout << "NO" << endl;
  return 0;
}
