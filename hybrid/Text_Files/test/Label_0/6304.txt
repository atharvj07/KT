int make_tree(int ter,int per){
	if(per!=-1){ko[ter].era(find(ko[ter].begin(),ko[ter].end(),per));}
	oya[ter]=per;
	int kos=1;
	for(auto it:ko[ter]){kos+=make_tree(it,ter);}
	siz[ter]=kos;
	return kos;
}
