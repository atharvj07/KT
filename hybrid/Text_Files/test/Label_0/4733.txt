#include <bits/stdc++.h>
using namespace std;
template <typename T>
void debug(T _a) {
  cout << _a << " ";
}
template <typename T1, typename T2>
void debug(pair<T1, T2> _p) {
  cout << "{" << _p.first << ": " << _p.second << "}" << endl;
}
template <typename T>
void debug(vector<T> _aa) {
  for (auto i : _aa) debug(i);
  cout << endl;
}
template <typename T>
void debug(set<T> _aa) {
  for (auto i : _aa) debug(i);
  cout << endl;
}
template <typename T1, typename T2>
void debug(map<T1, T2> _mm) {
  for (auto i : _mm) debug(i);
}
const long long int INF = 1e18, MOD = 1e9 + 7;
const int N = 1e5;
const int di[] = {-1, 0, 1, 0}, dj[] = {0, 1, 0, -1};
const string YN[] = {"NO", "YES"};
vector<string> a;
int n, m;
bool check(string s, int ind) {
  int cnt = 0;
  for (int i = 0; i < m; i++) cnt += s[i] != a[ind][i];
  return cnt <= 1;
}
void solve() {
  cin >> n >> m;
  a = vector<string>(n);
  for (string &i : a) cin >> i;
  string s = a[0];
  bool ok = false;
  for (int i = 0; i < m; i++) {
    char c = s[i];
    for (char ch = 'a'; ch <= 'z' && !ok; ch++) {
      s[i] = ch;
      bool ok2 = true;
      for (int j = 0; j < n; j++) {
        ok2 = ok2 & check(s, j);
      }
      ok = ok2;
    }
    if (ok)
      break;
    else
      s[i] = c;
  }
  if (ok)
    cout << s << endl;
  else
    cout << -1 << endl;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  int t = 1;
  cin >> t;
  for (int i = 1; i <= t; i++) solve();
}
