def otherChar(x, y):
	for c in ('a', 'b', 'c'):
		if c != x and c != y:
			return c

def solve(a, b, t):
	n = len(a)
	e = sum((1 for i in range(n) if a[i] == b[i]))
	d = n - e
	for x in range(0, n + 1):
		if d - x + x // 2 <= t and x % 2 == 0:
			break
	if not (d - x + x // 2 <= t and x % 2 == 0):
		return -1
	w = d - x
	y = t - (d - x + x // 2)
	z = e - y
	if z < 0:
		return -1
	c = len(a) * [0]
	j = 0
	for i in range(n):
		if a[i] != b[i]:
			if w > 0:
				c[i] = otherChar(a[i], b[i])
				w -= 1
			else:
				c[i] = a[i] if j % 2 else b[i]
				x -= 1
			j += 1
		elif y > 0:
			c[i] = otherChar(a[i], b[i])
			y -= 1
		else:
			c[i] = a[i]
			z -= 1
	if not w == x == y == z == 0:
		return -1
	return ''.join(map(str, c))

def check(a, b, t, c):
	assert sum((1 for i in range(len(a)) if a[i] != c[i])) == t
	assert sum((1 for i in range(len(a)) if b[i] != c[i])) == t
(_, t) = map(int, input().split())
a = input()
b = input()
print(solve(a, b, t))
