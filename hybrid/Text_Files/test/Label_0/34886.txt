
import java.util.Scanner;

public class ClassA {
	public static void main(String[] arges) {
		Scanner in=new Scanner(System.in);
		int T=in.nextInt();
		while(T!=0) {
			T--;
			int n=in.nextInt();
			int[] dp=new int[n];
			for(int i=0;i<n;i++) {
				dp[i]=in.nextInt();
			}
			for(int i=n-1;i>=0;i--) {
				if(i%2==1) {
					dp[i]=dp[i]*(-1);
					System.out.print(dp[i]+" ");
				}
					
				else if(i%2==0&&i!=0)
					System.out.print(dp[i]+" ");
				else System.out.println(dp[i]);
			}
		}
	}
}
