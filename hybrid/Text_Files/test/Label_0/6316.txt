#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  if (n % 2 == 0)
    puts("NO");
  else {
    puts("YES");
    int t = 1;
    for (int i = 1; i <= n; i++) {
      cout << t << " ";
      if (i % 2 == 1)
        t += 3;
      else if (i % 2 == 0)
        t++;
    }
    t = 2;
    for (int i = 1; i <= n; i++) {
      cout << t << " ";
      if (i % 2 == 1)
        t++;
      else if (i % 2 == 0)
        t += 3;
    }
  }
  return 0;
}
