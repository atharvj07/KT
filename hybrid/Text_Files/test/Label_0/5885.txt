#include <bits/stdc++.h>
using namespace std;
long long fact[200005];
long long bigmod(long long x, long long p) {
  if (p == 0) return 1LL;
  long long y = bigmod(x, p / 2);
  if (p % 2 == 1) {
    return ((((y % 1000000007) * (y % 1000000007)) % 1000000007) *
            (x % 1000000007)) %
           1000000007;
  }
  return (y % 1000000007) * (y % 1000000007) % 1000000007;
}
long long invmod(long long x) { return bigmod(x, 1000000007 - 2); }
void factorial() {
  fact[0] = 1;
  for (int i = 1; i <= 200000; i++) fact[i] = (fact[i - 1] * i) % 1000000007;
}
int main() {
  factorial();
  int n, m, g;
  cin >> n >> m >> g;
  if (n == 0) {
    if (m == 1)
      cout << g << endl;
    else if (g == 0)
      cout << 1 << endl;
    else
      cout << 0 << endl;
    return 0;
  }
  if (m == 0) {
    if (g != (n % 2))
      cout << 1 << endl;
    else
      cout << 0 << endl;
    return 0;
  }
  if (!g) {
    long long ans = 0;
    for (int i = 0; i <= n; i += 2) {
      int zr = n - i;
      int one = m - 1;
      long long temp = ((fact[zr + one] * invmod(fact[zr])) % 1000000007 *
                        invmod(fact[one])) %
                       1000000007;
      ans = (ans + temp) % 1000000007;
    }
    if (m == 1) {
      if (n % 2 == 0)
        ans = (ans - 1 + 1000000007) % 1000000007;
      else
        ans = (ans + 1) % 1000000007;
    }
    cout << ans << endl;
  } else {
    long long ans = 0;
    for (int i = 1; i <= n; i += 2) {
      int zr = n - i;
      int one = m - 1;
      long long temp = ((fact[zr + one] * invmod(fact[zr])) % 1000000007 *
                        invmod(fact[one])) %
                       1000000007;
      ans = (ans + temp) % 1000000007;
    }
    if (m == 1) {
      if (n % 2) {
        ans = (ans - 1 + 1000000007) % 1000000007;
      } else {
        ans = (ans + 1) % 1000000007;
      }
    }
    cout << ans << endl;
  }
}
