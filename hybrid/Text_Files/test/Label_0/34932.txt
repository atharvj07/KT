import java.util.*;
public class ProgressBar{
   public static void main(String[] Args){
      Scanner sc = new Scanner(System.in);
      int n = sc.nextInt();
      int k = sc.nextInt();
      int t = sc.nextInt();
      int r = (n*k*t)/100;
      for(int j =0; j < n;j++){
         if(r >= k){
            System.out.print(k+" ");
            r-=k;
         }
         else{
            System.out.print(r+" ");
            r = 0;
         }
      }
      System.out.println();
   }
}
