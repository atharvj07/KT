#include <bits/stdc++.h>
using namespace std;
int dp[100000] = {0};
int b[100000] = {0};
int main() {
  string s;
  cin >> s;
  int len = s.length();
  int i = 0;
  int j = len - 1;
  while (s[i] != 'b' && i < len) i++;
  while (s[j] != 'b' && j >= 0) j--;
  if (i == len) {
    cout << len;
    return 0;
  }
  int ii = i;
  int jj = j;
  b[0] = 0;
  dp[0] = 0;
  if (s[0] == 'a')
    dp[0] = 1;
  else
    b[0] = 1;
  for (i = 1; i < len; i++) {
    if (s[i] == 'b') {
      b[i] = 1 + b[i - 1];
      dp[i] = dp[i - 1];
    } else {
      b[i] = b[i - 1];
      dp[i] = 1 + dp[i - 1];
    }
  }
  int maxi = 0;
  int kk = -1;
  if (len >= 4 && s[0] == 'b' && s[1] == 'b' && s[2] == 'a' && s[3] == 'a') {
    kk = 0;
  }
  for (i = ii; i <= jj; i++) {
    for (j = i; j <= jj; j++) {
      if (i != 0)
        maxi = max(maxi, dp[i - 1] + dp[len - 1] - dp[j] + b[j] - b[i - 1]);
      else
        maxi = max(maxi, dp[len - 1] - dp[j] + b[j]);
    }
  }
  cout << maxi;
  return 0;
}
