t = int(input())
for _ in range(t):
	(a, b, x, y, n) = map(int, input().split())
	nn = n
	if a - x >= n:
		r1 = (a - n) * b
	else:
		n = n - (a - x)
		if b - y >= n:
			r1 = x * (b - n)
		else:
			r1 = x * y
	n = nn
	if b - y >= n:
		r2 = (b - n) * a
	else:
		n = n - (b - y)
		if a - x >= n:
			r2 = y * (a - n)
		else:
			r2 = x * y
	z = min(r1, r2)
	print(z)
