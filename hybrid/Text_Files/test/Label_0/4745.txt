#include <iostream>
#include <cstdio>

using namespace std;

typedef long long ll;

string s;
ll ans;
ll a[500005];
ll b[500005];

int main()
{
    cin>>s;
    for(ll i = 0; i < s.size(); i++)
    {
        if(s[i] == '<') a[i + 1] = a[i] + 1;
    }
    for(ll i = s.size() - 1; i >= 0; i--)
    {
        if(s[i] == '>') b[i] = b[i + 1] + 1;
    }
    for(ll i = 0; i <= s.size(); i++)
    {
        ans += max(a[i], b[i]);
    }
    cout<<ans;
    return 0;
}
