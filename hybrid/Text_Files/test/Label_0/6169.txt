#include <bits/stdc++.h>
typedef int ll;
 
using namespace std;
 
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
 	cout.tie(0);
	ll t;
	cin >> t;
	for (ll iii = 0; iii < t; ++iii) {
		ll n;
		cin >> n;
		string s;
		cin >> s;
		unordered_map<double, ll> m;
		ll K = 0, D = 0, now = 0;
		for (auto& i : s) {
			K += i == 'K';
			D += i == 'D';
			auto a = m.find(1.0 * K/D);
			if (a == m.end()) {
				now = 1;
				m[1.0 * K/D] = 1;
			} else {
				now = ++(a->second);
			}
			cout << now << ' ';
		}
		cout << '\n';
	}
}
