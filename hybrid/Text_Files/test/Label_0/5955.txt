#include <bits/stdc++.h>
using namespace std;
int grau[1000010];
int main() {
  int N, M;
  scanf(" %d %d", &N, &M);
  while (M--) {
    int i, j;
    scanf(" %d %d", &i, &j);
    grau[i - 1]++;
    grau[j - 1]++;
  }
  long long resp = 0;
  for (int i = 0; i < N; i++) {
    long long d = grau[i];
    long long e = N - 1 - d;
    resp += d * (d - 1LL) - d * e + e * (e - 1LL);
  }
  cout << resp / 6LL << endl;
  return 0;
}
