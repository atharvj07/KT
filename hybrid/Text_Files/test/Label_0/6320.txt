#include <bits/stdc++.h>

using namespace std;

int main()
{
    int a, b;
    
    while (scanf("%d %d", &a, &b) && b){
        int r[3];
        
        int d = b - a;
        r[2] = d / 1000;
        d -= r[2] * 1000;
        r[1] = d / 500;
        d -= r[1] * 500;
        r[0] = d / 100;
        
        printf("%d %d %d\n", r[0], r[1], r[2]);
    }
    
    return (0);
}
