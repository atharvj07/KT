#define _USE_MATH_DEFINES

#include <cstdio>
#include <cstdlib>
#include <iostream>
#include <cmath>
#include <algorithm>
#include <vector>
#include <queue>
#include <map>

using namespace std;

typedef pair<long long int, long long int> P;
long long int INF = 1e18;

struct edge{
	int to;
	long long int cost;
};

#define MAX_V 110000

vector<edge> E[MAX_V];

edge func(int u, int pre, int dist){
	edge e;
	e.to = u;
	e.cost = dist;
	for(int i = 0; i < E[u].size(); i++){
		if(E[u][i].to != pre){
			edge ee = func(E[u][i].to, u, dist + E[u][i].cost);
			if(ee.cost > e.cost){
				e = ee;
			}
		}
	}
	return e;
}

int main(){
	
	int N;
	cin >> N;
	
	for(int i = 0; i < N - 1; i++){
		int u, v;
		edge p;
		cin >> u >> v >> p.cost;
		p.to = v;
		E[u].push_back(p);
		p.to = u;
		E[v].push_back(p);
	}
	
	edge e = func(0, -1, 0);
	
	e = func(e.to, -1, 0);
	
	cout << e.cost << endl;
	
	return 0;
}
