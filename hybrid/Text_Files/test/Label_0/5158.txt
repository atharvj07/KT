#include<iostream>
#include<cstring>
#include<algorithm>
#include<cmath>


using namespace std;

int a[1000];

int main(){
	//freopen("1.in","r",stdin);
	ios::sync_with_stdio(false);
	int t;cin>>t;
	while(t--){
		int n;
		cin>>n;
		if(n%2){
			for(int i=0;i<n/2;i++){
				a[i]=1;
			}
			for(int i=n/2;i<n-1;i++){
				a[i]=-1;
			}
			int k=n-1;
			while(k--){
				for(int i=0;i<=k;i++){
					cout<<a[i]<<' ';
				}
			}
			cout<<endl;
		}else {
			for(int i=0;i<(n-1)/2;i++){
				a[i]=1;
			}
			a[n/2-1]=0;
			for(int i=n/2;i<n-1;i++){
				a[i]=-1;
			}
			int k=n-1;
			while(k--){
				for(int i=0;i<=k;i++){
					cout<<a[i]<<" ";
				}
			}
			cout<<endl;
		}

	}
	
}
