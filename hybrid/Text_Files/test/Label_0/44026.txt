import math
from heapq import heappush, heappop, heapify
import random
import string
from collections import deque, defaultdict
from bisect import bisect, bisect_left, bisect_right, insort
import sys
input = sys.stdin.readline
S = lambda : input().rstrip()
I = lambda : int(S())
M = lambda : map(int, S().split())
L = lambda : list(M())
mod1 = 1000000000 + 7
mod2 = 998244353
sys.setrecursionlimit(300000)
for _ in range(I()):
	(n, k) = M()
	a = L()
	adj = [[] for i in range(n)]
	for i in range(n - 1):
		(u, v) = M()
		adj[u - 1].append(v - 1)
		adj[v - 1].append(u - 1)
	axor = 0
	for i in a:
		axor ^= i
	if axor == 0:
		print('YES')
		continue
	if k == 2:
		print('NO')
		continue

	def bfs():
		q = deque()
		q.append(0)
		visited = [0] * n
		visited[0] = 1
		parent = [-1] * n
		bfstree = []
		while q:
			root = q.popleft()
			bfstree.append(root)
			for child in adj[root]:
				if not visited[child]:
					q.append(child)
					visited[child] = 1
					parent[child] = root
		return (bfstree, parent)
	cnt = 0
	(bfstree, parent) = bfs()
	while bfstree:
		cur = bfstree.pop()
		if a[cur] == axor:
			cnt += 1
		else:
			a[parent[cur]] ^= a[cur]
	if cnt >= 2:
		print('YES')
		continue
	print('NO')
