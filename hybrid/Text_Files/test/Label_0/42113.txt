import math
t = int(input())
for _ in range(t):
	(n, x) = map(int, input().split())
	max_damage = max_value = 0
	max_damage_head = max_value_head = 0
	for i in range(n):
		count = 0
		(damage, head) = map(int, input().split())
		if max_value < damage - head:
			max_value = damage - head
			max_value_head = head
		if max_damage < damage:
			max_damage = damage
			max_damage_head = head
	if max_value == 0:
		if max_damage < x:
			print(-1)
		else:
			print(1)
	else:
		if x - max_damage >= 0:
			count = 1 + math.ceil((x - max_damage) / max_value)
		else:
			count = 1
		print(count)
