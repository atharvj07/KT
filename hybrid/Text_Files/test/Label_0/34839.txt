import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.PriorityQueue;
import java.util.StringTokenizer;

public class E_209 {
	static BufferedReader stdin = new BufferedReader(new InputStreamReader(System.in));
	static StringTokenizer st = new StringTokenizer("");
	static StringBuilder s = new StringBuilder();
	static boolean[][] t;
	static boolean[][][] v;
	static ArrayList<PriorityQueue<Integer>> rowMin = new ArrayList();
	static ArrayList<PriorityQueue<Integer>> rowMax = new ArrayList();
	static ArrayList<PriorityQueue<Integer>> colMin = new ArrayList();
	static ArrayList<PriorityQueue<Integer>> colMax = new ArrayList();
	static int c = 0;
	static char[] a;
	
	public static void main(String[] args) throws Exception {
		int n = readInt();
		int y = readInt()-1;
		int x = readInt()-1;
		
		t = new boolean[n][n];
		v = new boolean[n][n][4];
		for(int i = 0; i < n; ++i) {
			rowMin.add(new PriorityQueue());
			rowMax.add(new PriorityQueue(1, Collections.reverseOrder()));
			colMin.add(new PriorityQueue());
			colMax.add(new PriorityQueue(1, Collections.reverseOrder()));
		}
		a = new char[] {'U','R','D','L'};
		for(int i = 0; i < n; ++i)
			for(int j = 0; j < n; ++j)
				if(readInt() == 1)
					light(j, i);
		
		if(!t[x][y])
			on(x, y);
		int d = 4;
		while(true) {
			int d2 = dir(x, y, d);
			if(d2 == 4)
				break;
			if(d != d2 && !t[x][y])
				on(x, y);
			d = d2;
			int[] pos = next(x, y, d);
			v[x][y][d] = true;
			if(v[pos[0]][pos[1]][(d+2)%4] && t[x][y])
				off(x, y);
			s.append(a[d]);
			x = pos[0];
			y = pos[1];
		}
		
		off(x, y);
		if(c == 0) {
			System.out.println("YES");
			System.out.println(s);
		} else
			System.out.println("NO");
	}
	
	static void on(int x, int y) {
		light(x, y);
		s.append(1);
	}
	
	static void off(int x, int y) {
		dark(x, y);
		s.append(2);
	}
	
	static int[] next(int x, int y, int d) {
		int[] pos = {x,y};
		if(d == 0)
			pos[1]--;
		if(d == 1)
			pos[0]++;
		if(d == 2)
			pos[1]++;
		if(d == 3)
			pos[0]--;
		return pos;
	}
	
	static int dir(int x, int y, int d) {
		d = (d+1)%4;
		for(int i = 0; i < 4; ++i) {
			int[] pos = next(x, y, d);
			if(pos[0] >= 0 && pos[0] < v.length && pos[1] >= 0 && pos[1] < v[0].length && (!visited(pos[0], pos[1]) || v[pos[0]][pos[1]][(d+2)%4])) {
				if(d == 0 && !colMin.get(x).isEmpty() && colMin.get(x).peek() < y)
					return d;
				if(d == 1 && !rowMax.get(y).isEmpty() && rowMax.get(y).peek() > x)
					return d;
				if(d == 2 && !colMax.get(x).isEmpty() && colMax.get(x).peek() > y)
					return d;
				if(d == 3 && !rowMin.get(y).isEmpty() && rowMin.get(y).peek() < x)
					return d;
			}
			d = (d+3)%4;
		}
		return 4;
	}
	
	static boolean visited(int x, int y) {
		for(int d = 0; d < 4; ++d)
			if(v[x][y][d])
				return true;
		return false;
	}
	
	static void light(int x, int y) {
		t[x][y] = true;
		rowMin.get(y).add(x);
		rowMax.get(y).add(x);
		colMin.get(x).add(y);
		colMax.get(x).add(y);
		++c;
	}
	
	static void dark(int x, int y) {
		t[x][y] = false;
		while(!rowMin.get(y).isEmpty() && !t[rowMin.get(y).peek()][y])
			rowMin.get(y).poll();
		while(!rowMax.get(y).isEmpty() && !t[rowMax.get(y).peek()][y])
			rowMax.get(y).poll();
		while(!colMin.get(x).isEmpty() && !t[x][colMin.get(x).peek()])
			colMin.get(x).poll();
		while(!colMax.get(x).isEmpty() && !t[x][colMax.get(x).peek()])
			colMax.get(x).poll();
		--c;
	}

	static String readString() throws Exception {
		while(!st.hasMoreTokens())
			st = new StringTokenizer(stdin.readLine());
		return st.nextToken();
	}
	static int readInt() throws Exception {
		return Integer.parseInt(readString());
	}
}
