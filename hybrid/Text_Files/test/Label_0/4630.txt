#include<bits/stdc++.h>
#define rep(i,n)for(int i=0;i<n;i++)
using namespace std;
 
int main() {
    int n;
    while (scanf("%d", &n), n) {
        unordered_map<string, int>mp;
        string s; int Max = 0;
        int d = 0;
        for (int i = 1; i <= n; i++) {
            string c; cin >> c;
            mp[c]++;
            if (mp[c] > Max) { s = c; Max = mp[c]; }
            int sec = 0;
            for (auto&& p : mp) {
                if (p.first != s&&sec < p.second) sec = p.second;
            }
            if (sec + n - i < Max && !d)d = i;
        }
        if (!d)puts("TIE");
        else cout << s << ' ' << d << endl;
    }
}
