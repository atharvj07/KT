ll my_log(ll n, int b) {
    ll i = 1;
    int ans = 0;
    while (1) {
        if (i > n)
        {
            ans--;
            break;
        }
        if (i == n)
            break;
        i *= b;
        ans++;
    }
    return ans;
}

ll pw(ll x, ll y, ll p) { 
    ll res = 1;      
    x = x % p;  
    while (y > 0) {
        if (y & 1) 
            res = (res*x) % p; 
        y = y>>1; 
        x = (x*x) % p;   
    } 
    return res; 
}
