        char[] small = new char[3];

            test = in.next();

//                big[i] = test.charAt(2);
                small[i] = test.charAt(0);
            } else {

            }

            test = "";

        for (int j = 0; j < 3; j++) {
            if (small[j] == 'A') {
                A++;
            } else if (small[j] == 'B') {
                B++;
            } else if (small[j] == 'C') {
                C++;
            }
        }
for(int i=2 ;i >=0; i--){
        if (C == i) {
            test += "C";
        } else if (B==i) {
            test += "B";
        }else if(A == i)
test += "A";
            }
    if(test.length()== 3)
            System.out.println(test);

            System.out.println("Impossible");
