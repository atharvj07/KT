import java.io.*;
import java.util.*;

public class dynasty {
	public static void main(String[] args) throws IOException{
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		int n = Integer.parseInt(br.readLine());
		int[][] dp = new int[26][26];
		//starting, ending
		int ans = 0;
		for (int i = 0; i < n; i++) {
			String name = br.readLine();
			int lval = ((int) name.charAt(name.length()-1))-97;
			int fval = ((int) name.charAt(0))-97;
			for (int j = 0; j < 26; j++) {
				if (dp[j][fval] != 0) {
					dp[j][lval] = Integer.max(dp[j][fval]+name.length(), dp[j][lval]);
					
				} else if (fval == j) {
					dp[j][lval] = Integer.max(name.length(), dp[j][lval]);
				}
			}
		}
		int c = 0;
		for (int r = 0; r < 26; r++) {
			ans = Integer.max(dp[r][c], ans);
			c++;
		}
		System.out.println(ans);
	}
}
