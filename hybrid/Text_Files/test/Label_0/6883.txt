#include <iostream>
#include <vector>
#include <algorithm>
#include <cassert>
#include <cmath>

using ll = long long;
#define MIN(a, b) (((a) < (b)) ? (a) : (b))
#define MAX(a, b) (((a) < (b)) ? (b) : (a))

int const nmax = 50;
int a[1 + nmax], b[1 + nmax];
int dp[1 + nmax][1 + nmax];

bool solve(int n, ll cost) {
  for(int i = 0;i <= nmax; i++)
    for(int j = 0;j <= nmax; j++)
      dp[i][j] = 0;
  for(int i = 0;i <= nmax; i++) {
    dp[i][i] = 1;
    for(int j = i; 0 <= j; j--) 
      for(int h = 1; h <= nmax; h++)
        if(0 < ((1LL << h) & cost))
          dp[i][j % h] |= dp[i][j];
  }
  for(int i = 1;i <= n; i++)
    if(dp[a[i]][b[i]] == 0)
      return false;
  return true;
}

int main() {
  int n;
  std::cin >> n;
  for(int i = 1;i <= n; i++) 
    std::cin >> a[i];
  for(int i = 1;i <= n; i++)
    std::cin >> b[i];
  ll result = (1LL << (nmax + 1)) - 2;
  if(solve(n, result) == 0) {
    std::cout << -1;
    return 0;
  }
  for(ll i = (1LL << nmax); 1 < i; i /= 2) {
    if(solve(n, result - i) == 1)
      result -= i;
  }
  std::cout << result;
  return 0;
}
