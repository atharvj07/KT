#include <bits/stdc++.h>
using namespace std;
int n, m, k, q;
vector<long long> u, p;
char str[50];
long long res;
int main() {
  scanf("%d%d%d%d", &n, &m, &k, &q);
  for (int i = 0; i < q; i++) {
    long long x;
    scanf("%I64d%s", &x, &str);
    if (str[0] == 'U')
      u.push_back(x);
    else
      p.push_back(x);
  }
  sort(p.begin(), p.end());
  sort(u.begin(), u.end());
  for (int i = 0; i < min((int)u.size(), n); i++) res += u[i];
  for (int i = 0; i < min((int)p.size(), m); i++) res += p[i];
  int x = min((int)u.size(), n), y = min((int)p.size(), m), t = 0;
  while (x != u.size() || y != p.size()) {
    if (t == k) break;
    long long r = 1e18;
    if (x != u.size()) r = min(r, u[x]);
    if (y != p.size()) r = min(r, p[y]);
    res += r;
    if (x != u.size() && r == u[x])
      x++;
    else
      y++;
    t++;
  }
  printf("%d %I64d", t + min((int)u.size(), n) + min((int)p.size(), m), res);
  return 0;
}
