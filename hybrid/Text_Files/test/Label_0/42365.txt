(x1, y1) = map(int, input().split())
(x2, y2) = map(int, input().split())
(x3, y3) = map(int, input().split())
if x1 == x2 and x2 == x3 or (y1 == y2 and y2 == y3):
	print(1)
elif x1 == x2:
	if y1 < y3 and y3 < y2 or (y1 > y3 and y3 > y2):
		print(3)
	else:
		print(2)
elif x1 == x3:
	if y1 < y2 and y2 < y3 or (y1 > y2 and y2 > y3):
		print(3)
	else:
		print(2)
elif x2 == x3:
	if y2 < y1 and y1 < y3 or (y2 > y1 and y1 > y3):
		print(3)
	else:
		print(2)
elif y1 == y2:
	if x1 < x3 and x3 < x2 or (x1 > x3 and x3 > x2):
		print(3)
	else:
		print(2)
elif y1 == y3:
	if x1 < x2 and x2 < x3 or (x1 > x2 and x2 > x3):
		print(3)
	else:
		print(2)
elif y2 == y3:
	if x2 < x1 and x1 < x3 or (x2 > x1 and x1 > x3):
		print(3)
	else:
		print(2)
else:
	print(3)
