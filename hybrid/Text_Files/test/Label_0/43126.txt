class Deque:

	def __init__(self):
		self.items = []

	def isEmpty(self):
		return self.items == []

	def addFront(self, item):
		self.items.append(item)

	def addRear(self, item):
		self.items.insert(0, item)

	def removeFront(self):
		return self.items.pop()

	def removeRear(self):
		return self.items.pop(0)

	def size(self):
		return len(self.items)

def give(lst, a, b):
	s0 = Deque()
	for i in range(1, a + 1):
		s0.addFront(i)
	while s0.size() != 1:
		index = int(s0.removeRear())
		c = lst[index - 1]
		if b < c:
			lst[index - 1] = lst[index - 1] - b
			s0.addFront(index)
	a = s0.removeRear()
	return a
(a, b) = input('').split(' ')
s = input()
a = int(a)
b = int(b)
nums = []
for i in s.split(' '):
	nums.append(int(i))
print(give(nums, a, b))
