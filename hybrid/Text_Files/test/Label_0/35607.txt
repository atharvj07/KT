import java.io.*;
import java.util.*;
import java.lang.Math;

public class Solution {
    static ArrayList<Integer>[] adjList;
    static int[] dpIn, dpOut;
    static int best;
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int N = Integer.parseInt(br.readLine());
        int i;
        adjList = new ArrayList[N];
        Arrays.setAll(adjList, x -> new ArrayList<>());
        for(i = 0; i < N-1; i++) {
            String[] sd = br.readLine().split("\\s+");
            int s = Integer.parseInt(sd[0])-1;
            int d = Integer.parseInt(sd[1])-1;
            adjList[s].add(d);
            adjList[d].add(s);
        }
        dpIn = new int[N];
        dpOut = new int[N];
        best = Integer.MIN_VALUE;
        getMaxNested(0, -1);
        System.out.println(best);
    }
    
    public static void getMaxNested(int start, int par) {
        dpIn[start] = 1;
        int size = adjList[start].size()-(par == -1 ? 0 : 1);
        dpOut[start] = size;
        int path1In = 0;
        int path2In = 0;
        int path1Out = 0;
        int path2Out = 0;
        for(Integer nei: adjList[start]) {
            if(nei != par) {
                getMaxNested(nei, start);
                dpIn[start] = Math.max(dpIn[start], dpOut[nei]+1);
                dpOut[start] = Math.max(dpOut[start], dpIn[nei]+size-1);
                dpIn[start] = Math.max(dpIn[start], dpOut[start]);
                if(dpIn[nei] > path1In) {
                    path2In = path1In;
                    path1In = dpIn[nei];
                }
                else if(dpIn[nei] > path2In)
                    path2In = dpIn[nei];
                if(dpOut[nei] > path1Out) {
                    path2Out = path1Out;
                    path1Out = dpOut[nei];
                }
                else if(dpOut[nei] > path2Out)
                    path2Out = dpOut[nei];
            }
        }
        int p = (par == -1) ? 0 : 1;
        best = Math.max(best, dpIn[start]);
        best = Math.max(best, dpOut[start]);
        best = Math.max(best, Math.max(1, size-2+p)+path1Out+path2Out);
        best = Math.max(best, size-2+path1In+path2In+p);
    }
}
