using namespace std;

/* constant */

const int MAX_N = 100000;
const int MAX_k = 17;

/* typedef */

typedef long long ll;

/* global variables */

ll dp[2][MAX_N + 1];

/* subroutines */

/* main */

int main() {
  int n, k;
  scanf("%d%d", &n, &k);

  if ((n >> (k - 1)) == 0) {
    puts("0");
    return 0;
  }

  fill(dp[0] + 1, dp[0] + n + 1, 1);
  int cur = 0, nxt = 1;

  for (int i = 1; i < k; i++) {
    memset(dp[nxt], 0, sizeof(dp[nxt]));
    int minj0 = 1 << (i - 1), maxj0 = n >> (k - i);
    int maxj1 = n >> (k - i - 1);

    for (int j = minj0; j <= maxj0; j++)
      for (int x = 2; j * x <= maxj1; x++)
	dp[nxt][j * x] += dp[cur][j];

    cur ^= 1, nxt ^= 1;
  }
  //for (int i = 1; i <= n; i++) printf("%lld ", dp[cur][i]); putchar('\n');

  ll sum = 0;
  for (int i = 1; i <= n; i++) sum += dp[cur][i];
  printf("%lld\n", sum);
  return 0;
}
