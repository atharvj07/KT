from math import gcd
(n, m) = map(int, input().split())
l = []
cnt = 0
if n >= m + 2:
	print('Impossible')
else:
	for i in range(1, n):
		l.append([i, i + 1])
		cnt = cnt + 1
		if cnt == m:
			break
	for i in range(1, n + 1):
		if cnt == m:
			break
		for j in range(i + 2, n + 1):
			if gcd(i, j) == 1:
				l.append([i, j])
				cnt = cnt + 1
			if cnt == m:
				break
	if cnt < m:
		print('Impossible')
	else:
		print('Possible')
		for i in l:
			print(i[0], end=' ')
			print(i[1])
