#include <bits/stdc++.h>
using namespace std;
int N;
int main() {
  scanf("%d", &N);
  long long debutN = 10, debutQ = 1, finN = 10000000, finQ = 1;
  for (int i = 0; i < N; i++) {
    int station = (i + 1) * 3.14;
    scanf("%d", &station);
    if ((station * 10 + 10) * finQ - (i + 1) * finN < 0) {
      finN = station * 10 + 10;
      finQ = i + 1;
    }
    if ((station * 10) * debutQ - (i + 1) * debutN > 0) {
      debutN = station * 10;
      debutQ = i + 1;
    }
    if (i == N - 1) {
      int nouv1 = (((N + 1) * debutN) / 10) / debutQ;
      int nouv2 = ((N + 1) * finN - 1) / (10 * finQ);
      if (nouv1 < nouv2)
        puts("not unique");
      else
        printf("unique\n%d\n", nouv1);
    }
  }
  return 0;
}
