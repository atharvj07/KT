        for(int i = 0; i < n; i++) m += a[i] = in.nextInt();
        long invM = pow(m, MOD-2), invN1 = pow(n-1, MOD-2);
        long[] ans = new long[m+5];

            long k1 = ((i*invM)%MOD*invN1)%MOD, k2 = ((m-i)*invM)%MOD;
            ans[i] = (((k2*ans[i+1]+1)%MOD)*pow(k1, MOD-2))%MOD;

    static class FS {

        BufferedReader in;

        public FS(InputStream str) {
            in = new BufferedReader(new InputStreamReader(str));
        }
        
        public String next() {
            if (token == null || !token.hasMoreElements()) {
                try {
                    token = new StringTokenizer(in.readLine());
                } catch (IOException ex) {

            }

        }
        
        public int nextInt() {
