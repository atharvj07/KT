#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll w[100009],b[100009],mod=1e9+7;
vector<int>v[100009];
void dfs(int x,int par)
{
    b[x]=1;w[x]=1;
    for(int i=0;i<v[x].size();i++)
    {
        int y=v[x][i];
        if(y==par)continue;
        dfs(y,x);
        b[x]*=w[y];
        b[x]%=mod;
        w[x]*=((w[y]+b[y]));
        w[x]%=mod;
    }
    return;
}
int n;
int main()
{
    cin>>n;
    int x,y;
    for(int i=0;i<n-1;i++)
    {
        cin>>x>>y;
        x--;y--;
        v[x].push_back(y);
        v[y].push_back(x);
    }
    dfs(0,0);
    cout<<(w[0]+b[0])%mod;
}
