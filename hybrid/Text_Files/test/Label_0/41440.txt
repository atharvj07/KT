def gcd(A, B):
	if B == 0:
		return A
	remainder = A % B
	return gcd(B, remainder)

def verify_input(input):
	if input[0] < input[1]:
		if input[0] > 0 and input[1] <= 10 ** 18:
			if input[1] - input[0] + 1 <= 3 * 10 ** 5:
				if (input[1] - input[0]) % 2 == 1:
					return True
	return False

def solution(S):
	for i in range(0, len(S), 2):
		print(S[i], S[i + 1])
S = [int(x) for x in input().split()]
verified = verify_input(S)
if verified:
	print('YES')
	solution(list(range(S[0], S[1] + 1)))
else:
	print('NO')
