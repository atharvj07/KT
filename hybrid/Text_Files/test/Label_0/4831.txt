template <class T1, class T2>

  return os << "(" << a.first << "," << a.second << ")";
}
template <class T>
void pv(T a, T b) {
  for (T i = a; i != b; ++i) cout << (*i) << " ";
  cout << endl;
}
template <class T>
void chmin(T& a, const T& b) {
  if (a > b) a = b;
}
template <class T>
void chmax(T& a, const T& b) {
  if (a < b) a = b;
}
int nextInt() {
  int x;
  scanf("%d", &x);
  return x;
}
long long nextLong() {
  long long x;
  scanf("%lld", &x);
  return x;
}
const int MAX_N = 1005;
int AtoB[MAX_N][MAX_N];
int BtoA[MAX_N][MAX_N];
int color[MAX_N][MAX_N];
int degA[MAX_N];
int degB[MAX_N];
void dfsB(int b, int s, int t);
void dfsA(int a, int s, int t) {
  int b = AtoB[a][s];
  if (b != -1) dfsB(b, t, s);
  swap(AtoB[a][t], AtoB[a][s]);
}
void dfsB(int b, int s, int t) {
  int a = BtoA[b][s];
  if (a != -1) dfsA(a, t, s);
  swap(BtoA[b][t], BtoA[b][s]);
}
void add_edge(int a, int b, int D) {
  for (int d = 0; d < D; d++) {
    if (AtoB[a][d] == -1 && BtoA[b][d] == -1) {
      AtoB[a][d] = b;
      BtoA[b][d] = a;
      return;
    }
  }
  int s = -1, t = -1;
  for (int d = 0; d < D; d++) {
    if (BtoA[b][d] == -1) s = d;
    if (AtoB[a][d] == -1) t = d;
  }
  dfsA(a, s, t);
  AtoB[a][s] = b;
  BtoA[b][s] = a;
  return;
}
int main2() {
  memset(AtoB, (-1), sizeof(AtoB));
  memset(BtoA, (-1), sizeof(BtoA));
  memset(color, (-1), sizeof(color));
  memset(degA, (0), sizeof(degA));
  memset(degB, (0), sizeof(degB));
  int A = nextInt();
  int B = nextInt();
  int M = nextInt();
  vector<pair<int, int>> es;
  for (int m = 0, _n = (int)(M); m < _n; ++m) {
    int a = nextInt() - 1;
    int b = nextInt() - 1;
    es.emplace_back(a, b);
    degA[a]++;
    degB[b]++;
  }
  int D = 0;
  for (int a = 0, _n = (int)(A); a < _n; ++a) chmax(D, degA[a]);
  for (int b = 0, _n = (int)(B); b < _n; ++b) chmax(D, degB[b]);
  for (auto e : es) {
    add_edge(e.first, e.second, D);
  }
  for (int a = 0, _n = (int)(A); a < _n; ++a) {
    for (int d = 0, _n = (int)(D); d < _n; ++d) {
      int b = AtoB[a][d];
      if (b != -1) {
        color[a][b] = d;
      }
    }
  }
  printf("%d\n", D);
  for (int i = 0, _n = (int)(es.size()); i < _n; ++i) {
    if (i > 0) printf(" ");
    printf("%d", (1 + color[es[i].first][es[i].second]));
  }
  printf("\n");
  return 0;
}
int main() {
  main2();
  return 0;
}
