from math import ceil
t = int(input())
for z in range(t):
	n = int(input())
	c = [int(x) for x in input().split(' ')]
	a = [int(x) for x in input().split(' ')]
	b = [int(x) for x in input().split(' ')]
	ma = 0
	sum = abs(a[1] - b[1]) + 2
	for i in range(1, n):
		ma = max(ma, sum + c[i] - 1)
		if i == n - 1:
			break
		abd = abs(a[i + 1] - b[i + 1])
		toabd = c[i] - abd - 1
		if abd == 0:
			sum = 2
			continue
		sum = max(sum + toabd + 2, abd + 2)
	print(ma)
