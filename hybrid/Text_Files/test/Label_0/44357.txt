import sys

def input():
	return sys.stdin.readline().strip()

def iinput():
	return int(input())

def rinput():
	return map(int, sys.stdin.readline().strip().split())

def get_list():
	return list(map(int, sys.stdin.readline().strip().split()))
n = iinput()
l = list(map(int, input().split()))
big = 0
s = 0
c1 = 0
for i in l:
	if i == 1:
		c1 += 1
		s = max(0, s - 1)
	else:
		s += 1
		big = max(big, s)
if c1 == n:
	print(n - 1)
else:
	print(big + c1)
