a = int(input())
t = list(map(int, input().split()))
t.sort()
p = []
for k in range(a - 1):
	if k == 0:
		p.append([t[1] - t[0], t[-1] - t[0]])
	else:
		p.append([min(t[k + 1] - t[k], t[k] - t[k - 1]), max(t[a - 1] - t[k], t[k] - t[0])])
p.append([t[-1] - t[-2], t[-1] - t[0]])
for s in p:
	print(*s)
