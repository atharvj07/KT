import java.util.Scanner;

public class Solve {

    public static void main(String[] args) {
        Scanner scan=new Scanner(System.in);
        long a=scan.nextLong();
        long b=scan.nextLong();

        if(a==b){
            System.out.println(a+" "+b);
            System.exit(0);
        }

        while(true){
            if(a==0 || b==0) break;

            if(a>=2*b){
                long x=b*2;
                long tmp=a/x;
                a-=tmp*x;
            }

            else if(b>=2*a){
                long x=a*2;
                long tmp=b/x;
                b-=tmp*x;
            }

            else break;
        }

        System.out.println(a+" "+b);

        System.exit(0);
    }
}
