#include <bits/stdc++.h>
using namespace std;
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  queue<pair<long long, long long>> q;
  set<long long> visited;
  long long n, m;
  cin >> n >> m;
  vector<long long> v(n);
  for (long long i = 0; i < n; i++) {
    cin >> v[i];
    q.push({v[i], 0});
  }
  long long ans = 0;
  while (visited.size() != (n + m)) {
    pair<long long, long long> p = q.front();
    q.pop();
    long long person = p.first;
    long long dist = p.second;
    if (visited.find(person) != visited.end()) continue;
    ans += dist;
    visited.insert(person);
    q.push({person + 1, dist + 1});
    q.push({person - 1, dist + 1});
  }
  for (long long i = 0; i < n; i++) visited.erase(v[i]);
  cout << ans << "\n";
  for (auto r : visited) cout << r << " ";
}
