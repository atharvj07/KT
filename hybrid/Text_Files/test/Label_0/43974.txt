for _ in range(int(input())):
	(x0, x1, x2) = [int(__) for __ in input().split()]
	(y0, y1, y2) = [int(___) for ___ in input().split()]
	ans = 0
	if x2 >= y1:
		ans += 2 * y1
		x2 -= y1
		if x2 <= y2:
			y2 -= x2
			if x0 <= y2:
				y2 -= x0
				ans -= 2 * y2
	else:
		ans += 2 * x2
		if y2 >= x0:
			y2 -= x0
			ans -= 2 * y2
	print(ans)
