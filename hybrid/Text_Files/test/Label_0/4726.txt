vector<int> ans;


int main(void){

    int N;
    cin >> N;

    vector<vector<int>> graph(N);

    vector<int> deg(N);

    int a, b;
    REP(i, N) {
        cin >> a >> b;
        graph[a-1].push_back(b-1);
        graph[b-1].push_back(a-1);
        deg[a-1]++; deg[b-1]++;
    }

    queue<int> que;
    vector<bool> is_pushed(N, false);
    REP(i, N) {
        if (deg[i] == 1) {
            que.push(i);
            is_pushed[i] = true;
        }
    }

    while (!que.empty()) {
        int p = que.front(); que.pop();

        is_pushed[p] = true;
        for (int c : graph[p]) {
            deg[c]--;
            if (deg[c] == 1) {
                que.push(c);
                is_pushed[c] = true;
            }
        }
    }

    int Q;
    cin >> Q;

    REP(i, Q) {
        cin >> a >> b;
        if (!is_pushed[a - 1] && !is_pushed[b - 1]) {
            cout << 2 << endl;
        }
        else {
            cout << 1 << endl;
        }
    }


    return 0;
}
