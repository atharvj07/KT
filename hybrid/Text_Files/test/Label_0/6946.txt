#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, h, k;
  cin >> n >> h >> k;
  vector<long long> potatoes(n);
  for (long long i = 0; i < n; i++) {
    cin >> potatoes[i];
  }
  long long ans = 0, accum = 0, x;
  for (long long i = 0; i < n; i++) {
    accum += potatoes[i];
    if (accum <= h) {
      x = accum / k;
      ans += x;
      accum -= (k * x);
    } else {
      accum -= potatoes[i];
      i--;
      while (1) {
        accum -= k;
        ans++;
        if (accum <= 0) break;
      }
      accum = 0;
    }
  }
  if (accum > 0) {
    while (1) {
      accum -= k;
      ans++;
      if (accum <= 0) break;
    }
  }
  cout << ans << endl;
  return 0;
}
