#include <iostream>
#include <cstdio>
#include <cstring>
using namespace std;
typedef long long ll;
const int mod=1e9+7;
struct P{
	int x,y;
}a[200005];
int n,c[8005][4005],f[4005][4005];
ll ans=0;
int add(int a,int b){
	return a+b>=mod?a+b-mod:a+b;
}
int main(){
	for(int i=0;i<=8002;i++){
		c[i][0]=1;
		for(int p=1;p<=min(i,4002);p++) c[i][p]=add(c[i-1][p],c[i-1][p-1]);
	}
	scanf("%d",&n);
	for(int i=1;i<=n;i++){
		scanf("%d%d",&a[i].x,&a[i].y);
		f[2001-a[i].x][2001-a[i].y]++;
		ans-=c[a[i].x+a[i].y+a[i].x+a[i].y][a[i].x+a[i].x];
	}
	ans%=mod;
	for(int i=1;i<=4002;i++){
		for(int p=1;p<=4002;p++){
			f[i][p]=add(f[i][p],add(f[i-1][p],f[i][p-1]));
		}
	}
	for(int i=1;i<=n;i++){
		ans+=f[2001+a[i].x][2001+a[i].y];
	}
	ans%=mod;
	ans=ans*(mod+1>>1)%mod;
	printf("%lld",(ans%mod+mod)%mod);
	return 0;
}
