n = int(input())
a = list(map(int, input().split()))
steps = 0
i = 0
b = [None] * n
while i < n - 1:
	j = i
	while j < n - 1 and a[j] != a[j + 1]:
		j += 1
	span = j - i + 1
	half_span = span // 2
	for k in range(i, i + half_span):
		b[k] = a[i]
	for k in range(i + half_span, i + span):
		b[k] = a[j]
	steps = max(steps, (span - 1) // 2)
	i = j + 1
	if i == n - 1:
		b[i] = a[i]
print(steps)
print(' '.join(map(str, b)))
