from sys import stdin
input = stdin.readline
m = 10 ** 9 + 7
R = lambda : map(int, input().split())
(r, c) = R()
p = 0
a = list(R())
b = list(R())
for i in range(b[0]):
	if a[i] == 0:
		exit(print(0))
for i in range(a[0]):
	if b[i] == 0:
		exit(print(0))
for i in range(1, r + 1):
	for j in range(1, c + 1):
		if j == a[i - 1] + 1 and i <= b[j - 1]:
			exit(print(0))
		elif i == b[j - 1] + 1 and j <= a[i - 1]:
			exit(print(0))
		if j > a[i - 1] + 1 and i > b[j - 1] + 1:
			p += 1
print(2 ** p % m)
