n = int(input())
arr = list(map(int, input().split()))
trees = [0] * (1 + n)
dic = [0] * (n + 1)
ans = [0] * n

def update(t, i, v):
	while i < len(t):
		t[i] += v
		i += lowbit(i)

def lowbit(x):
	return x & -x

def sum(t, i):
	ans = 0
	while i > 0:
		ans += t[i]
		i -= lowbit(i)
	return ans

def getmid(arr, l1, flag):
	(low, high) = (1, n)
	if l1 % 2 == 0 and flag:
		midv = l1 // 2
	else:
		midv = l1 // 2 + 1
	while low <= high:
		mid = (low + high) // 2
		ret = sum(arr, mid)
		if ret >= midv:
			high = mid - 1
		else:
			low = mid + 1
	return low
for i in range(n):
	dic[arr[i]] = i + 1
for i in range(1, n + 1):
	ans[i - 1] += sum(trees, n) - sum(trees, dic[i])
	if i >= 2:
		ans[i - 1] += ans[i - 2]
	update(trees, dic[i], 1)
visited = [0] * (1 + n)
mid = 0
last = 0
for i in range(1, n + 1):
	update(visited, dic[i], 1)
	mid = getmid(visited, i, dic[i] > mid)
	tt = sum(visited, dic[i])
	minus = min(tt - 1, i - tt)
	tmp = abs(dic[i] - mid - (tt - sum(visited, mid))) - minus
	ans[i - 1] += tmp + last
	last = tmp + last
print(' '.join(map(str, ans)))
