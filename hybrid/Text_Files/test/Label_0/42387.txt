from math import acos, sqrt, degrees

class BadPoint(Exception):
	pass

def vect(point1, point2):
	return [p2 - p1 for (p1, p2) in zip(point1, point2)]

def product(vect1, vect2):
	return sum([v1 * v2 for (v1, v2) in zip(vect1, vect2)])

def angle(vect1, vect2):
	return degrees(acos(product(vect1, vect2) / (sqrt(product(vect1, vect1)) * sqrt(product(vect2, vect2)))))
n = int(input())
points = []
for _ in range(n):
	points.append(list(map(int, input().split())))
if n > 15:
	print(0)
	exit(0)
cnt = []
for i in range(n):
	vectors = []
	for j in range(n):
		if i != j:
			vectors.append(vect(points[i], points[j]))
	try:
		for ii in range(len(vectors)):
			for jj in range(len(vectors)):
				if ii != jj:
					if angle(vectors[ii], vectors[jj]) < 90:
						raise BadPoint()
	except BadPoint:
		pass
	else:
		cnt.append(i)
print(len(cnt))
print(*sorted(map(lambda x: x + 1, cnt)), sep='\n')
