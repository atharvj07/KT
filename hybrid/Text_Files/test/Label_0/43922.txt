def main():
	(n, m) = map(int, input().split())
	s = input()
	k = sum((s[i] != s[i - 1] for i in range(1, n))) + 1
	x = i = 0
	while i < n - 1:
		if s[i] != s[i + 1]:
			j = i
			while i + 2 < n and s[i] == s[i + 2]:
				i += 1
			j = i - j + 2
			x += j * (j - 1) // 2
		i += 1
	ans = k * (m * n - n) - x
	print(ans)
main()
