n = int(input())
hused = []
yused = []
for x in range(n * n):
	if x == 0:
		a = list(map(int, input().split()))
		print(1, end=' ')
		hused.append(a[0])
		yused.append(a[1])
	else:
		a = list(map(int, input().split()))
		if hused.count(a[0]) == 0 and yused.count(a[1]) == 0:
			hused.append(a[0])
			yused.append(a[1])
			print(x + 1, end=' ')
