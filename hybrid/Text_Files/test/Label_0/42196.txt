for _ in range(int(input())):
	(n, m) = map(int, input().split())
	d = 1000000
	c = [0, 0]
	s = 0
	for i in range(n):
		for (j, x) in enumerate(map(int, input().split())):
			y = abs(x)
			s += y
			d = min(d, y)
			if x < 0:
				c[(i + j) % 2] += 1
	if c[0] % 2 == c[1] % 2:
		print(s)
	else:
		print(s - d * 2)
