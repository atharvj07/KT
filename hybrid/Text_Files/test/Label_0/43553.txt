def get_bal():
	bal = [0] * n
	bal[0] = u[0]
	for i in range(1, n):
		bal[i] = bal[i - 1] + u[i]
	min_b = min(bal)
	ans = 0
	for i in range(n):
		if bal[i] == min_b:
			ans += 1
	return ans
n = int(input())
u = list(input())
for i in range(n):
	if u[i] == '(':
		u[i] = 1
	else:
		u[i] = -1
if sum(u) != 0:
	print(0)
	print(1, 1)
	exit()
ind = (-1, -1)
ans = -1
for i in range(n):
	for j in range(i, n):
		(u[i], u[j]) = (u[j], u[i])
		ans_i = get_bal()
		(u[i], u[j]) = (u[j], u[i])
		if ans_i > ans:
			ans = ans_i
			ind = (i + 1, j + 1)
print(ans)
print(ind[0], ind[1])
