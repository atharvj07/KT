import sys
input = sys.stdin.readline

def solve():
	(n, u, v) = map(int, input().split())
	a = list(map(int, input().split()))
	res = 4000000000.0
	for i in range(1, len(a)):
		if a[i] + 1 < a[i - 1] or a[i - 1] + 1 < a[i]:
			return 0
		if a[i - 1] < a[i] or a[i] < a[i - 1]:
			res = min(res, min(v, u))
		if a[i] == a[i - 1]:
			res = min(res, v + min(v, u))
	return res
t = int(input())
r = [0] * t
for z in range(t):
	r[z] = str(solve())
print('\n'.join(r))
