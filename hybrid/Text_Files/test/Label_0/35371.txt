import java.util.*;
import java.lang.*;
import java.io.*;

/* Name of the class has to be "Main" only if the class is public. */
public class Main
{
	PrintWriter out = new PrintWriter(System.out);
	BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer tok = new StringTokenizer("");
    String next() throws IOException {
        if (!tok.hasMoreTokens()) { tok = new StringTokenizer(in.readLine()); }
        return tok.nextToken();
    }
    int ni() throws IOException { return Integer.parseInt(next()); }
    long nl() throws IOException { return Long.parseLong(next()); }
    
    
    void solve() throws IOException {
        int N=ni();
        int[] P = new int[N+1];
        ArrayList<Integer>[] A =new ArrayList[N+1];
        for (int n=1;n<=N;n++) A[n]=new ArrayList();
        for (int n=2;n<=N;n++) { P[n]=ni(); A[P[n]].add(n); }
        long[] V = new long[N+1];
        for (int n=1;n<=N;n++) V[n]=nl();
        
        Queue<Integer> Q = new LinkedList<>();
        Q.add(1);
        boolean F=true;
        long ans=0;
        while (Q.size()>0) {
            int n=Q.remove();
            int p=P[n];
            if (V[n]>-1) {
                if (V[n]<V[p]) {F=false; break; }
                ans+=V[n]-V[p];
                if (A[n].size()>0) {
                    for (int x=0;x<A[n].size();x++) Q.add(A[n].get(x));
                }
            }
            else {
                if (A[n].size()>0) {
                    V[n]=Integer.MAX_VALUE;
                    for (int x=0;x<A[n].size();x++) {
                        int c=A[n].get(x);
                        Q.add(c);
                        V[n]=Math.min(V[n],V[c]);
                    }
                    if (V[n]<V[p]) {F=false; break; }
                    ans+=V[n]-V[p];
                }
            }
        }
        if (F) System.out.println(ans);
        else System.out.println("-1");
    }
    
    
    public static void main(String[] args) throws IOException {
        new Main().solve();
    }
}
