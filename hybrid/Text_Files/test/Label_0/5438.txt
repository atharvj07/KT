#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> as(n);
    long long sum = 0;
    for (int i = 0; i < n; i++) {
        int a;
        cin >> a;
        as[i] = a;
        sum += a;
    }
    vector<vector<long long>> dp(n + 1, vector<long long>(5));
    for (int i = 0; i < n; i++) {
        dp[i + 1][0] = dp[i][0] + as[i];
        dp[i + 1][1] = min(min(dp[i][0], dp[i][1]) + (as[i] ? as[i] & 1 : 2), dp[i + 1][0]);
        dp[i + 1][2] = min(min(dp[i][1], dp[i][2]) + (as[i] & 1 ? 0 : 1), dp[i + 1][1]);
        dp[i + 1][3] = min(min(dp[i][2], dp[i][3]) + (as[i] ? as[i] & 1 : 2), dp[i + 1][2]);
        dp[i + 1][4] = min(min(dp[i][3], dp[i][4]) + as[i], dp[i + 1][3]);
    }
    cout << dp[n][4] << endl;
}
