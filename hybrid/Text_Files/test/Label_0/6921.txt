#include <bits/stdc++.h>
using namespace std;
const int maxN = 3e5 + 4;
int n, x;
int prevRed;
int prevGreen;
int prevBlue;
int maxxR, maxxB;
int main() {
  long long sol = 0;
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    char ch;
    scanf("%d %c", &x, &ch);
    if (ch == 'G') {
      if (prevRed) {
        sol += x - prevRed;
        maxxR = max(maxxR, x - prevRed);
      }
      if (prevBlue) {
        sol += x - prevBlue;
        maxxB = max(maxxB, x - prevBlue);
      }
      if (prevGreen) {
        if (maxxR + maxxB > x - prevGreen) {
          sol -= maxxR;
          sol -= maxxB;
          sol += x - prevGreen;
        }
      }
      prevRed = prevBlue = prevGreen = x;
      maxxR = maxxB = 0;
    }
    if (ch == 'R') {
      if (prevRed) {
        sol += x - prevRed;
        maxxR = max(maxxR, x - prevRed);
      }
      prevRed = x;
    }
    if (ch == 'B') {
      if (prevBlue) {
        sol += x - prevBlue;
        maxxB = max(maxxB, x - prevBlue);
      }
      prevBlue = x;
    }
  }
  printf("%I64d", sol);
  return 0;
}
