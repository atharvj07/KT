import sys
input = sys.stdin.readline

def find(a, x, j, p):
	d = []
	e = []
	if j < 0:
		return x
	for i in range(len(a)):
		if a[i] & 1 << j:
			d.append(a[i])
		else:
			e.append(a[i])
	if len(e) == 0:
		x = find(d, x, j - 1, p)
	elif len(d) == 0:
		x = find(e, x, j - 1, p)
	else:
		x = min(find(e, x, j - 1, p), find(d, x, j - 1, p)) | 1 << j
	return x
n = int(input())
a = [int(x) for x in input().split()]
print(find(a, 0, len(bin(max(a))[2:]), a))
