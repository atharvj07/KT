(n, k) = map(int, input().split())
s = 0
c = []
l = 0
for i in range(n):
	(a, b) = map(int, input().split())
	if a >= b:
		s += b
	else:
		s += a
		if 2 * a >= b:
			c.append(b - a)
			l += 1
		else:
			c.append(a)
			l += 1
c = sorted(c)
c = c[::-1]
if l < k:
	k = l
for i in range(k):
	s += c[i]
print(s)
