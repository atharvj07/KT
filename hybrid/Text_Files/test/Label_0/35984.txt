import java.math.BigInteger;
import java.util.Scanner;

public class DNA {

	public static void main(String args[]) {

		Scanner st = new Scanner(System.in);
		int n = st.nextInt();
		String s = st.next();

		if (n == 1) {
			System.out.println(1);
			return;
		}

		int count[] = new int[4];

		for (int i = 0; i < n; i++)
			if (s.charAt(i) == 'A')
				count[0]++;
			else if (s.charAt(i) == 'C')
				count[1]++;
			else if (s.charAt(i) == 'G')
				count[2]++;
			else if (s.charAt(i) == 'T')
				count[3]++;

		int w = 0;

		if (count[0] == 0)
			w++;
		if (count[1] == 0)
			w++;
		if (count[2] == 0)
			w++;
		if (count[3] == 0)
			w++;

		if (w == 3) {
			System.out.println(1);
			return;
		}

		int main = Math.max(count[3],
				Math.max(count[2], Math.max(count[1], count[0])));

		int how = 0;

		if (main == count[0])
			how++;

		if (main == count[1])
			how++;

		if (main == count[2])
			how++;

		if (main == count[3])
			how++;

		BigInteger b1 = new BigInteger(how + "");
		BigInteger b2 = new BigInteger(s.length() + "");
		BigInteger b3 = new BigInteger(1000000007 + "");

		BigInteger br = b1.modPow(b2, b3);

		System.out.println(br.toString());

		st.close();
	}
}
