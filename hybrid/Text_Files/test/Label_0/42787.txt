for _ in range(int(input())):
	n = int(input())
	l = []
	cnt = 0
	(a, b, c) = map(int, input().split())
	s = input()
	R = a - s.count('S')
	P = b - s.count('R')
	S = c - s.count('P')
	for i in s:
		if i == 'R':
			if b > 0:
				l.append('P')
				b -= 1
				cnt += 1
			elif R > 0:
				l.append('R')
				R -= 1
			else:
				l.append('S')
				S -= 1
		elif i == 'P':
			if c > 0:
				l.append('S')
				c -= 1
				cnt += 1
			elif P > 0:
				l.append('P')
				P -= 1
			else:
				l.append('R')
				R -= 1
		elif i == 'S':
			if a > 0:
				l.append('R')
				a -= 1
				cnt += 1
			elif S > 0:
				l.append('S')
				S -= 1
			else:
				l.append('P')
				P -= 1
	if cnt >= n // 2 + n % 2 and len(l) == n:
		print('YES')
		print(''.join(l))
	else:
		print('NO\n')
