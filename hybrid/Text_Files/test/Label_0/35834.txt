import java.util.*;
import java.io.*;

public class E
{
	private static Node [] nodes;
	public static void main(String[] args) throws Exception
	{
		BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

		StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
		int n = Integer.parseInt(tokenizer.nextToken());
		int k = Integer.parseInt(tokenizer.nextToken());

		tokenizer = new StringTokenizer(reader.readLine());
		int [] pos = new int[n];
		for(int i = 0 ; i < n ; i++)
			pos[Integer.parseInt(tokenizer.nextToken()) - 1] = i;

		tokenizer = new StringTokenizer(reader.readLine());
		boolean [] want = new boolean[n];
		for(int i = 0 ; i < k ; i++)
			want[Integer.parseInt(tokenizer.nextToken()) - 1] = true;

		nodes = new Node[n];
		Node root = make(0, n-1, null);

		TreeSet<Integer> intervals = new TreeSet<Integer>();
		intervals.add(-1);
		intervals.add(n);

		long res = 0;

		for(int i = 0 ; i < n ; i++)
		{
			if(want[i])
			{
				intervals.add(pos[i]);
			}
			else
			{
				int left = intervals.lower(pos[i]) + 1;
				int right = intervals.higher(pos[i]) - 1;

				res += (right - left + 1);
				res -= root.get(left, right);
				nodes[pos[i]].set(-1);
			}
		}

		System.out.println(res);
	}

	private static Node make(int from, int to, Node parent)
	{
		if(from == to)
			return nodes[from] = new Node(from, to, parent);

		Node res = new Node(from, to, parent);
		res.left = make(from, (from + to)/2, res);
		res.right = make((from+to)/2+1, to, res);

		return res;
	}

	static class Node
	{
		Node left, right, parent;
		int from, to;
		int value;

		public Node(int from, int to, Node parent)
		{
			this.from = from;
			this.to = to;
			this.parent = parent;
			this.value = 0;
		}

		public void set(int value)
		{
			this.value += (value == -1 ? 1 : 0);

			if(parent != null)
				parent.set(value);
		}

		public int get(int a, int b)
		{
			if(a <= from && b >= to)
				return value;
			else if(a > to || b < from)
				return 0;
			else
			{
				int x = left.get(a, b);
				int y = right.get(a, b);

				return x + y;
			}
		}
	}
}
