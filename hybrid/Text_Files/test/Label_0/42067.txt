n = int(input())
li = list(map(int, input().split()))
if n == 1:
	print(1)
elif n == 2:
	print(2)
else:
	count = 0
	maximum = 0
	for i in range(2, n):
		if li[i] == li[i - 1] + li[i - 2]:
			count += 1
		if li[i] != li[i - 1] + li[i - 2]:
			maximum = max(count, maximum)
			count = 0
	maximum = max(count, maximum)
	print(maximum + 2)
