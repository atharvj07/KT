for _ in range(int(input())):
	dic = {}
	(n, k) = map(int, input().split())
	arr = list(map(int, input().split()))
	arr.sort()
	score = sum(arr[:n - 2 * k])
	dic = {}
	for el in arr[n - 2 * k:]:
		if el not in dic:
			dic[el] = 1
		else:
			dic[el] += 1
	keys = list(dic.keys())
	tmax = 0
	psum = 0
	for k in range(len(keys)):
		if dic[keys[k]] > tmax:
			tmax = dic[keys[k]]
		psum += dic[keys[k]]
	if 2 * tmax > psum:
		score += (2 * tmax - psum) // 2
	print(score)
