#include <bits/stdc++.h>
using namespace std;
class dop {
 public:
  long long int n, m, a, b, c, d;
  void get();
};
void dop::get() {
  cin >> n >> m;
  bool y = false;
  for (long long int i = 0; i < n; i++) {
    cin >> a >> b >> c >> d;
    if (b == c) {
      y = true;
    }
  }
  if (m % 2 != 0) {
    cout << "NO" << endl;
    return;
  }
  if (y) {
    cout << "YES" << endl;
    return;
  }
  cout << "NO" << endl;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long int t = 1;
  cin >> t;
  dop obj[t];
  for (long long int i = 0; i < t; i++) {
    obj[i].get();
  }
  return 0;
}
