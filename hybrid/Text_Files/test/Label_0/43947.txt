for i in range(int(input())):
	n = int(input())
	l = [int(c) for c in input().split()]
	l = list(reversed(l))
	negative = 0
	o = 0
	for i in range(n):
		if l[i] > 0 and negative <= 0:
			o += l[i]
		elif l[i] > 0 and negative > 0:
			if l[i] > negative:
				o += l[i] - negative
				negative = 0
			else:
				negative = negative - l[i]
		elif l[i] <= 0:
			negative += abs(l[i])
	print(o)
