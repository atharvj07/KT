t = int(input())
for i in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	pmin = 0
	pmax = 0
	for i in range(n):
		if a[pmin] < a[i]:
			pmin = i
		if a[pmax] > a[i]:
			pmax = i
	left = max(pmin, pmax)
	right = max(n - pmin - 1, n - pmax - 1)
	left_right = min(pmin + n - pmax, pmax + n - pmin)
	print(min(left, right, left_right) + 1)
