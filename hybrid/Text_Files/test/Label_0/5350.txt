#include <bits/stdc++.h>
using namespace std;
const int dr[] = {-1, 0, 1, 0, -1, -1, 1, 1};
const int dc[] = {0, -1, 0, 1, -1, 1, -1, 1};
const int N = 1123;
int T;
int n, a[N];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> T;
  while (T--) {
    cin >> n;
    if (n <= 3)
      cout << -1 << '\n';
    else {
      if (n % 2 == 1) {
        for (int i = 1; i <= n; i += 2) cout << i << ' ';
        cout << n - 3 << ' ' << n - 1;
        for (int i = n - 5; i >= 2; i -= 2) cout << ' ' << i;
        cout << '\n';
      } else {
        for (int i = 2; i <= n; i += 2) cout << i << ' ';
        cout << n - 3 << ' ' << n - 1;
        for (int i = n - 5; i >= 1; i -= 2) cout << ' ' << i;
        cout << '\n';
      }
    }
  }
  return 0;
}
