#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int arr[n][4];
  multiset<int> s[4];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 4; j++) {
      cin >> arr[i][j];
      s[j].insert(arr[i][j]);
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 4; j++) {
      auto temp = s[j].find(arr[i][j]);
      s[j].erase(temp);
    }
    if (*s[0].rbegin() <= *s[2].begin() && *s[1].rbegin() <= *s[3].begin()) {
      cout << *s[0].rbegin() << " " << *s[1].rbegin() << endl;
      break;
    }
    for (int j = 0; j < 4; j++) s[j].insert(arr[i][j]);
  }
  return 0;
}
