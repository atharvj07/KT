from sys import stdin, stdout
from collections import Counter
from itertools import permutations
import bisect
import math
I = lambda : map(int, stdin.readline().split())
I1 = lambda : stdin.readline()

def solve():
	(n, k) = I()
	c = list(I())
	s = list(set(c))
	ans = 1000005
	for x in s:
		cnt = 0
		i = 0
		while i < n:
			if c[i] != x:
				i += k
				cnt += 1
			else:
				i += 1
		ans = min(ans, cnt)
	print(ans)
T = int(I1())
for _ in range(T):
	solve()
