import java.io.*;
import java.util.ArrayDeque;

public class P543C
{
	private static int n, k, totalProgress, doneSubmissions, timer;
	private static int[] a;
	private static ArrayDeque<Submission> submissionList, submissionQueue;
	private static Process[] processList;
	
	private static class Submission
	{
		private int testNumber;
		private boolean interesting;
		
		private Submission(int testNumber)
		{
			this.testNumber = testNumber;
			this.interesting = false;
		}
	}
	
	private static class Process
	{
		private final int length;
		private final Submission submission;
		private int progress;

		private Process(Submission submission)
		{
			this.submission = submission;
			this.length = submission.testNumber;
			this.progress = 0;
		}
		
	}
	
	private static void updateProgresses()
	{
		for (int i = 0; i < k; i++)
		{
			if (processList[i] != null)
			{
				processList[i].progress++;
			}
		}
	}
	
	private static void updateProcesses()
	{
		for (int i = 0; i < k; i++)
		{
			if (processList[i] == null)
			{
				if (submissionQueue.size() != 0)
				{
					processList[i] = new Process(submissionQueue.poll());
				}
			}
			else if (processList[i].progress == processList[i].length)
			{
				doneSubmissions++;
				totalProgress = (int) (100.0 * doneSubmissions / n + 0.5);
				if (submissionQueue.size() != 0)
				{
					processList[i] = new Process(submissionQueue.poll());
				}
				else
				{
					processList[i] = null;
				}
			}
		}
	}
	
	private static void checkInteresting()
	{
		for (int i = 0; i < k; i++)
		{
			if (processList[i] != null && processList[i].progress == totalProgress)
			{
				processList[i].submission.interesting = true;
			}
		}
	}
	
	public static void main(String[] args) throws IOException, NumberFormatException
	{
		BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
		String inputLine = input.readLine();
		String[] tokens = inputLine.split(" ");
		n = Integer.parseInt(tokens[0]);
		k = Integer.parseInt(tokens[1]);
		a = new int[n];
		submissionList = new ArrayDeque<Submission>(n);
		submissionQueue = new ArrayDeque<Submission>(n);
		processList = new Process[k];
		inputLine = input.readLine();
		tokens = inputLine.split(" ");
		for (int i = 0; i < n; i++)
		{
			a[i] = Integer.parseInt(tokens[i]);
			Submission current = new Submission(a[i]);
			submissionList.offer(current);
			submissionQueue.offer(current);
		}
		doneSubmissions = 0;
		timer = 0;
		updateProcesses();
		while (true)
		{
			timer++;
			updateProgresses();
			checkInteresting();
			updateProcesses();
			if (doneSubmissions == n)
			{
				break;
			}
		}
		int count = 0;
		for (Submission s : submissionList)
		{
			if (s.interesting)
			{
				count++;
			}
		}
		System.out.println(count);
	}
}
