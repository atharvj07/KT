public class Main {
    public static void main(String[] args){

        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n  = 0;
        List<Pair> res = new ArrayList<Pair>();
        try {
               n = Integer.parseInt(reader.readLine());

            for (int i = 0; i < n * 2 - 1; ++ i){
                String line = reader.readLine();
                String[] nums = line.split(" ");
                for (int j = 0; j < nums.length; ++ j){
                    Pair p = new Pair();
                    p.p1 = i + 1;
                    p.p2 = j;
                    p.weight = Integer.parseInt(nums[j]);
                    res.add(p);
                }
            }
        } catch (IOException e) {
            e.printStackTrace();  //To change body of catch statement use File | Settings | File Templates.
        }
        int[] summary = new int[n*2];
        Collections.sort(res, new Comparator<Pair>() {
            @Override
            public int compare(Pair o1, Pair o2) {
                return o2.weight - o1.weight;
            }
        });

        boolean[] used = new boolean[n*2];
        Arrays.fill(used, false);



        for (int i = 0; i < res.size(); ++ i){
            if (!used[res.get(i).p1] && !used[res.get(i).p2]){
                used[res.get(i).p1]=true;
                used[res.get(i).p2]=true;
                summary[res.get(i).p1] = res.get(i).p2;
                summary[res.get(i).p2] = res.get(i).p1;
            }
        }
        for (int i = 0; i < summary.length; ++ i){
            if (i > 0){
                System.out.print(" ");
            }
            System.out.print(summary[i] + 1);
        }
    }

    public static class Pair{
        int p1;
        int p2;
        int weight;
    }
}
