#include <stdio.h>
#define N 100010
int n,m,a[N];
int main() {
	scanf("%d%d%d",&n,&m,a);
	int p=0;
	for(int i=1;i<n;++i) {
		scanf("%d",a+i);
		if(a[i]+a[i-1]>=m) p=i;
	}
	if(!p) puts("Impossible");
	else {
		puts("Possible");
		for(int i=1;i<p;++i) printf("%d\n",i);
		for(int i=n-1;i>=p;--i) printf("%d\n",i);
	}
	return 0;
}
