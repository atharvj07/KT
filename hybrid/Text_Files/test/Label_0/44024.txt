import sys
import io, os
input = sys.stdin.readline
from collections import defaultdict
k = int(input())
n = int(input())
toid = {'white': 0, 'yellow': 1, 'green': 2, 'blue': 3, 'red': 4, 'orange': 5}
edge = [[2, 3, 4, 5], [2, 3, 4, 5], [0, 1, 4, 5], [0, 1, 4, 5], [0, 1, 2, 3], [0, 1, 2, 3]]
mod = 10 ** 9 + 7
predefine = {}
V = []
for i in range(n):
	(v, s) = map(str, input().split())
	v = int(v)
	v -= 1
	s = toid[s]
	V.append(v)
	predefine[v] = s
X = set()
for v in V:
	X.add(v)
	while v:
		v = (v - 1) // 2
		X.add(v)
X = list(X)
X.sort(reverse=True)
S = set(X)
dp = defaultdict(lambda : [0] * 6)
for v in X:
	if v in predefine:
		c = predefine[v]
		dp[v] = [0] * 6
		(nx1, nx2) = (2 * v + 1, 2 * v + 2)
		if nx1 in S and nx2 in S:
			(s1, s2) = (0, 0)
			for j in edge[c]:
				s1 += dp[nx1][j]
				s2 += dp[nx2][j]
			dp[v][c] += s1 * s2 % mod
		elif nx1 in S:
			for j in edge[c]:
				dp[v][c] += dp[nx1][j] % mod
		elif nx2 in S:
			for j in edge[c]:
				dp[v][c] += dp[nx2][j] % mod
		else:
			dp[v][c] = 1
	else:
		for c in range(6):
			(nx1, nx2) = (2 * v + 1, 2 * v + 2)
			if nx1 in S and nx2 in S:
				(s1, s2) = (0, 0)
				for j in edge[c]:
					s1 += dp[nx1][j]
					s2 += dp[nx2][j]
				dp[v][c] += s1 * s2 % mod
			elif nx1 in S:
				for j in edge[c]:
					dp[v][c] += dp[nx1][j] % mod
			elif nx2 in S:
				for j in edge[c]:
					dp[v][c] += dp[nx2][j] % mod
			else:
				dp[v][c] = 1
cnt = pow(2, k) - 1 - len(S)
ans = sum(dp[0]) * pow(4, cnt, mod)
ans %= mod
print(ans)
