#include <bits/stdc++.h>
using namespace std;
int iin() {
  int x;
  scanf("%d", &x);
  return x;
};
int n;
int main() {
  n = iin();
  int a, b;
  int x = 0, y = 0;
  while (n--) {
    a = iin(), b = iin();
    if (a > b) x++;
    if (b > a) y++;
  }
  if (x > y) printf("Mishka");
  if (y > x) printf("Chris");
  if (x == y) printf("Friendship is magic!^^");
}
