MOD = 998244353
(n, k) = input().split()
n = int(n)
k = int(k)
suffix_array = input().split()
for i in range(n):
	suffix_array[i] = int(suffix_array[i])
position = [0] * (n + 1)
for i in range(n):
	position[suffix_array[i]] = i
position[n] = -1
count = 0
for i in range(n - 1):
	if position[suffix_array[i] + 1] > position[suffix_array[i + 1] + 1]:
		count += 1
(numerador, denominador) = (1, 1)
for i in range(n):
	numerador = numerador * (k - count + n - 1 - i) % MOD
	denominador = denominador * (i + 1) % MOD
print(numerador * pow(denominador, MOD - 2, MOD) % MOD)
