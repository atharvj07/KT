k = int(input())
s1 = input()
b = input()
a = []
for elem in s1:
	a.append(elem)
i = 0
ans = 0
while i < len(a) - 1:
	f = a[i] + a[i + 1]
	g = b[i + 1] + b[i]
	if f == g and (f == '10' or f == '01'):
		(a[i], a[i + 1]) = (a[i + 1], a[i])
		i += 1
		ans += 1
	else:
		i += 1
for i in range(len(a)):
	if a[i] != b[i]:
		ans += 1
print(ans)
