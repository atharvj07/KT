n = int(input())
a = list(map(int, input().split()))
contador = dict()
for skill in a:
	if skill not in contador:
		contador[skill] = 0
	contador[skill] += 1
max_qtd = 0
for s in contador:
	total = 0
	for i in range(6):
		if s + i in contador:
			total += contador[s + i]
	max_qtd = max(max_qtd, total)
print(max_qtd)
