#include <bits/stdc++.h>
using namespace std;
int a = 1000, b = 2011;
bool ok(int p, int q) {
  int cn = 0;
  while (p) {
    if (p % 10 != q % 10) cn++;
    p /= 10;
    q /= 10;
  }
  if (cn <= 1)
    return true;
  else
    return false;
}
int main() {
  int i, j, k, l, m, n, c;
  cin >> n;
  vector<int> ans;
  for (i = 0; i < n; i++) {
    cin >> c;
    for (j = a; j <= 2011; j++) {
      if (ok(j, c)) {
        a = j;
        ans.push_back(a);
        break;
      }
    }
  }
  if (ans.size() == n) {
    for (auto x : ans) cout << x << endl;
  } else {
    cout << "No solution" << endl;
  }
}
