(a, b, c, d) = map(int, input().split())
A = [0] * a
for i in range(len(A)):
	A[i] = [0] * b
for i in range(c):
	(per1, per2) = map(int, input().split())
	A[per1 - 1][per2 - 1] = 1
ans = 0
for i in range(a):
	for j in range(b):
		for k in range(i, a):
			for t in range(j, b):
				cou = 0
				for s in range(i, k + 1):
					for m in range(j, t + 1):
						if A[s][m] == 1:
							cou += 1
				if cou >= d:
					ans += 1
print(ans)
