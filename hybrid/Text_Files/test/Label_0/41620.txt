(n, a, b) = map(int, input().split())
t = 2
o = []
while t <= n:
	l = min(a, b)
	k = max(a, b)
	a = l
	b = k
	r = n // t
	e = n - r
	l = min(r, e)
	k = max(r, e)
	r = l
	e = k
	d = a // r
	q = b // e
	o.append(min(d, q))
	while r < e:
		r = r + 1
		e = n - r
		d = a // r
		q = b // e
		o.append(min(d, q))
	t = t + 1
print(max(o))
