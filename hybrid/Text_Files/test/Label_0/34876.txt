import java.io.*;
import java.util.*;

public class CF1167G {
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		PrintWriter pw = new PrintWriter(System.out);
		StringTokenizer st = new StringTokenizer(br.readLine());
		int n = Integer.parseInt(st.nextToken());
		int d = Integer.parseInt(st.nextToken());
		st = new StringTokenizer(br.readLine());
		int[] aa = new int[n];
		for (int i = 0; i < n; i++)
			aa[i] = Integer.parseInt(st.nextToken());
		int m = Integer.parseInt(br.readLine());
		st = new StringTokenizer(br.readLine());
		while (m-- > 0) {
			int x = Integer.parseInt(st.nextToken());
			if (x == 0 || x == aa[n - 1] + 1) {
				pw.println(Math.PI / 2);
				continue;
			}
			int lower = 0, upper = n - 1;
			while (upper - lower > 1) {
				int i = (lower + upper) / 2;
				if (aa[i] < x)
					lower = i;
				else
					upper = i;
			}
			int di = x - aa[lower] - 1;
			int dj = aa[upper] - x;
			if (di == 0 && dj == 0) {
				pw.println(Math.PI);
				continue;
			} else if (di == 0 || dj == 0) {
				pw.println(Math.PI / 2);
				continue;
			}
			double alpha = Math.atan(1.0 / Math.min(di, dj));
			int d_ = (int) (1 / Math.tan(alpha / 2) + 10);
			for (int i = lower, j = upper; i >= 0 && j < n && (di = x - aa[i] - 1) < d_ && (dj = aa[j] - x) < d_; ) {
				if (Math.abs(di - dj) <= 1) {
					double beta = Math.atan(1.0 / Math.max(di, dj)) * 2;
					if (alpha < beta)
						alpha = beta;
					break;
				}
				if (di < dj)
					i--;
				else
					j++;
			}
			pw.println(alpha);
		}
		pw.close();
	}
}
