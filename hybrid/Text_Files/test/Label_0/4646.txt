#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main()
{
  	ios::sync_with_stdio(false);
  cin.tie(NULL);
	int n;
	cin>>n;
	ll a[n+1],pref[n+1];
	pref[0]=0;
	
	for(int i=1;i<=n;++i)
	{
		cin>>a[i];
		pref[i]=a[i] + pref[i-1];
	}
	ll ans=INT_MAX;
	for(int i=0;i<=n;++i)
	{
		ans=min(ans,abs(pref[n]-2*pref[i]));
	}
	cout<<ans;
}
