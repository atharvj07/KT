(_set, h) = ([], [])

def dfs(a, b, Len, h):
	if a == b:
		return True
	Len[a] = True
	for x in h[a]:
		if not Len[x] and dfs(x, b, Len, h):
			return True
	return False
for i in range(int(input())):
	(q, x, y) = map(int, input().split())
	if q == 1:
		_set.append((x, y))
		h.append([])
		for (i, part_set) in enumerate(_set):
			if x in range(part_set[0] + 1, part_set[1]) or y in range(part_set[0] + 1, part_set[1]):
				h[-1].append(i)
			if part_set[0] in range(x + 1, y) or part_set[1] in range(x + 1, y):
				h[i].append(len(_set) - 1)
	else:
		print('YES' if dfs(x - 1, y - 1, [False] * len(_set), h) else 'NO')
