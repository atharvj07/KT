using namespace std;

using ll=long long;
using R=double;
const R eps=1e-10;
using vec=valarray<R>;
using mat=valarray<vec>;

vec gauss_jordan(mat &a){
	const int n=a.size(),m=a[0].size();
	rep(i,n){
		int pivot=i;
		rep(j,i,n) if(abs(a[j][i])>abs(a[pivot][i])) pivot=j;
		swap(a[i],a[pivot]);

		if(abs(a[i][i])<eps) return vec();

		rep(j,i+1,m) a[i][j]/=a[i][i];
		rep(j,n) if(i!=j) rep(k,i+1,m) a[j][k]-=a[j][i]*a[i][k];
	}	

	vec x(0.0,n);
	rep(i,n) x[i]=a[i][n];
	return x;
}

ll comb[21][21];

inline ll calc(int a,int ai,int b,int bi){
	ll ret=1LL;
	rep(loop,ai) ret*=a;
	rep(loop,bi) ret*=b;
	return ret;
}

int main(void){
	rep(i,21)rep(j,i+1){
		if(j==0||j==i)
			comb[i][j]=1LL;
		else
			comb[i][j]=comb[i-1][j]+comb[i-1][j-1];
	}

	int a,m,b,n;
	while(cin >> a >> m >> b >> n,a){
		const int row=m*n,col=m*n+1;
		mat matrix(vec(0.0,col+1),row+1);

		for(int d=0;d<col;++d){
			for(int k=0;k<=d;++k){
				ll coef=comb[d][k];
				coef*=calc(a,(d-k)/m,b,k/n);
				matrix[((d-k)%m)*n+k%n+1][col-1-d]+=coef;
			}
		}

		matrix[0][0]=1;
		matrix[0][m*n+1]=1;

		vec ans=gauss_jordan(matrix);
		rep(i,col) cout << (i?" ":"") << int(round(ans[i]));
		cout << endl;
	}
	return 0;
}
