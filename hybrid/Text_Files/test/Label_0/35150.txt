	void work() throws Exception{
		x = Int(); k = Int(); p = Double()/100.0;
	    for(int i = 0; i <= k; i ++ ) 
	        for(int j = x + i; j % 2 == 0; j >>= 1 ) 
	            dp[0][i] += 1.0;
	    for(int i = 0; i < k; i ++ ) 
	        for(int j = 0; j <= k; j ++ )
	        {
	            dp[i + 1][j] += dp[i][j + 1] * ( 1.0 - p );
	            if (0==(j & 1) ) 
	                dp[i + 1][j] += ( 1.0 + dp[i][j >> 1] ) * p;	            
	        }	    
	    out.printf("%.10f\n", dp[k][0]);
	}
