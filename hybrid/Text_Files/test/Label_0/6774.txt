#include <bits/stdc++.h>
using namespace std;
const int N = 1e6 + 5;
const int MOD = 1e9 + 7;
const long long int INF = 1e18;
long long int powermod(long long int a, long long int b) {
  if (b == 0) return 1;
  long long int res = powermod(a, b / 2);
  res = (res * res) % MOD;
  if (b & 1) res = (res * a) % MOD;
  return res;
}
int v[N], fac[N], invfac[N];
void pre() {
  int i;
  fac[0] = 1;
  for (i = 1; i < N; i++) fac[i] = (fac[i - 1] * 1LL * i) % MOD;
  invfac[i - 1] = powermod(fac[i - 1], MOD - 2);
  for (i -= 2; i >= 0; i--) {
    invfac[i] = (invfac[i + 1] * 1LL * (i + 1)) % MOD;
    assert((fac[i] * 1LL * invfac[i]) % MOD == 1);
  }
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  pre();
  int n, m, i, j;
  cin >> n;
  for (i = 0; i < n; i++) cin >> v[i];
  sort(v, v + n);
  long long int ans = 0, cnt = 0, newans;
  for (i = n - 2; i >= 0; i--) {
    if (v[i] == v[i + 1]) {
      cnt++;
    } else
      cnt = 0;
    long long int curlen = n - i - 1;
    newans = (ans * (curlen + 1)) % MOD;
    newans = (newans +
              ((((curlen - cnt) * fac[curlen - 1]) % MOD) * 1LL * v[i]) % MOD) %
             MOD;
    ans = newans;
  }
  cout << ans << '\n';
  return 0;
}
