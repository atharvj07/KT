import java.io.*;
import java.math.BigInteger;
import java.util.*;

public class Ultimo {
	public static void main(String[] args)
    {
		Scanner in = new Scanner(System.in);
		
		long n = Long.parseLong(in.next());
		String ris = "NO";
		
		int cas = 1234567, aut = 123456, com = 1234;
		
		if(n % cas == 0)
		{
			ris = "YES";
		}else
		{
			for(int i = 0; i <= n / cas; i ++)
			{
				long totCas = i * cas;
				if((n - totCas) % aut == 0)
				{
					ris = "YES";
					break;
				}else
				{
					for(int j = 0; j <= (n - totCas) / aut; j ++)
					{
						long totAut = j * aut;
						if((n - totCas - totAut) % com == 0)
						{
							ris = "YES";
							break;
						}
					}
				}
			}
		}
		
		System.out.println(ris);
    }
}
