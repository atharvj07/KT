from collections import Counter
n = int(input())
t = list(map(int, input().split()))
st = Counter(t)
pows = [1]
mx = max(t)
while pows[-1] < 2 * mx:
	pows.append(2 * pows[-1])
ok = 0
for i in st:
	for p in pows:
		if p > i:
			if p - i in st and (i != p - i or st[i] > 1):
				ok += st[i]
				break
print(n - ok)
