#include<bits/stdc++.h>
using namespace std;
int main(){
  int m,n,x,y,lose[21]={};
  vector<int> vec[21];
  cin >> m >> n;
  for(int i=0;i<n;i++){
    cin >> x >> y;
    lose[y]++;
    vec[x].push_back(y);
  }
  queue<int> que;
  for(int i=1;i<=m;i++){
    if(lose[i] == 0) que.push(i);
  }
  while(!que.empty()){
    int z = que.front();
    que.pop();
    cout << z << endl;
    for(int i=0;i<vec[z].size();i++){
      lose[vec[z][i]]--;
      if(lose[vec[z][i]] == 0) que.push(vec[z][i]);
    }
  }
  return (0);
}
