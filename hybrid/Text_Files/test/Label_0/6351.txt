#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main()
{
	int num;
	cin >> num;
	vector<int> p;
	for (int i = 0;i < num;i++)
	{
		int n;
		cin >> n;
		p.push_back(n);
	}
	int oper;
	cin >> oper;
	while (oper--)
	{
		int head, end;
		cin >> head >> end;
		int k;
		cin >> k;
		vector<int>::iterator pp;
		for (pp = p.begin() + head;pp != p.begin() + end;pp++)
		{
			swap(*pp, *(pp + k - head));
		}
	}
	vector<int>::iterator temp = p.begin();
	while (temp != p.end())
	{
		cout << *temp;
		if (++temp == p.end())cout << endl;
		else cout << " ";
	}
}
