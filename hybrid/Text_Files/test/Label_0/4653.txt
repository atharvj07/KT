#include <bits/stdc++.h>
using namespace std;
template <class L, class R>
ostream& operator<<(ostream& cout, pair<L, R> P) {
  return cout << '(' << P.first << ',' << P.second << ')';
}
template <class T>
ostream& operator<<(ostream& cout, vector<T> V) {
  cout << "[ ";
  for (auto v : V) cout << v << ' ';
  return cout << ']';
}
typedef struct data {
  data* bit[2];
  long long cnt = 0;
} trie;
trie* head;
void insert(long long first) {
  trie* cur = head;
  for (long long i = 40; i >= 0; i--) {
    long long b = (first >> i) & 1;
    if (!cur->bit[b]) cur->bit[b] = new trie();
    cur = cur->bit[b];
    cur->cnt++;
  }
}
long long maxxor(long long first) {
  trie* cur = head;
  long long ans = 0;
  for (long long i = 40; i >= 0; i--) {
    long long b = (first >> i) & 1;
    if (cur->bit[!b] && cur->bit[!b]->cnt > 0) {
      ans += (1LL << i);
      cur = cur->bit[!b];
    } else
      cur = cur->bit[b];
  }
  return ans;
}
long long n;
vector<long long> inp;
signed main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> n;
  inp.resize(n);
  long long txor = 0;
  for (long long i = 0; i < n; ++i) {
    cin >> inp[i];
    txor ^= inp[i];
  }
  head = new trie();
  insert(0);
  long long ans = maxxor(txor);
  long long pres = 0;
  for (long long i = n - 1; i >= 0; --i) {
    pres ^= inp[i];
    insert(pres);
    txor ^= inp[i];
    ans = max(ans, maxxor(txor));
  }
  cout << ans << "\n";
}
