import sys
input = sys.stdin.readline
from sys import stdin, stdout
from collections import defaultdict, Counter
M = 10 ** 9 + 7
import heapq
MAX = 200005
INF = 10 ** 20

def MST(n, s, adj):
	dist = [INF] * n
	visited = [0] * n
	ans = 0
	dist[s] = 0
	heap = [[0, s]]
	while heap:
		head = heapq.heappop(heap)
		u = head[1]
		if visited[u]:
			continue
		ans += head[0]
		visited[u] = 1
		for i in range(len(adj[u])):
			elem = adj[u][i]
			disv = elem[1]
			v = elem[0]
			if not visited[v] and disv < dist[v]:
				dist[v] = disv
				heapq.heappush(heap, [disv, v])
	return ans

def main():
	for _ in range(int(input())):
		(n, m, k) = [int(s) for s in input().split()]
		adj = [[] for i in range(n)]
		minwk = 10 ** 10
		for i in range(m):
			(u, v, w) = [int(s) for s in input().split()]
			adj[u - 1].append([v - 1, max(0, w - k)])
			adj[v - 1].append([u - 1, max(w - k, 0)])
			minwk = min(minwk, abs(k - w))
		w = MST(n, 0, adj)
		if w == 0:
			print(minwk)
		else:
			print(w)
main()
