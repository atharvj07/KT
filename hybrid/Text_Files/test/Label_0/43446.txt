a = [int(i) for i in input().split()]
b = [int(i) for i in input().split()]
c = [0] * a[4]
for i in range(a[4]):
	cost = a[1]
	for j in range(i, a[4]):
		if cost > c[i]:
			c[i] = cost
		cost = cost - a[2] + a[3]
d = [0] * a[4]
for i in range(a[0]):
	d[b[i] - 1] += 1
ans = 0
for i in range(a[4]):
	ans += c[i] * d[i]
print(ans)
