#include <bits/stdc++.h>
using namespace std;
const int N = 300100;
const long long MOD = 998244353;
const double EBS = 1e-7;
int dX[] = {0, 0, 1, -1};
int dY[] = {1, -1, 0, 0};
int n, col[N], red[N], blue[N], ans;
vector<int> adj[N];
void dfs1(int cur = 1, int par = 0) {
  red[cur] += (col[cur] == 1);
  blue[cur] += (col[cur] == 2);
  for (auto x : adj[cur]) {
    if (x != par) {
      dfs1(x, cur);
      red[cur] += red[x];
      blue[cur] += blue[x];
    }
  }
}
void dfs2(int cur = 1, int par = 0) {
  for (auto x : adj[cur]) {
    if (x != par) {
      if (red[x] == 0 || blue[x] == 0) {
        if (red[1] - red[x] == 0 || blue[1] - blue[x] == 0) ans++;
      }
      dfs2(x, cur);
    }
  }
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; ++i) scanf("%d", &col[i]);
  for (int i = 0; i < n - 1; ++i) {
    int u, v;
    scanf("%d", &u);
    scanf("%d", &v);
    adj[v].push_back(u);
    adj[u].push_back(v);
  }
  dfs1();
  dfs2();
  printf("%d\n", ans);
  return 0;
}
