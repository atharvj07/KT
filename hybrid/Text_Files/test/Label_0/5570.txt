#include <bits/stdc++.h>
using namespace std;
vector<int> insertVector(int n) {
  vector<int> x;
  for (int i = 0; i < n; i++) {
    int y;
    cin >> y;
    x.push_back(y);
  }
  return x;
}
void displayVector(vector<int> x) {
  for (auto i : x) cout << i << " ";
  cout << "\n";
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n;
  cin >> n;
  int x = n % 4;
  if (x == 1)
    cout << 0 << " "
         << "A";
  else if (x == 2) {
    cout << 1 << " "
         << "B";
  } else if (x == 3) {
    cout << 2 << " "
         << "A";
  } else
    cout << 1 << " "
         << "A";
}
