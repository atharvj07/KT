#include <bits/stdc++.h>
using namespace std;
const long long mod = 1000000007;
const long long INF = (long long)1000000007 * 1000000007;
const long double eps = 1e-8;
const long double pi = acos(-1.0);
int n, m;
int a[400];
int need[400][400][400];
int mem[401];
void init() {
  for (int i = 0; i < 400; i++) {
    for (int j = 0; j < 400; j++) {
      for (int k = 0; k < 400; k++) {
        need[i][j][k] = mod;
      }
    }
  }
}
int main() {
  cin.tie(0);
  ios::sync_with_stdio(false);
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  init();
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 401; j++) {
      mem[j] = i + j;
    }
    need[i][i][0] = 0;
    for (int j = i + 1; j < n; j++) {
      for (int k = j - i; k >= 1; k--) {
        while (mem[k - 1] + 1 < j &&
               need[i][mem[k - 1] + 1][k - 1] < a[j] - a[mem[k - 1] + 1]) {
          mem[k - 1]++;
        }
        int id = mem[k - 1];
        need[i][j][k] =
            min(need[i][j][k], max(need[i][id][k - 1], a[j] - a[id]));
        if (id + 1 < j) {
          need[i][j][k] =
              min(need[i][j][k], max(need[i][id + 1][k - 1], a[j] - a[id + 1]));
        }
      }
      need[i][j][0] = a[j] - a[i];
    }
  }
  long long ans = 0;
  for (int i = 0; i < m; i++) {
    int l, r, c, d;
    cin >> l >> r >> c >> d;
    d = min(d, r - l);
    l--;
    r--;
    ans = max(ans, (long long)c * need[l][r][d]);
  }
  cout << ans << endl;
  return 0;
}
