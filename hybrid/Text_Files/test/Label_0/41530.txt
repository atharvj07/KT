import sys
import math
import collections

def set_debug(debug_mode=False):
	if debug_mode:
		fin = open('input.txt', 'r')
		sys.stdin = fin

def int_split_input():
	return list(map(int, input().split()))
t = int(input())
for ti in range(1, t + 1):
	(m, n) = int_split_input()
	(m_one, n_one) = (0, set())
	for i in range(m):
		cur = int_split_input()
		if cur.count(1) > 0:
			m_one += 1
		for i in range(len(cur)):
			if cur[i] == 1:
				n_one.add(i)
	c = min(m - m_one, n - len(n_one))
	if c % 2 == 1:
		print('Ashish')
	else:
		print('Vivek')
