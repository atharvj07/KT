size = int(input())
ss = input()
s = list()
for i in range(size):
	s.append(ss[i])
ones = twos = zeros = 0
zeroindx = list()
oneindex = list()
twoindex = list()
for i in range(size):
	if s[i] == '0':
		zeros = zeros + 1
		zeroindx.append(i)
	elif s[i] == '1':
		ones = ones + 1
		oneindex.append(i)
	else:
		twos = twos + 1
		twoindex.append(i)
div = int(size / 3)
a = len(zeroindx)
b = len(oneindex)
c = 0
if twos < div:
	while ones > div and twos < div:
		s[oneindex[b - 1]] = '2'
		b = b - 1
		twos = twos + 1
		ones = ones - 1
	while zeros > div and twos < div:
		s[zeroindx[a - 1]] = '2'
		a = a - 1
		twos = twos + 1
		zeros = zeros - 1
if twos > div:
	while zeros < div and twos > div:
		s[twoindex[c]] = '0'
		c = c + 1
		twos = twos - 1
		zeros = zeros + 1
	while ones < div and twos > div:
		s[twoindex[c]] = '1'
		c = c + 1
		twos = twos - 1
		ones = ones + 1
io = 0
while zeros < div:
	s[oneindex[io]] = '0'
	io = io + 1
	zeros = zeros + 1
while ones < div:
	s[zeroindx[a - 1]] = '1'
	a = a - 1
	ones = ones + 1
for i in s:
	print(i, end='')
