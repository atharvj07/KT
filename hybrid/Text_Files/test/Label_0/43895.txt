(n, m) = map(int, input().split())
ar = list(map(int, input().split()))
j = 0
while j != n:
	t = count = 0
	for i in range(j + 1):
		t += ar[i]
	if t < m:
		print(0, end=' ')
	else:
		ar1 = sorted(ar[0:j])
		sm = sum(ar1)
		while sm + ar[j] > m:
			sm -= ar1[-1]
			ar1.pop(-1)
			count += 1
		print(count, end=' ')
	j += 1
