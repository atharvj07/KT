mod = 10 ** 9 + 7

def modExp(x, y):
	if y == 0:
		return 1
	if y % 2 == 0:
		return modExp(x * x % mod, y // 2)
	return x * modExp(x * x % mod, (y - 1) // 2) % mod
t = int(input())
while t:
	t -= 1
	(n, k) = [int(x) for x in input().split()]
	if k == 0:
		print(1)
		continue
	res = 0
	z = modExp(2, n - 1)
	y = z * 2
	p = 1
	if n % 2 == 0:
		for i in range(k):
			x = 1
			x = x * modExp(y, k - i - 1) % mod
			x = x * p % mod
			res = (res + x) % mod
			p = p * (z - 1) % mod
		res = (res + p) % mod
	else:
		res = modExp(z + 1, k)
	print(res)
