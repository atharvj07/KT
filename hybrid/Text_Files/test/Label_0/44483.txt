a = int(input())
from heapq import *
for i in range(a):
	(n, w) = map(int, input().split())
	z = list(map(int, input().split()))
	z.sort(reverse=True)
	space = w
	rows = 1
	ans = []
	for i in range(len(z)):
		if space < z[i]:
			if len(ans) > 0:
				if abs(ans[0]) >= z[i]:
					t = abs(heappop(ans))
					t -= z[i]
					heappush(ans, -t)
					continue
			rows += 1
			heappush(ans, -space)
			space = w
		space -= z[i]
	print(rows)
