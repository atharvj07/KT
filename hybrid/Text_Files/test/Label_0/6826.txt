#include<iostream>
using namespace std;
int main()
{

    int t;
    cin>>t;
    while(t--)
    {
        int a,b;
        cin>>a>>b;
        int ans=10000000;
        for(int i=0;i<=35;i++)
        {
            int num=0;
            int sum=a;
            if(b+i==1)
               continue;
            num=i;
            while(sum)
            {
                sum/=(b+i);
                num++;
            }
            ans=min(ans,num);
        }
        cout<<ans<<endl;
    }
    return 0;
}
