(n, m) = map(int, input().strip().split())
bonus = set()
x = int(input())
for _ in range(x):
	(a, b) = map(int, input().strip().split())
	bonus.add((a, b) if n < m else (b, a))
if n > m:
	(n, m) = (m, n)
weapon_on_armor = [-1] * (n + 1)
weapon_on_armor[1] = 1
answer = 0
while weapon_on_armor[n] != m:
	weapon_on_armor_potential = weapon_on_armor.copy()
	for armor in range(1, n + 1):
		if weapon_on_armor[armor] == -1:
			continue
		current_power = armor + weapon_on_armor[armor] + ((armor, weapon_on_armor[armor]) in bonus)
		new_weapon = min(current_power, m)
		new_armor = min(current_power, n)
		weapon_on_armor_potential[armor] = max(weapon_on_armor_potential[armor], new_weapon)
		weapon_on_armor_potential[new_armor] = max(weapon_on_armor_potential[new_armor], weapon_on_armor[armor])
	answer += 1
	weapon_on_armor = weapon_on_armor_potential
print(answer)
