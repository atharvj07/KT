#include <bits/stdc++.h>
using namespace std;
int main() {
  int x;
  scanf("%d", &x);
  if (x & 1)
    printf("%d", (x - 1) >> 1);
  else {
    int limit;
    for (int i = 1; i <= 31; i++)
      if (1 << i > x) {
        limit = 1 << (i - 1);
        break;
      }
    printf("%d", (x - limit) >> 1);
  }
}
