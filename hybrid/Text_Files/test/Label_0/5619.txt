#include <bits/stdc++.h>
using namespace std;
int n, k;
struct dp {
  int v[1005];
  dp() {
    for (int i = (0); i < (k + 1); i++) v[i] = 0;
  }
  void add(pair<int, int> l) {
    int x = l.first, w = l.second;
    for (int i = k; i >= w; i--) v[i] = max(v[i], v[i - w] + x);
  }
};
int del[50005];
pair<int, int> ins[50005];
pair<int, int> pt[50005];
int t[50005];
int ind[50005];
int r[50005];
struct rt {
  vector<pair<int, int> > v[50000 * 4];
  void u(int i, int rs, int re, int qs, int qe, pair<int, int> p) {
    if (rs == qs && re == qe)
      v[i].push_back(p);
    else {
      int md = (rs + re) / 2;
      if (qs < md) u(2 * i + 1, rs, md, qs, min(qe, md), p);
      if (qe > md) u(2 * i + 2, md, re, max(qs, md), qe, p);
    }
  }
  void g(int i, int rs, int re, dp f) {
    for (auto p : v[i]) f.add(p);
    if (rs + 1 == re) {
      long long ans = 0;
      long long cp = 1;
      for (int j = (1); j < (k + 1); j++) {
        ans += cp * f.v[j];
        cp = (cp * 10000019ll) % 1000000007ll;
        ans %= 1000000007ll;
      }
      r[rs] = ans;
    } else {
      int md = (rs + re) / 2;
      g(2 * i + 1, rs, md, f);
      g(2 * i + 2, md, re, f);
    }
  }
};
rt a;
int main() {
  scanf("%d%d", &n, &k);
  for (int i = (0); i < (n); i++) {
    scanf("%d%d", &ins[i].first, &ins[i].second);
  }
  int q;
  scanf("%d", &q);
  for (int i = (0); i < (50005); i++) del[i] = q;
  int cind = n;
  for (int i = (0); i < (q); i++) {
    scanf("%d", &t[i]);
    if (t[i] == 1) {
      scanf("%d%d", &pt[i].first, &pt[i].second);
      ind[i] = ++cind;
    } else if (t[i] == 2) {
      int x;
      scanf("%d", &x);
      del[x] = i + 1;
    }
  }
  for (int i = (0); i < (n); i++) {
    a.u(0, 0, q, 0, del[i + 1], ins[i]);
  }
  for (int i = (0); i < (q); i++)
    if (t[i] == 1) {
      a.u(0, 0, q, i, del[ind[i]], pt[i]);
    }
  a.g(0, 0, q, dp());
  for (int i = (0); i < (q); i++)
    if (t[i] == 3) printf("%d\n", r[i]);
}
