import math
import string

def korea(m):
	return lambda x: x[m]

def convert_to_numbers(word, hash_function):
	l = []
	for i in range(len(word)):
		if i % 2 == 0:
			l.append(hash_function[word[i]])
		else:
			l.append(-hash_function[word[i]])
	return tuple(l)

def main_function():
	(n, m) = [int(i) for i in input().split(' ')]
	alphabet = list(string.ascii_uppercase)
	hash_function = {}
	for i in range(len(alphabet)):
		hash_function[alphabet[i]] = i
	hash_function_for_position = {}
	data = []
	for i in range(n):
		data.append(input())
	data_tuple = []
	for i in data:
		data_tuple.append(convert_to_numbers(i, hash_function))
	for i in range(len(data)):
		hash_function_for_position[data[i]] = i + 1
	data_tuple_sorted = sorted(data, key=lambda x: convert_to_numbers(x, hash_function))
	for i in range(len(data_tuple_sorted)):
		if i == len(data_tuple_sorted) - 1:
			print(hash_function_for_position[data_tuple_sorted[i]])
		else:
			print(hash_function_for_position[data_tuple_sorted[i]], end=' ')
main_function()
