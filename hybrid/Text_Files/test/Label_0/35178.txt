class FastScanner {

    BufferedReader br;
    StringTokenizer st;

    FastScanner(InputStreamReader reader) {
        br = new BufferedReader(reader);
        st = new StringTokenizer("");
    }

    String next() {
        while (!st.hasMoreElements()) {
            try {
                st = new StringTokenizer(br.readLine());
            } catch (IOException ex) {
                Task.MLE();
            }
        }
        return st.nextToken();
    }

    int nextInt() {
        return Integer.parseInt(next());
    }

    long nextLong() {
        return Long.parseLong(next());
    }
    double nextDouble() {
        return Double.parseDouble(next());
    }
}

class Solver {
    void aser(boolean OK) { 
        if(!OK)
            Task.MLE();
    }

    FastScanner sc;
    PrintWriter out;
    static PrintStream err = System.err;

    final long mod = (long)1e9 + 7;
    long[] pow2 = new long[(int)1e6]; //111111111111111111111111111
    long[] pow = new long[(int)1e6]; //111111111111111111111111111
    {
        pow2[0] = 1;
        for (int i = 1; i < pow2.length; i++) {
            pow2[i] = 2011 * pow2[i-1];
            //pow2[i] = (2L * pow2[i-1]) % 1000000007L;
            //pow2[i] = pow2[i-1] * 1000000007L;
            //pow2[i] = pow2[i-1] * 2010; //!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
        }
        pow[0] = 1;
        for (int i = 1; i < pow.length; i++) 
            pow[i] = 2 * pow[i-1] % mod;
    }
    int trMax = pow2.length - 1;
    
    class Rib{
        int to, pow2;

        public Rib(int to, int pow2) {
            this.to = to;
            this.pow2 = pow2;
        }
        
    }
    
    class Node implements Cloneable{
        boolean have0, paint0;
        long hash;
        Node l, r;

        Node myClone(){
            try {
                return (Node) clone();
            } catch (CloneNotSupportedException ex) {
                Task.MLE();
                return null;
            }
        }
        
    }
    
    void upd( Node t ){
        if( t == null ) return;
        t.hash   = t.l.hash   +  t.r.hash;
        t.have0  = t.l.have0  || t.r.have0;
        t.paint0 = t.l.paint0 && t.r.paint0;
    }
    
    int getPos0( Node t, int pos ){ return getPos0(t, pos, 0,trMax); }
    
    /// [pos .. trMax]
    int getPos0( Node t, int pos, int tl, int tr ){
        if( tr < pos || !t.have0 ) return -1;
        if( t.paint0 ) 
            return max(pos,tl);      
        
        if( tl == tr ){
            return tl;
        }
        else{
            int tm = (tl + tr) / 2;
            int ans = getPos0(t.l, pos, tl  , tm);
            if( ans == -1 )
                ans = getPos0(t.r, pos, tm+1, tr);
            return ans;
            
        }
    }
    
    Node set1( Node t, int pos ){ return set1(t,pos, 0,trMax); }
     
    Node set1( Node t, int pos, int tl, int tr ){
        t = t.myClone();
        if( tl == tr ){
            aser( tl == pos );
            t.have0 = t.paint0 = false;
            t.hash = pow2[pos];
        }
        else{
            if (t.paint0) {
                t.l = t.l.myClone();
                t.r = t.r.myClone();
                setZero(t.l);
                setZero(t.r);
            }
            int tm = (tl + tr)/2;
            if( pos <= tm ) t.l = set1( t.l, pos, tl  , tm );
            else            t.r = set1( t.r, pos, tm+1, tr );
            upd(t);
        }
        return t;
    }
    
    void setZero( Node t ){
        t.hash = 0;
        t.have0 = t.paint0 = true;
    }
    
    Node paint( Node t, int l, int r ){ return paint(t, l, r, 0, trMax);}
    
    Node paint( Node t, int l, int r, int tl, int tr ){
        if( l > r || t.paint0 ) return t;

        t = t.myClone();
        if( tl == l && tr == r ){
            setZero(t);
        }
        else{
            int tm = (tl + tr) / 2;
            t.l = paint(t.l,           l, min(r,tm), tl  , tm );
            t.r = paint(t.r, max(l,tm+1),         r, tm+1, tr );
            upd(t);
        }
        return t;
    }
    
    Node addPow2( Node t, int pow2 ){
        int pos = getPos0(t, pow2);
        aser( pos != -1 );
//        err.println( "addPow2: " + pos );
        if( pow2 <= pos-1 ) t = paint(t, pow2, pos-1);
        t = set1(t, pos);
        return t;
    }
    
    int cmp( Node a, Node b ){
        if( a == null && b==null ) return 0;
        aser( a != null && b!=null );
        long hashA = a.hash, hashB = b.hash;
        if( a.paint0 || b.paint0 || hashA == hashB )
            return Long.compare( abs(hashA), abs(hashB) );
        int res = cmp( a.r, b.r );
        if( res != 0 ) return res;
        return cmp( a.l, b.l );
    }
    
    Node build(){ return build(0,trMax); }
    
    Node build( int tl, int tr ){
        Node t = new Node();
        setZero(t);
        if( tl != tr ){
            int tm = (tl + tr) / 2;
            t.l = build( tl  , tm );
            t.r = build( tm+1, tr );
        }
        return t;
    }
    
    long getHash( Node t ){ return getHash(t,0,trMax); }
    
    long getHash( Node t, int tl, int tr ){
        if( t.paint0 )
            return 0;
        if( tl == tr ){
            return pow[tl];
        }
        int tm = (tl + tr) / 2;
        return (getHash(t.l, tl, tm) + getHash(t.r, tm+1, tr)) % mod;
    }
    
    int n, m, S, E;
    ArrayList<Rib>[] gr;
    Node[] dist;
    TreeSet<Integer> ts;
    int[] parent;
    
    void deb(){
        Node t = build();
        for (int i = 0; i < 108; i++) {
            t = addPow2(t, 0);
            err.println( t.hash );
        }
        
        Task.exit();
    }
    
    void solve(){
//        deb();
        
        n = sc.nextInt();
        m = sc.nextInt();
        gr = new ArrayList[n+1];
        for (int v = 1; v <= n; v++)
            gr[v] = new ArrayList<>();
        for (int iter = 0; iter < m; iter++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int p = sc.nextInt();
            gr[a].add( new Rib(b, p) );
            gr[b].add( new Rib(a, p) );
        }
        S = sc.nextInt();
        E = sc.nextInt();
        
        dist = new Node[n+1];
        dist[S] = build();
        ts = new TreeSet<>( (a,b) -> {
            int res = cmp( dist[a], dist[b] );
            if( res != 0 ) return res;
            return a - b;
        } );
        ts.add(S);
        parent = new int[n+1];
        fill( parent, -1 );
        while( !ts.isEmpty() ){
            int v = ts.pollFirst();
            for( Rib r : gr[v] ){
                Node toDist = addPow2( dist[v], r.pow2 );
                if( dist[r.to]==null ){
                    dist[r.to] = toDist;
                    ts.add(r.to);
                    parent[r.to] = v;
                }
                else if( cmp(dist[r.to],toDist) == 1 ){
                    ts.remove(r.to);
                    dist[r.to] = toDist;
                    ts.add(r.to);
                    parent[r.to] = v;
                }
            }
            /*/{
                for( int i = 1; i <= n; ++i ){
                    if( dist[i] == null )
                        err.println( i +") inf"  );
                    else
                        err.println( i +") "+ dist[i].hash );
                }
                err.println();
            }/**/
        }
        
        if( dist[E] == null ){
            out.println( "-1" );
        }
        else{
            ArrayList<Integer> path = new ArrayList<>();
            for(  int v = E; ; ){
                path.add( v );
                v = parent[v];
                if( v == -1 ) break;
            }
            Collections.reverse(path);
            
//            out.println(dist[E].hash);
            out.println(getHash(dist[E]));
            out.println(path.size());
            for (Integer v : path)
                out.print( " " + v );
            out.println();
        }
    }
