#include <bits/stdc++.h>
using namespace std;
typedef long long int ll;
typedef long double ld;
#define vc vector
#define mapll map<ll,ll>
#define ff first
#define ss second
#define pb push_back
#define mp make_pair
#define loopf(i,a,b) for(ll i=a;i<b;i++)
#define loopb(i,a,b) for(ll i=a;i>b;i--)

int main() {
cin.tie(NULL);
cout.tie(NULL);
ios_base::sync_with_stdio(false);    

    
ll t;
cin>>t;
while(t--){
    ll a,b;
    cin>>a>>b;
    string s;
    cin>>s;
    ll n=s.size();
    int flag=1;
    loopf(i,0,n){
        if(s[i]=='0'){
            if(s[n-1-i]=='1')flag=0;
            s[n-1-i]='0';
        }
        else if(s[i]=='1'){
            if(s[n-1-i]=='0')flag=0;
            s[n-1-i]='1';
        }
    }
    loopf(i,0,n){
        if(s[i]=='0')a--;
        else if(s[i]=='1')b--;
    }
    //cout<<flag<<a<<b<<endl;
    if(flag==0 || a<0 || b<0){
        cout<<"-1\n";
    }
    else{
        if(n%2==1){
            if(s[(n)/2]=='?'){
                //cout<<".";
                if((a-b)%2==0)flag=0;
                else{
                    if(a%2==1){
                        s[(n)/2]='0';
                        a--;
                    }
                    else{
                        s[(n)/2]='1';
                        b--;
                    }
                }
            }
            else{
                if(a%2==1)flag=0;
                else if(b%2==1)flag=0;
            }
        }
        //cout<<flag<<a<<b<<endl;
        loopf(i,0,n/2){
            if(s[i]=='?'){
                if(a>0){
                    s[i]='0';
                    s[n-1-i]='0';
                    a-=2;
                }
                else{
                    s[i]='1';
                    s[n-1-i]='1';
                    b-=2;
                }
            }
        }
        //cout<<flag<<a<<b<<endl;
        if(flag==0 || a!=0 || b!=0)cout<<"-1\n";
        else cout<<s<<endl;
    }
}
return 0;
}
