static boolean canWin(int start, int len) {
	if(start + len > n)return false;
	if(str[start] == '1') {
		return n0[start] - start + 1 > len;
	}else if(str[start] == '0') {
		return n1[start] - start + 1 > len;
	}else {
		return (n0[start] - start + 1 > len) || (n1[start] - start + 1 > len);
	}
}
static int go(int start, int len) {
	if(start >= n)return n;
	if(canWin(start, len))return start;
	int pos = nq[start];
	return nq[start] = go(pos, len);
}
