#include <bits/stdc++.h>
using namespace std;
int isPrime(int n) {
  for (int i = 2; i * i <= n; i++) {
    if (n % i == 0) return 0;
  }
  return 1;
}
int main() {
  int n, m;
  cin >> n >> m;
  int a[50][50];
  int flag = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> a[i][j];
      if (((i == 0 || i == (n - 1)) || (j == 0 || j == (m - 1))) &&
          a[i][j] == 1)
        flag = 1;
    }
  }
  if (flag == 0)
    cout << 4;
  else
    cout << 2;
  return 0;
}
