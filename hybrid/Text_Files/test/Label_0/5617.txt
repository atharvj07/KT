#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
long long a[maxn], n, m;
long long f(long long x) {
  long long sum = 0;
  for (int i = 1; i <= n; i++) {
    sum += max(1ll * 0, a[i] - (i - 1) / x);
  }
  return sum;
}
bool cmp(long long a, long long b) { return a > b; }
int main() {
  cin >> n >> m;
  long long sum = 0;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    sum += a[i];
  }
  if (sum < m) {
    cout << "-1\n";
    return 0;
  }
  sort(a + 1, a + n + 1, cmp);
  long long l = 1, r = m, mid;
  while (l < r) {
    mid = (l + r) / 2;
    if (f(mid) >= m)
      r = mid;
    else
      l = mid + 1;
  }
  cout << l << endl;
  return 0;
}
