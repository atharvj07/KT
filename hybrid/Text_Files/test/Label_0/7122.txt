#include <bits/stdc++.h>
using namespace std;
long long p;
const int maxn = 409;
long long fact[maxn];
long long alpha[maxn][maxn];
long long a[maxn];
long long b[maxn];
long long mod(long long& x) {
  x %= p;
  if (x < 0) x += p;
  return x;
}
int main() {
  cin.sync_with_stdio(0);
  cin.tie(0);
  cin.exceptions(cin.failbit);
  int t;
  cin >> t >> p;
  fact[0] = 1;
  for (int i = 1; i < (maxn); ++i) {
    fact[i] = fact[i - 1] * i;
    mod(fact[i]);
  }
  alpha[0][0] = 1;
  for (int i = 1; i < (maxn); ++i) {
    for (int j = 1; j < (maxn); ++j) {
      if (j > i) {
        continue;
      }
      for (int k = 1; k < (i + 1); ++k) {
        alpha[i][j] += fact[k] * alpha[i - k][j - 1];
        mod(alpha[i][j]);
      }
    }
  }
  a[0] = 0;
  b[0] = 0;
  a[1] = 1;
  b[1] = 1;
  for (int n = 2; n < (maxn); ++n) {
    b[n] = fact[n];
    for (int k = 1; k < (n); ++k) {
      b[n] -= b[k] * fact[n - k];
      mod(b[n]);
    }
    a[n] = 2 * b[n] - fact[n];
    for (int k = 4; k < (n); ++k) {
      a[n] -= alpha[n][k] * a[k];
      mod(a[n]);
    }
  }
  a[2] = 2;
  for (int i = 0; i < (t); ++i) {
    int nn;
    cin >> nn;
    cout << a[nn] << endl;
  }
}
