#include <bits/stdc++.h>
using namespace std;
const int INF = 0x3f3f3f3f;
const long long LINF = 0x3f3f3f3f3f3f3f3fll;
int dist[78499], par[78499], sieve[1000010];
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int n;
  cin >> n;
  vector<int> a(n);
  vector<int> primes;
  primes.push_back(1);
  for (long long i = 2; i <= 1e6; i++)
    if (sieve[i] == 0) {
      primes.push_back(i);
      sieve[i] = i;
      for (long long j = i * i; j <= 1e6; j += i) sieve[j] = i;
    }
  map<int, int> primeid;
  for (int i = 0; i < primes.size(); i++) primeid[primes[i]] = i;
  for (int& ai : a) {
    cin >> ai;
    map<int, int> factors;
    while (ai != 1) {
      factors[sieve[ai]]++;
      ai /= sieve[ai];
    }
    int left = 1;
    for (auto [b, times] : factors)
      if (times % 2 == 1) left *= b;
    ai = left;
  }
  map<int, int> test;
  for (auto u : a) test[u]++;
  int ans = INF;
  for (auto [u, times] : test) {
    if (u == 1)
      ans = 1;
    else if (times > 1)
      ans = min(ans, 2);
  }
  if (ans < INF) {
    cout << ans << '\n';
    exit(0);
  }
  vector<vector<int>> g(primes.size());
  for (int i = 0; i < n; i++) {
    int ida = primeid[sieve[a[i]]], idb = primeid[a[i] / sieve[a[i]]];
    g[ida].push_back(idb);
    g[idb].push_back(ida);
  }
  auto girth = [&](int x) {
    memset(dist, -1, sizeof dist);
    memset(par, -1, sizeof par);
    queue<int> q;
    q.push(x);
    dist[x] = 0;
    while (q.size()) {
      int ai = q.front();
      q.pop();
      for (auto b : g[ai])
        if (par[ai] != b) {
          if (dist[b] == -1) {
            dist[b] = dist[ai] + 1;
            q.push(b);
            par[b] = ai;
          } else
            return dist[ai] + dist[b] + 1;
        }
    }
    return INF;
  };
  for (int i = 0; i < primes.size(); i++)
    if (primes[i] <= 1000) {
      ans = min(ans, girth(i));
    }
  cout << (ans == INF ? -1 : ans) << '\n';
  return 0;
}
