#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

long long K, dp[555]; string S;

int main() {
	cin >> K;
	for (int i = 0; i <= 550; i++) {
		unsigned long long R = 1;
		for (int j = i + 1; j <= i + 7; j++) R *= j;
		R /= 5040;
		dp[i] = R;
	}
	for (int i = 549; i >= 0; i--) {
		while (K >= dp[i]) { S += "L"; K -= dp[i]; }
		S += "AVITSEF";
	}
	reverse(S.begin(), S.end());
	cout << S << endl;
	return 0;
}
