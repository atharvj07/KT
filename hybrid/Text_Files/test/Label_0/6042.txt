class UF{
public:
  int V;
  vector<int> par,rnk,sz;
  UF(){}
  UF(int V):V(V),par(V),rnk(V,0),sz(V,1){for(int i=0;i<V;i++)par[i]=i;}
  
  int find(int x){
    assert(x < V);
    if(par[x]==x)return x;
    return par[x]=find(par[x]);
  }
  
  void unite(int x,int y){
    x=find(x), y=find(y);
    if(x==y)return;
    if(x > y)par[x]=y, sz[y] += sz[x];
    else{
      par[y]=x;
      sz[x] += sz[y];
      if(rnk[x]==rnk[y])rnk[x]++;
    }
  }

  bool same(int x,int y){return find(x)==find(y);}

  int size(int x){return sz[find(x)];}
};

vector<string> rot90(const vector<string> &v){
  int h = v.size(), w = v[0].size();
  vector<string> res(w,string(h,'.'));
  for(int i=0;i<h;i++) for(int j=0;j<w;j++) res[j][h-i-1] = v[i][j];
  return res;
}

int h, w, n;
vector<vector<string> >mp;
int getIdx(int y,int x){return w * (w - 1 - x) + y;}

vector<int> getNext(vector<int> par, vector<string> &mp){

  assert((int)par.size() == h);
  vector<vector<int> > num(h, vector<int>(w));
  for(int i=0;i<h;i++)
    for(int j=0;j<w;j++) num[i][j] = getIdx(i, j) + h + 1;

  for(int i=0;i<h;i++) if(par[i] != -1) num[i][0] = par[i];
  
  UF uf(h * w + h + 1);
  for(int i=0;i<h;i++){
    for(int j=0;j<w;j++){
      if(mp[i][j] == '#') continue;
      if(i + 1 < h && mp[i+1][j] != '#') uf.unite(num[i][j], num[i+1][j]);
      if(j + 1 < w && mp[i][j+1] != '#') uf.unite(num[i][j], num[i][j+1]);
    }
  }

  for(int i=0;i<h;i++)
    for(int j=0;j<w;j++) num[i][j] = uf.find(num[i][j]);
  //for(int i=0;i<h;i++) pr(mp[i]);
  //for(int i=0;i<w;i++) pr(i, num[i]);


  vector<int> res(h, -1);
  vector<int> used(w * h + h + 1, -1);
  int cnt = 1;
  for(int i=0;i<h;i++){
    if(mp[i][w-1] == '#') continue;
    int p = num[i][w-1];
    if(p == 0) res[i] = 0;
    else if(used[p] != -1) res[i] = used[p];
    else res[i] = used[p] = cnt++;
  } 
  return res;
}

int check(vector<int> &par){
  for(int i=0;i<h;i++) if(par[i] == 0) return 1;
  return 0;
}

vector<map<vector<int>, int> >mem;
int dfs(int idx, vector<int> par){
  //pr(idx, par);
  if(check(par) == false) return 0;
  if(idx == n) return 1;
  if(mem[idx].count(par)) return mem[idx][par];
  
  int res = false;
  auto tmp = mp[idx];
  for(int i=0;i<4;i++){
    if(res) break;
    tmp = rot90(tmp);
    auto npar = getNext(par, tmp);
    res |= dfs(idx+1, npar);
  }
  return mem[idx][par] = res;
}
