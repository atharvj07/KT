(n, k) = map(int, input().split(' '))
(lt, ls) = (list(map(int, input().split(' '))), list(map(int, input().split(' '))))
s = 0
for i in enumerate(lt):
	if ls[i[0]]:
		s += i[1]
		lt[i[0]] = 0
curr_s = sum(lt[:k])
mx = curr_s
for i in range(1, n - k + 1):
	temp = curr_s - lt[i - 1] + lt[i + k - 1]
	curr_s = temp
	if temp > mx:
		mx = temp
print(s + mx)
