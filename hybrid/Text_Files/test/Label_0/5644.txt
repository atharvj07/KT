#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
using pii = pair<int, int>;
using vi = vector<int>;
using vvi = vector<vi>;
const int INF = 2e9 + 5;
int main() {
  cin.tie(0), cin.sync_with_stdio(0), cout.tie(0), cout.sync_with_stdio(0);
  int n;
  cin >> n;
  map<string, set<string>> m;
  while (n--) {
    string s;
    cin >> s;
    for (int i = 0; i < s.size(); i++)
      for (int j = i; j < s.size(); j++) m[s.substr(i, j - i + 1)].insert(s);
  }
  int q;
  cin >> q;
  while (q--) {
    string t;
    cin >> t;
    cout << m[t].size() << ' ' << (m[t].size() ? *m[t].begin() : "-") << '\n';
  }
  return cout << '\n', 0;
}
