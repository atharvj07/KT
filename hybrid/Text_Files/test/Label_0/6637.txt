#include <bits/stdc++.h>
using namespace std;
const int MAX = 5e3 + 5;
const int OO = 1e9;
int n, nextStart[MAX], memo[MAX][MAX];
int solve(int i, int k) {
  if (k == 0 || i == n) return 0;
  if (memo[i][k] != -1) return memo[i][k];
  return memo[i][k] = max(solve(i + 1, k),
                          solve(nextStart[i], k - 1) + nextStart[i] - i);
}
int main() {
  int k;
  cin >> n >> k;
  vector<int> v;
  for (int i = 0; i < n; i++) {
    int a;
    cin >> a;
    v.push_back(a);
  }
  sort(v.begin(), v.end());
  int be = 0;
  for (int i = 0; i < n; i++) {
    while (be < n && v[i] + 5 >= v[be]) be++;
    nextStart[i] = be;
  }
  memset(memo, -1, sizeof memo);
  cout << solve(0, k);
  return 0;
}
