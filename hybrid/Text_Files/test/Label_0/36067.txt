
import java.util.Arrays;
import java.util.Scanner;

public class B126 {
    public static void main(String args[])
    {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        
        int pos[]=new int[s.length()+1];
        int flg[]=new int[s.length()+1];
        
        int ind=0,l=s.length();
        for(int i=1;i<l;i++)
        {
            while(ind>0 && s.charAt(ind)!=s.charAt(i)){
                ind=pos[ind-1];
            }
            if(s.charAt(ind)==s.charAt(i))ind++;
            pos[i]=ind;
            flg[ind]++;
        }
        
        int len=pos[l-1];
        if(len>0 && flg[len]<2)len=pos[len-1];
        if(len==0)System.out.print("Just a legend");
        else System.out.print(s.substring(0,len));
    }
}
