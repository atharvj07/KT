a = int(input())
b = list(map(int, input().split()))
i = 0
ma = 0
mi = 0
while i < a:
	if i == 0 or i == a - 1:
		ma = abs(b[0] - b[a - 1])
		if i == 0:
			mi = abs(b[0] - b[1])
		else:
			mi = abs(b[a - 1] - b[a - 2])
	else:
		max1 = abs(b[i] - b[a - 1])
		max2 = abs(b[i] - b[0])
		min1 = abs(b[i] - b[i - 1])
		min2 = abs(b[i] - b[i + 1])
		if min1 > min2:
			mi = min2
		else:
			mi = min1
		if max1 > max2:
			ma = max1
		else:
			ma = max2
	print(mi, ma)
	i += 1
