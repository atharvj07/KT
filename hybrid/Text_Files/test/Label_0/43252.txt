N = 100
a = (0, 1, 5, N, N, 2, N, N, 8, N)

def g(s):
	(u, v, x, y) = (a[int(x)] for x in s if ':' > x)
	return h > x + y * 10 and v * 10 + u < m
R = lambda x=' ': map(int, input().split(x))
(t,) = R()
while t:
	t -= 1
	(h, m) = R()
	(x, y) = R(':')
	y += x * m
	print(next(filter(g, (f'{i // m % h:02}:{i % m:02}' for i in range(y, y + h * m)))))
