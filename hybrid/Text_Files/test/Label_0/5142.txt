bool complexSort(Point lhs,Point rhs){
    if (sgn(lhs.real()-rhs.real())){
        return lhs.real() < rhs.real();
    }
    else{
        return lhs.imag() < rhs.imag();
    }
}

double cross(Point z1,Point z2){
    return (conj(z1)*z2).imag();
}

vector<ll> convex_hull(vector<Point> p){
    vector<tuple<double,double,ll>> k;
    rep(i,p.size()){
        k.push_back(make_tuple(p[i].real(),p[i].imag(),i));
    }
    SORT(k);
    vector<ll> index;
    for (auto j:k){
        double a,b;
        ll c;
        tie(a,b,c) = j;
        index.push_back(c);
    }
    //array_output(all(index));
    SORT(k);
    int cnt = 0,n = p.size();
    sort(all(p),complexSort);
    vector<Point> s(n*2);
    vector<ll> ind(n*2);
    rep(i,n){
        while (cnt > 1 && sgn(cross(s[cnt-1]-s[cnt-2],p[i]-s[cnt-1])) <= 0) cnt--;
        s[cnt] = p[i];
        ind[cnt++] = index[i];
    }
    for (ll i = n-2,t = cnt;i >= 0;i--){
        while (cnt > t && sgn(cross(s[cnt-1]-s[cnt-2],p[i]-s[cnt-1])) <= 0) cnt--;
        s[cnt] = p[i];
        ind[cnt++] = index[i];
    }
    ind.resize(cnt-1);
    return ind;
}

class Prim{
    public:
    struct Node{
        vector<pair<double,ll>> edges; // cost node
        bool done;
        double cost;
    };
    ll V;
    vector<Node> nodes;
    Prim(ll v){
        V = v;
        nodes.resize(v);
    }
    void add_edge(ll v,ll node,double cost){
        nodes[v].edges.push_back(make_pair(cost,node));
    }
    double prim(){
        priority_queue<pair<double,ll>,vector<pair<double,ll>>,greater<pair<double,ll>>> mincost;
        rep(i,V){
            nodes[i].cost = INF;
            nodes[i].done = false;
        }
        double res = 0;
        mincost.push(make_pair(0.0,0));
        while(!mincost.empty()){
            pair<double,ll> p = mincost.top();mincost.pop();
            ll v = p.second;
            if (nodes[v].done) continue;
            nodes[v].done = true;
            res += p.first;
            for(pair<double,ll> u:nodes[v].edges){
                if (!nodes[u.second].done) mincost.push(make_pair(u.first,u.second));
            }
        }
        return res;
    }
};
