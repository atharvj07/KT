t = int(input())
for i in range(t):
	n = int(input())
	lst = list(map(int, input().split()))
	sums = sum(lst)
	if sums == 0:
		print('NO')
	else:
		print('YES')
		if sums < 0:
			lst.sort()
			for j in lst:
				print(j, end=' ')
			print()
		elif sums > 0:
			lst.sort(reverse=True)
			for j in lst:
				print(j, end=' ')
			print()
