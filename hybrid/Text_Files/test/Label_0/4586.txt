#include <bits/stdc++.h>
using namespace std;
template <typename T>
T sqr(T x) {
  return x * x;
}
template <typename T>
T abs(T x) {
  return x > 0 ? x : -x;
}
int gcd(int x, int y) {
  while (x != 0 && y != 0) {
    if (x > y) {
      x %= y;
    } else {
      y %= x;
    }
  }
  return x + y;
}
int main() {
  ios_base::sync_with_stdio(false);
  int n, l, r;
  cin >> n >> l >> r;
  if (n == 1) {
    cout << r - l + 1 << endl;
  } else if (n == 2) {
    cout << (r - l + 1) * 1ll * (r - l) << endl;
  } else {
    long long ans = 0;
    for (int x = 2; x * x <= r; x++) {
      for (int y = 1; y < x; y++) {
        if (gcd(x, y) == 1) {
          long long maxi = 1;
          for (int i = 1; i < n; i++) {
            maxi *= x;
            if (maxi > r) {
              break;
            }
          }
          if (maxi > r) {
            continue;
          }
          long long mini = 1;
          if (y > 1) {
            for (int i = 1; i < n; i++) {
              mini *= y;
            }
          }
          long long mini_delta = (l - 1) / mini + 1;
          long long maxi_delta = r / maxi;
          ans += 2 * max(maxi_delta - mini_delta + 1, 0ll);
        }
      }
    }
    cout << ans << "\n";
  }
  return 0;
}
