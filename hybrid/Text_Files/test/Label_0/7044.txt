#include <bits/stdc++.h>
using namespace std;
char _;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int t, n, a;
  cin >> t;
  for (int i = 0; i < t; i++) {
    cin >> n;
    int currentSum = 0;
    int currentCost = 0;
    int zeros = 0;
    for (int j = 0; j < n; j++) {
      cin >> a;
      if (a == 0) {
        zeros++;
      }
      currentSum += a;
    }
    if (currentSum != 0 && zeros == 0) {
      cout << "0" << endl;
    } else if (zeros > 0) {
      if (currentSum >= 0 || currentSum + zeros != 0) {
        cout << to_string(zeros) << endl;
      } else {
        cout << to_string(zeros + 1) << endl;
      }
    } else if (currentSum == 0) {
      cout << "1" << endl;
    }
  }
  return 0;
}
