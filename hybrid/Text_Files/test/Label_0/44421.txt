import sys
f = sys.stdin
n = f.readline()
n = int(n)
t = [[0], [1]]
for j in range(n):
	cur = [0] + t[-1]
	for (i, x) in enumerate(t[-2]):
		cur[i] += x
	if min(cur) < -1 or max(cur) > 1:
		cur = [0] + t[-1]
		for (i, x) in enumerate(t[-2]):
			cur[i] -= x
	t.append(cur)
print(len(t[n + 1]) - 1)
print(' '.join([str(x) for x in t[n + 1]]))
print(len(t[n]) - 1)
print(' '.join([str(x) for x in t[n]]))
