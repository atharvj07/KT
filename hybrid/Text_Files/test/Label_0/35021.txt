import java.util.*;
public class Program {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int q = sc.nextInt();
		String s, t;
		for(int h = 1; h <= q; h++){
		    s = sc.next(); 
		    t = sc.next();
		    int len = s.length();
		    String ans = "Yes";
		    //if they share at least one char in common, it's possible.
		    for(int i = 0; i < len; i++){
    		    if(t.indexOf(s.charAt(i)) != -1){
    		        ans = "YES";
    		        break;
    		    }
    		    ans = "No";
		    } 
		    System.out.println(ans);
		}
	}
}
