#include<iostream>
#include<vector>
#include<algorithm>

using namespace std;

void MakeData(int, int, bool, vector<int>, vector<int>&);
void NextData(vector<int>, vector<int>&);

int main(){
  int i, n;
  vector<int> coin, data;

  for(i=1; i<=17; ++i) coin.push_back(i*i);

  while(1){
    cin >> n;
    if(n == 0) break;
    MakeData(n, n, true, coin, data);

    for(i=1; data[0]!=1; ++i) NextData(coin, data);
    cout << i << endl;
    data.clear();
  }
  return 0;
}

void MakeData(int n, int m, bool is_upper,
	      vector<int> coin, vector<int>& data){
  vector<int>::iterator i;
  if(is_upper) i = upper_bound(coin.begin(), coin.end(), m);
  else i = lower_bound(coin.begin(), coin.end(), m);
  --i;
  while(1){
    if(n >= *i){
      data.push_back(*i);
      n -= *i;
      if(n == 0) return;
    }else{
      --i;
    }
  }
}

void NextData(vector<int> coin, vector<int>& data){
  int n, sum;
  vector<int>::iterator i, j;
  i = find(data.begin(), data.end(), 1);
  n = *(i-1);
  if(i == data.end()){
    data.pop_back();
    MakeData(n, n, false, coin, data);
  }else{
    for(j=i-1, sum=0; j!=data.end(); ++j) sum += *j;
    data.erase(i-1, data.end());
    MakeData(sum, n, false, coin, data);
  }
}
