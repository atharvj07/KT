def can(a, s, k):
	global cost
	costs = [a[i] + (i + 1) * k for i in range(n)]
	costs.sort()
	cost = sum(costs[:k])
	return cost <= s
cost = 0
(n, s) = map(int, input().split())
a = list(map(int, input().split()))
(lt, rt) = (0, n + 1)
while lt + 1 < rt:
	mid = (lt + rt) // 2
	if can(a, s, mid):
		lt = mid
	else:
		rt = mid
can(a, s, lt)
print(lt, cost)
