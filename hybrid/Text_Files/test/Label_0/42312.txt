result = 0
mod = 10 ** 6 + 3
(n, C) = map(int, input().split())

def fact(n):
	fact = 1
	for i in range(1, n + 1):
		fact = fact * i % mod
	return fact

def pow(a, b):
	p = 1
	for i in range(b):
		p = a * p % mod
	return p
result = fact(n + C) * pow(fact(n), mod - 2) * pow(fact(C), mod - 2) - 1
print(int(result % mod))
