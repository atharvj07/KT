#include <bits/stdc++.h>
using namespace std;
vector<pair<long long, long long> > G[300005];
set<pair<pair<long long, long long>, long long> > s;
set<pair<pair<long long, long long>, long long> >::iterator it;
long long f[300005];
long long n, k;
void insert(long long first, long long second, long long id) {
  it = s.lower_bound(make_pair(make_pair(first, 0), 0));
  if (it != s.begin()) {
    it--;
    if (it->first.second > first) {
      pair<pair<long long, long long>, long long> IT = *it;
      s.erase(it);
      s.insert(make_pair(make_pair(IT.first.first, first), IT.second));
      s.insert(make_pair(make_pair(first, IT.first.second), IT.second));
    }
  }
  it = s.lower_bound(make_pair(make_pair(second, 0), 0));
  if (it != s.begin()) {
    it--;
    if (it->first.second > second) {
      pair<pair<long long, long long>, long long> IT = *it;
      s.erase(it);
      s.insert(make_pair(make_pair(IT.first.first, second), IT.second));
      s.insert(make_pair(make_pair(second, IT.first.second), IT.second));
    }
  }
  it = s.lower_bound(make_pair(make_pair(first, 0), 0));
  set<pair<pair<long long, long long>, long long> >::iterator IT = it;
  if (it != s.end() && it->first.second <= second) {
    long long lst = first;
    do {
      long long L = IT->first.first;
      long long R = IT->first.second;
      if (L != lst) G[id].push_back(make_pair(1, L - lst));
      G[id].push_back(make_pair(IT->second + 1, R - L));
      lst = R;
      IT++;
    } while (IT != s.end() && IT->first.second <= second);
    if (lst != second) G[id].push_back(make_pair(1, second - lst));
    s.erase(it, IT);
  } else
    G[id].push_back(make_pair(1, second - first));
  s.insert(make_pair(make_pair(first, second), id));
}
long long calc_cnt(long long x) {
  for (long long i = 1; i <= n; i++) f[i] = 0;
  long long first = 0, cnt = 0;
  for (long long second = 1; second <= n; second++) {
    for (long long j = 0; j < G[second].size(); j++) {
      long long i = G[second][j].first;
      long long len = G[second][j].second;
      i = max(first, i);
      f[i] += len;
      f[second + 1] -= len;
    }
    while (first < second && f[first + 1] + f[first] >= x)
      f[first + 1] += f[first], first++;
    cnt += first;
  }
  return cnt;
}
long long calc_sum(long long x) {
  for (long long i = 1; i <= n; i++) f[i] = 0;
  long long first = 0, ans = 0, sum = 0;
  for (long long second = 1; second <= n; second++) {
    for (long long j = 0; j < G[second].size(); j++) {
      long long i = G[second][j].first;
      long long len = G[second][j].second;
      if (i <= first) ans += len * (first - i + 1);
      i = max(i, first);
      f[i] += len;
      f[second + 1] -= len;
    }
    while (first < second && f[first + 1] + f[first] >= x)
      f[first + 1] += f[first], ans += f[first + 1], first++;
    sum += ans;
  }
  return sum;
}
signed main() {
  scanf("%lld %lld", &n, &k);
  for (long long i = 1, first, second; i <= n; i++) {
    scanf("%lld %lld", &first, &second);
    insert(first, second, i);
  }
  long long first = 0, second = 1e9, ans;
  while (first <= second) {
    long long mid = (first + second) >> 1;
    if (calc_cnt(mid) >= k)
      ans = mid, first = mid + 1;
    else
      second = mid - 1;
  }
  printf("%lld\n", calc_sum(ans + 1) + (k - calc_cnt(ans + 1)) * ans);
  return 0;
}
