#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  int pos;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == '^') pos = i;
  }
  unsigned long long int left = 0;
  unsigned long long int right = 0;
  for (int i = 0; i < pos; i++) {
    if (s[i] != '=')
      left += (unsigned long long int)(pos - i) *
              ((unsigned long long int)s[i] - 48);
  }
  for (int i = pos; i < s.size(); i++) {
    if (s[i] != '=')
      right += (unsigned long long int)(i - pos) *
               ((unsigned long long int)s[i] - 48);
  }
  if (left == right) {
    cout << "balance";
  } else if (left > right)
    cout << "left";
  else {
    cout << "right";
  }
  return 0;
}
