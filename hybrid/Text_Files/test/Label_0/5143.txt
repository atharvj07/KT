  for (int i = 1; i <= n / 2; i++) {
    int b = (n - (i * 2 - 1)) / 2;
    for (int j = 0; j < b; j++) printf("*");
    for (int j = 0; j < i * 2 - 1; j++) printf("D");
    for (int j = 0; j < b; j++) printf("*");
    puts("");
  }
  for (int j = 0; j < n; j++) printf("D");
  puts("");
  for (int i = n / 2 + 2; i <= n; i++) {
    int p = n - i + 1;
    int b = (n - (p * 2 - 1)) / 2;
    for (int j = 0; j < b; j++) printf("*");
    for (int j = 0; j < p * 2 - 1; j++) printf("D");
    for (int j = 0; j < b; j++) printf("*");
    puts("");
  }
  return 0;
}
