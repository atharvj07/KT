#include <bits/stdc++.h>
using namespace std;
void solve() {
  int n, a, b, c;
  char ch;
  cin >> n;
  vector<int> gameOne(n * 2 + 1), cnt(n * 2 + 1);
  for (int i = 1; i <= n; i++) {
    cout << "? " << i << " " << i + n << endl;
    cin >> ch;
    if (ch == '>') {
      gameOne[i + n] = i;
    } else {
      gameOne[i] = i + n;
    }
  }
  for (int i = 1; i <= n * 2; i++) {
    for (int j = i + 1; j <= n * 2; j++) {
      if (gameOne[i] == 0 || gameOne[j] == 0) continue;
      cout << "? " << i << " " << j << endl;
      cin >> ch;
      if (ch == '>')
        cnt[i]++;
      else
        cnt[j]++;
    }
  }
  for (int i = 1; i <= n * 2; i++) {
    if (gameOne[i] == 0) continue;
    if (gameOne[i] && cnt[i] == n - 1)
      a = gameOne[i];
    else if (gameOne[i] && cnt[i] == n - 2)
      b = gameOne[i];
    else if (gameOne[i] && cnt[i] == n - 3)
      c = gameOne[i];
  }
  cout << "? " << b << " " << c << endl;
  cin >> ch;
  if (ch == '<') swap(b, c);
  for (int i = 1; i <= n * 2; i++) {
    if (i == a || i == b) continue;
    for (int j = i + 1; j <= n * 2; j++) {
      if (gameOne[i] && gameOne[j]) continue;
      if (j == a || j == b) continue;
      if (abs(i - j) == n) continue;
      cout << "? " << i << " " << j << endl;
      cin >> ch;
    }
  }
  cout << "!" << endl;
  return;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    solve();
  }
  return 0;
}
