T = int(input())
for i in range(T):
	puzzle = [input() for _ in range(9)]
	A = [0, 1, 2, 3, 4, 5, 6, 7, 8]
	B = [0, 3, 6, 1, 4, 7, 2, 5, 8]
	for (a, b) in zip(A, B):
		tmp = int(puzzle[a][b])
		if tmp == 9:
			tmp = 1
		else:
			tmp += 1
		tmp = str(tmp)
		puzzle[a] = puzzle[a][0:b] + tmp + puzzle[a][b + 1:9]
	for j in range(9):
		print(puzzle[j])
