import java.util.*;
public class FunnySubstrings {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int T = in.nextInt();
        for (int t = 0; t < T; t++) {
            int N = in.nextInt();
            HashMap<String, Long> count = new HashMap<>();
            HashMap<String, String> start = new HashMap<>(), end = new HashMap<>();
            long res = 0;
            for (int i = 0; i < N; i++) {
                String key = in.next();
                if (in.next().equals(":=")) {
                    String s = in.next();
                    long n = 0;
                    for (int j = 0; j + 3 < s.length(); j++) {
                        if (s.substring(j, j + 4).equals("haha")) {
                            n++;
                        }
                    }
                    count.put(key, n);
                    start.put(key, s.substring(0, Math.min(3, s.length())));
                    end.put(key, s.substring(Math.max(0, s.length() - 3)));
                    res = n;
                } else {
                    String a = in.next();
                    in.next();
                    String b = in.next();
                    long n = count.get(a) + count.get(b);
                    String mid = end.get(a) + start.get(b);
                    for (int j = 0; j + 3 < mid.length(); j++) {
                        if (mid.substring(j, j + 4).equals("haha")) {
                            n++;
                        }
                    }
                    count.put(key, n);
                    start.put(key, start.get(a).length() < 3 ? mid.substring(0, Math.min(3, mid.length())) : start.get(a));
                    end.put(key, end.get(b).length() < 3 ? mid.substring(Math.max(0, mid.length() - 3)) : end.get(b));
                    res = n;
                }
            }
            System.out.println(res);
        }
    }
}
