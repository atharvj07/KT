#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  int span = 1;
  cin >> n;
  int mid = n * n / 2;
  for (int j = mid; j > 0;) {
    for (int i = 0; i < n / 2; i++) {
      cout << j - i << " " << mid + span << " ";
      span++;
    }
    cout << "\n";
    j = j - (n / 2);
  }
  return 0;
}
