a = int(input())
from collections import *

def check(g1, g2):
	t = g2[0]
	m = g1[-1]
	if abs(ord(t) - ord(m)) == 1:
		return 0
	return 1

def printer(g1, g2, al):
	ans = []
	for i in range(len(g1)):
		for j in range(al[g1[i]]):
			ans.append(g1[i])
	for i in range(len(g2)):
		for j in range(al[g2[i]]):
			ans.append(g2[i])
	print(''.join(ans))
for t in range(a):
	s = input()
	al = defaultdict(int)
	for i in range(len(s)):
		al[s[i]] += 1
	g1 = []
	g2 = []
	for i in al:
		if ord(i) % 2 == 1:
			g1.append(i)
		else:
			g2.append(i)
	g1.sort()
	g2.sort()
	if len(g1) == 0 or len(g2) == 0:
		printer(g1, g2, al)
		continue
	m1 = g1[::-1]
	m2 = g2[::-1]
	if check(g1, g2):
		printer(g1, g2, al)
		continue
	if check(g1, m2):
		printer(g1, m2, al)
		continue
	if check(m1, g2):
		printer(m1, g2, al)
		continue
	if check(m1, m2):
		printer(m1, m2, al)
		continue
	print('No answer')
