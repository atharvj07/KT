#include <bits/stdc++.h>
using namespace std;
int n, m;
struct Vertex {
  int indeg = 0;
  int outdeg = 0;
  vector<int> ins;
};
Vertex vecs[100009];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n >> m;
  int a, b;
  long long ans = 0;
  for (int i = 0; i < m; i++) {
    cin >> a >> b;
    if (a < b) {
      swap(a, b);
    }
    vecs[b].ins.push_back(a);
    vecs[a].outdeg++;
    vecs[b].indeg++;
  }
  for (int i = 1; i <= n; i++) {
    ans += ((long long)vecs[i].indeg) * vecs[i].outdeg;
  }
  cout << ans << '\n';
  int q, v;
  long long old, now;
  cin >> q;
  while (q--) {
    cin >> v;
    old = ((long long)vecs[v].indeg) * vecs[v].outdeg;
    now = 0;
    for (auto idx : vecs[v].ins) {
      old += ((long long)vecs[idx].indeg) * vecs[idx].outdeg;
      now += ((long long)(++vecs[idx].indeg)) * (--vecs[idx].outdeg);
      vecs[idx].ins.push_back(v);
    }
    vecs[v].ins.clear();
    vecs[v].outdeg += vecs[v].indeg;
    vecs[v].indeg = 0;
    ans += now - old;
    cout << ans << '\n';
  }
}
