ull bitpow (ull a, ull x, ull modulo) {
	ull result = 1;
	while (x) {
		if (x & 1) {
			result *= a;
			result %= modulo;
		}
		x /= 2;
		a = (a * a) % modulo;
	}
	return result;
}

ull divide (ull a, ull b, ull modulo) {
	return (a * bitpow(b, modulo - 2, modulo)) % modulo;
}

void eval (sll v) {
	seg[v].first += lazy[v];

	if (v < base) {
		lazy[v * 2] += lazy[v];
		lazy[v * 2 + 1] += lazy[v];
	}

	lazy[v] = 0;
}
P add (sll l, sll r, sll x, sll beg = 0, sll end = base, sll i = 1) {
	eval(i);
	if (end <= l || r <= beg) return UNITY;
	if (l <= beg && end <= r) {
		lazy[i] += x;
		eval(i);
		return seg[i];
	}

	sll med = (beg + end) / 2;
	P vl = add(l, r, x, beg, med, i * 2);
	P vr = add(l, r, x, med, end, i * 2 + 1);
	seg[i] = merge(seg[i * 2], seg[i * 2 + 1]);
	return merge(vl, vr);
}
void upd (sll v, sll x) {
	x -= add(v, v + 1, 0).first;
	add(v, v + 1, x);

ull ufind (ull x) {

bool unite (ull a, ull b) {
	a = ufind(a);
	b = ufind(b);
	if (a == b) return false;

	if (size[a] < size[b]) {
		ull tmp = a;
		a = b;
		b = tmp;
	}

	size[a] += size[b];
	parent[b] = a;
	return true;
}
