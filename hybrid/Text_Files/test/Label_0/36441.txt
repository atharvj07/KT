import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.InputMismatchException;
import java.io.IOException;
import java.io.InputStream;
import java.util.LinkedList;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) throws Exception {
        Thread thread = new Thread(null, new TaskAdapter(), "", 1 << 27);
        thread.start();
        thread.join();
    }

    static class TaskAdapter implements Runnable {
        @Override
        public void run() {
            InputStream inputStream = System.in;
            OutputStream outputStream = System.out;
            FastReader in = new FastReader(inputStream);
            PrintWriter out = new PrintWriter(outputStream);
            BStrangeDefinition solver = new BStrangeDefinition();
            int testCount = Integer.parseInt(in.next());
            for (int i = 1; i <= testCount; i++)
                solver.solve(i, in, out);
            out.close();
        }
    }

    static class BStrangeDefinition {
        static final int MX = (int) 1e6;
        LinkedList<Integer>[] fac = new LinkedList[MX + 1];
        boolean[] vis = new boolean[MX + 1];
        boolean pre;
        int[] f = new int[MX + 1];

        void pre() {
            pre = true;
            for (int i = 0; i <= MX; i++) fac[i] = new LinkedList<>();
            for (int i = 2; i <= MX; i++) {
                if (!vis[i]) for (int j = i; j <= MX; j += i) {
                    fac[j].add(i);
                    vis[j] = true;
                }
            }
        }

        public void solve(int testNumber, FastReader s, PrintWriter w) {

            if (!pre) pre();

            int n = s.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) a[i] = s.nextInt();

            int[] nonRoot = new int[n];
            int mx = 0, mxEven = 0;

            for (int i = 0; i < n; i++) {
                int nonRootValue = 1;
                for (int j : fac[a[i]]) {
                    int cnt = 0;
                    while (a[i] % j == 0) {
                        a[i] /= j;
                        cnt++;
                    }
                    if (cnt % 2 == 1) nonRootValue *= j;
                }
                f[nonRootValue]++;
                nonRoot[i] = nonRootValue;

                mx = Math.max(mx, f[nonRootValue]);
            }

            mxEven += f[1];
            for (int i = 0; i < n; i++)
                if (nonRoot[i] != 1 && f[nonRoot[i]] % 2 == 0) mxEven++;
            for (int i = 0; i < n; i++) f[nonRoot[i]] = 0;

            int q = s.nextInt();

            while (q-- > 0) {
                long wi = s.nextLong();
                if (wi == 0) w.println(mx);
                else w.println(Math.max(mx, mxEven));
            }


        }

    }

    static class FastReader {
        private InputStream stream;
        private byte[] buf = new byte[1024];
        private int curChar;
        private int numChars;
        private FastReader.SpaceCharFilter filter;

        public FastReader(InputStream stream) {
            this.stream = stream;
        }

        public int read() {

            if (numChars == -1)
                throw new InputMismatchException();

            if (curChar >= numChars) {

                curChar = 0;

                try {
                    numChars = stream.read(buf);
                } catch (IOException e) {
                    throw new InputMismatchException();
                }

                if (numChars <= 0)
                    return -1;
            }

            return buf[curChar++];
        }

        public int nextInt() {

            int c = read();

            while (isSpaceChar(c))
                c = read();

            int sgn = 1;

            if (c == '-') {
                sgn = -1;
                c = read();
            }

            int res = 0;

            do {
                if (c < '0' || c > '9')
                    throw new InputMismatchException();

                res *= 10;
                res += c - '0';
                c = read();
            }
            while (!isSpaceChar(c));

            return res * sgn;
        }

        public long nextLong() {

            int c = read();

            while (isSpaceChar(c))
                c = read();

            int sgn = 1;

            if (c == '-') {
                sgn = -1;
                c = read();
            }

            long res = 0;

            do {
                if (c < '0' || c > '9')
                    throw new InputMismatchException();

                res *= 10;
                res += c - '0';
                c = read();
            }

            while (!isSpaceChar(c));
            return res * sgn;
        }

        public String next() {

            int c = read();

            while (isSpaceChar(c))
                c = read();

            StringBuilder res = new StringBuilder();

            do {
                res.appendCodePoint(c);
                c = read();
            }
            while (!isSpaceChar(c));

            return res.toString();
        }

        public boolean isSpaceChar(int c) {

            if (filter != null)
                return filter.isSpaceChar(c);

            return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
        }

        public interface SpaceCharFilter {
            public boolean isSpaceChar(int ch);

        }

    }
}
