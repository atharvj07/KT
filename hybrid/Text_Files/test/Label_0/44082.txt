from collections import Counter
n = int(input())
a = list(map(int, input().split()))
d = Counter(a)
v = list(d.values())
gt_1 = 0
for i in v:
	if i > 1:
		gt_1 = gt_1 + 1
s = 0
mn = 10000000000
for i in range(0, n):
	x = gt_1
	s = 0
	c = d.copy()
	for j in range(i, n):
		if x == 0:
			if mn > s:
				mn = s
			break
		s = s + 1
		c[a[j]] = c[a[j]] - 1
		if c[a[j]] == 1:
			x = x - 1
	if x == 0:
		if mn > s:
			mn = s
print(mn)
