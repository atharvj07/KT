#include<bits/stdc++.h>
#define rep(i, n) for(int i = 0; i < n; i++)
using namespace std;
int qmin(int a, int b){
	return (a < b) ? a : b;
}

int n;
char s[500500]; 
int Z[2][500500];
bool nok[500500];

void Zalgo(int dir){
	int l = 0, r = 0;
	Z[dir][0] = n;
	for(int i = 1; i < n; i++){
		if(i <= r) Z[dir][i] = qmin(Z[dir][i-l], r-i+1);
		while(i + Z[dir][i] < n && s[i + Z[dir][i]] == s[Z[dir][i]]) Z[dir][i]++;
		if(i+Z[dir][i]-1 > r){
			l = i;
			r = i+Z[dir][i]-1;
		}
	}
}

int main(){
	scanf("%s", s);
	n = strlen(s);
	bool all_same = 1;
	rep(i, n){
		if(s[i] != s[0]) all_same = 0;
	}
	if(all_same){
		cout << n << endl << "1" << endl;
		return 0;
	}
	Zalgo(0);
	for(int i = 0; i + i < n; i++){
		swap(s[i], s[n-i-1]); 
	}
	Zalgo(1);
	for(int i = 1; i <= n; i++){
		for(int j = 2; i*j <= n; j++){
			if(Z[0][i*(j-1)] >= i){
				nok[i*j] = 1;
			} else break;
		} 
		for(int j = 2; i*j <= n; j++){
			if(Z[1][i*(j-1)] >= i){
				nok[n - i*j] = 1;
			} else break;
		}
	}
	if(!nok[n]){
		cout << "1" << endl << "1" << endl; 
	} else {
		int cnt = 0;
		for(int i = 1; i < n; i++){
			if(!nok[i]) cnt++;
		}
		cout << "2" << endl << cnt << endl;
	}
	return 0;
}
