#include <bits/stdc++.h>
using namespace std;
int check(int num) {
  int t = sqrt(num);
  if (t * t == num)
    return 1;
  else
    return 0;
}
int main() {
  srand(time(NULL));
  int i, j, n, m, a, b, c, d;
  cin >> n >> m;
  while (1) {
    a = rand() % 100 + 1;
    b = rand() % 100 + 1;
    c = rand() % 100 + 1;
    d = rand() % 100 + 1;
    if (check((m - 1) * a * a + b * b) && check((n - 1) * a * a + c * c) &&
        check((n - 1) * b * b + d * d) && check((m - 1) * c * c + d * d)) {
      break;
    }
  }
  for (i = 0; i < n - 1; i++) {
    for (j = 0; j < m - 1; j++) {
      printf("%d ", a);
    }
    printf("%d\n", b);
  }
  for (j = 0; j < m - 1; j++) {
    printf("%d ", c);
  }
  printf("%d\n", d);
  return 0;
}
