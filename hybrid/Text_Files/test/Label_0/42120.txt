import math
import sys
from collections import Counter

def solve():
	n = int(input())
	T = [int(x) - 1 for x in input().split()]
	M = [[0] * n for i in range(n)]
	(curmin, ans) = ([0] * n, [0] * n)
	for i in range(n):
		ans[T[i]] += 1
		curmin[i] = T[i]
		M[i][T[i]] = 1
	for i in range(n):
		for j in range(i + 1, n):
			M[i][T[j]] += 1
			if M[i][T[j]] > M[i][curmin[i]] or (M[i][T[j]] == M[i][curmin[i]] and T[j] < curmin[i]):
				curmin[i] = T[j]
			ans[curmin[i]] += 1
	return ans
print(' '.join(map(str, solve())))
