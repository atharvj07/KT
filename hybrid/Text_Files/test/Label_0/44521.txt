import sys
import collections as cc
input = sys.stdin.readline
I = lambda : list(map(int, input().split()))
mod = 10 ** 9 + 7
input = sys.stdin.readline
mod = 10 ** 9 + 7
for tc in range(int(input())):
	(n,) = I()
	g = [[] for i in range(n + 1)]
	for i in range(n - 1):
		(x, y) = I()
		g[x].append(y)
		g[y].append(x)
	(m,) = I()
	l = sorted(I())
	if m < n - 1:
		l = [1] * (n - 1 - m) + l
	else:
		temp = 1
		for j in range(m - n + 1):
			temp = temp * l[-j - 1] % mod
		l[n - 2] = l[n - 2] * temp % mod
	odr = []
	st = [1]
	visi = [0] * (n + 1)
	visi[1] = 1
	par = [-1] * (n + 1)
	while st:
		x = st.pop()
		odr.append(x)
		for to in g[x]:
			if visi[to] == 0:
				st.append(to)
				par[to] = x
				visi[to] = 1
	tem = [1] * (n + 1)
	te = []
	for x in odr[1:][::-1]:
		te.append(tem[x] * (n - tem[x]))
		tem[par[x]] += tem[x]
	te.sort()
	ans = 0
	for i in range(n - 1):
		ans = (ans + te[i] * l[i]) % mod
	print(ans)
