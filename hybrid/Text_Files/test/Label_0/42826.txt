import sys
input = sys.stdin.readline
DEBUG = False

def Solve():
	n = int(input())
	S = input().strip()
	if n <= 1:
		return S[0] * 2
	if n >= 2:
		if S[0] == S[1]:
			return S[0] + S[1]
	for i in range(1, n):
		if S[i] <= S[i - 1]:
			continue
		else:
			return S[:i] + S[i - 1::-1]
	return S + S[::-1]
for _ in range(int(input())):
	print(Solve())
