#include <bits/stdc++.h>
using namespace std;
int q;
int b[200022];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int h = 0, t = 0;
  char op1;
  int op2;
  cin >> q >> op1 >> op2;
  b[op2] = h;
  for (int i = 0; i < q - 1; i++) {
    cin >> op1 >> op2;
    if (op1 == 'L')
      b[op2] = --h;
    else if (op1 == 'R')
      b[op2] = ++t;
    else if (op1 == '?') {
      int out = min(b[op2] - h, t - b[op2]);
      cout << out << '\n';
    }
  }
  return 0;
}
