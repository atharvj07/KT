#include <bits/stdc++.h>
using namespace std;
long long INF = 1000000000000000005LL;
long long power(long long x, long long y, long long p = 1000000007) {
  long long res = 1;
  x = x % p;
  while (y > 0) {
    if (y & 1) res = (res * x) % p;
    y = y >> 1;
    x = (x * x) % p;
  }
  return res;
}
long long inverse(long long a, long long m = 1000000007) {
  return power(a, m - 2, m);
}
vector<vector<long long> > matmul(const vector<vector<long long> > &a,
                                  const vector<vector<long long> > &b,
                                  long long M = 1000000007) {
  long long n = a.size(), m = a[0].size(), l = b[0].size();
  assert(m == b.size());
  vector<vector<long long> > c(n, vector<long long>(l, 0));
  for (long long i = 0; i < n; i++)
    for (long long j = 0; j < l; j++)
      for (long long k = 0; k < m; k++) {
        c[i][j] = (c[i][j] + a[i][k] * b[k][j]) % M;
      }
  return c;
}
vector<vector<long long> > matpow(vector<vector<long long> > a, long long p,
                                  long long M = 1000000007) {
  assert(a.size() == a[0].size());
  long long n = a.size();
  vector<vector<long long> > res(n, vector<long long>(n, 0));
  for (long long i = 0; i < n; i++) res[i][i] = 1;
  while (p > 0) {
    if (p & 1) res = matmul(res, a, M);
    a = matmul(a, a, M);
    p >>= 1;
  }
  return res;
}
long long dp[2001][2001] = {0};
long long n, m;
pair<long long, char> rec(long long x, long long y, vector<string> &gr) {
  if (x < 0 || x >= n) return {0, '0'};
  if (y < 0 || y >= m) return {0, '0'};
  if (dp[x][y] != 0) return {dp[x][y], gr[x][y]};
  pair<long long, long long> sq1 = rec(x + 1, y - 1, gr);
  pair<long long, long long> sq2 = rec(x - 1, y - 1, gr);
  pair<long long, long long> sq3 = rec(x, y - 1, gr);
  if (y < (2 * sq3.first) || gr[x][y - (2 * sq3.first)] != gr[x][y])
    sq3.first--;
  if (gr[x][y] == sq1.second && sq1.second == sq2.second &&
      sq2.second == sq3.second)
    return {dp[x][y] = min(min(sq1.first, sq2.first), sq3.first) + 1, gr[x][y]};
  else
    return {dp[x][y] = 1, gr[x][y]};
}
void solve() {
  cin >> n >> m;
  vector<string> gr(n);
  for (long long i = 0; i < n; i++) {
    cin >> gr[i];
  }
  long long ans = 0;
  for (long long i = 0; i < n; i++) {
    for (long long j = 0; j < m; j++) {
      ans += rec(i, j, gr).first;
    }
  }
  cout << ans << '\n';
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long T = 1;
  while (T--) {
    solve();
  }
  return 0;
}
