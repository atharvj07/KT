#include<iostream>
#include<string.h>
using namespace std;

int main()
{
  char x[1000],y[10];
  int sum=0;
  cin>>y;
  for(int i=0;i<strlen(y);i++)y[i]=tolower(y[i]);
  while(1){
    cin>>x;
    if(strcmp(x,"END_OF_TEXT")==0)break;
    for(int i=0;i<strlen(x);i++)x[i]=tolower(x[i]);
    if(strcmp(x,y)==0)sum++;
  }
  cout<<sum<<endl;
  return 0;
}
