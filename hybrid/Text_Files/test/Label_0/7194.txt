#include <bits/stdc++.h>
 
using namespace std;
typedef long long ll;
const int MAXN = 100100;

int N;
pair <int, int> loc[MAXN];

int main()
{
    ios_base::sync_with_stdio(0);

    int ans = 0;
    cin >> N;
    for (int i = 0; i < N; i++)
    {
        cin >> loc[i].first >> loc[i].second;
        loc[i].second++;

        ans = max (ans, loc[i].second - loc[i].first);
    }

    sort (loc, loc + N);
    reverse (loc, loc + N);

    int nfar = -1e9;
    for (int i = 0; i < N; i++)
        nfar = max (nfar, loc[i].second);

    int nbest = -1e9;
    for (int i = 0; i < N; i++)
    {
        int cv = loc[i].second;
        nbest = max (nbest, max (nfar, cv) - loc[i].first);

        nfar = min (nfar, cv);
    }

    int lh = -1e9, rl = 1.1e9;
    for (int i = 0; i < N; i++)
    {
        lh = max (lh, loc[i].first);
        rl = min (rl, loc[i].second);
    }
    ans = max (ans, nbest + rl - lh);
    cout << ans << "\n";
}
