#include <bits/stdc++.h>
using namespace std;
const int nsz = 2e3, msz = 1e5;
int n, m, sc, dp[nsz + 5][2], ans;
int sid[nsz + 5], w[nsz + 5], siz[nsz + 5];
map<pair<int, int>, bool> est;
struct graph {
  int sz, hd[nsz + 5], siz[nsz + 5];
  struct edge {
    int nxt, to;
    edge() {}
    edge(int nxt, int to) { this->nxt = nxt, this->to = to; }
  } e[2 * msz + 5];
  inline edge& operator[](int id) { return e[id]; }
  void inline link(int u, int v) {
    e[++sz] = edge(hd[u], v), hd[u] = sz, ++siz[u];
    e[++sz] = edge(hd[v], u), hd[v] = sz, ++siz[v];
  }
};
graph g, tr;
void tarjan(int u, int p = -1) {
  static int t, ss, vis[nsz + 5], stk[nsz + 5], dfn[nsz + 5], low[nsz + 5];
  dfn[u] = low[u] = ++t, vis[u] = 1, stk[ss++] = u;
  for (int i = g.hd[u]; i; i = g[i].nxt) {
    int v = g[i].to;
    if (v == p) continue;
    if (!vis[v])
      tarjan(v, u), low[u] = min(low[u], low[v]);
    else
      low[u] = min(low[u], dfn[v]);
  }
  if (dfn[u] != low[u]) return;
  ++sc;
  for (int v = -1; v != u; --ss) {
    v = stk[ss - 1], vis[v] = 2;
    sid[v] = sc, ++ans;
  }
  --ans;
}
void dfs(int u, int p = -1) {
  siz[u] = 1;
  dp[u][0] = dp[u][1] = 1;
  for (int i = tr.hd[u]; i; i = tr[i].nxt) {
    int v = tr[i].to;
    if (v == p) continue;
    dfs(v, u);
    siz[u] += siz[v];
    dp[u][1] = max(dp[u][1], dp[v][1]);
    dp[u][1] = max(dp[u][1], dp[u][0] + dp[v][0]);
    dp[u][0] = max(dp[u][0], dp[v][0] + 1);
  }
  dp[u][1] = max(dp[u][1], dp[u][0]);
  ans -= tr.siz[u] == 1;
}
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= m; ++i) {
    int u, v;
    scanf("%d%d", &u, &v);
    g.link(u, v);
  }
  for (int u = 1; u <= n; ++u)
    if (!sid[u]) tarjan(u);
  for (int u = 1; u <= n; ++u) {
    int su = sid[u];
    for (int i = g.hd[u]; i; i = g[i].nxt) {
      int v = g[i].to, sv = sid[v];
      if (su == sv || est[make_pair(su, sv)]) continue;
      est[make_pair(su, sv)] = est[make_pair(sv, su)] = 1;
      tr.link(su, sv);
    }
  }
  for (int u = 1; u <= sc; ++u) {
    if (siz[u]) continue;
    dfs(u);
    ans += siz[u];
    if (dp[u][1] > 1) {
      ans -= dp[u][1] - 3;
    }
  }
  --ans;
  printf("%d\n", ans);
}
