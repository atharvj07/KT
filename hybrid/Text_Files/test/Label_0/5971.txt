#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
const int inf = 1e8;

int main(){
	while(1){
		int n,m,s,g1,g2;
		cin >> n >> m >> s >> g1 >> g2;
		if(n==0) break;
		
		vector<vector<int> > adj(n+1, vector<int>(n+1, inf));
		for(int i=1; i<=n; i++){
			adj[i][i] = 0;
		}
		for(int i=0; i<m; i++){
			int f,t,c;
			cin >> f >> t >> c;
			adj[f][t] = c;
		}
		for(int k=1; k<=n; k++){
			for(int i=1; i<=n; i++){
				for(int j=1; j<=n; j++){
					adj[i][j] = min(adj[i][j], adj[i][k]+adj[k][j]);
				}
			}
		}
		
		int ans=inf;
		for(int i=1; i<=n; i++){
			ans = min(ans, adj[s][i]+adj[i][g1]+adj[i][g2]);
		}
		cout << ans << endl;
	}
	return 0;
}
