k1 = 0

def main():
	(n, k) = [int(x) for x in input().split()]
	if k & 1 == 0 or k > 2 * n - 1:
		print('-1')
	else:
		lst = [0] * n
		nonlocal k1
		k1 = k - 1

		def fill_arr(l, r, val):
			nonlocal k1
			if r <= l + 1:
				lst[l] = val
			elif k1 == 0:
				for i in range(l, r):
					lst[i] = val + i - l
			else:
				k1 -= 2
				m = (l + r) // 2
				fill_arr(l, m, val + r - m)
				fill_arr(m, r, val)
		fill_arr(0, n, 1)
		print(' '.join([str(x) for x in lst]))

def __starting_point():
	main()
__starting_point()
