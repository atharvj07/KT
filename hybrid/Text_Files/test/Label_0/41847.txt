from itertools import accumulate

def main():
	n = int(input())
	arr1 = list(accumulate(map(int, input().split())))
	m = int(input())
	arr2 = list(accumulate(map(int, input().split())))
	if arr1[-1] != arr2[-1]:
		print(-1)
		exit(0)
	s1 = set(arr1)
	cnt = 0
	for v in arr2:
		if v in s1:
			cnt += 1
	print(cnt)
main()
