template <typename T>
static T aabs(T x) {
  if (x < 0) return -x;
  return x;
}

template <typename T, typename U>
static void amax(T &x, U y) {
  if (x < y) x = y;
}
template <typename T, typename U>
static void amin(T &x, U y) {
  if (x > y) x = y;
}
template <typename T, typename U>
static void aswap(T &x, U &y) {
  T temp;
  temp = x;
  x = y;
  y = temp;
}
template <typename T, typename U>
static long long gcd(T a, U b) {
  long long r;
  while (b) {
    r = a % b;
    a = b;
    b = r;
  }
  return a;
}
template <typename T, typename U>
static long long lcm(T a, U b) {
  return a / gcd(a, b) * b;
}
template <typename T, typename U>
static long long fpow(T n, U k, int p = MOD) {
  long long r = 1;
  for (; k; k >>= 1) {
    if (k & 1) r = r * n % p;
    n = n * n % p;
  }
  return r;
}
