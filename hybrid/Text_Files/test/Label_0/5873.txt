#include <bits/stdc++.h>
using namespace std;
template <typename R, typename D>
inline void Min(R &a, D b) {
  if (a > b) a = b;
}
template <typename D, typename R>
inline void Max(D &a, R b) {
  if (a < b) a = b;
}
const long long mod = 1e9 + 7;
const int maxn = 300005;
int n, k, a[maxn], dem, lans, rans;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0), cout.tie(0);
  cin >> n >> k;
  for (int i = 1; i <= n; i++) cin >> a[i];
  int l = 1, r = 0;
  while (l <= n) {
    while (dem <= k && r < n) {
      r++;
      if (a[r] == 0) dem++;
      if (dem <= k && l <= r) {
        if (rans - lans <= r - l) rans = r, lans = l;
      }
    }
    while (((dem > k && l <= r) || r == n) && l <= n) {
      if (a[l] == 0) dem--;
      l++;
      if (dem <= k && l <= r) {
        if (rans - lans <= r - l) rans = r, lans = l;
      }
    }
  }
  if (rans == 0 && lans == 0)
    cout << 0 << "\n";
  else
    cout << rans - lans + 1 << "\n";
  for (int i = 1; i <= n; i++)
    cout << ((i >= lans && rans >= i) ? 1 : a[i]) << " ";
}
