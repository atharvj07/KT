#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 7;
const int MAXN = 5e5 + 7;
const int INF = 1e9 + 7;
const int MOD = 1e9 + 7;
int a[maxn];
int b[maxn];
int c[maxn];
int cnt[maxn];
int main() {
  int i, j, n, m, t, z;
  scanf("%d", &n);
  for (i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
    c[a[i]] = i;
  }
  for (i = 1; i <= n; i++) {
    scanf("%d", &b[i]);
    if (c[b[i]] <= i) {
      cnt[n - (i - c[b[i]])]++;
    } else
      cnt[c[b[i]] - i]++;
  }
  int ans = 0;
  for (i = 1; i <= n; i++) {
    ans = max(ans, cnt[i]);
  }
  printf("%d\n", ans);
  return 0;
}
