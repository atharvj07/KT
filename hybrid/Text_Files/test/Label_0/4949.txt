#include <cstdio>
#include <cstring>
#include <algorithm>
#include <cstdlib>
using namespace std;
#define GG puts("FUCK")
#define maxn 750
int map[750][750];
int n,m,a[750],b[750],tot;
int f[750][1550],vis[750];
void dfs(int x,int opt) {
	int i; vis[x]=opt;
	if(!opt) a[tot]++;
	else b[tot]++;
	for(i=1;i<=n;i++) {
		if(map[x][i]) {
			if(vis[i]==-1) {
				dfs(i,opt^1);
			}else if(vis[i]==opt) {
				puts("-1"); exit(0);
			}
		}
	}
}
int main() {
	scanf("%d%d",&n,&m);
	int i,j,x,y;
	memset(vis,-1,sizeof(vis));
	for(i=1;i<=n;i++) for(j=1;j<=n;j++) if(i!=j) map[i][j]=1;
	for(i=1;i<=m;i++) {
		scanf("%d%d",&x,&y); map[x][y]=map[y][x]=0;
	}
	for(i=1;i<=n;i++) {
		if(vis[i]==-1) {
			tot++;
			dfs(i,0);
		}
	}
	f[0][maxn]=1;
	for(i=0;i<tot;i++) {
		int del=a[i+1]-b[i+1];
		for(j=0;j<=maxn+maxn;j++) {
			if(f[i][j]) {
				if(j>=del) f[i+1][j-del]=1;
				if(j+del<=maxn+maxn) f[i+1][j+del]=1;
			}
		}
	}
	int mn=1<<30;
	for(i=maxn;i<=maxn+maxn;i++) {
		if(f[tot][i]) {
			mn=i-maxn; break;
		}
	}
	for(i=maxn;i>=0;i--) {
		if(f[tot][i]) {
			mn=min(mn,maxn-i); break;
		}
	}
	int s=(n+mn)>>1,t=n-s;
	printf("%d\n",s*(s-1)/2+t*(t-1)/2);
}
