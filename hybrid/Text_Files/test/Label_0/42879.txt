from sys import stdin
from sys import setrecursionlimit as SRL
SRL(10 ** 7)
rd = stdin.readline
rrd = lambda : map(int, rd().strip().split())
n = int(input())
a = list(rrd())
bit = [0] * (n + 100)

def ins(x):
	while x <= n:
		bit[x] += 1
		x += x & -x

def get(l):
	tot = 0
	while l:
		tot += bit[l]
		l -= l & -l
	return tot
ta = []
pans = 0
for i in range(len(a)):
	if a[i] > i:
		pans += 1
	ta.append([i + 1, a[i]])
ta.sort(key=lambda x: x[1])
ans = 0
now = 0
for x in ta:
	while now < min(n, x[1]):
		if a[now] <= n:
			ins(a[now])
		now += 1
	ans += now - get(x[0] - 1)
print((ans - pans) // 2)
