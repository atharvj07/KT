import java.util.*;
public class Solution{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String ans = "EASY";
        while(n-->0){
            int d = sc.nextInt();
            if(d !=0){
                ans = "HARD";
                break;
            }
        }
        System.out.println(ans);
    }
}
