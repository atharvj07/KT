import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.Scanner;

public class TwoStrings implements Runnable {
  @Override
  public void run() {
    try {
      BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
      String s = in.readLine();
      String t = in.readLine();

      int[] len = new int[s.length() + 1];
      len[0] = 0;
      for (int i = 1; i <= s.length(); i++) {
        len[i] = len[i - 1];
        if (len[i - 1] < t.length() && s.charAt(i - 1) == t.charAt(len[i - 1])) {
          len[i]++;
        }
      }

      int[][] num = new int[26][t.length() + 1];
      for (int i = 0; i < t.length(); i++) {
        for(int p = 0; p < 26; p++) {
          num[p][i + 1] = num[p][i];
        }
        num[(int)(t.charAt(i) - 'a')][i + 1]++;
      }

      int[] ren = new int[s.length() + 1];
      ren[s.length()] = 0;
      for (int i = s.length() - 1; i >= 0; i--) {
        ren[i] = ren[i + 1];
        if (ren[i + 1] < t.length() && s.charAt(i) == t.charAt(t.length() - 1 - ren[i + 1])) {
          ren[i]++;
        }
      }

      boolean ok = true;
      for (int i = 0; i < s.length(); i++) {
        int k = (int)(s.charAt(i) - 'a');
        int min = Math.max(1, t.length() - ren[i + 1]);
        int max = Math.min(len[i] + 1, t.length());
        //System.out.println(i + ": " + min + " " + max);
        if (min > max || num[k][max] <= num[k][min - 1]) {
          ok = false;
          break;
        }
      }
      if (ok) System.out.println("Yes");
      else System.out.println("No");
    } catch (Exception e) {
      throw new RuntimeException(e);
    }
  }

  public static void main(String[] args) {
    new Thread(new TwoStrings()).start();
  }
}
