import sys
import math

def answer(n, a):
	curr_step = 0
	if n == 1:
		return 0
	mx = a[0]
	mxd = 0
	for i in range(1, n):
		if a[i] > mx:
			mx = a[i]
		d = mx - a[i]
		if d > mxd:
			nsf0 = math.floor(math.log2(d)) + 1
			mxd = 2 ** nsf0 - 1
	return int(math.log2(mxd + 1))

def main():
	t = int(sys.stdin.readline())
	while t:
		n = int(sys.stdin.readline())
		a = tuple(map(int, sys.stdin.readline().split()))
		print(answer(n, a))
		t -= 1
	return
main()
