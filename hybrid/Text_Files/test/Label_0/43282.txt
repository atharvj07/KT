def readdata():
	global n, m, w, q
	(n, m) = [int(x) for x in input().split()]
	q = [0] * m
	w = [0] * m
	for i in range(m):
		(q[i], w[i]) = [int(x) for x in input().split()]

def podg():
	global summ
	w.sort(reverse=True)
	summ = [w[0]] * m
	for i in range(1, m):
		summ[i] = summ[i - 1] + w[i]

def may(k):
	if k % 2 == 1:
		return n > k * (k - 1) // 2
	return n > k * (k - 1) // 2 + (k - 2) // 2

def solve():
	for i in range(1, m + 1):
		if may(i) and (not may(i + 1)):
			print(summ[i - 1])
			return
	print(sum(w))
readdata()
podg()
solve()
