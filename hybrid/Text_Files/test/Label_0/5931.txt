#include <bits/stdc++.h>
using namespace std;

#define endl '\n'
#define int long long

int32_t main()
{
	ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);

	int tests; cin >> tests;
	while(tests --> 0) {
		int n; cin >> n;
		vector<int> vec(n);
		unordered_map<int, int> left, right;
		for(int &i : vec) cin >> i;
		for(int i = n - 1; i > 1; i --) 
			right[vec[i]] ++;
		int ans = 0;
		left[vec[0]] ++;
		for(int i = 1; i < n - 1; i ++) {
			ans += (left[vec[i]] * right[vec[i]]);
			for(int j = -2; j < 3; j ++) {
				for(int k = - 2; k < 3; k ++) {
					if(j == k and k == 0) continue;
					if(abs(j - k) > 2) continue;
					ans += (left[vec[i] + j] * right[vec[i] + k]);
				}
			}
			left[vec[i]] ++;
			right[vec[i + 1]] --;
		}
		cout << ans << endl;
	}

	return 0;
}
