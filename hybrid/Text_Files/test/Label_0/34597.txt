import static java.util.Arrays.deepToString;

import java.io.*;
import java.math.*;
import java.util.*;

public class D {

	static int solve(int k, long l, long r, int p) {
		if (p == 2)
			return k % 2 == 0 ? 1 : 0;
		if (k == 1)
			return 2;
		long res = k % 2 == 0 ? 1 : powFast(2, l - r, p);
		long mult = (powTwo(k, l, p) + p - 1) % p;
		if (mult != 0) {
			res = (res * (powTwo(k, r + 1, p) + p - 1)) % p;
			res = (res * (inverse(mult, p))) % p;
		} else {
			res = (res * powFast(2, r - l + 1, p)) % p;
		}
		return (int) res;
	}

	static long inverse(long a, long p) {
		return powFast(a, p - 2, p);
	}

	static long powFast(long a, long n, long p) {
		if (n < 0) {
			return inverse(powFast(a, -n, p), p);
		}
		if (n == 0)
			return 1;
		long t = powFast(a, n / 2, p);
		t = (t * t) % p;
		return n % 2 == 0 ? t : (t * a) % p;
	}

	static long powTwo(long a, long b, long p) {
		return a % p == 0 ? 0 : powFast(a, powFast(2, b, p - 1), p);
	}

	static int brute(int k, long l, long r, int p) {
		BigInteger LCM = BigInteger.ONE;
		for (long i = l; i <= r; i++) {
			BigInteger cur = val(k).pow(1 << i).add(BigInteger.ONE);
			LCM = lcm(LCM, cur);
		}
		return LCM.mod(val(p)).intValue();
	}

	static BigInteger lcm(BigInteger a, BigInteger b) {
		return a.multiply(b).divide(a.gcd(b));
	}

	static void maxTest() {
		int p = 1000000007;
		long MX = (long) 1e18;
		Random rng = new Random();
		long l = (rng.nextLong() % MX + MX) % MX, r = (rng.nextLong() % MX + MX) % MX;
		for (int it = 0; it < 100000; it++) {
			if (l > r) {
				long tmp = l;
				l = r;
				r = tmp;
			}
			int res = solve(199999, l, r, p);
			debug(199999, l, r, p, res);
		}
	}

	public static void main(String[] args) throws Exception {
		reader = new BufferedReader(new InputStreamReader(System.in));
		writer = new PrintWriter(System.out);

		int T = nextInt();
		for (int i = 0; i < T; i++) {
			int k = nextInt();
			long L = nextLong(), R = nextLong();
			int p = nextInt();
			int res = solve(k, L, R, p);
			writer.println(res);
		}

		writer.close();
	}

	static BufferedReader reader;
	static PrintWriter writer;
	static StringTokenizer tok = new StringTokenizer("");
	static long systemTime;

	static void debug(Object... o) {
		System.err.println(deepToString(o));
	}

	static void setTime() {
		systemTime = System.currentTimeMillis();
	}

	static void printTime() {
		System.err.println("Time consumed: " + (System.currentTimeMillis() - systemTime));
	}

	static void printMemory() {
		System.err.println("Memory consumed: " + (Runtime.getRuntime().totalMemory() - Runtime.getRuntime().freeMemory()) / 1000 + "kb");
	}

	static String next() {
		while (!tok.hasMoreTokens()) {
			String w = null;
			try {
				w = reader.readLine();
			} catch (Exception e) {
				e.printStackTrace();
			}
			if (w == null)
				return null;
			tok = new StringTokenizer(w);
		}
		return tok.nextToken();
	}

	static int nextInt() {
		return Integer.parseInt(next());
	}

	static long nextLong() {
		return Long.parseLong(next());
	}

	static double nextDouble() {
		return Double.parseDouble(next());
	}

	static BigInteger nextBigInteger() {
		return new BigInteger(next());
	}

	static BigInteger val(long x) {
		return BigInteger.valueOf(x);
	}
}
