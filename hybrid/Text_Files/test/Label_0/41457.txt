from collections import Counter
x = int(input())
k = list(map(int, input().split()))
if len(k) == 1:
	print('YES')
elif Counter(k).most_common(1)[0][1] <= len(k) // 2 and len(k) > 1 and (len(k) % 2 == 0):
	print('YES')
elif Counter(k).most_common(1)[0][1] <= len(k) // 2 + 1 and len(k) > 1 and (len(k) % 2 == 1):
	print('YES')
else:
	print('NO')
