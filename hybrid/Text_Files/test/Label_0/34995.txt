import java.io.PrintWriter;
import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class D {

    private static final int[] DELTAS = {-1, +1};

    private static int n, m;
    private static int[] students;

    private static boolean dfs(int i) {
        if (i == n * m)
            return true;

        int x = i / m;
        int y = i % m;

        for (int j = i; j < n * m; j++) {
            swap(i, j);

            Set<Integer> neighbors = new HashSet<>();
            for (int delta : DELTAS) {
                neighbors.add(students[i] + delta);
                neighbors.add(students[i] + delta*m);
            }

            if (
                (x == 0 || !neighbors.contains(students[(x-1)*m + y])) &&
                (y == 0 || !neighbors.contains(students[x*m + y-1]))
            ) {
                if (dfs(i + 1))
                    return true;
            }

            swap(j, i);
        }

        return false;
    }

    private static void swap(int i, int j) {
        int tmp = students[i];
        students[i] = students[j];
        students[j] = tmp;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);

        n = in.nextInt();
        m = in.nextInt();

        students = new int[n * m];
        for (int i = 0; i < n * m; i++)
            students[i] = i;

        if (dfs(0)) {
            out.println("YES");
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++)
                    out.print((students[i*m + j] + 1) + " ");
                out.println();
            }
            out.flush();
        } else
            System.out.println("NO");
    }

}
