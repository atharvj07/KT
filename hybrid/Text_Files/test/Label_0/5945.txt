#include <bits/stdc++.h>
using namespace std;
const int MOD = 1e9 + 7, N = 2e6 + 10;
int a[N];
void solve() {
  int n, k = 0;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i < n; i++) {
    if (a[i] <= a[i + 1]) {
      cout << "YES" << endl;
      return;
    }
  }
  cout << "NO" << endl;
}
int main() {
  int t = 1;
  cin >> t;
  while (t--) {
    solve();
  }
}
