import java.io.File;
import java.io.PrintWriter;
import java.util.Scanner;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();

        boolean found = false;
        int ans = 0;
        for (int i = 0; Math.pow(n, i) <= m; ++i) {
            if (Math.pow(n, i) == m) {
                found = true;
                ans = i;
                break;
            }
        }
        if (!found)
            System.out.println("NO");
        else {
            System.out.println("YES");
            System.out.println(ans-1);
        }
    }

}
