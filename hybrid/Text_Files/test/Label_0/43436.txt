def nextt(s):
	return chr(ord(s) + 1)
(n, k) = map(int, input().split())
s = [i for i in input()]
i = n - 1
s[i] = nextt(s[i])
while i >= 0 and i < n:
	if ord(s[i]) >= 97 + k:
		s[i] = 'a'
		i -= 1
		s[i] = nextt(s[i])
	elif i > 0 and s[i] == s[i - 1] or (i > 1 and s[i] == s[i - 2]):
		s[i] = nextt(s[i])
	else:
		i += 1
if i < 0:
	print('NO')
else:
	print(''.join(s))
