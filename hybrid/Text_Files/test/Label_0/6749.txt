#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int c1 = 0, c2 = 0;
  string s1, s2;
  cin >> s1 >> s2;
  for (int i = 0; i < s1.size(); i += 2) {
    if (s1[i] == '[' && s2[i] == '8') {
      c2++;
      continue;
    }
    if (s1[i] == '[' && s2[i] == '(') {
      c1++;
      continue;
    }
    if (s1[i] == '8' && s2[i] == '(') {
      c2++;
      continue;
    }
    if (s1[i] == '8' && s2[i] == '[') {
      c1++;
      continue;
    }
    if (s1[i] == '(' && s2[i] == '[') {
      c2++;
      continue;
    }
    if (s1[i] == '(' && s2[i] == '8') {
      c1++;
      continue;
    }
  }
  if (c1 > c2) {
    cout << "TEAM 1 WINS";
  }
  if (c2 > c1) {
    cout << "TEAM 2 WINS";
  } else if (c1 == c2) {
    cout << "TIE";
  }
  return 0;
}
