(n, m, k) = map(int, input().split())
s0 = list(map(int, input().split()))
launch = list(map(int, input().split()))
sequence = [[0, 0, 0] for i in range(n + 1)]
for i in range(n):
	if i == 0:
		sequence[s0[i]][0] = i + 1
		sequence[s0[i]][1] = 0
		sequence[s0[i]][2] = s0[i + 1]
	elif i == n - 1:
		sequence[s0[i]][0] = i + 1
		sequence[s0[i]][1] = s0[i - 1]
		sequence[s0[i]][2] = 0
	else:
		sequence[s0[i]][0] = i + 1
		sequence[s0[i]][1] = s0[i - 1]
		sequence[s0[i]][2] = s0[i + 1]
gestures = 0
for a in launch:
	gestures += (sequence[a][0] - 1) // k + 1
	b = sequence[a][1]
	d = sequence[a][2]
	c = sequence[b][1]
	if sequence[a][0] != 1:
		sequence[a][0] -= 1
		sequence[b][0] += 1
		sequence[a][1] = c
		sequence[c][2] = a
		sequence[b][2] = d
		sequence[d][1] = b
		sequence[a][2] = b
		sequence[b][1] = a
print(gestures)
