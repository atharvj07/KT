#include <bits/stdc++.h>
using namespace std;

  if (_a < _b) _a = _b;
}
template <typename T>
void mymin(T &_a, T _b) {
  if (_a > _b) _a = _b;
}
void print(int _x) { printf("%d\n", _x); }
template <typename T>
void print(T _data[], int _size) {
  for (int i = 1; i <= _size; i++) cout << _data[i] << " ";
  cout << endl;
}
int n;
int yuan[1010];
map<pair<double, double>, bool> mpa;
int main() {
  while (scanf("%d", &n) != EOF) {
    for (int i = 1; i <= n; i++) scanf("%d", &yuan[i]);
    if (n == 1) {
      printf("No\n");
      continue;
    }
    if (n == 2) {
      printf("Yes\n");
      continue;
    }
    mpa.clear();
    bool ok = 0;
    for (int i = 1; i < n; i++) {
      for (int j = i + 1; j <= n; j++) {
        double nowk = (double)(yuan[j] - yuan[i]) / (double)(j - i);
        double nowb = (double)yuan[j] - j * nowk;
        if (mpa.count(make_pair(nowk, nowb))) continue;
        mpa[make_pair(nowk, nowb)] = 1;
        bool fst = 0;
        double dis = (double)(2000000001);
        bool chg = 0;
        for (int k = 1; k <= n; k++) {
          if (k == i || k == j) continue;
          double nowdis = (double)yuan[k] - (double)k * nowk - nowb;
          if (!nowdis) continue;
          if (nowdis == dis) continue;
          if (dis != ((int)(2e9) + 1)) {
            fst = 1;
            break;
          }
          chg = 1;
          dis = nowdis;
        }
        if (!fst && chg) {
          ok = 1;
          break;
        }
      }
      if (ok) break;
    }
    if (ok)
      printf("Yes\n");
    else
      printf("No\n");
  }
  return 0;
}
