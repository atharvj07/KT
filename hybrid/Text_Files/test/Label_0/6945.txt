#include <bits/stdc++.h>
using namespace std;
typedef long long int ll;
typedef pair<ll,ll> P;

int main() {
    int N, T, L, B;
    while(cin >> N >> T >> L >> B, N){
        vector<int> masu(N+1);
        for(int i=0;i<L;i++){
            int x;
            cin >> x;
            masu[x] = 1;
        }
        for(int i=0;i<B;i++){
            int x;
            cin >> x;
            masu[x] = 2;
        }
        vector<vector<double>> dp(T+2, vector<double>(N+1,0));
        dp[0][0] = 1;
        for(int i=0;i<T;i++){
            for(int j=0;j<N;j++){
                for(int k=1;k<=6;k++){
                    int nxt = N-abs(N-(j+k));
                    if(masu[nxt] == 0){
                        dp[i+1][nxt] += dp[i][j]/6;
                    }
                    else if(masu[nxt] == 1){
                        dp[i+2][nxt] += dp[i][j]/6;
                    }
                    else{
                        dp[i+1][0] += dp[i][j]/6;
                    }
                }
            }
        }
        double ans = 0;
        for(int i=0;i<=T;i++){
            ans += dp[i][N];
        }
        printf("%.10lf\n",ans);
    }
    return 0;
}
