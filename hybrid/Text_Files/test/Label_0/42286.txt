t = int(input())
for _ in range(t):
	(n, k) = map(int, input().split())
	check = [1 for i in range(n - 1)]
	extra = 2
	for i in range(1, n - 1):
		check[i] = check[i - 1] + extra
		extra += 1
	l = 0
	r = n - 1
	ans = -1
	while l <= r:
		mid = l + (r - l) // 2
		if check[mid] >= k:
			ans = mid + 1
			r = mid - 1
		else:
			l = mid + 1
	if k == 1:
		print((n - 2) * 'a' + 'bb')
	else:
		diff = k - check[ans - 2]
		between = ans - diff
		final = (n - 1 - ans) * 'a' + 'b' + between * 'a' + 'b' + (diff - 1) * 'a'
		print(final)
