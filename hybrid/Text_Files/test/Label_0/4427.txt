#include <bits/stdc++.h>
using namespace std;
int main() {
  int l = 0, r = (1 << 29);
  int n;
  cin >> n;
  string str;
  for (int i = 0; i < min(29, n); i++) {
    cout << (l + r) / 2 << " " << 0 << endl;
    cin >> str;
    if (str == "black") {
      r = (l + r) / 2;
    } else {
      l = (l + r) / 2;
    }
  }
  if (n == 30) {
    cout << r << " " << 1 << endl;
    cin >> str;
    if (str == "black") {
      cout << r << " " << 2 << " " << r + 1 << " " << 6 << endl;
    } else {
      cout << r + 1 << " " << 1 << " " << r + 4 << " " << 3 << endl;
    }
  } else {
    cout << r << " " << 1 << " " << r + 1 << " " << 3 << endl;
  }
  return 0;
}
