import java.io.BufferedInputStream;
import java.io.Closeable;
import java.io.IOException;

/**
 * Created by Exsequi on 2016/9/7 0007.
 */
public class CF706C implements Runnable {
    public void solve(Reader in) throws IOException {
        int n = in.nextInt();
        long cost[] = new long[n];
        State oldState[] = new State[]{new State(), null};
        State newState[][] = new State[2][2];
        for (int i = 0; i < n; i++) {
            cost[i] = in.nextLong();
        }
        for (int i = 0; i < n; i++) {
            String string = in.next();
            for (int j = 0; j < 2; j++) {
                if (oldState[j] == null) {
                    newState[j][0] = null;
                    newState[j][1] = null;
                } else {
                    newState[j][0] = oldState[j].next(string, 0);
                    newState[j][1] = oldState[j].next(new StringBuilder(string).reverse().toString(), cost[i]);
                }
            }
            for (int j = 0; j < 2; j++) {
                if (newState[0][j] == null) {
                    oldState[j] = newState[1][j];
                } else if (newState[1][j] == null) {
                    oldState[j] = newState[0][j];
                } else if (newState[0][j].compareTo(newState[1][j]) <= 0) {
                    oldState[j] = newState[0][j];
                } else {
                    oldState[j] = newState[1][j];
                }
            }
            if (oldState[0] == null && oldState[1] == null) {
                break;
            }
        }
        if (oldState[0] == null && oldState[1] == null) {
            System.out.println("-1");
        } else if (oldState[0] == null) {
            System.out.println(oldState[1].cost);
        } else if (oldState[1] == null) {
            System.out.println(oldState[0].cost);
        } else if (oldState[0].compareTo(oldState[1]) <= 0) {
            System.out.println(oldState[0].cost);
        } else {
            System.out.println(oldState[1].cost);
        }
    }

    class State {
        String string;
        long cost;

        State() {
            string = "";
            cost = 0;
        }

        State(String string, long cost) {
            this.string = string;
            this.cost = cost;
        }

        State next(String string, long cost) {
            if (string.compareTo(this.string) >= 0) {
                return new State(string, this.cost + cost);
            } else {
                return null;
            }
        }

        int compareTo(State anotherState) {
            if (cost != anotherState.cost)
                return cost - anotherState.cost > 0 ? 1 : -1;
            else
                return string.compareTo(anotherState.string);
        }
    }

    public void run() {
        try (Reader in = new Reader()) {
            solve(in);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    @SuppressWarnings({"WeakerAccess", "unused"})
    private class Reader implements Closeable {
        private BufferedInputStream in;
        private int c;

        public Reader() throws IOException {
            in = new BufferedInputStream(System.in);
            c = in.read();
        }

        public boolean eof() throws IOException {
            skipWhiteSpace();
            return c == -1;
        }

        public int nextInt() throws IOException {
            int result = 0;
            boolean minus;
            skipWhiteSpace();
            if (c == '-') {
                minus = true;
                c = in.read();
            } else {
                minus = false;
            }
            while (c >= '0' && c <= '9') {
                result = result * 10 + (c - '0');
                c = in.read();
            }
            return minus ? -result : result;
        }

        public long nextLong() throws IOException {
            long result = 0;
            boolean minus;
            skipWhiteSpace();
            if (c == '-') {
                minus = true;
                c = in.read();
            } else {
                minus = false;
            }
            while (c >= '0' && c <= '9') {
                result = result * 10 + (c - '0');
                c = in.read();
            }
            return minus ? -result : result;
        }

        public String next() throws IOException {
            return next(33, 127);
        }

        public String next(int low, int high) throws IOException {
            StringBuilder builder = new StringBuilder();
            skipWhiteSpace();
            while (c >= low && c <= high) {
                builder.append((char) c);
                c = in.read();
            }
            if (builder.length() == 0)
                throw new IOException("This is not a String.");
            return builder.toString();
        }

        public int nextChar() throws IOException {
            int result = c;
            c = in.read();
            return result;
        }

        public void skipWhiteSpace() throws IOException {
            while (c == ' ' || c == '\r' || c == '\n')
                c = in.read();
        }

        @Override
        public void close() throws IOException {
            in.close();
        }
    }

    public static void main(String[] args) throws Exception {
        String className = Thread.currentThread().getStackTrace()[1].getClassName();
        new Thread((Runnable) Class.forName(className).newInstance()).start();
    }
}
