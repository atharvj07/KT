n = int(input())
arr = list(map(int, input().split()))
forward = arr[:]
cnt = arr[0] + 1
for i in range(1, n):
	forward[i] = max(cnt, arr[i])
	cnt = max(cnt, arr[i]) + 1
backward = arr[:]
cnt = arr[n - 1] + 1
for i in range(n - 2, -1, -1):
	backward[i] = max(cnt, arr[i])
	cnt = max(cnt, arr[i]) + 1
ans = 0
for i in range(n):
	if forward[i] < backward[i]:
		ans += forward[i] - arr[i]
	else:
		if abs(backward[i] - forward[i]) == 1:
			ans += 1
		ans += backward[i] - arr[i]
print(ans)
