import java.io.*;
import java.util.*;
import java.util.Map.Entry;
import java.math.*;
import java.awt.geom.*;

import static java.lang.Math.*;

public class Solution implements Runnable {
	
	static final int MOD = 1000000007;
	
	class State {
		
		int boat, c50, c100;
		
		public State(int boat, int c50, int c100) {
			this.boat = boat;
			this.c50 = c50;
			this.c100 = c100;
		}
		
	}
	
	public void solve() throws Throwable {
		int n = sc.nextInt();
		int w = sc.nextInt();
		int count50 = 0, count100 = 0;
		for (int i = 0; i < n; i++) {
			int num = sc.nextInt();
			if (num == 50)
				count50++;
			else
				count100++;
		}
		
		int dist [][][] = new int [2][count50 + 1][count100 + 1];
		for (int i = 0; i < 2; i++)
			for (int j = 0; j <= count50; j++)
				for (int k = 0; k <= count100; k++)
					dist[i][j][k] = Integer.MAX_VALUE / 2;
		dist[0][0][0] = 0;
		ArrayDeque<State> q = new ArrayDeque<>();
		q.add(new State(0, 0, 0));
		while (!q.isEmpty()) {
			State cur = q.poll();
			int boat = cur.boat;
			int c50 = cur.c50;
			int c100 = cur.c100;
			if (cur.boat == 0) {
				for (int to50 = 0; to50 <= count50 - c50 && 50 * to50 <= w; to50++)
					for (int to100 = 0; to100 <= count100 - c100 && 50 * to50 + 100 * to100 <= w; to100++) {
						if (to50 == 0 && to100 == 0) continue;
						if (dist[boat ^ 1][c50 + to50][c100 + to100] > dist[boat][c50][c100] + 1) {
							dist[boat ^ 1][c50 + to50][c100 + to100] = dist[boat][c50][c100] + 1;
							q.add(new State(boat ^ 1, c50 + to50, c100 + to100));
						}
					}
			} else {
				for (int to50 = 0; to50 <= c50 && 50 * to50 <= w; to50++)
					for (int to100 = 0; to100 <= c100 && 50 * to50 + 100 * to100 <= w; to100++) {
						if (to50 == 0 && to100 == 0) continue;
						if (dist[boat ^ 1][c50 - to50][c100 - to100] > dist[boat][c50][c100] + 1) {
							dist[boat ^ 1][c50 - to50][c100 - to100] = dist[boat][c50][c100] + 1;
							q.add(new State(boat ^ 1, c50 - to50, c100 - to100));
						}
					}
			}
		}
		if (dist[1][count50][count100] == Integer.MAX_VALUE / 2) {
			out.println(-1);
			out.println(0);
			return;
		}
		out.println(dist[1][count50][count100]);
		
		long c [][] = new long [n + 1][n + 1];
		for (int i = 0; i <= n; i++) {
			c[i][0] = c[i][i] = 1;
			for (int j = 1; j < i; j++)
				c[i][j] = (c[i - 1][j] + c[i - 1][j - 1]) % MOD;
		}
		
		int d = dist[1][count50][count100];
		long sum[][][] = new long[2][count50 + 1][count100 + 1];
		sum[1][count50][count100] = 1;
		for (int distance = d; distance > 0; distance--) {
			// boat 0 
			for (int c50 = 0; c50 <= count50; c50++)
				for (int c100 = 0; c100 <= count100; c100++)
					if (dist[0][c50][c100] == distance) {
						for (int to50 = 0; to50 <= count50 - c50 && 50 * to50 <= w; to50++)
							for (int to100 = 0; to100 <= count100 - c100 && 50 * to50 + 100 * to100 <= w; to100++) {
								if (to50 == 0 && to100 == 0) continue;
								if (dist[1][c50 + to50][c100 + to100] == dist[0][c50][c100] - 1) {
									long multiplier = (c[count50 - c50][to50] * c[count100 - c100][to100]) % MOD;
									multiplier = (multiplier * sum[0][c50][c100]) % MOD;
									sum[1][c50 + to50][c100 + to100] += multiplier;
									sum[1][c50 + to50][c100 + to100] %= MOD;
								}
							}
					}
			for (int c50 = 0; c50 <= count50; c50++)
				for (int c100 = 0; c100 <= count100; c100++)
					if (dist[1][c50][c100] == distance) {
						for (int to50 = 0; to50 <= c50 && 50 * to50 <= w; to50++)
							for (int to100 = 0; to100 <= c100 && 50 * to50 + 100 * to100 <= w; to100++) {
								if (to50 == 0 && to100 == 0) continue;
								if (dist[0][c50 - to50][c100 - to100] == dist[1][c50][c100] - 1) {
									long multiplier = (c[c50][to50] * c[c100][to100]) % MOD;
									multiplier = (multiplier * sum[1][c50][c100]) % MOD;
									sum[0][c50 - to50][c100 - to100] += multiplier;
									sum[0][c50 - to50][c100 - to100] %= MOD;
								}
							}
					}
				
		}
		out.println(sum[0][0][0]);
	}

	BufferedReader in;
	PrintWriter out;
	FastScanner sc;

	static Throwable uncaught;

	@Override
	public void run() {
		try {
			in = new BufferedReader(new InputStreamReader(System.in));
			out = new PrintWriter(System.out);
			sc = new FastScanner(in);
			solve();
		} catch (Throwable t) {
			Solution.uncaught = t;
		} finally {
			out.close();
		}
	}

	public static void main(String[] args) throws Throwable {
		Thread t = new Thread(null, new Solution(), "", (1 << 26));
		t.start();
		t.join();
		if (uncaught != null) {
			throw uncaught;
		}
	}

}

class FastScanner {

	BufferedReader reader;
	StringTokenizer strTok;

	public FastScanner(BufferedReader reader) {
		this.reader = reader;
	}

	public String nextToken() throws IOException {
		while (strTok == null || !strTok.hasMoreTokens()) {
			strTok = new StringTokenizer(reader.readLine().trim());
		}
		return strTok.nextToken();
	}

	public int nextInt() throws IOException {
		return Integer.parseInt(nextToken());
	}

	public long nextLong() throws IOException {
		return Long.parseLong(nextToken());
	}

	public double nextDouble() throws IOException {
		return Double.parseDouble(nextToken());
	}

}

class Treap {
	
	static Random rnd = new Random();
	
	long x, sum, pairSum;
	int prior, cnt;
	Treap l, r;
	
	public Treap(long x) {
		this.x = x;
		this.sum = x;
		this.pairSum = 0;
		this.cnt = 1;
		this.prior = rnd.nextInt();
	}
	
	public static long getSum (Treap t) {
		return t == null ? 0 : t.sum;
	}
	
	public static long getPairSum (Treap t) {
		return t == null ? 0 : t.pairSum;
	}
	
	public static int getCount (Treap t) {
		return t == null ? 0 : t.cnt;
	}
	
	public static void updateVertex (Treap t) {
		if (t == null) return;
		t.cnt = getCount(t.l) + getCount(t.r) + 1;
		t.sum = getSum(t.l) + getSum(t.r) + t.x;
		t.pairSum = getPairSum(t.l) + getPairSum(t.r);
		if (t.l != null)
			t.pairSum += t.x * getCount(t.l) - getSum(t.l);
		if (t.r != null)
			t.pairSum += getSum(t.r) - t.x * getCount(t.r);
		if (t.l != null && t.r != null)
			t.pairSum += getSum(t.r) * getCount(t.l) - getSum(t.l) * getCount(t.r);
	}
	
	public static Treap merge (Treap l, Treap r) {
		if (l == null) return r;
		if (r == null) return l;
		if (l.prior > r.prior) {
			l.r = merge (l.r, r);
			updateVertex(l);
			return l;
		} else {
			r.l = merge (l, r.l);
			updateVertex(r);
			return r;
		}
	}
	
	public static Treap [] split (Treap t, long key) {
		if (t == null) return new Treap [] {null, null};
		Treap[] res = null;
		if (key <= t.x) {
			res = split (t.l, key);
			t.l = res[1];
			res[1] = t;
		} else {
			res = split (t.r, key);
			t.r = res[0];
			res[0] = t;
		}
		updateVertex(res[0]);
		updateVertex(res[1]);
		return res;
	}
	
	public static void dfs (Treap t, StringBuilder sb) {
		if (t == null) return;
		dfs (t.l, sb);
		sb.append(t.x);
		sb.append(" ");
		dfs (t.r, sb);
	}
	
}
