n = int(input())
A = []
for i in range(n):
	(a, b) = map(int, input().split())
	A.append((b, a))
A.sort()
l = 0
r = n - 1
c = 0
res = 0
while l <= r:
	if c < A[l][0]:
		x = A[l][0] - c
		if x < A[r][1]:
			res += x * 2
			c += x
			A[r] = (A[r][0], A[r][1] - x)
		else:
			res += A[r][1] * 2
			c += A[r][1]
			r -= 1
	else:
		res += A[l][1]
		c += A[l][1]
		l += 1
print(res)
