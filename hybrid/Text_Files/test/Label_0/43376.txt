n = int(input())
p = list(map(int, input().split()))
slope = 1
maxi = 0
count = 0
end = {}
for i in range(1, n):
	if p[i] > p[i - 1]:
		slope += 1
	else:
		if slope > maxi:
			maxi = slope
			count = 1
			end = {i - 1: 1}
		elif slope == maxi:
			count += 1
		slope = 1
if slope > maxi:
	maxi = slope
	count = 1
	end = {i - 1: 1}
elif slope == maxi:
	count += 1
slope = 1
f = 0
for i in range(n - 2, -1, -1):
	if p[i] > p[i + 1]:
		slope += 1
	else:
		if slope > maxi:
			maxi = slope
			count = 1
		elif slope == maxi:
			count += 1
			if i + 1 in end:
				f = 1
		slope = 1
if slope > maxi:
	maxi = slope
	count = 1
elif slope == maxi:
	count += 1
	if i + 1 in end:
		f = 1
if count == 2:
	if f == 1:
		if maxi % 2 == 1:
			print(1)
		else:
			print(0)
	else:
		print(0)
else:
	print(0)
