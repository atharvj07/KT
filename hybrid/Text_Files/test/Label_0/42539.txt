import sys
read = lambda : map(int, sys.stdin.buffer.readline().split())
from math import *
(n, q) = read()
(x, y) = ([], [])
(cx, cy) = (0, 0)
deg = 0
(p1, p2) = (0, 1)

def pos(i):
	return (cx + x[i] * cos(deg) - y[i] * sin(deg), cy + y[i] * cos(deg) + x[i] * sin(deg))
for i in range(n):
	(_x, _y) = read()
	x.append(_x)
	y.append(_y)
s = 0
for i in range(2, n):
	_s = (x[i] - x[0]) * (y[i - 1] - y[0]) - (y[i] - y[0]) * (x[i - 1] - x[0])
	cx += (x[0] + x[i] + x[i - 1]) / 3 * _s
	cy += (y[0] + y[i] + y[i - 1]) / 3 * _s
	s += _s
cx /= s
cy /= s
for i in range(n):
	x[i] -= cx
	y[i] -= cy
for i in range(q):
	v = list(read())
	if v[0] == 1:
		if p1 == v[1] - 1:
			(p1, p2) = (p2, None)
		else:
			(p1, p2) = (p1, None)
		(cx, cy) = pos(p1)
		cy -= hypot(x[p1], y[p1])
		deg = pi / 2 - atan2(y[p1], x[p1])
		p2 = v[2] - 1
	else:
		print('%.15f %.15f' % pos(v[1] - 1))
