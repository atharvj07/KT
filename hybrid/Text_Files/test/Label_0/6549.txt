#include <bits/stdc++.h>
using namespace std;
bool cw(const pair<long long, long long> &a,
        const pair<long long, long long> &b,
        const pair<long long, long long> &c) {
  return (b.first - a.first) * (c.second - a.second) -
             (b.second - a.second) * (c.first - a.first) <
         0;
}
vector<pair<long long, long long> > convexHull(
    vector<pair<long long, long long> > p) {
  int n = p.size();
  if (n <= 1) return p;
  sort(p.begin(), p.end());
  int cnt = 0;
  vector<pair<long long, long long> > q(n * 2);
  for (int i = 0; i < n; q[cnt++] = p[i++])
    for (; cnt >= 2 && !cw(q[cnt - 2], q[cnt - 1], p[i]); --cnt)
      ;
  for (int i = n - 2, t = cnt; i >= 0; q[cnt++] = p[i--])
    for (; cnt > t && !cw(q[cnt - 2], q[cnt - 1], p[i]); --cnt)
      ;
  q.resize(cnt - 1 - (q[0] == q[1]));
  return q;
}
vector<pair<long long, long long> > vc, conv;
int main() {
  int n;
  double p, q;
  cin >> n >> p >> q;
  int mx = 0, my = 0;
  for (int i = 0; i < n; i++) {
    int x, y;
    cin >> x >> y;
    vc.push_back(make_pair(x, y));
    mx = max(mx, x);
    my = max(my, y);
  }
  vc.push_back(make_pair(0, 0));
  vc.push_back(make_pair(0, my));
  vc.push_back(make_pair(mx, 0));
  conv = convexHull(vc);
  int len = conv.size();
  double lo = 0, hi = 4e6;
  int ts = 400;
  while (ts--) {
    double mid = (lo + hi) / 2.0;
    int fl = 0;
    double tmp = (double)conv[len - 1].first * mid;
    if (tmp < p - 1e-12) fl = 1;
    tmp = (double)conv[1].second * mid;
    if (tmp < q - 1e-12) fl = 1;
    if (fl == 0) {
      fl = 1;
      for (int i = 2; i < len; i++) {
        double px = (double)conv[i - 1].first * mid;
        double py = (double)conv[i - 1].second * mid;
        double qx = (double)conv[i].first * mid;
        double qy = (double)conv[i].second * mid;
        if (px > p + 1e-12) break;
        if (qx < p - 1e-12) continue;
        if (abs(qx - p) < 1e-12 && qy > q - 1e-12) {
          fl = 0;
          break;
        }
        if (abs(qx - p) < 1e-12) {
          break;
        }
        double y = (p - qx) * (qy - py) / (qx - px);
        y += qy;
        if (y > q - 1e-12) {
          fl = 0;
          break;
        }
      }
    }
    if (fl == 0)
      hi = mid;
    else
      lo = mid;
  }
  cout << fixed << setprecision(10) << lo << endl;
}
