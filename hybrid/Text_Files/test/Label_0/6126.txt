#include <bits/stdc++.h>
using namespace std;
int main() {
  int t, len;
  bool check = true;
  string a;
  string b = "abc";
  cin >> t;
  for (; t; t--) {
    cin >> a;
    len = a.size();
    for (int i = 0; i < len; i++) {
      if (a[i] == '?') {
        for (int j = 0; j < 3; j++) {
          if ((a[i - 1] != b[j]) && (a[i + 1] != b[j])) {
            a[i] = b[j];
            break;
          }
        }
      }
    }
    for (int k = 0; k < len - 1; k++) {
      if (a[k] == '?' || a[k] == a[k + 1]) {
        check = false;
        cout << -1;
        break;
      }
    }
    if (check) {
      for (int k = 0; k < len; k++) cout << a[k];
    }
    cout << endl;
    check = true;
  }
  return 0;
}
