#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;
int n, x;
bool solve(int l, int r, vector<int> p, vector<int> q) {
	if (q.size() == 0) return false;
	int cnt = 0; bool ret = false;
	for (int i = 0; i < p.size(); i++) {
		if (p[i] == l - 1) {
			vector<int> p2 = p;
			p2.erase(p2.begin() + i);
			if (!solve(l - 1, r, q, p2)) ret = true;
			cnt++;
		}
		if (p[i] == r) {
			vector<int> p2 = p;
			p2.erase(p2.begin() + i);
			if (!solve(l, r + 1, q, p2)) ret = true;
			cnt++;
		}
	}
	if (cnt) return ret;
	return !solve(l, r, q, p);
}
int main() {
	cin >> n;
	while (n--) {
		vector<int> x(6), y;
		for (int i = 0; i < 6; i++) cin >> x[i];
		for (int i = 1; i <= 13; i++) {
			if (i != 7 && find(x.begin(), x.end(), i) == x.end()) y.push_back(i);
		}
		cout << (solve(7, 8, x, y) ? "yes" : "no") << endl;
	}
	return 0;
}
