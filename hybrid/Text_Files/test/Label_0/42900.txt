from collections import defaultdict
from collections import Counter
import sys
(r, c) = map(int, input().split())
m1 = []
m2 = []
for i in range(r):
	a = list(map(int, input().split()))
	m1.append(a)
for i in range(r):
	a = list(map(int, input().split()))
	m2.append(a)
d1 = defaultdict(list)
d2 = defaultdict(list)
for i in range(r):
	for j in range(c):
		d1[i + j].append(m1[i][j])
		d2[i + j].append(m2[i][j])
for (i, j) in zip(d1, d2):
	if not Counter(d1[i]) == Counter(d2[j]):
		print('NO')
		sys.exit()
print('YES')
