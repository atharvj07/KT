t = int(input())
for _ in range(t):
	n = int(input())
	u = list(map(int, input().split()))
	s = list(map(int, input().split()))
	us = [[] for j in range(n)]
	for i in range(n):
		us[u[i] - 1].append(s[i])
	s = [0] * n
	for i in range(n):
		if us[i] == []:
			continue
		us[i].sort()
		p = len(us[i])
		if p > 1:
			for j in range(p - 2, -1, -1):
				us[i][j] += us[i][j + 1]
		for j in range(1, p + 1):
			k = p % j
			s[j - 1] += us[i][k]
	print(*s)
