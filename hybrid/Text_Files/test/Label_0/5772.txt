#include <bits/stdc++.h>
int main() {
  int n, d;
  scanf("%d %d", &n, &d);
  int i, a[n];
  int sum = 0, count = 0, j = 0;
  for (i = 0; i < n; i++) {
    scanf("%d", &a[i]);
  }
  for (i = 0; i < n - 1; i++) {
    sum = sum + a[i] + 10;
    j += 2;
    if (sum > d) {
      count = -1;
      break;
    }
  }
  int temp;
  if (count == -1) {
    printf("%d", count);
  } else {
    sum = sum + a[n - 1];
    if (sum > d) {
      count = -1;
      printf("%d", count);
    } else {
      temp = d - sum;
      if (temp >= 5) {
        j += temp / 5;
        printf("%d", j);
      } else {
        printf("%d", j);
      }
    }
  }
}
