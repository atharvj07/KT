        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;

        PrintWriter out = new PrintWriter(outputStream);

    static class TaskD {
        Scanner in;
        PrintWriter out;

        public void solve(int testNumber, Scanner in, PrintWriter out) {
            this.in = in;
            this.out = out;
            run();
        }

        void run() {
            int n = in.nextInt();
            int m = in.nextInt();
            int x = in.nextInt();
            int[][] es = in.nextIntMatrix(m, 3);
            double l = 0, r = 1000000;
            for (int times = 0; times < 100; times++) {
                double mid = (l + r) / 2;
                Dinic.V[] vs = new Dinic.V[n];
                for (int i = 0; i < vs.length; i++) {
                    vs[i] = new Dinic.V();
                }
                for (int[] e : es) {
                    int cap = (int) (e[2] / mid);
                    if (cap > 0) vs[e[0] - 1].add(vs[e[1] - 1], cap);
                }
                int y = Dinic.dinic(vs[0], vs[n - 1]);
                if (y >= x) l = mid;
                else r = mid;
            }
            out.printf("%.10f%n", l * x);
        }

    }

    static class Dinic {
        public static final int INF = Integer.MAX_VALUE / 4;
        public static int p = 0;

        public static int dinic(Dinic.V s, Dinic.V t) {
            int flow = 0;
            for (p++; ; p++) {
                Queue<Dinic.V> que = new LinkedList<Dinic.V>();
                s.level = 0;
                s.p = p;
                que.offer(s);
                while (!que.isEmpty()) {
                    Dinic.V v = que.poll();
                    v.iter = v.es.size() - 1;
                    for (Dinic.E e : v.es)
                        if (e.to.p < p && e.cap > 0) {
                            e.to.level = v.level + 1;
                            e.to.p = p;
                            que.offer(e.to);
                        }
                }
                if (t.p < p) return flow;
                for (int f; (f = dfs(s, t, INF)) > 0; ) flow += f;
            }
        }

        public static int dfs(Dinic.V v, Dinic.V t, int f) {
            if (v == t) return f;
            for (; v.iter >= 0; v.iter--) {
                Dinic.E e = v.es.get(v.iter);
                if (v.level < e.to.level && e.cap > 0) {
                    int d = dfs(e.to, t, Math.min(f, e.cap));
                    if (d > 0) {
                        e.cap -= d;
                        e.rev.cap += d;
                        return d;
                    }
                }
            }
            return 0;
        }

        public static class V {
            public ArrayList<Dinic.E> es = new ArrayList<Dinic.E>();
            public int level;
            public int p;
            public int iter;

            public void add(Dinic.V to, int cap) {
                Dinic.E e = new Dinic.E(to, cap), rev = new Dinic.E(this, 0);
                e.rev = rev;
                rev.rev = e;
                es.add(e);
                to.es.add(rev);
            }

        }

        public static class E {
            public Dinic.V to;
            public Dinic.E rev;
            public int cap;

            public E(Dinic.V to, int cap) {
                this.to = to;
                this.cap = cap;
            }

        }

    }

    static class Scanner {
        BufferedReader br;
        StringTokenizer st;

        public Scanner(InputStream in) {
            br = new BufferedReader(new InputStreamReader(in));
            eat("");
        }

        private void eat(String s) {
            st = new StringTokenizer(s);
        }

        public String nextLine() {
            try {
                return br.readLine();
            } catch (IOException e) {
                return null;
            }
        }

        public boolean hasNext() {
            while (!st.hasMoreTokens()) {
                String s = nextLine();
                if (s == null)
                    return false;
                eat(s);
            }
            return true;
        }

        public String next() {
            hasNext();
            return st.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

        public int[] nextIntArray(int n) {
            int[] is = new int[n];
            for (int i = 0; i < n; i++) {
                is[i] = nextInt();
            }
            return is;
        }

        public int[][] nextIntMatrix(int row, int col) {
            int[][] mat = new int[row][];
            for (int i = 0; i < row; i++) {
                mat[i] = nextIntArray(col);
            }
            return mat;
        }

    }
}
