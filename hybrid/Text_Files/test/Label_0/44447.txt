n = int(input())
s = input().split(' ')
bit = [0] * 2 * n

def decbit(position):
	if position >= 2 * n:
		return
	bit[position] -= 1
	decbit(position + (position & -position))

def getsum(left, right):
	return get(right) - get(left - 1)

def get(position):
	if position < 0:
		return 0
	if position == 0:
		return bit[0]
	return bit[position] + get(position - (position & -position))
left = [-1] * n
right = [-1] * n
isLeft = [False] * 2 * n
for i in range(2 * n):
	s[i] = int(s[i]) - 1
	if left[s[i]] != -1:
		right[s[i]] = i
	else:
		left[s[i]] = i
		isLeft[i] = True
res = 0
for i in range(2 * n):
	if isLeft[i]:
		res += right[s[i]] - i - 1 + getsum(i, right[s[i]])
		decbit(right[s[i]])
print(res)
