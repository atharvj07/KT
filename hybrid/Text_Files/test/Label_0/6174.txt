#include<bits/stdc++.h>
#define fo(i,a,b)for(int i=a,_e=b;i<=_e;++i)
#define fd(i,a,b)for(int i=b,_e=a;i>=_e;--i)
#define ll long long
using namespace std;
const int N=25005,K=405,mo=1e9+7;
int n,k,m,mx,mx2,a[N],la[K],f[N][K],g[N][K],_[N];
ll ans;
int ksm(int x,int y){
	int t=1;
	for(;y;y>>=1,x=(ll)x*x%mo)if(y&1)t=(ll)t*x%mo;
	return t;
}
void go(int f[][K]){
	fo(i,1,n){
		fo(j,0,k-2)f[i][j+1]=(ll)f[i-1][j]*(k-j)%mo;
		int su=0;
		fd(j,1,k-1){
			su=(su+f[i-1][j])%mo;
			f[i][j]=(f[i][j]+su)%mo;
		}
	}
}
int main(){
	cin>>n>>k>>m;
	ans=(ll)(n-m+1)*ksm(k,n-m)%mo;
	fo(i,1,m){
		scanf("%d",&a[i]);
		mx=max(mx,la[a[i]]);
		if(mx&&!mx2)mx2=i-1;
		if(i-mx==k){
			cout<<ans;return 0;
		}
		la[a[i]]=i;
	}
	if(mx==0){
		f[0][0]=1;
		fo(i,1,n){
			fo(j,0,k-2)f[i][j+1]=(ll)f[i-1][j]*(k-j)%mo,g[i][j+1]=(ll)g[i-1][j]*(k-j)%mo;
			int su=0,su2=0;
			fd(j,1,k-1){
				su=(su+f[i-1][j])%mo;
				su2=(su2+g[i-1][j])%mo;
				f[i][j]=(f[i][j]+su)%mo;
				g[i][j]=(g[i][j]+su2)%mo;
			}
			fo(j,m,k-1)g[i][j]=(g[i][j]+f[i][j])%mo;
		}
		ll jc=1;
		fo(i,k-m+1,k)jc=jc*i%mo;
		_[0]=1;fo(i,1,n)_[i]=(ll)_[i-1]*k%mo;
		ans=ans*jc%mo;
		fo(i,1,k-1)ans=(ans-g[n][i])%mo;
		cout<<(ans+mo)*ksm(jc,mo-2)%mo;
		return 0;
	}
	f[0][mx2]=1;
	go(f);
	g[0][m-mx]=1;
	go(g);
	fo(i,0,n-m){
		int su=0,su2=0;
		fo(j,0,k)su=(su+f[i][j])%mo,su2=(su2+g[n-m-i][j])%mo;
		ans=(ans-(ll)su*su2)%mo;
	}
	cout<<(ans+mo)%mo;
}
