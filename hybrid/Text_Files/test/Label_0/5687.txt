#include <bits/stdc++.h>
using namespace std;
long n, m, i, j, l[5005][5005], rad[5005], k, h, ans, p;
char s[5005];
int main() {
  cin >> n >> m;
  for (i = 1; i <= n; i++) {
    scanf("%s", &s);
    for (j = 1; j <= m; j++)
      if (s[j - 1] == '0')
        l[i][j] = 0;
      else
        l[i][j] = l[i][j - 1] + 1;
  }
  ans = 0;
  for (j = 1; j <= m; j++) {
    for (i = 1; i <= m; i++) rad[i] = 0;
    for (i = 1; i <= n; i++) rad[l[i][j]]++;
    h = 0;
    for (i = m; i >= 1; i--)
      if (rad[i] > 0)
        for (k = 1; k <= rad[i]; k++) {
          h++;
          p = i * h;
          if (ans < p) ans = p;
        }
  }
  cout << ans;
  return 0;
}
