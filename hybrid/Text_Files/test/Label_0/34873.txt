
import java.util.Scanner;

public class C {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        double w = in.nextLong();
        double h = in.nextLong();
        double a = in.nextDouble();
        if (a > 90)
            a = 180 - a;
        if (a == 0) {
            System.out.println(w * h);
            return;
        }
        a = a * Math.PI / 180;
        if (h < w) {
            double temp = h;
            h = w;
            w = temp;
        }
        double beta = 2 * Math.atan(w / h);
        double p1x = h / 2 * Math.cos(a) + w / 2 * Math.sin(a);
        double p2y = h / 2 * Math.sin(a) + w / 2 * Math.cos(a);
        if (a < beta) {
            double t1x = (p1x - h / 2) / Math.cos(a);
            double t1y = (p1x - h / 2) / Math.sin(a);
            double t2x = (p2y - w / 2) / Math.cos(a);
            double t2y = (p2y - w / 2) / Math.sin(a);
            System.out.println(w * h - t1x * t1y - t2x * t2y);
        } else {
            double s = w / Math.sin(a);
            System.out.println(s * w);
        }
    }
}
