import java.util.Scanner;

public class A787 {

	public static void main(String[] args) {
		
		Scanner s = new Scanner(System.in);
		int a = s.nextInt();
		int b = s.nextInt();
		int c = s.nextInt();
		int d = s.nextInt();
		boolean[] reached = new boolean[20000];
		int k = 0;
		while (b + k * a < 20000) {
			reached[b + k * a] = true;
			k++;
		}
		k = 0;
		boolean found = false;
		while (d + k * c < 20000) {
			if (reached[d + k * c]) {
				System.out.println(d + k * c);
				found = true;
				break;
			}
			k++;
		}
		if (!found) System.out.println(-1);
		
	}
	
}
