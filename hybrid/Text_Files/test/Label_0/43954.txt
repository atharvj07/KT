import sys

def convert_to_bin(x):
	res = ''
	while x > 0:
		if x % 2 == 0:
			res += '0'
		else:
			res += '1'
		x = x // 2
	return res[::-1]
in_file = sys.stdin
(n, k) = map(int, in_file.readline().strip().split())
A = list(map(int, in_file.readline().strip().split()))
A = [convert_to_bin(a) for a in A]
lookup = {}
kids = []
parent = []
count = []
best = []
for a in A:
	t = ''
	for i in range(len(a)):
		prev = t
		t += a[i]
		if t not in lookup:
			lookup[t] = len(kids)
			if i != 0:
				kids[lookup[prev]].append(len(kids))
				parent.append(lookup[prev])
			else:
				parent.append(-1)
			kids.append([])
			count.append(0)
			best.append([-1 for i in range(k + 1)])
			best[-1][0] = 0
		if i == len(a) - 1:
			count[lookup[t]] += 1

def dp(i, k):
	if k < 0:
		return 0
	if best[i][k] != -1:
		return best[i][k]
	if count[i] >= k:
		best[i][k] = 0
		return 0
	best[i][k] = (len(A) + 1) * (len(A) + 1) * (len(A) + 1)
	if len(kids[i]) == 1:
		best[i][k] = k - count[i] + dp(kids[i][0], k - count[i])
	elif len(kids[i]) == 2:
		for j in range(k - count[i] + 1):
			best[i][k] = min(best[i][k], k - count[i] + dp(kids[i][0], j) + dp(kids[i][1], k - count[i] - j))
	return best[i][k]
m = (len(A) + 1) * (len(A) + 1) * (len(A) + 1)
for i in range(len(kids)):
	if m > dp(i, k):
		m = dp(i, k)
sys.stdout.write(str(m))
sys.stdout.flush()
