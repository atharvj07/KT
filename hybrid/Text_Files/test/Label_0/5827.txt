#include<iostream>
#include<vector>
#include<deque>
#include<algorithm>
using namespace std;
int N,K,S;
int X[3][2000];
struct UF{
	int N;
	vector<int>p,sz;
	UF(int n):N(n),p(n),sz(n,1){
		for(int i=0;i<N;i++)p[i]=i;
	}
	int find(int u)
	{
		if(p[u]==u)return u;
		else return p[u]=find(p[u]);
	}
	bool unite(int u,int v)
	{
		u=find(u),v=find(v);
		if(u==v)return false;
		if(sz[u]<sz[v])
		{
			p[u]=v;
			sz[v]+=sz[u];
		}
		else
		{
			p[v]=u;
			sz[u]+=sz[v];
		}
		return true;
	}
};
vector<long>A;
bool dfs(int u,int p,int out,vector<vector<pair<int,long> > >&G)
{
	if(u==out)
	{
		return true;
	}
	bool ret=false;
	for(pair<int,long>&q:G[u])
	{
		if(q.first!=p)
		{
			A.push_back(q.second);
			ret|=dfs(q.first,u,out,G);
		}
	}
	return ret;
}
int main()
{
	while(cin>>N>>K>>S,N)
	{
		for(int i=0;i<N;i++)
		{
			for(int j=0;j<3;j++)cin>>X[j][i];
		}
		vector<pair<pair<int,int>,long> >E;
		UF uf(N);
		vector<vector<pair<int,long> > >G(N);
		for(int i=0;i<N;i++)for(int j=i+1;j<N;j++)
		{
			int d[3];
			bool flag=true;
			for(int k=0;k<3;k++)
			{
				d[k]=S-abs(X[k][i]-X[k][j]);
				if(d[k]<=0)flag=false;
			}
			if(flag)
			{
				long t=d[0]*d[1]+d[1]*d[2]+d[2]*d[0];
				t*=2;
				E.push_back(make_pair(make_pair(i,j),t));
				uf.unite(i,j);
				G[i].push_back(make_pair(j,t));
				G[j].push_back(make_pair(i,t));
			}
		}
		long ans=-1;
		vector<bool>vis(N,false);
		for(int i=0;i<N;i++)
		{
			int u=uf.find(i);
			if(uf.sz[u]<K)vis[u]=true;
			if(vis[u])continue;
			A.clear();
			vis[u]=true;
			bool flag=false;
			for(pair<int,long>&q:G[u])
			{
				A.push_back(q.second);
				if(dfs(q.first,u,u,G))
				{
					flag=true;
					break;
				}
				reverse(A.begin(),A.end());
			}
			if(flag)
			{
				int n=A.size();
				for(int i=0;i<n;i++)A.push_back(A[i]);
			}
			int get=flag&&uf.sz[u]==K?K:K-1;
			vector<long>sum(A.size()+1,0);
			for(int i=0;i<A.size();i++)sum[i+1]=sum[i]+A[i];
			for(int i=get;i<=A.size();i++)ans=max(ans,sum[i]-sum[i-get]);
		}
		if(ans<0)cout<<-1<<endl;
		else cout<<6LL*S*S*K-ans<<endl;
	}
}
