int main(void){
	cout<<fixed<<setprecision(20);
	cin.tie(0);ios::sync_with_stdio(false);
	//seg木を使わずに行きたい
	//doubleはたぶんダメ
	int n,i,j;cin>>n;
	vector<pair<int,int>>zi;
	zi.reserve(n);
	for(i=0;i<n;i++){
		int T,A;cin>>T>>A;
		if(A==0){continue;}
		zi.pub(mp(T,A));
	}
	n=zi.size();
	vector<double>ma(n);
	for(i=0;i+20<=n;i++){
		double a=1;
		for(j=0;j<20;j++){a*=1-0.1*zi[i+j].sec;}
		ma[i]=a;
	}
	//+20のアレを使う
	//誤差を有効に使う
	int Q;cin>>Q;
	while(Q--){
		int L,R;cin>>L>>R;
		int Lb=LBI(zi,mp(L,0));
		int Rb=LBI(zi,mp(R,0));
		if(Rb-Lb>=400){cout<<0<<endl;continue;}//誤差なので
		double ans=1e9;
		while(Lb+20<=Rb){
			ans*=ma[Lb];
			Lb+=20;
		}
		while(Lb+1<=Rb){
			ans*=1-0.1*zi[Lb].sec;
			Lb++;
		}
		cout<<ans<<endl;
	}
	return 0;
}
