#include <bits/stdc++.h>
using namespace std;
vector<int> facs[1000009];
int dp[1000002][20], inv, mod = 1000000007;
int main() {
  for (int i = 0; i <= 20; i++) dp[0][i] = 2;
  dp[0][0] = 1;
  for (int i = 1; i <= 1000000; i++) {
    int pref = 0;
    for (int j = 0; j <= 20; j++) {
      pref = (pref + dp[i - 1][j]) % mod;
      dp[i][j] = pref;
    }
  }
  for (int i = 2; i <= 1000000; i++)
    if (!facs[i].size()) {
      for (int j = i; j <= 1000000; j += i) {
        facs[j].push_back(i);
      }
    }
  int q;
  scanf("%d", &q);
  while (q--) {
    int r, n;
    scanf("%d %d", &r, &n);
    int sol = 1;
    for (int i = 0; i < facs[n].size(); i++) {
      int c = n, x = 0;
      while (c % facs[n][i] == 0) {
        c /= facs[n][i];
        x++;
      }
      sol = (sol * 1LL * dp[r][x]) % mod;
    }
    printf("%d\n", sol);
  }
}
