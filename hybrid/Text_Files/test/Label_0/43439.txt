import math
import collections
from sys import stdin, stdout, setrecursionlimit
from bisect import bisect_left as bsl
from bisect import bisect_right as bsr
import heapq as hq
setrecursionlimit(2 ** 20)
t = 1
for _ in range(t):
	n = int(stdin.readline())
	a = list(map(int, stdin.readline().rstrip().split()))
	b = list(map(int, stdin.readline().rstrip().split()))
	d = {}
	for i in range(n):
		d[a[i]] = i
	r = {}
	l = {}
	zc = 0
	mx = 0
	for j in range(n):
		if d[b[j]] > j:
			rs = (d[b[j]] - j) % n
			ls = (j + n - d[b[j]]) % n
		elif d[b[j]] < j:
			rs = (d[b[j]] - j) % n
			ls = (j + n - d[b[j]]) % n
		else:
			zc += 1
			continue
		if rs not in r:
			r[rs] = 0
		if ls not in l:
			l[ls] = 0
		r[rs] += 1
		l[ls] += 1
		if r[rs] > mx:
			mx = r[rs]
		if l[ls] > mx:
			mx = l[ls]
	print(max(mx, zc))
