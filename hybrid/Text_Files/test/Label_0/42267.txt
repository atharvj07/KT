import sys
import math
input = sys.stdin.readline
imp = 'IMPOSSIBLE'
t = int(input())
for test in range(t):
	(n, m) = list(map(int, input().split(' ')))
	a = list(map(int, input().split(' ')))
	poc = [0 for i in range(n)]
	for i in range(n):
		kk = (i + 1 - a[i]) % n
		poc[kk] += 1
	moz = []
	for i in range(n):
		if poc[i] >= n - 2 * m:
			moz.append(i)
	ine = []
	res = []
	for mo in moz:
		ine = []
		for i in range(mo, n):
			if (i + 1 - a[i]) % n != mo:
				ine.append(a[i])
		for i in range(mo):
			if (i + 1 - a[i]) % n != mo:
				ine.append(a[i])
		sorine = sorted(enumerate(ine), key=lambda x: x[1])
		convine = [0 for i in range(len(ine))]
		for i in range(len(ine)):
			convine[sorine[i][0]] = i + 1
		jo = [0 for i in range(len(ine))]
		i = 0
		sets = 0
		while i < len(ine):
			if jo[i]:
				i += 1
			else:
				sets += 1
				k = i
				jo[k] = 1
				while convine[k] != i + 1:
					k = convine[k] - 1
					jo[k] = 1
				i += 1
		if len(ine) - sets <= m:
			res.append(str(mo))
	print(str(len(res)) + ' ' + ' '.join(res))
