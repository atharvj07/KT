for _ in range(int(input())):
	s = input()
	arr = []
	for i in range(2 ** 10):
		x = bin(i).split('b')[1]
		x = '0' * (10 - len(x)) + x
		f = 0
		for j in range(10):
			if s[j] == '?':
				continue
			if s[j] == x[j]:
				pass
			else:
				f = 1
		if f == 0:
			arr.append(x)
	ans = 10
	for i in arr:
		(a, b) = (0, 0)
		for j in range(10):
			if j % 2 == 0:
				a += int(i[j])
			else:
				b += int(i[j])
			if j % 2 == 0:
				rem = 9 - j
				rem = (rem + 1) // 2
				if rem + b < a:
					ans = min(ans, j + 1)
					break
				rem = (9 - j) // 2
				if a + rem < b:
					ans = min(ans, j + 1)
			else:
				rem = 9 - j
				rem = rem // 2
				if rem + a < b:
					ans = min(ans, j + 1)
					break
				if b + rem < a:
					ans = min(ans, j + 1)
	print(ans)
