import java.util.*;
import java.io.*;

public class ChessTourney {
	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);
		int n = scan.nextInt();
		int[] a = new int[2*n];
		for(int i = 0 ; i < 2*n ; i++)
			a[i] = scan.nextInt();
		Arrays.sort(a);
		System.out.println(a[n-1] < a[n]?"YES":"NO");
	}
}
