#include <cstdio>
int n, d, x;
int main() {
	scanf("%d%d%d", &n, &d, &x);
	double first = d, p = x;
	double ans = 0;
	for(int len = 2 * n; len > 0; len -= 2) {
		ans += (2 * first + (len - 1) * p) / 2;
		first = ((len + 2) * first + 5 * p) / len;
		p = (len + 4) * p / len;
	}
	printf("%f\n", ans);
	return 0;
}
