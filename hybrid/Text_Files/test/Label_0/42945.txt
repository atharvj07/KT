n = int(input())
prime = [True for i in range(n + 3)]

def SieveOfEratosthenes(n):
	p = 2
	while p * p <= n:
		if prime[p] == True:
			for i in range(p * 2, n + 1, p):
				prime[i] = False
		p += 1
SieveOfEratosthenes(n + 2)
p = []
for i in range(2, n + 2):
	if prime[i]:
		p.append(1)
	else:
		p.append(2)
s = set(p)
print(len(s))
for i in range(0, len(p)):
	print(p[i], end=' ')
