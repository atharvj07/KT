(n, s) = map(int, input().split())
mn = 101
zero = False
for i in range(n):
	(d, c) = map(int, input().split())
	if d == s and c == 0:
		zero = True
		continue
	if d < s and c < mn:
		if c == 0:
			zero = True
		else:
			mn = c
if mn > 100 and zero:
	print('0')
elif mn > 100:
	print('-1')
else:
	print(100 - mn)
