import sys
(n, m, k) = map(int, input().split())
save_k = k
if k > 4 * n * m - 2 * n - 2 * m:
	print('NO')
	sys.exit()
arr = []
for i in ['R', 'L']:
	if k < 1:
		break
	arr.append((m - 1, i))
	k -= m - 1
for i in range(n - 1):
	if k < 1:
		break
	arr.append((1, 'D'))
	arr.append((m - 1, 'R'))
	arr.append((m - 1, 'UDL'))
	k -= 4 * m - 3
if k > 0:
	arr.append((k, 'U'))
k = save_k
ans = []
for (i, j) in arr:
	if i == 0:
		continue
	if k > i * len(j):
		ans.append((i, j))
		k -= i * len(j)
		continue
	if k // len(j) != 0:
		ans.append((k // len(j), j))
	if k % len(j) != 0:
		ans.append((1, j[:k % len(j)]))
	break
print(f'YES\n{len(ans)}')
for (i, j) in ans:
	print(i, j)
