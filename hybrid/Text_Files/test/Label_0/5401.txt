long long int modex(long long int a, long long int b, long long int c) {
  long long int ans = 1;
  while (b != 0) {
    if (b % 2 == 1) ans = (ans * a) % c;
    a = (a * a) % c;
    b /= 2;
  }
  return ans;
}
int gcd(int a, int b) {
  if (a % b == 0) {
    return b;
  } else {
    return gcd(b, a % b);
  }
}
bool isprime[100005];
void sieve(int n) {
  for (int i = 0; i <= n; i++) {
    isprime[i] = true;
  }
  isprime[0] = false;
  isprime[1] = false;
  for (int i = 2; i * i <= n; i++) {
    if (isprime[i]) {
      for (int j = i * i; j <= n; j += i) isprime[j] = false;
    }
  }
  for (int i = 0; i <= n; i++) {
    if (isprime[i]) {
      cout << i << " ";
    }
  }
  cout << endl;
}
void code() {
  int nn;
  cin >> nn;
  long long int p = 0, n = 0;
  for (int i = 0; i < nn; i++) {
    long long int x, y;
    cin >> x >> y;
    if (x > 0) {
      p++;
    } else {
      n++;
    }
  }
  if (p <= 1 || n <= 1) {
    cout << "Yes" << endl;
  } else {
    cout << "No" << endl;
  }
}
