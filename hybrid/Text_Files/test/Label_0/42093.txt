t = int(input())
for i in range(t):
	(n, x) = map(int, input().split())
	mr = 0
	mch = 0
	ma = 0
	for j in range(n):
		(d, h) = map(int, input().split())
		if d > mch:
			mch = d
		if d - h > mr or (d - h == mr and d > ma):
			mr = d - h
			ma = d
	ans1 = 0
	if mch >= x:
		print(1)
	elif mr == 0:
		print(-1)
	else:
		ans1 = (x - mch) // mr
		if (x - mch) % mr == 0:
			ans1 += 1
		else:
			ans1 += 2
		print(ans1)
