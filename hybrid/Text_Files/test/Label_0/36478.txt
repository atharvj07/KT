import java.io.*;
import java.util.*;

public class E{
	
	static int n;
	static List<Integer>[] adj;
	static int[] odd;
	static int[] even;
	static int[] bip;
	static long[] v;
	static long mod = (long)1e9+7;
	static long ans;
	static void color(int i, int p, int c) {
		if (bip[i] != -1) return;
		bip[i] = c;
		for (int x: adj[i]) {
			if (x==p) continue;
			color(x,i,(c+1)%2);
		}
	}
	
	static void count(int i, int p) {
		// Count odds and evens per subtree, NOT INCLUDING ITSELF!
		for (int x: adj[i]) {
			if(x==p) continue;
			count(x,i);
			if (bip[x] == 0) even[i]++;
			else odd[i]++;
			odd[i]+=odd[x];
			even[i]+=even[x];
		}
		
	}
	
	static long solveCount(List<Integer> count) {
		long pref = 0;
		long res = 0;
		for (int x: count) pref+=x;
		for (int x: count) {
			pref-=x;
			res+=x*pref;
			res%=mod;
		}
		return res;
	}
	
	static int numEvens, numOdds;
	
	static void calc(int i, int p) {
		List<Integer> zeroCount = new ArrayList<Integer>();
		List<Integer> oneCount = new ArrayList<Integer>();
		zeroCount.add(numEvens-even[i]-(bip[i]==0?1:0));
		oneCount.add(numOdds-odd[i] - (bip[i]==1?1:0));
		for (int x: adj[i]) {
			if(x==p) continue;
			zeroCount.add(even[x] + (bip[x]==0?1:0));
			oneCount.add(odd[x] + (bip[x]==1?1:0));
			calc(x,i);
		}
		long bias = 0; // For the i-i case
		// Calculate all pairwise somehow
		// Handle same bipartite case differently. Include sum of same subtrees as well.
		// Odd paths WILL cancel each other paths. Only even paths matter, and double them.
		// Therefore try to only consider how many evens and odds funnel through this node.
		if (bip[i] == 0) for (int x: zeroCount) bias+=x;
		else for (int x: oneCount) bias+=x;
		//System.out.println("My initial bias: " + bias);
		// Alright, now big brain shit
		//System.out.println(zeroCount + " " + oneCount);
		//System.out.println(solveCount(zeroCount) + " " + solveCount(oneCount));
		
		long zc = solveCount(zeroCount);
		long oc = solveCount(oneCount);
		if (bip[i] == 0) {
			bias+=zc-oc;
		}
		else bias+=oc-zc;
		bias=bias*2+1;
		bias%=mod;
		bias = bias * v[i];
		bias %= mod;
		
		//System.out.println(bias);
		
		ans = (ans + bias + mod)%mod;
	}
	
	public static void main(String[] args) throws IOException {
		// br = new BufferedReader(new FileReader(".in"));
		// out = new PrintWriter(new FileWriter(".out"));
		//new Thread(null, new (), "peepee", 1<<28).start();
		n = readInt();
		adj = new List[n];
		for (int i = 0; i < n; i++) adj[i] = new ArrayList<Integer>();
		odd = new int[n];
		even = new int[n];
		bip = new int[n];
		v = new long[n];
		for (int i = 0; i < n; i++) v[i] =readInt();
		for (int i = 0; i < n-1; i++) {
			int x =readInt()-1;
			int y =readInt()-1;
			adj[x].add(y);
			adj[y].add(x);
		}
		Arrays.fill(bip, -1);
		color(0,0,0);
		count(0,0);
		for (int i = 0; i < bip.length; i++) if (bip[i] == 0) numEvens++; else numOdds++;
		calc(0,0);
		
		out.println(ans);
		out.close();
	}
	
	static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
	static PrintWriter out = new PrintWriter(new BufferedOutputStream(System.out));
	static StringTokenizer st = new StringTokenizer("");
	static String read() throws IOException{return st.hasMoreTokens() ? st.nextToken():(st = new StringTokenizer(br.readLine())).nextToken();}
	static int readInt() throws IOException{return Integer.parseInt(read());}
	static long readLong() throws IOException{return Long.parseLong(read());}
	static double readDouble() throws IOException{return Double.parseDouble(read());}
	
}
