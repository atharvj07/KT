import java.math.BigDecimal;

public class C {

    private static boolean planC(long n, long x, long y) {
        if (n > y) {
            return false;
        }

        long s = y - (n - 1);

        if (s * s + (n - 1) >= x) {
            for (int i = 0; i < n - 1; ++i) {
                System.out.println(1);
            }
            System.out.println(s);

            return true;
        }

        return false;
    }

    private static boolean planB(long n, long x, long y) {
        BigDecimal by = BigDecimal.valueOf(y);
        BigDecimal bx = BigDecimal.valueOf(x);
        BigDecimal bn = BigDecimal.valueOf(n);

        BigDecimal bs = by.divide(bn);
//        System.out.println("s: " + s);

        long start = bs.longValue();
        long next = by.subtract(BigDecimal.valueOf(start).multiply(bn)).longValue();
//        System.out.println("start: " + start);
//        System.out.println("next: " + next);

        long qs = quadSumm(n, start, next);
//        System.out.println("qs: " + qs);

        if (qs >= x) {
            for (long i = 0; i < (n - next); ++i) {
                System.out.println(start);
            }

            for (long i = 0; i < next; ++i) {
                System.out.println(start + 1);
            }
        } else {
            return false;
        }

        return true;
    }

    private static long quadSumm(long n, long start, long next) {
        return (start * start) * (n - next) + ((start + 1) * (start + 1)) * next;
    }

    private static void planA(long n, long x, long y) {
        long start = (long) start(n, y);

        System.out.println("s: " + start);

        long as = as(start + n - 1) - as(start - 1);
        long qs = qs(start + n - 1) - qs(start - 1);

        System.out.println("as: " + as);
        System.out.println("qs: " + qs);

        if (qs >= x && start > 0) {
            System.out.println("YES");
            for (long i = start; i < start + n; ++i) {
                System.out.println(i);
            }
        } else {
            System.out.println(-1);
        }
    }

    private static double start(long n, long y) {
        return (double) y / (double) n - ((double) n - 1d) / 2d;
    }

    private static long qs(long n) {
        return (n * (n + 1) * (2 * n + 1)) / 6;
    }

    private static long as(long n) {
        return ((1 + n) * n) / 2;
    }

}
