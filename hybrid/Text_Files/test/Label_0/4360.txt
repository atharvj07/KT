#include <bits/stdc++.h>
struct P {
  long long x, y;
  P() {}
  P(long long x, long long y) : x(x), y(y) {}
  P operator-(P p) { return P(x - p.x, y - p.y); }
  P operator+(P p) { return P(x + p.x, y + p.y); }
  long long operator^(P p) { return x * p.y - y * p.x; }
  void eat() { scanf("%I64d%I64d", &x, &y); }
};
const int N = 2030600;
P p[N];
int n, t;
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; i++) p[i].eat(), p[i + n] = p[i];
  scanf("%d", &t);
  while (t--) {
    P q;
    q.eat();
    long long ans = (long long)n * (n - 1) / 2 * (n - 2) / 3;
    bool ok = 1;
    for (int i = 0; i < n; i++) ok &= ((p[i] - q) ^ (p[i + 1] - q)) < 0;
    if (!ok) {
      puts("0");
      continue;
    }
    for (int i = 0, j = 1; i < n; i++) {
      while (j < i + n && ((p[i] - q) ^ (p[j] - q)) <= 0) ++j;
      long long c = j - i - 1;
      ans -= c * (c - 1) / 2;
    }
    printf("%I64d\n", ans);
  }
}
