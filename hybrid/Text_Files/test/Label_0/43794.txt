for _ in range(int(input())):
	n = int(input())
	x = list(map(int, input().split()))
	if n == 1:
		print('YES')
	elif n == 2:
		if x[0] <= x[1]:
			print('YES')
		else:
			print('NO')
	else:
		ans = 'NO'
		temp = [0] * n
		for i in range(n):
			temp[x[i] - 1] += 1
			if temp[x[i] - 1] > 1:
				ans = 'YES'
				break
		if ans == 'NO':
			p = 0
			for i in range(n):
				while x[i] != i + 1:
					temp1 = x[i]
					temp2 = x[x[i] - 1]
					x[x[i] - 1] = temp1
					x[i] = temp2
					p += 1
			if p % 2 == 0:
				ans = 'YES'
		print(ans)
