#include <bits/stdc++.h>
using namespace std;
int main(){
  string s;
  cin >> s;
  for(int i = 0;i < s.size();i++){
    if(i == s.size()-1)cout << -1 << " " << -1 << endl;
    else if(s[i] == s[i+1]){
      cout << i+1 << " " << i+2 << endl;
      break;
    }else if(i < s.size()-1 && s[i] == s[i+2]){
      cout << i+1 << " " << i+3 << endl;
      break;
    }
  }
}
