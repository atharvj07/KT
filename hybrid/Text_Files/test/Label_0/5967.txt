#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:66777216")
using namespace std;
int a, b, c, d, n, m, k;
pair<int, int> mas[102];
int can[102], best[102];
inline bool check(int xx) {
  memset(best, -1, sizeof(best));
  best[0] = xx;
  queue<int> q;
  q.push(0);
  while (!q.empty()) {
    int v = q.front();
    q.pop();
    for (int _n((n)-1), i(0); i <= _n; i++) {
      if (i == v) continue;
      if (i == 0) continue;
      int need = d * (abs(mas[i].first - mas[v].first) +
                      abs(mas[i].second - mas[v].second));
      if (need > best[v]) continue;
      int cur = best[v] - need + can[i];
      if (cur > best[i]) {
        best[i] = cur;
        q.push(i);
      }
    }
  }
  return best[n - 1] != -1;
}
int main() {
  scanf("%d%d", &n, &d);
  for (int _n(n - 2), i(1); i <= _n; i++) {
    scanf("%d", &can[i]);
  }
  for (int _n((n)-1), i(0); i <= _n; i++) {
    scanf("%d%d", &mas[i].first, &mas[i].second);
  }
  int l = -1, r = 1000000000;
  while (r - l > 1) {
    int xx = (r + l) / 2;
    if (check(xx))
      r = xx;
    else
      l = xx;
  }
  printf("%d\n", r);
}
