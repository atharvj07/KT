#include <bits/stdc++.h>
using namespace std;
inline int rei() {
  int x;
  cin >> x;
  return x;
}
inline long long rel() {
  long long x;
  cin >> x;
  return x;
}
inline string res() {
  string x;
  cin >> x;
  return x;
}
vector<int> G[100000];
int depth[100000];
int Num[100001];
void Calc() {
  int N = rei();
  for (int i = 0; i < N - 1; i++) {
    int v = rei() - 1;
    int t = rei() - 1;
    G[v].push_back(t);
    G[t].push_back(v);
  }
  stack<pair<int, int>> sp;
  sp.push({0, -1});
  while (!sp.empty()) {
    int v = sp.top().first;
    int f = sp.top().second;
    sp.pop();
    if (f == -1) {
      depth[v] = 1;
    } else {
      depth[v] = depth[f] + 1;
    }
    for (int i = 0; i < G[v].size(); i++) {
      int t = G[v][i];
      if (t != f) {
        sp.push({t, v});
      }
    }
  }
  for (int i = 0; i < N; i++) {
    Num[depth[i]]++;
  }
  double ans = 0;
  for (int i = N; i > 0; i--) {
    double e = 1;
    e /= i;
    e *= Num[i];
    ans += e;
  }
  printf("%1.10lf\n", ans);
}
int main(int argc, char** argv) {
  ios::sync_with_stdio(false), cin.tie(0);
  cout.tie(0);
  Calc();
  return 0;
}
