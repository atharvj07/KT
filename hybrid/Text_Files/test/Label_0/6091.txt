#include <bits/stdc++.h>
using namespace std;
const long long int inf = (long long int)1e18 + 10;
map<long long int, long long int> m, m2;
map<pair<long long int, long long int>, long long int> done;
long long int a[1003];
long long int mx = 0;
long long int get(long long int y, long long int x) {
  long long int cnt = 2, temp;
  m2[x] += 1;
  m2[y] += 1;
  while (abs(x) <= mx + 1) {
    temp = x;
    x = x + y;
    y = temp;
    m2[x] += 1;
    if (m.find(x) != m.end() && m2[x] <= m[x])
      cnt++;
    else
      break;
  }
  m2.clear();
  return cnt;
}
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long int n, ans = 2;
  cin >> n;
  for (long long int i = 0; i < n; i++) {
    cin >> a[i];
    m[a[i]] += 1;
    mx = max(mx, abs(a[i]));
  }
  for (long long int i = 0; i < n; i++) {
    for (long long int j = i + 1; j < n; j++) {
      if (done[make_pair(a[i], a[j])] != 1)
        ans = max(ans, get(a[i], a[j])), done[make_pair(a[i], a[j])] = 1;
      if (done[make_pair(a[j], a[i])] != 1)
        ans = max(ans, get(a[j], a[i])), done[make_pair(a[j], a[i])] = 1;
    }
  }
  cout << ans << endl;
}
