for _ in range(int(input())):
	(n, u, v) = map(int, input().split())
	obstacle = list(map(int, input().split()))
	cost = 0
	diff = 0
	for i in range(n - 1):
		if abs(obstacle[i + 1] - obstacle[i]) > diff:
			diff = abs(obstacle[i + 1] - obstacle[i])
	if diff == 0:
		print(min(2 * v, u + v))
	elif diff == 1:
		print(min(u, v))
	else:
		print(0)
