(n, m, k) = map(int, input().split())
a = [int(i) for i in input().split()]
b = [int(i) for i in input().split()]
c = {}
j = 0
for i in a:
	c[i] = j
	j += 1
num = 0
for i in b:
	d = c[i]
	num += (d + k) // k
	if c[i] != 0:
		e = a[d - 1]
		(c[i], c[e]) = (c[e], c[i])
		(a[d], a[d - 1]) = (a[d - 1], a[d])
print(num)
