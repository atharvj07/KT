from itertools import groupby
for _ in range(int(input())):
	n = int(input())
	o = input()
	p = input()
	op = list(zip(o, p))
	ans = 0
	i = 0
	while i < n:
		(r1, r2) = op[i]
		if r1 != r2:
			ans += 2
			i += 1
		elif r1 == r2 == '0':
			try:
				if set(op[i + 1]) == {'1'}:
					ans += 2
					i += 2
				else:
					ans += 1
					i += 1
			except:
				ans += 1
				i += 1
		elif r1 == r2 == '1':
			try:
				if set(op[i + 1]) == {'0'}:
					ans += 2
					i += 2
				else:
					i += 1
			except:
				i += 1
	print(ans)
