#include<iostream>
#include<string>
#include<algorithm>
using namespace std;
int main(void){
  string m[1010];
  int n,ans,dp[1010][1010];
  while(cin >> n,n){
    for(int i=0;i<n;i++) cin >> m[i];
    ans=0;
    for(int i=0;i<n;i++){
      for(int j=0;j<n;j++){
	if(m[i][j]=='.') dp[i][j]=1;
	else dp[i][j]=0;
      }
    }
    for(int i=1;i<n;i++){
      for(int j=1;j<n;j++){
        if(m[i][j]=='.'){
	  dp[i][j]=min(dp[i-1][j],min(dp[i][j-1],dp[i-1][j-1]))+1;
	  ans=max(ans,dp[i][j]);
	}
      }
    }
    cout << ans << endl;
  }
}
