l = [input() for _ in range(4)]
m = list(map(lambda x: len(x[2:]), l))
res = []
for i in l:
	y = 1
	r1 = True
	r2 = True
	for j in l:
		if i != j:
			if len(i[2:]) >= len(j[2:]) * 2 and r1:
				y *= 1
				r2 = False
			elif len(i[2:]) <= len(j[2:]) / 2 and r2:
				r1 = False
				y *= 1
			else:
				y *= 0
				break
	res.append(y)
if res.count(0) == 4:
	print('C')
elif res.count(1) > 1:
	print('C')
else:
	print('ABCD'[res.index(1)])
