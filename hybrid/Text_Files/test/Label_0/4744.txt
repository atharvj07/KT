#include <bits/stdc++.h>

const int MAXN = 1e3 + 10, INF = 1e9;

int n, m;
int id[MAXN][MAXN], fa[MAXN];

struct NetWork {
	static const int MAXM = MAXN * MAXN + MAXN * 10;
	int B, E, sum, N,  MaxFlow, MinCost;
	int last[MAXN << 1], vis[MAXN << 1], cur[MAXN << 1], dis[MAXN << 1];
	struct Edge { int link, to, flow, cost; } edge[MAXM << 1];
	inline void Init(int _, int __, int ___) {
		B = _, E = __, N = ___, sum = -1;
		std::fill(last, last + N + 1, -1);
	}
	inline void Add(int u, int v, int f, int c) { edge[++sum] = (Edge) { v, last[u], f, c }, last[u] = sum; }
	inline void AddEdge(int u, int v, int f, int c) { Add(u, v, f, c), Add(v, u, 0, -c); }
	inline int SPFA() {
		std::queue <int> Q;
		std::fill(dis, dis + N + 1, INF), dis[B] = 0, Q.push(B);
		while (!Q.empty()) {
			int x = Q.front(); Q.pop(); vis[x] = 0;
			for (int i = last[x]; ~i; i = edge[i].to) {
				int y = edge[i].link, c = edge[i].cost;
				if (dis[y] > dis[x] + c && edge[i].flow) {
					dis[y] = dis[x] + c;
					if (!vis[y]) vis[y] = 1, Q.push(y);
				}
			}
		}
		return dis[E] != INF;
	}
	int Dfs(int x, int dist) {
		int flown = 0, f;
		if (x == E || !dist) return dist; vis[x] = 1;
		for (int &i = cur[x]; ~i; i = edge[i].to) {
			int y = edge[i].link;
			if (!vis[y] && dis[y] - dis[x] == edge[i].cost && ((f = Dfs(y, std::min(dist, edge[i].flow))) > 0)) {
				edge[i].flow -= f, edge[i ^ 1].flow += f;
				flown += f, dist -= f;
				if (!dist) break;
			}
		}
		vis[x] = 0;
		return flown;
	}
	inline void ZKW() {
		MaxFlow = MinCost = 0;
		while (SPFA()) {
			for (int i = 0; i <= N; ++i) cur[i] = last[i], vis[i] = 0;
			int d = Dfs(B, INF);
			MaxFlow += d, MinCost += d * dis[E];
		}
	}
} G;
int find(int x) { return x == fa[x] ? x : fa[x] = find(fa[x]); }

int main() {
	scanf("%d%d", &n, &m);
	for (int i = 1, x, y; i <= m; ++i) scanf("%d%d", &x, &y), id[x][y] = i;
	for (int i = 1; i <= m; ++i) fa[i] = i;
	int cnt = 0;
	for (int i = 1; i <= n; ++i)
		for (int j = 1, lst = 0; j <= n; ++j) if (id[i][j]) {
			if (lst) {
				int x = find(lst), y = find(id[i][j]);
				if (x != y) fa[y] = x, ++cnt;
			}
			lst = id[i][j];
		}
	for (int j = 1; j <= n; ++j)
		for (int i = 1, lst = 0; i <= n; ++i) if (id[i][j]) {
			if (lst) {
				int x = find(lst), y = find(id[i][j]);
				if (x != y) fa[y] = x, ++cnt;
			}
			lst = id[i][j];
		}
	int S = n * 2 + 1, T = S + 1, N = T;
	G.Init(S, T, N);
	for (int i = 1; i <= n; ++i) G.AddEdge(S, i, 1, 0);
	for (int i = 1; i <= n; ++i) G.AddEdge(i + n, T, 1, 0);
	for (int i = 1; i <= n; ++i)
		for (int j = 1; j <= n; ++j) G.AddEdge(i, j + n, 1, id[i][j] ? 1 : 2);
	G.ZKW();
	printf("%d %d\n", m - (2 * n - G.MinCost), cnt);
	return 0;
}
