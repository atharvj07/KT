def solution():
	(n, m) = map(int, input().strip().split())
	q = int(input())
	a_b = set()
	for _ in range(q):
		(a, b) = map(int, input().strip().split())
		a_b.add((a, b) if n < m else (b, a))
	if n > m:
		(n, m) = (m, n)
	dp = [-1] * (n + 1)
	dp[1] = 1
	result = 0
	while dp[n] != m:
		new_dp = dp[:]
		for i in range(1, n + 1):
			if dp[i] == -1:
				continue
			x = i + dp[i] + ((i, dp[i]) in a_b)
			new_dp[i] = max(new_dp[i], min(x, m))
			new_dp[min(x, n)] = max(new_dp[min(x, n)], dp[i])
		dp = new_dp
		result += 1
	return result
print(solution())
