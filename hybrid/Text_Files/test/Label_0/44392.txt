def solve(s, n, k):
	if k == 0:
		return True
	if n % 2 == 0 and k == n // 2:
		return False
	for i in range(k):
		if s[i] != s[n - 1 - i]:
			return False
	return True
t = int(input())
for tc in range(t):
	(n, k) = map(int, input().split())
	s = input()
	result = solve(s, n, k)
	print('YES' if result else 'NO')
