#include <iostream>
 
using namespace std;
 
int main() {
    int a, d = 0; cin >> a;
    for (int i = 0; i < a; i++) {
        int b; cin >> b;
        d++;
        for(int j = 2; j * j <= b; j++) {
            if (b % j == 0) {
                d--;
                break;
            }
        }
    }
    cout << d << endl;
}
