import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.OutputStream;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.StringTokenizer;

public class CFTemplate {

	public static class Move {
		int type;
		long c;
		long l;
		int x;
	}
	
	public static void main(String[] args) {
		InputStream inputStream = System.in;
		OutputStream outputStream = System.out;
		InputReader in = new InputReader(inputStream);
		PrintWriter out = new PrintWriter(outputStream);

		int stageCount = in.nextInt();
		Move[] moves = new Move[stageCount];
		for (int i = 0; i < moves.length; i++) {
			Move move = new Move();
			move.type = in.nextInt();
			if (move.type == 1) {
				move.x = in.nextInt();
			} else {
				move.l = in.nextInt();
				move.c = in.nextInt();
			}
			
			moves[i] = move;
		}
		
		int indexCount = in.nextInt();
		long[] indexes = new long[indexCount];
		for (int i = 0; i < indexes.length; i++) {
			indexes[i] = in.nextLong();
		}
		
		List<Integer> seq = new ArrayList<Integer>();
		int currentMove = 0;
		while (seq.size() < 200000 && currentMove < moves.length) {
			if (moves[currentMove].type == 1) {
				seq.add(moves[currentMove].x);
				currentMove++;
			} else {
				for (int i = 0; i < moves[currentMove].l; i++) {
					seq.add(seq.get(i));
				}
				moves[currentMove].c--;
				if (moves[currentMove].c == 0) currentMove++;
			}
		}
		
		long currentSize = seq.size();
		for (int i = 0; i < indexes.length; i++) {
			long ind = indexes[i] - 1;
			
			boolean ok = false;
			while(!ok) {
				if (seq.size() <= ind) {
					Move currMove = moves[currentMove];
					if (ind == (long) seq.size() && currMove.type == 1) {
						ok = true;
						out.print(currMove.x + " ");
						currentMove++;
						currentSize++;
					} else if (currMove.type == 1) {
						currentSize++;
						currentMove++;
					} else if (currMove.type == 2 && (ind < currentSize + currMove.l * currMove.c)) {
						ok = true;
						out.print(seq.get(((int) (ind - currentSize)) % (int) currMove.l) + " ");
					} else {
						currentMove++;
						currentSize += currMove.l * currMove.c;
					}
				} else {
					ok = true;
					out.print(seq.get((int) ind) + " ");
				}
			}
		}
		
		out.close();
	}
}

class InputReader {
	public BufferedReader reader;
	public StringTokenizer tokenizer;

	public InputReader(InputStream stream) {
		reader = new BufferedReader(new InputStreamReader(stream));
		tokenizer = null;
	}

	public String next() {
		while (tokenizer == null || !tokenizer.hasMoreTokens()) {
			try {
				tokenizer = new StringTokenizer(reader.readLine());
			} catch (IOException e) {
				throw new RuntimeException(e);
			}
		}
		return tokenizer.nextToken();
	}

	public int nextInt() {
		return Integer.parseInt(next());
	}

	public long nextLong() {
		return Long.parseLong(next());
	}
}
