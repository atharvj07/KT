n = int(input())
s = list(input().rstrip())
(z, o, t) = (s.count('0') - n // 3, s.count('1') - n // 3, s.count('2') - n // 3)
(zcnt, ocnt, tcnt) = (s.count('0'), s.count('1'), s.count('2'))
(z1, o1, t1) = (0, 0, 0)
for i in range(n):
	if z == o == t == 0:
		break
	if s[i] == '0':
		z1 += 1
		if z <= 0:
			continue
	elif s[i] == '1':
		o1 += 1
		if o <= 0:
			continue
	else:
		if t <= 0:
			continue
		t1 += 1
	if z < 0:
		repl = '0'
	elif o < 0:
		repl = '1'
	elif t < 0:
		repl = '2'
	if int(repl) < int(s[i]):
		if s[i] == '0':
			z -= 1
		elif s[i] == '1':
			o -= 1
		elif s[i] == '2':
			t -= 1
		if repl == '0':
			z += 1
		elif repl == '1':
			o += 1
		else:
			t += 1
		s[i] = repl
	elif int(repl) > int(s[i]):
		if repl == '1':
			if s[i] == '0':
				if zcnt - z1 < z:
					s[i] = repl
					z -= 1
					o += 1
		else:
			if s[i] == '0':
				if zcnt - z1 < z:
					s[i] = repl
					z -= 1
					t += 1
			if s[i] == '1':
				if ocnt - o1 < o:
					s[i] = repl
					o -= 1
					t += 1
print(''.join(s))
