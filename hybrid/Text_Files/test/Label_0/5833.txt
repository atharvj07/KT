#include <bits/stdc++.h>
using namespace std;
template <typename Arg1>
void __f(const char* name, Arg1&& arg1) {
  cerr << name << " : " << arg1 << std::endl;
}
template <typename Arg1, typename... Args>
void __f(const char* names, Arg1&& arg1, Args&&... args) {
  const char* comma = strchr(names + 1, ',');
  cerr.write(names, comma - names) << " : " << arg1 << " | ";
  __f(comma + 1, args...);
}
const int N = 100000;
vector<pair<int, int> > E[N];
vector<pair<int, int> > bfsE[N];
int L0[N], P0[N];
int Ls[N];
int fP[N];
int extra0[N];
int main() {
  int n, m;
  scanf("%d", &n);
  scanf("%d", &m);
  for (int i = 0; i < m; i++) {
    int u, v, w;
    scanf("%d", &u);
    scanf("%d", &v);
    scanf("%d", &w);
    E[u].push_back(make_pair(v, w));
    swap(u, v);
    E[u].push_back(make_pair(v, w));
  }
  memset(L0, -1, sizeof(L0));
  memset(Ls, -1, sizeof(Ls));
  L0[n - 1] = 0;
  P0[n - 1] = n - 1;
  queue<int> bfs;
  bfs.push(n - 1);
  while (!bfs.empty()) {
    int x = bfs.front();
    bfs.pop();
    for (auto c : E[x]) {
      if (L0[c.first] == -1 and c.second == 0) {
        L0[c.first] = 1 + L0[x];
        P0[c.first] = x;
        bfs.push(c.first);
      }
    }
  }
  Ls[0] = 0;
  bfs.push(0);
  while (!bfs.empty()) {
    int x = bfs.front();
    bfs.pop();
    for (auto c : E[x]) {
      if (Ls[c.first] == -1) {
        Ls[c.first] = 1 + Ls[x];
        bfs.push(c.first);
      }
      if (Ls[c.first] == 1 + Ls[x] and (L0[x] == -1 or L0[c.first] == -1))
        bfsE[c.first].push_back(make_pair(x, c.second));
    }
  }
  memset(extra0, -1, sizeof(extra0));
  int mnl = n;
  for (int i = 0; i < n; i++)
    if (L0[i] != -1) mnl = min(mnl, Ls[i]);
  for (int i = 0; i < n; i++)
    if (L0[i] != -1 and Ls[i] == mnl) {
      extra0[i] = L0[i];
      bfs.push(i);
      fP[i] = i;
    }
  string cs = "";
  while (!bfs.empty()) {
    vector<int> lvl;
    while (!bfs.empty()) {
      int x = bfs.front();
      bfs.pop();
      lvl.push_back(x);
    }
    int mn = 10;
    for (auto x : lvl)
      for (auto c : bfsE[x]) mn = min(mn, c.second);
    if (mn == 10) break;
    cs += (mn + '0');
    for (auto x : lvl)
      for (auto c : bfsE[x])
        if (c.second == mn) {
          if (extra0[c.first] == -1 or extra0[c.first] > extra0[x]) {
            if (extra0[c.first] == -1) bfs.push(c.first);
            extra0[c.first] = extra0[x];
            fP[c.first] = x;
          }
        }
  }
  if (cs.empty())
    cout << 0 << endl;
  else
    cout << cs << endl;
  cout << cs.length() + extra0[0] + 1 << endl;
  int x = 0;
  while (1) {
    if (L0[x] != -1) break;
    printf("%d ", x);
    x = fP[x];
  }
  while (1) {
    printf("%d ", x);
    if (P0[x] == x) break;
    x = P0[x];
  }
  cout << endl;
  return 0;
}
