#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
bool ok(char c, char b, char a) {
  string res = "";
  res += c;
  res += b;
  res += a;
  sort(res.begin(), res.end());
  return res == "ABC";
}
string get(string x) {
  for (int i = 0; i < x.size(); ++i) {
    if (x[i] == '.') x[i] = '1';
  }
  return x;
}
void solve(const string& msg) {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  bool found = false;
  int x, y;
  for (int i = 0; i < n; ++i) {
    for (int j = i; j < n; j += k) {
      if (s[j] != s[i]) {
        found = true;
        x = i;
        y = j;
        break;
      }
    }
  }
  if (found) {
    if (s[x] == '.') {
      s[x] = s[y] == '1' ? '0' : '1';
    } else if (s[y] == '.') {
      s[y] = s[x] == '1' ? '0' : '1';
    }
    for (int i = 0; i < n; ++i) {
      if (s[i] == '.') {
        s[i] = '0';
      }
    }
    cout << s << endl;
    cerr << msg << endl;
  } else {
    for (int i = 0; i < n; ++i) {
      for (int j = i + k; j < n; j += k) {
        if (s[i] == '.' and s[i] == s[j]) {
          s[i] = '1';
          s[j] = '0';
          for (int id = 0; id < n; ++id) {
            if (s[id] == '.') {
              s[id] = '1';
            }
          }
          cout << s << endl;
          return;
        }
      }
    }
    cout << "No" << endl;
  }
}
int main() {
  ios_base ::sync_with_stdio(false);
  cin.tie(NULL);
  solve(" Successfully done! ");
  return 0;
}
