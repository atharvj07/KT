//package codeforces.cf153;
import java.io.*;

public class ProblemD {
    BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
    PrintWriter writer = new PrintWriter(new OutputStreamWriter(System.out));

    int[] readInts() throws IOException {
        String[] strings = reader.readLine().split(" ");
        int[] ints = new int[strings.length];
        for(int i = 0; i < ints.length; i++) {
            ints[i] = Integer.parseInt(strings[i]);
        }
        return ints;
    }

    long[] readLongs() throws IOException {
        String[] strings = reader.readLine().split(" ");
        long[] ints = new long[strings.length];
        for(int i = 0; i < ints.length; i++) {
            ints[i] = Long.parseLong(strings[i]);
        }
        return ints;
    }

    boolean check(int[][] a) {
        int m = a.length;
        int n = a[0].length;
        for(int i = 0; i < m - 1; i++) {
            int j = 0;
            while(j < n && a[i][j] == 0) j++;
            if(j < n && a[m - 1][j] == 1) {
                for(int k = j; k < n; k++) {
                    a[m - 1][k] ^= a[i][k];
                }
            }
        }
        int j = 0;
        while(j < n && a[m - 1][j] == 0) j++;
        if(j < n - 1) {
            for(int i = 0; i < m - 1; i++) {
                if(a[i][j] == 1) {
                    for(int k = j; k < n; k++) {
                        a[i][k] ^= a[m - 1][k];
                    }
                }
            }
        }
        return j != n - 1;
    }

    int[][] append(int[][]a, int[] b) {
        if(a == null) {
            return new int[][]{b};
        }
        int[][] ret = new int[a.length + 1][];
        for(int i = 0; i < a.length; i++) {
            ret[i] = a[i];
        }
        ret[ret.length - 1] = b;
        return ret;
    }

    boolean isBit(long a, int bit) {
        return (a & (1L << bit)) > 0;
    }

    int[] makeEquation(long[] nums, int bit) {
        int n = nums.length;
        int[] ret = new int[n + 1];
        ret[n] = 1;
        for(int i = 0; i < n; i++) {
            if(isBit(nums[i], bit)) {
                ret[i] = 1;
                ret[n] ^= 1;
            }
        }
        return ret;
    }

    void solve() throws IOException {
        int n = readInts()[0];
        long[] a = readLongs();
        long or = 0, xor = 0;
        for(long num: a) {
            or |= num;
            xor ^= num;
        }
        int[][] eq = null;
        for(int bit = 62; bit >= 0; bit--) {
            if(isBit(or, bit) && !isBit(xor, bit)) {
                int[][] eq2 = append(eq, makeEquation(a, bit));
                if(check(eq2)) {
                    eq = eq2;
                }
            }
        }
        for(int bit = 62; bit >= 0; bit--) {
            if(isBit(or, bit) && isBit(xor, bit)) {
                int[][] eq2 = append(eq, makeEquation(a, bit));
                if(check(eq2)) {
                    eq = eq2;
                }
            }
        }
        int ans[] = new int[n];
        for(int i = 0; i < n; i++) ans[i] = 2;
        for(int i = 0; i < eq.length; i++) {
            int j = 0;
            while(j < n && eq[i][j] == 0) j++;
            if(j < n) ans[j] = 2 - eq[i][n];
        }
        for(int num: ans) {
            writer.printf("%d ", num);
        }
        writer.println();
        writer.flush();
    }

    public static void main(String[] args) throws IOException{
        new ProblemD().solve();
    }
}
