#include <bits/stdc++.h>
using namespace std;
vector<pair<int, int>> res;
int n, k, nl;
vector<int> p, q;
void Input() {
  cin >> n >> k;
  nl = n - k;
}
void Solve() {
  nl--;
  int x = 1, len = 0;
  p.push_back(1);
  while (nl >= k) {
    int d = k;
    len++;
    while (d--) {
      nl--;
      x++;
      res.push_back({p.back(), x});
      q.push_back(x);
      if (p.size() > 1) p.pop_back();
    }
    p.swap(q);
    q.clear();
  }
  len = (len + 1) * 2;
  len += min(2, nl);
  if (p.size() == 1) {
    for (int i = 0; i < nl; i++) {
      x++;
      res.push_back({p[0], x});
      p.push_back(x);
    }
  } else {
    for (int i = 0; i < nl; i++) {
      x++;
      res.push_back({p[i], x});
      p[i] = x;
    }
  }
  while (k--) {
    x++;
    res.push_back({p.back(), x});
    if (p.size() > 1) p.pop_back();
  }
  cout << len << '\n';
  for (auto a : res) cout << a.first << ' ' << a.second << '\n';
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  Input();
  Solve();
}
