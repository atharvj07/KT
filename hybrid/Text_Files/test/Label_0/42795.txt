import sys
input = sys.stdin.readline
from collections import Counter

def convert(diff, k):
	(lena, lenb) = (0, 0)
	for letter in sorted(diff.keys()):
		n_letters = diff[letter]
		if n_letters > 0:
			lenb += n_letters
		else:
			lena += abs(n_letters)
		if lena < lenb:
			return 'NO'
	for (key, val) in diff.items():
		if abs(val) % k != 0:
			return 'NO'
	return 'YES'
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	a = Counter(input()[:-1])
	b = Counter(input()[:-1])
	if a == b:
		print('YES')
		continue
	b.subtract(a)
	print(convert(b, k))
