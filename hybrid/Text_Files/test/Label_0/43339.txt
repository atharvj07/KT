cases = int(input())
for z in range(cases):
	(n, k) = input().split()
	n = int(n)
	k = int(k)
	if n == k:
		for i in range(1, n + 1):
			print(i, end=' ')
		print('')
	else:
		num = 2 * k - n
		for i in range(1, num):
			print(i, end=' ')
		temp = k
		for i in range(num, k + 1):
			print(temp, end=' ')
			temp -= 1
		print('')
