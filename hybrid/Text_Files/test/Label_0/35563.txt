import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.Arrays;
import java.util.Comparator;
import java.util.Map.Entry;
import java.util.StringTokenizer;
import java.util.TreeMap;

public class Main {

	private void solve() throws IOException {

		TreeMap<Integer, Double> mp = new TreeMap<Integer, Double>();

		int n = ni();
		int m = ni();
		for (int i = 0; i < n; i++) {
			int a = ni();
			int h = ni();
			double l = ni();
			if (l == 100)
				l = 99.999;
			double r = ni();
			if (r == 100)
				r = 99.999;
			mp.put(a - h, mp.getOrDefault(a - h, 0.0) + Math.log(1 - l / 100));
			mp.put(a, mp.getOrDefault(a, 0.0) - Math.log(1.0 - l / 100));
			mp.put(a + 1, mp.getOrDefault(a + 1, 0.0) + Math.log(1.0 - r / 100));
			mp.put(a + h + 1, mp.getOrDefault(a + h + 1, 0.0) - Math.log(1.0 - r / 100));
			// Helper.tr(mp);
		}
		mp.put(Integer.MAX_VALUE, 0.0);
		mp.put(Integer.MIN_VALUE, 0.0);
		int a[][] = new int[m][2];
		for (int i = 0; i < m; i++) {
			a[i][0] = ni();
			a[i][1] = ni();
		}
		Arrays.sort(a, new Comparator<int[]>() {
			public int compare(int[] o1, int[] o2) {
				return o1[0] - o2[0];
			}
		});

		int i = 0;
		double ans = 0;
		double curProb = 0;
		Entry<Integer, Double>[] probs = mp.entrySet().toArray(new Entry[0]);
		for (int j = 0; j < probs.length; j++) {
			curProb += probs[j].getValue();
			// Helper.tr(curProb, i,j );
			for (; i < a.length && a[i][0] < probs[j + 1].getKey(); i++) {
				ans += Math.exp(curProb) * a[i][1];
				// Helper.tr(i,a[i][1],curProb, probs[j].getKey(), ans);
			}
		}

		out.println(ans);
		out.close();

	}

	static BufferedReader in;
	static PrintWriter out;
	static StringTokenizer tok;

	private int[][] na(int n) throws IOException {
		int[][] a = new int[n][2];
		for (int i = 0; i < n; i++) {
			a[i][0] = ni();
			a[i][1] = i;
		}
		return a;
	}

	String ns() throws IOException {
		while (!tok.hasMoreTokens()) {
			tok = new StringTokenizer(in.readLine(), " ");
		}
		return tok.nextToken();
	}

	int ni() throws IOException {
		return Integer.parseInt(ns());
	}

	long nl() throws IOException {
		return Long.parseLong(ns());
	}

	double nd() throws IOException {
		return Double.parseDouble(ns());
	}

	String[] nsa(int n) throws IOException {
		String[] res = new String[n];
		for (int i = 0; i < n; i++) {
			res[i] = ns();
		}
		return res;
	}

	int[] nia(int n) throws IOException {
		int[] res = new int[n];
		for (int i = 0; i < n; i++) {
			res[i] = ni();
		}
		return res;
	}

	long[] nla(int n) throws IOException {
		long[] res = new long[n];
		for (int i = 0; i < n; i++) {
			res[i] = nl();
		}
		return res;
	}

	public static void main(String[] args) throws IOException {
		in = new BufferedReader(new InputStreamReader(System.in));
		out = new PrintWriter(System.out);
		tok = new StringTokenizer("");
		Main main = new Main();
		main.solve();
		out.close();
	}
}
