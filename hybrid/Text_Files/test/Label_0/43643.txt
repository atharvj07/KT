N = int(input())
a = input().split()
put = []
for i in a:
	put.append(int(i))
average = sum(put) / N
step1 = 0
step2 = 0
for i in put:
	if int(average) == average:
		if i > average:
			step1 += i - average
		else:
			step2 += average - i
	if int(average) != average:
		if i > int(average) + 1:
			step1 += i - int(average + 1)
		if i < int(average):
			step2 += int(average) - i
print(int(max(step1, step2)))
