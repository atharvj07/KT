public class Es
{
	static class P implements Comparable
	{
		int position;
		char color;
		
		public P(int p, char c)
		{
			position = p;
			color = c;
		}
		
		public int compareTo(Object o)
		{
			P p2 = (P)o;
			if(this.position < p2.position)
				return -1;
			if(this.position > p2.position)
				return 1;
			return 0;
		}
	}
	
    public static void main(String[] args) throws IOException
    {
    	long n = in.nextInt();
    	Vector <Long> r = new Vector <Long> ();
    	Vector <Long> g = new Vector <Long> ();
    	Vector <Long> b = new Vector <Long> ();
    	for(int i = 0; i < n; i ++)
    	{
    		long p = in.nextLong();
    		char c = in.next().charAt(0);
    		if(c == 'R')
    			r.add(p);
    		else if(c == 'G')
    			g.add(p);
    		else
    			b.add(p);
    	}
    	long ans = 0;
    	if(g.size() == 0)
    	{
    		if(r.size() > 1)
    			ans += r.get(r.size() - 1) - r.get(0);
    		if(b.size() > 1)
    			ans += b.get(b.size() - 1) - b.get(0);
    		sop(ans);
    		return;
    	}
    	if(r.size() > 0)
	    	if(r.get(0) < g.get(0))
	    		ans += g.get(0) - r.get(0);
    	if(b.size() > 0)
	    	if(b.get(0) < g.get(0))
	    		ans += g.get(0) - b.get(0);
    	int j = 0, z = 0;
    	for(int i = 0; i < g.size() - 1; i ++)
    	{
    		long cnt_1 = Long.MAX_VALUE;
    		for(; j < r.size(); j ++)
    		{
    			if(r.get(j) > g.get(i + 1))
    				break;
    			if(r.get(j) < g.get(i))
    				continue;
    			long extra = 0;
    			if(j < r.size() - 1)
    				if(r.get(j + 1) < g.get(i + 1))
    					extra = g.get(i + 1) - r.get(j + 1);
    			cnt_1 = Math.min(cnt_1, r.get(j) - g.get(i) + extra);
    			extra = 0;
    			if(j > 0)
    				if(r.get(j - 1) > g.get(i))
    					extra = r.get(j - 1) - g.get(i);
    			cnt_1 = Math.min(cnt_1, g.get(i + 1) - r.get(j) + extra);
    		}
    		if(cnt_1 == Long.MAX_VALUE)
    			cnt_1 = 0;
    		long cnt_2 = Long.MAX_VALUE;
    		for(; z < b.size(); z ++)
    		{
    			if(b.get(z) > g.get(i + 1))
    				break;
    			if(b.get(z) < g.get(i))
    				continue;
    			long extra = 0;
    			if(z < b.size() - 1)
    				if(b.get(z + 1) < g.get(i + 1))
    					extra = g.get(i + 1) - b.get(z + 1);
    			cnt_2 = Math.min(cnt_2, b.get(z) - g.get(i) + extra);
    			extra = 0;
    			if(z > 0)
    				if(b.get(z - 1) > g.get(i))
    					extra = b.get(z - 1) - g.get(i);
    			cnt_2 = Math.min(cnt_2, g.get(i + 1) - b.get(z) + extra);
    		}
    		if(cnt_2 == Long.MAX_VALUE)
    			cnt_2 = 0;
    		ans += Math.min(g.get(i + 1) - g.get(i) + cnt_1 + cnt_2, 2 * (g.get(i + 1) - g.get(i)));
    	}
    	if(r.size() > 0)
	    	if(r.get(r.size() - 1) > g.get(g.size() - 1))
	    		ans += r.get(r.size() - 1) - g.get(g.size() - 1);
    	if(b.size() > 0)
	    	if(b.get(b.size() - 1) > g.get(g.size() - 1))
	    		ans += b.get(b.size() - 1) - g.get(g.size() - 1);
    	sop(ans);
    }
    
	static class FastReader
    {
        BufferedReader br;
        StringTokenizer st;
 
        public FastReader()
        {
            br = new BufferedReader(new InputStreamReader(System.in));
        }
        
        String next()
        {
            while (st == null || !st.hasMoreElements())
            {
                try
                {
                    st = new StringTokenizer(br.readLine());
                }
                catch (IOException  e)
                {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
 
        int nextInt()
        {
            return Integer.parseInt(next());
        }
 
        long nextLong()
        {
            return Long.parseLong(next());
        }
 
        double nextDouble()
        {
            return Double.parseDouble(next());
        }
 
        String nextLine()
        {
            String str = "";
            try
            {
                str = br.readLine();
            }
            catch (IOException e)
            {
                e.printStackTrace();
            }
            return str;
        }
    }
	
    static FastReader in = new FastReader();
	
    public static void sop(Object o)
    {
    	System.out.print(o);
    }
