import sys
input = lambda : sys.stdin.readline().strip()
t = int(input())
while t:
	t -= 1
	n = int(input())
	a = list(map(int, input().split()))
	cnt_l = [0] * (n + 1)
	ans = 0
	for i in range(n):
		cnt_r = [0] * (n + 1)
		for j in range(n - 1, i, -1):
			ans += cnt_l[a[j]] * cnt_r[a[i]]
			cnt_r[a[j]] += 1
		cnt_l[a[i]] += 1
	print(ans)
