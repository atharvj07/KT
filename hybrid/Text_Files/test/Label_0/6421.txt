vector<pair<int,int> > all;
vector<pair<int,int> > bi[75];
void add(int x,int y)
{
	all.push_back(make_pair(x,y));
	if (prex[x]) all.push_back(make_pair(prex[x],y));
	if (nxtx[x]) all.push_back(make_pair(nxtx[x],y));
	if (prey[y]) all.push_back(make_pair(x,prey[y]));
	if (nxty[y]) all.push_back(make_pair(x,nxty[y]));
	if (prex[prex[x]]) all.push_back(make_pair(prex[prex[x]],y));
	if (nxtx[nxtx[x]]) all.push_back(make_pair(nxtx[nxtx[x]],y));
	if (prey[prey[y]]) all.push_back(make_pair(x,prey[prey[y]]));
	if (nxty[nxty[y]]) all.push_back(make_pair(x,nxty[nxty[y]]));
	if (prex[x]&&prey[y]) all.push_back(make_pair(prex[x],prey[y]));
	if (prex[x]&&nxty[y]) all.push_back(make_pair(prex[x],nxty[y]));
	if (nxtx[x]&&prey[y]) all.push_back(make_pair(nxtx[x],prey[y]));
	if (nxtx[x]&&nxty[y]) all.push_back(make_pair(nxtx[x],nxty[y]));
}
ll dijkstra(int x,int y)
{
	memset(dis,18,sizeof(dis));
	dis[x]=0;
	priority_queue<pair<ll,ll> > pq;
	pq.push(make_pair(0,x));
	while (!pq.empty())
	{
		x=pq.top().second;pq.pop();
		foreach(it,bi[x])
		{
			if (dis[it->fi]>dis[x]+it->se)
			{
				dis[it->fi]=dis[x]+it->se;
				pq.push(make_pair(-dis[it->fi],it->fi));
			}
		}
	}
	if (dis[y]==dis[54]) return -1;
	return dis[y];
}
void solve(int x,int y,int xx,int yy)
{
	sort(all.begin(),all.end());
	all.resize(unique(all.begin(),all.end())-all.begin());
//	foreach(it,all) cerr<<it->fi<<' '<<it->se<<endl;
	int i,j,ss,tt;
	for (i=0;i<all.size();i++) if (all[i].first==x&&all[i].second==y) ss=i;
	for (i=0;i<all.size();i++) if (all[i].first==xx&&all[i].second==yy) tt=i;
	for (i=0;i<all.size();i++) bi[i].clear();
	for (i=0;i<all.size();i++) for (j=0;j<all.size();j++) if (i!=j)
	{
		if (all[i].first==all[j].first&&((s[all[i].first]=='E')==(all[i].second<all[j].second)))
		{
			bi[i].push_back(make_pair(j,abs(all[i].second-all[j].second)));
		}
		if (all[i].second==all[j].second&&((t[all[i].second]=='S')==(all[i].first<all[j].first)))
		{
			bi[i].push_back(make_pair(j,abs(all[i].first-all[j].first)));
		}
	}
	printf("%lld\n",dijkstra(ss,tt));
}

	lst1=lst2=0;
	rep(i,n){if (s[i]=='W'){prex[i]=lst2;lst1=i;}else{prex[i]=lst1;lst2=i;}}
	lst1=lst2=0;
	rep2(i,n){if (s[i]=='W'){nxtx[i]=lst2;lst1=i;}else{nxtx[i]=lst1;lst2=i;}}
	lst1=lst2=0;
	rep(i,m){if (t[i]=='N'){prey[i]=lst2;lst1=i;}else{prey[i]=lst1;lst2=i;}}
	lst1=lst2=0;
	rep2(i,m){if (t[i]=='N'){nxty[i]=lst2;lst1=i;}else{nxty[i]=lst1;lst2=i;}}
	while (q--)
	{
		scanf("%d%d%d%d",&x,&y,&xx,&yy);

		add(x,y);add(x,yy);add(xx,y);add(xx,yy);
