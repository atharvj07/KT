#include <iostream>
#include <algorithm>
#define rep(i, n) for(i = 0; i < n; i++)
#define int long long
using namespace std;

const int DEPTH = 17;
const int INF = 1e+16;
struct SegTree {
	int SegSumMax[1 << (DEPTH + 1)];
	int SegSumMaxL[1 << (DEPTH + 1)];
	int SegSumMaxR[1 << (DEPTH + 1)];
	int Sum[1 << (DEPTH + 1)];
	
	SegTree() {
		int i;
		rep(i, (1 << (DEPTH + 1))) {
			SegSumMax[i] = -INF;
			SegSumMaxL[i] = -INF;
			SegSumMaxR[i] = -INF;
			Sum[i] = 0;
		}
	}
	
	void update(int pos, int x) {
		pos += (1 << DEPTH) - 1;
		SegSumMax[pos] = x;
		SegSumMaxL[pos] = x;
		SegSumMaxR[pos] = x;
		Sum[pos] = x;
		while (pos > 0) {
			pos = (pos - 1) / 2;
			SegSumMax[pos] = max({	SegSumMax[pos * 2 + 1],
									SegSumMax[pos * 2 + 2],
									SegSumMaxR[pos * 2 + 1] + SegSumMaxL[pos * 2 + 2]});
			SegSumMaxL[pos] = max(SegSumMaxL[pos * 2 + 1], Sum[pos * 2 + 1] + SegSumMaxL[pos * 2 + 2]);
			SegSumMaxR[pos] = max(SegSumMaxR[pos * 2 + 2], Sum[pos * 2 + 2] + SegSumMaxR[pos * 2 + 1]);
			Sum[pos] = Sum[pos * 2 + 1] + Sum[pos * 2 + 2];
		}
	}
	
	int query() {
		return max(0LL, SegSumMax[0]);
	}
};

int n, q;
int a[100000];
int k, x;
SegTree seg;

signed main() {
	int i;
	
	cin >> n >> q;
	rep(i, n) cin >> a[i];
	rep(i, n) seg.update(i, a[i]);
	cout << seg.query() << endl;
	
	rep(i, q) {
		cin >> k >> x;
		k--;
		seg.update(k, x);
		cout << seg.query() << endl;
	}
	return 0;
}
