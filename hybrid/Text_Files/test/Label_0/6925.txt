#include <bits/stdc++.h>
using namespace std;
class testcase {
  vector<map<int, int>> m;
  vector<int> nxt;
  vector<long long> dp;

 public:
  long long solve(int n, vector<int>& a) {
    nxt.resize(n, -1);
    m.resize(n + 1);
    dp.resize(n + 1, 0);
    for (int i = n - 1; i >= 0; i--) {
      nxt[i] = (m[i + 1].count(a[i]) ? m[i + 1][a[i]] + 1 : -1);
      if (nxt[i] != -1) {
        m[i].swap(m[nxt[i] + 1]);
        if (nxt[i] + 1 < n) m[i][a[nxt[i] + 1]] = nxt[i];
      }
      m[i][a[i]] = i - 1;
    }
    long long ans = 0;
    for (int i = n - 1; i >= 0; i--) {
      if (nxt[i] != -1) dp[i] = dp[nxt[i] + 1] + 1;
      ans += dp[i];
    }
    return ans;
  }
};
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    testcase tc;
    cout << tc.solve(n, a) << "\n";
  }
  return 0;
}
