import java.io.*;
import java.util.*;
import java.math.*;

public class Main {
	public static void main(String[] args) 
		throws IOException
	{
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int k = in.nextInt();

		System.out.println((6*n - 1) * k);

		for (int i = 0; i < n; i++) {
			System.out.println(k * (6*i + 1) + " " + k * (6*i + 2) + " " + k * (6*i + 3) + " " + k * (6*i + 5));
		}

		in.close();
	}
}
