#include <bits/stdc++.h>
using namespace std;
const int MAXN = 200010;
const int LOGN = 21;
const int MOD = 998244353;
vector<int> x[2][MAXN];
int f[MAXN][LOGN];
pair<int, int> g[MAXN][2];
int p2[MAXN];
void Dijkstra1(int n) {
  for (int i = 1; i <= n; ++i) {
    for (int j = 0; j < LOGN; ++j) {
      f[i][j] = MAXN;
    }
  }
  for (int j = 0; j < LOGN; ++j) {
    set<pair<int, int>> s;
    if (j == 0) {
      f[1][0] = 0;
      s.insert(make_pair(0, 1));
    } else {
      for (int i = 1; i <= n; ++i) {
        if (f[i][j - 1] == MAXN) continue;
        f[i][j] = f[i][j - 1];
        s.insert(make_pair(f[i][j], i));
      }
    }
    const int o = j % 2;
    while (!s.empty()) {
      auto [dist, i] = *s.begin();
      s.erase(s.begin());
      if (dist > f[i][j]) continue;
      for (int t : x[o][i]) {
        if (f[t][j] > f[i][j] + 1) {
          f[t][j] = f[i][j] + 1;
          s.insert(make_pair(f[t][j], t));
        }
      }
    }
  }
}
void Dijkstra2(int n) {
  for (int i = 1; i <= n; ++i) {
    g[i][0] = g[i][1] = make_pair(MAXN, MAXN);
  }
  set<tuple<int, int, int, int>> s;
  g[1][0] = make_pair(0, 0);
  s.insert(make_tuple(0, 0, 1, 0));
  while (!s.empty()) {
    auto [rv, dist, i, j] = *s.begin();
    s.erase(s.begin());
    if (make_pair(rv, dist) > g[i][j]) continue;
    for (int t : x[j][i]) {
      if (g[t][j] > make_pair(rv, dist + 1)) {
        g[t][j] = make_pair(rv, dist + 1);
        s.insert(make_tuple(rv, dist + 1, t, j));
      }
    }
    if (g[i][j ^ 1] > make_pair(rv + 1, dist)) {
      g[i][j ^ 1] = make_pair(rv + 1, dist);
      s.insert(make_tuple(rv + 1, dist, i, j ^ 1));
    }
  }
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < m; ++i) {
    int a, b;
    cin >> a >> b;
    x[0][a].push_back(b);
    x[1][b].push_back(a);
  }
  Dijkstra1(n);
  p2[0] = 0;
  p2[1] = 1;
  for (int i = 2; i < MAXN; ++i) {
    p2[i] = (p2[i - 1] + p2[i - 1]) % MOD;
  }
  int ans = MOD;
  for (int j = 0; j < LOGN; ++j) {
    if (f[n][j] != MAXN) ans = min(ans, (f[n][j] + p2[j + 1] - 1) % MOD);
  }
  if (ans != MOD) {
    cout << ans << endl;
    return 0;
  }
  Dijkstra2(n);
  if (g[n][0] < g[n][1]) {
    ans = (g[n][0].second + p2[g[n][0].first + 1] - 1) % MOD;
  } else {
    ans = (g[n][1].second + p2[g[n][1].first + 1] - 1) % MOD;
  }
  cout << ans << endl;
  return 0;
}
