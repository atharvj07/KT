Char = 'WB'
Index = 0
First = 0
X = list(map(int, input().split()))
Sentence = []
for i in range(X[0]):
	Index = (First + 1) % 2
	Temp = list(input())
	for j in range(X[1]):
		if Temp[j] == '.':
			Temp[j] = Char[Index]
		Index = (Index + 1) % 2
	Sentence.append(Temp)
	First = (First + 1) % 2
for i in range(X[0]):
	print(*Sentence[i], sep='')
