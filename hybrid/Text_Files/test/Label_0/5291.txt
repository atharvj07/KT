#include<bits/stdc++.h>
using namespace std;

using Graph=vector<vector<int>>;
using ll=__int128;

class Solver{
    int n;
    Graph g;
    vector<int> cnt;
    const ll INF=1e20;
    int dfs(int v,int pre,int dep){
        cnt[dep]=max(cnt[dep],(int)(g[v].size()-1));
        int res=dep;
        for(auto to:g[v]){
            if(to!=pre) res=max(res,dfs(to,v,dep+1));
        }
        return res;
    }
    pair<int,ll> calc(vector<int> vs,vector<int> pre){
        cnt.assign(n,1);
        int dep=0;
        for(auto v:vs){
            int prev=-1;
            for(int i=0;i<pre.size();i++) if(pre[i]!=v) prev=pre[i];
            dep=max(dep,dfs(v,prev,0)+1);
        }
        if(pre.size()==1) dep++;
        ll res=1;
        for(int i=0;i<n;i++) res=min(res*cnt[i],INF);
        return make_pair(dep,res*vs.size());
    }
    public:
    Solver(int n):n(n),g(n){
        for(int i=0;i+1<n;i++){
            int a,b;
            cin>>a>>b;
            a--,b--;
            g[a].push_back(b);
            g[b].push_back(a);
        }
    }
    pair<int,ll> solve(){
        pair<int,ll> res={n+1,INF};
        for(int i=0;i<n;i++){
            vector<int> vs;
            for(auto to:g[i]){
                vs.push_back(to);
            }
            res=min(res,calc(vs,vector<int>{i}));
            for(auto to:g[i]){
                res=min(res,calc(vector<int>{i,to},vector<int>{i,to}));
            }
        }
        return res;
    }
};

int main(){
    int n;
    cin>>n;
    Solver solver(n);
    auto ret=solver.solve();
    cout<<ret.first<<" "<<(long long)(ret.second)<<endl;
}
