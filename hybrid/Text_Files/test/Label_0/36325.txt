import java.io.*;
import java.util.*;
public class Competencia2 {
	public static void main(String[] args) {
		  int modulo = 1000000007;
		  int max = 1000;
		  Scanner scan = new Scanner(System.in);
		    
		  long[][] combinatoria = new long[max][max];
		  combinatoria[0][0] = 1;
		  for (int i = 1; i < max; i++) {
			  combinatoria[i][0] = 1;
			  	for (int j = 1; j <= i; j++) {
			  		combinatoria[i][j] = (combinatoria[i-1][j] + combinatoria[i-1][j-1]) % modulo;
			  	}
		    }
		    
		    int K = scan.nextInt();
		    int[] color = new int[K];
		    for (int i = 0; i < K; i++) color[i] = scan.nextInt();
		    
		    long res = 1;
		    int total = 0;
		    for (int i = 0; i < K; i++) {
		      res = (res * combinatoria[total + color[i] - 1][color[i] - 1]) % modulo;
		      total += color[i];
		    }
		    
		   System.out.println(res);
		   scan.close();
	}
}
