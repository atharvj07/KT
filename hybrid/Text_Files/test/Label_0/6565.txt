#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  string a, b, result;
  int nb = 0;
  cin >> t >> a >> b;
  for (int i = b.size() - 1; i >= 0; i--) {
    int j = 0;
    for (; j <= i && b[i] != a[j]; j++)
      ;
    if (j == i + 1) {
      nb = -1;
      break;
    }
    for (; j < i; j++) {
      nb++;
      result += to_string(j + 1) + " ";
      char temp = a[j];
      a[j] = a[j + 1];
      a[j + 1] = temp;
    }
  }
  if (nb != -1)
    cout << nb << endl << result << endl;
  else
    cout << nb << endl;
  return 0;
}
