from operator import itemgetter
(n, m) = map(int, input().split())
food = list(map(int, input().split()))
cost = list(map(int, input().split()))
arr = [0] * n
for i in range(n):
	arr[i] = [0] * 2
	arr[i][0] = i
	arr[i][1] = cost[i]
arr = sorted(arr, key=itemgetter(1))
index = 0
for i in range(m):
	ans = 0
	(t, d) = map(int, input().split())
	if food[t - 1] >= d:
		ans += cost[t - 1] * d
		food[t - 1] -= d
		d = 0
	else:
		d -= food[t - 1]
		ans += cost[t - 1] * food[t - 1]
		food[t - 1] = 0
		if d > 0:
			ind = index
			for j in range(ind, n):
				y = min(d, food[arr[j][0]])
				ans += arr[j][1] * y
				food[arr[j][0]] -= y
				d -= y
				if food[arr[j][0]] == 0:
					index += 1
				if d == 0:
					break
	if d == 0:
		print(ans)
	else:
		print(0)
