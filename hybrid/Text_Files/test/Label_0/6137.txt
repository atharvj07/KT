#include <bits/stdc++.h>
using namespace std;
using namespace std;
long long n;
bool a[20][20];
vector<long long> ans(50000);
vector<long long> v[20][50000];
void work() {
  cin >> n;
  for (long long i = 0; i < n; i++) {
    for (long long j = 0; j < n; j++) {
      char c;
      cin >> c;
      a[i][j] = (c == '1');
    }
  }
  long long m = (1 << n);
  for (long long i = 0; i < m; i++) {
    long long nm = 0;
    for (long long j = 0; j < n; j++) {
      if (i & (1 << j)) {
        nm += 1;
      }
    }
    for (long long j = 0; j < n; j++) {
      if (!(i & (1 << j))) {
        long long ch = i | (1 << j);
        if (v[j][ch].empty()) v[j][ch] = vector<long long>(1 << nm);
        if (nm == 0) {
          v[j][ch][0] = 1;
          continue;
        }
        for (long long k = 0; k < n; k++) {
          if (k == j) continue;
          if (i & (1 << k)) {
            long long f = (long long)(a[j][k]) * (1 << (nm - 1));
            for (long long b = 0; b < (1 << (nm - 1)); b++) {
              long long b1 = b + f;
              v[j][ch][b1] += v[k][i][b];
            }
          }
        }
      }
    }
    for (long long j = 0; j < n; j++) {
      if (i == m - 1) {
        for (long long b = 0; b < m / 2; b++) {
          ans[b] += v[j][i][b];
        }
      }
      v[j][i].clear();
    }
  }
  for (long long i = 0; i < m / 2; i++) {
    cout << ans[i] << " ";
  }
  cout << endl;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long t;
  t = 1;
  for (long long i = 0; i < t; i++) {
    work();
  }
}
