n = int(input())
l = list(map(int, input().split()))
k = 0
k1 = 0
z = 0
t = 1
while k1 < n:
	if t == 1:
		for i in range(n):
			if l[i] <= z and l[i] != -1:
				z = z + 1
				k1 = k1 + 1
				l[i] = -1
	if t == 2:
		for i in range(n - 1, -1, -1):
			if l[i] <= z and l[i] != -1:
				z = z + 1
				k1 = k1 + 1
				l[i] = -1
	if t == 2:
		t = 1
	else:
		t = 2
	if k1 >= n:
		break
	else:
		k = k + 1
print(k)
