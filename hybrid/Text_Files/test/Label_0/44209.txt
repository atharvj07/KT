import sys
import collections
import math
import bisect
import heapq
inf = sys.maxsize

def get_ints():
	return map(int, sys.stdin.readline().strip().split())

def get_array():
	return list(map(int, sys.stdin.readline().strip().split()))

def input():
	return sys.stdin.readline().strip()
mod = 1000000007
for _ in range(int(input())):
	(n, k) = get_ints()
	dp = [[0 for i in range(1005)] for j in range(1005)]
	for i in range(1, k + 1):
		for j in range(n + 1):
			if i == 1:
				dp[i][j] = 1
			elif j == 0:
				dp[i][j] = 1
			else:
				dp[i][j] = (dp[i][j - 1] % mod + dp[i - 1][n - j] % mod) % mod
	print(dp[k][n] % mod)
