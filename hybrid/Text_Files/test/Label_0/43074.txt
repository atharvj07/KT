def solution():
	a = list(map(int, list(input())))
	n = len(a)
	answer = len(a) - 2
	counter = [0] * 10
	for ai in a:
		counter[ai] += 1
	for i in range(10):
		answer = min(answer, n - counter[i])
	for i in range(10):
		if counter[i] == 0:
			continue
		for j in range(i + 1, 10):
			if counter[j] == 0:
				continue
			cur_ans = 0
			last = -1
			for k in range(n):
				if a[k] == i:
					if last == i:
						cur_ans += 1
					last = i
				elif a[k] == j:
					if last == j:
						cur_ans += 1
					last = j
				else:
					cur_ans += 1
			if (n - cur_ans) % 2 == 1:
				cur_ans += 1
			answer = min(answer, cur_ans)
	print(answer)
t = int(input())
for _ in range(t):
	solution()
