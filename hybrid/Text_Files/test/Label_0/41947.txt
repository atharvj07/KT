a = list(map(int, input().split()))
b = list(map(int, input().split()))
c = list(map(int, input().split()))
p1 = 0
m = 0
cur = 0
for i in range(a[0] - 1, -1, -1):
	if c[i] == 1:
		p1 += b[i]
	if c[i] == 0:
		cur += b[i]
	if i + a[1] < a[0]:
		if c[i + a[1]] == 0:
			cur -= b[i + a[1]]
			if cur < 0:
				cur = 0
	if cur > m:
		m = cur
print(p1 + m)
