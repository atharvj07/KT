def repaint(n, k, array):
	s = set(array)
	min_ans = n
	for x in s:
		l = 0
		r = k + l - 1
		ans = 0
		while l < n:
			if array[l] == x:
				l += 1
				r = k + l - 1
			else:
				ans += 1
				l = l + k
				r = k + l - 1
		if ans < min_ans:
			min_ans = ans
	return min_ans
t = int(input())
final = []
for _ in range(t):
	(n, k) = map(int, input().split())
	l = list(map(int, input().split()))
	final.append(repaint(n, k, l))
for _ in range(t):
	print(final[_])
