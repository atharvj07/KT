import java.util.*;
import java.io.*;

public class Pmain0 {

	static FastReader input = new FastReader();
	static PrintWriter out = new PrintWriter(System.out);
	static int n, m, a[], mo[];

	public static void main(String[] args) throws NumberFormatException, IOException {

		int t = input.nextInt();
		while (t-- > 0) {
			n = input.nextInt();
			mo = new int[n];

			int bestmo = -1;
			for (int i = 0; i < n; i++) {
				mo[i] = input.nextInt();
				bestmo = Math.max(bestmo, mo[i]);
			}

			m = input.nextInt();
			a = new int[n + 5];
			int best = -1;
			for (int i = 0; i < m; i++) {
				int x = input.nextInt();
				int y = input.nextInt();
				a[y] = Math.max(a[y], x);
				best = Math.max(best, x);
			}
			for (int i = n; i >= 1; i--) { a[i] = Math.max(a[i], a[i + 1]); }
			if (bestmo > best) {
				out.println(-1);
			} else {

				int i = 0;
				int j = 0;
				int ans = 0;
				//out.println(Arrays.toString(a));
				while (i < n) {
					int choice = (int) (1e9) + 10;
					int max = mo[j];

					while (j < n && choice >= max) {
						int e = j - i + 1;
						choice = a[e];
						max = Math.max(max, mo[j]);
						if (choice >= max) {
							j++;
						}
					}

					i = j;
					ans++;

				}
				out.println(ans);

			}

		}
		out.flush();

	}

	static class Shop implements Comparable<Shop> {
		int x, q, r;

		public Shop(int x, int q, int r) {
			this.x = x;
			this.q = q;
			this.r = r;
		}

		@Override
		public int compareTo(Shop o) {
			if (this.r == o.r && this.q == o.q)
				return this.x - o.x;
			else if (this.r == o.r)
				return this.q - o.q;
			return this.r - o.r;
		}
	}

	static class Customer {
		int y, d;

		public Customer(int y, int d) {
			this.y = y;
			this.d = d;
		}
	}

	static ArrayList<Integer> sieve(int n) {
		ArrayList<Integer> ret = new ArrayList<>();
		boolean notPrime[] = new boolean[n + 1];
		for (int i = 2; i * i <= n; i++) {
			if (!notPrime[i]) {
				for (int j = i * i; j <= n; j += i) { notPrime[j] = true; }
			}
		}
		for (int i = 2; i < notPrime.length; i++) {
			if (!notPrime[i]) {
				ret.add(i);
			}
		}
		out.println(ret);
		return ret;
	}

	static class FastReader {
		BufferedReader br;
		StringTokenizer st;

		public FastReader() {
			br = new BufferedReader(new InputStreamReader(System.in));
			try {
				// br = new BufferedReader(new FileReader(new File("business.in")));
			} catch (Exception e) {
				e.printStackTrace();
			}
		}

		String next() throws IOException {
			while (st == null || !st.hasMoreElements()) { st = new StringTokenizer(br.readLine()); }
			return st.nextToken();
		}

		int nextInt() throws NumberFormatException, IOException { return Integer.parseInt(next()); }

		long nextLong() throws NumberFormatException, IOException { return Long.parseLong(next()); }

		double nextDouble() throws NumberFormatException, IOException { return Double.parseDouble(next()); }

		String nextLine() throws IOException {
			String str = "";
			str = br.readLine();
			return str;
		}
	}

	static public class Pair implements Comparable<Pair> {
		int x, y;

		Pair(int x, int y) {
			this.x = x;
			this.y = y;
		}

		@Override
		public int compareTo(Pair o) {
			if (this.x == o.x)
				return o.y - this.y;

			return o.x - this.x;
		}
	}

}
