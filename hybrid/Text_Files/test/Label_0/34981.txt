import java.io.*;
import java.util.*;
import java.lang.*;

public class GFG {
	public static void main (String[] args) {
		Scanner s=new Scanner(System.in);
		int n=s.nextInt();
		String str=s.next();
		char[] ch=str.toCharArray();
		int min=0;
		for(int i=1;i<n;i++){
		    if(ch[i]==ch[i-1]){
		        min++;
		        char c=ch[i];
		        if(c=='R'){
		            if(i==n-1)ch[i]='G';
		            else{
		                if(ch[i+1]=='G')ch[i]='B';
		                else ch[i]='G';
		            }
		        }
		        else if(c=='B'){
		            if(i==n-1)ch[i]='G';
		            else{
		                if(ch[i+1]=='G')ch[i]='R';
		                else ch[i]='G';
		            }
		        }
		        else{
		            if(i==n-1)ch[i]='R';
		            else{
		                if(ch[i+1]=='R')ch[i]='B';
		                else ch[i]='R';
		            }
		        }
		    }
		    
		    
		}
		 System.out.println(min);
		  System.out.println(ch);
		
	}
}
