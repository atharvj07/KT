s = input()
d = {}
for i in s:
	if i in d:
		d[i] += 1
	else:
		d[i] = 1
x = sorted(list(d.keys()))
i = 0
j = len(x) - 1
while i < j:
	if d[x[i]] % 2 != 0 and d[x[j]] % 2 != 0:
		d[x[i]] += 1
		d[x[j]] -= 1
		i += 1
		j -= 1
	elif d[x[i]] % 2 != 0:
		while d[x[j]] % 2 == 0 and j >= 0:
			j -= 1
		if j != -1:
			d[x[i]] += 1
			d[x[j]] -= 1
			i += 1
	elif d[x[j]] % 2 != 0:
		while d[x[i]] % 2 == 0 and i < len(x):
			i += 1
		if i != len(x):
			d[x[i]] += 1
			d[x[j]] -= 1
			j -= 1
	else:
		i += 1
		j -= 1
s = ''
x = ''
for i in sorted(list(d.keys())):
	if d[i] % 2 != 0:
		x = i
	s = s + i * (d[i] // 2)
print(s + x + s[::-1])
