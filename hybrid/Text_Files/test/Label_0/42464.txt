t = int(input())
for i in range(t):
	(n, k) = [int(x) for x in input().split()]
	s = [int(x) for x in list(input())]
	ln = len(s)
	mp = ln
	for j in range(k):
		(sm, mn, md) = (0, 0, 0)
		for c in range(j, ln, k):
			sm += 2 * s[c] - 1
			if sm <= mn:
				mn = sm
			if sm - mn > md:
				md = sm - mn
		mp = min(mp, -md)
	print(sum(s) + mp)
