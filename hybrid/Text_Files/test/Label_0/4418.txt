#include <bits/stdc++.h>
using namespace std;
long long n, k;
int main() {
  ios_base::sync_with_stdio(false);
  cin >> n >> k;
  long long tmp = 0;
  long long sz = 1;
  long long num = 1;
  while (n--) {
    tmp += sz;
    sz *= 2LL;
    sz++;
    num++;
    if (tmp >= k) break;
  }
  long long levo = 1;
  long long desno = sz;
  long long mid;
  while (levo <= desno) {
    mid = (levo + desno) / 2LL;
    if (mid == k) {
      cout << num << endl;
      return 0;
    }
    if (mid < k) {
      levo = mid + 1;
    } else {
      desno = mid - 1;
    }
    num--;
  }
  return 0;
}
