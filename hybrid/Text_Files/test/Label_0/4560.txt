#include <iostream>
#include <string>
#include <vector>
#include <cstdio>
#include <algorithm>
#include <cassert>
using namespace std;

int main(){

  while(1){
    int n;
    cin >> n;
    if(n == 0) break;

    vector <string> str;
    string a;
    char ch;
    scanf("%c",&ch);
    for(int i=0;i<n;i++){

      assert(ch != ' '); 
      while(1){
	scanf("%c",&ch);
	
	if(ch == ' ') {
	  str.push_back(a);
	  a="";
	  continue;
	}
	if(ch == '\n') {
	  str.push_back(a);
	  a="";
	  break;
	}
	a += ch;
      }

    }
    cin >>a;
    vector <pair<int,string> > b;
    sort(str.begin(),str.end());
    for(int i=0;i<str.size();i++){
      if(a[0] == str[i][0]) {
	int cnt = 1;
	while(1) {
	  if(i+1 == str.size()) break;
	  if(str[i] == str[i+1])cnt++,i++;
	  else break;
	}
	pair <int,string> t;
	t.first = -cnt;
	t.second = str[i];
	  b.push_back(t);
      }
    }
    
    sort(b.begin(),b.end());
    for(int i=0;i<b.size();i++){
      if(i == 5) break;
      cout << b[i].second;
      if(i != b.size()-1) cout <<" ";
    }
    if(b.size() == 0) cout << "NA";
    cout << endl;
    
    
  }
  return 0;
}
