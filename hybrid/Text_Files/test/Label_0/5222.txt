#include <iostream>
using namespace std;
int a[110];
int main(){
  int n,m;
  cin>>n>>m;
  int s=0;
  for(int i=0;i<n;i++){
    cin>>a[i];
    s+=a[i];
  }
  int ans=0;
  for (int i=0;i<n;++i){
    if(a[i]*4*m>=s){
      ans+=1;
    }
  }
  
  if(ans>=m) cout<<"Yes";
  else cout<<"No";

  
  
      
}
