t = int(input())
for _ in range(t):
	(x, y) = [int(s) for s in input().split()]
	lastMove = ''
	moveCount = 0
	actualX = 0
	actualY = 0
	firstMove = 'x' if x > y else 'y'
	while actualX != x or actualY != y:
		if firstMove == 'y':
			if actualX < x and lastMove == 'right' and (actualY == y) or (actualY < y and lastMove == 'up' and (actualX == x)):
				lastMove = 'none'
				moveCount += 1
			elif actualY < y and lastMove != 'up' or (actualX < x and lastMove == 'right'):
				lastMove = 'up'
				actualY += 1
				moveCount += 1
			elif actualX < x and lastMove != 'right' or (actualY < y and lastMove == 'up'):
				lastMove = 'right'
				actualX += 1
				moveCount += 1
		elif actualX < x and lastMove == 'right' and (actualY == y) or (actualY < y and lastMove == 'up' and (actualX == x)):
			lastMove = 'none'
			moveCount += 1
		elif actualX < x and lastMove != 'right' or (actualY < y and lastMove == 'up'):
			lastMove = 'right'
			actualX += 1
			moveCount += 1
		elif actualY < y and lastMove != 'up' or (actualX < x and lastMove == 'right'):
			lastMove = 'up'
			actualY += 1
			moveCount += 1
	print(moveCount)
