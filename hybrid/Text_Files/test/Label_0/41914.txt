from sys import stdin, stdout

def main():
	n = int(stdin.readline())
	a = [int(x) for x in stdin.readline().strip().split()]
	d = 2
	ans = -1
	if n <= 2:
		stdout.write(str(n))
		return
	for i in range(2, len(a)):
		if a[i] != a[i - 1] + a[i - 2]:
			d = 2
		else:
			d += 1
		ans = max(ans, d)
	stdout.write(str(ans))
main()
