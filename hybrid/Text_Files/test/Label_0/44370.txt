import sys
reader = (s.rstrip() for s in sys.stdin)
input = reader.__next__

def printGrid(grid):
	for row in grid:
		for cell in row:
			print(cell, end='')
		print('')

def changeBoard(board, n, idx):
	for i in range(n):
		for j in range(n):
			if (i + j) % 3 == idx and board[i][j] == 'X':
				board[i][j] = 'O'
t = int(input())
for test in range(t):
	n = int(input())
	board = []
	for i in range(n):
		board.append(list(map(str, input())))
	counts = [0, 0, 0]
	for i in range(n):
		for j in range(n):
			cell = board[i][j]
			if cell == 'X':
				counts[(i + j) % 3] = counts[(i + j) % 3] + 1
	minIdx = 0
	for i in range(3):
		if counts[minIdx] > counts[i]:
			minIdx = i
	changeBoard(board, n, minIdx)
	printGrid(board)
