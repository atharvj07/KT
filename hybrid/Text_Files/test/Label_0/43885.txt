(a, b, c) = map(int, input().split())
ab = (a + b - c) / 2
bc = (b + c - a) / 2
ca = (c + a - b) / 2
i = 0
if ab == 0:
	i += 1
if bc == 0:
	i += 1
if ca == 0:
	i += 1
if i >= 2:
	print('Impossible')
elif ab != int(ab) or bc != int(bc) or ca != int(ca):
	print('Impossible')
elif ab >= 0 and bc >= 0 and (ca >= 0):
	print(int(ab), int(bc), int(ca))
else:
	print('Impossible')
