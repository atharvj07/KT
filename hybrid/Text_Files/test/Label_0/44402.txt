def predict(x, p):
	if x == 1:
		if p - a - f >= 0:
			return 'Go'
		return 'Fuel'
	elif p - (a + a - f) >= 0:
		return 'Go'
	else:
		return 'Fuel'
(a, b, f, k) = map(int, input().split())
t = 0
r = 0
www = b
for i in range(k):
	if i == k - 1:
		if b >= a:
			continue
	if i % 2 == 0:
		d = predict(i % 2, b)
		if d == 'Go':
			b -= a
		elif b >= f and a - b <= a - f and (b >= 0):
			b = www
			b -= a - f
			if b < 0:
				r = 1
				break
			t += 1
		else:
			r = 1
			break
	else:
		d = predict(i % 2, b)
		if d == 'Go':
			b -= a
		elif b >= a - f:
			b = www
			b -= f
			if b < 0:
				r = 1
				break
			t += 1
		else:
			r = 1
			break
if r == 1:
	print('-1')
else:
	print(t)
