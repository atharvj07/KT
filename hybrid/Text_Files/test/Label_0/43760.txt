T = int(input())
for _ in range(T):
	(n, k) = map(int, input().split())
	l = list(map(int, input().split()))
	a = []
	b = []
	d = []
	for i in range(0, n // 2):
		b.append(l[i] + l[n - i - 1])
		d.append(max(l[i], l[n - i - 1]) + k)
		a.append(min(l[i], l[n - i - 1]))
	b.sort()
	d.sort()
	a.sort()
	c = []
	last = -1
	cnt = 0
	for i in b:
		if last == -1:
			last = i
			cnt = 1
		elif last == i:
			cnt += 1
		else:
			c.append([last, cnt])
			cnt = 1
			last = i
	c.append([last, cnt])
	ans = n - 2
	curr = n
	i = 0
	j = 0
	k = 0
	sa = len(a)
	sc = len(c)
	sd = len(d)
	while i < sa and j < sc and (k < sd):
		if curr < ans:
			ans = curr
		if c[j][0] <= a[i] and c[j][0] <= d[k]:
			if curr - c[j][1] < ans:
				ans = curr - c[j][1]
			j += 1
		elif a[i] <= d[k]:
			curr -= 1
			i += 1
		else:
			curr += 1
			k += 1
	while j < sc and k < sd:
		if curr < ans:
			ans = curr
		if c[j][0] <= d[k]:
			if curr - c[j][1] < ans:
				ans = curr - c[j][1]
			j += 1
		else:
			curr += 1
			k += 1
	if curr < ans:
		ans = curr
	print(ans)
