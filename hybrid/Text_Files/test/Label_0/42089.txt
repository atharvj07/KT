def refPos(twodigitstr):
	se = {'1': '1', '0': '0', '8': '8', '2': '5', '5': '2'}
	(a, b) = (twodigitstr[0], twodigitstr[1])
	if a in se and b in se:
		return ''.join([se[b], se[a]])
	return False
for _ in range(int(input())):
	(H, M) = map(int, input().split())
	s = input()
	ans = refPos(s[:2])
	flag = False
	if ans:
		if int(ans) < M:
			for m in range(int(s[3:]), M):
				strM = '0' * (2 - len(str(m))) + str(m)
				ans1 = refPos(strM)
				if ans1:
					if int(ans1) < H:
						flag = True
						print(s[:3] + strM)
						break
	if not flag:
		for h in range(int(s[:2]) + 1, H):
			strH = '0' * (2 - len(str(h))) + str(h)
			ans = refPos(strH)
			if ans:
				if int(ans) < M:
					flag = True
					print(strH + ':' + '00')
					break
	if not flag:
		print('00:00')
