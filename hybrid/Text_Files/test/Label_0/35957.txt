import java.util.Scanner;
public class D
{
    public void solve()
    {
        Scanner cin = new Scanner(System.in);
        int N = cin.nextInt();
        int K = cin.nextInt();
        int x = cin.nextInt();
        int[] a = new int[N];
        long[] b = new long[N];
        long[] sumA = new long[N + 1];
        long[] sumB = new long[N + 1];
        for(int i = 0;i < N;i++)
        {
            int j = cin.nextInt();
            a[i] = j;
            b[i] = j;
        }

        for(int i = 0;i < N;i++)
        {
            sumA[i + 1] = sumA[i] | a[i];
        }

        for(int i = N - 1;i >= 0;i--)
        {
            sumB[i] = sumB[i + 1] | a[i];
        }
        long ans = 0;
        for(int i = 0;i < N;i++)
        {
            for(int j = 0;j < K;j++)
            {
                b[i] *= x;
            }

            long max = sumA[i] | b[i] | sumB[i + 1];
            ans = Math.max(ans,max);
        }
        System.out.println(ans);
    }
    public static void main(String[] args) 
    {
        new D().solve();    
    }
}
