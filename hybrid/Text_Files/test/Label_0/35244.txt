import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int t=sc.nextInt();
        for (int i = 0; i <t ; i++) {
            int h=sc.nextInt();
            int m=sc.nextInt();
            int hm=h*60;
            int sum=hm+m;
            int sheng=1440-sum;
            System.out.println(sheng);
        }
    }
}
