#include <bits/stdc++.h>
using namespace std;
int main(){
  long long int n,k[100005],sum=0;
  cin>>n;
  for(int i=0;i<n;++i){
    cin>>k[i];
    sum+=k[i];
  }
  if(sum%(((1+n)*n)/2)){
    cout<<"NO"<<endl;
    return 0;
  }
  int mon=sum/(((1+n)*n)/2);
  long long int mo[100005];
  for(int i=0;i<n;++i){
    if(i!=n-1) mo[i]=k[i+1]-k[i]-mon;
    else mo[i]=k[0]-k[i]-mon;
    
    if(mo[i]>0||mo[i]%n){
      cout<<"NO"<<endl;
      return 0;
    }
  }
  
  cout<<"YES"<<endl;
}
