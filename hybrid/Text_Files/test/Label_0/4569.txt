#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n, m, d;
  cin >> n >> d >> m;
  long long int a[n];
  vector<long long int> r, l;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    if (a[i] > m)
      r.push_back(a[i]);
    else
      l.push_back(a[i]);
  }
  sort(r.rbegin(), r.rend());
  sort(l.rbegin(), l.rend());
  long long int nl = l.size(), nr = r.size();
  vector<long long int> sr(n + 1), sl(n + 1);
  for (int i = 0; i < nl; i++) sl[i + 1] = sl[i] + l[i];
  for (int i = 0; i < nr; i++) sr[i + 1] = sr[i] + r[i];
  long long int ans = 0;
  for (int i = 0; i <= nl; i++) {
    int x = n - i;
    int id = min(nr, (x + d) / (d + 1));
    ans = max(ans, sr[id] + sl[i]);
  }
  cout << ans << endl;
  return 0;
}
