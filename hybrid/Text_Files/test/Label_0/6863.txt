#include <bits/stdc++.h>
using namespace std;
map<string, int> m;
map<string, int>::iterator it;
vector<pair<int, string> > v;
bool myfun(pair<int, string> a, pair<int, string> b) {
  if (a.first > b.first)
    return true;
  else if (a.first < b.first)
    return false;
  else {
    return (a.second < b.second);
  }
}
string name;
int n;
string name1, verb, name2, smiec;
int points;
int main() {
  cin >> name;
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    cin >> name1;
    cin >> verb;
    if (verb == "likes") {
      points = 5;
      cin >> name2;
      name2.erase(name2.end() - 2, name2.end());
      cin >> smiec;
    } else if (verb == "posted") {
      points = 15;
      cin >> name2;
      cin >> name2;
      name2.erase(name2.end() - 2, name2.end());
      cin >> smiec;
    } else if (verb == "commented") {
      points = 10;
      cin >> name2;
      cin >> name2;
      name2.erase(name2.end() - 2, name2.end());
      cin >> smiec;
    }
    if (name1 == name || name2 == name) {
      if (name1 == name)
        m[name2] += points;
      else
        m[name1] += points;
    } else {
      m[name1] += 0;
      m[name2] += 0;
    }
  }
  for (it = m.begin(); it != m.end(); it++) {
    v.push_back(make_pair((*it).second, (*it).first));
  }
  sort(v.begin(), v.end(), myfun);
  for (int i = 0; i < v.size(); i++) {
    cout << v[i].second << endl;
  }
  return 0;
}
