import math
ref = {'0': '0', '1': '1', '2': '5', '5': '2', '8': '8'}

def isValid(hr, minu, h, m):
	invalid = {'3', '4', '6', '7', '9'}
	for ch in hr:
		if ch in invalid:
			return False
	for ch in minu:
		if ch in invalid:
			return False
	(refH, refM) = ('', '')
	for ch in hr:
		refH = ref[ch] + refH
	for ch in minu:
		refM = ref[ch] + refM
	if int(refM) >= h or int(hr) >= h or int(refH) >= m or (int(minu) >= m):
		return False
	if int(refM) * m + int(refH) > h * m + m:
		return False
	return True
for test in range(int(input())):
	(h, m) = map(int, input().split())
	(hrs, minu) = map(str, input().split(':'))
	while not isValid(hrs, minu, h, m):
		temp = int(minu) + 1
		minu = str(temp)
		if int(minu) >= m:
			t1 = int(hrs) + 1
			hrs = str(t1)
			minu = '00'
		if int(hrs) >= h:
			hrs = '00'
			minu = '00'
		if len(hrs) == 1:
			hrs = '0' + hrs
		if len(minu) == 1:
			minu = '0' + minu
	print(hrs + ':' + minu)
