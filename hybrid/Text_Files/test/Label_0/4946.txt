#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int c = 0;
  int ans = 0;
  bool started = false;
  for (int i = 0; i <= n; i++) {
    if (s[i] == 'x') {
      if (started)
        c++;
      else {
        started = true;
        c = 1;
      }
    } else {
      started = false;
      if (c > 2) ans += (c - 2);
      c = 0;
    }
  }
  cout << ans;
}
