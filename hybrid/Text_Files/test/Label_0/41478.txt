import sys
import math
from collections import defaultdict

def solve(test):
	(n, k) = map(int, input().split())
	a = list(map(int, input().split()))
	a += [0, 0]
	if n == 1 and a[0] == k:
		print('yes')
		return
	(flag1, flag2) = (0, 0)
	for i in range(n):
		if a[i] == k:
			flag1 = 1
		if a[i] >= k:
			if a[i + 1] >= k or a[i + 2] >= k:
				flag2 = 1
	print('yes' if flag1 and flag2 else 'no')
test_cases = int(input())
for t in range(1, test_cases + 1):
	solve(t)
