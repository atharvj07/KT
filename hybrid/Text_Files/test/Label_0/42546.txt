t = int(input())
for _ in range(t):
	s = list(str(input()))
	x = int(input())
	w = ['1' for _ in range(len(s))]
	for i in range(len(s)):
		if s[i] == '0':
			if i - x >= 0:
				w[i - x] = '0'
			if i + x < len(s):
				w[i + x] = '0'
	flag = False
	for i in range(len(s)):
		if s[i] == '1':
			if i - x >= 0 and i + x < len(s):
				if w[i - x] == '0' and w[i + x] == '0':
					flag = True
			elif i - x >= 0 and i + x >= len(s):
				if w[i - x] == '0':
					flag = True
			elif i - x < 0 and i + x < len(s):
				if w[i + x] == '0':
					flag = True
			elif i - x < 0 and i + x >= len(s):
				flag = True
	if flag == True:
		print(-1)
	else:
		res = ''
		for i in range(len(w)):
			res += w[i]
		print(res)
