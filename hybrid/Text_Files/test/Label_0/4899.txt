#include <bits/stdc++.h>
using namespace std;
typedef struct {
  string name;
  int s1;
  int s2;
  int s3;
} Team;
bool comp(const Team &x, const Team &y) {
  if (x.s1 > y.s1)
    return true;
  else if (x.s1 < y.s1)
    return false;
  else if (x.s2 > y.s2)
    return true;
  else if (x.s2 < y.s2)
    return false;
  else if (x.s3 > y.s3)
    return true;
  else
    return false;
}
bool comp2(const Team &x, const Team &y) {
  if (x.name < y.name)
    return true;
  else
    return false;
}
int main() {
  int n, i, j, len, x, y;
  string t1, t2, name1, name2;
  cin >> n;
  Team T[n];
  for (i = 0; i < n; i++) {
    cin >> T[i].name;
    T[i].s1 = T[i].s2 = T[i].s3 = 0;
  }
  for (i = 0; i < n * (n - 1) / 2; i++) {
    cin >> t1 >> t2;
    j = t1.find('-');
    name1 = t1.substr(0, j);
    name2 = t1.substr(j + 1, 80);
    j = t2.find(':');
    x = atoi(t2.substr(0, j).c_str());
    y = atoi(t2.substr(j + 1, 80).c_str());
    for (j = 0; j < n; j++) {
      if (name1 == T[j].name) {
        T[j].s1 += (x > y) ? 3 : (x == y) ? 1 : 0;
        T[j].s2 += x - y;
        T[j].s3 += x;
      } else if (name2 == T[j].name) {
        T[j].s1 += (y > x) ? 3 : (x == y) ? 1 : 0;
        T[j].s2 += y - x;
        T[j].s3 += y;
      }
    }
  }
  sort(T, T + n, comp);
  sort(T, T + n / 2, comp2);
  for (i = 0; i < n / 2; i++) cout << T[i].name << endl;
  return 0;
}
