from collections import defaultdict
import heapq
T = int(input())
for _ in range(T):
	(n, K) = map(int, input().split())
	A = list(map(int, input().split()))
	A.sort(reverse=True)
	B = [0]
	a = heapq.heapify(B)
	ans = 0
	for i in range(n):
		v = heapq.heappop(B)
		v = -1 * v
		if v >= A[i]:
			v = v - A[i]
		else:
			ans += 1
			heapq.heappush(B, -1 * (K - A[i]))
		heapq.heappush(B, -1 * v)
	print(ans)
