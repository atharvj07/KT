#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 2e7+10;
int T,a,b,c;
int tot,p[maxn],f[maxn],g[maxn];
void seive(int n){
    f[1] = g[1] = 1;
    for (int i = 2; i <= n; i++) {
        if(!f[i]) { p[++tot] = i,g[i] = 2; }
        for (int j = 1; p[j]*i <= n && j <= tot; j++) {
            f[i*p[j]] = 1;
            if(i%p[j]==0) { 
                g[i*p[j]] = g[i]; 
                break; 
            }
            g[i*p[j]] = g[i]*2;
        }
    }
}

int cal(int u) {
    if((u+b)%a) return 0;
    return g[(u+b)/a];
}

int main() {
    seive(maxn-10);
    for (cin >> T; T; T--) {
        int ans = 0;
        cin >> a >> b >> c;
        for (int i = 1; i*i <= c; i++) {
            if(c%i) continue;
            ans += cal(i) + (i*i!=c?cal(c/i):0);
        }
        cout << ans << endl;
    }
    return 0;
}
