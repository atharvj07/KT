#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  int T;
  cin >> T;
  for (int t = 0; t < T; ++t) {
    long long X, Y, K;
    cin >> X >> Y >> K;
    X--;
    long long ans = 0;
    ans += (K + Y * K - 1) / X + ((K + Y * K - 1) % X != 0);
    ans += K;
    cout << ans << '\n';
  }
  return 0;
}
