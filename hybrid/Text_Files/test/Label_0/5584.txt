#include <bits/stdc++.h>
using namespace std;
long long a[1200];
int main() {
  long long n, m, sum = 0;
  cin >> n >> m;
  long long k = n % m;
  long long q = n / m;
  for (int i = 1; i <= m; i++) {
    a[i] = q;
    if (i <= k) {
      a[i]++;
    }
  }
  for (int i = 1; i <= m; i++) {
    for (int j = 1; j <= m; j++) {
      if ((i * i + j * j) % m == 0) {
        sum += a[i] * a[j];
      }
    }
  }
  cout << sum;
  return 0;
}
