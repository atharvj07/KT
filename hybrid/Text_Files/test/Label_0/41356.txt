import math
n = int(input())
s = input()
combination = ('RGB', 'RBG', 'GRB', 'GBR', 'BGR', 'BRG')
summa = [0, 0, 0, 0, 0, 0]

def diff(input, pattern):
	count = 0
	if len(input) == 1:
		if input[0] != pattern[0]:
			count += 1
		return count
	elif len(input) == 2:
		if input[0] != pattern[0]:
			count += 1
		if input[1] != pattern[1]:
			count += 1
		return count
	else:
		if input[0] != pattern[0]:
			count += 1
		if input[1] != pattern[1]:
			count += 1
		if input[2] != pattern[2]:
			count += 1
		return count
number_of_ut = 0
number_of_ut = len(s) // 3
for i in range(number_of_ut):
	tmp = s[3 * i:3 * (i + 1)]
	for i in range(len(combination)):
		summa[i] += diff(tmp, combination[i])
ostatok = len(s) - number_of_ut * 3
if ostatok > 0:
	tmp = s[len(s) - ostatok:]
	for i in range(len(combination)):
		summa[i] += diff(tmp, combination[i])
best_comb = combination[summa.index(min(summa))]
ans = ''
for i in range(number_of_ut):
	ans += best_comb
ans += best_comb[:ostatok]
print(summa[summa.index(min(summa))])
print(ans)
