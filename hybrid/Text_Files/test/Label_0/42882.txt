def compare(a, b):
	last = 0
	for i in range(0, len(b), 1):
		if a.find(b[i:i + 1], last, len(a)) == -1:
			return -1
		else:
			last = a.find(b[i:i + 1], last, len(a)) + 1
	return len(a) - len(b)
string = input()
ans = []
i = 1
while int(i * i) <= int(string):
	n = compare(string, str(i * i))
	if n != -1:
		ans.append(n)
	i += 1
if len(ans) == 0:
	print(-1)
else:
	print(min(ans))
