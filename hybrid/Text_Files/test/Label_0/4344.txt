#include <bits/stdc++.h>
using namespace std;
int n, k, i, a[2000001];
int main() {
  while (~scanf("%d%d", &n, &k)) {
    for (i = 1; i <= n * 2; i++) {
      a[i] = i;
    }
    for (i = 1; i <= n * 2; i += 2) {
      swap(a[i], a[i + 1]);
    }
    for (i = 1; i <= n * 2; i += 2) {
      if (k) {
        swap(a[i], a[i + 1]);
        k--;
      }
    }
    for (i = 1; i <= n * 2; i++) {
      if (i != n * 2)
        printf("%d ", a[i]);
      else
        printf("%d\n", a[i]);
    }
  }
}
