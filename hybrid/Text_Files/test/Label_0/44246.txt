for _ in range(int(input())):
	s = list(input())
	t = list(input())
	p = list(input())
	j = 0
	arr = []
	pos = 1
	for i in range(len(s)):
		while j < len(t):
			if s[i] != t[j]:
				arr.append(t[j])
				j += 1
			else:
				j += 1
				break
		else:
			print('NO')
			pos = 0
			break
	if j <= len(t) - 1:
		arr += t[j:]
	if pos:
		for k in range(len(arr)):
			if arr[k] in p:
				del p[p.index(arr[k])]
			else:
				print('NO')
				pos = 0
				break
		if pos:
			print('YES')
