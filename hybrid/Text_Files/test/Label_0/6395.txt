#include<bits/stdc++.h>
typedef long long i64;
const int P = 1e9 + 7;
int N, M;
int dp[1 << 15];
int ouset[1 << 15];

int main() {
  if(fopen("input.txt","r")) {
    freopen("input.txt","r",stdin);
    freopen("output.txt","w",stdout);
  }
  
  scanf("%d%d", &N, &M);
  while(M--) {
    int u, v; scanf("%d%d", &u, &v);
    --u, --v; ouset[u] |= 1 << v;
  }

  dp[0] = 1;
  for(int S = 0;S < 1<<N; ++S) {
    if(!dp[S]) continue;
    int T = ((1 << N) - 1) ^ S;
    for(int S0 = T; S0; (--S0) &= T) {
      if((S0 & 3) == 3) continue;
      int w = dp[S];
      for(int k = 0;k < N; ++k) {
	if(S >> k & 1) continue;
	if(S0 >> k & 1) w = (i64) w * (1ll << __builtin_popcount(ouset[k] & ~(S | S0))) % P;
	else w = (i64) w * ((1 << __builtin_popcount(ouset[k] & S0)) - 1) % P;
      }
      (dp[S | S0] += w) %= P;
    }
  }

  printf("%d\n", dp[(1 << N) - 1]);
  
  return 0;
}
