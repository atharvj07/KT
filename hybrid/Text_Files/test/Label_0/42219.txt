s = input()
num = [str(i) for i in range(10)]
any = 0
alpha = []
for i in s:
	if i == '?':
		any += 1
	elif i not in num and (not i in alpha):
		alpha.append(i)
result = 1
if s[0] in num:
	c = 10
	while alpha != []:
		result *= c
		alpha.pop()
		c -= 1
	result *= 10 ** any
	print(result)
elif s[0] != '?':
	result *= 9
	alpha.pop()
	c = 9
	while alpha != []:
		result *= c
		alpha.pop()
		c -= 1
	result *= 10 ** any
	print(result)
else:
	c = 10
	result *= 9
	while alpha != []:
		result *= c
		alpha.pop()
		c -= 1
	result *= 10 ** (any - 1)
	print(result)
