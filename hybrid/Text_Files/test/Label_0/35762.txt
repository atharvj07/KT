import java.io.BufferedReader;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.util.HashSet;
import java.util.StringTokenizer;

public class d {

	public static void main(String[] Args) throws Exception {
		fscan sc = new fscan(System.in);
		int n = sc.nextInt();
		pre();
		long index = sc.nextLong() - 1;
		int tot = 0;
		for (int k = 0; k < n; k++) {
			if (cyc[n - k - 1] > index) {
				if (k != 0) {
					System.out.print(" ");
				}
				System.out.print((k + 1));
			} else {
				int taken = 2;
				int rem = n-k-taken;
				index -= cyc[rem + 1];
				while(index >= cyc[rem]){
					index-=cyc[rem];
					rem--;
					taken++;
				}
				for(int j = 0; j < taken;j++){
					if (k+j != 0) {
						System.out.print(" ");
					}
					System.out.print(stuff[taken-1][j]+k+1);
				}
				k+=taken-1;
			}
		}
		System.out.println();
	}

	public static void pre() {
		cyc[0] = 1;
		cyc[1] = 1;
		for (int k = 2; k < 51; k++) {
			cyc[k] = cyc[k-1] +cyc[k-2];
		}
		stuff = new int[51][51];
		for(int k = 0; k < 51; k++){
			for(int j = 0; j <= k; j++){
				if(k - j*2 >= 0){
					stuff[k][j] = k-j*2;
				}
			}
			for(int j = k,i = 1; j >= 0; j--,i+=2){
				if(k-i >= 0){
					stuff[k][j] = k-i;
				}
			}
		}
	}

	public static long[] cyc = new long[51];
	public static int[][] stuff;

	public static class fscan {
		StringTokenizer st;
		BufferedReader br;

		fscan(InputStream in) throws Exception {
			br = new BufferedReader(new InputStreamReader(in));
			st = new StringTokenizer(br.readLine());
		}

		String next() throws Exception {
			if (st.hasMoreTokens()) {
				return st.nextToken();
			}
			st = new StringTokenizer(br.readLine());
			return next();
		}

		int nextInt() throws Exception {
			return Integer.parseInt(next());
		}

		long nextLong() throws Exception {
			return Long.parseLong(next());
		}
	}
}
