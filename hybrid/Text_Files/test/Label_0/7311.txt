#include<iostream>
#include<complex>
#include<cmath>
#include<utility>
#include<cstring>
#include<vector>
#include<queue>
#define rep(i,n) for(int i=0;i<(int)n;i++)
#define fs first
#define sc second
using namespace std;
typedef double D;
typedef complex<D> P;
typedef pair<P,P> L;
const D EPS = 1e-9;
const D PI = acos(-1);

D dot(P x,P y){return real(conj(x)*y);}
D cross(P x,P y){return imag(conj(x)*y);}
D arg(P a,P b,P c){return acos(dot(b-a,c-a)/(abs(b-a)*abs(c-a)));}

int c,w;
D x,y;
P p[110];
pair<int,int> l[330];
int g[330][2];
vector<int> wall[110];
int s,t;
int dis[1100];

int main(){
  while(cin >> c >> w, c){
    rep(i,c){
      cin >> x >> y; p[i] = P(x,y);
      wall[i].clear();
    }

    rep(i,w){
      cin >> s >> t; s--; t--;
      l[i] = make_pair(s,t);
      wall[s].push_back(t); wall[t].push_back(s);
    }

    rep(i,c){
      for(int j=1;j<(int)wall[i].size();j++){
	for(int k=j+1;k<(int)wall[i].size();k++){
	  D s1 = arg(p[i],p[wall[i][0]],p[wall[i][j]]);
	  if(cross(p[wall[i][0]]-p[i],p[wall[i][j]]-p[i])<EPS)s1 = 2*PI-s1;
	  D s2 = arg(p[i],p[wall[i][0]],p[wall[i][k]]); 
	  if(cross(p[wall[i][0]]-p[i],p[wall[i][k]]-p[i])<EPS)s2 = 2*PI-s2;
	  if(s1<s2)swap(wall[i][j],wall[i][k]);
	}
      }
    }
    /*
    rep(i,c){
      rep(j,wall[i].size())cout << wall[i][j] << " " ;cout << endl;
    }
    cout << endl;
    */

    memset(dis,-1,sizeof(dis));
    memset(g,-1,sizeof(g));
    int group = 0;

    rep(i,c){
      rep(j,wall[i].size()){
	int cur = -1, dir = -1;
	rep(k,w){
	  if(i == l[k].fs && wall[i][j] == l[k].sc){
	    cur = k; dir = 0; break;
	  }
	  if(i == l[k].sc && wall[i][j] == l[k].fs){
	    cur = k; dir = 1;  break;
	  }
	}
	if(g[cur][dir]>=0)continue;

	D area = 0;
	while(true){
	  //cout << cur+1 << " " << dir << endl;
	  g[cur][dir] = group;
	  
	  int node = dir?l[cur].fs:l[cur].sc;
	  int prv = dir?l[cur].sc:l[cur].fs;
	  //cout << "[" << prv+1 << ", " << node+1 << "]" << endl; 
	  rep(k,wall[node].size()){
	    if(wall[node][k] == prv){
	      int nxt = wall[node][(k+1)%wall[node].size()];
	      area += cross(p[prv]-p[node],p[nxt]-p[node]);
	      prv = node;
	      node = nxt;
	      break;
	    }
	  }

	  rep(k,w){
	    if(prv == l[k].fs && node == l[k].sc){
	      cur = k; dir = 0; break;
	    }
	    if(prv == l[k].sc && node == l[k].fs){
	      cur = k; dir = 1;  break;
	    }
	  }
	  if(g[cur][dir]>=0)break;
	}
	//cout << area << endl;
	//cout << endl;
	if(area>EPS)dis[group] = 0;
	group++;
      }
    }

    //cout << group << endl;

    bool adj[1100][1100];
    memset(adj,0,sizeof(adj));
    rep(i,w){
      adj[g[i][0]][g[i][1]] = adj[g[i][1]][g[i][0]] = 1;
    }

    /*
    rep(i,group){
      rep(j,group)cout << adj[i][j] << " ";cout << endl;
    }
    */

    queue<int> q;
    rep(i,group){
      if(dis[i] == 0)q.push(i);
    }

    int ans = 0;
    while(q.size()){
      int now = q.front(); q.pop();
      rep(i,group){
	if(adj[now][i] && dis[i]<0){
	  ans = dis[i] = dis[now] + 1;
	  q.push(i);
	}
      }
    }
    cout << ans << endl;
  }
}
