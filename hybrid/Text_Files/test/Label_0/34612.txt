import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.Stack;
import java.util.StringTokenizer;

public class Task {

    StringTokenizer st;
    BufferedReader in;
    PrintWriter out;

    public static void main(String[] args) throws NumberFormatException,
            IOException {
        Task task = new Task();
        task.open();
        task.solve();
        task.close();
    }

    public void open() throws IOException {
        in = new BufferedReader(new InputStreamReader(System.in));
        out = new PrintWriter(System.out);

    }

    public String nextToken() throws IOException {
        while (st == null || !st.hasMoreTokens()) {
            String line = in.readLine();
            if (line == null)
                return null;
            st = new StringTokenizer(line);
        }
        return st.nextToken();
    }

    public int nextInt() throws NumberFormatException, IOException {
        return Integer.parseInt(nextToken());
    }



    class Edge implements Comparable<Edge> {
        int first;
        int second;
        int ind;

        boolean opposite = false;
        boolean visited = false;

        public Edge(int first, int second, int ind) {
            this.first = first;
            this.second = second;
            this.ind = ind;
        }

        public void setOpposite(boolean opposite) {
            this.opposite = opposite;
        }

        public void setVisited(boolean visited) {
            this.visited = visited;
        }

        @Override
        public int compareTo(Edge o) {
            return Integer.compare(this.ind, o.ind);
        }
    }

    public void solve() throws NumberFormatException, IOException {
        int cntVertexes = nextInt();
        int cntEdges = nextInt();

        LinkedList<Edge>[] graph = new LinkedList[cntVertexes];
        int[] cntAlwaysIncome = new int[cntVertexes];
        int[] cntAlwaysOutcome = new int[cntVertexes];

        for (int i = 0; i < cntVertexes; i++) {
            graph[i] = new LinkedList<Edge>();
        }

        ArrayList<Edge> edges = new ArrayList<Edge>();

        for (int i = 0; i < cntEdges; i++) {
            int a = nextInt() - 1;
            int b = nextInt() - 1;

            edges.add(new Edge(a, b, i));

            graph[a].add(edges.get(i));
            graph[b].add(edges.get(i));

            if (a == b) {
                cntAlwaysIncome[a]++;
                cntAlwaysOutcome[a]++;
            }
        }

        ArrayList<Integer> oddVertexes = new ArrayList<Integer>();

        for (int i = 0; i < cntVertexes; i++) {
            if (graph[i].size() % 2 != 0) oddVertexes.add(i);
        }

        for (int i = 0; i < oddVertexes.size() - 1; i += 2) {
            Edge edge = new Edge(oddVertexes.get(i), oddVertexes.get(i + 1), cntEdges);
            graph[oddVertexes.get(i)].add(edge);
            graph[oddVertexes.get(i + 1)].add(edge);
            edges.add(edge);
            cntEdges++;
        }

        if (cntEdges % 2 != 0) {
            Edge edge = new Edge(0, 0, cntEdges);
            graph[0].add(edge);
            edges.add(edge);
            cntEdges++;

            cntAlwaysIncome[0]++;
            cntAlwaysOutcome[0]++;
        }

        Stack<Integer> st = new Stack<Integer>();
        int startVertex = edges.get(0).first;
        st.push(startVertex);

        ArrayList<Integer> answ = new ArrayList<Integer>();

        while (!st.isEmpty()) {
            int currentVertex = st.peek();

            while (graph[currentVertex].size() > 0 && graph[currentVertex].getLast().visited)
                graph[currentVertex].pollLast();

            if (graph[currentVertex].size() == 0) {
                answ.add(currentVertex);
                st.pop();
            } else {
                Edge curEdge = graph[currentVertex].pollLast();
                curEdge.setVisited(true);

                if (curEdge.first == currentVertex) st.push(curEdge.second);
                else {
                    st.push(curEdge.first);
                    curEdge.setOpposite(true);
                }
            }

        }

        out.println(cntEdges);

        for (int i = 0; i < answ.size() - 1; i++) {
            if (i % 2 == 0) {
                out.println((answ.get(i) + 1) + " " + (answ.get(i + 1) + 1));
            } else out.println((answ.get(i + 1) + 1) + " " + (answ.get(i) + 1));
        }
    }



    public void close() {
        out.flush();
        out.close();
    }
}
