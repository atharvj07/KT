#include <bits/stdc++.h>
using namespace std;
long long max(long long x, long long y) { return x > y ? x : y; }
long long min(long long x, long long y) { return x < y ? x : y; }
void _(long long& a, long long b) {
  a = (a % 1000000007 + b % 1000000007) % 1000000007;
  a = (a + 1000000007) % 1000000007;
}
long long n;
string s;
long long dp[500][500];
long long dfs(long long i, long long j) {
  if (i == j) return 1;
  if (i > j) return 0;
  if (dp[i][j] != -1) return dp[i][j];
  long long ans = dfs(i + 1, j) + 1;
  for (long long u = i + 1; u <= j; u++) {
    if (s[i] == s[u]) ans = min(ans, dfs(i + 1, u - 1) + dfs(u, j));
  }
  return dp[i][j] = ans;
}
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  cin >> n;
  cin >> s;
  memset(dp, -1, sizeof(dp));
  cout << dfs(0, n - 1);
}
