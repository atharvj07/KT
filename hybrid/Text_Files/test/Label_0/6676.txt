class Add_Segment_Tree {
	vector<long long int>v;
	long long int ret;
	int num;
	long long int Update(int place) {
		if (place >= v.size() / 2) {
			return v[place];
		}
		v[place] = Update(place * 2) + Update(place * 2 + 1);
		return v[place];
	}
public:
	Add_Segment_Tree(int n) {
		n++;
		num = 1;
		while (num < n * 2)num *= 2;
		v.resize(num, 0);
	}
	void Add(int place, long long int num, bool update) {
		place += v.size() / 2;
		v[place] += num;
		if (!update)return;
		place /= 2;
		while (place) {
			v[place] = v[place * 2] + v[place * 2 + 1];
			place /= 2;
		}
	}
	void TopDown() {
		Update(1);
	}
	long long int Sum(int a, int b) {
		ret = 0;
		b++;
		for (a += num / 2, b += num / 2; a < b; a >>= 1, b >>= 1) {
			if (a & 1)ret += v[a++];
			if (b & 1)ret += v[--b];
		}
		return ret;
	}
};

int main() {
	ios::sync_with_stdio(false);

	cin >> N;
	vector<long long int>v(N);
	for (auto &i : v)cin >> i;
	for (int i = 1; i < N; i++) {
		v[i] += v[i - 1];
	}
	vector<long long int>w(N);
	for (auto &i : w)cin >> i;
	for (int i = 1; i < N; i++) {
		w[i] += w[i - 1];
	}
	auto u = v;
	sort(u.begin(), u.end());
	if (w[0] > u[0]) {
		cout << -1 << endl;
		return 0;
	}
	for (int i = 1; i < N; i++) {
		if (w[i] - w[i - 1] > u[i] - u[i - 1]) {
			cout << -1 << endl;
			return 0;
		}
	}
	long long int ans = 0;
	Add_Segment_Tree asg(N);
	for (int i = 0; i < N; i++) {
		int idx = lower_bound(u.begin(), u.end(), v[i]) - u.begin();
		ans += asg.Sum(idx, N - 1);
		asg.Add(idx, 1, true);
	}
	cout << ans  << endl;
}
