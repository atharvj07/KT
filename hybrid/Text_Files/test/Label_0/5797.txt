#include <bits/stdc++.h>
const int MOD = 998244353;
const int ny2 = (MOD + 2) / 2;
const int N = 2005;
long long f[N], g[N], h[N][N];
long long ksm(long long x, long long dep) {
  long long res = 1;
  x = (x % MOD + MOD) % MOD;
  for (; dep; dep >>= 1) {
    (dep & 1) ? (res = res * x % MOD) : 0;
    x = x * x % MOD;
  }
  return res;
}
int main(void) {
  int n, a, b;
  scanf("%d%d%d", &n, &a, &b);
  long long p = a * ksm(b, MOD - 2) % MOD;
  h[1][0] = h[1][1] = 1;
  for (long long i = 2; i <= n; ++i) {
    h[i][0] = 1;
    for (long long j = 1; j <= i; ++j) {
      h[i][j] = (h[i - 1][j] * ksm(1 + MOD - p, j) % MOD +
                 h[i - 1][j - 1] * ksm(p, i - j) % MOD) %
                MOD;
    }
  }
  g[1] = 1;
  for (long long i = 2; i <= n; ++i) {
    for (long long j = 1; j <= i - 1; ++j)
      g[i] = (g[i] + g[j] * h[i][j] % MOD) % MOD;
    g[i] = (MOD + 1 - g[i]) % MOD;
  }
  f[1] = 0;
  for (long long i = 2; i <= n; ++i) {
    f[i] = g[i] * h[i][i] % MOD * i % MOD * (i - 1) % MOD * ny2 % MOD;
    for (long long j = 1; j <= i - 1; ++j) {
      f[i] = (f[i] + (j * (j - 1) % MOD * ny2 % MOD + (i - j) * j % MOD + f[j] +
                      f[i - j]) %
                         MOD * g[j] % MOD * h[i][j] % MOD) %
             MOD;
    }
    f[i] = f[i] * ksm(1 - g[i] * h[i][i] % MOD, MOD - 2) % MOD;
  }
  printf("%lld\n", f[n]);
  return 0;
}
