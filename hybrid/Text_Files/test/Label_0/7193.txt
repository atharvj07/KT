#include <bits/stdc++.h>
using namespace std;
string s;
long long ans = 0;
long long fact[2005];
long long dp[2005][2005][3];
long long hatena[2005][2005];
int main(void) {
  cin >> s;
  int n = s.size();
  fact[0] = 1;
  for (int i = 1; i <= n; i++) {
    fact[i] = fact[i - 1] * 2LL % 998244353LL;
  }
  for (int i = 0; i < n; i++) {
    for (int j = i; j < n; j++) {
      hatena[i][j + 1] = hatena[i][j];
      if (s[j] == '?') {
        hatena[i][j + 1]++;
      }
    }
  }
  dp[0][n - 1][0] = 1;
  for (int i = 0; i < n; i++) {
    for (int j = n - 1; j > i; j--) {
      for (int k = 0; k < 3; k++) {
        if (s[i] == '(') {
          if (k > 0) continue;
          if (s[j] == ')') {
            long long val = dp[i][j][k];
            val = (long long)val * fact[hatena[i + 1][j]] % 998244353LL;
            ans += val;
            if (ans >= 998244353LL) {
              ans -= 998244353LL;
            }
            dp[i + 1][j - 1][0] += dp[i][j][k];
            if (dp[i + 1][j - 1][0] >= 998244353LL) {
              dp[i + 1][j - 1][0] -= 998244353LL;
            }
          } else if (s[j] == '(') {
            dp[i][j - 1][0] += dp[i][j][k];
            if (dp[i][j - 1][0] >= 998244353LL) {
              dp[i][j - 1][0] -= 998244353LL;
            }
          } else if (s[j] == '?') {
            long long val = dp[i][j][k];
            val = (long long)val * fact[hatena[i + 1][j]] % 998244353LL;
            ans += val;
            if (ans >= 998244353LL) {
              ans -= 998244353LL;
            }
            dp[i + 1][j - 1][0] += dp[i][j][k];
            if (dp[i + 1][j - 1][0] >= 998244353LL) {
              dp[i + 1][j - 1][0] -= 998244353LL;
            }
            dp[i][j - 1][0] += dp[i][j][k];
            if (dp[i][j - 1][0] >= 998244353LL) {
              dp[i][j - 1][0] -= 998244353LL;
            }
          }
        }
        if (s[i] == ')') {
          if (k > 0) continue;
          if (s[j] == '(') {
            dp[i + 1][j - 1][0] += dp[i][j][k];
            if (dp[i + 1][j - 1][0] >= 998244353LL) {
              dp[i + 1][j - 1][0] -= 998244353LL;
            }
          } else {
            dp[i + 1][j][0] += dp[i][j][0];
            if (dp[i + 1][j][0] >= 998244353LL) {
              dp[i + 1][j][0] -= 998244353LL;
            }
          }
        }
        if (s[i] == '?') {
          if (k == 0) {
            dp[i][j][1] += dp[i][j][0];
            dp[i][j][1] %= 998244353LL;
            dp[i][j][2] += dp[i][j][0];
            dp[i][j][2] %= 998244353LL;
          } else if (k == 1) {
            if (s[j] == '(') {
              dp[i][j - 1][1] += dp[i][j][k];
              dp[i][j - 1][1] %= 998244353LL;
            } else if (s[j] == ')') {
              long long val = dp[i][j][k];
              val = (long long)val * fact[hatena[i + 1][j]] % 998244353LL;
              ans += val;
              if (ans >= 998244353LL) {
                ans -= 998244353LL;
              }
              dp[i + 1][j - 1][0] += dp[i][j][k];
              dp[i + 1][j - 1][0] %= 998244353LL;
            } else if (s[j] == '?') {
              long long val = dp[i][j][k];
              val = (long long)val * fact[hatena[i + 1][j]] % 998244353LL;
              ans += val;
              if (ans >= 998244353LL) {
                ans -= 998244353LL;
              }
              dp[i + 1][j - 1][0] += dp[i][j][k];
              dp[i + 1][j - 1][0] %= 998244353LL;
              dp[i][j - 1][1] += dp[i][j][k];
              dp[i][j - 1][1] %= 998244353LL;
            }
          } else if (k == 2) {
            if (s[j] == '(') {
              dp[i + 1][j - 1][0] += dp[i][j][k];
              dp[i + 1][j - 1][0] %= 998244353LL;
            } else if (s[j] == ')') {
              dp[i + 1][j][0] += dp[i][j][k];
              dp[i + 1][j][0] %= 998244353LL;
            } else if (s[j] == '?') {
              dp[i + 1][j][0] += dp[i][j][k];
              dp[i + 1][j][0] %= 998244353LL;
            }
          }
        }
      }
    }
  }
  ans %= 998244353LL;
  printf("%lld\n", ans);
  return 0;
}
