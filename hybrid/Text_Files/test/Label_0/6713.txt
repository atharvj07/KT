#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  int size = s.size();
  stack<char> t1;
  stack<char> t2;
  for (int i = 0; i < size; i++) {
    t1.push(s[i]);
  }
  while (!t1.empty()) {
    if (t2.empty()) {
      t2.push(t1.top());
      t1.pop();
      continue;
    }
    if (t1.top() == t2.top()) {
      t1.pop();
      t2.pop();
    } else {
      t2.push(t1.top());
      t1.pop();
    }
  }
  while (!t2.empty()) {
    cout << t2.top();
    t2.pop();
  }
  return 0;
}
