#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n;
  cin >> n;
  long long int a[n];
  for (long long int i = 0; i < n; i++) cin >> a[i];
  long long int sum1 = a[0], sum2 = a[n - 1];
  long long int i = 0, j = n - 1;
  long long int res = 0;
  while (i < j) {
    if (sum1 == sum2) {
      res = max(sum1, res);
      sum1 += a[++i];
      sum2 += a[--j];
    } else if (sum1 > sum2)
      sum2 += a[--j];
    else
      sum1 += a[++i];
  }
  cout << res << endl;
}
