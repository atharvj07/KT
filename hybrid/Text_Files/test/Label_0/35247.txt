/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class yamp {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        FastScanner in = new FastScanner(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        MATHYAMP solver = new MATHYAMP();
        solver.solve(1, in, out);
        out.close();
    }

    static class MATHYAMP {
        int N;
        int K;
        int[] nums;
        long oo = 987654231031415L;
        long[][] memo;
        int[] count;
        int pL;
        int pR;
        long cost;

        long cost(int a, int b) {
            while (pL < a) cost -= --count[nums[pL++]];
            while (pL > a) cost += count[nums[--pL]]++;
            while (pR > b) cost -= --count[nums[--pR]];
            while (pR < b) cost += count[nums[pR++]]++;
            return cost;
        }

        void solve(int L, int R, int sL, int sR, int k) {
            if (L > R) return;
            int mid = (L + R) / 2;
            int cap = Math.min(mid, sR);
            long best = memo[k - 1][cap] + cost(cap, mid);
            int bind = cap--;
            while (cap >= sL) {
                if (memo[k - 1][cap] + cost(cap, mid) < best) {
                    best = memo[k - 1][cap] + cost(cap, mid);
                    bind = cap;
                }
                cap--;
            }

            memo[k][mid] = best;
            solve(L, mid - 1, sL, bind, k);
            solve(mid + 1, R, bind, sR, k);

        }

        public void solve(int testNumber, FastScanner s, PrintWriter out) {

            N = s.nextInt();
            K = s.nextInt();
            pL = pR = 0;
            cost = 0L;
            nums = new int[N + 1];
            for (int i = 0; i < N; i++)
                nums[i] = s.nextInt();
            count = new int[1000001];

            memo = new long[K + 1][N + 1];
            for (long[] a : memo)
                Arrays.fill(a, oo);

            memo[0][0] = 0;
            for (int k = 1; k <= K; k++) {
                solve(0, N, 0, N, k);
            }

//        for (long[] row : memo)
//            out.println(Arrays.toString(row));

            out.println(memo[K][N]);

//        out.println(badans(0, N, K));

        }

    }

    static class FastScanner {
        private InputStream stream;
        private byte[] buf = new byte[1024];
        private int curChar;
        private int numChars;

        public FastScanner(InputStream stream) {
            this.stream = stream;
        }

        int read() {
            if (numChars == -1)
                throw new InputMismatchException();
            if (curChar >= numChars) {
                curChar = 0;
                try {
                    numChars = stream.read(buf);
                } catch (IOException e) {
                    throw new InputMismatchException();
                }
                if (numChars <= 0)
                    return -1;
            }
            return buf[curChar++];
        }

        boolean isSpaceChar(int c) {
            return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

        public String next() {
            int c = read();
            while (isSpaceChar(c))
                c = read();
            StringBuilder res = new StringBuilder();
            do {
                res.appendCodePoint(c);
                c = read();
            } while (!isSpaceChar(c));
            return res.toString();
        }

    }
}
