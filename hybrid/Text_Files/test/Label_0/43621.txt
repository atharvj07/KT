m = int(input())
l = []
for i in range(m):
	(a, b) = map(int, input().split())
	if b % 2 == 0:
		j = 1
	else:
		j = 0
	l.append(a - b + b // 2)
	s = ''
	for i in range(a, b, -1):
		s += str(i) + ' '
	for i in range(b, b // 2 - j, -1):
		if i != b:
			s += str(i) + ' '
	l.append(s[:-1])
for i in l:
	print(i)
