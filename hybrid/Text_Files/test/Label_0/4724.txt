#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, m;
  cin >> n >> m;
  if (n - m == 1) {
    for (int i = 0; i < n; i++) {
      cout << "0";
      if (i != n - 1) {
        cout << "1";
      }
    }
  } else if (m == n) {
    for (int i = 0; i < n; i++) {
      cout << "10";
    }
  } else if (m - n == 1) {
    for (int i = 0; i < m; i++) {
      cout << "1";
      if (i != m - 1) {
        cout << "0";
      }
    }
  } else if ((n + 1) * 2 >= m && m >= (n + 1)) {
    int rem = m - (n + 1);
    for (int i = 0; i < (n + 1); i++, rem--) {
      cout << "1";
      if (rem > 0) {
        cout << "1";
      }
      if (i != n) {
        cout << "0";
      }
    }
  } else {
    cout << "-1";
  }
  return 0;
}
