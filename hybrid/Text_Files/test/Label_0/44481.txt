(n, m) = map(int, input().split())
a = list(map(int, input().split()))
k = 0
have = [0] * n
need = set(range(1, n + 1))
for i in range(m):
	have[a[i] - 1] += 1
	if a[i] in need:
		need.remove(a[i])
	if len(need) == 0:
		print(1, end='')
		k += 1
		for i in range(n):
			if have[i] - k == 0:
				need.add(i + 1)
	else:
		print(0, end='')
