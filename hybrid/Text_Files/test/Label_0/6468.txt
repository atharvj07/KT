#include <bits/stdc++.h>
using namespace std;
int vis[300];
char s[205];
char ans[205];
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    memset(vis, 0, sizeof(vis));
    memset(ans, '0', sizeof(ans));
    int minn = 100, maxx = 100;
    scanf(" %s", s);
    int pd = 0;
    vis[s[0]] = 1;
    int now = 100;
    ans[now] = s[0];
    for (int i = 1; i < strlen(s); i++) {
      if (s[i] == s[i - 1]) continue;
      if (vis[s[i]] == 0) {
        if (ans[now + 1] == '0') {
          now++;
          ans[now] = s[i];
        } else if (ans[now - 1] == '0') {
          now--;
          ans[now] = s[i];
        } else {
          pd = 1;
          break;
        }
        vis[s[i]] = 1;
      } else {
        if (ans[now + 1] == s[i]) {
          now++;
          continue;
        }
        if (ans[now - 1] == s[i]) {
          now--;
          continue;
        }
        pd = 1;
        break;
      }
      maxx = max(maxx, now);
      minn = min(minn, now);
    }
    if (pd == 1) {
      printf("NO\n");
      continue;
    } else {
      printf("YES\n");
      for (int i = minn; i <= maxx; i++) printf("%c", ans[i]);
      for (int i = 'a'; i <= 'z'; i++) {
        if (vis[i] == 0) printf("%c", i);
      }
      printf("\n");
    }
  }
}
