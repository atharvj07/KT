import io, os
JJJ = input
input = lambda : JJJ().encode()
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	S = [x - ord('a') for x in input().strip()]
	E = [x - ord('a') for x in input().strip()]
	s = [0 for x in range(26)]
	e = [0 for x in range(26)]
	for c in S:
		s[c] += 1
	for c in E:
		e[c] += 1
	for i in range(26):
		if i == 25 and s[i] != e[i]:
			print('No')
			break
		elif i == 25:
			continue
		diff = s[i] - e[i]
		if diff < 0 or diff % k != 0:
			print('No')
			break
		s[i] -= diff
		s[i + 1] += diff
	else:
		print('Yes')
