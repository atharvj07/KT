	private static boolean isPrime(long x) {
		int top = (int) Math.sqrt(x)+1;
		for(int a=2;a<=top;a++){
			if(x%a==0)return false;
		}
		return true;
	}

	private static boolean isPal(String rev) {
//		String other = new StringBuilder(rev).reverse().toString();
//		if(other.compareTo(rev)==0)return true;
//		else return false;
		int N = rev.length()/2;
		for(int a=0;a<N;a++){
			if(rev.charAt(a)!=rev.charAt(rev.length()-1-a))return false;
		}
		return true;
	}
