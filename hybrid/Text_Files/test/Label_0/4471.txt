#include <iostream>
#include <vector>
#include <utility>
using namespace std;
int a[100010];
vector<pair<int,int>> v;
int main(){
	int i,m,n;
	cin >> m >> n;
	if(m==2){
		for(i=0;i<n;i++){
			cin >> a[i];
		}
		int cnt = n,cnt1 = 0;
		for(i=0;i<n;i++){
			if(i&1 && a[i]==1) cnt1++;
			if(!(i&1) && a[i]==2) cnt1++;
		}
		cnt = min(cnt,cnt1);
		cnt1 = 0;
		for(i=0;i<n;i++){
			if(i&1 && a[i]==2) cnt1++;
			if(!(i&1) && a[i]==1) cnt1++;
		}
		cnt = min(cnt,cnt1);
		cout << cnt << endl;
		return 0;
	}
	for(i=0;i<n;i++){
		int a; cin >> a;
		if(i==0 || v.back().first!=a){
			v.push_back({a,1});
		}else{
			v.back().second++;
		}
	}
	int cnt = 0;
	for(i=0;i<v.size();i++){
		cnt += v[i].second/2;
	}
	cout << cnt << endl;
}
