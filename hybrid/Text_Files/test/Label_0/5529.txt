#include <bits/stdc++.h>
using namespace std;
long long n, na, cf, cm, nm;
pair<int, int> p[100005];
int a[100005];
long long suff[100005], pre[100005];
int tim(int x, int sz) {
  int dau = 1, cuoi = sz;
  while (dau <= cuoi) {
    int giua = (dau + cuoi) >> 1;
    if (p[giua].first >= x)
      cuoi = giua - 1;
    else
      dau = giua + 1;
  }
  return cuoi;
}
long long tinh(int cnt) {
  long long res = cnt * cf;
  long long money = nm - suff[n - cnt + 1];
  long long dau = p[1].first, cuoi = na, giua, pos;
  while (dau <= cuoi) {
    giua = (dau + cuoi) >> 1;
    pos = tim(giua, n - cnt);
    long long need = giua * pos - pre[pos];
    if (need <= money)
      dau = giua + 1;
    else
      cuoi = giua - 1;
  }
  return res + cuoi * cm;
}
void solve(int cnt) {
  long long res = cnt * cf;
  long long money = nm - suff[n - cnt + 1];
  long long dau = p[1].first, cuoi = na, giua, pos;
  while (dau <= cuoi) {
    giua = (dau + cuoi) >> 1;
    pos = tim(giua, n - cnt);
    long long need = giua * pos - pre[pos];
    if (need <= money)
      dau = giua + 1;
    else
      cuoi = giua - 1;
  }
  res += cuoi * cm;
  pos = tim(cuoi, n - cnt);
  printf("%I64d\n", res);
  int i;
  for ((i) = (1); (i) <= (pos); (i)++) a[p[i].second] = cuoi;
  for ((i) = (n - cnt + 1); (i) <= (n); (i)++) a[p[i].second] = na;
  for ((i) = (1); (i) <= (n); (i)++) printf("%d ", a[i]);
}
void input() {
  int i, cntmax, resmax;
  long long res, val;
  scanf("%I64d %I64d %I64d %I64d %I64d", &n, &na, &cf, &cm, &nm);
  for ((i) = (1); (i) <= (n); (i)++)
    scanf("%d", &a[i]), p[i] = make_pair(a[i], i);
  sort(p + 1, p + n + 1);
  cntmax = 0;
  i = n;
  while (i >= 1 && p[i].first == na) cntmax++, i--;
  suff[n + 1] = 0;
  for ((i) = (n); (i) >= (1); (i)--) suff[i] = suff[i + 1] + (na - p[i].first);
  for ((i) = (1); (i) <= (n); (i)++) pre[i] = pre[i - 1] + p[i].first;
  res = -1000111000111000111LL;
  for ((i) = (0); (i) <= (n - cntmax); (i)++) {
    if (nm >= suff[n - cntmax - i + 1]) {
      val = tinh(cntmax + i);
      if (val > res) res = val, resmax = cntmax + i;
    } else
      break;
  }
  solve(resmax);
}
int main() {
  input();
  return 0;
}
