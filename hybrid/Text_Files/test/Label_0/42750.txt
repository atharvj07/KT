s = input().split()
(n, m) = (int(s[0]), int(s[1]))
cl = list(map(int, input().split()))
st = set()
count = 1
pos = n * [0]
res = ''
for i in range(m):
	st.add(cl[i])
	pos[cl[i] - 1] += 1
	if len(st) == n:
		res += '1'
		st = set()
		count += 1
		for i in range(n):
			if pos[i] >= count:
				st.add(i + 1)
	else:
		res += '0'
print(res)
