def list2d(a, b, c):
	return [[c] * b for i in range(a)]

def list3d(a, b, c, d):
	return [[[d] * c for j in range(b)] for i in range(a)]

def list4d(a, b, c, d, e):
	return [[[[e] * d for j in range(c)] for j in range(b)] for i in range(a)]

n = int(ri())
arr = []
dic = {}
no = -1
dic1 = {}
ind = [-1] * (n + 1)
for i in range(n - 2):
	temp = Ri()
	arr.append(temp)
	for j in temp:
		ind[j] = i
		if j in dic:
			dic[j] += 1
		else:
			dic[j] = 1
	if tuple(sorted((temp[0], temp[1]))) in dic1:
		dic1.get(tuple(sorted((temp[0], temp[1])))).append(temp[2])
	else:
		dic1[tuple(sorted((temp[0], temp[1])))] = [temp[2]]
	if tuple(sorted((temp[2], temp[1]))) in dic1:
		dic1.get(tuple(sorted((temp[2], temp[1])))).append(temp[0])
	else:
		dic1[tuple(sorted((temp[2], temp[1])))] = [temp[0]]
	if tuple(sorted((temp[0], temp[2]))) in dic1:
		dic1.get(tuple(sorted((temp[0], temp[2])))).append(temp[1])
	else:
		dic1[tuple(sorted((temp[0], temp[2])))] = [temp[1]]
for i in dic:
	if dic[i] == 1:
		no = i
		break
ans = []
ans.append(no)
tup = arr[ind[no]]
(x, y) = (-1, -1)
for i in tup:
	if x == -1 and i != no:
		x = i
		continue
	if y == -1 and i != no:
		y = i
(fi, se) = (-1, -1)
if dic[y] == 2:
	ans.append(y)
	fi = y
	se = x
else:
	ans.append(x)
	fi = x
	se = y
ans.append(se)
while len(ans) < n:
	th = dic1[tuple(sorted((ans[-1], ans[-2])))]
	for i in th:
		if i != ans[-3]:
			th = i
			break
	ans.append(th)
print(*ans)
