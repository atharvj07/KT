#include <bits/stdc++.h>
using namespace std;
int siz[100005], ch[100005][2], cnt, fa[100005], root;
long long f[100005], tag[100005], ans, now, K, Y;
int get(int x) { return ch[fa[x]][1] == x; }
void pushup(int x) { siz[x] = siz[ch[x][0]] + siz[ch[x][1]] + 1; }
void rotate(int x) {
  int f = fa[x], g = fa[f], c = get(x);
  if (!f) return;
  fa[x] = g;
  if (g) ch[g][get(f)] = x;
  ch[f][c] = ch[x][c ^ 1];
  if (ch[x][c ^ 1]) fa[ch[x][c ^ 1]] = f;
  ch[x][c ^ 1] = f;
  fa[f] = x;
  pushup(f), pushup(x);
}
void pushr(int x, long long y) {
  if (x) f[x] += y, tag[x] += y;
}
void pushdown(int x) {
  if (tag[x]) pushr(ch[x][0], tag[x]), pushr(ch[x][1], tag[x]), tag[x] = 0;
}
void splay(int x) {
  while (fa[x]) {
    int f = fa[x];
    if (fa[f]) rotate(get(f) == get(x) ? f : x);
    rotate(x);
  }
  root = x;
}
void insert(int &now, int pr = 0) {
  if (!now) {
    now = ++cnt;
    f[now] = (K + 1) * Y;
    siz[now] = 1;
    fa[now] = pr;
    splay(now);
    return;
  }
  pushdown(now), siz[now]++;
  if ((K + siz[ch[now][0]] + 1) * Y >= f[now])
    f[now] += Y, pushr(ch[now][1], Y), insert(ch[now][0], now);
  else
    K += siz[ch[now][0]] + 1, insert(ch[now][1], now);
}
void dfs(int x) {
  if (!x) return;
  pushdown(x);
  dfs(ch[x][0]);
  now += f[x];
  ans = max(ans, now);
  dfs(ch[x][1]);
}
int main() {
  int n, x;
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d", &x), K = 0, Y = x, insert(root);
  dfs(root), printf("%lld", ans);
  return 0;
}
