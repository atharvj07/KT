from itertools import permutations
n = int(input())
nums = [int(x) for x in input().split(' ')]
cnt_100 = 0
cnt_200 = 0
for num in nums:
	if num == 100:
		cnt_100 += 1
	else:
		cnt_200 += 1
if cnt_100 % 2 == 0 and cnt_200 % 2 == 0 and (n > 1):
	print('YES')
elif cnt_200 % 2 != 0 and cnt_100 % 2 != 0 and (n > 1):
	print('NO')
elif cnt_100 < cnt_200 and cnt_100 % 2 != 0 and (cnt_200 % 2 == 0) and (n > 1):
	print('NO')
elif cnt_100 > cnt_200 and cnt_100 % 2 == 0 and (cnt_200 % 2 != 0) and (n > 1):
	print('YES')
elif cnt_200 > cnt_100 and cnt_200 % 2 != 0 and (cnt_100 % 2 == 0) and (cnt_100 >= 2):
	print('YES')
else:
	print('NO')
