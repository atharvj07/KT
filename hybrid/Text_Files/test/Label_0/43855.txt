n = int(input())
s = input()
res = 0
(ri, rj) = (0, 0)
for i in range(n):
	for j in range(i + 1, n):
		ss = s[:i] + s[j] + s[i + 1:j] + s[i] + s[j + 1:]
		(k_mn, mn) = (0, 0)
		teck = 0
		for g in range(n):
			if ss[g] == '(':
				teck += 1
			else:
				teck -= 1
			if k_mn == 0 or teck < mn:
				(mn, k_mn) = (teck, 1)
			elif teck == mn:
				k_mn += 1
		if teck == 0:
			if k_mn > res:
				res = max(res, k_mn)
				ri = i
				rj = j
print(res)
print(ri + 1, rj + 1)
