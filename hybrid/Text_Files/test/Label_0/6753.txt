#define mod 1000000007 //1e9+7(prime number)
#define INF 1000000000 //1e9
#define LLINF 2000000000000000000LL //2e18
#define SIZE 100010


vector<int> way[SIZE];

pair<int,int> dfs(int now, int back=-1,int h=0){
  pair<int,int> ret = {h,now};

  for(int i=0;i<way[now].size();i++){
    if(way[now][i] != back)
      ret = max(dfs(way[now][i],now,h+1),ret);
  }

  return ret;
}

int ans[SIZE];

void dfs2(int now, int back=-1,int h=0){

  ans[now] = max(ans[now],h);
  for(int i=0;i<way[now].size();i++){
    if(way[now][i] != back)
      dfs2(way[now][i],now,h+1);
  }

  return;
}

int main(){
  int n;
  int u,v;
  
  scanf("%d",&n);

  for(int i=0;i<n-1;i++){
    scanf("%d%d",&u,&v);
    v--; u--;
    way[u].push_back(v);
    way[v].push_back(u);
  }

  int p1,p2;
  
  pair<int,int> ret;
  ret = dfs(0);
  p1 = ret.second;
  ret = dfs(p1);
  p2 = ret.second;

  dfs2(p1);
  dfs2(p2);

  for(int i=0;i<n;i++){
    printf("%d\n",(n-1)*2-ans[i]);
  }
  
  return 0;
}
