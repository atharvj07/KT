import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class B198 {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int N = in.nextInt();
        int K = in.nextInt();
        char[] L = in.next().toCharArray();
        char[] R = in.next().toCharArray();
        boolean[][] seen = new boolean[2][N];
        Queue<Data> queue = new LinkedList<>();
        Data data = new Data(true, 0, 0);
        queue.add(data);
        boolean found = false;
        while (!queue.isEmpty()) {
            data = queue.poll();
            int jumpHeight = data.height + K;
            if (jumpHeight >= N) {
                found = true;
                break;
            } else {
                Data newData = new Data(!data.left, jumpHeight, data.time+1);
                if ((newData.left ? L : R)[newData.height] == '-' && !seen[newData.left ? 0 : 1][newData.height]) {
                    queue.add(newData);
                    seen[newData.left ? 0 : 1][newData.height] = true;
                }
            }
            if ((data.left ? L : R)[data.height+1] == '-' && !seen[data.left ? 0 : 1][data.height+1]) {
                queue.add(new Data(data.left, data.height+1, data.time+1));
                seen[data.left ? 0 : 1][data.height+1] = true;
            }
            Data newData = new Data(data.left, data.height-1, data.time+1);
            if (newData.height >= newData.time && (newData.left ? L : R)[newData.height] == '-' && !seen[newData.left ? 0 : 1][newData.height]) {
                queue.add(newData);
                seen[newData.left ? 0 : 1][newData.height] = true;
            }
        }
        System.out.println(found ? "YES" : "NO");
    }

    static class Data {
        boolean left;
        int height;
        int time;

        Data(boolean left, int height, int time) {
            this.left = left;
            this.height = height;
            this.time = time;
        }
    }

}
