import java.io.*;
import java.util.*;

public class GB2015E
{
    public static StringTokenizer st;
    public static void nextLine(BufferedReader br) throws IOException
    {
        st = new StringTokenizer(br.readLine());
    }
    
    public static String next()
    {
        return st.nextToken();
    }
    
    public static int nextInt()
    {
        return Integer.parseInt(st.nextToken());
    }
    
    public static long nextLong()
    {
        return Long.parseLong(st.nextToken());
    }
    
    public static double nextDouble()
    {
        return Double.parseDouble(st.nextToken());
    }
    
    public static void main(String[] args) throws IOException
    {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        nextLine(br);
        int n = nextInt();
        nextLine(br);
        int[] nums = new int[] {nextInt(), nextInt(), nextInt()};
        Arrays.sort(nums);
        
        int a = nums[0];
        int b = nums[1];
        int c = nums[2];
        boolean flag = c > a+b;
        
        nextLine(br);
        int[] crim = new int[n];
        for (int i = 0; i < n; i++)
        {
            crim[i] = nextInt();
        }
        Arrays.sort(crim);
        int[] limits = new int[8];
        int[] cat = new int[8];
        limits[0] = a;
        limits[1] = b;
        limits[2] = c;
        limits[3] = a+b;
        limits[4] = a+c;
        limits[5] = b+c;
        limits[6] = a+b+c;
        limits[7] = 1000000000;
        Arrays.sort(limits);
        int cc = 0;
        for (int i = 0; i < n; i++)
        {
            while (crim[i] > limits[cc])
            {
                cc++;
            }
            cat[cc]++;
        }
        //cat = new int[] {48040, 33882, 23920, 23846, 23970, 34305, 12037, 0};
        //System.out.println(Arrays.toString(cat));
        if (cat[7] > 0)
        {
            System.out.println(-1);
            return;
        }
        int ans = 0;
        ans += cat[6]; // remove a+b+c
        ans += cat[5]; // remove b+c, with optional a
        cat[0] = Math.max(0, cat[0] - cat[5]);
        ans += cat[4]; // remove a+c, with optional b (taking from a if excess)
        if (cat[1] >= cat[4])
        {
            cat[1] -= cat[4];
        }
        else
        {
            cat[0] = Math.max(0, cat[0] - (cat[4] - cat[1]));
            cat[1] = 0;
        }
        //System.out.println(ans + " " + Arrays.toString(cat));
        
        if (flag)
        {
            // c > a+b, try to kill triplets first
            int rem = Math.min(cat[3], Math.min(cat[0], cat[1]));
            ans += rem;
            cat[3] -= rem;
            cat[0] -= rem;
            cat[1] -= rem;
            
            if (cat[0] > 0 && cat[3] > 0)
            {
                rem = Math.min(cat[0] / 2, cat[3]);
                ans += rem;
                cat[0] -= rem * 2;
                cat[3] -= rem;
            }
            // now kill rest of cs, taking out a+bs or remaining a,bs
            rem = cat[3];
            ans += rem;
            for (int i = 2; i >= 0; i--)
            {
                if (cat[i] >= rem)
                {
                    cat[i] -= rem;
                    break;
                }
                else
                {
                    rem -= cat[i];
                    cat[i] = 0;
                }
            }
        }
        else
        {
            // a+b > c. All a+bs must be gotten rid of, with optional c,b or a
            int rem = cat[3];
            ans += rem;
            for (int i = 2; i >= 0; i--)
            {
                if (cat[i] >= rem)
                {
                    cat[i] -= rem;
                    break;
                }
                else
                {
                    rem -= cat[i];
                    cat[i] = 0;
                }
            }
        }
        //System.out.println(ans + " " + Arrays.toString(cat));
        // at this point only cat[0-2] are left, and cat[2] can be taken out by either a+b or c
        // Kill triplets
        int rem = Math.min(cat[2], Math.min(cat[0], cat[1]));
        ans += rem;
        cat[2] -= rem;
        cat[0] -= rem;
        cat[1] -= rem;
        if (cat[0] == 0)
        {
            ans += (cat[1] + cat[2] + 1) / 2;
        }
        else if (cat[1] == 0)
        {
            if (cat[2] * 2 <= cat[0])
            {
                ans += (cat[0] + cat[2] + 2) / 3;
            }
            else
            {
                rem = cat[0] / 2;
                ans += rem;
                cat[2] -= rem;
                cat[0] -= rem*2;
                ans += (cat[0] + cat[2] + 1) / 2;
            }
        }
        else if (cat[2] == 0)
        {
            if (cat[1] <= cat[0] * 2)
            {
                ans += (cat[0] + cat[1] + 2) / 3;
            }
            else
            {
                ans += cat[0];
                cat[1] -= 2*cat[0];
                ans += (cat[1] + 1) / 2;
            }
        }
        
        System.out.println(ans);
    }
}
