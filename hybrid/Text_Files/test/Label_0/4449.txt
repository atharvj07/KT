#include <iostream>
#define INF 1e+9
using namespace std;

int n,w[1000],c[1000],imos[1001] = {};
bool dp1[1000][1000];
int dp2[1001];

int main(){
	cin >> n;
	for(int i = 0;i < n;i++){
		cin >> c[i] >> w[i];
		imos[i + 1] = imos[i] + w[i];
		dp1[i][i] = true;
		dp2[i + 1] = INF;
	}
	for(int i = 1;i < n;i++){
        for(int j = 0;j < n - i;j++){
            if((c[j + i] >= imos[j + i] - imos[j] && dp1[j][j + i - 1]) || (c[j] >= imos[j + i + 1] - imos[j + 1] && dp1[j + 1][j + i])) dp1[j][j + i] = true;
            else dp1[j][i + j] = false;
        }
    }
	for(int i = 1;i <= n;i++){
		for(int j = 0;j <= i;j++){
			if(dp1[j][i - 1]) dp2[i] = min(dp2[i],dp2[j] + 1);
		}
	}
	cout << dp2[n] << endl;
	return 0;
}
