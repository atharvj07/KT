#include <bits/stdc++.h>
using namespace std;
int main(int argc, char** argv) {
  int t;
  cin >> t;
  while (t--) {
    int num;
    cin >> num;
    int num_3 = 0, num_5 = 0, num_7 = 0;
    int res = num % 10;
    if (res == 0) {
      num_5 = num / 5;
    }
    if (res == 1) {
      num_3 = 2;
      num_5 = (num - 11) / 5 + 1;
    }
    if (res == 2) {
      num_3 = 4;
      num_5 = (num - 12) / 5;
    }
    if (res == 3) {
      num_3 = 1;
      num_5 = (num - 3) / 5;
    }
    if (res == 4) {
      num_5 = (num - 14) / 5;
      num_7 = 2;
    }
    if (res == 5) {
      num_5 = num / 5;
    }
    if (res == 6) {
      num_3 = 2;
      num_5 = (num - 6) / 5;
    }
    if (res == 7) {
      num_7 = 1;
      num_5 = (num - 7) / 5;
    }
    if (res == 8) {
      num_3 = 1;
      num_5 = (num - 8) / 5 + 1;
    }
    if (res == 9) {
      num_3 = 3;
      num_5 = (num - 9) / 5;
    }
    if (num_3 < 0 || num_5 < 0 || num_7 < 0)
      cout << -1 << endl;
    else
      cout << num_3 << " " << num_5 << " " << num_7 << " " << endl;
  }
  return 0;
}
