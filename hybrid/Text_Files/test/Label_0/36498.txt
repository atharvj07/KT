import java.io.*;
import java.util.*;
 
public class A
{
   public static void main(String[] args)
   {
      Scanner scan = new Scanner(System.in);
      int n = scan.nextInt();
      boolean[] pass = new boolean[n+1];
      
      int p = scan.nextInt();
      for(int i = 0; i<p; i++)
      {
         int x = scan.nextInt();
         pass[x] = true;
      }
      
      p = scan.nextInt();
      for(int i = 0; i<p; i++)
      {
         int x = scan.nextInt();
         pass[x] = true;
      }
      
      boolean flag = true;
      
      for(int i = 1; i<=n; i++)
      {
         if(pass[i] == false)
            flag = false;
      }
      if(flag)
      {
         System.out.println("I become the guy.");
      }
      else
      {
         System.out.println("Oh, my keyboard!");
      }
   }
}
