import java.util.HashSet;
import java.util.Scanner;
import java.io.*;
import java.util.Set;

public class ColorfulGraph {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        int m = in.nextInt();
        int[] colors = new int[n];

        int maxColor = Integer.MAX_VALUE;
        int maxNum = 0;

        for (int i = 0; i < n; i++) {
            colors[i] = in.nextInt() - 1;
            if (colors[i] < maxColor) {
                maxColor = colors[i];
            }
        }
        int[] from = new int[m];
        int[] to = new int[m];
        Set<Integer>[] tree = new HashSet[100000];



        for (int i = 0; i < m; i++) {
            from[i] = in.nextInt() - 1;
            to[i] = in.nextInt() - 1;
        }

        for (int i = 0; i < 100000; i++) {
            tree[i] = new HashSet<>();
        }



        for (int i = 0; i < m; i++) {
            //take from[i] convert to color, and then take to[i] convert to color. then add eachother to set. check length of set
            int color1 = colors[from[i]];
            int color2 = colors[to[i]];

            if (!tree[color1].contains(color2)) {
                if (color1 != color2) {
                    tree[color1].add(color2);

                    if (tree[color1].size() > maxNum) {
                        maxColor = color1;
                        maxNum = tree[color1].size();
                    } else if (tree[color1].size() == maxNum) {
                        if (maxColor > color1) {
                            maxColor = color1;
                        }
                    }
                }
            }

            if (!tree[color2].contains(color1)) {
                if (color2 != color1) {
                    tree[color2].add(color1);

                    if (tree[color2].size() > maxNum) {
                        maxColor = color2;
                        maxNum = tree[color2].size();
                    } else if (tree[color2].size() == maxNum) {
                        if (maxColor > color2) {
                            maxColor = color2;
                        }
                    }
                }
            }
        }
        
        System.out.println(maxColor + 1);
    }
}
