#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 10;
const long long MOD = 1e9 + 7;
long long inline in() {
  long long x = 0, flag = 1;
  char ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') flag = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9')
    x = (x << 3) + (x << 1) + (ch ^ 48), ch = getchar();
  return x * flag;
}
long long n, x;
long long tot, p[MAXN];
bool flag[MAXN];
void filter() {
  for (int i = 2; i < MAXN; ++i) {
    if (!flag[i]) p[++tot] = i;
    for (int j = 1; j <= tot && i * p[j] < MAXN; ++j) {
      flag[i * p[j]] = true;
      if (i % p[j] == 0) break;
    }
  }
}
int cnt;
long long rec[MAXN];
long long ans = 1;
long long fpow(long long a, long long b) {
  long long ret = 1;
  while (b) {
    if (b % 2 == 1)
      b--, (ret *= a) %= MOD;
    else
      b /= 2, (a *= a) %= MOD;
  }
  return ret;
}
int main() {
  x = in();
  n = in();
  filter();
  for (int i = 1; i <= tot; ++i) {
    if (x % p[i] == 0) rec[++cnt] = p[i];
    while (x % p[i] == 0 && x != 0) x /= p[i];
  }
  if (x != 1) rec[++cnt] = x;
  for (int i = 1; i <= cnt; ++i) {
    for (long long now = rec[i]; now <= n; now *= rec[i]) {
      (ans *= fpow(now % MOD, n / now - n / now / rec[i])) %= MOD;
      if (n / now < rec[i]) break;
    }
  }
  printf("%lld\n", ans);
  return 0;
}
