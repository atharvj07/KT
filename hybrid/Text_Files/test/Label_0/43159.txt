for _ in range(int(input())):
	(n, k) = map(int, input().split())
	dp = []
	for i in range(0, 1005):
		dp.append([0] * 1005)
	ans = 1
	for i in range(1, k + 1):
		for j in range(0, n + 1):
			if j == 0 or i == 1:
				dp[i][j] = 1
			else:
				dp[i][j] = (dp[i - 1][n - j] + dp[i][j - 1]) % 1000000007
	print(dp[k][n])
