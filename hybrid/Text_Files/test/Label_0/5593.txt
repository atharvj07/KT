long long rnd(long long x, long long y) {
  static auto gen = std::bind(std::uniform_int_distribution<long long>(),
                              std::mt19937(960172));
  return gen() % (y - x + 1) + x;
}
long long gcd(long long a, long long b) {
  while (b > 0) {
    long long t = a % b;
    a = b;
    b = t;
  }
  return a;
}

int const N = 1003;
int const LG = 11;
int a[N][N];
short t[LG][LG][N][N];
int logs[N];
int get_max(int x1, int y1, int x2, int y2) {
  if (x2 <= x1 || y2 <= y1) {
    return 0;
  }
  int xx = logs[x2 - x1];
  int yy = logs[y2 - y1];
  return max(max(t[xx][yy][x1][y1], t[xx][yy][x1][y2 - (1 << yy)]),
             max(t[xx][yy][x2 - (1 << xx)][y1],
                 t[xx][yy][x2 - (1 << xx)][y2 - (1 << yy)]));
}
void solve() {
  logs[1] = 0;
  for (int i = 2; i < N; ++i) {
    logs[i] = 1 + logs[i / 2];
  }
  int n, m;
  scanf("%d%d", &n, &m);
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      scanf("%d", a[i] + j);
    }
  }
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      if (!a[i][j]) {
        t[0][0][i][j] = 0;
        continue;
      }
      if (i > 0 && j > 0) {
        int len = min(t[0][0][i - 1][j], t[0][0][i][j - 1]);
        if (a[i - len][j - len]) {
          ++len;
        }
        t[0][0][i][j] = len;
      } else {
        t[0][0][i][j] = 1;
      }
    }
  }
  for (int i = 0; 1 << i <= n; ++i) {
    int xlen = 1 << i;
    for (int j = 0; 1 << j <= m; ++j) {
      int ylen = 1 << j;
      for (int x = 0; x + xlen <= n; ++x) {
        for (int y = 0; y + ylen <= m; ++y) {
          if (i == 0) {
            if (j > 0) {
              t[i][j][x][y] =
                  max(t[i][j - 1][x][y], t[i][j - 1][x][y + ylen / 2]);
            }
          } else {
            t[i][j][x][y] =
                max(t[i - 1][j][x][y], t[i - 1][j][x + xlen / 2][y]);
          }
        }
      }
    }
  }
  int q;
  scanf("%d", &q);
  for (int it = 0; it < q; ++it) {
    int x1, y1, x2, y2;
    scanf("%d%d%d%d", &x1, &y1, &x2, &y2);
    --x1, --y1;
    int l = 0;
    int r = n + 1;
    while (r - l > 1) {
      int m = (l + r) / 2;
      int cur = get_max(x1 + m - 1, y1 + m - 1, x2, y2);
      if (cur >= m) {
        l = m;
      } else {
        r = m;
      }
    }
    printf("%d\n", l);
  }
}
int main() {
  cout.precision(15);
  cout << fixed;
  cerr.precision(6);
  cerr << fixed;
  solve();
}
