import java.util.Scanner;
public class Ishu

{
    
	public static void main(String[] args)
    
	{
        
	 Scanner scan=new Scanner(System.in);    
 
	 int x1,x2,y1,y2,sum=0,d1,d2,sd;
	 char c='a';
	 x1=scan.nextInt();
	 y1=scan.nextInt();
	 x2=scan.nextInt();
	 y2=scan.nextInt();	
	 d1=x2-x1;
	 d2=y2-y1;
	 if(d1<0)	
		d1*=(-1);
	 if(d2<0)
		d2*=(-1);
	 sd=d1<=d2?d1:d2;
	 if(sd==d1)
		c='y';
	 if(sd==d2)
		c='x';
	 sum+=sd;	
	 if(c=='y')
		sum+=(d2-sd);
	 if(c=='x')
		sum+=(d1-sd);
	 System.out.println(sum);
	}
}
