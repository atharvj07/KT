#include <bits/stdc++.h>
using namespace std;
int ans1,ans2,a[200001],b[200001],r[200001],MAX;
int sc[200001];
int xc[200001];
int main()
{
	int n;
	scanf("%d",&n);
	for(int i=1;i<=n;i++)
	{
		scanf("%d%d",&a[i],&b[i]);
		sc[a[i]]++;
		xc[b[i]]++;
		r[a[i]]++;
		r[b[i]]--;
		MAX=max(MAX,b[i]);
	}
	for(int i=1;i<=MAX;i++)
	{
		sc[i]+=sc[i-1];
		xc[i]+=xc[i-1];
	}
	for(int i=1;i<=n;i++)
	ans1=max(ans1,sc[b[i]-1]-xc[a[i]]);
	for(int i=1;i<=MAX;i++)
	{
		r[i]+=r[i-1];
		ans2=max(ans2,r[i]);
	}
	printf("%d %d\n",ans1,ans2);
	return 0;
}
