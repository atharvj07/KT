import io
import os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
(n, q) = map(int, input().split())
for _ in range(q):
	x = int(input())
	for r in range(100):
		num = 2 * n - 2 ** r - x
		den = 2 ** (r + 1)
		if num % den == 0 and n - num // den >= 1 and (n - num // den <= n):
			print(n - num // den)
			break
