#include <bits/stdc++.h>
using namespace std;
int main(){
  unsigned long long flg = 0;
  int n;
  cin >> n;
  vector<unsigned long long> M(n, 0);
  for (int i = 0; i < n; i++){
    int k;
    cin >> k;
    for (int j = 0; j < k; j++){
      int b;
      cin >> b;
      M[i] |= (1LL << b);
    }
  }
  int q;
  cin >> q;
  for (int i = 0; i < q; i++){
    int t;
    cin >> t;
    if (t == 0){
      int i;
      cin >> i;
      cout << ((flg >> i) & 1LL) << endl;
    } else {
      int m;
      cin >> m;
      if (t == 1) flg |= M[m];
      if (t == 2) flg &= (~M[m]);
      if (t == 3) flg ^= M[m];
      if (t == 4) cout << ((flg & M[m]) == M[m]) << endl;
      if (t == 5) cout << ((flg & M[m]) != 0) << endl;
      if (t == 6) cout << ((flg & M[m]) == 0) << endl;
      if (t == 7) cout << __builtin_popcountll(flg & M[m]) << endl;
      if (t == 8) cout << (flg & M[m]) << endl;
    }
  }
}
