from collections import *
import sys

def ri():
	return int(input())

def rl():
	return list(map(int, input().split()))
t = ri()
for _ in range(t):
	n = ri()
	aa = rl()
	ans = 0
	i = 0
	j = 1
	while i < n:
		if aa[i] <= 0:
			ans -= aa[i]
			i += 1
		else:
			if j <= i:
				j = i + 1
			while j < n and aa[i] > 0:
				if aa[i] + aa[j] > 0:
					aa[i] += aa[j]
					aa[j] = 0
					j += 1
				else:
					aa[j] += aa[i]
					aa[i] = 0
					break
			i += 1
	print(ans)
