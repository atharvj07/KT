import java.io.*;
import java.util.*;

public class E {

	BufferedReader in;
	StringTokenizer st;
	PrintWriter out;

	String next() throws IOException {
		while (st == null || !st.hasMoreTokens())
			st = new StringTokenizer(in.readLine());
		return st.nextToken();
	}

	int nextInt() throws Exception {
		return Integer.parseInt(next());
	}

	long nextLong() throws Exception {
		return Long.parseLong(next());
	}

	double nextDouble() throws Exception {
		return Double.parseDouble(next());
	}

	static int n;
	static int a[];
	static int b[];
	static long c;

	static long foo(long x) {
		long ans = 0;

		for (int i = 0; i < n; i++) {
			double add = a[i];
			add *= x;
			add /= b[i];
			if (add > 1e10)
				return Long.MAX_VALUE;
			ans += Math.floor(add);
		}
		return ans + n;
	}

	static long MAXV = Long.MAX_VALUE / 2;

	public static long LB(long key) {
		long lo = -1;
		long hi = MAXV;
		while (hi - lo > 1) {
			long mid = (lo + hi) / 2;
			long midVal = foo(mid);
			if (midVal < key) {
				lo = mid;
			} else {
				hi = mid;
			}
		}
		return hi;
	}

	public static long UB(long key) {
		long lo = -1;
		long hi = MAXV;
		while (hi - lo > 1) {
			long mid = (lo + hi) / 2;
			long midVal = foo(mid);
			if (midVal <= key) {
				lo = mid;
			} else {
				hi = mid;
			}
		}
		return hi;
	}

	void solve() throws Exception {
		n = nextInt();
		c = nextLong();
		a = new int[n];
		b = new int[n];
		for (int i = 0; i < n; i++) {
			a[i] = nextInt();
			b[i] = nextInt();
			if (a[i] == 0) {
				out.println(-1);
				return;
			}
		}
		if (n > c) {
			out.println(0);
			return;
		}

		// for (int i = 0; i < n; i++)
		// b[i] = nextInt();

		// if (binarySearchFirstB() == binarySearchFirstE())
		// out.println(0);
		// System.err.println(LB(c) + " " + UB(c));
		long fix = 0;
		if (foo(0) == c)
			fix++;
		out.println(Math.max(UB(c) - LB(c) - fix, 0));
	}

	void run() {
		try {
			Locale.setDefault(Locale.US);
			boolean oj = System.getProperty("ONLINE_JUDGE") != null;
			Reader reader = oj ? new InputStreamReader(System.in) : new FileReader("input.txt");
			Writer writer = oj ? new OutputStreamWriter(System.out) : new FileWriter("output.txt");
			in = new BufferedReader(reader);
			out = new PrintWriter(writer);
			solve();
			out.close();
		} catch (Exception e) {
			e.printStackTrace();
			System.exit(1);
		}
	}

	public static void main(String[] args) {
		new E().run();

	}

}
