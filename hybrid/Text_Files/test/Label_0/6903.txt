#include <bits/stdc++.h>
using namespace std;
const int C = 4000;
int n, p, m;
int pre[4050][4050], suf[4050][4050];
struct data {
  int cost, value, time, id;
  bool operator<(const data b) const { return time < b.time; }
} a[4050];
int read() {
  char c = getchar();
  int x = 0;
  while (!isdigit(c)) c = getchar();
  while (isdigit(c)) {
    x = (x << 3) + (x << 1) + c - '0';
    c = getchar();
  }
  return x;
}
void build() {
  for (int i = 1; i <= n; ++i)
    if (a[i].id != a[i - 1].id) {
      for (int j = C; j >= a[i].cost; --j) pre[i][j] = a[i].value;
    } else {
      memcpy(pre[i], pre[i - 1], sizeof(pre[i]));
      for (int j = C; j >= a[i].cost; --j)
        pre[i][j] = max(pre[i][j], pre[i - 1][j - a[i].cost] + a[i].value);
    }
  for (int i = n; i >= 1; --i)
    if (a[i].id != a[i + 1].id) {
      for (int j = C; j >= a[i].cost; --j) suf[i][j] = a[i].value;
    } else {
      memcpy(suf[i], suf[i + 1], sizeof(suf[i]));
      for (int j = C; j >= a[i].cost; --j)
        suf[i][j] = max(suf[i][j], suf[i + 1][j - a[i].cost] + a[i].value);
    }
}
int main() {
  n = read();
  p = read();
  for (int i = 1; i <= n; ++i) {
    a[i].cost = read();
    a[i].value = read();
    a[i].id = (a[i].time = read()) / p + 1;
  }
  sort(a + 1, a + n + 1);
  build();
  m = read();
  while (m--) {
    int cur = read(), cash = read(), ans = 0, l = 0, r = 0;
    for (int i = 1; i <= n; ++i)
      if (a[i].time <= cur && a[i].time + p > cur) {
        if (!l) l = i;
        r = i;
      }
    if (a[l].id == a[r].id) {
      if (r < n && a[r].id == a[r + 1].id)
        l = 0;
      else
        r = 0;
    }
    for (int i = 0; i <= cash; ++i)
      ans = max(ans, suf[l][i] + pre[r][cash - i]);
    printf("%d\n", ans);
  }
}
