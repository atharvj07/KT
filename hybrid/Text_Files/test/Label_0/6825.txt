#include <bits/stdc++.h>
using namespace std;
const int UNDEF = -1;
const int INF = 2000000000;
const double EPS = 1e-9;
const double PI = 3.1415926535897932384626433832795;
template <typename T>
T sqr(const T &x) {
  return x * x;
}
template <typename T>
string GetString(const T &val) {
  ostringstream sout;
  sout << val;
  return sout.str();
}
long long n;
long long resMin = 9223372032559808512;
long long resMax = -resMin;
void F(long long x) {
  long long tmp;
  long long border = (long long)(sqrt(x + 0.0) + 0.001);
  for (long long i = 1; i <= border; ++i) {
    if (!(x % i)) {
      tmp = (n / x + 1) * (i + 2) * (x / i + 2) - n;
      resMin = min(resMin, tmp);
      resMax = max(resMax, tmp);
    }
  }
}
int main() {
  cin >> n;
  long long border = (long long)(sqrt(n + 0.0) + 0.001);
  for (long long i = 1; i <= border; ++i) {
    if (!(n % i)) {
      F(i);
      F(n / i);
    }
  }
  cout << resMin << ' ' << resMax << endl;
  return 0;
}
