import java.util.*;
public class TwoNeighbourDistinct {

    /**
     * @param args
     */
    public static void main(String[] args) {
        Scanner in = new Scanner (System.in);
        int n = in.nextInt();
        
        int[] numCount= new int [1001];
        
        int max=0;
        
        for (int i = 0; i < n; i++) {
            int num=in.nextInt();
            numCount[num]++;
            max =Math.max(max, numCount[num]);
        }
int half =n/2;
if (n%2!=0)
{
    half++;
}
    if (max>half)
    {
        System.out.println("NO");
    }else
    {
        System.out.println("YES");
    }
    
    
    }

}
