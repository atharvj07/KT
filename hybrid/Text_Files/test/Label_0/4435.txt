#include <bits/stdc++.h>
using namespace std;
int sta[200005], arr[200005], l[200005], r[200005], ma[200005];
int main() {
  int n, tp = 0;
  scanf("%d", &n);
  arr[0] = arr[n + 1] = 0x3f3f3f3f;
  sta[++tp] = 0;
  for (int i = 1; i <= n; i++) {
    scanf("%d", &arr[i]);
    ma[arr[i]] = i;
    while (tp && arr[sta[tp]] < arr[i]) tp--;
    l[i] = sta[tp] + 1;
    sta[++tp] = i;
  }
  int ans = 0;
  tp = 0;
  sta[++tp] = n + 1;
  for (int i = n; i >= 1; i--) {
    while (tp && arr[sta[tp]] < arr[i]) tp--;
    r[i] = sta[tp] - 1;
    sta[++tp] = i;
  }
  for (int i = 1; i <= n; i++)
    if (i - l[i] < r[i] - i) {
      for (int j = l[i]; j < i; j++)
        if (ma[arr[i] - arr[j]] > i && ma[arr[i] - arr[j]] <= r[i]) ans++;
    } else
      for (int j = i + 1; j <= r[i]; j++)
        if (ma[arr[i] - arr[j]] >= l[i] && ma[arr[i] - arr[j]] < i) ans++;
  printf("%d\n", ans);
  return 0;
}
