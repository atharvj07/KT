#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int t;
  cin >> t;
  for (int i = (0); (1) > 0 ? i < (t) : i >= (t); i += (1)) {
    int n, x;
    cin >> n >> x;
    vector<int> num(n);
    for (auto& j : num) cin >> j;
    sort((num).begin(), (num).end());
    int cont = 1;
    int ans = 0;
    for (int j = (n - 1); (-1) > 0 ? j < (0) : j >= (0); j += (-1)) {
      if (num[j] * cont >= x)
        ++ans, cont = 1;
      else
        ++cont;
    }
    cout << ans << "\n";
  }
  return 0;
}
