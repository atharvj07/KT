#include <bits/stdc++.h>

#define REP(i,n) for(int i=0;i<(int)(n);i++)

using namespace std;

typedef int Weight;
Weight INF = 1000000000;
struct Edge{
  int src, dest; Weight weight;
  bool operator < (const Edge &rhs) const {return weight > rhs.weight;}
};

typedef vector<Edge> Edges;
typedef vector<Edges> Graph;
typedef vector<Weight> Array;
typedef vector<Array> Matrix;

void add_edge(Graph &g, int src, int dest, Weight weight) {
  g[src].push_back((Edge){src, dest, weight});
}

// Dijkstra (Verified: AOJ2005)
void dijkstra(Graph &g, Matrix &d, int s) {
  REP(i,g.size()) d[i].assign(10, INF);
  d[s][0] = 0;
  typedef pair<Weight,int> P;
  priority_queue<P, vector<P>, greater<P> > que;
  que.push(P(0, s));
  while (!que.empty()) {
    Weight dist = que.top().first;
    int v = que.top().second;
    que.pop();
    auto itr=find(begin(d[v]),end(d[v]),dist);
    if (itr == end(d[v])) continue;
    REP(i, g[v].size()) {
      Edge e = g[v][i];
      REP(j,10) {
        if (d[e.dest][j] > dist + e.weight) {
          d[e.dest].insert(begin(d[e.dest])+j, dist + e.weight);
          d[e.dest].pop_back();
          que.push(P(d[e.dest][j], e.dest));
          break;
        }
      }
    }
  }
}

int main() {
  while(1){
    int n,m,l,k;
    cin>>n>>m>>l>>k;
    if(!n)break;
    --k;
    int z=1<<l;
    Graph g(n*z);
    vector<tuple<int,int,int>> es;
    REP(i,m){
      int u,v,c;
      cin>>u>>v>>c;
      --u;--v;
      es.emplace_back(u,v,c);
    }
    int s,t;
    cin>>s>>t;
    --s;--t;
    vector<int> db(l);
    REP(i,l)cin>>db[i];
    REP(i,l)--db[i];
    REP(i,m) {
      int u,v,c;
      tie(u,v,c)=es[i];
      REP(j,z){
        bitset<7> bs(j);
        bool tku = false;
        bool tkv = false;
        REP(p,l){
          if (!bs[p]) {
            if (db[p]==u) {
              add_edge(g,v*z+j,u*z+j+(1<<p),c);
              tku = true;
            }
            if(db[p]==v) {
              add_edge(g,u*z+j,v*z+j+(1<<p),c);
              tkv = true;
            }
          }
        }
        if(!tku) add_edge(g,v*z+j,u*z+j,c);
        if(!tkv) add_edge(g,u*z+j,v*z+j,c);
      }
    }
    Matrix dist(n*z);
    REP(i,dist.size()) dist[i]=Array(10);
    auto itr = find(begin(db),end(db),s);
    if (itr == end(db))
      dijkstra(g,dist,s*z);
    else
      dijkstra(g,dist,s*z+(1<<(itr-begin(db))));
    int ti=t*z+z-1;
    if(dist[ti][k] < INF)
      cout<<dist[ti][k]<<endl;
    else
      cout<<"NA"<<endl;
  }
  return 0;
}
