a = input() + '1'
n = len(a) - 1
s = [0] * (n + 1)
for q in range(n - 1, -1, -1):
	s[q] = s[q + 1]
	if a[q] == '1':
		s[q] += 1
d = [0] * (n + 1)
for q in range(n - 1, -1, -1):
	d[q] = max(s[q], d[q + 1])
	if a[q] == '0':
		d[q] = max(d[q], d[q + 1] + 1)
t = []
for q in range(n):
	if a[q] == '0':
		t.append('0')
	elif d[q] == d[q + 1] + 1:
		t.append('0')
	else:
		t.append('1')
print(''.join(t))
