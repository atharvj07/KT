

import java.util.Scanner;

public class Chips {

	public static void main(String[] args) {
		Scanner scan=new Scanner(System.in);
		int n=scan.nextInt();
		int m=scan.nextInt();
		int temp=0;
		while(m>0) {
			for(int i=1;i<=n;i++) {
				if(m>=i) {
					m=m-i;
				}
				else {
					System.out.println(m);temp=1;
					break;
				}
			}
			if(temp==1)break;
		}
		if(temp==0)System.out.println(0);
	}

}
