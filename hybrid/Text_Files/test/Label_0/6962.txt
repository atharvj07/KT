#include <bits/stdc++.h>
using namespace std;
vector<string> const elems = {
    "H",  "He", "Li", "Be", "B",  "C",  "N",  "O",  "F",  "Ne", "Na", "Mg",
    "Al", "Si", "P",  "S",  "Cl", "Ar", "K",  "Ca", "Sc", "Ti", "V",  "Cr",
    "Mn", "Fe", "Co", "Ni", "Cu", "Zn", "Ga", "Ge", "As", "Se", "Br", "Kr",
    "Rb", "Sr", "Y",  "Zr", "Nb", "Mo", "Tc", "Ru", "Rh", "Pd", "Ag", "Cd",
    "In", "Sn", "Sb", "Te", "I",  "Xe", "Cs", "Ba", "La", "Ce", "Pr", "Nd",
    "Pm", "Sm", "Eu", "Gd", "Tb", "Dy", "Ho", "Er", "Tm", "Yb", "Lu", "Hf",
    "Ta", "W",  "Re", "Os", "Ir", "Pt", "Au", "Hg", "Tl", "Pb", "Bi", "Po",
    "At", "Rn", "Fr", "Ra", "Ac", "Th", "Pa", "U",  "Np", "Pu", "Am", "Cm",
    "Bk", "Cf", "Es", "Fm", "Md", "No", "Lr"};
int elem_to_num(string const& elem) {
  return find(elems.begin(), elems.end(), elem) - elems.begin() + 1;
}
string num_to_elem(int num) { return elems[num - 1]; }
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n), b(k);
  for (int& x : a) {
    string s;
    cin >> s;
    x = elem_to_num(s);
  }
  for (int& x : b) {
    string s;
    cin >> s;
    x = elem_to_num(s);
  }
  vector<int> sum(1 << n);
  for (int i = 1; i < 1 << n; ++i) {
    int j = 0;
    while (((i >> j) & 1) == 0) ++j;
    sum[i] = sum[i ^ (1 << j)] + a[j];
  }
  vector<int> dp(1 << n);
  vector<int> from(1 << n);
  for (int i = 1; i < 1 << n; ++i) {
    for (int j = 0; j < n; ++j)
      if ((i >> j) & 1) {
        if (dp[i ^ (1 << j)] > dp[i]) {
          dp[i] = dp[i ^ (1 << j)];
          from[i] = i ^ (1 << j);
        }
      }
    if (dp[i] == k) continue;
    int num = b[dp[i]];
    for (int j = i;; j = (j - 1) & i) {
      if (dp[j] == dp[i] && sum[i ^ j] == num) {
        ++dp[i];
        from[i] = j;
        break;
      }
      if (!j) break;
    }
  }
  int i = max_element(dp.begin(), dp.end()) - dp.begin();
  if (dp[i] != k) {
    cout << "NO\n";
  } else {
    cout << "YES\n";
    while (i) {
      if (dp[i] > dp[from[i]]) {
        int cur = i ^ from[i];
        bool was = false;
        for (int x = 0; x < n; ++x) {
          if ((cur >> x) & 1) {
            if (was) cout << '+';
            was = true;
            cout << num_to_elem(a[x]);
          }
        }
        cout << "->" << num_to_elem(b[dp[i] - 1]) << '\n';
      }
      i = from[i];
    }
  }
}
