#include<cstdio>
#include<cstdlib>
#include<cstring>
#include<cmath>
#include<iostream>
#include<sstream>
#include<utility>

#include<vector>
#include<queue>
#include<algorithm>
#include<set>
using namespace std;
typedef long long ll;
typedef pair<int,int>P;

int N,x[21];
set<vector<int> >res;

int vm(int *d)
{
	int mx;
	for(mx=400;mx;mx--)
		if(d[mx])break;
	return mx;
}

void dfs(int j,int *d)
{
	if(j==N)
	{
		int r[20];
		memcpy(r,x,sizeof(r));
		sort(r,r+N);
		vector<int>v;
		for(int i=0;i<N-1;i++)v.push_back(r[i+1]-r[i]);
		res.insert(v);
		return;
	}

	int tm[401];
	int l=vm(d);
	for(int i=0;i<2;i++)
	{
		int a;
		memcpy(tm,d,sizeof(tm));
		x[j]=(i?l:(x[1]-l));
		for(a=0;a<j;a++)
		{
			int dd=abs(x[j]-x[a]);
			if(!tm[dd])
				break;
			tm[dd]--;
		}
		if(a==j)
			dfs(j+1,tm);
	}
}

int main()
{
	for(;scanf("%d",&N),N;)
	{
		int d[401]={0};
		for(int i=0;i<N*(N-1)/2;i++)
		{
			int t;
			scanf("%d",&t);
			d[t]++;
		}
		res.clear();
		int l=vm(d);
		x[0]=0;x[1]=l;d[l]--;
		dfs(2,d);
		for(set<vector<int> >::iterator it=res.begin();it!=res.end();it++)
			for(int i=0;i<N-1;i++)
				printf("%d%s",(*it)[i],i==N-2?"\n":" ");
		puts("-----");
	}
}
