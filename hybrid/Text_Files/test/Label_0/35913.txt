
import java.io.*;
import java.util.*;

public class Solution {

    static int gds(int a, int b) {
        if (a == b)
            return a;
        if (a > b) {
            return gds(a-b, b);
        } else return gds(a,b-a);
    }
    
    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        //PrintWriter out = new PrintWriter(new BufferedWriter(new FileWriter("output.txt")));

        int a = sc.nextInt();

        int s = 0;

        for (int i = 2; i < a; i++) {
            int k = a;

            while (k > 0) {
                s += k % i;
                k = k / i;
            }
        }
        
        int gds = gds(s, a-2);
        s = s / gds;
        int c = (a-2) / gds;
        System.out.println(s+"/"+c);
        
        sc.close();
    }
}
