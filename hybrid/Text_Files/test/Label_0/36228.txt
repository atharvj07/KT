import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.StringTokenizer;

public class kok {
	public static void main(String[] args) throws IOException {
		FastReader sc = new FastReader(System.in); 
		
		int len = sc.nextInt();
		
		int[] wanted = new int[len];
		
		for(int i = 0; i < len; i++)
			wanted[i] = sc.nextInt();
		
		int[] given = new int[len];
		
		for(int i = 0; i < len; i++)
			given[i] = sc.nextInt();
		
		ArrayList<Pair> result = new ArrayList<>();
		
		for(int i = 0; i < len; i++){
			if(given[i] != wanted[i]){
				int best = i + 1;
				while(wanted[i] != given[best]) best++;
				while(best > i){
					result.add(new Pair(best - 1, best));
					int temp = given[best - 1];
					given[best - 1] = given[best];
					given[best] = temp;
					best--;
				}
			}
		}
		
		System.out.println(result.size());
		for(Pair i : result)
			System.out.println((i.x + 1) + " " + (i.y + 1));
		
		//for(int i = 0; i < len; i++)
		//	System.out.print(given[i] + " ");
		
	}
}

class Pair { 
	public int x;
	public int y;
	
	public Pair(int x, int y){
		this.x = x;
		this.y = y;
	}
}

class FastReader {
	public BufferedReader reader;
    public StringTokenizer tokenizer;

    public FastReader(InputStream stream) {
        reader = new BufferedReader(new InputStreamReader(stream), 32768);
        tokenizer = null;
    }

    public String next() {
        while (tokenizer == null || !tokenizer.hasMoreTokens()) {
            try {
                tokenizer = new StringTokenizer(reader.readLine());
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
        }
        return tokenizer.nextToken();
    }
    
    public int nextInt() throws IOException{
        return Integer.parseInt(next());
    }
     public long nextL() throws IOException{
        return Long.parseLong(next());
    }
}
