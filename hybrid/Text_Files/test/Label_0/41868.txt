read = lambda : map(int, input().split())
(n, m, k) = read()
mat = [[0] * m for i in range(n)]
row = [(0, -1)] * n
col = [(0, -1)] * m
for i in range(k):
	(a, b, c) = read()
	if a == 1:
		row[b - 1] = (c, i)
	else:
		col[b - 1] = (c, i)
for i in range(n):
	for j in range(m):
		if row[i][1] > col[j][1]:
			mat[i][j] = row[i][0]
		else:
			mat[i][j] = col[j][0]
for i in mat:
	print(*i)
