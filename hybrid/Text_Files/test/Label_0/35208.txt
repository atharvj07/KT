import java.lang.*;
import java.util.*;

public class C{
	public static void main(String args[]){
		Scanner sc = new Scanner(System.in);
		double total = 0;
		int memSum= 0;
		int n = sc.nextInt();
		int[] seq = new int[n+1];
		int[] mem = new int[n+1];
		int len = 1 ;
		for(int i=0;i<n;i++){
			int op = sc.nextInt() ;
			if(op==1){
				int ai= sc.nextInt();
				int xi = sc.nextInt();
				mem[ai-1]+=xi ;
				total+=(ai*xi);
				System.out.println(total/len);
			}
			else if(op==2){
				int k = sc.nextInt();
				seq[len] = k;
				len++;
				total+=k;
				System.out.println(total/len);
			}
			else{
				len--;
				total-=(seq[len]+mem[len]);
				mem[len-1] += mem[len] ;
				mem[len]=0;
				seq[len]=0;
				System.out.println(total/len);
			}
		}
		
	}
}
