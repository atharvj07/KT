#include <bits/stdc++.h>
using namespace std;

long long c[51][51], ans;
int n, a, b;
vector<double> v;
double avg;
int main() {
	scanf("%d%d%d",&n,&a,&b);
	c[0][0] = 1;
	for(int i = 1; i <= n; i++) {
		c[i][0] = c[i][i] = 1;
		for(int j = 1; j < i; j++) c[i][j] = c[i-1][j-1]+c[i-1][j];
	}
	v.resize(n);
	for(int i = 0; i < n; i++) scanf("%lf",&v[i]);
	sort(v.begin(), v.end(), greater<double>());
	for(int i = 0; i < a; i++) avg += v[i];
	avg /= a;
	int x = 0, y = 0;
	for(int i = 0; i < n; i++) if(v[i] == v[a-1]) {
		x++;
		if(i < a) y++;
	}
	if(v[0] != v[a-1]) ans = c[x][y];
	else {
		for(int k = a; k <= min(x, b); k++) ans += c[x][k];
	}
	printf("%lf\n%lld\n", avg, ans);
}
