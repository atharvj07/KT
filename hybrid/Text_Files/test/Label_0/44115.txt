n = int(input())
nums = input()
nums = [int(i) for i in nums.split()]
ans = 0
for i in range(0, len(nums), 2):
	target = nums[i]
	other = i + 1
	for j in range(i + 1, len(nums)):
		if nums[j] == target:
			other = j
	swaps = other - i - 1
	ans += swaps
	for j in range(other, i + 1, -1):
		(nums[j], nums[j - 1]) = (nums[j - 1], nums[j])
print(ans)
