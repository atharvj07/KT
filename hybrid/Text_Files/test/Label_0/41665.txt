n = int(input())
p = [-1] * n
c = [-1] * n
d = [False] * n
for i in range(n):
	(p[i], c[i]) = list(map(int, input().split()))
for i in range(n):
	if p[i] == -1:
		continue
	if c[i] == 0:
		d[i] = True
		d[p[i] - 1] = True
flag = False
o = []
for i in range(n):
	if p[i] == -1:
		continue
	if not d[i]:
		flag = True
		o.append(i + 1)
if not flag:
	print(-1)
else:
	print(' '.join((str(x) for x in o)))
