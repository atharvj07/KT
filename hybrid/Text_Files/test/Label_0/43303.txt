import sys

def problem1(n, m, a, b, time_list):
	if b < a:
		a = n - a + 1
		b = n - b + 1
	max_to_recall = b - a - 1
	time_list.sort()
	max_time_for_dazhao_to_recall = b - 2
	time_list = time_list[:max_to_recall]
	answer = 0
	for i in range(max_to_recall):
		if max_time_for_dazhao_to_recall >= time_list[-1]:
			answer += 1
			max_time_for_dazhao_to_recall -= 1
		time_list.pop()
		if time_list == []:
			break
	return answer
T = int(sys.stdin.readline().strip())
while T > 0:
	T -= 1
	(n, m, a, b) = sys.stdin.readline().strip().split(' ')
	l = sys.stdin.readline().strip()
	l = l.split(' ')
	l = sorted(list(map(lambda x: int(x), l)), reverse=False)
	print(problem1(int(n), int(m), int(a), int(b), l))
