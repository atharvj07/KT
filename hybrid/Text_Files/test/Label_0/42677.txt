(n, m, k) = map(int, input().split())
rowclr = [[-1, -1]] * n
colclr = [[-1, -1]] * m
oprt = {}
mat = [[0 for j in range(m)] for i in range(n)]
for i in range(k):
	(a, b, c) = map(int, input().split())
	oprt[b - 1, a] = [i, c]
d = {k: oprt[k] for k in sorted(oprt, key=oprt.get)}
for (k, v) in d.items():
	if k[1] == 1:
		row = k[0]
		add = v[1]
		for j in range(m):
			mat[row][j] = add
	else:
		col = k[0]
		add = v[1]
		for i in range(n):
			mat[i][col] = add
for i in range(n):
	print(*mat[i])
