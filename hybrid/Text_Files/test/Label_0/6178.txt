#include<cstdio>
#include<functional>
#include<algorithm>
using namespace std;

int main(void)
{
	int y1,m1,d1,y2,m2,d2,x;
	scanf("%d %d %d",&y1,&m1,&d1);
	scanf("%d %d %d",&y2,&m2,&d2);
	if(y1>y2 || (y1==y2 && m1>m2) || (y1==y2 && m1==m2 && d1>d2))	{
		swap(y1,y2);	swap(m1,m2);	swap(d1,d2);
	}
	x=y2-y1;
	if(m2>m1 || (m2==m1 && d2>d1))	x++;
	printf("%d\n",x);	
	return 0;
}
