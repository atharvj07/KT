#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  for (int j = 0; j < t; j++) {
    long long n, sum = 0, temp;
    char t;
    string s, ans;
    cin >> n;
    cin >> s;
    for (int i = 0; i < n; i++) {
      t = s[i];
      string c;
      c = c + t;
      temp = stoi(c);
      if (temp % 2 == 1) ans = ans + t;
      if (ans.length() == 2) break;
    }
    if (ans.length() == 2)
      cout << ans << endl;
    else
      cout << -1 << endl;
  }
}
