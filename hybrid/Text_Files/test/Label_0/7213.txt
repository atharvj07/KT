#include <bits/stdc++.h>
int main() {
  int n;
  std::string s;
  std::cin >> n >> s;
  std::vector<int> mx(26, 0);
  std::vector<int> dp(n, 1);
  for (int i = 0; i < n; ++i) {
    for (int c = 25; c > s[i] - 'a'; --c) dp[i] = std::max(dp[i], mx[c] + 1);
    mx[s[i] - 'a'] = std::max(mx[s[i] - 'a'], dp[i]);
  }
  std::cout << *std::max_element(mx.begin(), mx.end()) << std::endl;
  for (int i = 0; i < n; ++i) std::cout << dp[i] << " ";
  std::cout << std::endl;
}
