import java.util.*;

import javax.print.attribute.IntegerSyntax;

import java.io.*;
public class Problem {
	static int [] buyPrice;
	static int [] profit;
	static int initial;
	public static void main(String[] args)throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		PrintWriter out = new PrintWriter(new OutputStreamWriter(System.out));
		StringTokenizer s = new StringTokenizer(br.readLine());
		int a = Integer.parseInt(s.nextToken());
		int b = Integer.parseInt(s.nextToken());
		int lowestBase = Math.max(highDigit(a), highDigit(b)) + 1;
		int newTotal = nB(a, lowestBase) + nB(b, lowestBase);
//		System.out.println(newTotal);
		out.println(highPower(newTotal, lowestBase));
		out.close();
	}
	public static int highDigit(int n){
		int d = 0;
		while(n > 0){
			d = Math.max(d, n % 10);
			n /= 10;
		}
		return d;
	}
	public static int highPower(int n, int b){
		int exponent = 0;
		while(Math.pow(b, exponent) <= n){
			exponent ++;
		}
		return exponent;
	}
	public static int nB(int a, int base){
		String n = Integer.toString(a);
		int total = 0;
		for(int i = 0; i < n.length(); i++){
			total += Integer.parseInt(n.substring(i, i + 1)) * Math.pow(base, n.length() - 1 - i);
		}
		return total;
	}
}
