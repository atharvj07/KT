import java.util.*;
import java.math.*;
public class Main 
{
	public static void main(String args[])
	{
		BigInteger f[]=new BigInteger[605];
		f[0]=BigInteger.ONE;
		for(int i=1;i<=600;i++) f[i]=f[i-1].multiply(BigInteger.valueOf(12));
		Scanner in=new Scanner(System.in);
		BigInteger n=in.nextBigInteger();
		for(int i=0;i<=300;i++)
			for(int j=i;j<=300;j++)
				if (n.equals(f[i].add(f[j])))
				{
					System.out.printf("YES\n1\n%d\n%d\n", i+j+1,(i+j)/2);
					for(int k=(i+j)/2;k>=0;k--) if (k!=i) System.out.println(f[k].add(f[i+j-k]));
					return;
				}
		System.out.println("NO");
	}
}
