cnt = int(input())
for _ in range(cnt):
	(leng, s1, s2) = (int(input()), input(), input())
	(s, i) = (0, 0)
	while i < leng:
		if int(s1[i]) + int(s2[i]) == 1:
			s += 2
		elif int(s1[i]) + int(s2[i]) == 2:
			for j in range(i + 1, leng):
				i += 1
				if int(s1[i]) + int(s2[i]) <= 1:
					s += 2
					break
		elif i < leng - 1 and int(s1[i + 1]) + int(s2[i + 1]) == 2:
			s += 2
			i += 1
		else:
			s += 1
		i += 1
	print(s)
