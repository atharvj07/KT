from sys import stdin
from collections import deque

def main():
	(n, m) = [int(x) for x in stdin.readline().split()]
	a = [int(x) for x in stdin.readline().split()]
	q = deque()
	for i in range(0, n):
		q.append((i, 0))
	ans = -1
	while q:
		student = q.popleft()
		if student[1] + m >= a[student[0]]:
			ans = student[0]
		else:
			q.append((student[0], student[1] + m))
	print(ans + 1)
main()
