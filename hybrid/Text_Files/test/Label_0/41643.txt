def process_input():
	(n, m, k) = map(int, input().split())
	rows = {}
	cols = {}
	for p in range(1, k + 1):
		(o, i, a) = input().split()
		if o == '1':
			rows[int(i)] = (a, p)
		else:
			cols[int(i)] = (a, p)
	return (n, m, rows, cols)

def generate_output(n, m, rows, cols):
	for r in range(1, n + 1):
		(ra, rp) = rows[r] if r in rows else ('0', 0)
		row = []
		for c in range(1, m + 1):
			(ca, cp) = cols[c] if c in cols else ('0', 0)
			row.append(ca if cp > rp else ra)
		print(' '.join(row))
(n, m, rows, cols) = process_input()
generate_output(n, m, rows, cols)
