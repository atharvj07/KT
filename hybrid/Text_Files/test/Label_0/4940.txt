#include <bits/stdc++.h>
using namespace std;
vector<int> ans;
int mas[200005];
int main() {
  long long n, finish, s, d, i;
  cin >> n >> finish >> s >> d;
  for (i = 0; i < n; i++) cin >> mas[i];
  sort(mas, mas + n);
  i = 0;
  int start = 0;
  while (i < n) {
    if (mas[i] - start - 1 < s) {
      cout << "IMPOSSIBLE";
      return 0;
    }
    ans.push_back(mas[i] - start - 1);
    int j = i;
    while (j < n - 1 && mas[j + 1] - mas[j] - 2 < s) j++;
    if (mas[j] - mas[i] + 2 > d) {
      cout << "IMPOSSIBLE";
      return 0;
    }
    ans.push_back(mas[j] - mas[i] + 2);
    start = mas[j] + 1;
    i = j + 1;
  }
  if (finish - start) ans.push_back(finish - start);
  for (i = 0; i < ans.size(); i++) {
    if (i % 2 == 0)
      cout << "RUN ";
    else
      cout << "JUMP ";
    cout << ans[i] << endl;
  }
  return 0;
}
