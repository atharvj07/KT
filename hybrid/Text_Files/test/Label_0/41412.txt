def judge(s):
	for i in s:
		if i == '7':
			return True
	return False

def main():
	x = int(input())
	l = input().split()
	h = int(l[0])
	hh = h
	m = int(l[1])
	mm = m
	while True:
		if judge(str(m)) or judge(str(h)):
			break
		m -= x
		if m < 0:
			m += 60
			h -= 1
	print(((hh - h) * 60 + (mm - m)) // x)
main()
