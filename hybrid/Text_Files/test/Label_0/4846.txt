typedef double Real;

typedef complex<Real> P;

Real cross(P a,P b){
  return imag(conj(a)*b);
}

Real sur(P a,P b,P c){
  b-=a; c-=a;
  return abs(cross(b,c))/2.0;
}

P ppp[252521];

// count triangle s.t. surface < s
ll calc(int n,Real s){
  Real u = 2.0*M_PI/(Real)n;
  ll ans = 0;
  int a = 0;
  P x = ppp[a];
  ll bai = n;
  if(n%3==0)bai/=3;
  FOR(b,1,n-1){
    P y = ppp[b];
    int low = b+1, high = n-1;
    int cnt = (high-low+1);
    // seisannkakkei guard
    if(n%3==0 && b==n/3){
      cnt--;
    }
    if(cnt%2==1){
      int cent = low + cnt/2;
      P z = ppp[cent];
      if(sur(x,y,z) < s){
        ans += cnt;
        continue;
      }
    }
    cnt /= 2;
    high = low + cnt - 1 + 1;
    low--;
    while(low+1<high){
      int mid = (low+high)/2;
      P z = ppp[mid];
      if(sur(x,y,z) < s){
        low = mid;
      }else{
        high = mid;
      }
    }
    ans += 2ll*(low-b);
  }
  // DEBUG(ans);
  ans /= 3;
  ans *= n;
  return ans;
}
