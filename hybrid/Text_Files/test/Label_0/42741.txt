t = int(input())
for _ in range(t):
	(w, h) = map(int, input().split())
	(x1, y1, x2, y2) = map(int, input().split())
	(w1, h1) = map(int, input().split())
	a = max(x1, w - x2)
	b = max(y1, h - y2)
	if w1 <= a or h1 <= b:
		print(0)
	elif w - (x2 - x1) < w1 and h - (y2 - y1) < h1:
		print(-1)
	elif w - (x2 - x1) < w1:
		print(h1 - b)
	elif h - (y2 - y1) < h1:
		print(w1 - a)
	else:
		print(min(w1 - a, h1 - b))
