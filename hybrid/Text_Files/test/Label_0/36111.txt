import java.io.*;
import java.util.*;

public class D
{
	public static void main(String [] args) throws IOException
	{
		BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
		PrintWriter writer = new PrintWriter(new OutputStreamWriter(System.out));

		StringTokenizer tokenizer = new StringTokenizer(reader.readLine());

		int n = Integer.parseInt(tokenizer.nextToken());
		if(n == 1)
		{
			writer.println(1);
			writer.flush();
			writer.close();
			return;
		}
		int k = Integer.parseInt(tokenizer.nextToken());

		tokenizer = new StringTokenizer(reader.readLine());

		int [] fools = new int[n];
		for(int i = 0 ; i < n ; i++)
			fools[i] = Integer.parseInt(tokenizer.nextToken());

		int [] max = new int[n+1];
		for(int i = n-1 ; i >= 0 ; i--)
			max[i] = Math.max(max[i+1], fools[i]);

		boolean [][] visited = new boolean[n+2][n+2];
		visited[0][1] = true;
		int [][] best = new int[n+2][n+2];
		for(int [] fill : best)
			Arrays.fill(fill, Integer.MAX_VALUE/2);
		best[0][1] = 0;

		Queue<Integer> queue = new LinkedList<Integer>();

		queue.offer(0);
		queue.offer(1);

		int res = 0;
		while(!queue.isEmpty())
		{
			int left = queue.poll();
			int right = queue.poll();
			if(best[left][right] <= k)
				res++;
			else
				break;

			if(left >= n || right >= n)
				continue;

			if(fools[left] == 0 && max[right] == 0)
			{

			}
			else if(max[right] == 0)
			{
				if(!visited[left][right+1])
				{
					queue.offer(left);
					queue.offer(right+1);
					best[left][right+1] = best[left][right] + 1;
					visited[left][right+1] = true;
				}
			}
			else if(fools[left] == 0)
			{
				if(!visited[right][right+1])
				{
					queue.offer(right);
					queue.offer(right+1);
					best[right][right+1] = best[left][right] + 1;
					visited[right][right+1] = true;
				}
			}
			else
			{
				if(!visited[right+1][right+2])
				{
					queue.offer(right+1);
					queue.offer(right+2);
					best[right+1][right+2] = best[left][right] + 1;
					visited[right+1][right+2] = true;
				}
				if(fools[left] == 100 && max[right] == 100)
				{

				}
				else if(fools[left] == 100)
				{
					if(!visited[left][right+1])
					{
						queue.offer(left);
						queue.offer(right+1);
						best[left][right+1] = best[left][right] + 1;
						visited[left][right+1] = true;
					}

				}
				else if(max[right] == 100)
				{
					if(!visited[right][right+1])
					{
						queue.offer(right);
						queue.offer(right+1);
						best[right][right+1] = best[left][right] + 1;
						visited[right][right+1] = true;
					}
				}
				else
				{
					if(!visited[left][right+1])
					{
						queue.offer(left);
						queue.offer(right+1);
						best[left][right+1] = best[left][right] + 1;
						visited[left][right+1] = true;
					}

					if(!visited[right][right+1])
					{
						queue.offer(right);
						queue.offer(right+1);
						best[right][right+1] = best[left][right] + 1;
						visited[right][right+1] = true;
					}
				}
			}
		}

		System.out.println(res);


	}
}
