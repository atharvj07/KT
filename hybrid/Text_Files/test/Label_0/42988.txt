import sys
input = sys.stdin.readline
from collections import defaultdict
for _ in range(int(input())):
	n = int(input())
	A = list(map(int, input().split()))
	mod = 10 ** 9 + 7
	dp = defaultdict(int)
	dp[0] = 1
	ans = 1
	cur = 0
	for a in A:
		extra = dp[cur]
		dp[cur] = ans
		cur -= a
		ans = (2 * ans - extra) % mod
	print(ans)
