#include<iostream>
#include<vector>
#include<iomanip>
#include<cmath>
#include<algorithm>

using namespace std;
using P = pair<double, pair<int, int> >;

double p[200][3];
double v[200][3];
double r[200];
double rv[200];
double life[200];
bool die[200];
double tdie[200];

int n;

double coll_time(int i, int j) {
    double R = r[i] + r[j];
    double RV = rv[i] + rv[j];
    double ce2 = - RV * RV;
    double ce1 = 2 * R * RV;
    double ce0 = - R * R;

    for (int k = 0; k < 3; ++k) {
        double P = p[i][k] - p[j][k];
        double V = v[i][k] - v[j][k];
        ce2 += V * V;
        ce1 += 2 * P * V;
        ce0 += P * P;
    }

    double ans;
    if (ce2 != 0) {
        double D = ce1 * ce1 - 4 * ce0 * ce2;
        if (D < 0) return -1;
        ans = (- ce1 - sqrt(D)) / (2 * ce2);
    } else {
        ans = - ce0 / ce1;
    }
    if (ans < 0) return -1;
    return ans;
}

void solve() {
    vector<P> death;
    for (int i = 0; i < n; ++i) {
        cin >> p[i][0] >> p[i][1] >> p[i][2] >> v[i][0] >> v[i][1] >> v[i][2] >> r[i] >> rv[i];
        life[i] = r[i] / rv[i];
        death.push_back(make_pair(life[i], make_pair(i, -1)));
        die[i] = false;
    }

    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            double collt = coll_time(i, j);
            if (collt < 0) continue;
            death.push_back(make_pair(collt, make_pair(i, j)));
        }
    }

    sort(death.begin(), death.end());

    int cnt = 0;
    for (P d : death) {
        double t = d.first;
        int i = d.second.first;
        int j = d.second.second;

        //cout << t << ", " << i << ", " << j << endl;
        if (j == -1 && !die[i]) {
            die[i] = true;
            tdie[i] = t;
            cnt++;
        } else if (j != -1 && !die[i] && !die[j]) {
            die[i] = true;
            die[j] = true;
            tdie[i] = t;
            tdie[j] = t;
            cnt += 2;
        }
        if (cnt >= n) break;
    }

    for (int i = 0; i < n; ++i) {
        cout << fixed << setprecision(10) << tdie[i] << endl;
    }
}

int main() {
    while (cin >> n) {
        if (n == 0) return 0;
        //cout << "=============" << endl;
        solve();
    }
}
