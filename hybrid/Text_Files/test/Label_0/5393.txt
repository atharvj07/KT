#include<bits/stdc++.h>
using namespace std;
typedef struct{
  int x,y;
}P;

int dx[4]={0,1,0,-1};
int dy[4]={-1,0,1,0};
int d[100][100];
char mp[100][100];
#define INF 100000
int main(){
  int h,w,cnt=0;
  cin>>h>>w;
  for(int i=0;i<h;i++){
    for(int j=0;j<w;j++){
      cin>>mp[i][j];
      if(mp[i][j]=='.')cnt++;
      d[i][j]=INF;
    }
  }
  queue<P>que;
  que.push((P){0,0});
  d[0][0]=0;

  while(!que.empty()){
    P p=que.front();
    que.pop();
    int x=p.x;
    int y=p.y;
    if(x==w-1&&y==h-1)break;
    for(int i=0;i<4;i++){
      int nx=x+dx[i];
      int ny=y+dy[i];
      if(0<=nx&&nx<w&&0<=ny&&ny<h&&mp[ny][nx]!='#'&&d[ny][nx]==INF){
        d[ny][nx]=d[y][x]+1;
        que.push((P){nx,ny});
      }
    }
  }
  if(d[h-1][w-1]==INF)cout<<-1<<endl;
  else cout<<cnt-1-d[h-1][w-1]<<endl;
  return(0);
}
