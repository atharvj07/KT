import java.util.Arrays;
import java.util.List;
import java.util.Scanner;
import java.util.stream.Collectors;
import java.util.stream.Stream;

public class Application {
	public static void main(String[] args) {
		List<Integer> solution = solution();
		System.out.println(solution.get(0) + " " + solution.get(1));
	}

	private static List<Integer> solution() {
		Scanner scanner = new Scanner(System.in);
		List<Integer> collect = Stream.of(scanner.nextLine().split(" ")).map(Integer::parseInt).collect(Collectors.toList());
		int n =  collect.get(0);
		int k =  collect.get(1);
		String[] split = scanner.nextLine().split(" ");
		List<Integer> a = Stream.of(split).map(Integer::parseInt).collect(Collectors.toList());
		
		int count = 0;
		int[] cnt = new int[100001];
		for (int r = 0; r < n; r++ ) {
			cnt[a.get(r)] ++;
			if (cnt[a.get(r)] == 1) {
				count++;
			}
			int l = 0;
			while (count == k) {
				cnt[a.get(l)]--;
				if (cnt[a.get(l)] == 0) {
					return Arrays.asList(l + 1, r + 1);
				}
				l++;
			}
		}
		return Arrays.asList(-1, -1);
		
	}
}
