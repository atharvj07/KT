import java.io.*;
import java.util.*;
import java.math.*;

public class Main
{
	static InputStream inputstream = System.in ;
	static PrintStream outputstream = System.out ;
	static InputReader in = new InputReader ( inputstream ) ;
	static PrintWriter out = new PrintWriter ( outputstream ) ;
	
	public static void main ( String args[] )
	{
		Main task = new Main() ;
		task.main() ;
		out.close() ;
	}
	int[] vis = new int[222222] ;
	int[] a = new int[222222] ;
	void main ()
	{
		int n , i , j , k ;
		n = in.nextInt() ;
		if ( n % 2 == 1 )
		{
			out.println("-1") ;
			return ;
		}
		for ( i = 0 ; i <= n ; i ++ ) vis[i] = 0 ;
		a[n] = n / 2 ;
		vis[a[n]] = 1 ;
		for ( i = n - 1 ; i >= 1 ; i -- )
		{
			if ( a[i+1] % 2 == 1 )
			{
				if ( vis[(a[i+1]+n-1)/2] == 1 && vis[(a[i+1]-1)/2] == 1 )
				{
					out.println( "-1" ) ;
					return ;
				}
				else if ( vis[(a[i+1]+n-1)/2] == 1 ) 
					a[i] = (a[i+1]-1)/2 ;
				else a[i] = ( a[i+1] + n - 1 ) / 2 ;
			}
			else
			{
				if ( vis[(a[i+1]+n)/2] == 1 && vis[a[i+1]/2] == 1 )
				{
					out.println( "-1" ) ;
					return ;
				}
				else if ( vis[(a[i+1]+n)/2] == 1 ) a[i] = a[i+1]/2 ;
				else a[i] = ( a[i+1] + n ) / 2 ;
			}
			vis[a[i]] = 1 ;
		}
		for ( i = 1 ; i <= n ; i ++ ) out.print( a[i] + " " ) ;
		out.println( "0" ) ;
	}
}
class InputReader {
	BufferedReader reader;
	StringTokenizer tokenizer;

	InputReader(InputStream stream) {
		reader = new BufferedReader(new InputStreamReader(stream));
		tokenizer = null;
	}

	String next() {
		if (!hasNext())
			throw new RuntimeException();
		return tokenizer.nextToken();
	}

	boolean hasNext() {
		while (tokenizer == null || !tokenizer.hasMoreTokens())
			try {
				tokenizer = new StringTokenizer(reader.readLine());
			} catch (Exception e) {
				return false;
			}
		return true;
	}

	long nextLong() {
		return Long.parseLong(next());
	}

	int nextInt() {
		return Integer.parseInt(next());
	}
}
