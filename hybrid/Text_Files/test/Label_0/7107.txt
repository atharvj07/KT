#include <iostream>
#include <vector>

using namespace std;

void rebuild(int i,int K,vector <int> &m,vector <int> &t){
  i /=2;
  while(i > 0){
    int l = i*2;// left  child
    int r = l+1;// right child
    t[i] = t[l] * t[r];
    m[i] = ( m[l] + t[l]*m[r] ) % K;
    i /= 2;
  }
}

int main(){
  int K,N,Q;
  cin >> K >> N >> Q;

  // segment tree
  //        1
  //    2       3
  //  4   5   6   7

  int nleaf = 1;
  while(nleaf < N){
    nleaf <<= 1;
  }

  vector <int> m(nleaf*2);// segment tree of move
  vector <int> t(nleaf*2);// segment tree of turn (-1:turn 1:stay )

  for(int i = 0; i < nleaf*2; i++){// make tree
    t[i] = 1;
    m[i] = 0;
  }

  for(int i = 0; i < N; i++){// store leaves
    int j = nleaf +i;// nleaf:left edge leaves
    cin >> m[j];
    if(m[j] == 0) t[j] = -1;
  }

  for(int i = nleaf-1; i > 0; i--){// build tree
    int l = i*2;// left  child
    int r = l+1;// right child
    t[i] = t[l] * t[r];
    m[i] = ( m[l] + t[l]*m[r] ) % K;
  }

  for(int i = 0; i < Q; i++){
    int L,R;
    cin >> L >> R;
    L = L + nleaf - 1;;
    R = R + nleaf - 1;
    swap( m[L], m[R] );
    swap( t[L], t[R] );
    rebuild(L,K,m,t);
    rebuild(R,K,m,t);
    cout << t[1] * ((K-m[1])%K+1) << endl;
  }
  
  return 0;
}
