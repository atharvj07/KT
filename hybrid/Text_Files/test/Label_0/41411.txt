def run_case():
	(a, b, x, y, n) = map(int, input().split())
	if a - x + (b - y) >= n:
		t = []
		temp = a - x
		if temp >= n:
			t.append((a - n) * b)
		else:
			t.append(x * (b - (n - temp)))
		temp = b - y
		if temp >= n:
			t.append((b - n) * a)
		else:
			t.append(y * (a - (n - temp)))
		print(min(t))
	else:
		print(x * y)
flag = 1
if flag:
	t = int(input())
	for i in range(t):
		run_case()
else:
	run_case()
