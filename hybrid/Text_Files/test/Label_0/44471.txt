(n, k, a, b) = map(int, input().split())
s = 'GB' * min(a, b)
a -= len(s) // 2
b -= len(s) // 2
if a != 0:
	s += 'G'
	a -= 1
elif b != 0:
	s = 'B' + s
	b -= 1
cur = 0
m = 1
while cur != len(s):
	if m != k:
		if s[cur] == 'G' and a != 0:
			s = s[:cur + 1] + 'G' + s[cur + 1:]
			a -= 1
			m += 1
		elif s[cur] == 'B' and b != 0:
			s = s[:cur + 1] + 'B' + s[cur + 1:]
			b -= 1
			m += 1
	else:
		m = 1
	cur += 1
if a == 0 and b == 0:
	print(s)
else:
	print('NO')
