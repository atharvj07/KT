template <class T>
inline void ckmin(T& a, T b) {
  if (b < a) a = b;
}
template <class T>
inline void ckmax(T& a, T b) {
  if (b > a) a = b;
}
template <class T>
inline T sqr(T x) {
  return x * x;
}
using VI = vector<int>;
int query(int x, int y) {
  printf("? %d %d\n", x + 1, y + 1);
  fflush(stdout);
  int r;
  scanf("%d", &r);
  return r;
}
namespace scc {
VI index;
VI low;
VI in_stack;
stack<int> s;
int counter;
vector<VI> components;
void dfs(const vector<VI>& g, int v) {
  index[v] = low[v] = counter++;
  s.push(v);
  in_stack[v] = 1;
  for (int w : g[v])
    if (w != v)
      if (index[w] < 0) {
        dfs(g, w);
        low[v] = min(low[v], low[w]);
      } else if (in_stack[w])
        low[v] = min(low[v], index[w]);
  if (index[v] == low[v]) {
    VI component;
    int w;
    do {
      w = s.top();
      s.pop();
      in_stack[w] = 0;
      component.push_back(w);
    } while (w != v);
    components.emplace_back(std::move(component));
  }
}
vector<VI> tarjan(const vector<VI>& g) {
  int n = ((int)g.size());
  index = VI(n, -1);
  low = VI(n);
  in_stack = VI(n, 0);
  s = stack<int>();
  components.clear();
  counter = 0;
  for (int i = 0; i < n; i++)
    if (index[i] < 0) dfs(g, i);
  return components;
}
}  // namespace scc
int main() {
  int n, m;
  cin >> n >> m;
  vector<VI> g(n);
  for (int i = 0; i < (m); ++i) {
    int a, b;
    cin >> a >> b;
    --a;
    --b;
    g[a].push_back(b);
  }
  vector<VI> sc = scc::tarjan(g);
  VI idx(n);
  m = ((int)sc.size());
  vector<VI> g2(m);
  for (int i = 0; i < (m); ++i)
    for (int x : sc[i]) idx[x] = i;
  for (int i = 0; i < (n); ++i)
    for (int x : g[i])
      if (idx[i] != idx[x]) g2[idx[i]].push_back(idx[x]);
  VI in_deg(m);
  for (int i = 0; i < (m); ++i)
    for (int x : g2[i]) ++in_deg[x];
  VI que;
  for (int i = 0; i < (m); ++i)
    if (in_deg[i] == 0) que.push_back(i);
  int key = que[0];
  for (int cl = 0; cl < (((int)que.size())); ++cl) {
    int other = que[cl];
    if (key == other) continue;
    while (((int)sc[key].size()) > 0 && ((int)sc[other].size()) > 0)
      if (query(sc[key].back(), sc[other].back()))
        sc[other].pop_back();
      else
        sc[key].pop_back();
    if (((int)sc[key].size()) == 0) swap(key, other);
    for (int x : g2[other])
      if ((--in_deg[x]) == 0) que.push_back(x);
  }
  printf("! %d\n", sc[key].back() + 1);
  fflush(stdout);
  return 0;
}
