import java.util.*;

public class B {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int L1 = in.nextInt();
        int L2 = in.nextInt();
        char[] A = in.next().toCharArray();
        char[] B = in.next().toCharArray();
        
        int[][] S = new int[B.length][];
        int pos=0;
        int cnt=0;
        for(int i=0;i<L1;i++){
            if(S[pos]!=null){
                int diffAmt  = cnt-S[pos][0]; 
                int diffTime = i-S[pos][1];
                int step = (L1-(i+1))/diffTime;
                
                i+= step*diffTime;
                cnt+=step*diffAmt;
            }
            
            S[pos] = new int[]{cnt,i};
            for(int j=0;j<A.length;j++){
                if(A[j]==B[pos]){
                    pos++;
                    if(pos==B.length){
                        cnt++;
                        pos=0;
                    }
                }
                
            }
        }
        System.out.println(cnt/L2);
    }
}
