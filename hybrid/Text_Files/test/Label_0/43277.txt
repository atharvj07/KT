(n, m) = map(int, input().split())
i = 1
startingPoint = n * 2 + 1
lst = []
if n < m:
	for j in range(0, m):
		if startingPoint <= m:
			lst.append(startingPoint)
		lst.append(i)
		startingPoint = startingPoint + 1
		i = i + 1
	if lst[m - 1] > m:
		lst[m - 1] = lst[m]
		print(*lst[:m], sep=' ')
	else:
		print(*lst[:m], sep=' ')
else:
	for item in range(1, n + 1):
		lst.append(item)
	print(*lst[:m], sep=' ')
