import math

def fn(a):
	res = (math.sqrt(1 + 8 * a) - 1) / 2
	return res

def rev(n):
	return n * (n + 1) / 2

def change(s, i, j):
	return s[:i] + 'b' + s[i + 1:j] + 'b' + s[j + 1:]
t = int(input())
for _ in range(t):
	(n, k) = map(int, input().split())
	perm = math.ceil(fn(k))
	first_ind = n - 1 - perm
	temp = int(k - rev(perm))
	sec_ind = first_ind + (1 - temp)
	res = 'a' * n
	print(change(res, first_ind, sec_ind))
