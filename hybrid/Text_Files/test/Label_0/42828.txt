import math

def completar_lista(lista, n):
	X = []
	s = 0
	cadena = ''
	for x in lista:
		parar = True
		for div1 in range(int(math.sqrt(x)), 0, -1):
			if x % div1 == 0:
				div2 = x // div1
				if (div1 + div2) % 2 == 0:
					s_1 = ((div2 - div1) // 2) ** 2
					if s_1 > s:
						temp = s_1 - s
						X.append(temp)
						X.append(x)
						s = s + temp + x
						parar = False
						break
		if parar:
			return None
	return X
n = int(input())
lista = map(int, input().split())
lista_entera = completar_lista(lista, n // 2)
if lista_entera is None:
	print('No')
else:
	print('Yes')
	print(*lista_entera)
