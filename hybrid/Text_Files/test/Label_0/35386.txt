import java.io.*;
import java.util.*;

public class test{
      public static void main(String aa[]){
            Scanner sc=new Scanner(System.in);
            int t=sc.nextInt();
            String cmp="";
            int result=0,count=0;
            String arr[]=new String[t];
            for(int i=0;i<t;i++){
                  arr[i]=sc.next();
            }
            for(int i=0;i<t;i++){
		  count=0;
                  cmp=arr[i];
                  for(int j=0;j<t;j++){
                        if(cmp.equals(arr[j])){
                              count++;
                        }
                        
                  }
                  if(result<count){result=count;}
                  
            }
            
            System.out.println(result);
      }
      
      
      
      
}
