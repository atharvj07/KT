#include <bits/stdc++.h>
#define int long long
#define r(i,n) for(int i=0;i<n;i++)
using namespace std;

vector<int>s;
bool use[1111111];

void MAKE(){
  use[0]=use[1]=1;
  for(int i=2;i<1111111;i++){
    if(!use[i]){
      for(int j=i+i;j<1111111;j+=i) use[j]=1;
    }
  }
  r(i,1111111) if(!use[i]) s.push_back(i);
}

int get(int a,int b){
  int l=-1,r=1e13;
  while(r-l>1){
    int mid=(l+r)/2;
    if(mid*a>=b) r=mid;
    else l=mid;
  }
  return r;
}

int sum[1000009],A[1000009];

signed main(){
  MAKE();
  int l,r;
  cin>>l>>r;
  r(i,r-l+1)A[i]=i+l;
  for(int i=0;i<s.size();i++){
    int N=get(s[i],l);
    N*=s[i];//if(i<=10)cout<<N<<endl;
    while(N<=r){
      while(A[N-l]%s[i]==0){
        sum[N-l]++;
        A[N-l]/=s[i];
      }
      N+=s[i];
    }
  }
  int ans=0;
  r(i,r-l+1){
    if(1<A[i])sum[i]++;
    if(!use[sum[i]]) ans++;
  }
  cout<<ans<<endl;
}
