(n, m) = [int(x) for x in input().split()]
res = 0
tmp1 = []
tmp2 = []
for i in range(1, 6):
	if n >= i:
		tmp1.append(int((n - i) / 5) + 1)
	else:
		tmp1.append(0)
	if m >= i:
		tmp2.append(int((m - i) / 5) + 1)
	else:
		tmp2.append(0)
for i in range(4):
	res += tmp1[i] * tmp2[3 - i]
res += tmp1[4] * tmp2[4]
print(res)
