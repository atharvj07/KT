t = int(input())
for i in range(t):
	(n, p, k) = map(int, input().split())
	l = list(input())
	(x, y) = map(int, input().split())
	val = [0] * n
	ans = x * n + y * n
	for j in range(n - 1, -1, -1):
		if l[j] == '0':
			val[j] = x + val[j + k] if j + k < n else x
		else:
			val[j] = val[j + k] if j + k < n else 0
	for j in range(p - 1, n, 1):
		ans = min(ans, (j + 1 - p) * y + val[j])
	print(ans)
