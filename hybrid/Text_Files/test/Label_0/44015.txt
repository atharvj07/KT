from collections import defaultdict
import math

class SegmentTree:

	def __init__(self, n):
		self.n = self.next_power(n)
		self.size = self.n * 2 - 1
		self.ZERO = 0
		self.tree = [self.ZERO] * self.size

	def next_power(self, x):
		return 1 if x == 0 else 2 ** math.ceil(math.log2(x))

	def parent(self, i):
		return (i - 1) // 2 if i != 0 else None

	def l_child(self, i):
		return i * 2 + 1

	def r_child(self, i):
		return i * 2 + 2

	def set(self, i, val):
		i = self.n - 1 + i
		self.tree[i] = self.set_val(val)
		self.up(i)

	def set_val(self, val):
		return val

	def up(self, i):
		p_i = self.parent(i)
		if p_i is not None:
			l_i = self.l_child(p_i)
			r_i = self.r_child(p_i)
			self.tree[p_i] = self.update(self.tree[l_i], self.tree[r_i])
			self.up(p_i)

	def update(self, l_val, r_val):
		return l_val + r_val

	def get(self, l, r):
		return self.down(0, l, r, 0, self.n)

	def down(self, i, l, r, lx, rx):
		if lx >= l and rx <= r:
			return self.tree[i]
		if rx <= l or lx >= r:
			return 0
		m = (rx + lx) // 2
		(l_i, r_i) = (self.l_child(i), self.r_child(i))
		return self.update(self.down(l_i, l, r, lx, m), self.down(r_i, l, r, m, rx))

def get_coords(pos, m):
	row = pos // m
	col = pos - row * m
	return (row, col)

def solve(n, m, a):
	cnt = defaultdict(list)
	for i in range(len(a) - 1, -1, -1):
		cnt[a[i]].append(i)
	matrix = [[0] * m for _ in range(n)]
	start_pos = 0
	keys = sorted(cnt.keys())
	for key in keys:
		while cnt[key]:
			(curr_row, curr_col) = get_coords(start_pos, m)
			max_col = min(m, curr_col + len(cnt[key]))
			for col in range(max_col - 1, curr_col - 1, -1):
				matrix[curr_row][col] = cnt[key].pop()
				start_pos += 1
	num_inconv = 0
	for row in matrix:
		row_with_index = [(num, i) for (i, num) in enumerate(row)]
		row_with_index.sort()
		tree = SegmentTree(len(row_with_index))
		for (_, i) in row_with_index:
			num_inconv += tree.get(0, i)
			tree.set(i, 1)
	return num_inconv
t = int(input())
for _ in range(t):
	(n, m) = [int(val) for val in input().split()]
	a = [int(val) for val in input().split()]
	print(solve(n, m, a))
