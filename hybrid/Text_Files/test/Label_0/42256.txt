n = int(input())
a = [*map(int, input().split())]
b = [*map(int, input().split())]
mpa = dict()
mpb = dict()
for i in range(n):
	mpa[a[i]] = i
	mpb[b[i]] = i
mp1 = dict()
for i in range(n):
	j = mpb[a[i]]
	try:
		if j > i:
			mp1[i + n - j] += 1
		else:
			mp1[i - j] += 1
	except KeyError:
		if j > i:
			mp1[i + n - j] = 1
		else:
			mp1[i - j] = 1
mp2 = dict()
for i in range(n):
	j = mpb[a[i]]
	try:
		if j >= i:
			mp2[j - i] += 1
		else:
			mp2[j + n - i] += 1
	except KeyError:
		if j >= i:
			mp2[j - i] = 1
		else:
			mp2[j + n - i] = 1
res = 0
for i in mp1.values():
	res = max(res, i)
for i in mp2.values():
	res = max(res, i)
print(res)
