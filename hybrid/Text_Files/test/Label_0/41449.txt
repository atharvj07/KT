from math import *
from collections import *
import sys
input = sys.stdin.readline

def inp():
	return int(input())

def st():
	return input().rstrip('\n')

def lis():
	return list(map(int, input().split()))

def ma():
	return map(int, input().split())
t = inp()
while t:
	t -= 1
	n = inp()
	dp = [[] for i in range(n)]
	u = lis()
	s = lis()
	res = [0 for i in range(n)]
	pre = [[0] for i in range(n)]
	co = 0
	for i in u:
		dp[i - 1].append(s[co])
		co += 1
	for i in dp:
		if i:
			i.sort(reverse=True)
	for i in range(n):
		for j in dp[i]:
			pre[i].append(pre[i][-1] + j)
	for i in range(n):
		z = len(dp[i])
		for j in range(len(dp[i])):
			if z % (j + 1) == 0:
				res[j] += pre[i][-1]
			else:
				x = z - z % (j + 1)
				res[j] += pre[i][x]
	print(*res)
