#include <bits/stdc++.h>
using namespace std;
int main() {
  int i, n, m, t = 1, k;
  long long s = 0;
  scanf("%d %d", &n, &m);
  for (i = 1; i <= m; i++) {
    scanf("%d", &k);
    if (k >= t) {
      s += k - t;
    } else {
      s += n - t + k;
    }
    t = k;
  }
  printf("%lld\n", s);
  return 0;
}
