n = int(input())
a = [int(i) for i in input().split()]
if n == 1 and a[0] == 1:
	print(0)
elif n == 1 and a[0] == 0:
	print(1)
elif n == sum(a):
	print(n - 1)
else:
	max = sum(a)
	for i in range(n):
		for j in range(i + 1, n + 1):
			x = sum(a[0:i]) + (j - i) - sum(a[i:j]) + sum(a[j:])
			if x > max:
				max = x
	print(max)
