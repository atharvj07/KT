[n, m] = list(map(int, input().split()))
p = []
for i in range(n):
	p.append(input())
minx = 1000
miny = 1000
maxx = 0
maxy = 0
for i in range(n):
	for j in range(m):
		if p[i][j] == 'X':
			miny = min(miny, i)
			maxy = max(maxy, i)
			minx = min(minx, j)
			maxx = max(maxx, j)
for i in range(miny, maxy + 1):
	for j in range(minx, maxx + 1):
		if p[i][j] != 'X':
			print('NO')
			exit(0)
print('YES')
