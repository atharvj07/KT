#include <bits/stdc++.h>
using namespace std;
long long fastPow(long long a, long long pw, long long mod) {
  long long res = 1;
  a %= mod;
  while (pw > 0) {
    if (pw & 1) res = (res * a) % mod;
    a = (a * a) % mod;
    pw >>= 1;
  }
  return res;
}
const long long mod = 1e6 + 3;
void gauss(vector<vector<long long>> &a) {
  long long n = a.size();
  long long m = a[0].size() - 1;
  for (long long col = 0, row = 0; col < m && row < n; ++col) {
    long long sel = row;
    for (long long i = row + 1; i < n; ++i)
      if (abs(a[i][col]) > abs(a[sel][col])) sel = i;
    if (abs(a[sel][col]) == 0) {
      continue;
    }
    for (long long j = 0; j <= m; ++j) swap(a[sel][j], a[row][j]);
    long long s = fastPow(a[row][col], mod - 2, mod);
    for (long long j = 0; j <= m; ++j) a[row][j] = (a[row][j] * s) % mod;
    for (long long i = 0; i < n; ++i)
      if (i != row && abs(a[i][col]) != 0) {
        long long t = a[i][col];
        for (long long j = 0; j <= m; ++j)
          a[i][j] = (a[i][j] - (a[row][j] * t) + mod) % mod;
      }
    ++row;
  }
}
long long query(long long x) {
  cout << "? " << x << '\n';
  cout << flush;
  long long res;
  cin >> res;
  if (res == -1) exit(0);
  if (res == 0) {
    cout << "! " << x << flush;
    exit(0);
  }
  return res;
}
signed main() {
  vector<vector<long long>> mat(11, vector<long long>(12));
  for (long long i = 1; i <= 11; ++i) {
    mat[i - 1][11] = query(i);
    for (long long j = 0; j < 11; ++j) {
      mat[i - 1][j] = fastPow(i, j, mod);
    }
  }
  gauss(mat);
  for (long long x = 0; x < mod; ++x) {
    long long y = 0;
    for (long long i = 0; i < (long long)mat.size(); ++i) {
      y += fastPow(x, i, mod) * mat[i].back() % mod;
      y = (y % mod + mod) % mod;
    }
    assert(0 <= y && y < mod);
    if (y == 0) {
      cout << "! " << x;
      return 0;
    }
  }
  cout << "! " << -1;
}
