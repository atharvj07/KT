from sys import stdin, stdout
import math

def main():
	t = int(input())
	while t:
		(n, k) = map(int, input().split())
		x = 1
		(j1, j2) = (n - 2, n - 1)
		while 1:
			if x * (x + 1) // 2 >= k:
				break
			x += 1
		if x * (x + 1) == 2 * k:
			j1 = j1 - (x - 1)
			j2 = j2 - (x - 1)
		else:
			j1 = j1 - (x - 1)
			j2 = j2 - (x - 1) + (x * (x + 1) // 2 - k)
		s = ''
		for i in range(n):
			if i == j1 or i == j2:
				s += 'b'
			else:
				s += 'a'
		print(s)
		t -= 1
main()
