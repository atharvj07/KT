#include<bits/stdc++.h>
using namespace std;
typedef long long lint;
template<typename TP>inline void read(TP &tar)
{
	TP ret=0,f=1;char ch=getchar();
	while(ch<'0'||ch>'9'){if(ch=='-')f=-1;ch=getchar();}
	while(ch>='0'&&ch<='9'){ret=ret*10+(ch-'0');ch=getchar();}
	tar=ret*f;
}
template<typename TP,typename... Args>inline void read(TP& t,Args&... args){read(t),read(args...);}
namespace RKK
{
const int N=200011;

struct pat
{
	int x,y,p;
	pat(int x=0,int y=0,int p=0):x(x),y(y),p(p){}
	bool operator<(const pat &a)const{return x*a.y-a.x*y<0;}
}pt;
int n,fa[N],a[N],bc[N],f[N],size[N];int find(int x){return f[x]==f[f[x]]?f[x]:f[x]=find(f[x]);}
priority_queue<pat> q;
lint ans;
int main()
{
	read(n);for(int i=2;i<=n;i++) read(fa[i]);for(int i=1;i<=n;i++) read(a[i]),f[i]=i,size[i]=1;
	for(int i=2;i<=n;i++)if(!a[i]) f[i]=find(fa[i]),size[f[i]]+=size[i];
	for(int i=1;i<=n;i++)if(a[i]) bc[i]=1,size[i]--,ans+=size[i];
	for(int i=2;i<=n;i++)if(a[i]) q.push(pat(size[i],bc[i],i));
	for(int i=1;i<=n;i++) fa[i]=find(fa[i]);for(int i=1;i<=n;i++) f[i]=i;
	while(!q.empty())
	{
		pt=q.top(),q.pop();
		int x=pt.p,y=find(fa[x]);
		if(!y||pt.x!=size[x]) continue;
		ans+=1ll*bc[y]*size[x],size[y]+=size[x],bc[y]+=bc[x],f[x]=y;
		q.push(pat(size[y],bc[y],y));
	}
	printf("%lld\n",ans);
	return 0;
}
}
int main(){return RKK::main();}
