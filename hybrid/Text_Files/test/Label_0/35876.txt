import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;
import java.io.*;
import java.math.*;
import java.io.File;
import java.io.FileInputStream;
import java.io.FileNotFoundException;
 
public class Main {
  static ArrayList<Integer> a[] = new ArrayList[100005];
  static int []deg = new int[100005];
  static int []col = new int[100005];
  static int [][]cost = new int[100005][3];

  static long dfs(int n, int pre, int c1, int c2)
  {
    int c = 3 - c1 - c2;
    //System.out.println("n=" + n + " c=" + c + " c1=" + c1 + " c2=" + c2);
    col[n] = c + 1;
    long ans = cost[n][c];
    for (int j : a[n]) {
      if (j != pre) ans += dfs(j, n, c, c1);
    }
    return ans;
  }
  
	public static void main(String[] args) throws Exception {
    MScanner in = new MScanner(System.in);
    int n = in.nextInt();
    for (int i = 1; i <= n; i++) cost[i][0] = in.nextInt();
    for (int i = 1; i <= n; i++) cost[i][1] = in.nextInt();
    for (int i = 1; i <= n; i++) cost[i][2] = in.nextInt();

    for (int i = 0; i <= n; i++) {
      a[i] = new ArrayList<>();
    }
    int root = 0;
    int bad = 0;
    for (int i = 1;i <= n - 1;++i) {
      int i1 = in.nextInt();
      int i2 = in.nextInt();
      a[i1].add(i2);
      a[i2].add(i1);
      deg[i1] += 1;
      deg[i2] += 1;
      if (deg[i1] >= 3) bad = 1;
      if (deg[i2] >= 3) bad = 1;
    }
    for (int i = 1;i <= n;++i) {
      if (deg[i] == 1) root = i;
    }
    if (bad > 0) {
      System.out.println(-1);
      return;
    }
    long min = -1;
    int mi = 0;
    int mj = 0;
    for (int i = 0;i <= 2;++i) {
      for (int j = 0;j <= 2;++j) {
        if (i == j) continue;
        long ans = dfs(root, 0, i, j);
        if (min == -1 || ans < min) {
          min = ans;
          mi = i;
          mj = j;
        }
      }
    }
    dfs(root, 0, mi, mj);
    System.out.println(min);
    for (int i = 1;i <= n;++i) {
      System.out.print(col[i] + " ");
    }
	}
  static class MScanner {
    StringTokenizer st;
    BufferedReader br;
    public MScanner(InputStream system) {
      br = new BufferedReader(new InputStreamReader(system));
    }

    public MScanner(String file) throws Exception {
      br = new BufferedReader(new FileReader(file));
    }

    public String next() throws IOException {
      while (st == null || !st.hasMoreTokens())
	st = new StringTokenizer(br.readLine());
      return st.nextToken();
    }
    public int[] takearr(int n) throws IOException {
      int[]in=new int[n];for(int i=0;i<n;i++)in[i]=nextInt();
      return in;
    }
    public long[] takearrl(int n) throws IOException {
      long[]in=new long[n];for(int i=0;i<n;i++)in[i]=nextLong();
      return in;
    }
    public Integer[] takearrobj(int n) throws IOException {
      Integer[]in=new Integer[n];for(int i=0;i<n;i++)in[i]=nextInt();
      return in;
    }
    public Long[] takearrlobj(int n) throws IOException {
      Long[]in=new Long[n];for(int i=0;i<n;i++)in[i]=nextLong();
      return in;
    }
    public String nextLine() throws IOException {
      return br.readLine();
    }

    public int nextInt() throws IOException {
      return Integer.parseInt(next());
    }

    public double nextDouble() throws IOException {
      return Double.parseDouble(next());
    }

    public char nextChar() throws IOException {
      return next().charAt(0);
    }

    public Long nextLong() throws IOException {
      return Long.parseLong(next());
    }

    public boolean ready() throws IOException {
      return br.ready();
    }

    public void waitForInput() throws InterruptedException {
      Thread.sleep(3000);
    }
  }
}
