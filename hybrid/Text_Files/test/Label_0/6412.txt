#include <bits/stdc++.h>
using namespace std;
int dis[1001][1002];
int mov_x[] = {0, 1, 0, -1};
int mov_y[] = {1, 0, -1, 0};
bool check(int i, int j, int n, int m, char grid[][1002]) {
  if (i < 0 || j < 0 || i >= n || j >= m || grid[i][j] == 'T') return false;
  return true;
}
void bfs(int a, int b, char grid[][1002], int n, int m) {
  int i, j;
  for (i = 0; i <= n - 1; i++) {
    for (j = 0; j <= m - 1; j++) dis[i][j] = INT_MAX;
  }
  dis[a][b] = 0;
  queue<pair<int, int> > q;
  q.push(make_pair(a, b));
  while (!q.empty()) {
    pair<int, int> p = q.front();
    q.pop();
    for (i = 0; i <= 3; i++) {
      int x = p.first + mov_x[i];
      int y = p.second + mov_y[i];
      if (check(x, y, n, m, grid) && dis[x][y] > dis[p.first][p.second] + 1) {
        dis[x][y] = dis[p.first][p.second] + 1;
        q.push(make_pair(x, y));
      }
    }
  }
  return;
}
int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  int i, j;
  char grid[1001][1002];
  for (i = 0; i <= n - 1; i++) {
    cin >> grid[i];
  }
  int count = 0;
  int a, b, x, y;
  for (i = 0; i <= n - 1; i++) {
    for (j = 0; j <= m - 1; j++) {
      if (grid[i][j] == 'S') a = i, b = j;
      if (grid[i][j] == 'E') x = i, y = j;
    }
  }
  bfs(x, y, grid, n, m);
  for (i = 0; i <= n - 1; i++) {
    for (j = 0; j <= m - 1; j++) {
      if (grid[i][j] - '0' > 0 && grid[i][j] - '0' < 10) {
        if (dis[i][j] <= dis[a][b]) count += grid[i][j] - '0';
      }
    }
  }
  cout << count << endl;
}
