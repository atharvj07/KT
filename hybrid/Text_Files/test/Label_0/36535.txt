import java.util.*;

public class B {

    public static void main(String[] args) {
        Scanner scn = new Scanner(System.in);
        int t = scn.nextInt();
        scn.nextLine();
        while (t-- > 0) {
            char c[] = scn.nextLine().toCharArray();
            StringBuilder sb = new StringBuilder();
            int r = 0, p = 0, s = 0;
            for (int i = 0; i < c.length; ++i) {
                if (c[i] == 'R') {
                    r++;
                } else if (c[i] == 'P') {
                    p++;
                } else {
                    s++;
                }
            }
            if(r >= p && r >= s){
                for(int i=0;i<c.length;++i){
                    System.out.print("P");
                }
            } else if(p >= r && p >= s){
                for(int i=0;i<c.length;++i){
                    System.out.print("S");
                }
            } else {
                for(int i=0;i<c.length;++i){
                    System.out.print("R");
                }
            }
            System.out.println();
        }
    }
}
