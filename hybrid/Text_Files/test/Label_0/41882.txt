n = int(input())
arr = list(map(int, input().split()))
arr.sort()
seen = {}
cost = 0
for i in range(len(arr)):
	cur = arr[i]
	if cur in seen:
		found = False
		val = cur + 1
		cost += 1
		while not found:
			if val not in seen:
				seen[val] = True
				found = True
			else:
				val += 1
				cost += 1
	else:
		seen[cur] = True
print(cost)
