(a, b) = map(int, input().split())
s = str(input())
t = str(input())
main = []
for i in range(b - a + 1):
	c = 0
	p = []
	for j in range(a):
		if t[i] != s[j]:
			c += 1
			p.append(j)
		i += 1
	p.insert(0, c)
	main.append(p)
main.sort()
print(main[0][0])
for t in main[0][1:]:
	print(t + 1, end=' ')
print('\r')
