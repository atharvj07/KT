#include<iostream>
#include<vector>
#include<string>
using namespace std;
int main(){
  int n,m;
  while(true){
  cin>>n>>m;
  if(!n&&!m)break;
  vector<int> money(n);
  vector<string> bingo(n);
  vector<string> buy(m);
  for(int i=0;i<n;i++){
    cin>>bingo[i]>>money[i];
  }
  for(int i=0;i<m;i++){
    cin>>buy[i];
  }
  int sum = 0;
  for(int i=0;i<n;i++){
    for(int j=0;j<m;j++){
      int k=0;
      for(k=0;k<8;k++){
	if(bingo[i][k]!='*' && bingo[i][k]!=buy[j][k])break;
      }
      if(k==8)
	sum+=money[i];
    }
  }
  cout<<sum<<endl;
  }
  return 0;
}
