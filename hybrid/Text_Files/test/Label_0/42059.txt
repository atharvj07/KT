t = int(input())
bar = [False for x in range(t)]

def turn(index):
	if bar[index] == True:
		return
	bar[index] = True
num1 = list(map(int, input().split()[1:]))
for i in num1:
	turn(i - 1)
num2 = list(map(int, input().split()[1:]))
for i in num2:
	turn(i - 1)
if False not in bar:
	print('I become the guy.')
else:
	print('Oh, my keyboard!')
