import os
import random
import sys

def solve():
	(have,) = rl(1)
	(want,) = rl(1)
	spare = 0
	for index in range(3):
		if have[index] > want[index]:
			diff = have[index] - want[index]
			spare += diff // 2
		else:
			more = want[index] - have[index]
			spare -= more
	print('Yes' if spare >= 0 else 'No')

def rv():
	return map(int, input().split())

def rl(n):
	return [list(map(int, input().split())) for _ in range(n)]
if os.path.exists('test.txt'):
	sys.stdin = open('test.txt')
solve()
