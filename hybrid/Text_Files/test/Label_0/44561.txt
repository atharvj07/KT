n = int(input())
S = input()
pa = -1
pb = -1
cnt = 0
for (i, s) in enumerate(S):
	if s == 'A':
		if pa == -1:
			pa = i
		else:
			cnt += pa + 1
			if S[i - 1] == 'A' and pb != -1:
				cnt -= 1
			pa = i
	if s == 'B':
		if pb == -1:
			pb = i
		else:
			cnt += pb + 1
			if S[i - 1] == 'B' and pa != -1:
				cnt -= 1
			pb = i
print(cnt)
