#include <bits/stdc++.h>
using namespace std;
struct Node {
  Node() {}
  Node(string s, string p) : s(s), p(p) {}
  string s, p;
} a;
int i, j, k, n, m;
char c;
inline Node Solve(int n) {
  Node Tmp;
  if (n == 1) {
    c = 'a';
    return Node("a", "a");
  }
  if (n == 2) {
    c = 'b';
    return Node("abb", "ab");
  }
  if (n % 2 == 0) {
    Tmp = Solve(n / 2 - 1);
    c++;
    Tmp.s.insert((int)Tmp.p.size(), string(2, c));
    Tmp.s += string(2, c);
    Tmp.p += string(1, c);
    return Tmp;
  }
  Tmp = Solve(n / 2);
  c++;
  Tmp.s.insert((int)Tmp.p.size(), string(1, c));
  Tmp.s += string(2, c);
  Tmp.p += string(1, c);
  return Tmp;
}
int main() {
  scanf("%d", &n);
  a = Solve(n);
  for (i = 0; i < a.s.size(); i++) putchar(a.s[i]);
  putchar(' ');
  for (i = 0; i < a.p.size(); i++) putchar(a.p[i]);
  puts("");
  return 0;
}
