#include <bits/stdc++.h>
using namespace std;
const long long MOD = (long long)1e9 + 7;
const long double PI = 3.141592653589793238462643383279502884197;
priority_queue<int, vector<int>, greater<int> > pq;
set<int> ma[2001];
int lo[2001];
char visit[5001];
char ans[2001];
char two[2001][2001];
void add(char *p, char *a, char *b) {
  int n = strlen(a) - 1, m = strlen(b) - 1;
  vector<char> v;
  int carry = 0;
  while (n >= 0 || m >= 0) {
    int sum = 0;
    if (n >= 0) sum += a[n] - '0';
    if (m >= 0) sum += b[m] - '0';
    sum += carry;
    carry = sum / 10;
    sum %= 10;
    v.push_back(sum + '0');
    n--;
    m--;
  }
  if (carry) v.push_back('1');
  int t = 0;
  memset(p, 0, sizeof(ans));
  for (int i = (int)(v).size() - 1; i >= 0; i--) p[t++] = v[i];
}
int main() {
  memset(lo, -1, sizeof(lo));
  int n;
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    char s[5];
    int a;
    scanf("%s %d", s, &a);
    if (strcmp(s, "win") == 0)
      ma[a].insert(i);
    else
      lo[a] = i;
  }
  two[0][0] = '1';
  for (int i = 1; i <= 2000; i++) add(two[i], two[i - 1], two[i - 1]);
  ans[0] = '0';
  for (int i = 2000; i >= 0; i--) {
    if (lo[i] == -1) continue;
    auto k = ma[i].lower_bound(lo[i]);
    if (k == ma[i].begin()) continue;
    k--;
    int flag = 0;
    for (int j = *k; j <= lo[i]; j++) flag += visit[j];
    if (flag) continue;
    for (int j = *k; j <= lo[i]; j++) visit[j] = 1;
    add(ans, ans, two[i]);
  }
  printf("%s", ans);
}
