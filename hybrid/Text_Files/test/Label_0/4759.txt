#include <stdio.h>
#include <vector>
#include <algorithm>

int main(void) {
  int i, j, k, l, m, n;
  long long ans = 0;
  std::vector<int> vx, vy, vd;
  scanf("%d%d", &n, &k);
  int x1[n], y1[n], d1[n], x2[n], y2[n], d2[n];
  for(i = 0; i < n; ++i) {
    scanf("%d%d%d%d%d%d", &x1[i], &y1[i], &d1[i], &x2[i], &y2[i], &d2[i]);
    vx.push_back(x1[i]);
    vx.push_back(x2[i]);
    vy.push_back(y1[i]);
    vy.push_back(y2[i]);
    vd.push_back(d1[i]);
    vd.push_back(d2[i]);
  }
  std::sort(vx.begin(), vx.end());
  std::sort(vy.begin(), vy.end());
  std::sort(vd.begin(), vd.end());
  vx.erase(std::unique(vx.begin(), vx.end()), vx.end());
  vy.erase(std::unique(vy.begin(), vy.end()), vy.end());
  vd.erase(std::unique(vd.begin(), vd.end()), vd.end());
  for(i = 1; i < vx.size(); ++i) {
    for(j = 1; j < vy.size(); ++j) {
      for(l = 1; l < vd.size(); ++l) {
        int t = 0;
        for(m = 0; m < n; ++m) {
          if(x1[m] <= vx[i - 1] && vx[i] <= x2[m] && y1[m] <= vy[j - 1] && vy[j] <= y2[m] && d1[m] <= vd[l - 1] && vd[l] <= d2[m]) ++t;
        }
        if(t >= k) ans += (long long)(vx[i] - vx[i - 1]) * (long long)(vy[j] - vy[j - 1]) * (long long)(vd[l] - vd[l - 1]);
      }
    }
  }
  printf("%lld\n", ans);
  return 0;
}
