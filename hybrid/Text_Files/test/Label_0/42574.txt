[n, k1, k2] = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
c = [0 for i in range(n)]
for i in range(n):
	c[i] = a[i] - b[i] if a[i] > b[i] else b[i] - a[i]
c.sort(reverse=True)
k = k1 + k2
while k > 0:
	i = 0
	while i != n - 1 and c[i] <= c[i + 1]:
		i += 1
	c[i] = abs(c[i] - 1)
	k -= 1
result = 0
for x in c:
	result += x * x
print(result)
