def solution():
	n = int(input())
	adj = [[] for _ in range(n)]
	for _ in range(n - 1):
		(u, v) = map(lambda x: int(x) - 1, input().strip().split())
		adj[u].append(v)
		adj[v].append(u)
	colors = [-1] * n
	stk = [0]
	colors[0] = 0
	while stk:
		u = stk.pop()
		for v in adj[u]:
			if colors[v] != -1:
				continue
			colors[v] = colors[u] ^ 1
			stk.append(v)
	(w, color) = (sum(colors), 1)
	if w > n - w:
		(w, color) = (n - w, 0)
	result = [0] * n
	idx1 = [i for (i, c) in enumerate(colors) if c == color]
	idx2 = [j for (j, c) in enumerate(colors) if c != color]
	bit = 1
	i = j = 0
	while bit <= n:
		if w & bit:
			for x in range(bit, min(2 * bit - 1, n) + 1):
				result[idx1[i]] = x
				i += 1
		else:
			for x in range(bit, min(2 * bit - 1, n) + 1):
				result[idx2[j]] = x
				j += 1
		bit <<= 1
	return ' '.join(map(lambda x: str(x), result))
for case in range(int(input())):
	print('%s' % solution())
