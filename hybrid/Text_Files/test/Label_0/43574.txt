from collections import *
from sys import stdin, stdout
input = stdin.buffer.readline
print = stdout.write

def ri():
	return int(input())

def rl():
	return list(map(int, input().split()))
n = ri()
h = rl()
sum = 0
for i in range(n):
	sum += h[i]
sum -= int(n * (n - 1) / 2)
x = sum // n + 1
a = sum % n
for i in range(a):
	print(str(x + i) + ' ')
for i in range(a, n):
	print(str(x - 1 + i) + ' ')
print('\n')
