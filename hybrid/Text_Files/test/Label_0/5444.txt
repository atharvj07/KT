#include <bits/stdc++.h>
using namespace std;
long long a[200100];
long long m;
long long n, k;
bool f(long long x) {
  long long s = 0;
  for (int i = m; i < n; i++) {
    if (a[i] < x) {
      s += x - a[i];
    }
  }
  if (s <= k) {
    return 1;
  }
  return 0;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin >> n >> k;
  m = n / 2;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a, a + n);
  long long l = a[m], r = 2000000001;
  long long an = a[m];
  while (l <= r) {
    long long mi = (l + r) / 2;
    if (f(mi)) {
      an = max(an, mi);
      l = mi + 1;
    } else {
      r = mi - 1;
    }
  }
  cout << an;
  return 0;
}
