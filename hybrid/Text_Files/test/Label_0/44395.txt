import os, io, math
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
n = int(input())
a = list(map(int, input().split()))
cuml = [0] * n
cumr = [0] * n
curr = 0
maxi = 0
for i in range(1, n):
	if a[i] > a[i - 1]:
		curr += 1
	else:
		curr = 0
	cuml[i] = curr
	maxi = max(maxi, curr)
curr = 0
for i in range(n - 2, -1, -1):
	if a[i] > a[i + 1]:
		curr += 1
	else:
		curr = 0
	cumr[i] = curr
	maxi = max(maxi, curr)
l = set()
for i in range(n):
	if cuml[i] == maxi or cumr[i] == maxi:
		l.add(i)
if len(l) >= 2:
	print(0)
	exit()
if maxi % 2 == 0:
	for i in l:
		if cuml[i] == cumr[i]:
			print(1)
		else:
			print(0)
else:
	print(0)
