(m, x) = map(int, input().split())
if x == 1:
	print(1)

	def divide(n):

		def generator(n):
			divisors = []
			for i in range(1, int(n ** (1 / 2)) + 1):
				if n % i == 0:
					yield i
					if i * i != n:
						divisors.append(n / i)
			for divisor in reversed(divisors):
				yield int(divisor)
		return list(generator(n))[1:-1]
	factors = divide(x)
	for i in range(len(factors)):
		if factors[i] > m:
			j = factors.index(x // factors[i])
			factors = factors[j + 1:i]
			break
	occ = len(factors)
	if x <= m:
		occ += 2
	print(occ)
