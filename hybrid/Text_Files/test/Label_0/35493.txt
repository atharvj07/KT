import java.io.*;
import java.util.*;

public class D implements Runnable{
	public static void main (String[] args) {new Thread(null, new D(), "_cf", 1 << 28).start();}

	public void run() {
		FastScanner fs = new FastScanner();
		PrintWriter out = new PrintWriter(System.out);
		System.err.println("");

		int n = fs.nextInt(), k = fs.nextInt();
		long[] a = new long[n], b = new long[n];
		for(int i = 0; i < n; i++) a[i] = fs.nextLong();
		for(int i = 0; i < n; i++) b[i] = fs.nextLong();
		
		long res = -1;
		long lo = 0, hi = (long)1e13;
		int[] lastUpdate = new int[n];
		ArrayDeque<Laptop>[] list = new ArrayDeque[k];
		for(int i = 0; i < k; i++) list[i] = new ArrayDeque<>();
		
		for(int bs = 0; bs < 100 && lo <= hi; bs++) {
			long mid = (lo + hi) / 2;
			Arrays.fill(lastUpdate, 0);
			for(int i = 0; i < n; i++) {
				Laptop l = new Laptop(a[i], b[i], i);
				l.when = a[i]/b[i];
				if(l.when < k-1) list[(int)l.when].add(l);
			}
			boolean bad = false;
			int j = 0;
			for(int i = 0; i < k-1 && !bad; i++) {
				while(j < k-1 && list[j].isEmpty()) j++;
				if(j == k-1) break;
				Laptop now = list[j].pollFirst();
				if(now.when < i) { bad = true; break; }
				
				now.charge -= now.rate*(i-lastUpdate[now.id]+1);
				lastUpdate[now.id] = i+1;
				now.charge += mid;
				if(now.charge < 0) { bad = true; break; }
				now.when = i + (now.charge/now.rate)+1;
				if(now.when < k-1) {
					list[(int)now.when].add(now);
				}
			}
			while(j < k-1 && list[j].isEmpty()) j++;
			for(int i = 0; i < k; i++) list[i].clear();
			if(j < k-1) bad = true;
			
			if(bad) { lo = mid+1; }
			else {
				res = mid;
				hi = mid-1;
			}
		}
		
		out.println(res);
		out.close();
	}

	class Laptop implements Comparable<Laptop> {
		long charge, rate, when;
		int id;
		Laptop(long c, long r, int i) {
			charge = c; rate = r;
			id = i;
		}
		
		public int compareTo(Laptop l) {
			return Long.compare(when, l.when);
		}
		public String toString() {
			return String.format("ID %d (%d %d) -- %d", id, charge, -rate, when);
		}
	}

	class FastScanner {
		public int BS = 1<<16;
		public char NC = (char)0;
		byte[] buf = new byte[BS];
		int bId = 0, size = 0;
		char c = NC;
		double num = 1;
		BufferedInputStream in;

		public FastScanner() {
			in = new BufferedInputStream(System.in, BS);
		}

		public FastScanner(String s) {
			try {
				in = new BufferedInputStream(new FileInputStream(new File(s)), BS);
			}
			catch (Exception e) {
				in = new BufferedInputStream(System.in, BS);
			}
		}

		public char nextChar(){
			while(bId==size) {
				try {
					size = in.read(buf);
				}catch(Exception e) {
					return NC;
				}                
				if(size==-1)return NC;
				bId=0;
			}
			return (char)buf[bId++];
		}

		public int nextInt() {
			return (int)nextLong();
		}

		public long nextLong() {
			num=1;
			boolean neg = false;
			if(c==NC)c=nextChar();
			for(;(c<'0' || c>'9'); c = nextChar()) {
				if(c=='-')neg=true;
			}
			long res = 0;
			for(; c>='0' && c <='9'; c=nextChar()) {
				res = (res<<3)+(res<<1)+c-'0';
				num*=10;
			}
			return neg?-res:res;
		}

		public double nextDouble() {
			double cur = nextLong();
			return c!='.' ? cur:cur+nextLong()/num;
		}

		public String next() {
			StringBuilder res = new StringBuilder();
			while(c<=32)c=nextChar();
			while(c>32) {
				res.append(c);
				c=nextChar();
			}
			return res.toString();
		}

		public String nextLine() {
			StringBuilder res = new StringBuilder();
			while(c<=32)c=nextChar();
			while(c!='\n') {
				res.append(c);
				c=nextChar();
			}
			return res.toString();
		}

		public boolean hasNext() {
			if(c>32)return true;
			while(true) {
				c=nextChar();
				if(c==NC)return false;
				else if(c>32)return true;
			}
		}

		public int[] nextIntArray(int n) {
			int[] res = new int[n];
			for(int i = 0; i < n; i++) res[i] = nextInt();
			return res;
		}

	}

}
