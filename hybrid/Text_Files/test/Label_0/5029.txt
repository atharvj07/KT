#include <bits/stdc++.h>
#pragma GCC optimize(3)
using namespace std;
inline long long read() {
  long long first = 0, f = 1;
  char ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') {
    first = first * 10 + ch - '0';
    ch = getchar();
  }
  return first * f;
}
char F[200];
inline void out(long long first) {
  if (first == 0) return (void)(putchar('0'));
  long long tmp = first > 0 ? first : -first;
  if (first < 0) putchar('-');
  int cnt = 0;
  while (tmp > 0) {
    F[cnt++] = tmp % 10 + '0';
    tmp /= 10;
  }
  while (cnt > 0) putchar(F[--cnt]);
}
long long ksm(long long a, long long b, long long p) {
  long long res = 1;
  while (b) {
    if (b & 1) res = res * a % p;
    a = a * a % p;
    b >>= 1;
  }
  return res;
}
const int inf = 0x3f3f3f3f, mod = 1e9 + 7;
const long long INF = 0x3f3f3f3f3f3f3f3f;
const int maxn = 1e7 + 7, maxm = 3e6 + 7;
long long n, a[maxn];
struct Xortrie {
  int node;
  int son[maxn][2];
  int L[maxn], R[maxn];
  void Xortrie_init() { node = 0; }
  void add(long long first, int id) {
    int rt = 0;
    for (int i = 32; i >= 0; i--) {
      int tmp = (first >> i & 1ll) ? 1 : 0;
      int &tt = son[rt][tmp];
      if (!tt) tt = ++node;
      rt = tt;
      if (!L[rt]) L[rt] = id;
      R[rt] = id;
    }
  }
  long long qaskpos(int rt, int pos, long long first) {
    long long res = 0;
    for (int i = pos; i >= 0; i--) {
      int tmp = (first >> i & 1ll) ? 1 : 0;
      if (son[rt][tmp])
        rt = son[rt][tmp];
      else {
        res += (1ll << i);
        rt = son[rt][!tmp];
      }
    }
    return res;
  }
  long long Divide(int rt, int pos) {
    if (son[rt][0] && son[rt][1]) {
      int first = son[rt][0], second = son[rt][1];
      long long minn = INF;
      if (R[first] - L[first] + 1 <= R[second] - L[second] + 1) {
        for (int k = L[first]; k <= R[first]; k++)
          minn = min(minn, qaskpos(second, pos - 1, a[k]) + (1ll << pos));
      } else {
        for (int k = L[second]; k <= R[second]; k++)
          minn = min(minn, qaskpos(first, pos - 1, a[k]) + (1ll << pos));
      }
      return minn + Divide(first, pos - 1) + Divide(second, pos - 1);
    } else if (son[rt][0]) {
      return Divide(son[rt][0], pos - 1);
    } else if (son[rt][1]) {
      return Divide(son[rt][1], pos - 1);
    }
    return 0ll;
  }
  void debug(int rt) {
    printf("%d %d\n", L[rt], R[rt]);
    if (son[rt][0]) debug(son[rt][0]);
    if (son[rt][1]) debug(son[rt][1]);
  }
} Fuck;
int main() {
  Fuck.Xortrie_init();
  n = read();
  for (int i = 1; i <= n; i++) a[i] = read();
  sort(a + 1, a + 1 + n);
  for (int i = 1; i <= n; i++) Fuck.add(a[i], i);
  long long q = Fuck.Divide(0, 32);
  out(q);

}
