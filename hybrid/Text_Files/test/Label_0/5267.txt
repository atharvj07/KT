#include<bits/stdc++.h>
#define L long long
#define pb push_back
using namespace std;
int n,p;
vector<int> a,b;
map<L,int> h;
set<int> f[100010];
int main()
{
	//freopen(".in","r",stdin);
	//freopen(".out","w",stdout);
	int i,j,k;
	scanf("%d",&n);
	for(i=1;i<=2*n-2;i++)
	  {
	   scanf("%d%d",&j,&k);
	   h[(L)j*n+k]++;
	   h[(L)k*n+j]++;
	   f[j].insert(k);
	   f[k].insert(j);
	   if(h[(L)j*n+k]==2)
	     a.pb(j),b.pb(k);
	  }
	for(i=0;i<a.size();i++)
	  if(h[(L)a[i]*n+b[i]]==2)
	    {
		 p++;
		 if(f[a[i]].size()<f[b[i]].size())
		   swap(a[i],b[i]);
		 while(f[b[i]].size())
		   {
			j=*f[b[i]].begin();
			h[(L)a[i]*n+j]+=h[(L)b[i]*n+j];
			h[(L)j*n+a[i]]+=h[(L)b[i]*n+j];
			h[(L)b[i]*n+j]=h[(L)j*n+b[i]]=0;
			f[a[i]].insert(j);
			f[j].erase(b[i]);
			f[j].insert(a[i]);
			if(h[(L)a[i]*n+j]==2)
			  a.pb(a[i]),b.pb(j);
			f[b[i]].erase(j);
		   }
		}
	if(p==n-1)
	  printf("YES\n");
	else
	  printf("NO\n");
	return 0;
}
