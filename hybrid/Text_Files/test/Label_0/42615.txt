class UnionFind:

	def __init__(self, n):
		self.state = [-1] * n
		self.size_table = [1] * n
		self.cnt = n

	def root(self, u):
		stack = []
		while self.state[u] >= 0:
			stack.append(u)
			u = self.state[u]
		for v in stack:
			self.state[v] = u
		return u

	def merge(self, u, v):
		u = self.root(u)
		v = self.root(v)
		if u == v:
			return
		du = -self.state[u]
		dv = -self.state[v]
		if du < dv:
			(u, v) = (v, u)
		if du == dv:
			self.state[u] -= 1
		self.state[v] = u
		self.cnt -= 1
		self.size_table[u] += self.size_table[v]
		return

	def same(self, u, v):
		return self.root(u) == self.root(v)

	def size(self, u):
		return self.size_table[self.root(u)]
for _ in range(II()):

	ai.sort(key=lambda x: x[0])

	ans = 0
	fin = [False] * n
	for (a, i) in ai:

			if aa[j] % a:
				break
			if uf.same(i, j):
				break
			ans += a
			uf.merge(i, j)
			fin[j] = True
		for j in range(i + 1, n):
			if aa[j] % a:
				break
			if uf.same(i, j):
				break
			ans += a
			uf.merge(i, j)
			fin[j] = True
	ans += (uf.cnt - 1) * p
