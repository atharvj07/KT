#include <bits/stdc++.h>
using namespace std;
int n, k, power[100010], out;
int main() {
  scanf("%d%d", &n, &k);
  for (int i = 0; i < n; i++) {
    int tmp;
    scanf("%d", &tmp);
    power[i % k] += tmp;
  }
  for (int i = 1; i < k; i++)
    if (power[i] < power[out]) out = i;
  printf("%d\n", out + 1);
  return 0;
}
