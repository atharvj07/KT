n = int(input())
a = int(input())
b = int(input())
c = int(input())
n -= 1
s = 1
an = 0
while n:
	if s == 1:
		if a < b:
			s = 2
			an += a
		else:
			s = 3
			an += b
		n -= 1
	elif s == 2:
		if a < c:
			s = 1
			an += a
		else:
			s = 3
			an += c
		n -= 1
	else:
		if b < c:
			s = 1
			an += b
		else:
			s = 2
			an += c
		n -= 1
print(an)
