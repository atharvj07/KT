import math
for _ in range(int(input())):
	(a, b, x, y, n) = list(map(int, input().split()))
	ans = 0
	if n >= a - x + b - y:
		ans = x * y
	elif a - x >= n:
		ans = (a - n) * b
	else:
		ans = x * (b - (n - (a - x)))
	(a, b) = (b, a)
	(x, y) = (y, x)
	ans1 = 0
	if n >= a - x + b - y:
		ans1 = x * y
	elif a - x >= n:
		ans1 = (a - n) * b
	else:
		ans1 = x * (b - (n - (a - x)))
	print(min(ans, ans1))
