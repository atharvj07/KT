#include <bits/stdc++.h>
using namespace std;
const int mm = 22;
int f[mm][mm][mm][2];
int main() {
  memset(f, 0, sizeof(f));
  for (int i = 1; i < 5; i++) f[1][1][i][0] = 1;
  for (int k = 2; k <= 20; k++)
    for (int l = 1; l <= 10; l++)
      for (int i = 1; i <= 4; i++) {
        for (int j = 1; j < i; j++)
          f[k][l][i][0] += f[k - 1][l - 1][j][1] + f[k - 1][l][j][0];
        if (k == 2)
          f[k][l][i][1] = 0;
        else
          for (int j = i + 1; j < 5; j++)
            f[k][l][i][1] += f[k - 1][l][j][1] + f[k - 1][l][j][0];
      }
  int n, t;
  int ans;
  while (cin >> n >> t) {
    ans = 0;
    for (int i = 1; i <= 4; i++) ans += f[n][t][i][1];
    cout << ans << "\n";
  }
}
