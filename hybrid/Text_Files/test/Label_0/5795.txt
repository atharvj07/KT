#include <bits/stdc++.h>
using namespace std;
void _main(int tc) {
  int n, m;
  cin >> n >> m;
  deque<pair<int, int> > q;
  for (int i = 0; ((0 < n ? 1 : -1) < 0) ? i > n : i < n;
       i += (0 < n ? 1 : -1)) {
    int x;
    cin >> x;
    q.push_back({x, 0});
  }
  set<int> vis;
  for (int i = 0; ((0 < int(q.size()) ? 1 : -1) < 0) ? i > int(q.size())
                                                     : i < int(q.size());
       i += (0 < int(q.size()) ? 1 : -1)) {
    vis.insert(q[i].first);
  }
  long long int sumAns = 0;
  vector<int> ans;
  while (int(q.size())) {
    auto u = q.front();
    q.pop_front();
    if (!vis.count(u.first + 1)) {
      vis.insert(u.first + 1);
      q.push_back({u.first + 1, u.second + 1});
      ans.push_back(u.first + 1);
      sumAns += u.second + 1;
    }
    if (int(ans.size()) == m) break;
    if (!vis.count(u.first - 1)) {
      vis.insert(u.first - 1);
      q.push_back({u.first - 1, u.second + 1});
      ans.push_back(u.first - 1);
      sumAns += u.second + 1;
    }
    if (int(ans.size()) == m) break;
  }
  cout << sumAns << '\n';
  for (int i = 0; ((0 < m ? 1 : -1) < 0) ? i > m : i < m;
       i += (0 < m ? 1 : -1)) {
    cout << ans[i] << " \n"[i + 1 == m];
  }
}
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0);
  _main(0), exit(0);
  int tc;
  cin >> tc;
  for (int i = 0; ((0 < tc ? 1 : -1) < 0) ? i > tc : i < tc;
       i += (0 < tc ? 1 : -1))
    _main(i + 1);
}
