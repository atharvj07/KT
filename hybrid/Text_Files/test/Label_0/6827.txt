#include <bits/stdc++.h>
using namespace std;
const int N = 101;
const long long P = 1000000007;
int i, j, n, m, x, y, L, R;
vector<int> g[N];
long long f[N][22][22], tmp[22][22], newtmp[22][22], ans;
void cal(int i, int j, int k, int p) {
  if (i == -1 && k == -1)
    L = -1;
  else if (i != -1 && k != -1)
    L = min(i, k);
  else if (i == -1)
    L = k;
  else
    L = i;
  R = max(j, p);
  if (L != -1 && L + R + 2 <= m) R = -1;
  L++;
  R++;
}
void dfs(int x, int fa) {
  int flag = 0, i, j, p, k;
  for (auto y : g[x])
    if (y != fa) dfs(y, x), flag = 1;
  if (!flag)
    f[x][1][0] = f[x][0][1] = 1;
  else {
    flag = 0;
    for (auto y : g[x])
      if (y != fa) {
        if (!flag) {
          flag = 1;
          for (i = 0; i <= m + 1; i++)
            for (j = 0; j <= m + 1; j++) tmp[i][j] = f[y][i][j];
        } else {
          memset(newtmp, 0, sizeof(newtmp));
          for (i = 0; i <= m + 1; i++)
            for (j = 0; j <= m + 1; j++) {
              if (!tmp[i][j]) continue;
              for (k = 0; k <= m + 1; k++)
                for (p = 0; p <= m + 1; p++) {
                  if (!f[y][k][p]) continue;
                  cal(i - 1, j - 1, k - 1, p - 1);
                  newtmp[L][R] = (newtmp[L][R] + tmp[i][j] * f[y][k][p]) % P;
                }
            }
          for (i = 0; i <= m + 1; i++)
            for (j = 0; j <= m + 1; j++) tmp[i][j] = newtmp[i][j];
        }
      }
    for (i = 0; i <= m + 1; i++)
      for (j = 0; j <= m; j++) {
        f[x][1][0] = (f[x][1][0] + tmp[i][j]) % P;
        if (i == 0 || i == m + 1)
          L = 0;
        else
          L = i + 1;
        if (j == 0) {
          if (L == 0)
            R = 1;
          else
            R = 0;
        } else
          R = j + 1;
        f[x][L][R] = (f[x][L][R] + tmp[i][j]) % P;
      }
  }
  return;
  cout << x << ":" << endl;
  for (i = 0; i <= m + 1; i++)
    for (j = 0; j <= m + 1; j++)
      if (f[x][i][j]) printf("%d,%d: %d\n", i - 1, j - 1, f[x][i][j]);
  cout << endl;
}
int main() {
  cin >> n >> m;
  if (m == 0) {
    cout << 1;
    return 0;
  }
  for (i = 1; i < n; i++) {
    cin >> x >> y;
    g[x].push_back(y);
    g[y].push_back(x);
  }
  dfs(1, 0);
  for (i = 0; i <= m + 1; i++) ans = (ans + f[1][i][0]) % P;
  cout << ans << endl;
}
