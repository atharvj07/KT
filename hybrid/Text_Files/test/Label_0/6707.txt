#include <bits/stdc++.h>
using namespace std;
int main() {
  int k;
  scanf("%d", &k);
  if (k < 10)
    printf("%d\n", k);
  else if (k < 190) {
    k -= 10;
    if (k % 2 == 1)
      printf("%d\n", (k >> 1) % 10);
    else
      printf("%d\n", (k >> 1) / 10 + 1);
  } else if (k < 2890) {
    k -= 190;
    if (k % 3 == 0)
      printf("%d\n", k / 3 / 100 + 1);
    else if (k % 3 == 1)
      printf("%d\n", k / 3 % 100 / 10);
    else
      printf("%d\n", k / 3 % 10);
  } else {
    k -= 2890;
    if (k % 4 == 0)
      printf("%d\n", (k >> 2) / 1000 + 1);
    else if (k % 4 == 1)
      printf("%d\n", (k >> 2) % 1000 / 100);
    else if (k % 4 == 2)
      printf("%d\n", (k >> 2) % 100 / 10);
    else
      printf("%d\n", (k >> 2) % 10);
  }
  return 0;
}
