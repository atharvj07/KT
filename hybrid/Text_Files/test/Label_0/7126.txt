#include <bits/stdc++.h>
using namespace std;
int n;
double p[100010];
double len, res;
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> p[i];
    res += p[i] * (2 * len + 1);
    len = p[i] * (len + 1);
  }
  cout << setprecision(9) << fixed << showpoint << res << endl;
  return 0;
}
