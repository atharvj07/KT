import sys
(n, m) = map(int, input().split())
cnt = 0
r = c = 500
r1 = c1 = 0
for i in range(n):
	s = input()
	for j in range(m):
		if s[j] == 'X':
			cnt += 1
			if i < r:
				r = i
			if i > r1:
				r1 = i
			if j < c:
				c = j
			if j > c1:
				c1 = j
if (r1 - r + 1) * (c1 - c + 1) == cnt:
	print('YES')
else:
	print('NO')
