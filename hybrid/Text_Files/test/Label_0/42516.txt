import sys
input = sys.stdin.readline
from collections import *
for _ in range(int(input())):
	s = input()[:-1]
	cnt = Counter(list(s))
	M = max(list(cnt.values()))
	if M % 2 == 0:
		ans = len(s) - (M - 1)
	else:
		ans = len(s) - M
	for a in range(10):
		for b in range(10):
			L = 0
			flag = True
			for si in s:
				if flag and int(si) == a:
					flag = False
					L += 1
				elif not flag and int(si) == b:
					flag = True
					L += 1
			ans = min(ans, len(s) - L // 2 * 2)
	print(ans)
