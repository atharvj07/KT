#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); ++i)
using namespace std;
using ll = long long;
using P = pair<int,int>;

int main() {
	string s,st;
	int q;
	cin >> s >> q;
	rep(i,q){
		int t;
		cin >> t;
		if(t == 1) swap(s,st);
		else{
			int f;
			char c;
			cin >> f >> c;
			if(f == 1){
				st.push_back(c);
			}else{
				s.push_back(c);
			}
		}
	}
	string ans;
	reverse(st.begin(),st.end());
	ans = st + s;
	cout << ans << endl;
}
