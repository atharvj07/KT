#include <bits/stdc++.h>
using namespace std;
#pragma comment(linker, "/stack:64000000")
const long long int MOD = 1000 * 1000 * 1000 + 7;
const long long int INF = MOD * 1000 * 1000;
const double EPS = 1e-9;
const long double PI = acos(-1.0);
int n;
int f(int a) { return a % n + 1; }
int main() {
  std::ios_base::sync_with_stdio(false);
  int t;
  cin >> t;
  while (t--) {
    int p;
    cin >> n >> p;
    for (int i = 0; i < n; ++i) {
      cout << f(i) << " " << f(i + 1) << endl;
      cout << f(i) << " " << f(i + 2) << endl;
    }
    for (int k = 3; p > 0; ++k) {
      for (int i = 0; p > 0 && i < n; ++i, --p) {
        cout << f(i) << " " << f(i + k) << endl;
      }
    }
  }
  return 0;
}
