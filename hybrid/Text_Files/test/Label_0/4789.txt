#include <bits/stdc++.h>
using namespace std;
const int N = 110;
const int mod = 1e9 + 7;
int gi() {
  int x = 0, o = 1;
  char ch = getchar();
  while (!isdigit(ch) && ch != '-') ch = getchar();
  if (ch == '-') o = -1, ch = getchar();
  while (isdigit(ch)) x = x * 10 + ch - '0', ch = getchar();
  return x * o;
}
int n, m, k, C[N][N], f[2][N][N][N];
void inc(int &a, int b) { a = a + b >= mod ? a + b - mod : a + b; }
int main() {
  cin >> n >> m >> k;
  for (int i = 0; i <= n; i++) {
    C[i][0] = 1;
    for (int j = 1; j <= i; j++)
      C[i][j] = min(C[i - 1][j] + C[i - 1][j - 1], k + 1);
  }
  for (int i = 1; i <= n; i++) f[1][i][i][1] = 1;
  int ans = 0;
  for (int i = 2; i <= m; i++) {
    int now = i & 1, lst = now ^ 1;
    memset(f[now], 0, sizeof(f[now]));
    for (int j = 1; j <= n; j++)
      for (int x = 1; x <= j; x++)
        for (int y = 1; y <= k; y++)
          if (f[lst][j][x][y])
            for (int z = x; j + z <= n && y * C[z - 1][x - 1] <= k; z++)
              inc(f[now][j + z][z - x][y * C[z - 1][x - 1]], f[lst][j][x][y]);
    for (int j = 1; j <= n; j++)
      for (int y = 1; y <= k; y++)
        ans = (ans + 1ll * (m - i + 1) * f[now][j][0][y]) % mod;
  }
  cout << ans;
  return 0;
}
