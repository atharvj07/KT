#include<bits/stdc++.h>
typedef long long int ll;
typedef unsigned long long int ull;
#define BIG_NUM 2000000000
#define HUGE_NUM 99999999999999999
//#define MOD 1000000007
#define EPS 0.000000001
using namespace std;



#define MOD 998244353
#define NUM 100005

ll uniformity[8*NUM],partial[8*NUM];
ll N = 1;
ll first_N,K,P;

void init(ll first_N){
	while(N < first_N)N *= 2;
}

void add(ll left,ll right,ll value,ll node_id,ll node_left,ll node_right){

	if(right <= node_left || left >= node_right)return;
	else if(left <= node_left && right >= node_right){
		uniformity[node_id] += value;
		uniformity[node_id] %= MOD;
	}else{
		partial[node_id] += (min(right,node_right)-max(left,node_left))*value;
		partial[node_id] %= MOD;
		add(left,right,value,2*node_id+1,node_left,(node_left+node_right)/2);
		add(left,right,value,2*node_id+2,(node_left+node_right)/2,node_right);
	}
}

ll getSum(ll left,ll right,ll node_id,ll node_left,ll node_right){
	if(right <= node_left || left >= node_right)return 0;
	else if(left <= node_left && right >= node_right){
		return ((node_right-node_left)*uniformity[node_id]+partial[node_id])%MOD;

	}else{

		ll sum = (min(right,node_right)-max(left,node_left))*uniformity[node_id];
		sum %= MOD;

		sum += getSum(left,right,2*node_id+1,node_left,(node_left+node_right)/2);
		sum %= MOD;

		sum += getSum(left,right,2*node_id+2,(node_left+node_right)/2,node_right);
		sum %= MOD;

		return sum;
	}
}

ll extgcd(ll a,ll b,ll &x,ll &y){
	ll d = a;
	if(b != 0){
		d = extgcd(b,a%b,y,x);
		y -= (a/b)*x;
	}else{
		x = 1;
		y = 0;
	}
	return d;
}

ll mod_inverse(ll a,ll m){
    ll x,y;
    extgcd(a,m,x,y);
    return (m+x%m)%m;
}

int main(){

	scanf("%lld %lld %lld",&first_N,&K,&P);

	init(K+first_N);

	for(ll i = 0; i <= 2*N-2; i++){
		uniformity[i] = 0;
		partial[i] = 0;
	}

	ll inverse_N = mod_inverse(first_N,MOD);
	ll mult = mod_inverse(100*first_N,MOD)*P;
	mult %= MOD;

	//クリア状態
	for(ll i = K; i <= K+first_N-1; i++){

		add(i,i+1,1,0,0,N);
	}

	//i枚ある状態からクリアする確率を求める
	for(ll i = K-1; i >= 0; i--){

		ll tmp = 0;

		//1発でクリア状態に行ける場合
		if(i+first_N >= K){

			tmp += getSum(K,i+first_N+1,0,0,N);
			tmp *= inverse_N;
			tmp %= MOD;
		}

		ll tmp2 = 0;

		//クリア状態ではない状態に行ける場合
		if(i+1 <= K-1){

			tmp2 += getSum(i+1,min(K-1,i+first_N)+1,0,0,N);
			tmp2 *= mult;
			tmp2 %= MOD;
		}
		tmp += tmp2;
		tmp %= MOD;
		add(i,i+1,tmp,0,0,N);
	}

	printf("%lld\n",getSum(0,1,0,0,N)); //★注意:開区間★

	return 0;
}
