#include <bits/stdc++.h>
using namespace std;
int32_t main() {
  long long n, d;
  cin >> n >> d;
  long long m;
  cin >> m;
  while (m--) {
    long long x, y;
    cin >> x >> y;
    bool f = 1;
    long long a = x + y;
    long long b = x - y;
    if (a < d || a > (2 * n - d)) f = 0;
    if (b > d || b < -d) f = 0;
    if (!f)
      cout << "NO" << endl;
    else
      cout << "YES" << endl;
  }
  return 0;
}
