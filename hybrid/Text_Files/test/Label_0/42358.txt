import sys
n = int(sys.stdin.readline())
arr = list(map(int, sys.stdin.readline().split()))
left = [0 for _ in range(n)]
right = [0 for _ in range(n)]
left[0] = 1
right[-1] = 1
if n == 1:
	print(1)
	print('L')
else:
	for i in range(1, n):
		if arr[i - 1] < arr[i]:
			pass
		else:
			break
	count = 1
	for j in range(i - 1, -1, -1):
		left[j] = count
		count += 1
	for i in range(n - 2, -1, -1):
		if arr[i + 1] < arr[i]:
			pass
		else:
			break
	count = 1
	for j in range(i + 1, n):
		right[j] = count
		count += 1
	ans = [0 for _ in range(n)]
	(l, r) = (0, n - 1)
	i = 0
	a = [0 for _ in range(n)]
	z = True
	lim = 0
	last = -float('inf')
	while l <= r and z:
		if arr[l] < arr[r]:
			if arr[l] > last and left[l] > 0:
				last = arr[l]
				ans[i] = arr[l]
				a[i] = 'L'
				l += 1
				lim += 1
				i += 1
				continue
			if arr[r] > last and right[r] > 0:
				last = arr[r]
				ans[i] = arr[r]
				a[i] = 'R'
				r -= 1
				lim += 1
				i += 1
				continue
			z = False
		if arr[r] < arr[l]:
			if arr[r] > last and right[r] > 0:
				last = arr[r]
				ans[i] = arr[r]
				a[i] = 'R'
				r -= 1
				lim += 1
				i += 1
				continue
			if arr[l] > last and left[l] > 0:
				last = arr[l]
				a[i] = 'L'
				ans[i] = arr[l]
				l += 1
				lim += 1
				i += 1
				continue
		z = False
	if arr[l] == arr[r]:
		if left[l] > right[r]:
			c = left[l]
			while c > 0:
				ans[i] = arr[l]
				l += 1
				a[i] = 'L'
				lim += 1
				i += 1
				c -= 1
		else:
			c = right[r]
			while c > 0:
				ans[i] = arr[r]
				r -= 1
				lim += 1
				a[i] = 'R'
				i += 1
				c -= 1
	print(lim)
	print(''.join((x for x in a[:lim])))
