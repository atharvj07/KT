using ll = long long;

#define rrep(i, n) for (int i = 1; i <= (n); ++i)

template <class T, class U>
ostream &operator<<(ostream &os, const pair<T, U> &p) {
  os << "(" << p.first << "," << p.second << ")";
  return os;
}

template <class T> ostream &operator<<(ostream &os, const vector<T> &v) {
  os << "{";
  rep(i, (int)v.size()) {
    if (i)
      os << ",";
    os << v[i];
  }
  os << "}";
  return os;
}

template <typename T, size_t S> void printArray(const T (&array)[S]) {
  for (auto val : array)
    std::cout << val << ", ";
  std::cout << "\n";
}

const int mod = 1e9 + 7;
const int inf = 1e9 + 5;

void solve(int n, int m) {
  vi vec(n);
  rep(i, m) rep(j, n) {
    int a; std::cin >> a;
    vec[j] += a;
  }
  int ans = *max_element(rng(vec));
  std::cout << ans << "\n";
}

int main() {
  cin.tie(0);
  ios::sync_with_stdio(false);
  cout << std::setprecision(10);

  while (1) {
    int n, m; std::cin >> n >> m;
    if (n == 0 && m == 0) break;
    solve(n, m);
  }
}
