max = min
n = int(input())
a = [int(x) for x in input().split()]
a = [100000000000] + a + [1000000000000]
f = max(a)
indi = []
for i in range(1, n + 1):
	if a[i] <= a[i + 1] and a[i] <= a[i - 1] and (a[i] == f):
		indi.append(i)
best = 10000000000000000000
for i in range(len(indi) - 1):
	best = max(best, indi[i + 1] - indi[i])
print(best)
