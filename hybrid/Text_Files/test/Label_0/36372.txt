
import java.util.ArrayList;
import java.util.HashMap;
import java.util.Scanner;
import java.util.TreeSet;

public class Eelephant {
	static final int EPS = (int) 1e10;

public static void main(String[] args) {
	Scanner sc   = new Scanner(System.in);
	int n = sc.nextInt();
	HashMap<Integer, Integer> front = new HashMap<>();
	HashMap<Integer, Integer> back = new HashMap<>();
	ArrayList<Integer> a = new ArrayList<>();
	
	for (int i = 0; i < n; i++) {
		int x = sc.nextInt();
		int y = sc.nextInt();
		if (front.get(x) == null) {
			front.put(x, 0);
			a.add(x);
			
		}
		if (front.get(y) == null) {
			front.put(y, 0);
			a.add(y);
			
		}
		front.put(x, front.get(x) + 1);
		if (back.get(y) == null) {
			a.add(y);
			back.put(y, 0);
		}
		
		if (back.get(x) == null) {
			a.add(x);
			back.put(x, 0);
		}
		if (x != y)
			back.put(y, back.get(y) + 1);
	}
	
	int min = EPS;
	for (int i = 0; i < a.size(); i++) {
		int x = a.get(i);
		if(front.get(x) >= (n+1)/2)
		{
			min = 0;
			break;
		}else{
			
			if( back.get(x) + front.get(x) >= (n + 1) / 2){
				int b = (n + 1) / 2 - front.get(x);
				if(b < min)
					min = b;
			}
			
		}
		
	}
	
	if(min == EPS)
		System.out.println(-1);
	else
		System.out.println(min);
	
	
	
	
	
	
}
}
