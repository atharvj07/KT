#include <bits/stdc++.h>
using namespace std;
int N;
long long arr[100010];
long long dp[100010];
int32_t main() {
  ios_base::sync_with_stdio(false);
  if (fopen("cf756b.in", "r")) {
    freopen("cf756b.in", "r", stdin);
    freopen("cf756b.out", "w", stdout);
  }
  cin >> N;
  arr[0] = -1000000001;
  dp[0] = 0;
  for (int i = 1; i <= N; i++) {
    cin >> arr[i];
  }
  for (int i = 1; i <= N; i++) {
    dp[i] = dp[i - 1] + 20;
    int idx90 = upper_bound(arr, arr + i + 1, arr[i] - 90) - arr;
    dp[i] = min(dp[i], dp[idx90 - 1] + 50);
    int idx1440 = upper_bound(arr, arr + i + 1, arr[i] - 1440) - arr;
    dp[i] = min(dp[i], dp[idx1440 - 1] + 120);
  }
  for (int i = 1; i <= N; i++) {
    cout << dp[i] - dp[i - 1] << '\n';
  }
  return 0;
}
