
var arr = readline().split(" ").map(v => +v.trim());


var l = arr[0];
var u = arr[1];


var cur = 1;

while (true)
{
	if (cur * 3 > u)
	{
		break;
	}
	cur *= 3;
}

var res = 0;

while (cur >= 1)
{
	var cur2 = 1;
	while (true)
	{
		if (cur2 * cur >= l)
		{
			++res; 
		}
		
		if (cur2 * 2 * cur > u)
		{
			break;
		}
		
		cur2 *= 2;
	}
	
	cur = Math.floor(cur / 3);
}

print(res);
