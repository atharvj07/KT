t = int(input())
for i in range(t):
	n = int(input())
	e = 0
	l = []
	if n == 1:
		print('NO')
	elif n == 2:
		print('YES')
	else:
		r = int(pow(n, 1 / 3)) + 1
		for k in range(1, r):
			l.append(int(pow(k, 3)))
		s = set()
		for i in range(0, len(l)):
			s.add(l[i])
			temp = n - l[i]
			if temp in s:
				print('YES')
				e = 1
				break
		if e == 0:
			print('NO')
