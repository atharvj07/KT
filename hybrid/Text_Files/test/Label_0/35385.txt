
import java.util.Arrays;
import java.util.Scanner;

public class EnergyExchange {
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for(int i=0;i<n;i++){
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        System.out.println(solve(a, 0, n-1, k));
    }
    static double solve(int[] a, int l, int r, double k){
        if (l<=r){
            int m = l + (r-l)/2;
            double in=0,out=0;
            for(int i=0;i<= m;i++){
                in+=((double)(a[m]-a[i]))/(1 - k/100);
            }
            for(int i=m+1;i<a.length;i++){
                out+=a[i] - a[m];
            }
            if(in < out){
                return solve(a, m+1, r, k);
            }
            else if (in > out){
                return solve(a, l, m-1,k);
            } else {
                return a[m];
            }
        }else {
            double in=0, out = 0;
            for(int i=0;i<=r;i++){
                in+=((double) a[i])/(1 - k/100);
            }
            for(int i=l;i<a.length;i++){
                out+=a[i];
            }
            double sum = in + out;
            return sum/(((double)(r+1))/(1 - k/100)+ (a.length-r-1));
        }

    }
}
