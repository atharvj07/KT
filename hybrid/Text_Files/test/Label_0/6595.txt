#include <bits/stdc++.h>
using namespace std;
int n, m, k, h, u, v, p;
bool hp[1000001];
int main() {
  scanf("%d%d%d", &n, &m, &k);
  for (int i = 0; i < m; i++) {
    scanf("%d", &h);
    hp[h] = 1;
  }
  p = 1;
  for (int i = 0; i < k; i++) {
    scanf("%d%d", &u, &v);
    if (hp[p]) break;
    if (p == u)
      p = v;
    else if (p == v)
      p = u;
  }
  cout << p;
  return 0;
}
