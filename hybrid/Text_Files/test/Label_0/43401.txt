n = int(input())
d = {'z': 0, 'e': 0, 'r': 0, 'o': 0, 'n': 0}
s = input()
for i in range(n):
	d[s[i]] += 1
seq = []
for i in range(n):
	if d['o'] >= 1 and d['n'] >= 1 and (d['e'] >= 1):
		seq.append(1)
		d['o'] -= 1
		d['n'] -= 1
		d['e'] -= 1
	if d['o'] == 0 and d['n'] == 0 and (d['e'] == 0):
		print(' '.join((str(x) for x in sorted(seq, reverse=True))))
		break
if d['o'] >= 1:
	for i in range(n):
		if d['z'] >= 1 and d['e'] >= 1 and (d['r'] >= 1) and (d['o'] >= 1):
			seq.append(0)
			d['z'] -= 1
			d['e'] -= 1
			d['r'] -= 1
			d['o'] -= 1
	print(' '.join((str(x) for x in sorted(seq, reverse=True))))
