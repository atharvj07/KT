n = int(input())
s = list(input())
place = [0, 0]
(underneath, above) = (0, 0)
total = 0
for x in s:
	if x == 'R':
		place[0] += 1
	else:
		place[1] += 1
	if underneath and place[1] < place[0]:
		total += 1
	elif above and place[0] < place[1]:
		total += 1
	if place[0] < place[1]:
		underneath = 1
		above = 0
	elif place[0] > place[1]:
		above = 1
		underneath = 0
print(total)
