#include <bits/stdc++.h>
#define REP(i,n) for(int i=0; i<(int)(n); ++i)

using namespace std;
int main(){
    int n;
    int toint[256];
    for(int i = 0; i < 10; i++){
        toint[i + '0'] = i;
    }
    for(int i = 0; i < 26; i++){
        toint[i + 'A'] = 10 + i;
    }
    for(int i = 0; i < 26; i++){
        toint[i + 'a'] = 36 + i;
    }
    scanf("%d", &n);
    const int BUF_SIZE = 4194304;
    char buf[BUF_SIZE];
    fread(buf, sizeof(char), BUF_SIZE, stdin);
    int sum = 0;
    int idx = 0;
    for(int i = 0; i < n; i++){
        int p = 0;
        for(; buf[idx] != ' '; idx++){
            p = p * 10 + buf[idx] - '0';
        }
        idx++;
        int x = 0;
        for(; buf[idx] != '\n'; idx++){
            x = x * p + toint[buf[idx]];
        }
        idx++;
        sum = sum ^ x;
    }
    if(sum == 0){
        cout << "lose" << endl;
    }else{
        cout << "win" << endl;
    }
    return 0;
}
