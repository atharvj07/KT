#include <bits/stdc++.h>
#define che
#define debug(...) fprintf(stderr,__VA_ARGS__)
using namespace std;

typedef long long LL;

const int maxn= 500000 + 10;

char s[maxn];
int f[maxn], zer[maxn], N;

int main(){
	scanf("%d", &N);scanf("%s", s+1);
	for (int i= 1, lst=0; i<=N; i++){
		zer[i] = lst;
		if ( s[i]=='0') lst= i;
	}
	f[0] = 0;
	for (int i=1; i<=N; i++){
		f[i] = f[i-1];
		if ( s[i]=='0') continue;	else
		if ( !zer[i] || s[ zer[i]-1] !='1' ) continue;	else{
			f[i] = max( f[i]  , f[zer[i]-2]+ i-zer[i]);
			if ( zer[i]==i-1){
				int zzr = zer[ zer[i]];
				f[i] =max(f[i], f[zzr] + i-zzr-2);
				if ( zzr+1<zer[i]) f[i] = max(f[i], f[zzr+1]+i-zzr-3);
			}
		}
	}
	printf("%d\n", f[N]);
	return 0;
}
