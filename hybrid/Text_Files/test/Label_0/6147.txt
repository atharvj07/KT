#include <bits/stdc++.h>
using namespace std;
long long a[100005], b[100005], sa, sb;
int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    sa += a[i];
  }
  for (int i = 0; i < m; i++) cin >> b[i];
  sort(a, a + n);
  sort(b, b + m);
  long long ans = LLONG_MAX;
  long long j = m - 1;
  for (int i = n - 1; i >= 0; i--) {
    sa -= a[i];
    while (j >= 0 && b[j] > a[i]) {
      sb += b[j];
      j--;
    }
    long long tmp = (a[i] * i) - sa + sb - (a[i] * (m - j - 1));
    ans = min(ans, tmp);
  }
  sa = 0, sb = 0;
  for (int i = 0; i < m; i++) sb += b[i];
  j = 0;
  for (int i = 0, p = m - 1; i < m; i++, p--) {
    sb -= b[i];
    while (j < n && a[j] < b[i]) {
      sa += a[j];
      j++;
    }
    long long tmp = (b[i] * j) - sa + sb - (b[i] * p);
    ans = min(ans, tmp);
  }
  cout << ans;
}
