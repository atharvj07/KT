#include <bits/stdc++.h>
using namespace std;
std::vector<long long int> v;
long long int val;
void countfacts(long long int n) {
  v.clear();
  long long int count = 0;
  long long int k = 2;
  long long int num = n;
  while (n > 1) {
    if (n % k == 0) {
      n = n / k;
      v.push_back(k);
      count++;
    } else
      k++;
  }
}
int main() {
  long long int n, k;
  cin >> n >> k;
  countfacts(n);
  if (k > v.size()) {
    cout << -1 << endl;
  } else {
    long long int i = 0;
    long long int mul = 1;
    while (i < k - 1) {
      mul = mul * v[i];
      cout << v[i] << " ";
      i++;
    }
    cout << n / mul << endl;
  }
  return 0;
}
