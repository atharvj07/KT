#include <bits/stdc++.h>
using namespace std;
const int N = 1e2 + 5;
const long long mod = 1e9 + 7;
long long powermodm(long long x, long long n, long long M) {
  long long result = 1;
  if (x == 0) return result;
  while (n > 0) {
    if (n % 2 == 1) result = (result * x) % M;
    x = (x * x) % M;
    n = n / 2;
  }
  return result;
}
long long power(long long x, long long y) {
  long long z = 1;
  while (y) {
    if (y % 2 == 1) z = (z * x);
    y /= 2;
    x = (x * x);
  }
  return z;
}
long long gcd(long long a, long long b) {
  if (a == 0) return b;
  return gcd(b % a, a);
}
long long lcm(long long a, long long b) {
  return (max(a, b) / gcd(a, b)) * min(a, b);
}
int main() {
  long long n;
  cin >> n;
  long long a[n];
  for (long long i = 0; i < n; i++) cin >> a[i];
  queue<long long> q2, q3;
  vector<pair<long long, long long>> ans;
  long long cnt = 0;
  for (long long i = 0; i < n; i++) {
    if (a[i] == 0) continue;
    if (a[i] == 1) {
      if (!q2.empty()) {
        long long x = q2.front();
        ans.push_back(make_pair(x - 1, i));
        q2.pop();
      } else if (!q3.empty()) {
        long long x = q3.front();
        ans.push_back(make_pair(x - 1, i));
        ans.push_back(make_pair(cnt++, i));
        q3.pop();
      } else {
        ans.push_back(make_pair(cnt++, i));
      }
    } else {
      if (!q3.empty()) {
        long long x = q3.front();
        ans.push_back(make_pair(x - 1, i));
        q3.pop();
      }
      ans.push_back(make_pair(cnt++, i));
      if (a[i] == 2)
        q2.push(cnt);
      else
        q3.push(cnt);
    }
  }
  if (!q2.empty() || !q3.empty()) {
    cout << -1;
    return 0;
  }
  cout << ans.size() << endl;
  for (long long i = 0; i < ans.size(); i++) {
    cout << ans[i].first + 1 << " " << ans[i].second + 1 << endl;
  }
  return 0;
}
