n = int(input())
a = list(map(int, input().split()))
a.sort()
if len(a) % 2 != 0:
	list1 = a[1:len(a) // 2 + 1]
	list2 = a[len(a) // 2 + 1:len(a)]
	res = [[] for i in range(n)]
	res[0] = a[0]
	k1 = 0
	k2 = len(list2) - 1
if len(a) % 2 == 0:
	list1 = a[0:len(a) // 2]
	list2 = a[len(a) // 2:len(a)]
	res = [[] for i in range(n)]
	res[0] = a[0]
	k1 = 1
	k2 = len(list2) - 1
for i in range(2, len(a) + 1):
	if i % 2 != 0:
		res[i - 1] = list1[k1]
		k1 = k1 + 1
	if i % 2 == 0:
		res[i - 1] = list2[k2]
		k2 = k2 - 1
print(*res)
