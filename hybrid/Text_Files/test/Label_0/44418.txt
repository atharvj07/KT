import sys
import math
from collections import defaultdict, Counter, deque
n = int(input())
s = list(input())
colour = [1] * n
dp = [0] * 26
for i in range(n):
	for j in range(25, ord(s[i]) - 97, -1):
		colour[i] = max(colour[i], dp[j] + 1)
	dp[ord(s[i]) - 97] = max(dp[ord(s[i]) - 97], colour[i])
print(max(colour))
print(*colour)
