def get_new_str(s, n):
	ns = ''
	for i in range(n):
		if i > 0 and s[i] == 'A' and (s[i - 1] == 'A'):
			continue
		else:
			ns = ns + s[i]
	return ns
import sys
t = int(sys.stdin.readline().strip())
for _ in range(t):
	n = int(sys.stdin.readline().strip())
	s = sys.stdin.readline().strip()
	new_str = get_new_str(s, n)
	chunks = new_str.split('A')[1:]
	if chunks:
		len_chunks = [len(i) for i in chunks]
		print(max(len_chunks))
	else:
		print(0)
