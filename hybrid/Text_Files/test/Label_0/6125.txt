#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n;
  cin >> n;
  int u = 0;
  if (n == 2) {
    cout << 1;
    return 0;
  }
  if (n % 2 == 0) {
    cout << 2;
    return 0;
  }
  for (int i = 2; i * i <= n; i++) {
    if (n % i == 0) {
      u++;
    }
  }
  if (u == 0) {
    cout << 1;
    return 0;
  }
  n -= 2;
  u = 0;
  for (int i = 2; i * i <= n; i++) {
    if (n % i == 0) {
      u++;
    }
  }
  if (u == 0) {
    cout << 2;
    return 0;
  }
  cout << 3;
  return 0;
}
