#define MOD 1000000007LL
using namespace std;
typedef long long ll;
typedef pair<int,int> P;

int n;
int k[20001],bit[20001];

int dp[1<<22];
set<int> cnt[1<<22];

int main(void){
	scanf("%d",&n);
	for(int i=0;i<n;i++){
		scanf("%d",&k[i]);
		for(int j=0;j<k[i];j++){
			int a=0;
			scanf("%d",&a);
			a--;
			bit[i]+=(1<<a);
		}
		dp[bit[i]]=bit[i];
		if(cnt[bit[i]].size()<3)cnt[bit[i]].insert(i);
	}
	int res=0;
	for(int i=(1<<22)-1;i>=0;i--){
		for(int j=0;j<22;j++){
			if(!((i>>j) & 1)){
				dp[i]|=dp[i+(1<<j)];
				set<int>::iterator it=cnt[i+(1<<j)].begin();
				while(cnt[i].size()<3 && it!=cnt[i+(1<<j)].end()){
					cnt[i].insert(*it);
					it++;
				}
			}
		}
		if(cnt[i].size()==3){
			int v=__builtin_popcount(i)*__builtin_popcount(dp[i]);
			res=max(res,v);
		}
	}
	printf("%d\n",res);
	return 0;
}
