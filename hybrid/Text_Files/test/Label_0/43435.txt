n = int(input())
s = input()
c = {x: n // 3 - s.count(x) for x in ['2', '1', '0']}
if c['0'] < 0:
	if c['2'] > 0:
		r = min(-c['0'], c['2'])
		c['0'] += r
		c['2'] -= r
		s = s[::-1].replace('0', '2', r)[::-1]
	if c['1'] > 0:
		r = min(-c['0'], c['1'])
		c['0'] += r
		c['1'] -= r
		s = s[::-1].replace('0', '1', r)[::-1]
if c['2'] < 0:
	if c['0'] > 0:
		r = min(c['0'], -c['2'])
		c['0'] -= r
		c['2'] += r
		s = s.replace('2', '0', r)
	if c['1'] > 0:
		r = min(c['1'], -c['2'])
		c['1'] -= r
		c['2'] += r
		s = s.replace('2', '1', r)
if c['1'] < 0:
	if c['0'] > 0:
		r = min(-c['1'], c['0'])
		c['1'] += r
		c['0'] -= r
		s = s.replace('1', '0', r)
	if c['2'] > 0:
		r = min(-c['1'], c['2'])
		c['1'] += r
		c['2'] -= r
		s = s[::-1].replace('1', '2', r)[::-1]
print(s)
