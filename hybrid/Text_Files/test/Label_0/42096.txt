from collections import defaultdict
(X, MyDict) = (list(map(int, input().split())), defaultdict(int))
for i in range(X[0]):
	Temp = input()
	MyDict[Temp if MyDict.get(Temp[::-1]) == None else Temp[::-1]] += 1
(Answer, Middle) = ('', '')
for key in MyDict.keys():
	Answer += MyDict[key] // 2 * key
	MyDict[key] %= 2
	if MyDict[key] == 1 and key[::-1] == key and (len(key) > len(Middle)):
		Middle = key
Answer = Answer + Middle + Answer[::-1]
print(len(Answer))
print(Answer)
