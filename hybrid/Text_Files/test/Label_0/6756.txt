#include <bits/stdc++.h>
using namespace std;
int n, a[200001];
set<pair<int, int>> s;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  for (int i = 1; i <= n; i++) {
    int t = 1 << ((int)log2(a[i]) + 1), tot = 0;
    for (int j = i + 1; j <= n - 1; j++) {
      tot += a[j];
      if (tot > t) break;
      if (tot == (a[i] ^ a[j + 1])) s.insert(pair<int, int>(i, j + 1));
    }
  }
  for (int i = n; i >= 1; --i) {
    int t = 1 << ((int)log2(a[i]) + 1), tot = 0;
    for (int j = i - 1; j > 1; --j) {
      tot += a[j];
      if (tot > t) break;
      if (tot == (a[i] ^ a[j - 1])) s.insert(pair<int, int>(j - 1, i));
    }
  }
  cout << s.size();
  return 0;
}
