import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.io.PrintWriter;
import java.util.TreeMap;
import java.util.TreeSet;


public class Main {
    public static void main(String[] args) throws NumberFormatException, IOException {
        BufferedReader r = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(new OutputStreamWriter(System.out));
        
        N = Integer.parseInt(r.readLine());
        
        F = new TreeMap<Integer, Integer>();
        B = new TreeMap<Integer, Integer>();
        S = new TreeSet<Integer>();
        
        for(int i = 0; i < N; i++){
            String[] line = r.readLine().split("[ ]+");
            
            int f = Integer.parseInt(line[0]);
            int b = Integer.parseInt(line[1]);
            
            S.add(f);S.add(b);
            
            if(F.containsKey(f))F.put(f, F.get(f) + 1);
            else F.put(f, 1);
            
            if(f != b){
                if(B.containsKey(b))B.put(b, B.get(b) + 1);
                else B.put(b, 1);
            }
        }
        
        int h = (N + 1) / 2;
        int min = N + 1;
        for(int c : S){
            int f = F.containsKey(c) ? F.get(c) : 0;
            int b = B.containsKey(c) ? B.get(c) : 0;
            
            if(f + b < h)continue;
            else min = Math.min(min, Math.max(0, h - f));
        }
        if(min == N + 1)out.println(-1);
        else out.println(min);
        out.close();
    }
    static TreeMap<Integer, Integer> F, B;
    static TreeSet<Integer> S;
    static int N;
}
