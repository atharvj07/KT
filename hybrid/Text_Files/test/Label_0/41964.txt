import sys
import math
from fractions import Fraction
from decimal import *
input = sys.stdin.readline
inp = input().strip()
arr = [int(x) for x in inp.split(' ')]
n = arr[0]
t = arr[1]
par = [0] + [i + 1 for i in range(n)]
sizes = [0] + [1 for i in range(n)]
extra = 0
while t > 0:
	inp = input().strip()
	arr = [int(x) for x in inp.split(' ')]
	x = arr[0]
	y = arr[1]
	px = x
	while par[px] != px:
		px = par[px]
	py = y
	while par[py] != py:
		py = par[py]
	if px == py:
		extra += 1
	else:
		par[py] = px
		sizes[px] += sizes[py]
		sizes[py] = -1
	cursizes = []
	for s in sizes[1:]:
		if s != -1:
			cursizes.append(s)
	cursizes.sort(key=lambda x: -x)
	i = 0
	for j in range(extra):
		cursizes[i + 1] += cursizes[i]
		i += 1
		if i == len(cursizes) - 1:
			break
	print(cursizes[i] - 1)
	t -= 1
