#include <bits/stdc++.h>
using namespace std;
const int mxn = 4444;
int n, c, r, x[mxn], y[mxn], d[mxn], f[mxn][mxn], dp[mxn][mxn], s[mxn][mxn], t;
vector<int> v[mxn];
inline int go(int l, int r) {
  if (l > r) return 0;
  if (~dp[l][r]) return dp[l][r];
  dp[l][r] = go(l + 1, r);
  for (int k = 0; k < v[l].size(); ++k)
    if (y[v[l][k]] < r and go(l, y[v[l][k]]) + go(y[v[l][k]], r) > dp[l][r])
      dp[l][r] = dp[l][y[v[l][k]]] + dp[y[v[l][k]]][r], s[l][r] = y[v[l][k]];
  return dp[l][r] += bool(f[l][r]);
}
void dfs(int l, int r) {
  if (l > r) return;
  if (s[l][r])
    dfs(l, s[l][r]), dfs(s[l][r], r);
  else
    dfs(l + 1, r);
  if (f[l][r]) printf("%d ", f[l][r]);
}
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) {
    scanf("%d%d", &c, &r);
    x[i] = c - r, y[i] = c + r;
    d[i] = x[i], d[i + n] = y[i];
  }
  sort(d, d + n * 2);
  t = unique(d, d + n * 2) - d;
  for (int i = 0; i < n; ++i) {
    x[i] = upper_bound(d, d + t, x[i]) - d,
    y[i] = upper_bound(d, d + t, y[i]) - d;
    v[x[i]].push_back(i);
    f[x[i]][y[i]] = i + 1;
  }
  memset(dp, -1, sizeof(dp));
  printf("%d\n", go(1, t));
  dfs(1, t);
  return 0;
}
