from math import ceil as ceil
(n, k, a, b) = [int(i) for i in input().strip().split()]
if (min(a, b) + 1) * k < max(a, b):
	print('NO')
else:
	(l, m) = ('G', 'B')
	if a > b:
		(l, m) = (m, l)
	s = ''
	(a, b) = (max(a, b), min(a, b))
	while a > 0 and b > 0:
		x = ceil(a / (b + 1))
		s = s + m * x
		a -= x
		if not a:
			break
		s = s + l
		b -= 1
	if a:
		s = s + m * a
	if b:
		s = s + l * b
	print(s)
