#include <bits/stdc++.h>
using namespace std;
const long maxn = 1e6 + 5;
struct Node {
  long low, high;
  vector<long> v;
};
array<long, maxn> a, b;
vector<long> pos[maxn + 1];
long n, k;
Node tree[maxn * 2];
void build(long root, long l, long r) {
  tree[root].low = l, tree[root].high = r;
  tree[root].v.resize(r - l + 1);
  if (l == r)
    tree[root].v[0] = b[l];
  else {
    long mid = (l + r) / 2, left = root * 2 + 1, right = root * 2 + 2;
    build(left, l, mid), build(right, mid + 1, r);
    merge(tree[left].v.begin(), tree[left].v.end(), tree[right].v.begin(),
          tree[right].v.end(), tree[root].v.begin());
  }
}
inline long f(vector<long> &v, long t) {
  return lower_bound(v.begin(), v.end(), t) - v.begin();
}
long query(long root, long l, long r, long t) {
  if (tree[root].low == l && tree[root].high == r) return f(tree[root].v, t);
  long left = root * 2 + 1, right = root * 2 + 2;
  if (r <= tree[left].high) return query(left, l, r, t);
  if (l >= tree[right].low) return query(right, l, r, t);
  return query(left, l, tree[left].high, t) +
         query(right, tree[right].low, r, t);
}
int main() {
  scanf("%ld%ld", &n, &k);
  for (long i = 0; i < n; i++) {
    scanf("%ld", &a[i]);
    if (pos[a[i]].size() >= k)
      b[i] = *(pos[a[i]].rbegin() + k - 1);
    else
      b[i] = -1;
    pos[a[i]].push_back(i);
  }
  build(0, 0, n - 1);
  long q;
  scanf("%ld", &q);
  long ans = 0;
  while (q--) {
    long x, y, l, r;
    scanf("%ld%ld", &x, &y);
    l = (x + ans) % n, r = (y + ans) % n;
    if (l > r) swap(l, r);
    printf("%ld\n", ans = query(0, l, r, l));
  }
  return 0;
}
