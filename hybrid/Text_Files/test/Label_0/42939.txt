d = 0
n = int(input())
p = [int(i) for i in input().split()]
q = [int(i) for i in input().split()]
a = []
for i in range(1, len(p)):
	a.append(p[i])
for j in range(1, len(q)):
	a.append(q[j])
b = set(a)
for i in range(1, n + 1):
	if i in b:
		d = d + 1
if d == n:
	print('I become the guy.')
else:
	print('Oh, my keyboard!')
