from fractions import gcd

def valid(x):
	if l <= x <= r:
		return True
	return False
(l, r, x, y) = map(int, input().split())
if y % x != 0:
	print(0)
	exit(0)
p = y // x
i = 1
ans = 0
while i * i <= p:
	if p % i == 0:
		f1 = i
		f2 = p // i
		if f1 != f2:
			if gcd(f1, f2) == 1 and valid(x * f1) and valid(x * f2):
				ans += 2
		elif gcd(f1, f2) == 1 and valid(x * f1) and valid(x * f2):
			ans += 1
	i += 1
print(ans)
