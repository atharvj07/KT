import java.util.Scanner;

public class C {

	static int[] heights;
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		
		int h = in.nextInt();
		heights = new int[h+1];
		
		boolean prev = false;
		
		boolean amb = false;
		int index = -1;
		
		for(int i = 0; i <=h; ++i) {
			heights[i] = in.nextInt();
			if(prev && heights[i] > 1) {
				index = i-1;
				amb = true;
			}
			
			if(heights[i] > 1 ) prev = true;
			else prev = false;
		}
		
		if(!amb) System.out.println("perfect");
		else {
			System.out.println("ambiguous");
			System.out.print(0);
			
			int counter  = 1;
			for(int i = 1; i <= h; ++i) {
				for(int j = 0; j < heights[i]; ++j) {
					System.out.print(" " + counter);
				}
				counter += heights[i];
			}
			System.out.println();
			
			System.out.print(0);
			counter = 1;
			
			for(int i = 1; i <= index; ++i) {
				for(int j = 0; j < heights[i]; ++j) {
					System.out.print(" " + counter);
				}
				counter += heights[i];
			}

			for(int j = 0; j < heights[index+1]-1; ++j) {
				System.out.print(" " + counter);
			}
			System.out.print(" " + (counter-1));
			counter+= heights[index+1];
			
			for(int i = index+2; i <= h; ++i) {
				for(int j = 0; j < heights[i]; ++j) {
					System.out.print(" " + counter);
				}
				counter += heights[i];
			}
			System.out.println();
			
		}
		
		
		
		in.close();
	}

}
