import math

def solve():
	a = int(input())
	l = [int(i) for i in input().split()]
	one = 0
	uv = 0
	two = 0
	for i in l:
		if i == 1:
			if one > two:
				one += 1
			else:
				two += 1
			uv += 1
		if i == 2:
			if one < two:
				one -= 1
			else:
				two -= 1
		if i == 3:
			if one >= 0:
				one += 1
				uv += 1
			elif two >= 0:
				two += 1
				uv += 1
			elif two < 0 and one < 0:
				if two < one:
					two -= 1
				else:
					one -= 1
	print(uv)

def main():
	for i in range(int(input())):
		solve()
main()
