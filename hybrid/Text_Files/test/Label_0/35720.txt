function trim(s) {
	return s.replace(/^\s+|\s+$/gm, '');
}

function tokenize(s) {
	return trim(s).split(/\s+/);
}

function tokenizeIntegers(s) {
	var tokens = tokenize(s);
	for (var i = 0; i < tokens.length; i += 1) {
		tokens[i] = parseInt(tokens[i], 10);
	};
	return tokens;
}

function main() {
	var data = tokenizeIntegers(readline()),
		n = data[0], serving = data[1],
		children = tokenizeIntegers(readline()),
		pos = 0, count = 0;
	while (true) {
		while (children[pos] <= 0) {
			pos = (pos+1)%n;
		}
		children[pos] -= serving;
		if (children[pos] <= 0) {
			if (++count == n) {
				print(pos+1);
				return;
			}
		}
		pos = (pos+1)%n;
	}
}

main();
