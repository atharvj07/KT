def ii():
	return int(input())

def si():
	return input()

def mi():
	return map(int, input().split())

def msi():
	return map(str, input().split())

def li():
	return list(mi())
t = ii()
for _ in range(t):
	(n, m) = mi()
	mat = []
	for z in range(n):
		mat.append(list(si()))
	(r, w) = ([], [])
	for i in range(n):
		for j in range(m):
			if mat[i][j] == 'R':
				r.append(i + j)
			if mat[i][j] == 'W':
				w.append(i + j)
	flag = 0
	if len(list(filter(lambda x: x % 2 == 0, r))) == 0 and len(list(filter(lambda x: x % 2 == 1, w))) == 0:
		flag = 1
	if len(list(filter(lambda x: x % 2 == 1, r))) == 0 and len(list(filter(lambda x: x % 2 == 0, w))) == 0:
		flag = 2
	if flag > 0:
		for i in range(n):
			for j in range(m):
				if mat[i][j] == '.':
					if (i + j) % 2 == 0:
						if flag == 1:
							mat[i][j] = 'W'
						else:
							mat[i][j] = 'R'
					elif flag == 1:
						mat[i][j] = 'R'
					else:
						mat[i][j] = 'W'
		print('YES')
		for row in mat:
			print(''.join(row))
	else:
		print('NO')
