#include <bits/stdc++.h>
using namespace std;
struct Laptop {
  int n;
  int p1, p2, p3;
  int t1, t2;
  int* time;
};
Laptop a;
int sum(Laptop a, int n) {
  int j = 0, s = 0;
  int* dif = new int[n];
  int* rest = new int[n - 1];
  for (int i = 0; i < n; i++) {
    dif[i] = (a.time[j + 1]) - (a.time[j]);
    s += dif[i] * a.p1;
    j += 2;
  }
  if (n > 1) {
    j -= 2;
    for (int i = 0; i < n - 1; i++) {
      rest[i] = a.time[j] - a.time[j - 1];
      if (rest[i] > a.t1) {
        s += a.t1 * a.p1;
        rest[i] -= a.t1;
        if (rest[i] > a.t2) {
          s += a.t2 * a.p2;
          rest[i] -= a.t2;
          s += rest[i] * a.p3;
        } else
          s += rest[i] * a.p2;
      } else
        s += rest[i] * a.p1;
      j -= 2;
    }
  }
  delete[] rest;
  delete[] dif;
  return s;
}
int main() {
  int n;
  cin >> n;
  a.time = new int[n * 2];
  cin >> a.p1 >> a.p2 >> a.p3;
  cin >> a.t1 >> a.t2;
  int j = 0;
  for (int i = 0; i < n; i++) {
    cin >> a.time[j] >> a.time[j + 1];
    j += 2;
  }
  cout << sum(a, n);
  delete[] a.time;
  return 0;
}
