bool cyclesize(long long int node, long long int neww) {
  int x = node;
  vector<int> ans;
  while (x != neww) {
    ans.push_back(x);
    x = parent[x];
  }
  ans.push_back(x);
  if (ans.size() > k) {
    cout << ans.size() << ("\n");
    for (int i : ans) cout << i << " ";
    exit(0);
  } else
    return false;
}
bool dfs(long long int node, long long int par) {
  vis[node] = 1;
  parent[node] = par;
  for (auto it : adj[node]) {
    if (vis[it] != 1) {
      if (dfs(it, node)) return true;
    } else {
      if (it != par && cyclesize(node, it)) {
        return true;
      }
    }
  }
  return false;
}

bool isPrime(long long int n) {
  if (n <= 1) return false;
  if (n <= 3) return true;
  if (n % 2 == 0 || n % 3 == 0) return false;
  for (long long int i = 5; i * i <= n; i = i + 6)
    if (n % i == 0 || n % (i + 2) == 0) return false;
  return true;
}
long long int pw(long long int x, long long int y) {
  long long int i = 0;
  long long int ans = 1;
  for (i = 1; i <= y; ++i) ans = (ans * x) % 998244353;
  return ans;
}
long long int fact(long long int n) {
  long long int ans = 1;
  for (long long int i = 1; i <= n; ++i) ans = (ans * i) % 1000000007;
  return ans;
}
