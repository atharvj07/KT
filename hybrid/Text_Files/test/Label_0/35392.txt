class TaskD {

    long MAX = (long) 1e15;

    public void solve(int testNumber, InputStreamReader inSt, PrintWriter out) {
        InputReader in = new InputReader(inSt);
        long n = in.nextLong();
        long m = in.nextLong();
        long k = in.nextLong();
        long[] a = new long[(int) k];
        for (int i = 0; i < k; i++) {
            a[i] = in.nextLong();
        }

        long lcm = 1;
        for (int i = 0; i < a.length; i++) {
            lcm = lcmM(lcm, a[i]);
            if (lcm == MAX) {
                out.println("NO");
                return;
            }
        }

        if (lcm > n) {
            out.println("NO");
            return;
        }

        long rPriv = 0;
        long mPriv = 1;
        long rs = -1;
        for (int i = 0; i < a.length; i++) {
            rs = solve(rPriv, mPriv, -i, a[i]);
            if (rs == -1) {
                out.println("NO");
                return;
            }

            rPriv = rs;
            mPriv = lcmM(mPriv, a[i]);
        }

        if (rs > m - k + 1) {
            out.println("NO");
            return;
        }

        for (int i = 0; i < k; i++) {
            long g = gcd(lcm, rs + i);
            if (g != a[i]) {
                out.println("NO");
                return;
            }
        }

        out.println("YES");

    }

    long lcmM(long a, long b) {
        long g = gcd(a, b);
        a /= g;

        return mult(a, b);
    }


    long solve(long r1, long m1, long r2, long m2) {
        Tuple rs = eGCD(m1, m2);

        if ((r2 - r1) % rs.g != 0) {
            return -1;
        }

        long df = (r2 - r1) / rs.g;

        long x0 = rs.x * df;
        long y0 = rs.y * df;

        long bg = m2 / rs.g;

        long k = -(x0 / bg);
        long resx = x0 + k * bg;
        if (resx < 0) {
            resx += bg;
        }

        long X = m1 * resx + r1;
        if (X == 0) {
            X = lcmM(m1, m2);
        }

        return X;
    }

    long mult(long a, long b) {
        if (MAX / a <= b) {
            return MAX;
        }

        return a * b;
    }


    class Tuple {
        long g;
        long x;
        long y;

        Tuple(long g, long x, long y) {
            this.g = g;
            this.x = x;
            this.y = y;
        }
    }

    Tuple eGCD(long a, long b) {
        if (a == 0) {
            return new Tuple(b, 0, 1);
        }

        Tuple tmp = eGCD(b % a, a);

        Tuple result = new Tuple(tmp.g, tmp.y - (b / a) * tmp.x, tmp.x);

        return result;
    }


    long gcd(long a, long b) {
        if (b == 0) {
            return a;
        }

        return gcd(b, a % b);
    }

    class InputReader {
        public BufferedReader reader;
        private String[] currentArray;
        int curPointer;

        public InputReader(InputStreamReader inputStreamReader) {
            reader = new BufferedReader(inputStreamReader);
        }

        public String next() {
            try {
                currentArray = null;
                return reader.readLine();
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
        }

        public void nextChars(char[] t) {
            try {
                currentArray = null;
                reader.read(t);
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
        }

        public char nextChar() {
            try {
                currentArray = null;
                return (char) reader.read();
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
        }

        public int nextInt() {
            if ((currentArray == null) || (curPointer >= currentArray.length)) {
                try {
                    currentArray = reader.readLine().split(" ");
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
                curPointer = 0;
            }
            return Integer.parseInt(currentArray[curPointer++]);
        }

        public long nextLong() {
            if ((currentArray == null) || (curPointer >= currentArray.length)) {
                try {
                    currentArray = reader.readLine().split(" ");
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
                curPointer = 0;
            }
            return Long.parseLong(currentArray[curPointer++]);
        }

    }
}
