#include<bits/stdc++.h>
#define reg register
#define ll long long
using namespace std;
multiset<string>s;
ll a,b,c;
int main()
{
	scanf("%lld%lld%lld",&a,&b,&c);
	for(reg int i=1;i<=a;i++)s.insert("a");
    for(reg int i=1;i<=b;i++)s.insert("b");
    for(reg int i=1;i<=c;i++)s.insert("c");
    while(s.size()>1)
	{
        string a=*s.begin(),b=*s.rbegin();
        s.erase(s.lower_bound(a));
        s.erase(s.lower_bound(b));
        s.insert(a+b);
    }
    cout<<(*s.begin()).c_str()<<endl;
    return 0;
}
