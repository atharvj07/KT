#include <bits/stdc++.h>
int main(void) {
  long long int i, j, h, acu1, acu2, acu3, cantidad = 0;
  long long int n, suma = 0, div, aux0 = 0;
  scanf("%I64d", &n);
  int numeros[n];
  for (i = 0; i < n; ++i) {
    scanf("%I64d", &numeros[i]);
    if (numeros[i] != 0) aux0 = 1;
    suma += numeros[i];
  }
  i = 0;
  acu1 = 0;
  acu2 = 0;
  acu3 = 0;
  div = suma / 3;
  if (suma == 0 && aux0 == 0 && n >= 3)
    cantidad = ((n - 1) * (n - 2)) / (long long int)2;
  else if (n >= 3 && suma % 3 == 0) {
    while (i < n - 2) {
      acu1 += numeros[i];
      ++i;
      j = i;
      if (acu1 == div) {
        while (j < n - 1) {
          acu2 += numeros[j];
          ++j;
          acu3 = suma - acu1 - acu2;
          if (acu1 == acu2 && acu2 == acu3) {
            ++cantidad;
          }
        }
      }
      acu2 = 0;
    }
  }
  printf("%I64d\n", cantidad);
}
