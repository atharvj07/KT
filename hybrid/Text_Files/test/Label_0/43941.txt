def input_num_list():
	s = input().split(' ')
	L = []
	for e in s:
		L.append(int(e))
	return L

def inti():
	return int(input())
T = inti()
for t in range(1, T + 1):
	(n, a, b, c, d) = input_num_list()
	grain_min = a - b
	grain_max = a + b
	pkg_min = c - d
	pkg_max = c + d
	pkg_total_min = n * grain_min
	pkg_total_max = n * grain_max
	if pkg_total_max < pkg_min or pkg_max < pkg_total_min:
		print('No')
	else:
		print('Yes')
