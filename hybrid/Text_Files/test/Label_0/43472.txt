(m, n) = map(int, input().split())
b = []
for i in range(m):
	b.append(list(map(int, input().split())))
a = [[1 for j in range(n)] for i in range(m)]
for i in range(m):
	for j in range(n):
		if b[i][j] == 0:
			for k in range(m):
				a[k][j] = 0
			for k in range(n):
				a[i][k] = 0
f = 1
temp = [[0 for j in range(n)] for i in range(m)]
for i in range(m):
	if 1 in a[i]:
		for j in range(n):
			temp[i][j] = 1
for i in range(n):
	if 1 in [a[j][i] for j in range(m)]:
		for j in range(m):
			temp[j][i] = 1
for i in range(m):
	for j in range(n):
		if b[i][j] != temp[i][j]:
			f = 0
			print('NO')
			break
	if f == 0:
		break
if f == 1:
	print('YES')
	for i in range(m):
		print(*a[i])
