#include <iostream>
#include <vector>
#include <algorithm>
#include <cstring>

using namespace std;

int n;
int coins[100];
int dp[51][10001];
const int INF=1000000000;

int dfs(int pos,int money){
	if(pos==n){
		if(money==0)return 0;
		return INF;
	}
	if(dp[pos][money]!=-1)return dp[pos][money];
	int res=INF;
	// ¥¦éÈçÎ¥¤
	if(money-coins[pos]>=0){
		int tmp=dfs(pos,money-coins[pos]);
		if(tmp!=INF)
		res=min(res,tmp+1);
	}
	// ¡ñÍ¥¤àzðÏ¦é¾¯
	int tmp=dfs(pos+1,money);
	if(tmp!=INF)res=min(res,tmp);
	return dp[pos][money]=res;
}

int main(){
	int cnt=0;
	while(cin>>n&&n){
		bool con=false;
		cnt++;
		cout<<"Case #"<<cnt<<": ";
		for(int i=0;i<n;i++)cin>>coins[i];
		sort(coins,coins+n);
		memset(dp,-1,sizeof(dp));
		for(int j=1;j<=10000;j++)dfs(0,j);
		for(int i=1;i<=10000;i++){
			if(dp[0][i]==INF){
				cout<<"Cannot pay some amount"<<endl;
				con=true;
				break;
			}
		}
		if(con)continue;
		// »ê¼êÌàzðgreedyÅ¥¤
		for(int i=1;i<=10000;i++){
			int leftm=i;
			int allc=0;
			for(int j=n-1;j>=0;j--){
				int cnt=leftm/coins[j];
				leftm-=cnt*coins[j];
				allc+=cnt;
			}
			if(dp[0][i]!=allc){
				int a=dp[0][i];
				cout<<"Cannot use greedy algorithm"<<endl;
				con=true;
				break;
			}
		}
		if(con)continue;
		cout<<"OK"<<endl;
	}

	return 0;
}
