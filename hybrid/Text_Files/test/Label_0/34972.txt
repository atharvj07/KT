import java.util.Arrays;
import java.util.Scanner;

public class ReplacingDigits {

    /**
     * @param args
     */
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        StringBuilder number = new StringBuilder(scan.next());

        char[] chrs = scan.next().toCharArray();
        Arrays.sort(chrs);

        int indx = 0;
        for (int ii = chrs.length - 1; ii >= 0; ii--) {
            int tmpIndx = indx;
            for (int i = indx; i < number.length(); i++) {
                if (chrs[ii] > number.charAt(i)) {
                    number.setCharAt(i, chrs[ii]);
                    indx = i;
                    break;
                }
            }
            if (tmpIndx == indx && ii!=chrs.length-1) {
                break;
            }
        }
        System.out.println(number);

    }
}
