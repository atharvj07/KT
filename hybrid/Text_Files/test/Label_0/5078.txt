#include <bits/stdc++.h>

const int INF = 1 << 20;
const int MOD = 1e9 + 7;
int main() {
  string s;
  cin >> s;
  string str = "aiueo";
  for (int i = 0; i < (s.size() - 1); i++) {
    if ((s[i] != 'a' && s[i] != 'i' && s[i] != 'u' && s[i] != 'e' &&
         s[i] != 'o')) {
      if (s[i] != 'n' &&
          (s[i + 1] != 'a' && s[i + 1] != 'i' && s[i + 1] != 'u' &&
           s[i + 1] != 'e' && s[i + 1] != 'o')) {
        cout << "NO" << endl;
        return 0;
      }
    }
  }
  int i = s.size() - 1;
  if ((s[i] != 'a' && s[i] != 'i' && s[i] != 'u' && s[i] != 'e' &&
       s[i] != 'o') &&
      s[i] != 'n') {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  return 0;
}
