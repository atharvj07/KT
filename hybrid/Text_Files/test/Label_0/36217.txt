import java.util.*;
public class RobotMove {

	public static void main(String[] args) {
		Scanner s= new Scanner(System.in);
		int t = s.nextInt();
		while(t-->0) {
			int a = s.nextInt();
			int b =  s.nextInt();
			int ans = 0;
			if(a==b) {
				ans = a+b;
			}
			else if(a>b) {
				int diagonal = a-b;
				int di = a-b;
				ans = 1 + 2*(di-1);
				ans+=2*b;
			}
			else {
				int diagonal = b-a;
				int di = b-a;
				ans = 1 + 2*(di-1);
				ans+=2*a;
			}
			System.out.println(ans);
		}
	}

}
