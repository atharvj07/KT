#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long int n;
  cin >> n;
  string s, s1;
  cin >> s >> s1;
  long long int i, cnt = 0;
  for (i = 0; i < s.size() - 1; i++) {
    if (s[i] != s1[i]) {
      if (s[i] != s[i + 1] && s[i + 1] != s1[i + 1]) {
        cnt++;
        i++;
      } else
        cnt++;
    }
  }
  if (s[i] != s1[i]) cnt++;
  cout << cnt;
  return 0;
}
