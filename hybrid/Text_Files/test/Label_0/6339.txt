#include <bits/stdc++.h>
using namespace std;
const int N = 200100, mod = 1e8;
inline int read() {
  int x = 0, c = getchar(), f = 0;
  for (; c > '9' || c < '0'; f = c == '-', c = getchar())
    ;
  for (; c >= '0' && c <= '9'; c = getchar()) x = (x << 1) + (x << 3) + c - '0';
  return f ? -x : x;
}
void write(long long x) {
  if (x > 9) write(x / 10);
  putchar(x % 10 + '0');
}
long long d[N];
int fa[N], a[N], ans[N], h[N], f[N][20], ne[N], to[N], L, w[N];
void add(int k, int x, int y) {
  ne[++L] = h[x];
  h[x] = L;
  to[L] = y;
  w[L] = k;
}
void dfs(int x, int y) {
  f[x][0] = y;
  y = x;
  int k;
  for (k = 0; f[x][k]; k++) f[x][k + 1] = f[f[x][k]][k];
  for (k--; k >= 0; k--)
    if (d[x] - d[f[y][k]] <= a[x]) y = f[y][k];
  ans[fa[x]]++;
  ans[fa[y]]--;
  for (k = h[x]; k; k = ne[k]) {
    y = to[k];
    d[y] = d[x] + w[k];
    dfs(y, x);
    ans[x] += ans[y];
  }
}
int main() {
  int i, n = read();
  for (i = 1; i <= n; i++) a[i] = read();
  for (i = 2; i <= n; i++) add(read(), fa[i] = read(), i);
  dfs(1, 0);
  for (i = 1; i <= n; i++) printf("%d ", ans[i]);
  return 0;
}
