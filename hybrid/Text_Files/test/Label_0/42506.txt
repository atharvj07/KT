(n, k) = map(int, input().split())
cities = input().split()
cap = input().split()
capitals = [False] * len(cities)
j = 0
sumBeauties = 0
sumRoads = 0
for i in range(len(cities)):
	cities[i] = int(cities[i])
	sumBeauties += cities[i]
	if j >= len(cap):
		continue
	if int(cap[j]) == i + 1:
		capitals[i] = True
		j += 1
for i in range(len(cities)):
	if capitals[i]:
		sumBeauties -= cities[i]
		sumRoads += cities[i] * sumBeauties
	elif not capitals[i - 1]:
		sumRoads += cities[i] * cities[i - 1]
print(sumRoads)
