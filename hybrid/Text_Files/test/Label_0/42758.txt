def diffchar(a, b):
	for i in range(26):
		c = chr(ord('a') + i)
		if c != a and c != b:
			return c
while True:
	try:
		(n, t) = map(int, input().split())
		(a, b) = (input() for _ in range(2))
	except:
		raise SystemExit
	same = [i for i in range(n) if a[i] == b[i]]
	diff = [i for i in range(n) if a[i] != b[i]]
	if 2 * t < len(diff):
		print(-1)
		continue
	(scnt, dcnt) = (n - t, t)
	res = ['.'] * n
	for si in same:
		if scnt == 0:
			break
		res[si] = a[si]
		scnt -= 1
	if scnt > 0:
		for (i, di) in enumerate(diff):
			if scnt == 0:
				break
			res[di] = a[di] if i % 2 == 0 else b[di]
			scnt -= 0.5
	for i in range(n):
		if res[i] == '.':
			res[i] = diffchar(a[i], b[i])
	print(''.join(res))
