#include <bits/stdc++.h>
using namespace std;
vector<vector<int>> g;
map<pair<int, int>, set<int>> nogo;
void bfs(int v, int n) {
  queue<pair<int, int>> que;
  set<int> empty_set;
  que.push(make_pair(-1, 0));
  map<pair<int, int>, pair<int, int>> paths_taken;
  paths_taken[make_pair(0, -1)] = make_pair(-1, -1);
  while (que.size()) {
    int curr = que.front().second;
    int par = que.front().first;
    que.pop();
    0;
    if (curr == n - 1) {
      vector<int> res;
      for (pair<int, int> t = make_pair(curr, par); t.first != -1;
           t = paths_taken[t]) {
        0;
        res.push_back(t.first);
      }
      int dist = res.size();
      cout << dist - 1 << endl;
      for (int i = dist - 1; i >= 0; --i) {
        cout << res[i] + 1 << " ";
      }
      cout << endl;
      return;
    }
    set<int>* nexts;
    if (nogo.count(make_pair(par, curr))) {
      nexts = &nogo[make_pair(par, curr)];
    } else {
      nexts = &empty_set;
    }
    for (auto to : g[curr]) {
      if (nexts->count(to)) continue;
      if (paths_taken.count(make_pair(to, curr))) {
        continue;
      }
      paths_taken[make_pair(to, curr)] = make_pair(curr, par);
      que.push(make_pair(curr, to));
      0;
    }
  }
  cout << -1 << endl;
}
void solve() {
  int n, m, k;
  cin >> n >> m >> k;
  g.resize(n);
  for (int i = 0; i < m; ++i) {
    int t1, t2;
    cin >> t1 >> t2;
    t1--;
    t2--;
    g[t1].push_back(t2);
    g[t2].push_back(t1);
  }
  for (int i = 0; i < k; ++i) {
    int a, b, c;
    cin >> a >> b >> c;
    a--;
    b--;
    c--;
    nogo[make_pair(a, b)].insert(c);
  }
  bfs(0, n);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t = 1;
  for (int i = 1; i <= t; ++i) {
    solve();
  }
  return 0;
}
