import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;



public class hals {
	
	public static void main(String[] args)throws Exception {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		String s = br.readLine();
		int m = Integer.parseInt(br.readLine());
		long x1=0,x2=0;
		int cnt = 0,last=-1;
		int[] arr = new int[m];
		boolean flag=false;
		for(int w=0;w<10;w++){
			if(s.charAt(w)=='0')continue;
			x1=0;
			x2=0;
			cnt=1;
			arr[0]=w+1;
			last=w+1;
			x1+=w+1;
			flag=true;
			while(cnt<m){
				flag=false;
				for(int i=0;i<10;i++){
					if(s.charAt(i)=='1'){
						if(cnt%2==0){
							if(x1+i+1>x2 && (i+1)!=last){
								x1+=i+1;
								last=i+1;
								flag=true;
								arr[cnt++]=i+1;
								break;
							}
						}
						else{
							if(x2+i+1>x1 && (i+1)!=last){
								x2+=i+1;
								last=i+1;
								flag=true;
								arr[cnt++]=i+1;
								break;
							}
						}
					}
				}
				if(!flag){
					break;
				}
			}
			if(flag)
				break;
		}
		if(!flag)
			System.out.println("NO");
		else{
			System.out.println("YES");
			System.out.print(arr[0]);
			for(int i=1;i<m;i++)
				System.out.print(" "+arr[i]);
			System.out.println();
		}
	}
}
