(n, m) = map(int, input().split())
l = []
for i in range(n):
	l.append(input())
x = [[0 for i in range(m)] for i in range(n)]
y = [[0 for i in range(m)] for i in range(n)]
for i in range(n):
	for j in range(1, m):
		if l[i][j] == '#':
			x[i][j] = x[i][j - 1]
		elif l[i][j - 1] == '.':
			x[i][j] = 1 + x[i][j - 1]
		else:
			x[i][j] = x[i][j - 1]
for i in range(m):
	for j in range(1, n):
		if l[j][i] == '#':
			y[j][i] = y[j - 1][i]
		elif l[j - 1][i] == '.':
			y[j][i] = 1 + y[j - 1][i]
		else:
			y[j][i] = y[j - 1][i]
mod_x = [[0 for i in range(m)] for i in range(n)]
mod_y = [[0 for i in range(m)] for i in range(n)]
for i in range(n):
	for j in range(m):
		if j == 0:
			mod_y[i][j] = y[i][j]
		else:
			mod_y[i][j] += mod_y[i][j - 1] + y[i][j]
for i in range(m):
	for j in range(n):
		if j == 0:
			mod_x[j][i] = x[j][i]
		else:
			mod_x[j][i] += mod_x[j - 1][i] + x[j][i]
q = int(input())
for i in range(q):
	(a, b, c, d) = map(int, input().split())
	a -= 1
	b -= 1
	c -= 1
	d -= 1
	if a == 0:
		ans = mod_x[c][d] - mod_x[c][b]
	else:
		ans = mod_x[c][d] - mod_x[a - 1][d] - mod_x[c][b] + mod_x[a - 1][b]
	if b == 0:
		ans += mod_y[c][d] - mod_y[a][d]
	else:
		ans += mod_y[c][d] - mod_y[c][b - 1] - mod_y[a][d] + mod_y[a][b - 1]
	print(ans)
