#include<bits/stdc++.h>
typedef long long int ll;
typedef unsigned long long int ull;
#define BIG_NUM 2000000000
#define HUGE_NUM 99999999999999999
#define MOD 1000000007
#define EPS 0.000000001
using namespace std;



#define SIZE 605
struct Edge{

	int from,to;
	char buf[6];
	int len;
};

int N1,N2,M1,M2;
ll dp[SIZE][SIZE][6][6];
Edge edge1[SIZE],edge2[SIZE];
vector<int> G1[SIZE],G2[SIZE];

ll recursive(int e1,int loc1,int e2,int loc2){

	if(dp[e1][e2][loc1][loc2] != -1){

		return dp[e1][e2][loc1][loc2];
	}

	if(edge1[e1].buf[loc1] != edge2[e2].buf[loc2]){

		return dp[e1][e2][loc1][loc2] = 0;
	}

	ll ret = 0;

	if(loc1+1 < edge1[e1].len && loc2+1 < edge2[e2].len){ //両方非終端

		ret = recursive(e1,loc1+1,e2,loc2+1);

	}else if(loc1+1 == edge1[e1].len && loc2+1 < edge2[e2].len){ //DAG1が終端

		if(edge1[e1].to == N1-1){

			ret = 0;

		}else{ //DAG1が新しいノードへ

			for(int i = 0; i < G1[edge1[e1].to].size(); i++){

				ret += recursive(G1[edge1[e1].to][i],0,e2,loc2+1);
				ret %= MOD;
			}
		}

	}else if(loc1+1 < edge1[e1].len && loc2+1 == edge2[e2].len){ //DAG2が終端

		if(edge2[e2].to == N2-1){

			ret = 0;

		}else{ //DAG2が新しいノードへ

			for(int i = 0; i < G2[edge2[e2].to].size(); i++){

				ret += recursive(e1,loc1+1,G2[edge2[e2].to][i],0);
				ret %= MOD;
			}
		}

	}else{ //両方終端

		if((edge1[e1].to == N1-1 && edge2[e2].to != N2-1) ||
				(edge1[e1].to != N1-1 && edge2[e2].to == N2-1)){ //片方だけ文章完成

			ret = 0;

		}else if(edge1[e1].to == N1-1 && edge2[e2].to == N2-1){ //両方文章完成

			ret = 1;

		}else{ //両方新しいノードへ

			for(int i = 0; i < G1[edge1[e1].to].size(); i++){
				for(int k = 0; k < G2[edge2[e2].to].size(); k++){

					ret += recursive(G1[edge1[e1].to][i],0,G2[edge2[e2].to][k],0);
					ret %= MOD;
				}
			}
		}
	}

	return dp[e1][e2][loc1][loc2] = ret;
}


void func(){

	for(int i = 0; i < SIZE; i++){
		G1[i].clear();
		G2[i].clear();
	}

	scanf("%d %d",&N1,&M1);

	for(int i = 0; i < M1; i++){

		scanf("%d %d %s",&edge1[i].from,&edge1[i].to,edge1[i].buf);
		for(edge1[i].len = 0; edge1[i].buf[edge1[i].len] != '\0'; edge1[i].len++);

		G1[edge1[i].from].push_back(i);
	}

	scanf("%d %d",&N2,&M2);

	for(int i = 0; i < M2; i++){

		scanf("%d %d %s",&edge2[i].from,&edge2[i].to,edge2[i].buf);
		for(edge2[i].len = 0; edge2[i].buf[edge2[i].len] != '\0'; edge2[i].len++);

		G2[edge2[i].from].push_back(i);
	}

	for(int i = 0; i < M1; i++){
		for(int k = 0; k < M2; k++){
			for(int a = 0; a < 6; a++){
				for(int b = 0; b < 6; b++){
					dp[i][k][a][b] = -1;
				}
			}
		}
	}

	ll ans = 0;
	int root = 0;

	for(int i = 0; i < G1[root].size(); i++){
		for(int k = 0; k < G2[root].size(); k++){

			ans += recursive(G1[root][i],0,G2[root][k],0);
			ans %= MOD;
		}
	}

	printf("%lld\n",ans);
}

int main(){

	int num_case;
	scanf("%d",&num_case);

	for(int loop = 0; loop < num_case; loop++){

		func();
	}

	return 0;
}
