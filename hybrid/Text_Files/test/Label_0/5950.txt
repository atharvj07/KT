/** Problem1244 : Molecular Formula **/
typedef long long ll;
map<string, int> list;
string s;
int pos;

ll formula()
{
	if ('A' <= s[pos] && s[pos] <= 'Z') {
		string t;
		t += s[pos++];
		if ('a' <= s[pos] && s[pos] <= 'z')
			t+=s[pos++];
		
		int times=0;
		while (isdigit(s[pos])) {
			times*=10;
			times+=s[pos]-'0';
			pos++;
		}
		
		return list[t]*(times!=0?times:1);
	} else if (s[pos] == '(') {
		pos++;
		ll a = 0;
		while (s[pos]!=')') {
			a+= formula();
		}
		pos++;
				
		int times=0;
		while (isdigit(s[pos])) {
			times*=10;
			times+=s[pos]-'0';
			pos++;
		}
		return a*(times!=0?times:1);
	}
}

int main()
{
	string t;
	while (cin>>t) {
		if (t == "END_OF_FIRST_PART")
			break;
		int a; cin>>a;
		list[t] = a;
	}
	
	while (cin>>s, s!="0") {
		s+='*';
		pos = 0;
		ll ans=0;
		while (s[pos] != '*') {
			ll a = formula();
			if (a==0) {
				cout << "UNKNOWN" << endl;
				goto end;
			} else {
				ans += a;
			}
		}
		cout << ans << endl;
	end:;
	}
}
