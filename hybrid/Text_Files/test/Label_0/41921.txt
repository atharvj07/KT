def check(grid, x, y, n, m):
	for i in range(n):
		if i != x and grid[i][y] == 1:
			return False
	for i in range(m):
		if i != y and grid[x][i] == 1:
			return False
	return True
for _ in range(int(input())):
	(n, m) = map(int, input().split())
	grid = []
	for i in range(n):
		grid.append(list(map(int, input().split())))
	cur = 0
	for i in range(n):
		for j in range(m):
			if grid[i][j] == 0 and check(grid, i, j, n, m):
				cur += 1
				grid[i][j] = 1
				break
	if cur % 2:
		print('Ashish')
	else:
		print('Vivek')
