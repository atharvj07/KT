#include<bits/stdc++.h>
using namespace std;
int main()
{
	
	int t;
	cin>>t;
	while(t--)
	{
		char a[405][405];
		int b[4][2];
		int n;
		cin>>n;
		memset(b,0,sizeof b);
		getchar();
	for(int i=0;i<n;i++)
	{
		gets(*(a+i));
	}
	int x=0;
	for(int i=0;i<n;i++)
	{
		for(int j=0;j<n;j++)
		{
			if(a[i][j]=='*')
			{
				b[x][0]=i,b[x++][1]=j;
			}
		}
	}
	if(b[0][0]==b[1][0])
	{
		if(b[0][0]>=1)
		{
			a[b[0][0]-1][b[0][1]]='*',a[b[1][0]-1][b[1][1]]='*';
		}
		else a[b[0][0]+1][b[0][1]]='*',a[b[0][0]+1][b[1][1]]='*';
	}
	else if(b[0][1]==b[1][1])
	{
		if(b[0][1]>=1)
		a[b[0][0]][b[0][1]-1]='*',a[b[1][0]][b[0][1]-1]='*';
		else a[b[0][0]][b[0][1]+1]='*',a[b[1][0]][b[0][1]+1]='*';
	}
	else
	{
		a[b[0][0]][b[1][1]]='*',a[b[1][0]][b[0][1]]='*';
	}
	for(int i=0;i<n;i++)
	{
		puts(a[i]);
	}
    }
}
