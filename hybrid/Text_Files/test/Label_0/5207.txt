#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    char str[1000];
    priority_queue<int> pq;
    int visited[50];
    memset(visited, -1, sizeof(visited));
    scanf("%s", str);
    long long int c = 0;
    long long int l = strlen(str);
    for (int i = 0; i < l; i++) {
      int no = str[i] - 'a' + 1;
      if (visited[no] == -1) {
        visited[no] = 1;
        pq.push(no);
      } else if (visited[no] == 1) {
        c++;
        break;
      }
    }
    if (c != 0) {
      cout << "No" << endl;
    } else {
      c = 0;
      int s1 = pq.top();
      pq.pop();
      while (!pq.empty()) {
        int s2 = pq.top();
        pq.pop();
        if (s1 - s2 != 1) {
          c++;
        } else {
          s1 = s2;
        }
      }
      if (c == 0) {
        cout << "Yes" << endl;
      } else {
        cout << "No" << endl;
      }
    }
  }
  return 0;
}
