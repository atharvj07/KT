n = int(input())
s = input()
x = []
temp_x = 0
y = []
temp_y = 0
ans = 0
slope = 0
if s[0] == 'R':
	slope = 0.5
else:
	slope = 2
for i in range(0, len(s)):
	if s[i] == 'U':
		temp_y += 1
	elif s[i] == 'R':
		temp_x += 1
	x.append(temp_x)
	y.append(temp_y)
for i in range(1, n):
	if x[i] != 0:
		sl = y[i] / x[i]
	else:
		sl = slope
	if slope > 1 and sl < 1:
		slope = sl
		ans += 1
	elif slope < 1 and sl > 1:
		slope = sl
		ans += 1
print(ans)
