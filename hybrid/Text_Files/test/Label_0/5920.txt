#include<cstdio>
const int MaxN=200010;
int N,p[MaxN],deg[MaxN],stk[MaxN];
struct edge{int to;edge*next;}E[MaxN],*fir[MaxN];
bool vis[MaxN];
int f[MaxN][2],cir[MaxN],len,nxt[MaxN*2];
int dfn[MaxN*2],now=1;
void work(){
	
	for(int i=1;i<=N;i++){
		scanf("%d",p+i);
		deg[p[i]]++;
		E[i]=(edge){i,fir[p[i]]};
		fir[p[i]]=E+i;
	}
	int*tp=stk;
	for(int i=1;i<=N;i++)if(!deg[i])*tp++=i;
	while(tp>stk){
		int x=*--tp;
		for(edge*e=fir[x];e;e=e->next)vis[*f[e->to]]=1;
		while(vis[*f[x]])++*f[x];
		for(edge*e=fir[x];e;e=e->next)vis[*f[e->to]]=0;
		if(!--deg[p[x]])*tp++=p[x];
	}
	int o=0;
	for(int i=1;!o;i++)if(deg[i])o=i;
	for(int i=o;;){
		cir[len++]=i;
		for(edge*e=fir[i];e;e=e->next)if(!deg[e->to])vis[*f[e->to]]=1;
		while(vis[*f[i]])++*f[i];
		for(f[i][1]=*f[i]+1;vis[f[i][1]];++f[i][1]);
		for(edge*e=fir[i];e;e=e->next)if(!deg[e->to])vis[*f[e->to]]=0;
		if((i=p[i])==o)break;
	}
	for(int i=0;i<len;i++){
		nxt[i]=(i+1)%len+(f[cir[i]][0]==f[cir[(i+1)%len]][0])*len;
		nxt[i+len]=(i+1)%len+(f[cir[i]][1]==f[cir[(i+1)%len]][0])*len;
	}
	for(int i=0;i<len;i++){
		int la=now,x=i,mv=now;
		while(!dfn[x]){
			dfn[x]=now++;
			if(x<len)mv=dfn[x];
			x=nxt[x];
		}
		if(dfn[x]>=la&&now-dfn[x]==len&&mv>=dfn[x]){
			puts("POSSIBLE");
			return;
		}
	}
	puts("IMPOSSIBLE");
}
int main(){
	//freopen("in.txt","r",stdin);
	//freopen("out.txt","w",stdout);
	while(scanf("%d",&N)==1){
		for(int i=1;i<=N;i++)fir[i]=0,deg[i]=*f[i]=dfn[i-1]=dfn[i+N-1]=0;
		now=1;len=0;
		work();
	}
}
