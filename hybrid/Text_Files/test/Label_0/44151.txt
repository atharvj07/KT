from math import ceil, floor
from collections import defaultdict
test = int(input())
for t in range(test):
	n = int(input())
	arr = list(map(int, input().split()))
	S = sum(arr) / n
	if ceil(2 * S) != floor(2 * S):
		print(0)
		continue
	S = floor(2 * S)
	mp = defaultdict(int)
	cnt = 0
	for i in range(n):
		if S - arr[i] in mp:
			cnt += mp[S - arr[i]]
		mp[arr[i]] += 1
	print(cnt)
