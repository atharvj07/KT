def reduce_consec(xs):
	result = []
	stack = []
	for x in xs:
		if stack:
			if stack[-1] + 1 == x:
				stack.append(x)
			else:
				result.append((stack[0], len(stack)))
				stack = [x]
		else:
			stack.append(x)
	if stack:
		result.append((stack[0], len(stack)))
	return result

def solve(xs, n):
	zigzag = []
	for (i, x) in enumerate(xs):
		if i == 0 or i == n - 1:
			continue
		if xs[i - 1] != x and xs[i + 1] != x:
			zigzag.append(i)
	zigzag_start_length_pairs = reduce_consec(zigzag)
	count = 0
	result = xs[:]
	for (i, n) in zigzag_start_length_pairs:
		nHalf = n // 2
		count = max(count, (n + 1) // 2)
		if n % 2 == 0:
			for j in range(i, i + nHalf):
				result[j] = xs[i - 1]
			for j in range(i + nHalf, i + n):
				result[j] = 1 - xs[i - 1]
		else:
			for j in range(i, i + n):
				result[j] = xs[i - 1]
	return (count, result)

def solve0(xs, n):

	def transform(ps):
		result = []
		for i in range(n):
			if i == 0 or i == n - 1:
				result.append(ps[i])
			else:
				median = int(sum(ps[i - 1:i + 2]) >= 2)
				result.append(median)
		return tuple(result)
	xs = tuple(xs)
	seen = set()
	seen.add(xs)
	ys = transform(xs)
	count = 0
	while ys != xs:
		if ys in seen:
			return (-1, xs)
		xs = ys
		seen.add(xs)
		count += 1
		ys = transform(xs)
	return (count, xs)
