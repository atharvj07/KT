def all_children_disrespect(i, tree):
	return tree[i][2] == tree[i][4]
n = int(input())
tree = [[0, 0, 0, 0, 0] for i in range(n + 1)]
for i in range(n):
	(parent, c) = map(int, input().split())
	tree[i + 1][0] = parent
	tree[i + 1][1] = c
	if parent != -1:
		if c == 0:
			tree[parent][3] += 1
		else:
			tree[parent][4] += 1
		tree[parent][2] += 1
disrespectful = list(filter(lambda x: x[1] == 1, tree))
delet = []
for i in range(1, len(tree)):
	if tree[i][1] == 1 and all_children_disrespect(i, tree):
		delet.append(i)
if delet:
	delet = map(lambda x: str(x), delet)
	print(' '.join(delet))
else:
	print(-1)
