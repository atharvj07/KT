(n, k) = map(int, input().split())
a = [[0] * n for _ in range(n)]
s = 0
c = n * n
for i in range(n):
	for j in range(n - 1, k - 2, -1):
		a[i][j] = c
		if j == k - 1:
			s += c
		c -= 1
for i in range(n):
	for j in range(k - 2, -1, -1):
		a[i][j] = c
		c -= 1
print(s)
print('\n'.join([' '.join(map(str, _)) for _ in a]))
