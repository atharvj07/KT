#include<cstdio>
#include<algorithm>
#include<map>
#define pii pair<int,int>
using namespace std;
int curU, curD;
int n;
map<pii, int>Map, T;
map<int, pii>Z;
void Put(int a, int b, int c) {
	if (!T.count({ a,b }) || T[{a, b}] > c)T[{a, b}] = c;
}
int main() {
	int i;
	scanf("%d", &n);
	Map[{0, 0}] = 0;
	for (i = 1; i <= n; i++) {
		int a;
		T.clear();
		scanf("%d", &a);
		for (auto &t : Map) {
			int A = t.first.first;
			int B = t.first.second;
			int c = t.second;
			if (A <= a) {
				pii tp = { A,a - A };
				int c2 = c;
				if ((B == 0 && a - A > 0) || B > a - A)c2++;
				Put(A, a - A, c2);
			}
			if (a >= B) {
				pii tp = { a - B,B };
				int c2 = c;
				if (A < a - B)c2++;
				Put(a - B, B, c2);
			}
			pii tp = { 0, a };
			int c2 = c;
			if ((B == 0 && a > 0) || B > a)c2++;
			Put(0, a, c2);

			tp = { a,0 };
			c2 = c;
			if (A < a)c2++;
			Put(a, 0, c2);
		}
		Z.clear();
		Map.clear();
		for (auto &t : T) {
			pii tp = t.first;
			if (tp.first == 0 || tp.second == 0) {
				Map[t.first] = t.second;
				continue;
			}
			int u = t.second;
			if (Z.count(u)) {
				pii t2 = Z[u];
				if (tp.first <= t2.first)continue;
			}
			Z[u] = tp;
		}
		int Mn = 1e9;
		for (auto &z : Z) {
			if (Mn > z.first)Mn = z.first;
			if (z.first > Mn + 1) continue;
			Map[z.second] = z.first;
		}
	}
	int res = 1e9;
	for (auto &t : Map) {
		res = min(res, t.second);
	}
	printf("%d\n", res);
	
}
