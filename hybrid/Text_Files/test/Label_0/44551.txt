import sys
input = lambda : sys.stdin.readline().rstrip()
T = int(input())
for _ in range(T):
	(N, M) = map(int, input().split())
	X = [[0 if a == 'a' else 1 if a == 'b' else -1 for a in input()] for _ in range(N)]
	if M % 2:
		print('YES')
		print(*[1, 2] * (M // 2 + 1))
		continue
	f = 0
	for i in range(N):
		for j in range(i):
			if X[i][j] == X[j][i]:
				print('YES')
				print(*[i + 1] + [j + 1, i + 1] * (M // 2))
				f = 1
				break
		if f:
			break
	if f:
		continue
	if N == 2:
		print('NO')
		continue
	if X[0][1] == X[1][2]:
		(i, j, k) = (1, 2, 3)
	elif X[1][2] == X[2][0]:
		(i, j, k) = (2, 3, 1)
	else:
		(i, j, k) = (3, 1, 2)
	print('YES')
	if M % 4:
		print(*[i] + [j, i] * (M // 4) + [j, k] * (M // 4 + 1))
	else:
		print(*[j, i] * (M // 4) + [j, k] * (M // 4) + [j])
