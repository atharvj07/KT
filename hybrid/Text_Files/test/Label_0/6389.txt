#include <bits/stdc++.h>
using namespace std;
const int N = 1e6 + 5;
const long long MOD = 1e9 + 7;
int n, k, pr[N], nx[N];
long long ans, cnt[N];
vector<int> q, ms;
void prepare() {
  stack<int> s;
  for (int i = 0; i < n; ++i) {
    while (!s.empty() && ms[s.top()] <= ms[i]) s.pop();
    pr[i] = (s.empty() ? 0 : s.top() + 1);
    s.push(i);
  }
  while (!s.empty()) s.pop();
  for (int i = n - 1; i >= 0; --i) {
    while (!s.empty() && ms[s.top()] < ms[i]) s.pop();
    nx[i] = (s.empty() ? n - 1 : s.top() - 1);
    s.push(i);
  }
  for (int i = 1; i * (k - 1) + 1 <= n; ++i) q.push_back(i * (k - 1) + 1);
  int p = 0;
  long long sum = 0;
  for (int i = 1; i <= n; ++i) {
    while (p < (int)q.size() && q[p] <= i) sum += q[p++];
    cnt[i] = (1ll * i * p - sum + p) % MOD;
  }
}
int main() {
  scanf("%d%d", &n, &k);
  for (int i = 0; i < n; ++i) {
    int x;
    scanf("%d", &x);
    ms.push_back(x);
  }
  prepare();
  for (int i = 0; i < n; ++i) {
    ans += 1ll * ms[i] *
           (cnt[nx[i] - pr[i] + 1] - cnt[i - pr[i]] - cnt[nx[i] - i]) % MOD;
    ans %= MOD;
  }
  printf("%lld", ans);
  return 0;
}
