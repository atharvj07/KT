import sys
input = sys.stdin.readline
n = int(input())
a = list(map(int, input().split()))
Power = 0
res = -1
E = [0] * n
while Power != n:
	res += 1
	for i in range(n):
		if E[i] == 0 and Power >= a[i]:
			Power += 1
			E[i] = 1
	if Power != n:
		res += 1
		for i in range(n - 1, -1, -1):
			if E[i] == 0 and Power >= a[i]:
				Power += 1
				E[i] = 1
print(res)
