n = int(input())
tree = [[] for _ in range(n)]
p = [0] * n
c = [0] * n
for i in range(n):
	(p[i], c[i]) = map(int, input().split())
	if p[i] == -1:
		continue
	p[i] -= 1
	tree[p[i]].append(i)
ans = []
for i in range(n):
	if c[i] and all((c[j] for j in tree[i])):
		ans.append(i + 1)
if ans:
	print(*ans)
else:
	print(-1)
