import sys
import math
import bisect
import itertools
import random

def main():
	(n, m) = map(int, input().split())
	A = [-1] * n
	B = list(map(int, input().split()))
	for i in range(m):
		b = B[i]
		for j in range(b - 1, n):
			if A[j] == -1:
				A[j] = b - 1
	print(' '.join(list((str(a + 1) for a in A))))
main()
