(r, c) = map(int, input().split())
b = []
rcnt = 0
ccnt = 0
for i in range(r):
	x = list(map(int, input().split()))
	b.append(x)
a = [[0 for i in range(c)] for j in range(r)]
cmp = {}
rmp = {}
for i in range(r):
	rmp[i] = 0
	count = 0
	for j in range(c):
		if b[i][j] == 1:
			count += 1
		else:
			break
	if count == c:
		rmp[i] = 1
		rcnt += 1
for i in range(c):
	cmp[i] = 0
	count = 0
	for j in range(r):
		if b[j][i] == 1:
			count += 1
		else:
			break
	if count == r:
		cmp[i] = 1
		ccnt += 1
chk = 0
for i in range(r):
	for j in range(c):
		if b[i][j] == 1:
			if rmp[i] == 1 or cmp[j] == 1:
				continue
			else:
				chk = 1
				break
if chk != 1:
	if rcnt == ccnt or (rcnt >= 1 and ccnt >= 1):
		print('YES')
		for i in rmp.keys():
			for j in cmp.keys():
				if rmp[i] == 1 and cmp[j] == 1:
					a[i][j] = 1
		for i in range(r):
			for j in range(c):
				print(a[i][j], end=' ')
			print()
	else:
		print('NO')
else:
	print('NO')
