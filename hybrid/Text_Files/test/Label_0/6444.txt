#include <iostream>
#include <cstdio>
#include <vector>
#define il inline
#define ri register
#define ll long long
#define Size 100050
using namespace std;
struct point{
	point*next;int to,w;
}*head[Size];
ll sum,ans;
int sz[Size],n,G[3],gt;
void dfs(int,int);
il void read(int&),link(int,int,int);
int main(){
	read(n);
	for(int i(1),u,v,w;i<n;++i)
		read(u),read(v),read(w),
			link(u,v,w),link(v,u,w);dfs(1,0);
	if(gt==2){
		for(point*i(head[G[1]]);i!=NULL;i=i->next)
			if(i->to==G[2]){ans=sum-i->w;break;}}
	else for(point*i(head[G[1]]);i!=NULL;i=i->next)
			 ans=max(ans,sum-i->w);
	return printf("%lld",ans),0;
}
void dfs(int x,int pa){int part(0);
	for(point*i(head[x]);i!=NULL;i=i->next){if(i->to==pa)continue;
		dfs(i->to,x),sz[x]+=sz[i->to],part=max(part,sz[i->to]),
			sum+=(ll)(min(sz[i->to],n-sz[i->to])<<1)*i->w;
	}++sz[x],part=max(part,n-sz[x]);if((part<<1)<=n)G[++gt]=x;
}
il void link(int u,int v,int w){
	head[u]=new point{head[u],v,w};
}
il void read(int &x){
	x^=x;ri char c;while(c=getchar(),c<'0'||c>'9');
	while(c>='0'&&c<='9')x=(x<<1)+(x<<3)+(c^48),c=getchar();
}
