from sys import stdin
input = stdin.readline

def solve(a, k, n):
	b = []
	has_k = False
	for item in a:
		if item == k:
			b.append(1)
			has_k = True
		elif item < k:
			b.append(0)
		else:
			b.append(2)
	if not has_k:
		return False
	if n == 1:
		return True
	for i in range(n):
		if b[i] > 0:
			if i + 1 < n and b[i + 1] > 0:
				return True
			if i + 2 < n and b[i + 2] > 0:
				return True
	return False
for i in range(int(input())):
	(n, k) = map(int, input().split())
	a = [int(x) for x in input().split()]
	print('yes' if solve(a, k, n) else 'no')
