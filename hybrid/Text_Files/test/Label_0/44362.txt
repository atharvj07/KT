import sys
import math
input = sys.stdin.readline
from collections import deque
from queue import LifoQueue
for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	to = 0
	for i in range(n):
		if a[i] > 0:
			to += a[i]
		if a[i] < 0 and to >= -a[i]:
			to += a[i]
			a[i] = 0
		elif a[i] < 0 and to > 0:
			a[i] += to
			to = 0
	ans = 0
	for i in range(n):
		if a[i] < 0:
			ans += a[i]
	print(-ans)
