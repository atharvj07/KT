    static node merge(node left,node right){
        if(left==null)return right;
        if(right==null)return left;
        int min=Math.min(left.cur,right.cur);
        node ans=new node();
        for(int i=0;i<m;i++){
            if(left.val[i]!=0&&right.val[i]!=0){
                min=Math.min(min,left.val[i]+right.val[i]);
            }
            if(left.val[i]==0)ans.val[i]=right.val[i];
            else if(right.val[i]==0)ans.val[i]=left.val[i];
            else ans.val[i]=Math.min(left.val[i],right.val[i]);
        }
        ans.cur=min;
        return ans;
    }
    static class segmentTree{
        int n=0;
        int[] lo,hi;
        node[] value;
        segmentTree(int n){
            this.n=n;
            lo=new int[4*n+1];
            hi=new int[4*n+1];
            value=new node[4*n+1];
            init(1,1,n);
        } 
        void init(int i,int left,int right){
            lo[i]=left;
            hi[i]=right;
            if(left==right){
                return;
            }
            int mid=(left+right)/2;
            init(2*i,left,mid);
            init(2*i+1,mid+1,right);
        }
        void update(int left,int right,int val){
            update(1,left,right,val);
        }
        node query(int left,int right){
            return query(1,left,right);
        }
        void update(int i){
            value[i]=merge(value[2*i],value[2*i+1]);
        }
        void update(int i,int left,int right,int val){
            if(left>hi[i]||right<lo[i]){
                return;
            }
            if(lo[i]==hi[i]){
                value[i]=new node();
                int cnt=0;
                String str=Integer.toString(val);
                int p=str.length();
                for(int k=0;k<p;k++){
                    if(str.charAt(p-1-k)!='0')
                        value[i].val[k]=val;
                }
                return ;
            }
            update(2*i,left,right,val);
            update(2*i+1,left,right,val);
            update(i);
        }
        node query(int i,int left,int right){
            if(left>hi[i]||right<lo[i]){
                return null;
            }
            if(lo[i]>=left&&hi[i]<=right){
               // debug(value[i]);
                return value[i];
            }
            node l=query(2*i,left,right);
            node r=query(2*i+1,left,right);
            return merge(l,r);
        }
    }

    static class Pair implements Comparable<Pair> {
        int u;
        int v;
        int index=-1;
        public Pair(int u, int v) {
            this.u = u;
            this.v = v;
        }
        public int hashCode() {
            int hu = (int) (u ^ (u >>> 32));
            int hv = (int) (v ^ (v >>> 32));
            return 31 * hu + hv;
        }
        public boolean equals(Object o) {
            Pair other = (Pair) o;
            return u == other.u && v == other.v;
        }
        public int compareTo(Pair other) {
            if(index!=other.index)
                return Long.compare(index, other.index);
            return Long.compare(v, other.v)!=0?Long.compare(v, other.v):Long.compare(u, other.u);
        }
        public String toString() {
            return "[u=" + u + ", v=" + v + "]";
        }
    }

    static long modulo(long a,long b,long c) {
        long x=1;
        long y=a;
        while(b > 0){
            if(b%2 == 1){
                x=(x*y)%c;
            }
            y = (y*y)%c; // squaring the base
            b /= 2;
        }
        return  x%c;
    }
/********************************************GCD**********************************************************/
    static long gcd(long x, long y)
    {
        if(x==0)
            return y;
        if(y==0)
            return x;
        long r=0, a, b;
        a = (x > y) ? x : y; // a is greater number
        b = (x < y) ? x : y; // b is smaller number
        r = b;
        while(a % b != 0)
        {
            r = a % b;
            a = b;
            b = r;
        }
        return r;
    }

    static void sieveMake(int n){
        sieve=new boolean[n];
        Arrays.fill(sieve,true);
        sieve[0]=false;
        sieve[1]=false;
        for(int i=2;i*i<n;i++){
            if(sieve[i]){
                for(int j=i*i;j<n;j+=i){
                    sieve[j]=false;
                }
            }
        }
        primes=new ArrayList<Integer>();
        for(int i=0;i<n;i++){
            if(sieve[i]){
                primes.add(i);
            }
        }        
    }
