#include <bits/stdc++.h>
using namespace std;
struct edge {
  double x, y;
} G[10001];
int n;
bool check(double xx) {
  for (int i = 1; i <= n; i++) {
    int ll = i - 1, rr = i + 1;
    if (ll == 0) ll = n;
    if (rr == n + 1) rr = 1;
    if (fabs(G[ll].x - G[rr].x) < 0.00000001) {
      if (fabs(G[rr].x - G[i].x) / 2 < xx) return 0;
      continue;
    }
    double k = (G[rr].y - G[ll].y) / (G[rr].x - G[ll].x);
    double b = (G[ll].y * G[rr].x - G[rr].y * G[ll].x) / (G[rr].x - G[ll].x);
    if (fabs((k * G[i].x - G[i].y + b) / sqrt(k * k + 1) / 2) < xx) return 0;
  }
  return 1;
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%lf %lf", &G[i].x, &G[i].y);
  }
  double ans = 100000000000000.0;
  for (int i = 1; i <= n; i++)
    for (int j = i + 1; j <= n; j++)
      ans = min(ans, sqrt((G[i].x - G[j].x) * (G[i].x - G[j].x) +
                          (G[i].y - G[j].y) * (G[i].y - G[j].y)) /
                         2);
  double l = 0, r = ans;
  for (int i = 1; i <= 200; i++) {
    double mid = (l + r) / 2.0;
    if (check(mid))
      l = mid;
    else
      r = mid;
  }
  printf("%.8f\n", l);
  return 0;
}
