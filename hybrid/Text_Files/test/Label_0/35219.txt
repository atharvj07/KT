import java.util.Scanner;

public class BerOSFS {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String path = in.nextLine();
        path = path.replaceAll("//*", "/");
        if (path.length() > 1) {
            path = path.replaceAll("//*$", "");
        }
        System.out.println(path);
    }
}
