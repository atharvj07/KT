import java.util.Scanner; 

public class MaxHeap
{
private int size;  
private int next; 
private int[] data; 
 
public MaxHeap(int s)
{
next = 1; 
size = s; 
data = new int[s+1]; 
}

private void sift_down()
{
int i = 1; 
while(2*i < next)
{
//left child exist
if(2*i+1 < next)
{
//right child exists
if(data[i] >= data[2*i] && data[i] >= data[2*i+1]) return; //Heap is alright; 
else
{
if(data[2*i] >= data[2*i+1])
{
int temp = data[i]; 
data[i] = data[2*i]; 
data[2*i] = temp; 
i = 2*i; 
}
else
{
int temp = data[i]; 
data[i] = data[2*i+1]; 
data[2*i+1] = temp; 
i = 2*i+1;
} 
}				
}
else
{
//no right child
if(data[i] >= data [2*i]) return; //Heap is alright; 
else
{
int temp = data[i]; 
data[i] = data[2*i]; 
data[2*i] = temp; 
i = 2*i; 
}
}
}
}

private void sift_up()
{
int i = next; 
while (i > 1)
{
if(data[i/2] < data[i])
{
int temp = data[i/2]; 
data[i/2] = data[i]; 
data[i] = temp; 
i = i/2; 
}
else return; 
}
}

public void add(int elem)
{
data[next] = elem; 
sift_up(); 
next++; 
}

public int delete()
{
int result = data[1]; 
data[1] = data[next-1]; 
next--; 
sift_down(); 
return result; 
}

public void printHeap()
{
int i = 1; 
while(i < next)
{
System.out.print("" + data[i] + " "); 
i++; 
}
}

public static void main(String[] args)
{
Scanner input = new Scanner(System.in); 
int size = input.nextInt(); 
MaxHeap data = new MaxHeap(size); 
int total = 0; 
for(int i = 0; i < size; i++)
{
int value = input.nextInt(); 
total = total + value; 
data.add(value); 
} 
input.close(); 
int val = total/size; 
int rem = total%size;
int count = 0; 
int result= 0;
int elem = data.delete(); 
while(elem > val)
{
//System.out.println(result); 
if(count < rem && rem != 0)
{
result = result + elem - 1 - val; 
count++; 
}
else
result = result + elem - val; 
elem = data.delete(); 
}
//System.out.println("final result"); 
System.out.println("" + result);  

}
}
