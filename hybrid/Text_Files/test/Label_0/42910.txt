n = int(input())
a = [int(i) for i in input().split()]
cnt = [0] * (n + 1)
for i in a:
	cnt[i] += 1
rem = []
for i in range(1, n + 1):
	if cnt[i] == 0:
		rem.append(i)
rem.reverse()
print(len(rem))
skipped = [0] * (n + 1)
for i in range(n):
	if len(rem) == 0:
		break
	if cnt[a[i]] > 1 and (a[i] > rem[-1] or skipped[a[i]]):
		cnt[a[i]] -= 1
		a[i] = rem.pop()
	elif cnt[a[i]] > 1 and a[i] < rem[-1]:
		skipped[a[i]] = 1
print(*a)
