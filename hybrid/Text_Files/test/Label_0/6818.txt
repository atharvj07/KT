#include <bits/stdc++.h>
#define REP(i,n) for(int i=0; i<(int)(n); ++i)

using namespace std;

const int INF = 1000000000;

int main(){
    int n;
    while(cin >> n){
        vector<int> x(n);
        vector<int> y(n);
        REP(i, n) cin >> x[i] >> y[i];

        int dp[1001][1001] = {};
        REP(i, n) REP(j, n + 1) dp[i][j] = (i + 1 == j ? 0 : INF);

        for(int len = 2; len <= n; len++) {
            for(int i = 0; i + len <= n; i++) {
                int j = i + len;
                for(int k = i + 1; k < j; k++) {
                    dp[i][j] = min(dp[i][j], dp[i][k] + dp[k][j] + x[k] - x[i] + y[k - 1] - y[j - 1]);
                }
            }
        }
        
        cout << dp[0][n] << endl;
    }

    return 0;
}
