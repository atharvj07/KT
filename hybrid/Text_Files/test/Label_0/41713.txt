import sys
from math import sqrt, gcd, ceil, log
from collections import defaultdict, Counter, deque
input = sys.stdin.readline
read = lambda : list(map(int, input().strip().split()))

def main():
	ans = []
	(n, q) = read()
	par = [0, 0] + read()
	adj = defaultdict(list)
	for i in range(2, n + 1):
		adj[par[i]].append(i)
	for i in adj:
		adj[i].sort(reverse=True)
	ord_arr = []
	order = defaultdict(int)
	stk = [1]
	while stk:
		node = stk.pop()
		ord_arr.append(node)
		order[node] = len(ord_arr)
		for child in adj[node]:
			if child != par[node]:
				stk.append(child)
	sub_tree = defaultdict(int)
	for i in range(n, 0, -1):
		sub_tree[i] += 1
		sub_tree[par[i]] += sub_tree[i]
	for i in range(q):
		(u, v) = read()
		if sub_tree[u] < v:
			ans.append(str(-1))
		else:
			ans.append(str(ord_arr[order[u] - 1 + v - 1]))
	print('\n'.join(ans))
main()
