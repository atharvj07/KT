n = int(input())
a = list(map(int, input().split()))
ans = 0
flag = 0
while True:
	for i in range(n):
		if a[i] <= i:
			flag = 1
			ans = i
			break
	if flag:
		print(ans + 1)
		exit()
	mina = min(a)
	if n > mina:
		minus = n
	else:
		minus = mina // n * n
	for i in range(n):
		a[i] -= min(a[i], minus)
