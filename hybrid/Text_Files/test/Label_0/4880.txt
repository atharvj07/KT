#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, w = 0, x = 0, y = 0, d = 0, ans = 0, temp, tw = 0;
  char c;
  cin >> n >> m;
  int Min[n + 1], Max[n + 1], pos[n + 1];
  for (int i = 0; i < n; i++) {
    w = 0;
    Min[i] = 1e9 + 10;
    Max[i] = 0;
    for (int j = 0; j < m; j++) {
      cin >> c;
      if (c == 'W') {
        Min[i] = min(Min[i], j);
        Max[i] = max(Max[i], j);
        w++;
        tw++;
      }
    }
    pos[i] = w;
  }
  Min[n] = 1e9 + 10;
  Max[n] = 0;
  if (tw == 0) {
    cout << 0 << endl;
    return 0;
  }
  while (tw != 0) {
    ans++;
    if (d == 0) {
      temp = y;
      y = max(Max[x], Max[x + 1]);
      if (y == 0) {
        y = temp;
        tw -= pos[x];
        x++;
        d = 1 - d;
        continue;
      }
      tw -= pos[x];
      x++;
      d = 1 - d;
      ans += abs(y - temp);
      continue;
    } else {
      temp = y;
      y = min(Min[x], Min[x + 1]);
      if (y == 1e9 + 10) {
        y = temp;
        tw -= pos[x];
        x++;
        d = 1 - d;
        continue;
      }
      tw -= pos[x];
      x++;
      d = 1 - d;
      ans += abs(temp - y);
      continue;
    }
  }
  cout << ans - 1 << endl;
  return 0;
}
