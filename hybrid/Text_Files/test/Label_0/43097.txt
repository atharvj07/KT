for _ in range(int(input())):
	n = int(input())
	a = input().split()
	res = a[0]
	if n == 3:
		print('b' + res)
	else:
		for i in range(1, len(a)):
			str2 = a[i]
			str1 = a[i - 1]
			if str2[0] == str1[1]:
				res += str2[1]
			else:
				res += a[i]
		if len(res) != n:
			print(res + 'b')
		else:
			print(res)
