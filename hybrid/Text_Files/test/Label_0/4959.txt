#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  int sum1 = 0;
  int b[n];
  int sum2 = 0;
  for (int i = 0; i < n; i++) {
    b[i] = a[i] / 2;
    if (a[i] % 2 == 0) {
      sum1 += b[i];
    } else {
      sum2 += b[i];
    }
  }
  if (sum1 + sum2) {
    sum2 += sum1;
    for (int i = 0; i < n && sum2 < 0; i++) {
      if (a[i] % 2 == 1 && a[i] > 0) {
        b[i]++;
        sum2++;
      }
    }
    for (int i = 0; i < n && sum2 > 0; i++) {
      if (a[i] < 0 && a[i] % 2 == -1) {
        b[i]--;
        sum2--;
      }
    }
  }
  for (int i = 0; i < n; i++) cout << b[i] << '\n';
  return 0;
}
