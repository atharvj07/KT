n = int(input())
k = int(input())
a = int(input())
b = int(input())
cost = 0
if k == 1:
	print((n - 1) * a)
	exit()
while n != 1:
	if n % k == 0:
		if b > (n - n // k) * a:
			cost += (n - n // k) * a
			n = n // k
		else:
			cost += b
			n = n // k
	elif n < k:
		cost += (n - 1) * a
		n = 1
	else:
		cost += a * (n % k)
		n -= n % k
print(cost)
