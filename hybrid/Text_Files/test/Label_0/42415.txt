(n, a, b) = [int(x) for x in input().split()]
mp = {}
spmp = {}
for i in range(n):
	(x, vx, vy) = [int(x) for x in input().split()]
	vv = a * vx - vy
	spr = str([vx, vy])
	if vv in mp:
		mp[vv] += 1
	else:
		mp[vv] = 1
	if spr in spmp:
		spmp[spr] += 1
	else:
		spmp[spr] = 1
ans = 0
for v in mp.values():
	ans += v * (v - 1)
for v in spmp.values():
	ans -= v * (v - 1)
print(ans)
