import java.util.*;
import java.io.*;
public class Solution {
  public static void main(String[] args) {
    Scanner in = new Scanner(new BufferedReader(new InputStreamReader(System.in)));
    PrintWriter out = new PrintWriter(new OutputStreamWriter(System.out));
    int n = in.nextInt();
    int m = in.nextInt();
    int[][] b = new int[n][m];
    for (int i = 0; i < n; ++i) {
      for (int j = 0; j < m; ++j) {
        b[i][j] = in.nextInt();
      }
    }
    int[][] rst = solve(b);
    for (int i = 0; i < n; ++i) {
      for (int j = 0; j < m; ++j) {
        out.print(rst[i][j]);
        if (j < m - 1) {
          out.print(" ");
        }
      }
      out.println();
    }
    out.flush();
  }
  public static int[][] solve(int[][] b) {
    int n = b.length;
    int m = b[0].length;
    int[][] h = new int[n][m];
    int[][] v = new int[n][m];
    int[][] rst = new int[n][m];
    Integer[] row = new Integer[m];
    Integer[] col = new Integer[n];
    int[] hh = new int[n];
    int[] vh = new int[m];
    init(row);
    init(col);
    for (int i = 0; i < n; ++i) {
      final int ii = i;
      Arrays.sort(row, (e0, e1) -> b[ii][e0] - b[ii][e1]);
      int rank = 1;
      for (int j = 0; j < row.length; ++j) {
        if (j > 0 && b[i][row[j]] > b[i][row[j - 1]]) {
          rank++;
        }
        h[i][row[j]] = rank;
      }
      hh[i] = rank;
    }
    for (int i = 0; i < m; ++i) {
      final int ii = i;
      Arrays.sort(col, (e0, e1) -> b[e0][ii] - b[e1][ii]);
      int rank = 1;
      for (int j = 0; j < col.length; ++j) {
        if (j > 0 && b[col[j]][i] > b[col[j - 1]][i]) {
          rank++;
        }
        v[col[j]][i] = rank;
      }
      vh[i] = rank;
    }
    for (int i = 0; i < n; ++i) {
      for (int j = 0; j < m; ++j) {
        int rh = Math.max(hh[i] - h[i][j], vh[j] - v[i][j]);
        int rcur = Math.max(h[i][j], v[i][j]);
        rst[i][j] = rcur + rh;
      }
    }
    return rst;
  }
  public static void init(Integer[] arr) {
    for (int i = 0; i < arr.length; ++i) {
      arr[i] = i;
    }
  }
}
