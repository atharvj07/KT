import math

def cal(s):
	lenS = len(s)
	sheeps = []
	ans = 0
	for i in range(lenS):
		if s[i] == '*':
			sheeps.append(i)
	if not sheeps:
		return ans
	mid = math.ceil(len(sheeps) / 2)
	midInd = sheeps[mid - 1]
	cur = 1
	for i in range(lenS):
		if s[i] == '*':
			ans += abs(midInd - i - (mid - cur))
			cur += 1
	return ans
for i in range(int(input())):
	n = int(input())
	s = input()
	print(cal(s))
