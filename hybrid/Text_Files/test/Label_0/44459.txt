import os
import sys
from atexit import register
from io import StringIO
sys.stdout = StringIO()
register(lambda : os.write(1, sys.stdout.getvalue().encode()))
import math
n = int(input())
s = input()
x = [ord(c) - 97 for c in s]
c = [1]
res = 1
r = [x[0]]
for k in x[1:]:
	appended = False
	for cl in range(res):
		if k >= r[cl]:
			c.append(cl + 1)
			r[cl] = k
			appended = True
			break
	if not appended:
		res += 1
		r.append(k)
		c.append(res)
print(res)
print(' '.join([str(k) for k in c]))
