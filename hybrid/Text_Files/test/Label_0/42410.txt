import sys
input = sys.stdin.readline
for _ in range(int(input())):
	(n, d) = map(int, input().split())
	ar = list(map(int, input().split()))
	se = set({})
	li = []
	for i in range(n):
		if not i in se:
			li.append([ar[i]])
			se.add(i)
			j = (i + d) % n
			while j != i:
				li[-1].append(ar[j])
				se.add(j)
				j = (j + d) % n
			li[-1] = li[-1] + li[-1]
	ans = 0
	flag = True
	for i in li:
		le = len(i)
		count = i.count(1)
		if count == le:
			flag = False
			break
		count = 0
		for j in i:
			if j == 1:
				count += 1
			else:
				count = 0
			ans = max(ans, count)
	if flag:
		print(ans)
	else:
		print(-1)
