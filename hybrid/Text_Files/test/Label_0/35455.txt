import java.io.*;
import java.math.*;
import java.util.*;

public class Main {
	public static void main(String[] args) throws Exception {
		InputStream inputStream = System.in;
		OutputStream outputStream = System.out;
		Scanner in = new Scanner(inputStream);
		PrintWriter out = new PrintWriter(outputStream);
		new Task().solve(1,in,out);
		out.close();
	}
}

class Task {
	void solve(int ca,Scanner in,PrintWriter out) {
		int n = in.nextInt();
		int m = in.nextInt();
		int[][] col = new int[n][m];
		char[][] mat = new char[n][m];
		for (int i = 0; i < n; ++ i) {
			mat[i] = in.next().toCharArray();
		}
		int tim = 233;
		for (int i = 0; i < n; ++ i) {
			for (int j = 1; j < m; ++ j) {
				if (mat[i][j]=='#' || mat[i][j-1]=='#') continue;
				if (col[i][j]>0 || col[i][j-1]>0) continue;
				col[i][j] = col[i][j-1] = ++ tim;
			}
		}
		for (int i = 1; i < n; ++ i) {
			for (int j = 0; j < m; ++ j) {
				if (mat[i][j]=='#' || mat[i-1][j]=='#') continue;
				if (col[i][j]>0 || col[i-1][j]>0) continue;
				col[i][j] = col[i-1][j] = ++ tim;
			}
		}
		for (int i = 0; i < n; ++ i) {
			for (int j = 0; j < m; ++ j) {
				if (mat[i][j]=='#') continue;
				if (col[i][j]>0) continue;
				if (i>0 && col[i-1][j]>0)
					col[i][j] = col[i-1][j];
				else if (j>0 && col[i][j-1]>0)
					col[i][j] = col[i][j-1];
				else if (i<n-1 && col[i+1][j]>0)
					col[i][j] = col[i+1][j];
				else if (j<m-1 && col[i][j+1]>0)
					col[i][j] = col[i][j+1];
				else {
					out.println(-1);
					return ;
				}
			}
		}
		int[] que = new int[n*m*2];
		int[] dx = {1,-1,0,0};
		int[] dy = {0,0,1,-1};
		boolean[][] vis = new boolean[n][m];
		for (int i = 0; i < n; ++ i) {
			for (int j = 0; j < m; ++ j) {
				if (mat[i][j]=='#') continue;
				if (vis[i][j]) continue;
				int qf = 0,qe = 0;
				int mask = 0;
				que[qe++] = i;
				que[qe++] = j;
				vis[i][j] = true;
				while (qf != qe) {
					int x = que[qf++];
					int y = que[qf++];
					for (int k = 0; k < 4; ++ k) {
						int tx = x+dx[k];
						int ty = y+dy[k];
						if (tx<0 || tx>=n || ty<0 || ty>=m) continue;
						if (mat[tx][ty]=='#') continue;
						if (col[tx][ty] != col[i][j]) {
							if (col[tx][ty] < 10)
								mask |= 1<<col[tx][ty];
							continue;
						}
						if (vis[tx][ty]) continue;
						vis[tx][ty] = true;
						que[qe++] = tx;
						que[qe++] = ty;
					}
				}
				int c = 0;
				while ((mask>>c&1)==1) ++ c;
				for (int I = 0; I < qe; I += 2) {
					int x = que[I];
					int y = que[I+1];
					col[x][y] = c;
				}
			}
		}
		for (int i = 0; i < n; ++ i) {
			for (int j = 0; j < m; ++ j) {
				if (mat[i][j] == '#') {
					out.print('#');
				} else {
					out.print(col[i][j]);
				}
			}
			out.println();
		}
	}
}
