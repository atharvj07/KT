def check(n, s):
	for i in range(1, n):
		if s[i] == s[i - 1] and s[i] == 1:
			return False
	return True
n = int(input())
st = input()
s = []
for i in range(n):
	s.append(int(st[i]))
result = check(n, s)
for i in range(n):
	if s[i] == 0:
		s[i] = 1
		if result:
			result = not check(n, s)
		s[i] = 0
if result:
	print('Yes')
else:
	print('No')
