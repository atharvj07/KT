def comb(n, r):
	if r > n or r < 0:
		return 0
	r = min(r, n - r)
	result = 1
	for i in range(1, r + 1):
		result = result * (n + 1 - i) // i
	return result

def F(n, t):
	if t == 0:
		return 1
	elif n == 0:
		return 0
	elif n == 1:
		return int(t == 1)
	m = len(bin(n)) - 3
	return F(n - (1 << m), t - 1) + comb(m, t)
(n, t) = map(int, input().strip().split())
T = len(bin(t)) - 3
if 1 << T != t:
	print(0)
else:
	print(F(n + 1, T + 1) - int(t == 1))
