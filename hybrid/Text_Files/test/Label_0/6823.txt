#include <bits/stdc++.h>
using namespace std;
using P=pair<int,int>;

int main(){
  int h,w; cin>>h>>w;
  vector<string> m(h);
  for(int i=0;i<h;i++){
    cin>>m[i];
  }long long ans=0;
  int my[4]={0,0,-1,1};
  int mx[4]={-1,1,0,0};
  vector<vector<bool>> flag(h,vector<bool>(w,false));
  for(int i=0;i<h;i++){
    for(int j=0;j<w;j++){
      if(flag[i][j]) continue;
      queue<P> que;
      que.push(P(i,j));
      flag[i][j]=true;
      long long a=0,b=0;
      while(!que.empty()){
        P p=que.front(); que.pop();
        int y=p.first,x=p.second;
        char base=m[y][x];
        if(base=='#') b++;
        else a++;
        for(int k=0;k<4;k++){
          int ny=y+my[k],nx=x+mx[k];
          if(ny>=h ||ny<0 ||nx<0 ||nx>=w) continue;
          if(flag[ny][nx] || m[ny][nx]==base) continue;
          flag[ny][nx]=true;
          que.push(P(ny,nx));
        }
      }
      ans+=a*b;
    }
  }
  cout<<ans<<endl;
}
