#include <bits/stdc++.h>
#define rep(i,n) for(int i=0;i<(n);++i)
#define ALL(A) A.begin(), A.end()

using namespace std;

typedef long long ll;
typedef pair<int, int> P;

int main()
{
	ios_base::sync_with_stdio(0);
	cin.tie(0);
	int theta1, theta2; cin >> theta1 >> theta2;

	double t1 = (theta2 - theta1) / 2.;
	double t2 = (theta2 - 360 - theta1) / 2.;
	double t3 = (theta2 - theta1 + 360) / 2.;
	double t = 360.;
	t = (abs(t) > abs(t1) ? t1 : t);
	t = (abs(t) > abs(t2) ? t2 : t);
	t = (abs(t) > abs(t3) ? t3 : t);

	double res = theta1 + t;
	res = (res < 0. ? res + 360 : res);
	res = (res > 360 ? res - 360 : res);

	printf("%.1lf\n", res);

	return 0;
}
