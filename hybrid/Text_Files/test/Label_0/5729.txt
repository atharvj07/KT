#include <bits/stdc++.h>
using namespace std;
int ct = 0;
char test(int x, int y) {
  printf("? %d %d\n", x, y);
  fflush(stdout);
  char go[10];
  scanf("%s", go);
  if (go[0] == 'e') exit(0);
  return go[0];
}
void solve() {
  int l = 1, r = -1, x = 2;
  for (int i = 1; i <= 29; i++) {
    if (test(x * 2, x) == 'x')
      l = x, x = x * 2;
    else {
      l = x;
      r = 2 * x;
      break;
    }
  }
  if (x == 1 || r == -1) {
    if (test(2, 1) == 'y') {
      printf("! 2\n");
      fflush(stdout);
      return;
    }
    if (test(1, 2) == 'x') {
      printf("! 1\n");
      fflush(stdout);
      return;
    }
  }
  int ans = r;
  int base = l;
  while (l < r) {
    int mid = (l + r) >> 1;
    if (test(mid, base) == 'y')
      ans = min(ans, mid), r = mid;
    else {
      l = mid + 1;
    }
  }
  printf("! %d\n", ans);
  fflush(stdout);
  return;
}
int main() {
  char s[100];
  while (1) {
    scanf("%s", s);
    if (s[0] == 's')
      solve();
    else
      break;
  }
  return 0;
}
