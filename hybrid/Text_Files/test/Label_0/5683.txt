#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long t, n, x, y, m, i, j, k, l;
  string s;
  t = 1;
  while (t--) {
    cin >> n >> s;
    long long ans = 0, len = s.length();
    for (i = 0; i < len; i++) {
      for (j = i; j < len; j++) {
        x = 0;
        y = 0;
        for (k = i; k <= j; k++) {
          if (s[k] == 'U')
            y++;
          else if (s[k] == 'D')
            y--;
          else if (s[k] == 'R')
            x++;
          else if (s[k] == 'L')
            x--;
        }
        if (x == 0 && y == 0) ans++;
      }
    }
    cout << ans << '\n';
  }
  return 0;
}
