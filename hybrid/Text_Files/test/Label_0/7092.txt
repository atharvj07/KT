#include <bits/stdc++.h>
using namespace std;
const int INF = 1e9;
const int maxm = 100005;
int x[maxm], xr[maxm];
int main() {
  int n, i, j, k, sum, m, cnt = 0, ans, t;
  int x1, x2, y;
  scanf("%d%d", &n, &m);
  for (i = 0; i < n; i++) scanf("%d", &x[i]);
  ans = t = 0;
  sort(x, x + n);
  for (i = 1; i <= m; i++) {
    scanf("%d%d%d", &x1, &x2, &y);
    if (x1 == 1) {
      xr[t++] = x2;
      if (x2 == INF) ans++;
    }
  }
  sort(xr, xr + t);
  ans += n;
  i = 0, j = 0;
  cnt = t;
  while (i < n) {
    while (j < t && xr[j] < x[i]) {
      --cnt;
      ++j;
    }
    ans = min(ans, cnt + i);
    if (ans == 0) break;
    ++i;
  }
  printf("%d\n", ans);
  return 0;
}
