import math

def check(n, d):
	ms = math.sqrt(d)
	f = math.floor(ms)
	c = math.ceil(ms)
	if math.ceil(f + d / f - 1) <= n or math.ceil(c + d / c - 1) <= n:
		return 1
	return 0
for _ in range(int(input())):
	(n, d) = map(int, input().split())
	if d <= n or check(n, d):
		print('YES')
	else:
		print('NO')
