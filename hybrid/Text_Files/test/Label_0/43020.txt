import sys
from collections import defaultdict
input = lambda : sys.stdin.readline().rstrip()
n = int(input())
d = defaultdict(int)
for _ in range(n):
	s = input()
	(e, mask) = (defaultdict(int), 0)
	for ch in s:
		e[ch] += 1
	for ch in e:
		v = 1 << ord(ch) - ord('a')
		if e[ch] & 1:
			mask |= v
	d[mask] += 1
ans = 0
aa = list(d.keys())
for mask in aa:
	ans += d[mask] * (d[mask] - 1) // 2
	for bit in range(26):
		v = 1 << bit
		if mask & v > 0:
			ans += d[mask ^ v] * d[mask]
print(ans)
