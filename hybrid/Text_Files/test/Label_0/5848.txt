#include <bits/stdc++.h>
using namespace std;
inline long long area(pair<pair<int, int>, pair<int, int> > a) {
  return (a.second.second - a.first.second) * 1ll *
         (a.second.first - a.first.first);
}
pair<pair<int, int>, pair<int, int> > inter(
    pair<pair<int, int>, pair<int, int> > a,
    pair<pair<int, int>, pair<int, int> > b) {
  pair<pair<int, int>, pair<int, int> > ans = {{0, 0}, {0, 0}};
  if (a.first.second >= b.second.second) return ans;
  if (b.first.second >= a.second.second) return ans;
  if (a.first.first >= b.second.first) return ans;
  if (b.first.first >= a.second.first) return ans;
  if (b.first.second < a.first.second) swap(a, b);
  ans.first.second = b.first.second;
  if (b.second.second < a.second.second)
    ans.second.second = b.second.second;
  else
    ans.second.second = a.second.second;
  if (b.first.first < a.first.first) swap(a, b);
  ans.first.first = b.first.first;
  if (b.second.first < a.second.first)
    ans.second.first = b.second.first;
  else
    ans.second.first = a.second.first;
  return ans;
}
bool solve() {
  pair<pair<int, int>, pair<int, int> > w;
  cin >> w.first.second >> w.first.first >> w.second.second >> w.second.first;
  pair<pair<int, int>, pair<int, int> > b1;
  cin >> b1.first.second >> b1.first.first >> b1.second.second >>
      b1.second.first;
  pair<pair<int, int>, pair<int, int> > b2;
  cin >> b2.first.second >> b2.first.first >> b2.second.second >>
      b2.second.first;
  pair<pair<int, int>, pair<int, int> > wb1 = inter(w, b1);
  pair<pair<int, int>, pair<int, int> > wb2 = inter(w, b2);
  pair<pair<int, int>, pair<int, int> > com = inter(wb1, wb2);
  return area(w) > area(wb1) + area(wb2) - area(com);
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int T = 1;
  while (T--) cout << (solve() ? "YES\n" : "NO\n");
  return 0;
}
