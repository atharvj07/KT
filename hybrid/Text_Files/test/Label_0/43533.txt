__author__ = 'sm'

def readln():
	return list(map(int, input().split()))
s = input()
len_s = len(s)
count_x = [0] * len_s
count_y = [0] * len_s
count_z = [0] * len_s
for (i, val) in enumerate(s):
	if val == 'x':
		if i == 0:
			count_x[0] = 1
		else:
			count_x[i] = count_x[i - 1] + 1
			count_y[i] = count_y[i - 1]
			count_z[i] = count_z[i - 1]
	if val == 'y':
		if i == 0:
			count_y[0] = 1
		else:
			count_x[i] = count_x[i - 1]
			count_y[i] = count_y[i - 1] + 1
			count_z[i] = count_z[i - 1]
	if val == 'z':
		if i == 0:
			count_z[0] = 1
		else:
			count_x[i] = count_x[i - 1]
			count_y[i] = count_y[i - 1]
			count_z[i] = count_z[i - 1] + 1
m = readln()[0]
output = []
for i in range(m):
	(l, r) = readln()
	l -= 1
	r -= 1
	if r - l + 1 < 3:
		output.append('YES')
		continue
	num_x = 0
	num_y = 0
	num_z = 0
	if l == 0:
		num_x = count_x[r]
		num_y = count_y[r]
		num_z = count_z[r]
	else:
		num_x = count_x[r] - count_x[l - 1]
		num_y = count_y[r] - count_y[l - 1]
		num_z = count_z[r] - count_z[l - 1]
	ans = [num_x, num_y, num_z]
	ans.sort()
	ans[1] -= ans[0]
	ans[2] -= ans[0]
	ans[0] -= ans[0]
	if sum(ans) <= 2 and max(ans) <= 1:
		output.append('YES')
	else:
		output.append('NO')
print('\n'.join(output))
