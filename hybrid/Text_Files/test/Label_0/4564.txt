#include <bits/stdc++.h>
using namespace std;
template <typename T, typename U>
ostream& operator<<(ostream& os, const pair<T, U>& p) {
  os << "(" << p.first << "," << p.second << ")";
  return os;
}
template <typename T>
ostream& operator<<(ostream& os, const vector<T>& v) {
  os << "[";
  for (__typeof((v).begin()) i = (v).begin(); i != (v).end(); ++i) {
    if (i != v.begin()) os << ", ";
    os << *i;
  }
  os << "]";
  return os;
}
int main() {
  int n, m, k;
  scanf("%d%d%d", &n, &m, &k);
  vector<vector<double> > C(n + 1);
  for (int i = (int)(0); i <= (int)(n); ++i) {
    C[i].resize(i + 1);
    C[i][0] = 1;
    for (int j = (int)(1); j <= (int)(i - 1); ++j)
      C[i][j] = C[i - 1][j - 1] + C[i - 1][j];
    C[i][i] = 1;
  }
  vector<double> P(((n) * (n)) + 1, 0);
  P[0] = 1.0;
  for (int i = (int)(1); i <= (int)(((n) * (n))); ++i)
    P[i] = P[i - 1] * (k - i + 1) / (m - i + 1);
  double res = 0.0;
  for (int a = (int)(0); a <= (int)(n); ++a)
    for (int b = (int)(0); b <= (int)(n); ++b)
      res += C[n][a] * C[n][b] * P[(a + b) * n - a * b];
  printf("%.10lf\n", min(res, 1e99));
}
