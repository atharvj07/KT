#include <bits/stdc++.h>
using namespace std;
const long long Mod = 1e9 + 7;
const long long N = 300005;
long long dp[N][2], T, n;
vector<long long> vec[N];
long long tmp[2];
inline long long read() {
  long long x = 0, f = 1;
  char ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') {
    x = (x << 1) + (x << 3) + (ch ^ 48);
    ch = getchar();
  }
  return x * f;
}
void dfs(long long x) {
  dp[x][1] = 1;
  dp[x][0] = 0;
  long long a = 1, b0 = 1, b1 = 0;
  for (long long i = 0; i < vec[x].size(); i++) {
    long long y = vec[x][i];
    dfs(y);
    tmp[0] = dp[x][0];
    tmp[1] = dp[x][1];
    dp[x][0] =
        (dp[x][0] + tmp[1] * dp[y][1] % Mod + tmp[0] * dp[y][0] % Mod) % Mod;
    dp[x][1] =
        (dp[x][1] + tmp[1] * dp[y][0] % Mod + tmp[0] * dp[y][1] % Mod) % Mod;
    a = a * (dp[y][0] + 1) % Mod;
    tmp[0] = b0;
    tmp[1] = b1;
    b0 = (b0 + tmp[1] * dp[y][1] % Mod) % Mod;
    b1 = (b1 + tmp[0] * dp[y][1] % Mod) % Mod;
  }
  dp[x][0] = (dp[x][0] * 2 - b1 + Mod) % Mod;
  dp[x][1] = (dp[x][1] * 2 - a + Mod) % Mod;
}
signed main() {
  n = read();
  for (long long i = 1; i <= n; i++) vec[i].clear();
  for (long long i = 2; i <= n; i++) {
    long long x = read();
    vec[x].push_back(i);
  }
  dfs(1);
  printf("%lld\n", (dp[1][0] + dp[1][1]) % Mod);
  return 0;
}
