#include <bits/stdc++.h>
using namespace std;
int test;
int main() {
  cin >> test;
  for (int i = 0; i < test; i++) {
    int n;
    cin >> n;
    vector<pair<long long, pair<long long, int> > > v;
    long long total[2] = {0, 0};
    for (int j = 0; j < 2 * n - 1; j++) {
      long long a, o;
      cin >> a >> o;
      v.push_back(make_pair(a, make_pair(o, j + 1)));
      total[0] += a;
      total[1] += o;
    }
    sort(v.begin(), v.end());
    long long sum[2] = {0, 0};
    for (int j = 0; j < 2 * n - 1; j += 2) {
      sum[0] += v[j].first;
      sum[1] += v[j].second.first;
    }
    if (sum[0] >= (total[0] + 1) / 2 && sum[1] >= (total[1] + 1) / 2) {
      cout << "YES" << endl;
      for (int j = 0; j < 2 * n - 1; j += 2) cout << v[j].second.second << " ";
      cout << endl;
    } else {
      sum[0] = 0;
      sum[1] = 0;
      for (int j = 1; j < 2 * n - 1; j += 2) {
        sum[0] += v[j].first;
        sum[1] += v[j].second.first;
      }
      sum[0] += v[2 * n - 2].first;
      sum[1] += v[2 * n - 2].second.first;
      if (sum[0] >= (total[0] + 1) / 2 && sum[1] >= (total[1] + 1) / 2) {
        cout << "YES" << endl;
        for (int j = 1; j < 2 * n - 1; j += 2)
          cout << v[j].second.second << " ";
        cout << v[2 * n - 2].second.second << endl;
      } else {
        cout << "NO" << endl;
        cout << endl;
      }
    }
  }
  return 0;
}
