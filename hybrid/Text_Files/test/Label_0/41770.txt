for _ in range(int(input())):
	arr = list(map(int, input().rstrip().rsplit()))
	(n, a, b, c, d) = (arr[0], arr[1], arr[2], arr[3], arr[4])
	(w1, w2, w3, w4) = (n * (a - b), n * (a + b), c - d, c + d)
	if w1 <= w4 and w2 >= w3:
		if w1 <= w3 and w2 <= w4 or (w1 >= w3 and w2 >= w4) or (w1 < w3 and w2 > w4) or (w1 > w3 and w2 < w4):
			print('Yes')
		else:
			print('No')
	else:
		print('No')
