#include <bits/stdc++.h>
using namespace std;
long long in() {
  int32_t x;
  scanf("%d", &x);
  return x;
}
const long long maxn = 1e5 + 10;
long long mp(char ch) {
  return ch >= 'A' && ch <= 'Z' ? (ch - 'A') + 26 : (ch - 'a');
}
const long long inf = 1e18 + 10;
long long cnt[53][53];
long long source, sink, input[maxn], output[maxn];
vector<long long> g[maxn];
struct edge {
  long long to, cap, weight;
} e[maxn];
long long ecnt = 0;
long long edgeid[60][60];
inline void add_edge(long long v, long long u, long long cap,
                     long long weight) {
  g[v].push_back(ecnt);
  e[ecnt++] = {u, cap, weight};
  g[u].push_back(ecnt);
  e[ecnt++] = {v, 0, -weight};
}
bool used[maxn];
long long di[maxn];
long long cur;
long long q[maxn], t, h;
long long pre[maxn];
long long res[maxn];
inline long long mcmf() {
  long long ret = 0;
  while (true) {
    for (long long i = 0; i <= cur; i++) used[i] = 0, di[i] = inf;
    t = h = 0;
    q[t++] = source;
    di[source] = 0;
    while (h < t) {
      long long v = q[h++];
      used[v] = false;
      for (auto u : g[v]) {
        if (e[u].cap == 0) continue;
        long long nxt = e[u].to;
        long long cst = di[v] + e[u].weight;
        if (cst < di[nxt]) {
          di[nxt] = cst;
          pre[nxt] = u;
          if (!used[nxt]) {
            used[nxt] = true;
            q[t++] = nxt;
          }
        }
      }
    }
    if (di[sink] == inf) break;
    long long mini = inf;
    long long cur = sink;
    while (cur != source) {
      long long preedgeid = pre[cur];
      mini = min(mini, e[preedgeid].cap);
      cur = e[preedgeid ^ 1].to;
    }
    cur = sink;
    ret += mini * di[sink];
    while (cur != source) {
      long long preedgeid = pre[cur];
      e[preedgeid].cap -= mini;
      e[preedgeid ^ 1].cap += mini;
      cur = e[preedgeid ^ 1].to;
    }
  }
  return ret;
}
int32_t main() {
  long long n = in(), k = in();
  string s;
  cin >> s;
  string answers;
  cin >> answers;
  for (long long i = 0; i < n; i++) cnt[mp(s[i])][mp(answers[i])]++;
  for (long long i = 0; i < k; i++) input[i] = cur++, output[i] = cur++;
  source = cur++, sink = cur++;
  for (long long i = 0; i < k; i++) {
    add_edge(source, input[i], 1, 0);
    add_edge(output[i], sink, 1, 0);
    for (long long j = 0; j < k; j++) {
      edgeid[i][j] = ecnt;
      add_edge(input[i], output[j], 1, -cnt[i][j]);
    }
  }
  cout << -mcmf() << endl;
  ;
  for (long long i = 0; i < k; i++) {
    for (long long j = 0; j < k; j++) {
      long long edgeid = ::edgeid[i][j];
      if (e[edgeid].cap == 0) res[i] = j;
    }
    cout << (res[i] < 26 ? char('a' + res[i]) : char('A' + (res[i] - 26)));
  }
}
