n = int(input())
s = list(input())
cnt_0 = s.count('0')
cnt_1 = s.count('1')
cnt_2 = s.count('2')
k = n // 3
d = {'0': cnt_0, '1': cnt_1, '2': cnt_2}
for i in range(n):
	if d[s[i]] > k:
		if s[i] == '1' and cnt_1 > k:
			if cnt_0 < k:
				s[i] = '0'
				cnt_0 += 1
				cnt_1 -= 1
		elif s[i] == '2' and cnt_2 > k:
			if cnt_0 < k:
				s[i] = '0'
				cnt_0 += 1
				cnt_2 -= 1
			elif cnt_1 < k:
				s[i] = '1'
				cnt_1 += 1
				cnt_2 -= 1
for j in range(n - 1, -1, -1):
	if d[s[j]] > k:
		if s[j] == '0' and cnt_0 > k:
			if cnt_2 < k:
				s[j] = '2'
				cnt_2 += 1
				cnt_0 -= 1
			elif cnt_1 < k:
				s[j] = '1'
				cnt_1 += 1
				cnt_0 -= 1
		elif s[j] == '1' and cnt_1 > k:
			if cnt_2 < k:
				s[j] = '2'
				cnt_2 += 1
				cnt_1 -= 1
print(*s, sep='')
