from collections import defaultdict
n = int(input())
if n > 11:
	print(292 + 49 * (n - 11))
else:
	d = defaultdict(list)
	for L in range(n + 1):
		for X in range(n + 1 - L):
			for V in range(n + 1 - X - L):
				I = n - X - V - L
				s = L * 50 + X * 10 + V * 5 + I
				d[s].append((L, X, V, I))
	print(len(d))
