import sys
(n, m) = map(int, input().split())
dist = [n] * n
candy_cnt = [0] * n
for (a, b) in (map(int, l.split()) for l in sys.stdin):
	if a > b:
		b += n
	if dist[a - 1] > b - a:
		dist[a - 1] = b - a
	candy_cnt[a - 1] += 1
dist *= 2
candy_cnt *= 2
ans = [0] * n
for i in range(n):
	time = 0
	for j in range(i, i + n):
		if candy_cnt[j] == 0:
			continue
		t = (candy_cnt[j] - 1) * n + dist[j] + j - i
		if time < t:
			time = t
	ans[i] = time
print(*ans)
