import sys
input = lambda : sys.stdin.readline().strip()

def mi():
	return map(int, input().split())

def li():
	return list(mi())

def ii():
	return int(input())

def fi():
	return float(input())

def si():
	return input().strip()
for _ in range(ii()):
	(n, k) = mi()
	a = [x + 1 for x in range(k)] + [x for x in range(k - 1, k - (n - k) - 1, -1)]
	p = [x + 1 for x in range(k)]

	p = p[:k - count] + list(reversed(p[k - count:]))
