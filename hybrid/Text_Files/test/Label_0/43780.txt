(n, m) = map(int, input().split())
l = []
for i in range(n):
	l.append(list(map(int, input().split())))
ans = [[0] * m for _ in range(n)]
a = []
su = []
f = 1
for j in range(m):
	x = 0
	for i in range(n):
		if l[i][j] == 0:
			break
		x += 1
	if x == n:
		a.append(j)
for i in range(n):
	if l[i].count(1) == m:
		for j in a:
			ans[i][j] = 1
			su.append([i, j])
k = [[0] * m for _ in range(n)]
for i in su:
	k[i[0]] = [1] * m
	for j in range(n):
		k[j][i[1]] = 1
if l != k:
	print('NO')
else:
	print('YES')
	for i in range(n):
		print(*ans[i])
