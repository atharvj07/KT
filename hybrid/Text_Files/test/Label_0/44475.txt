s = input()
count = 0
n = len(s)
res = []
last = s[0]
for i in range(n):
	if s[i] == last and count == 0:
		count = 1
		res.append(s[i])
	elif s[i] == last and count == 1:
		count = 2
		res.append(s[i])
	elif s[i] != last and count == 1:
		res.append(s[i])
	elif s[i] != last and count == 2:
		count = 3
		res.append(s[i])
	elif s[i] != last and count == 3:
		count = 1
		res.append(s[i])
	last = s[i]
print(''.join(res))
