import java.math.BigInteger;
import java.util.Scanner;

public class D {

	public static void main(String[] args) {

		Scanner s = new Scanner(System.in);
		
		String[] firstLine = s.nextLine().split(" ");
		
		boolean[] done = new boolean[Integer.parseInt(firstLine[0])];
		int K = Integer.parseInt(firstLine[1]);
		
		BigInteger ways = new BigInteger("1");
		
		if (K != 0 && K != 1) {
		
		int[] cycles = new int[done.length];
		
		for (int i = 1; i < done.length; i++) {
			
			if (!done[i]) {
				
				done[i] = true;
				int cycleLength = 1;
				long start = (i * (long) K) % done.length;
				
				while (start != i) {
					if (start < 0 || start > done.length) {
						
						System.out.println(start);
						
					}
					done[(int) start] = true;
					start *= K;
					start %= done.length;
					cycleLength++;
					
				}
				
				cycles[cycleLength]++;
				
			}
			
		}
		long way = 1;
		for (int i = 0; i < cycles.length; i++) {
			
			if (cycles[i] != 0) {
				
				for (int j = 0; j < cycles[i]; j++) {
					
					way *= cycles[i] * i + 1;
					way %= 1000000007;
					
				}
				
			}
			
		}
		System.out.println(way);
		
		}else if (K == 1) {
			
			int p = done.length;
			ways = new BigInteger(String.valueOf(p));
			ways = ways.modPow(new BigInteger(String.valueOf(p)), new BigInteger("1000000007"));

			System.out.println(ways);
			
		}else{
			
			int p = done.length;
			ways = new BigInteger(String.valueOf(p));
			ways = ways.modPow(new BigInteger(String.valueOf(p - 1)), new BigInteger("1000000007"));

			System.out.println(ways);
			
		}
		
		s.close();
		System.exit(0);
		
	}

}
