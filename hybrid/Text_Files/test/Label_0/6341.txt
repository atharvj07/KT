#include <iostream>
#include <bits/stdc++.h>
typedef long long ll;
using namespace std;

int main()
{   
    ios::sync_with_stdio(0);
    cin.tie(0);cout.tie(0);
    int tests=1;
    cin>>tests;
    while(tests--)
    {
        ll l,r;
        cin>>l>>r;
        ll ans=0;
        while(l!=0 || r!=0){
            ans+=(r-l);
            r=r/10;
            l=l/10;
        }
        cout<<ans<<endl;
        

        
        
    }

    return 0;
}
