#include <bits/stdc++.h>
using namespace std;
int n, sq;
vector<vector<int> > edges;
bool inTree[100001];
int depth[100001] = {};
int anc[100001];
bool inSet[100001];
void dfs(int v, int iter) {
  inTree[v] = true;
  int cnt = 0;
  depth[v] = iter;
  for (int i = 0; i < edges[v].size(); i++) {
    if (!inTree[edges[v][i]]) {
      anc[edges[v][i]] = v;
      dfs(edges[v][i], iter + 1);
    }
    if (depth[edges[v][i]] < depth[v]) cnt++;
  }
  if (cnt >= sq - 1) {
    int cur = v;
    cout << 2 << endl;
    vector<int> ans;
    ans.push_back(cur + 1);
    int trg;
    for (int i = 0; i < edges[v].size(); i++) {
      if (depth[v] - depth[edges[v][i]] >= sq - 1 && inTree[edges[v][i]]) {
        trg = edges[v][i];
        break;
      }
    }
    while (cur != trg) {
      cur = anc[cur];
      ans.push_back(cur + 1);
    }
    cout << ans.size() << endl;
    for (int i = 0; i < ans.size(); i++) {
      cout << ans[i] << " ";
    }
    exit(0);
  }
  for (int i = 0; i < edges[v].size(); i++) {
    if (inSet[edges[v][i]]) return;
  }
  inSet[v] = true;
}
int main() {
  int m;
  cin >> n >> m;
  edges.resize(n);
  sq = ceil(sqrt(n));
  for (int i = 0; i < m; i++) {
    int boof1, boof2;
    cin >> boof1 >> boof2;
    boof1--;
    boof2--;
    edges[boof1].push_back(boof2);
    edges[boof2].push_back(boof1);
  }
  dfs(0, 0);
  cout << 1 << endl;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (inSet[i]) {
      cout << i + 1 << " ";
      cnt++;
    }
    if (cnt == sq) break;
  }
}
