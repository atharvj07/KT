t = int(input())
for _ in range(t):
	n = int(input())
	c = list(map(int, input().split()))
	n1 = 0
	n2 = 0
	mx = float('inf')
	my = float('inf')
	cst = float('inf')
	sum = 0
	for i in range(1, n + 1):
		sum += c[i - 1]
		if i % 2 == 0:
			n1 += 1
			mx = min(mx, c[i - 1])
		else:
			n2 += 1
			my = min(my, c[i - 1])
		cst = min(cst, sum + mx * (n - n1) + my * (n - n2))
	print(cst)
