def quick_mod(b, p):
	s = 1
	while p != 0:
		if p & 1:
			s = s * b % mod
		p = p // 2
		b = b * b % mod
	return s
s = input().split()
n = int(s[0])
N = 100000
mod = 1000000007
enter = []
for i in range(1, N + 5):
	enter.append(0)
for i in range(1, n):
	s1 = input().split()
	enter[int(s1[0])] += 1
	enter[int(s1[1])] += 1
tot = 0
for i in range(1, n + 1):
	if enter[i] == 1:
		tot += 1
print((n + tot) * quick_mod(2, n - tot) % mod)
