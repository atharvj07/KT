def inputSeperator(inputStr):
	return list(map(int, inputStr.split(' ')))
temp = inputSeperator(input())
n = temp[0]
x = temp[1]
y = temp[2]
number = list(input())
number = list(map(int, number))
number = number[n - x:n]
if number[x - y - 1] == 1:
	print(sum(number) - 1)
else:
	print(sum(number) + 1)
