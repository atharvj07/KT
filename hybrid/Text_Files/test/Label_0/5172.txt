#include <bits/stdc++.h>
using namespace std;
int a[1005], b[1005], c[1005];
int main() {
  int n, p;
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &b[i]);
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    memset(a, 0, sizeof(a));
    a[i]++;
    p = b[i];
    while (1) {
      a[p]++;
      if (a[p] == 2) {
        c[++ans] = p;
        break;
      }
      p = b[p];
    }
  }
  for (int i = 1; i <= ans; i++) {
    printf("%d ", c[i]);
  }
  printf("\n");
  return 0;
}
