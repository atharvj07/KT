n = int(input())
a = list((i for i in input()))
b = list((i for i in input()))
c = list(a)
d = list(b)
do_nothing = 0
a.append('2')
b.append('2')
cn = 0
for i in range(n):
	if a[i] == b[i]:
		do_nothing = 1
	elif a[i] == b[i + 1] and a[i + 1] == b[i]:
		cn += 1
		(a[i], a[i + 1]) = (a[i + 1], a[i])
	elif a[i] != b[i]:
		cn += 1
		a[i] = b[i]
print(cn)
