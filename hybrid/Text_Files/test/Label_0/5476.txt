#include <iostream>
#include <algorithm>
using namespace std;

int main(){
	int n, c[100][100];
	cin >> n;
	for(int i = 0; i < n; ++i)
	for(int j = 0; j < n; ++j){
		cin >> c[i][j];
	}
	long long s = 0;
	for(int i = 1; i < n; ++i)
	for(int j = 0; j < i; ++j){
		s += min(c[i][j], c[j][i]);
	}
	cout << s << endl;
}
