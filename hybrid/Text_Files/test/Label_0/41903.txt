import io
import os
from collections import Counter, defaultdict, deque

def solve(XYZ1, XYZ2):
	(X1, Y1, Z1) = XYZ1
	(X2, Y2, Z2) = XYZ2
	ans = 0
	temp = min(Z1, Y2)
	Z1 -= temp
	Y2 -= temp
	ans += temp * 2
	temp = max(Y1 + Z2 - (X1 + Y1 + Z1), 0)
	ans -= temp * 2
	return ans
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
TC = int(input())
for tc in range(1, TC + 1):
	XYZ1 = [int(x) for x in input().split()]
	XYZ2 = [int(x) for x in input().split()]
	ans = solve(XYZ1, XYZ2)
	print(ans)
