#include <bits/stdc++.h>
const long long LL_INF = (long long)2e18 + 5;
using namespace std;
const long long N = 3e3 + 10;
vector<string> moves = {"D", "L", "R", "U"};
long long n, m, can;
char v[1003][1003];
long long dx[] = {0, -1, 1, 0}, dy[] = {1, 0, 0, -1};
bool chk(long long y, long long x) {
  return y < n and y >= 0 and x < m and x >= 0 and v[y][x] != '*';
}
void solve() {
  cin >> n >> m >> can;
  string ans;
  pair<long long, long long> rob = {-1, -1};
  vector<vector<long long>> d(n, vector<long long>(m, -1));
  for (long long i = (long long)(0); i < (long long)(n); i++)
    for (long long j = (long long)(0); j < (long long)(m); j++)
      cin >> v[i][j], v[i][j] == 'X' ? rob = {i, j} : rob = rob;
  bool ok = 0;
  for (long long i = (long long)(0); i < (long long)(4); i++)
    ok |= chk(rob.first + dy[i], rob.second + dx[i]);
  ok &= (can & 1 ^ 1);
  if (!ok) {
    cout << "IMPOSSIBLE\n";
    return;
  }
  auto b = [&](long long y, long long x) {
    queue<pair<long long, long long>> q;
    q.push({y, x});
    d[y][x] = 0;
    while (!q.empty()) {
      y = q.front().first, x = q.front().second;
      q.pop();
      for (long long i = (long long)(0); i < (long long)(4); i++) {
        long long yy = y + dy[i], xx = x + dx[i];
        if (chk(yy, xx) and d[yy][xx] == -1) {
          q.push({yy, xx}), d[yy][xx] = d[y][x] + 1;
        }
      }
    }
  };
  long long y = rob.first, x = rob.second;
  b(y, x);
  for (long long i = (long long)(0); i < (long long)(can); i++) {
    ok = 0;
    for (long long j = (long long)(0); j < (long long)(4); j++) {
      long long yy = y + dy[j], xx = x + dx[j];
      if (chk(yy, xx) and d[yy][xx] <= can - i - 1) {
        y += dy[j], x += dx[j];
        ans += moves[j];
        ok = 1;
        break;
      }
    }
    if (!ok) {
      cout << "IMPOSSIBLE\n";
      return;
    }
  }
  cout << ans << '\n';
}
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  solve();
}
