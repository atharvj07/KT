from collections import defaultdict, deque
from sys import stdin, stdout

def input():
	return stdin.buffer.readline().rstrip()

def out(var):
	stdout.write(str(var) + '\n')
t = int(input())
(stats, arr, unchosen) = ([], [], [])
for _ in range(t):
	(numv, nume) = map(int, input().split())
	d = defaultdict(list)
	u = []
	for _ in range(nume):
		(t, x, y) = map(int, input().split())
		if t == 1:
			d[x].append(y)
		elif t == 0:
			u.append((x, y))
	stats.append(numv)
	unchosen.append(u)
	arr.append(d)
for i in range(len(arr)):
	(d, u, numv) = (arr[i], unchosen[i], stats[i])

	def get_indegrees(graph):
		indegree = {node: 0 for node in range(1, numv + 1)}
		for nodes in graph.values():
			for node in nodes:
				indegree[node] += 1
		return indegree

	def top_sort_bfs(graph):
		ans = []
		indegree = get_indegrees(graph)
		q = deque([node for node in indegree if indegree[node] == 0])
		while q:
			node = q.popleft()
			ans.append(node)
			for neighbor in graph[node]:
				indegree[neighbor] -= 1
				if indegree[neighbor] == 0:
					q.append(neighbor)
		return [] if len(ans) != len(graph) else ans
	toporder = top_sort_bfs(d)
	if len(toporder) == 0:
		out('NO')
		continue
	out('YES')
	for head in d:
		for tail in d[head]:
			out(f'{head} {tail}')
	position = {node: idx for (idx, node) in enumerate(toporder)}
	for (v1, v2) in u:
		if position[v1] < position[v2]:
			out(f'{v1} {v2}')
		else:
			out(f'{v2} {v1}')
