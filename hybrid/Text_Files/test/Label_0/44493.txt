import time

def main():
	t = i_input()
	for tt in range(t):
		(W, H) = li_input()
		(x1, y1, x2, y2) = li_input()
		(x1, x2) = (min(x1, x2), max(x1, x2))
		(y1, y2) = (min(y1, y2), max(y1, y2))
		(w, h) = li_input()
		if w <= W - (x2 - x1):
			rx = max(x1, W - x2)
			mx = max(0, w - rx)
		else:
			mx = 10 ** 9
		if h <= H - (y2 - y1):
			ry = max(y1, H - y2)
			my = max(0, h - ry)
		else:
			my = 10 ** 9
		out = min(mx, my)
		print(out if out != 10 ** 9 else -1)

def i_input():
	return int(input())

def l_input():
	return input().split()

def li_input():
	return list(map(int, l_input()))

def il_input():
	return list(map(int, l_input()))
TT = time.time()
main()
