from collections import defaultdict
from itertools import accumulate
import sys
input = sys.stdin.readline
inf = 100000000000000000
mod = 998244353
for CASES in range(int(input())):
	space = input().split()
	(n, m) = map(int, space)
	MAP = []
	for i in range(n):
		MAP.append(input().strip())
	black = True
	ans = []
	cir1 = -2
	cir2 = -2
	for i in range(n):
		for j in range(n):
			if i > j and MAP[i][j] == MAP[j][i]:
				(cir1, cir2) = (i, j)
	charm = -2
	chara = -2
	charb = -2
	for i in range(n):
		if len(set(MAP[i])) == 3:
			charm = i
			for j in range(n):
				if MAP[i][j] == 'a':
					chara = j
				elif MAP[i][j] == 'b':
					charb = j
	charm += 1
	chara += 1
	charb += 1
	cir1 += 1
	cir2 += 1
	if m % 2 == 1:
		for i in range(m + 1):
			if i % 2 == 1:
				ans.append(1)
			else:
				ans.append(2)
	elif cir1 != -1:
		for i in range(m + 1):
			if i % 2 == 1:
				ans.append(cir1)
			else:
				ans.append(cir2)
	elif charm != -1:
		if m % 4 == 0:
			for i in range(m + 1):
				if i % 2 == 0:
					ans.append(charm)
				elif i % 4 == 1:
					ans.append(chara)
				elif i % 4 == 3:
					ans.append(charb)
		else:
			for i in range(m + 1):
				if i % 2 == 1:
					ans.append(charm)
				elif i % 4 == 0:
					ans.append(chara)
				elif i % 4 == 2:
					ans.append(charb)
	else:
		black = False
	if not black:
		print('NO')
		continue
	print('YES')
	for i in range(m + 1):
		print(ans[i], end=' ')
	print()
