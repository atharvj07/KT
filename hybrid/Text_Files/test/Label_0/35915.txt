import java.util.*;

public class CodeForces {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for(int u = 0; u < t; u++)
        {
            int n = scanner.nextInt(), m = scanner.nextInt();
            Cell[][] table = new Cell[n][m];
            ArrayList<Cell> list = new ArrayList<Cell>();
            for(int i = 0; i < n; i++)
            {
                for(int j = 0; j < m; j++)
                {
                    int x = scanner.nextInt();
                    table[i][j] = new Cell(i, j, x);
                    list.add(table[i][j]);
                }
            }
            Collections.sort(list);
            int indexCounter = 0;
            for(int i = 0; i < m; i++)
            {
                swap(table, list.get(indexCounter), table[list.get(indexCounter).x][i]);
                indexCounter++;
            }
            for(int i = 0; i < n; i++)
            {
                for(int j = 0; j < m; j++)
                {
                    System.out.print(table[i][j].value + " ");
                }
                System.out.println();
            }
        }
    }

    public static void swap(Cell[][] table, Cell cell1, Cell cell2)
    {
        table[cell1.x][cell1.y] = cell2;
        table[cell2.x][cell2.y] = cell1;
        int x1 = cell1.x, x2 = cell2.x, y1 = cell1.y, y2 = cell2.y;
        cell1.x = x2;
        cell1.y = y2;
        cell2.x = x1;
        cell2.y = y1;
    }
}

class Cell implements Comparable<Cell>
{
    int x, y, value;
    public Cell(int x, int y, int value)
    {
        this.x = x;
        this.y = y;
        this.value = value;
    }
    @Override
    public int compareTo(Cell cell)
    {
        if(this.value > cell.value)
            return 1;
        else if(this.value < cell.value)
            return -1;
        else return 0;
    }
}
