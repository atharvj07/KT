import  java.util.*;
public class Main {
	final static int maxn=(int)1e5+10;
	static long a[]=new long [maxn];
	static long sum[]=new long [maxn];
	static long gkd[]=new long [maxn];
	static long gcd(long x,long y)
	{
		if(x==0)return y;
		else if(y==0)return x;
		else
			return gcd(y,x%y);
	}
 public static void main(String args[])
 {
	 Scanner input=new Scanner(System.in);
		 int n=input.nextInt();
		 for(int i=1;i<=n;i++)a[i]=input.nextLong();
		for(int i=1;i<=n;i++)sum[i]=0;
		sum[n]=a[n];
		for(int i=n-1;i>=1;i--)
		{
			sum[i]=gcd(sum[i+1],a[i]);
		}
		long ans=1;
		for(int i=1;i<=n-1;i++)
		{
			//System.out.println(sum[i]+" ");
			gkd[i]=(a[i]*sum[i+1])/gcd(a[i],sum[i+1]);
		}
		ans=gkd[1];
		for(int i=2;i<=n-1;i++)
			ans=gcd(ans,gkd[i]);
		System.out.println(ans);;
 }
}
