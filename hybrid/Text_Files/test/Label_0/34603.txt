import java.util.*;
public class Main {
    public static void main(String args[]) {
        Scanner s=new Scanner(System.in);
        int T=s.nextInt();
        for(int t=0;t<T;t++)
        {
            int strength=s.nextInt();
            int intelligence=s.nextInt();
            int experience=s.nextInt();
            System.out.println(numCharacters(strength,intelligence,experience));
        }
    }
    public static int numCharacters(int strength,int intelligence,int experience){
        if(experience==0)
        {
            if(strength>intelligence)
                return 1;
            else return 0;
        }
        int minX=(int)Math.ceil((0.0+experience+(intelligence-strength))/2);
        if((experience+(intelligence-strength))%2==0)
        {
            minX++;
        }
        if(minX>0)
        {
            int ans=experience-minX+1;
            if(ans<0)
            {
                return 0;
            }
            else return ans;
        }
        return experience+1;
    }
}
