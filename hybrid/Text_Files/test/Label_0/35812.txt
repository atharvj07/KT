import java.util.Scanner;
public class code{
public static void main(String args[]){
Scanner sc  = new Scanner(System.in);
int n = sc.nextInt();
String a =sc.next();
boolean ok=true;


for(int i=0;i<a.length();i++){
	if(a.charAt(i)!='4' && a.charAt(i)!='7'){
		ok=false;
		break;
	}}
	
if(ok){
	int q=0;
	int w=0;
	for(int i=0;i<n/2;i++){
		q+=Integer.parseInt(a.charAt(i)+"");
	}
	for(int i=n/2;i<n;i++){
		w+=Integer.parseInt(a.charAt(i)+"");
	}
	if(q==w){
		System.out.println("YES");
	}
	else{
		System.out.println("NO");
	}}
	


else{
		System.out.println("NO");
}}}
