import math
T = int(input())
for t in range(T):
	N = int(input())
	(ra, pa, sa) = map(int, input().split())
	seq = input()
	rb = 0
	pb = 0
	sb = 0
	for n in range(N):
		if seq[n] == 'R':
			rb += 1
		elif seq[n] == 'P':
			pb += 1
		else:
			sb += 1
	win = 0
	out = ['' for n1 in range(N)]
	for n2 in range(N):
		curr = seq[n2]
		if curr == 'R':
			if pa > 0:
				out[n2] = 'P'
				pa -= 1
				win += 1
		elif curr == 'P':
			if sa > 0:
				out[n2] = 'S'
				sa -= 1
				win += 1
		elif ra > 0:
			out[n2] = 'R'
			ra -= 1
			win += 1
	if win >= math.ceil(N / 2):
		print('YES')
		for n3 in range(N):
			if out[n3] == '':
				if pa > 0:
					out[n3] = 'P'
					pa -= 1
				elif sa > 0:
					out[n3] = 'S'
					sa -= 1
				else:
					out[n3] = 'R'
					ra -= 1
		print(''.join(out))
	else:
		print('NO')
