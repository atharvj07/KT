#include <bits/stdc++.h>
double const PI = 4 * atan(1.0);
using namespace std;
long long a[5];
char ch[5];
long long tinh(long long u, char dau, long long v) {
  if (dau == '+') return u + v;
  return u * v;
}
long long tinh1() {
  long long res = 1ll << 50;
  long long A = tinh(a[0], ch[1], a[1]);
  long long B = tinh(a[2], ch[2], a[3]);
  res = min(res, tinh(A, ch[3], B));
  A = tinh(A, ch[2], a[2]);
  res = min(res, tinh(A, ch[3], a[3]));
  return res;
}
int main() {
  for (int i = 0; i < 4; i++) cin >> a[i];
  for (int i = 1; i < 4; i++) cin >> ch[i];
  long long MIN = 1ll << 50;
  sort(a, a + 4);
  do {
    MIN = min(MIN, tinh1());
  } while (next_permutation(a, a + 4));
  cout << MIN;
}
