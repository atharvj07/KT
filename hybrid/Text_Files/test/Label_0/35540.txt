import java.util.HashMap;
import java.util.Scanner;
import java.util.StringTokenizer;


public class CodeA {

	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		HashMap <Character,Integer>map = new HashMap<Character,Integer>();
			
		String one =in.nextLine();
		int i;
		
		for(i=0; i < one.length(); i++) {
			if(map.containsKey(one.charAt(i)))
				map.put(one.charAt(i), map.get(one.charAt(i))+1);
			else
				map.put(one.charAt(i), 1);
		}
		
		String two = in.nextLine();
		
		for(i=0; i < two.length(); i++) {
			if(map.containsKey(two.charAt(i)))
				map.put(two.charAt(i), map.get(two.charAt(i))+1);
			else
				map.put(two.charAt(i), 1);
		}
		
		String fin= in.nextLine();
		
		boolean flag=true;
		
		for(i=0; i < fin.length(); i++) {
			if(map.containsKey(fin.charAt(i)))
				map.put(fin.charAt(i), map.get(fin.charAt(i))-1);
			else{
				flag=false;
				break;
			}
		}
		
		
		for(char j: map.keySet()) if(map.get(j)!=0) {flag=false;}
		
		System.out.println((flag)?"YES":"NO");

	}

}
