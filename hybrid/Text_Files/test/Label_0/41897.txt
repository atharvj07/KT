import math
for _ in range(int(input())):
	(aa, bb) = map(int, input().split())
	if aa != bb:
		a = max(aa, bb)
		b = min(aa, bb)
		xx = a // b
		if a % b == 0 and (xx and (not xx & xx - 1)):
			ans = 0
			ans += int(math.log(xx, 8))
			xx //= 8 ** int(math.log(xx, 8))
			ans += int(math.log(xx, 4))
			xx //= 4 ** int(math.log(xx, 4))
			ans += int(math.log(xx, 2))
			print(ans)
		else:
			print(-1)
	else:
		print(0)
