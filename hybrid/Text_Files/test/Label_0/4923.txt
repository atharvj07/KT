#include<iostream>
#include<string>
#define r(i,n) for(int i=0;i<n;i++)
using namespace std;
int a[26],b[10],ans;
string s;
int main(){
  cin>>s;
  r(i,s.size())
    if(isdigit(s[i]))b[s[i]-'0']++;
    else a[s[i]-'a']++;
  r(i,10){
    while(b[i]){
      int p=i,t=0;
      while(b[p]&&p<10)p++,t++;
      for(int j=i;j<p;j++)b[j]--;
      if(t<=2)ans+=t;
      else ans+=3;
    }
  }
  r(i,26){
    while(a[i]){
      int p=i,t=0;
      while(a[p]&&p<26)p++,t++;
      for(int j=i;j<p;j++)a[j]--;
      if(t<=2)ans+=t;
      else ans+=3;
    }
  }
  cout<<ans<<endl;
}
