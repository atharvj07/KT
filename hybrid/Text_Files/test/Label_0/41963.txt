(x1, y1) = map(int, input().split())
(x2, y2) = map(int, input().split())
n = int(input())
wind = input()
prepx = [0] * (n + 1)
prepy = [0] * (n + 1)
for i in range(1, n + 1):
	prepx[i] = prepx[i - 1]
	prepy[i] = prepy[i - 1]
	if wind[i - 1] == 'U':
		prepy[i] += 1
	elif wind[i - 1] == 'D':
		prepy[i] -= 1
	elif wind[i - 1] == 'L':
		prepx[i] -= 1
	else:
		prepx[i] += 1

def vaivar(dienas):
	daudz = dienas // n
	x = daudz * prepx[n]
	y = daudz * prepy[n]
	atlik = dienas % n
	x += prepx[atlik]
	y += prepy[atlik]
	dx = x1 - x2
	dy = y1 - y2
	if dienas >= abs(x + dx) + abs(y + dy):
		return True
	return False
lower = 0
uper = int(10 ** 18)
if not vaivar(uper):
	print(-1)
	exit(0)
while lower < uper:
	guess = (lower + uper) // 2
	if vaivar(guess):
		uper = guess
	else:
		lower = guess + 1
print(uper)
