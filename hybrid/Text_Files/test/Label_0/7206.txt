#include <bits/stdc++.h>
using namespace std;
int a, b, f, g, h, i, j;
double c, d, e, u, v;
bool E;
struct Point {
  double x, y;
  double angle;
  Point() {}
  Point(double x, double y) : x(x), y(y) {}
  friend bool operator<(const Point &x, const Point &y) {
    return x.angle - y.angle < (1e-9);
  }
  friend struct Point operator-(const Point &x, const Point &y) {
    struct Point xx;
    xx = Point(x.x - y.x, x.y - y.y);
    return xx;
  }
};
set<Point> G;
set<Point>::iterator T;
set<Point>::iterator ch;
set<Point>::iterator Y;
set<Point>::iterator RRR;
Point w[9];
Point o;
double X(Point x, Point y) { return x.x * y.y - x.y * y.x; }
double JiJiao(Point x) { return atan2(x.x - o.x, x.y - o.y); }
set<Point>::iterator last(set<Point>::iterator x) {
  set<Point>::iterator ii;
  set<Point>::iterator jj;
  set<Point>::iterator kk;
  kk = G.end();
  kk--;
  ii = x;
  if (ii == G.begin())
    jj = kk;
  else
    jj = --ii;
  return jj;
}
set<Point>::iterator next(set<Point>::iterator x) {
  set<Point>::iterator ii;
  set<Point>::iterator jj;
  ii = x;
  jj = ++ii;
  if (jj == G.end()) jj = G.begin();
  return jj;
}
bool Mid(Point x, Point y, Point z) { return X(y - x, z - x) <= 1e-9; }
int main() {
  scanf("%d", &a);
  a -= 3;
  for (i = 1; i <= 3; i++) {
    scanf("%d%lf%lf", &b, &c, &d);
    u += c;
    v += d;
    w[i] = Point(c, d);
  }
  o = Point(u / 3.0, v / 3.0);
  for (i = 1; i <= 3; i++) w[i].angle = JiJiao(w[i]);
  for (i = 1; i <= 3; i++) G.insert(w[i]);
  while (a--) {
    scanf("%d%lf%lf", &b, &c, &d);
    if (b == 1) {
      w[1] = Point(c, d);
      w[1].angle = JiJiao(w[1]);
      Y = G.lower_bound(w[1]);
      if (Y == G.end()) Y = G.begin();
      T = last(Y);
      if (Mid(w[1], (*T), (*Y)))
        continue;
      else {
        ch = Y;
        Y = next(Y);
        while (Y != ch && Mid((*last(Y)), w[1], (*Y))) {
          G.erase(last(Y));
          Y = next(Y);
        }
        ch = T;
        T = last(T);
        while (T != ch && Mid((*next(T)), (*T), w[1])) {
          G.erase(next(T));
          T = last(T);
        }
        G.insert(w[1]);
      }
    } else {
      w[1] = Point(c, d);
      w[1].angle = JiJiao(w[1]);
      Y = G.lower_bound(w[1]);
      if (Y == G.end()) Y = G.begin();
      T = last(Y);
      if (Mid(w[1], (*T), (*Y)))
        printf("YES\n");
      else
        printf("NO\n");
    }
  }
  return 0;
}
