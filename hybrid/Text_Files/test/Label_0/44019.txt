def calc_range(x, y, d):
	l1 = x - d
	r1 = x + d
	l2 = y - d
	r2 = y + d
	points = sorted([l1 * l2, l1 * r2, r1 * l2, r1 * r2])
	return (points[0], points[3])

def main():
	(a, b) = map(int, input().split())
	(c, d) = map(int, input().split())
	l = 0.0
	r = 10000000000.0
	for iter in range(100):
		m = (l + r) / 2.0
		(minA, maxA) = calc_range(a, d, m)
		(minB, maxB) = calc_range(b, c, m)
		if maxA < minB or maxB < minA:
			l = m
		else:
			r = m
	print(r)
main()
