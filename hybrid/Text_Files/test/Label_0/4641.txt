#include <bits/stdc++.h>
using namespace std;
long long n, q, temp;
long long get(long long sz, long long temp, long long delta) {
  long long k = sz;
  if ((temp <= k) && (temp % 2 == 1)) {
    return temp / 2 + 1 + delta;
  } else {
    if (sz % 2 == 0) return get(sz / 2, temp / 2, sz / 2 + delta);
    if (sz % 2 == 1) return get(sz / 2 + sz % 2, temp / 2 + 1, sz / 2 + delta);
  }
}
int main() {
  cin >> n >> q;
  for (long long i = 1; i <= q; i++) {
    cin >> temp;
    cout << get(n, temp, 0) << endl;
    ;
  }
}
