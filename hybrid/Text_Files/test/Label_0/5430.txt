#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, b, bot[105];
  cin >> n >> b;
  for (int i = 0; i < n; ++i) cin >> bot[i];
  int maxi = 0;
  for (int i = 0; i < n; ++i) maxi = max(bot[i], maxi);
  bool bo = 1;
  for (int i = 0; i < n; ++i) {
    b -= maxi - bot[i];
  }
  if (b < 0)
    cout << "-1\n";
  else {
    for (int i = 0; i < n; ++i) {
      printf("%.6f\n", maxi - bot[i] + b * 1.0 / n);
    }
  }
  return 0;
}
