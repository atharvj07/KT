import java.io.*;
import java.util.*;

public class Main implements Runnable {

	public void _main() throws IOException {
		long[] k = new long[3];
		for (int i = 0; i < 3; i++)
			k[i] = nextInt();
		long[] t = new long[3];
		for (int i = 0; i < 3; i++)
			t[i] = nextInt();
		int n = nextInt();
		Queue<Event> events = new PriorityQueue<Event>();
		Queue<Person>[] waiting = new Queue[3];
		for (int i = 0; i < 3; i++)
			waiting[i] = new PriorityQueue<Person>();
		Person[] p = new Person[n];
		for (int i = 0; i < n; i++) {
			int c = nextInt();
			p[i] = new Person(c, i);
			events.offer(new Event(c, 0, i, true));
		}
		long res = 0;
		while (!events.isEmpty()) {
			Event e = events.poll();
			int tp = e.type;
			if (e.comes) {
				if (k[tp] > 0) {
					--k[tp];
					events.offer(new Event(e.time + t[tp], tp, e.id, false));
				}
				else {
					waiting[tp].offer(p[e.id]);
				}
			}
			else {
				++k[tp];
				if (tp < 2) {
					events.offer(new Event(e.time, tp + 1, e.id, true));
				}
				else {
					res = Math.max(res, e.time - p[e.id].c);
				}
			}
			for (tp = 0; tp < 3; tp++)
				while (k[tp] > 0 && !waiting[tp].isEmpty()) {
					Person z = waiting[tp].poll();
					--k[tp];
					events.offer(new Event(e.time + t[tp], tp, z.id, false));
				}
		}
		out.println(res);
	}
	
	class Person implements Comparable<Person> {
		long c;
		int id;
		Person(long c, int id) {
			this.c = c;
			this.id = id;
		}
		public int compareTo(Person o) {
			if (c != o.c)
				return c < o.c ? -1 : 1;
			return 0;
		}
	}
	
	class Event implements Comparable<Event> {
		boolean comes; //false if leaves
		int type;
		long time;
		int id;
		
		Event(long time, int type, int id, boolean comes) {
			this.time = time;
			this.type = type;
			this.id = id;
			this.comes = comes;			
		}
		public int compareTo(Event o) {
			if (time != o.time)
				return time < o.time ? -1 : 1;
			if (comes ^ o.comes)
				return comes ? 1 : -1;
			return 0;
		}
	}


	private BufferedReader in;
	private PrintWriter out;
	private StringTokenizer st;

	private String next() throws IOException {
		while (st == null || !st.hasMoreTokens()) {
			String rl = in.readLine();
			if (rl == null)
				return null;
			st = new StringTokenizer(rl);
		}
		return st.nextToken();
	}

	private int nextInt() throws IOException {
		return Integer.parseInt(next());
	}

	private long nextLong() throws IOException {
		return Long.parseLong(next());
	}

	private double nextDouble() throws IOException {
		return Double.parseDouble(next());
	}

	public static void main(String[] args) {
		Locale.setDefault(Locale.UK);
		new Thread(new Main()).start();
	}

	public void run() {
		try {
			in = new BufferedReader(new InputStreamReader(System.in));
			out = new PrintWriter(System.out);

			_main();

			out.close();
		} catch (Exception e) {
			e.printStackTrace();
			System.exit(202);
		}
	}

}
