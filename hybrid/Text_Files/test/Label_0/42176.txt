import math

def isper(n):
	z = int(math.sqrt(n))
	if z ** 2 == n:
		return True
	else:
		return False
t = int(input())
for _ in range(t):
	n = int(input())
	if n % 2 != 0:
		print('NO')
	elif isper(n / 2) == True:
		print('YES')
	elif n % 4 != 0:
		print('NO')
	elif isper(n / 4) == True:
		print('YES')
	else:
		print('NO')
