#include <bits/stdc++.h>
using namespace std;
void solve() {
  int a, b, c;
  cin >> a >> b >> c;
  int ans = 0;
  if (a) a--, ans++;
  if (b) b--, ans++;
  if (c) c--, ans++;
  if (a < b) swap(a, b);
  if (a < c) swap(a, c);
  if (a && b) a--, b--, ans++;
  if (a && c) a--, c--, ans++;
  if (b && c) b--, c--, ans++;
  if (a && b && c) a--, b--, c--, ans++;
  cout << ans << "\n";
  return;
}
int main() {
  long long t;
  cin >> t;
  while (t--) {
    solve();
  }
  return 0;
}
