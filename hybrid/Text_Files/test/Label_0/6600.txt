#define ll long long
const int MAX=1e5+10;
char p[5]="ANOT";
char a[MAX];
int n;
int pre[5];
int A,O,T,N;
char ck[5];
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        //cin>>n;
        scanf("%s",a+1);
        n=strlen(a+1);
        A=count(a+1,a+n+1,'A');
        O=count(a+1,a+n+1,'O');
        T=count(a+1,a+n+1,'T');
        N=count(a+1,a+n+1,'N');
        ll minn=0x3f3f3f3f3f3f;
        for(int L=1;L<=24;L++)
        {
            ll sum=0;
            memset(pre,0,sizeof pre);//cout<<n<<endl;
            for(int i=1;i<=n;i++)
            {
                for(int j=0;j<4;j++)
                {
                    if(p[j]==a[i])
                    {
                        pre[j]++;
                        break;
                    }
                    sum+=pre[j];

                }

            }
            if(sum<minn)
            {
                minn=sum;
                memcpy(ck,p,sizeof p);
            }

            next_permutation(p,p+4);
            //cout<<sum<<endl;
        }
        for(int i=0;i<4;i++)
        {
            if(ck[i]=='A')
            {
                for(int i=1;i<=A;i++)putchar('A');
            }
            else if(ck[i]=='O')
            {
                for(int i=1;i<=O;i++)putchar('O');
            }
            else if(ck[i]=='N')
            {
                for(int i=1;i<=N;i++)putchar('N');
            }
            else if(ck[i]=='T')
            {
                for(int i=1;i<=T;i++)putchar('T');
            }
        }cout<<'\n';
    }
}
