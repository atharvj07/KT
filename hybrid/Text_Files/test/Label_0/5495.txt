template<class T> inline string toString(const vector<T>& x) {
	stringstream ss;
	REP(i,x.size()){
		if(i!=0)ss<<" ";
		ss<< x[i];
	}
	return ss.str();
}
template<class T> inline string toString(const vector<vector<T> >& map) {
	stringstream ss;
	REP(i,map.size()){
		if(i!=0)ss<<BR;
		ss<< toString(map[i]);
	}
	return ss.str();
}

ll INF=1LL<<58;
class Main{
public:	
	void run(){
                        int N;cin >> N;
                        vector<int> ts(N);REP(i,N) cin >> ts[i];

                        sort(ALL(ts));
                        ll res=0;

                        int Mi=N-1;
                        for(;Mi-1>=2;Mi-=2){
                            res+=ts[0]+ts[Mi]+min(ts[0]+ts[Mi-1],2*ts[1]);
                        }
                        if(Mi>=2){//odd
                            res+=ts[0]+ts[1]+ts[2];
                        }else if(Mi==1){//even
                            res+=ts[1];    
                        }else{
                            res+=ts[0];
                        }
		cout << res <<endl;
	}
};
 int main(){
