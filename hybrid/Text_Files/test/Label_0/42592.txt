n = int(input())
lst = list(map(int, input().strip().split(' ')))
lst.sort()
l = [[lst[0]]]
for i in range(1, n):
	j = 0
	while True:
		if j == len(l):
			l.append([lst[i]])
			break
		elif l[j][-1] != lst[i]:
			l[j].append(lst[i])
			break
		else:
			j += 1
s = 0
for i in range(len(l)):
	s += len(l[i]) - 1
print(s)
