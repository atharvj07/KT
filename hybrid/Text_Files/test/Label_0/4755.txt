#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
typedef long double ld;

inline int in(){int x, y; y = scanf("%d", &x); return x;}

const int N = 120021;

int d[N];
vector <int> g[N];
bool mark[N];

void dfs(int v) {
	mark[v] = true;
	for(int u : g[v])
		if(!mark[u])
			dfs(u);
}

int main() {
	int n = in(), m = in();
	while(m--) {
		int u = in() - 1, v = in() - 1;
		g[u].push_back(v);
		g[v].push_back(u);
		d[u]++;
		d[v]++;
	}
	bool bad = false;
	int cnt = 0;
	int a = -1, b = -1;
	for(int i = 0; i < n; i++) {
		bad |= d[i]%2;
		cnt += (d[i] - 2)/2;
		if(d[i] == 4)
			if(a == -1)
				a = i;
			else
				b = i;
	}
	if(!bad && (cnt >= 3 || (cnt == 2 && b == -1)))
		cout << "Yes\n";
	else if(bad || cnt < 2)
		cout << "No\n";
	else {
		mark[a] = true;
		dfs(b);
		cerr << 1 + a << ", " << b + 1 << (*min_element(mark, mark + n)) << endl;
		if(*min_element(mark, mark + n) == false)
			cout << "Yes\n";
		else
			cout << "No\n";
	}
}
