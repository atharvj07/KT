#include <bits/stdc++.h>
using namespace std;
const int N = 1e4 + 5;
vector<int> ke[N], we[N];
int t, n, m, res, M[N], vis[N];
bool DFS(int u, int& MAX) {
  vis[u] = t;
  for (int i = 0; i < ke[u].size(); i++) {
    int v = ke[u][i], w = we[u][i];
    if (w > MAX) continue;
    if (M[v] == 0) {
      M[v] = u;
      return true;
    }
  }
  for (int i = 0; i < ke[u].size(); i++) {
    int v = ke[u][i], w = we[u][i];
    if (w > MAX) continue;
    if (vis[M[v]] < t && DFS(M[v], MAX) == true) {
      M[v] = u;
      return true;
    }
  }
  return false;
}
bool check(int& MAX) {
  for (int i = 1; i <= n; i++) {
    M[i] = vis[i] = 0;
  }
  res = 0;
  t = 1;
  for (int i = 1; i <= n; i++) {
    if (DFS(i, MAX) == true) t++, res++;
  }
  return res == n;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  cin >> n >> m;
  vector<int> S;
  for (int i = 1; i <= m; i++) {
    int u, v, w;
    cin >> u >> v >> w;
    S.push_back(w);
    ke[u].push_back(v);
    we[u].push_back(w);
  }
  sort(S.begin(), S.end());
  S.erase(unique(S.begin(), S.end()), S.end());
  int l = 0, r = S.size() - 1, ans = -1;
  while (l <= r) {
    int m = (l + r) / 2;
    if (check(S[m]) == true) {
      ans = S[m];
      r = m - 1;
    } else
      l = m + 1;
  }
  cout << ans;
}
