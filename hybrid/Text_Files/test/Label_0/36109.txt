

import java.io.*;
import java.util.*;
public class D1000 {
    public static void main(String args[])throws IOException
    {
        Scanner sc=new Scanner(System.in);
        long mod=998244353;
        long comb[][]=new long[1001][1001];
       
        for(int i=1;i<=1000;i++)
        {
            for(int j=0;j<=1000;j++)
            {
                if(j==1)
                    comb[i][j]=i;
                else if(j==0)
                    comb[i][j]=1;
                else
                    comb[i][j]=(comb[i-1][j]+comb[i-1][j-1])%mod;
            }
        }
        //System.out.println(comb[10][5]);
        int n=sc.nextInt();
        int arr[]=new int[n+1];
        long dp[]=new long[n+1];dp[n]=1;
        for(int i=0;i<n;i++)
        {
            arr[i]=sc.nextInt();
        }
        long tot=0;
        for(int i=n-1;i>=0;i--)
        {
            if(arr[i]>0)
            {
                for(int j=i+arr[i]+1;j<=n;j++)
                {
                    dp[i]=dp[i]+(comb[j-i-1][arr[i]]*dp[j]);
                    dp[i]%=mod;
                }
            }
            tot=(tot+dp[i])%mod;
        }
        System.out.println(tot);
    }

}
