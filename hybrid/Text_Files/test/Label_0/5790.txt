int N, M, s, t;
vector<int> G[100010];
int ca[100010], cb[100010];

vector<int> dijkstra(int s) {
    vector<int> ret(N, INF);
    ret[s] = 0;

    // dist, point
    priority_queue< pii, vector<pii>, greater<pii> > q;
    q.push(pii(0, s));

    while(!q.empty()) {
        pii t = q.top(); q.pop();
        int cost = t.first, cur = t.second;
        for(int to : G[cur]) {
            if(ret[to] > ret[cur] + 1) {
                ret[to] = ret[cur] + 1;
                q.push(pii(ret[to], to));
            }
        }
    }

    return ret;
}

signed main() {
    cin >> N >> M >> s >> t; s--; t--;
    rep(i,0,M) {
        int x, y; cin >> x >> y; x--; y--;
        G[x].push_back(y);
        G[y].push_back(x);
    }

    vector<int> da = dijkstra(s), db = dijkstra(t);
    rep(i,0,N) if(da[i] != INF) ca[ da[i] ]++;
    rep(i,0,N) if(db[i] != INF) cb[ db[i] ]++;

    
    int dist = da[t], ans = 0;
    repq(i,0,dist-2) {
        int j = dist - 2 - i;
        ans += ca[i] * cb[j];
    }
