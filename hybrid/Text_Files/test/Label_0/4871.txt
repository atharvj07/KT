#include <bits/stdc++.h>
#pragma warning(disable : 4786)
using namespace std;
int s[5][5];
int n = 5, k;
int nod = 0;
int f[5] = {0, 1, 2, 3, 4};
void init() {
  int bull = 0;
  memset(s, 0, sizeof(s));
  for (int i = 0; i < n; i++)
    for (int j = 0; j < n; j++) {
      cin >> s[i][j];
    }
  int bk = 120;
  while (bk--) {
    for (int i = bull = 0; i < n; i++)
      for (int j = i + 1; j < n; j += 2)
        bull += s[f[j]][f[j - 1]] + s[f[j - 1]][f[j]];
    if (bull > nod) nod = bull;
    next_permutation(f, f + n);
  }
}
void solve() { cout << nod << endl; }
int main() {
  init();
  solve();
  return 0;
}
