t = int(input().strip())
for _ in range(t):
	(n, x) = map(int, input().strip().split(' '))
	a = tuple(map(int, input().strip().split(' ')))
	len = n + x + 1
	b = [False] * len
	for e in a:
		if e < len:
			b[e] = True
	ans = None
	for ans in range(1, len):
		if b[ans] == False:
			if x > 0:
				x -= 1
			else:
				ans -= 1
				break
	print(ans)
