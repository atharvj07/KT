l = int(input())
n = input()
res = int(n)
toCheck = [l // 2, (l + 1) // 2]
while toCheck[0] > 1 and n[toCheck[0]] == '0':
	toCheck[0] -= 1
while toCheck[1] < l - 1 and n[toCheck[1]] == '0':
	toCheck[1] += 1
for pos in toCheck:
	v1 = n[:pos]
	v2 = n[pos:]
	if v1[0] != '0' and v2[0] != '0':
		res = min(res, int(v1) + int(v2))
print(res)
