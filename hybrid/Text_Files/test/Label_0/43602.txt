from sys import *
inp = lambda : stdin.readline()

def main():
	(n, y, an) = (int(inp()), 0, 0)
	l = [int(i) for i in inp().split()]
	a = [0 for i in range(1001)]
	for i in l:
		a[i] += 1
	ans = []
	while y < n:
		for i in range(1001):
			if a[i]:
				a[i] -= 1
				ans.append(i)
				y += 1
	for i in range(1, len(ans)):
		if ans[i] > ans[i - 1]:
			an += 1
	print(an)
main()
