for i in range(int(input())):
	(n, x) = map(int, input().split())
	a = list(map(int, input().split()))
	(total_sum, flag) = (0, 0)
	(first_index, last_index) = (-1, 0)
	for j in range(n):
		total_sum += a[j]
		if a[j] % x != 0:
			flag = 1
			if first_index == -1:
				first_index = j + 1
			last_index = j
	if flag == 0:
		print(-1)
	elif total_sum % x != 0:
		print(n)
	else:
		print(max(n - first_index, last_index))
