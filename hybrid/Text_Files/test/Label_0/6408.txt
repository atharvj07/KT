long long gcd(long long a, long long b) { return (a == 0) ? b : gcd(b % a, a); }
long long cnt1(long long x) {
  long long cnt = 0;
  while (x > 0) x &= x - 1, cnt++;
  return cnt;
}
template <class T>
istream &operator>>(istream &is, vector<T> &v) {
  for (long long i = 0; i < v.size(); i++) is >> v[i];
  return is;
}
template <class T>
ostream &operator<<(ostream &os, vector<T> v) {
  for (long long i = 0; i < v.size(); i++) os << v[i] << ' ';
  os << '\n';
  return os;
}
struct coord {
  long long x, y;
  friend istream &operator>>(istream &is, coord &a) { return is >> a.x >> a.y; }
  coord operator+(coord &other) { return {x + other.x, y + other.y}; }
  coord operator-(coord &other) { return {x - other.x, y - other.y}; }
  coord() { x = y = 0; }
  coord(long long x_, long long y_) {
    x = x_;
    y = y_;
  }
  long long size() { return ((x) * (x)) + ((y) * (y)); }
  coord operator-() { return {-x, -y}; }
};
signed main() {
  srand(time(0));
  ios_base::sync_with_stdio(0), cin.tie(0);
  long long n;
  cin >> n;
  vector<pair<long long, coord>> a(n);
  for (long long i = 0; i < n; i++) {
    cin >> a[i].second;
    a[i].first = i;
  }
  coord cur;
  vector<long long> ans(n);
  do {
    random_shuffle(a.begin(), a.end());
    ans[a[0].first] = 1;
    cur = a[0].second;
    for (long long i = 1; i < n; i++) {
      coord a1 = cur + a[i].second, a2 = cur - a[i].second;
      if (a1.size() < a2.size()) {
        cur = a1;
        ans[a[i].first] = 1;
      } else {
        cur = a2;
        ans[a[i].first] = -1;
      }
    }
  } while (cur.size() > ((1500000LL) * (1500000LL)));
  cout << ans;
}
