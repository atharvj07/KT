n = int(input())
c = list(map(int, input().split()))
c.append(-5)
arr = [0]
for i in range(n):
	if c[i] != c[i + 1]:
		arr.append(c[i])
arr_r = [0] + arr[::-1]
l = len(arr)
store = [0] * l
up = [0] * l
for i in range(1, l):
	for j in range(1, l):
		if arr[i] == arr_r[j]:
			store[j] = up[j - 1] + 1
		else:
			store[j] = max(store[j - 1], up[j])
	up = store
	store = [0] * l
print(l - 1 - 1 - up[-1] // 2)
