//ICPC B
#include <bits/stdc++.h>
using namespace std;
using ll=long long;
const ll mod=1e9+7;

int main(){
    vector<int> answers;
    while(1){
        int h,w;
        cin>>h>>w;
        if(h==0&&w==0) break;
        string r;
        pair<int,int> pos[256];
        for(int i=0;i<h;i++){
            cin>>r;
            for(int j=0;j<w;j++){
                if(r[j]!='_') pos[r[j]]={i,j};
            }
        }
        string s;
        cin>>s;
        int x=0,y=0;
        int ans=s.length();
        for(int i=0;i<s.length();i++){
            ans+=abs(y-pos[s[i]].first)+abs(x-pos[s[i]].second);
            y=pos[s[i]].first;
            x=pos[s[i]].second;
        }
        answers.push_back(ans);
    }

    for(auto ans:answers){
        cout<<ans<<endl;
    }
    return 0;
}
