#include <bits/stdc++.h>
using namespace std;
vector<vector<int>> graph;
vector<int> used, dist, pr;
int mdfs(int v) {
  used[v] = true;
  int res = v;
  for (int u : graph[v]) {
    if (used[u]) {
      continue;
    }
    pr[u] = v;
    dist[u] = dist[v] + 1;
    int cur = mdfs(u);
    if (dist[cur] > dist[res]) {
      res = cur;
    }
  }
  return res;
}
bool check(int v, int k) {
  used[v] = true;
  int deg = 0;
  for (int u : graph[v]) {
    if (used[u]) {
      continue;
    }
    ++deg;
    if (!check(u, k - 1)) {
      return false;
    }
  }
  if (deg == 0) {
    return k == 0;
  }
  return deg >= 3;
}
int main() {
  int n, k;
  cin >> n >> k;
  graph.resize(n);
  int from, to;
  for (int i = 0; i < n - 1; ++i) {
    cin >> from >> to;
    --from;
    --to;
    graph[from].push_back(to);
    graph[to].push_back(from);
  }
  used.resize(n);
  dist.resize(n);
  pr.resize(n);
  int l1 = mdfs(0);
  used.assign(n, 0);
  dist.assign(n, 0);
  pr.assign(n, 0);
  int l2 = mdfs(l1);
  if (dist[l2] != 2 * k) {
    cout << "No";
    return 0;
  }
  for (int i = 0; i < k; ++i) {
    l2 = pr[l2];
  }
  used.assign(n, 0);
  cout << (check(l2, k) ? "Yes" : "No");
  return 0;
}
