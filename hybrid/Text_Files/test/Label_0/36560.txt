import java.util.Scanner;

public class Arrow {
    double x,y;
    Arrow(double x, double y){
        this.x=x;
        this.y=y;
    }
    static void rotate(double x, double y, double cos, double sin, double px, double py){
        double x1= x*cos - y*sin+px;
        double y1= x*sin + y*cos+py;
        System.out.println(x1+" "+y1);
    }

    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        double px= cin.nextInt();
        double py= cin.nextInt();
        double vx= cin.nextInt();
        double vy= cin.nextInt();
        double a= cin.nextInt();
        double b= cin.nextInt();
        double c= cin.nextInt();
        double d= cin.nextInt();
        Arrow[] arrows= new Arrow[7];
        arrows[0]= new Arrow(b,0);
        arrows[1]= new Arrow(0,a/2);
        arrows[2]= new Arrow(0,c/2);
        arrows[3]= new Arrow(-d,c/2);
        arrows[4] = new Arrow(-d,-c/2);
        arrows[5]= new Arrow(0,-c/2);
        arrows[6]= new Arrow(0,-a/2);
        double v= Math.sqrt(vx*vx + vy*vy);
        double cos= vx/v;
        double sin= vy/v;
        for(int i=0;i<7;i++){
            rotate(arrows[i].x,arrows[i].y,cos,sin,px,py);
        }
    }
}
