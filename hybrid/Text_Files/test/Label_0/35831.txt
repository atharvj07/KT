import java.util.*;
import java.io.*;
public class CF691F {
	public static void main(String[] args) throws IOException {
		in.init(System.in);
		int n = in.nextInt();
		HashMap<Integer, Integer> vals = new HashMap<Integer, Integer>();
		for(int i = 0; i < n; i ++)
		{
			int x = in.nextInt();
			vals.put(x, vals.containsKey(x)? vals.get(x)+1: 1);
		}
		int m = in.nextInt();
		int b = 3_000_001;
		long max = ((long)n)*(n-1l);
		long[] ways = new long[b];
		
		for(int e: vals.keySet())
			for(int i = e; i < b; i += e)
				if(vals.containsKey(i/e) && i/e!=e)
					ways[i] += (long)vals.get(i/e)*vals.get(e);

		for(int e: vals.keySet())
			if(vals.get(e)>1 && e*e<b)
				ways[e*e] += ((long)vals.get(e)*(vals.get(e)-1l));
		
		for(int i = 1; i<b; i ++)
			ways[i] += ways[i-1];
		PrintWriter out = new PrintWriter(System.out);
		for(int i = 0; i < m; i ++)
			out.println(max-ways[in.nextInt()-1]);
		out.close();
	}
	public static class in {
		static BufferedReader reader;
		static StringTokenizer tokenizer;
		static void init(InputStream in) {
			reader = new BufferedReader(
			new InputStreamReader(in));
			tokenizer = new StringTokenizer("");
		}

		static String next() throws IOException {
			while (!tokenizer.hasMoreTokens()) {
				tokenizer = new StringTokenizer(
				reader.readLine());
			}
			return tokenizer.nextToken();
		}
		static int nextInt() throws IOException {
			return Integer.parseInt(next());
		}
		static double nextDouble() throws IOException {
			return Double.parseDouble(next());
		}
		static long nextLong() throws IOException {
			return Long.parseLong(next());
		}
	}
}
