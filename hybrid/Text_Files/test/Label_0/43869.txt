s = input()
a = 0
r = ''
l = []
for i in s:
	if 48 <= ord(i) <= 57 or ord(i) == 46:
		r += i
	elif len(r) > 0:
		if len(r) > 3 and r[-3] == '.':
			a += float(''.join(r[:-3].split('.')) + r[-3:])
		else:
			a += float(''.join(r.split('.')) + '.00')
		r = ''
if len(r) > 0:
	if len(r) > 3 and r[-3] == '.':
		a += float(''.join(r[:-3].split('.')) + r[-3:])
	else:
		a += float(''.join(r.split('.')) + '.00')
a = round(float(a), 2)
sa = str(a)
li = sa.split('.')
r = ''
for i in range(1, len(li[0]) + 1):
	r += li[0][-i]
	if i % 3 == 0 and i < len(li[0]):
		r += '.'
r = r[::-1]
if int(li[1]) > 0:
	r += '.' + li[1]
	if len(li[1]) == 1:
		r += '0'
