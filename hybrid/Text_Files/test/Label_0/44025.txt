import bisect
import collections
import copy
import enum
import functools
import heapq
import itertools
import math
import random
import re
import sys
import time
import string
from typing import List
sys.setrecursionlimit(3001)
input = sys.stdin.readline
n = int(input())
ans = 0
sl = []
arr = list(map(int, input().split()))
brr = list(map(int, input().split()))
g = collections.defaultdict(list)
for (i, c) in enumerate(brr):
	g[c].append(i + 1)
q = collections.deque()
for c in g[-1]:
	q.append(c)
while q:
	ind = q.popleft()
	sl.append(ind)
	for nc in g[ind]:
		q.append(nc)
pre = []
aft = []
crr = arr[:]
for ind in sl[::-1]:
	if crr[ind - 1] > 0:
		ans += crr[ind - 1]
		if brr[ind - 1] != -1:
			crr[brr[ind - 1] - 1] += crr[ind - 1]
		pre.append(ind)
	else:
		ans += crr[ind - 1]
		aft.append(ind)
print(ans)
print(*pre + aft[::-1])
