#include<iostream>
using namespace std;
int main(){
  while(1){
    int n,w,h,x,y,max=0,m[1000][1000]={};
    cin>>n;
    if(n==0)break;
    cin>>w>>h;
    for(int i=0;i<n;i++){
      cin>>x>>y;
      m[x][y]++;
    }
    cin>>x>>y;
    for(int i=1;i<=w-x+1;i++){
      for(int j=1;j<=h-y+1;j++){
	int cnt=0;
	for(int k=i;k<i+x;k++){
	  for(int l=j;l<j+y;l++){
	    if(m[k][l]==1)cnt++;
	  }
	}
	if(max<cnt)max=cnt;
      }
    }
    cout<<max<<endl;
  }
  return 0;
}
