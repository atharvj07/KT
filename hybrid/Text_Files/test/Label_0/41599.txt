import sys

def input():
	return sys.stdin.readline()[:-1]

def main():
	for _ in range(int(input())):
		(n, m) = map(int, input().split())
		B = [list(map(int, input().split())) for k in range(n)]
		R = [B[k].count(1) for k in range(n)]
		C = [0] * m
		for l in range(m):
			f = 0
			for k in range(n):
				f += B[k][l]
			C[l] = f
		count = 0
		while True:
			prev = count
			for l in range(m):
				for k in range(n):
					if R[k] == 0 and C[l] == 0:
						count += 1
						R[k] = 1
						C[l] = 1
			if count == prev:
				break
		print('Ashish' if count % 2 == 1 else 'Vivek')
main()
