N = int(5000.0 + 3)
n = int(input())
res = [0] * N
cum = list(([0] * N for _ in range(N)))
a = list(map(int, input().split()))
for i in range(n):
	c = [0] * N
	for j in range(i, n):
		c[a[j]] += 1
		cum[i][j] = c[a[j]]
for i in range(n):
	(curr_max, curr_idx) = (0, N - 1)
	j = 0
	while cum[i][j] == 0:
		j += 1
	while j < n:
		if cum[i][j] > curr_max or (cum[i][j] == curr_max and a[j] < curr_idx):
			curr_max = cum[i][j]
			curr_idx = a[j]
		res[curr_idx] += 1
		j += 1
print(*res[1:n + 1])
