signed main(){
  Int n,d;
  cin>>n>>d;
  vector<Int> t(n),f(n);
  for(Int i=0;i<n;i++) cin>>t[i]>>f[i],f[i]--;

  auto NG=[](){cout<<-1<<endl;exit(0);};
  Int ans=0,tmp=0,lst=0,num=0;
  for(Int i=0;i<n;i++){
    if(abs(tmp-f[i])>(t[i]-lst)) NG();
    if(tmp+f[i]<=(t[i]-lst)){
      ans+=tmp*num;
      num=1;
    }else{
      if(num==d) NG();
      ans+=abs(t[i]-lst)*num;
      num++;      
    }
    lst=t[i];
    tmp=f[i];
  }
  // cout<<ans<<" "<<tmp<<" "<<num<<endl;
  ans+=tmp*num;
  cout<<ans<<endl;
  return 0;
}
