import java.io.*;
import java.util.*;

public class C {
  public static void main(String[] args) {
    int n;
    int[] a = null, sorted = null, mark = null;
    List<List> outputs = null;

    try (BufferedReader in = new BufferedReader(new InputStreamReader(System.in))) {
      String line = in.readLine();
      n = Integer.parseInt(line);
      a = new int[n];
      sorted = new int[n];
      mark = new int[n];
      outputs = new LinkedList<List>();

      line = in.readLine();
      String[] numbers = line.split("\\s+");
      for (int i = 0; i < n; i++) {
        a[i] = Integer.parseInt(numbers[i]);
        sorted[i] = a[i];
      }
    } catch (IOException e) {
      return;
    }
    
    Arrays.sort(sorted);
    PrintWriter out = new PrintWriter(System.out);
    int subSequenceCount = 0;

    for (int i = 0; i < n; i++) {
      if (mark[i] > 0) { continue; }
      
      List<Integer> outputLine = new LinkedList<Integer>();
      outputs.add(outputLine);
      subSequenceCount++;
      int j = i, count = 0;
      
      while (mark[j] == 0) {
        mark[j] = subSequenceCount;
        outputLine.add(j);
        j = findIndexOf(a[j], n, sorted);
      }
    }

    out.println(subSequenceCount);
    for (List<Integer> list : outputs) {
      out.printf("%d ", list.size());
      for (Integer i : list) {
        out.printf("%d ", i + 1);
      }
      out.println();
    }
    out.flush();
    out.close();
  }

  private static int findIndexOf(int value, int n, int[] a) {
    int begin = 0, end = n - 1;
    while (begin <= end) {
      if (begin == end) {
        if (a[begin] == value) {
          return begin;
        } else {
          return -1;
        }
      }
      int mid = begin + (end - begin)/2;

      if (value == a[mid]) {
        return mid;
      } else if (value < a[mid]) {
        end = mid - 1;
      } else {
        begin = mid + 1;
      }
    }
    return -1;
  }
}
