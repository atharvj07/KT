for t in range(int(input())):
	n = int(input())
	c = list(map(int, input().split()))
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	curlen = 0
	lstcur = 0
	ans = 0
	for i in range(1, n):
		curlen = abs(a[i] - b[i]) + 1 + c[i]
		if a[i] != b[i]:
			curlen = max(curlen, c[i] + 1 + lstcur - abs(a[i] - b[i]))
			lstcur = curlen
		ans = max(ans, curlen)
		lstcur = curlen
	print(ans)
