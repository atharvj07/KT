import sys
import math
from collections import defaultdict as dd
mod = 1000000007
T = 1
T = int(sys.stdin.readline())
for _ in range(T):
	(n, x) = map(int, input().split())
	l = list(map(int, input().split()))
	ff = 1
	for i in range(n - 1):
		if l[i] > l[i + 1]:
			ff = 0
			break
	if ff:
		print('0')
		continue
	ans = 0
	flg = 1
	for i in range(n - 1):
		if l[i] > x:
			(x, l[i]) = (l[i], x)
			ans += 1
		if l[i] > l[i + 1]:
			flg = 0
			break
		t = 1
		for j in range(i, n - 1):
			if l[j] > l[j + 1]:
				t = 0
				break
		if t:
			break
	if flg:
		print(ans)
	else:
		print(-1)
