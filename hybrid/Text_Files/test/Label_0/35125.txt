import java.util.*;
import java.io.*;

public class C {
  static int[] uf;
  static int find(int x) {
    if(uf[x]==x) { return x; }
    return uf[x]=find(uf[x]);
  }
  static void mix(int x, int y) {
    uf[find(x)] = find(y);
  }
  static int i(String s) { return Integer.parseInt(s); }

  static List<List<Integer>> E;
  static int[] P;
  static boolean[] B;

  static void root(int x) {
    for(Integer y : E.get(x)) {
      if(y==P[x]) { continue; }
      P[y] = x;
      root(y);
    }
  }

  static List<Integer> ANS = new ArrayList<Integer>();
  static void walk(int x) {
    B[x] = !B[x];
    ANS.add(x);
  }

  static void f(int x) {
    walk(x); // IN
    for(Integer y : E.get(x)) {
      if(y==P[x]) { continue; }
      f(y); // visit child
      walk(x); // back to root
    }

    if(B[x] && P[x]==-1) {
      ANS.remove(ANS.size()-1);
      B[x] = !B[x];
    } else if(B[x]) {
      walk(P[x]);
      walk(x);
    }
  }

  public static void main(String[] args) throws Exception {
    BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    String[] arr = in.readLine().split(" ");
    int n = i(arr[0]);
    int m = i(arr[1]);

    E = new ArrayList<List<Integer>>();
    for(int i=0; i<n; i++) {
      E.add(new ArrayList<Integer>());
    }
    uf = new int[n];
    for(int i=0; i<n; i++) {
      uf[i] = i;
    }

    for(int i=0; i<m; i++) {
      arr = in.readLine().split(" ");
      int x = i(arr[0])-1;
      int y = i(arr[1])-1;
      if(find(x) != find(y)) {
        E.get(x).add(y);
        E.get(y).add(x);
        mix(x, y);
      }
    }

    arr = in.readLine().split(" ");
    B = new boolean[n];
    for(int i=0; i<n; i++) {
      B[i] = i(arr[i])==1;
    }

    boolean[] HAS_BLACK = new boolean[n];
    for(int i=0; i<n; i++) {
      if(B[i]) {
        HAS_BLACK[find(i)] = true;
      }
    }
    int rt = 0;
    int has_black = 0;
    for(int i=0; i<n; i++) {
      if(HAS_BLACK[i]) {
        has_black++;
        rt = i;
      }
    }
    if(has_black >= 2) {
      System.out.println(-1);
      return;
    }

    P = new int[n];
    P[rt] = -1;
    root(rt);
    f(rt);

    StringBuilder out = new StringBuilder();
    out.append(String.format("%d\n", ANS.size()));
    for(int i=0; i<ANS.size(); i++) {
      out.append(String.format("%d%c", 1+ANS.get(i), i==ANS.size()-1?'\n':' '));
    }
    System.out.print(out);
  }
}
