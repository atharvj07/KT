for _ in range(int(input())):
	(m, n) = list(map(int, input().split()))
	lst = []
	for i in range(m):
		lst.append(list(map(int, input().split())))
	row = set()
	col = set()
	for i in range(m):
		for j in range(n):
			if lst[i][j] == 1:
				row.add(i)
				col.add(j)
	chang = True
	c = 0
	while chang:
		for i in range(m):
			for j in range(n):
				if lst[i][j] == 0 and i not in row and (j not in col):
					c += 1
					row.add(i)
					col.add(j)
					continue
		else:
			chang = False
	if c % 2 == 0:
		print('Vivek')
	else:
		print('Ashish')
