构造题
%2L不影响答案
绕两下不影响答案 
	贪心绕，先把两边都可以绕的分配给只能绕前面的，如果碰到个只能绕后面的就改成和绕后面的配
特判是否需要额外一轮来保证每个点都被经过
2018.4.11 
*/
#include <bits/stdc++.h>
#define rep(i, s, t) for(int i = s; i <= t; ++i)

using namespace std;
 
const int M = 3e5 + 10;
 
typedef long long LL;
int n, both, res; LL ans;
LL L, x[M], t;
bool oL[M], oR[M];
 
int main() {
	scanf("%d%lld", &n, &L); L <<= 1;
	ans = L;
	rep(i, 1, n) scanf("%lld", &x[i]);
	
	rep(i, 1, n) {
		scanf("%lld", &t);
		ans += (t + L - 1) / L * L;
		t %= L;
		if(t > 0 && t <= 2 * x[i]) oL[i] = 1;
		if(t > 0 && t <= L - 2 * x[i]) oR[i] = 1;
	}
	
	rep(i, 1, n-1) {
		if(res && oR[i]) {
    		if(oL[i]) ++both;
			--res; ans -= L;
		} else
		if(!oL[i] && oR[i]) {
			if(both) --both, ++res;
		} else
		if(oL[i]) ++res;
	}
	if(oR[n]) ans -= L;
	printf("%lld\n", ans);
	return 0;
}
