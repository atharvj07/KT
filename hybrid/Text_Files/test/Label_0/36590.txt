import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.util.*;
import java.lang.*;

public class CF442B 
{
    public static void main(String[] args) throws Exception {
        Scanner scanner = new Scanner(System.in);
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter writer = new BufferedWriter(new OutputStreamWriter(System.out));

        String str = "*" + reader.readLine() + "*";

        int[] suma = new int[5005];
        int[] sumb = new int[5005];

        suma[0] = sumb[0] = 0;
        for (int ii = 1; ii < str.length(); ii++) {
            suma[ii] = suma[ii-1] + (str.charAt(ii) == 'a' ? 1 : 0);
            sumb[ii] = sumb[ii-1] + (str.charAt(ii) == 'b' ? 1 : 0);
        }

        int ans = 0;
        for (int ii = 0; ii < str.length(); ii++) {
            for (int jj = ii; jj < str.length(); jj++) {
                if (ii > 0 && jj == ii) continue;
                int sum = 0;
                if (ii > 0) sum += suma[ii];
                if (jj > 0) sum += sumb[jj] - sumb[ii];
                if (jj < str.length()-2) sum += suma[str.length()-1] - suma[jj];
                ans = Math.max(ans, sum);
            }
        }

        writer.write(ans + "\n");
        writer.flush();
    }
}
