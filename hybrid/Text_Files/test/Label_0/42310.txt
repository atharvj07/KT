import io, os
from sys import stdin, stdout
for _ in range(int(stdin.readline())):
	xxx = stdin.readline()
	(n, m) = map(int, stdin.readline().split())
	ar = []
	for i in range(n):
		ar.append(list(map(int, stdin.readline().split())))
	a = [[] for i in range(m)]
	for i in range(m):
		for j in range(n):
			a[i].append([ar[j][i], j])
		a[i].sort()
	mx = 0
	for i in range(n):
		mx = max(mx, max(ar[i]))
	l = 1
	r = mx
	ans = 0
	while l <= r:
		mid = l + (r - l) // 2
		d = {}
		check = False
		check1 = False
		for i in range(m):
			for j in range(n):
				if a[i][j][0] >= mid:
					if a[i][j][1] not in d:
						d[a[i][j][1]] = True
					else:
						check = True
						break
			if a[i][-1][0] < mid:
				check1 = True
				break
		if check1 or not check:
			r = mid - 1
		else:
			ans = mid
			l = mid + 1
	print(ans)
