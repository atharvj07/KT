import java.util.Scanner;

public class Problem {

    public static void main(String[] args) {
    
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        int t = cin.nextInt();
        int k = cin.nextInt();
        int d = cin.nextInt();
 
        if (n - (d / t)*k > k)
            System.out.println("YES");
        else
            System.out.println("NO");
       
    }
}
