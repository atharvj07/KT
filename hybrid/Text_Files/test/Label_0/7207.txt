#include<iostream>
using namespace std;
int main()
{
	int il[4] = { 0 };
	int ir[4] = { 0 };
	double r, l;
	while (cin >> l >> r){
		if (l >= 1.1){
			il[0]++;
		}
		else if(l>=0.6){
			il[1]++;
		}
		else if (l >= 0.2){
			il[2]++;
		}
		else{
			il[3]++;
		}
		if (r >= 1.1){
			ir[0]++;
		}
		else if (r >= 0.6){
			ir[1]++;
		}
		else if (r >= 0.2){
			ir[2]++;
		}
		else{
			ir[3]++;
		}
	}
	for (int i = 0; i < 4; i++){
		cout << il[i] << " " << ir[i] << endl;
	}
}
