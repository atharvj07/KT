#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 7;
const int oo = 1e9 + 7;
void init() {
  ios_base::sync_with_stdio(0);
  cin.tie(NULL);
  cout.tie(NULL);
}
pair<int, int> a[maxn];
int main() {
  init();
  long long n, P1, P2, P3, T1, T2;
  cin >> n >> P1 >> P2 >> P3 >> T1 >> T2;
  for (int i = 1; i <= n; i++) cin >> a[i].first >> a[i].second;
  long long s = (a[1].second - a[1].first) * P1;
  for (int i = 2; i <= n; i++) {
    s += (a[i].second - a[i].first) * P1;
    int t = a[i].first - a[i - 1].second;
    t -= T1;
    if (t < 0) {
      s += (t + T1) * P1;
      continue;
    }
    s += T1 * P1;
    t -= T2;
    if (t < 0) {
      s += (t + T2) * P2;
      continue;
    }
    s += T2 * P2 + t * P3;
  }
  cout << s;
}
