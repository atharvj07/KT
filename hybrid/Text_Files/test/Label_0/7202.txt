#include <bits/stdc++.h>
using namespace std;
int main() {
  long n, k, count = 0;
  scanf("%ld%ld", &n, &k);
  long a[n];
  for (long i = 0; i < n; i++) {
    scanf("%ld", &a[i]);
  }
  long i = (n - 1);
  while (i > -1) {
    if (a[i] >= 2 * k) {
      count += a[i] / (2 * k);
      a[i] = a[i] % (2 * k);
      if (a[i] == 0) {
        i--;
      }
    } else {
      if (a[i] > k && a[i] < (2 * k)) {
        count++;
        a[i] = 0;
        i--;
      } else {
        count++;
        a[i] = 0;
        i--;
        if (i == -1) {
          break;
        }
        if (a[i] <= k) {
          a[i] = 0;
          i--;
        } else {
          a[i] -= k;
        }
      }
    }
  }
  printf("%ld", count);
  return 0;
}
