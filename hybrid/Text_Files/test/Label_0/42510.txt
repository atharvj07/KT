(N, M) = map(int, input().split())
X = [0] * N
Y = [-N] * N
Z = [0] * N
for _ in range(M):
	(a, b) = map(int, input().split())
	a -= 1
	b -= 1
	X[a] += 1
	if Y[a] < 0:
		Y[a] = (b - a) % N
	else:
		Y[a] = min((b - a) % N, Y[a])
for i in range(N):
	Z[i] = (X[i] - 1) * N + Y[i]
ANS = []
for i in range(N):
	ma = 0
	for k in range(N):
		j = (i + k) % N
		ma = max(Z[j] + k, ma)
	ANS.append(str(ma))
print(' '.join(ANS))
