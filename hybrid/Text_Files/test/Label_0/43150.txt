def calculaArestas(vertices, arestas, governos):
	ligacoes = {}
	for i in range(arestas):
		(x, y) = list(map(int, input().split(' ')))
		if x - 1 not in ligacoes:
			ligacoes[x - 1] = [y - 1]
		else:
			ligacoes[x - 1].append(y - 1)
		if y - 1 not in ligacoes:
			ligacoes[y - 1] = [x - 1]
		else:
			ligacoes[y - 1].append(x - 1)
	visitados = {}
	ciclos = {0: [], 1: []}
	for i in range(vertices):
		is_governos = False if i not in governos else True
		if i in visitados:
			continue
		ciclo = [i]
		if i not in ligacoes:
			visitados[i] = 1
			if is_governos:
				ciclos[0].append(ciclo)
			else:
				ciclos[1].append(ciclo)
			continue
		path = ligacoes[i]
		while len(path) > 0:
			no = path.pop(0)
			if no in visitados:
				continue
			visitados[no] = 1
			if no in governos:
				is_governos = True
			path = ligacoes[no] + path
			ciclo.append(no)
		if is_governos:
			ciclos[0].append(list(set(ciclo)))
		else:
			ciclos[1].append(list(set(ciclo)))
	ciclo_ordenado = sorted(ciclos[0], key=len)
	tamanho = len(ciclo_ordenado[-1])
	ciclo_ordenado.pop(len(ciclo_ordenado) - 1)
	for ciclo in ciclos[1]:
		tamanho += len(ciclo)
	total = tamanho * (tamanho - 1) // 2
	for j in ciclo_ordenado:
		total += len(j) * (len(j) - 1) // 2
	return total - arestas
(n, m, k) = list(map(int, input().split()))
governos = {c - 1: 1 for c in list(map(int, input().split()))}
print(calculaArestas(n, m, governos))
