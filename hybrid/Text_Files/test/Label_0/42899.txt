from collections import Counter
t = int(input())
for i in range(t):
	a = list(input())
	n = len(a)
	freq = Counter(a)
	ans = n - max(freq.values())
	for i in range(10):
		for j in range(10):
			c = 0
			maxi = 0
			for k in range(n):
				if c == 0 and a[k] == str(i):
					maxi += 1
					c = 1
				elif c == 1 and a[k] == str(j):
					maxi += 1
					c = 0
			if maxi % 2 != 0:
				maxi -= 1
			ans = min(ans, n - maxi)
	print(ans)
