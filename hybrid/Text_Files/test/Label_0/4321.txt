#include <bits/stdc++.h>
using namespace std;
#define N 2100
int len,ans;
char a[N],b[N];
int pre[N],nex[N],v[N];
int dis(int x,int y)
{
	if(x>y)swap(x,y);
	return min(y-x,x+len-y);
}
int main()
{
	//freopen("tt.in","r",stdin);
	scanf("%s%s",a,b);
	len=strlen(a);
	int fir=len,las=-1;
	for(int i=0;i<len;i++)if(b[i]=='1')
	{
		fir=min(fir,i);
		las=max(las,i);
	}
	if(fir==len)
	{
		for(int i=0;i<len;i++)if(a[i]=='1')
			return puts("-1"),0;
		return puts("0"),0;
	}
	int p=-1;
	for(int i=0;i<len;i++)
	{
		if(b[i]=='1')p=i;
		if(p==-1)pre[i]=len+i-las;
		else pre[i]=i-p;
	}
	p=len;
	for(int i=len-1;i>=0;i--)
	{
		if(b[i]=='1')p=i;
		if(p==len)nex[i]=len-i+fir;
		else nex[i]=p-i;
	}
	ans=1<<30;
	for(int i=0;i<len;i++)
	{
		int sum=0,tans=1<<30;
		memset(v,0,sizeof(v));
		for(int j=0;j<len;j++)
		{
			int t=(j+i)%len;
			if(a[j]=='1'&&b[t]=='0')
			{
				if(pre[j])
					v[pre[j]-1]=max(v[pre[j]-1],nex[j]);
				sum++;
			}
			if(a[j]=='0'&&b[t]=='1')
				sum++;
		}
		for(int j=len-1;j>=0;j--)
		{
			v[j]=max(v[j],v[j+1]);
			tans=min(tans,j+j+v[j]+dis(v[j],i));
			tans=min(tans,v[j]+v[j]+j+dis((len-j)%len,i));
		}
		ans=min(ans,tans+sum);
	}
	printf("%d\n",ans);
	return 0;
}
