import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.StringTokenizer;

public class SinglePush { // Template for CF
    public static class ListComparator implements Comparator<List<Integer>> {

        @Override
        public int compare(List<Integer> l1, List<Integer> l2) {
            for (int i = 0; i < l1.size(); ++i) {
                if (l1.get(i).compareTo(l2.get(i)) != 0) {
                    return l1.get(i).compareTo(l2.get(i));
                }
            }
            return 0;
        }
    }

    public static void main(String[] args) throws IOException {
        // Check for int overflow!!!!
        // Should you use a long to store the sum or smthn?
        BufferedReader f = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(System.out);

        int T = Integer.parseInt(f.readLine());
        for (int i = 0; i < T; i++) {
            int N = Integer.parseInt(f.readLine());
            StringTokenizer st = new StringTokenizer(f.readLine());
            List<Integer> listA = new ArrayList<>();
            for (int j = 0; j < N; j++) {
                int a = Integer.parseInt(st.nextToken());
                listA.add(a);
            }
            List<Integer> listB = new ArrayList<>();
            StringTokenizer ab = new StringTokenizer(f.readLine());
            for (int j = 0; j < N; j++) {
                int a = Integer.parseInt(ab.nextToken());
                listB.add(a);
            }
            List<Integer> listC = new ArrayList<>();
            for (int j = 0; j < N; j++) {
                int a = listB.get(j) - listA.get(j);
                listC.add(a);
            }
            boolean check = true;
            boolean pushed = false;
            int prev = 0;
            for (int j = 0; j < N; j++) {
                // System.out.println(listC.get(j));
                if (listC.get(j) < 0) {
                    check = false;
                    break;
                } else if (prev != 0 && (listC.get(j) != prev && listC.get(j) != 0)) {
                    check = false;
                    break;
                } else if (pushed && (listC.get(j) != prev && listC.get(j) != 0)) {
                    check = false;
                    break;
                }
                if (listC.get(j) != 0) {
                    pushed = true;
                }
                prev = listC.get(j);
            }
            if (check) {
                out.println("YES");
            } else {
                out.println("NO");
            }
        }
        out.close();
    }

}
