#include <iostream>
#include <algorithm>
using namespace std;
int main(void){
  int n,i,p[110][2],h[110],cx,cy,t,ch,u;
  cin >> n;
  for (i=0;i<n;i++) cin >> p[i][0] >> p[i][1] >> h[i];
  t=0;
  while (h[t]==0) t++;
  for (cx=0;cx<=100;cx++){
    for (cy=0;cy<=100;cy++){
      ch=abs(cx-p[t][0])+abs(cy-p[t][1])+h[t];
      for (i=0;i<n;i++){
        u = max(ch-abs(cx-p[i][0])-abs(cy-p[i][1]) , 0);
        if (u!=h[i]) break;
      }
      if (i==n) break;
    }
    if (cy!=101) break;
  }
  printf("%d %d %d\n",cx,cy,ch);
  return 0;
}
