      public static void dfs(int curr, int[] ls, boolean[] added)
      {
         //iterative bc Im dead inside

         IntegerStack stack = new IntegerStack();
         EdgeStack active = new EdgeStack();

   class IntegerStack
   {
      private int[] arr;
      int size = 0;
      
      public IntegerStack()
      {
         arr = new int[600000];
      }
      public int size()
      {
         return size;
      }
      public void push(int v)
      {
         arr[size++] = v;
      }
      public int pop()
      {
         int v = arr[--size];
         return v;
      }
   }
   class EdgeStack
   {
      public Edge[] arr;
      int size = 0;
      
      public EdgeStack()
      {
         arr = new Edge[600000];
      }
      public int size()
      {
         return size;
      }
      public void push(Edge v)
      {
         arr[size++] = v;
      }
      public Edge pop()
      {
         Edge v = arr[--size];
         return v;
      }
   }
