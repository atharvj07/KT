#include <bits/stdc++.h>
using namespace std;
string baca;
bool cek(string x) {
  for (int i = 0; i < x.length(); i++) {
    if (x[i] != x[x.length() - i - 1]) {
      return false;
    }
  }
  return true;
}
int main() {
  cin >> baca;
  for (int i = 1; i < baca.length(); i++) {
    string a = baca.substr(0, i);
    string b = baca.substr(i, baca.length() - i);
    string c = b + a;
    if (c != baca && cek(c)) {
      cout << "1\n";
      return 0;
    }
  }
  bool sama = true;
  for (int i = 0; i < baca.length() / 2; i++) {
    if (baca[i] != baca[0]) {
      sama = false;
      break;
    }
  }
  if (sama) {
    cout << "Impossible\n";
    return 0;
  }
  cout << "2\n";
}
