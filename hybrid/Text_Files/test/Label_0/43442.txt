t = int(input())
for i in range(t):
	(n, k) = [int(i) for i in input().split()]
	mt = [int(i) for i in input().split()]
	peak = [0 for i in range(n)]
	maxP = 0
	pos = 0
	for i in range(1, k - 1):
		if mt[i - 1] < mt[i] and mt[i] > mt[i + 1]:
			maxP += 1
			peak[i] = 1
	i = k - 1
	nowP = maxP
	while i < n - 1:
		if peak[i - (k - 2)] == 1:
			nowP -= 1
		if mt[i - 1] < mt[i] and mt[i] > mt[i + 1]:
			nowP += 1
			peak[i] = 1
		if nowP > maxP:
			maxP = nowP
			pos = i - (k - 2)
		i += 1
	print(maxP + 1, pos + 1)
