#include <bits/stdc++.h>
using namespace std;
int n, ret;
int a[201];
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int n;
  cin >> n;
  for (int i = 1; i <= 2 * n; i++) cin >> a[i];
  for (int i = 1; i <= 2 * n; i++) {
    if (a[i] == a[i + 1]) continue;
    for (int j = 2 * n - 1; j > i; j--)
      if (a[j + 1] == a[i]) {
        swap(a[j], a[j + 1]);
        ret++;
      }
  }
  cout << ret << '\n';
  return 0;
}
