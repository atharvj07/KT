#include <iostream>
#include <cmath>
#include <functional>
#include <vector>
using namespace std;

template <typename T>
struct SegmentTree{
private:
  using F = function<T(T,T)>;
  int n;
  F f;
  T ti;
  vector<T> dat;
public:
  SegmentTree(){};
  SegmentTree(F f,T ti) : f(f),ti(ti) {}
  void build(int n_){
    n = n_;
    dat.assign(2*n,ti);
  }
  void build(const vector<T> &v){
    int n_ = v.size();
    build(n_);
    for(int i = 0; i < n; ++i) dat[n+i]=v[i];
    for(int i = n-1; i >= 0; --i)
      dat[i]=f(dat[2*i+0],dat[2*i+1]);
  }
  void set_val(int k,T x){
    dat[k+=n] = x;
    while(k > 0){
      k = k/2;
      dat[k]=f(dat[2*k+0],dat[2*k+1]);
    }
  }
  T query(int a,int b){
    T vl = ti, vr = ti;
    for(int l = a+n, r = b+n; l < r; l >>= 1, r >>= 1){
      if(l&1) vl = f(vl,dat[l++]);
      if(r&1) vr = f(dat[--r],vr);
    }
    return f(vl,vr);
  }
};


int main(){
  using T = double;
  function<T(T,T)> f = [](T a, T b){
                         return min(a,b);
                       };
  T ti = 1e9;
  SegmentTree<T> st(f,ti);
  int n;
  double k;
  cin >> n >> k;
  vector<double> S(n);
  for(int i = 0; i < n; ++i) cin >> S[i];
  for(int i = 0; i < n; ++i){
    if(S[i] < 1e-3){
      cout << n << endl;
      return 0;
    }
  }
  double lk = log(k);
  vector<double> L(n+1);
  for(int i = 0; i < n; ++i)
    L[i+1] = log(S[i]);
  for(int i = 0; i < n; ++i)
    L[i+1] += L[i];
  st.build(L);
  int ans = 0;
  for(int i = 0; i < n; ++i){
    int ok = i, ng = n+2;
    while(ng-ok > 1){
      int m = (ok+ng)/2;
      double mi = st.query(m,n+1);
      if(mi-L[i] > lk) ng = m;
      else ok = m;
    }
    ans = max(ans,ok-i);
  }
  cout << ans << endl;
}
