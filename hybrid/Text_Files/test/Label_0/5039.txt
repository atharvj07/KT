#define FOR(i,a,b) for(int i=(a);i<(b);++i)  
#define REP(i,n) FOR(i,0,n)  
#define FORE(i,a,b) for(int i=(a);i<=(b);++i)

typedef long long ll;
typedef unsigned long long ull;
ll gcd(ll a,ll b) { return b==0?a:gcd(b,a%b); }

const int MAXN = 15;
const int MAXM = MAXN*(MAXN - 1) / 2;
const int MOD = 1000000007;

int n, m;
int a[MAXM], b[MAXM];

int p2[MAXM + 1];
int nedges[1 << MAXN];

int ways[2][1 << MAXN];


void printmask(int mask) { REP(i, n) printf("%c", mask&(1 << i) ? 'x' : '.'); }


void run() {
	scanf("%d%d", &n, &m); REP(i, m) scanf("%d%d", &a[i], &b[i]), --a[i], --b[i];

	p2[0] = 1; FORE(i, 1, MAXM) p2[i] = (ll)p2[i - 1] * 2 % MOD;
	REP(i, 1 << n) { nedges[i] = 0; REP(j, m) if ((i&(1 << a[j])) != 0 && (i&(1 << b[j])) != 0) ++nedges[i]; }

	FOR(amask, 1, 1 << n) if ((amask&(1 << 0)) != 0 && (amask&(1 << 1)) == 0) {
		ways[0][amask] = p2[nedges[amask]];
		int othermask = (1 << n) - 1 - amask;
		for (int pmask = (othermask + 1)&amask; pmask != amask; pmask = (pmask + othermask + 1)&amask) if ((pmask&(1 << 0)) != 0) {
			ways[0][amask] = (ways[0][amask] + MOD - (ll)ways[0][pmask] * p2[nedges[amask - pmask]] % MOD) % MOD;
		}
		//printf("A"); printmask(amask); printf(" = %d (nedges=%d)\n", ways[0][amask],nedges[amask]);
	}
	FOR(bmask, 1, 1 << n) if ((bmask&(1 << 1)) != 0 && (bmask&(1 << 0)) == 0) {
		ways[1][bmask] = p2[nedges[bmask]];
		int othermask = (1 << n) - 1 - bmask;
		for (int pmask = (othermask + 1)&bmask; pmask != bmask; pmask = (pmask + othermask + 1)&bmask) if ((pmask&(1 << 1)) != 0) {
			ways[1][bmask] = (ways[1][bmask] + MOD - (ll)ways[1][pmask] * p2[nedges[bmask - pmask]] % MOD) % MOD;
		}
		//printf("B"); printmask(bmask); printf(" = %d (nedges=%d)\n", ways[1][bmask], nedges[bmask]);
	}

	int ret = 0;
	REP(amask, 1 << n) if ((amask&(1 << 0)) != 0 && ((amask&(1 << 1)) == 0)) {
		int forbid = amask;
		REP(j, m) { if (amask&(1 << a[j])) forbid |= 1 << b[j]; if (amask&(1 << b[j])) forbid |= 1 << a[j]; }
		for (int bmask = 0; bmask < (1 << n); bmask = (bmask + forbid + 1)&~forbid) if ((bmask&(1 << 1)) != 0 && (bmask&(1 << 0)) == 0) {
			int cmask = (1 << n) - 1 - amask - bmask;
			int cur = (ll)ways[0][amask] * ways[1][bmask] % MOD*p2[nedges[cmask]] % MOD;
			//printf("sol "); printmask(amask); printf(" "); printmask(bmask); printf(" "); printmask(cmask); printf(" -> %d\n", cur);
			ret = (ret + cur) % MOD;
		}
	}
	ret = (p2[m] + MOD - ret) % MOD;
	printf("%d\n", ret);
}

int main() {
	run();
	return 0;
}
