#include <bits/stdc++.h>
using namespace std;
template <typename T>
void out(T x) {
  cout << x << endl;
  exit(0);
}
using ll = long long;
const ll mod = 1e9 + 7;
const int maxn = 1e6 + 5;
int dp[25][25];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int n;
  cin >> n;
  string s = "abcdefghijk";
  for (int i = 0; i < 12; i++) {
    for (int j = 0; j < 12; j++) {
      if (i == 0 || j == 0) {
        dp[i][j] = 1;
      } else {
        dp[i][j] = dp[i - 1][j] + dp[i][j - 1];
      }
    }
  }
  vector<int> take(20, 0);
  for (int i = 11; i >= 0; i--) {
    while (n >= dp[11][i]) {
      take[i]++;
      n -= dp[11][i];
    }
  }
  for (int i = 0; i < 12; i++) {
    cout << s;
    for (int j = 0; j < take[i]; j++) cout << "l";
  }
  cout << " " << s << "l\n";
  return 0;
}
