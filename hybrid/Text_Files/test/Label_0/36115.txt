import java.io.BufferedReader;
import java.io.InputStreamReader;

/**
 * Created by Anton Berezin on 1/21/2017.
 */
public class Orkestra {
    public static void main(String[] args) throws Exception {
        try(BufferedReader reader = new BufferedReader(new InputStreamReader(System.in))){
            String[] line = reader.readLine().split(" ");
            int r = Integer.parseInt(line[0]);
            int c = Integer.parseInt(line[1]);
            int n = Integer.parseInt(line[2]);
            int k = Integer.parseInt(line[3]);
            int[] altRow = new int[n];
            int[] altCol = new int[n];
            for(int i = 0; i<n;i++) {
                line = reader.readLine().split(" ");
                altRow[i] = Integer.parseInt(line[0]) - 1;
                altCol[i] = Integer.parseInt(line[1]) - 1;
            }
            int answer = 0;
            for(int rowStart = 0; rowStart < r; rowStart++) {
                for(int colStart = 0; colStart < c; colStart++) {
                    for(int rowEnd = rowStart; rowEnd < r; rowEnd++) {
                        for(int colEnd = colStart; colEnd < c; colEnd++) {
                            int cnt = 0;
                            for(int i = 0;  i< n; i++) {
                                if (altRow[i]>=rowStart && altRow[i]<=rowEnd && altCol[i]>=colStart && altCol[i]<= colEnd) {
                                    cnt++;
                                }
                            }
                            if(cnt>= k) {
                                answer++;
                            }
                        }
                    }
                }
            }
            System.out.println(answer);
        }
    }
}
