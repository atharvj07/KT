from sys import stdin, stdout
(n, m) = map(int, stdin.readline().split())
a = [list(map(int, stdin.readline().split())) for i in range(n)]
isDiff = False
(p, q) = (-1, -1)
for i in range(n):
	cval = a[i][0]
	for j in range(1, m):
		if cval is not a[i][j]:
			isDiff = True
			(p, q) = (i, j)
			break
	if isDiff:
		break
if not isDiff:
	x = 0
	for i in range(n):
		x ^= a[i][0]
	if not x:
		print('NIE')
	else:
		print('TAK')
		for i in range(n):
			print('1', end=' ')
		print()
else:
	x = 0
	for i in range(n):
		x ^= a[i][0]
	if x:
		print('TAK')
		for i in range(n):
			print('1', end=' ')
		print()
	else:
		print('TAK')
		for i in range(n):
			print(q + 1, end=' ') if i == p else print('1', end=' ')
		print()
