#include <cstdio>
#include <cstring>
const int N = 4e2 + 100;
int f[N];
int ff(int x) {
	return f[x] == x ? x : f[x] = ff(f[x]);
}
int n, h;
int in[N], ou[N], ok[N];

int main() {
	scanf("%d%d", &n, &h);
	for (int i = 1; i <= h + h; ++i)
		f[i] = i;
	for (int i = 1; i <= n; ++i) {
		int a, b, c, d, l, r;
		scanf("%d%d%d%d", &a, &b, &c, &d);
		l = c == 0 ? h + a : c;
		r = d == 0 ? b : h + d;
		++ou[l], ++in[r];
		if (ff(l) != ff(r))
			f[ff(l)] = ff(r);
	}
	for (int i = 1; i <= h; ++i)
		if (ou[i] > in[i])
			return puts("NO"), 0;
	for (int i = h + 1; i <= h + h; ++i)
		if (in[i] > ou[i])
			return puts("NO"), 0;
	for (int i = 1; i <= h + h; ++i)
		if (in[i] != ou[i])
			ok[ff(i)] = true;
	for (int i = 1; i <= h + h; ++i)
		if (ff(i) == i && !ok[i] && in[i] + ou[i] != 0)
			return puts("NO"), 0;
	return puts("YES"), 0;
}
