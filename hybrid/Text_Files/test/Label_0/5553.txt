#include <bits/stdc++.h>
using namespace std;
void _make_Bin_V();
void PrintV();
void sol() {
  double pi = 3.1415926536;
  int n;
  cin >> n;
  vector<float> v;
  for (int i = 0; i < n; i++) {
    float inp;
    cin >> inp;
    v.push_back(inp);
  }
  sort((v).begin(), (v).end());
  float sum = 0;
  if (n & 1) {
    sum += v[0] * v[0];
    for (int i = 1; i < n; i += 2) {
      sum += v[i + 1] * v[i + 1] - v[i] * v[i];
    }
  } else {
    for (int i = 0; i < n; i += 2) {
      sum += ((v[i + 1] * v[i + 1]) - (v[i] * v[i]));
    }
  }
  cout << pi * sum << endl;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  sol();
}
void PrintV(const vector<long long int> v) {
  for (auto it = v.begin(); it != v.end(); it++) {
    cout << *it << " ";
  }
  cout << endl;
}
