#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
int a[N], b[N], c[N];
int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  long long sa = 0, sb = 0;
  for (int i = 0; i < n; i++) {
    scanf("%d%d", &a[i], &b[i]);
    c[i] = a[i] - b[i];
    sa += a[i];
    sb += b[i];
  }
  if (sa <= m)
    cout << "0" << endl;
  else if (sb > m)
    cout << "-1" << endl;
  else {
    sort(c, c + n);
    int ans = 0;
    for (int i = n - 1; i >= 0; i--) {
      ans++;
      sa -= c[i];
      if (sa <= m) break;
    }
    cout << ans << endl;
  }
  return 0;
}
