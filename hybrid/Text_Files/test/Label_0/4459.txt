#include <iostream>
#include <vector>
#include <algorithm>
#include <utility>
using namespace std;
typedef long long int lli;

lli dfs(int v, int prev, vector<vector<pair<int,lli>>> &adj, vector<lli> &res){
    vector<lli> costs;
    for(auto next: adj[v]){
        if(next.first == prev) continue;
        costs.push_back(next.second + dfs(next.first, v, adj, res));
    }
    if(costs.empty()) return 0;
    sort(costs.begin(), costs.end(), greater<lli>());
    for(int i=1; i<(int)costs.size(); i++){
        res.push_back(costs[i]);
    }
    return costs[0];
}

int main(){
    int n,m;
    cin >> n >> m;
    vector<vector<pair<int,lli>>> adj(n);
    for(int i=0; i<n-1; i++){
        int a,b,c;
        cin >> a >> b >> c;
        a--; b--;
        adj[a].emplace_back(b, c);
        adj[b].emplace_back(a, c);
    }
    vector<lli> s(n);
    for(int i=0; i<n; i++){
        cin >> s[i];
    }

    vector<lli> cost;
    for(int i=0; i<n; i++){
        vector<lli> dist;
        dist.push_back(dfs(i, -1, adj, dist));
        for(auto d: dist){
            cost.push_back(d*s[i]);
        }
    }
    sort(cost.begin(), cost.end(), greater<lli>());
    lli ans = 0;
    for(int i=0; i<min(m, (int)cost.size()); i++){
        ans += cost[i];
    }
    cout << ans << endl;
    return 0;
}
