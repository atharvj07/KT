n = int(input())
p = [int(i) - 1 for i in input().split()]
chd = [0] * (n + 2)
dev = 0
dd = 0
for i in range(n):
	shi = p[i] - i
	dev += abs(shi)
	if p[i] <= i:
		dd += 1
		shi = n + shi
		chd[shi] += 2
	else:
		dd -= 1
		chd[shi] += 2
bdev = dev
bsh = 0
for i in range(1, n):
	dev -= abs(p[-i] - (n - 1))
	dev += abs(p[-i] - 0)
	dd -= 1
	dev += dd
	dd -= 1
	dd += chd[i]
	if dev < bdev:
		bsh = i
		bdev = dev
print(bdev, bsh)
