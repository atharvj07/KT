n = int(input())
h = list(map(int, input().split()))
a = [0] * n
a1 = [0]
a2 = [0]
for i in range(1, n):
	a[i] = a[i - 1] + 1
	while len(a1) and h[i] >= h[a1[-1]]:
		x = a1[-1]
		a1.pop()
		if h[i] > h[x] and len(a1):
			a[i] = min(a[i], a[a1[-1]] + 1)
	while len(a2) and h[i] <= h[a2[-1]]:
		x = a2[-1]
		a2.pop()
		if h[i] < h[x] and len(a2):
			a[i] = min(a[i], a[a2[-1]] + 1)
	a1.append(i)
	a2.append(i)
print(a[-1])
