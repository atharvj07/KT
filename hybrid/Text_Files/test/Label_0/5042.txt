#include <bits/stdc++.h>
using namespace std;
int main() {
  int q;
  cin >> q;
  for (int z = 0; z < q; z++) {
    int m = 0;
    string s;
    cin >> s;
    if (s.size() == 1) {
      if (s[0] == '?') cout << 'a' << endl;
    } else {
      if (s[0] == '?') {
        if (s[1] == 'a')
          s[0] = 'b';
        else if (s[1] == 'b')
          s[0] = 'c';
        else if (s[1] == 'c')
          s[0] = 'a';
        else
          s[0] = 'a';
      }
      for (int i = 1; i < s.size() - 1; i++) {
        if (s[i] == '?') {
          if (s[i - 1] == 'a') {
            if (s[i + 1] == 'b')
              s[i] = 'c';
            else
              s[i] = 'b';
          } else if (s[i - 1] == 'b') {
            if (s[i + 1] == 'a')
              s[i] = 'c';
            else
              s[i] = 'a';
          } else if (s[i - 1] == 'c') {
            if (s[i + 1] == 'b')
              s[i] = 'a';
            else
              s[i] = 'b';
          } else {
            if (s[i + 1] == 'a') {
              s[i - 1] = 'a';
              s[i] = 'b';
            } else if (s[i + 1] == 'b') {
              s[i - 1] = 'b';
              s[i] = 'a';
            } else if (s[i + 1] == 'c') {
              s[i - 1] = 'c';
              s[i] = 'a';
            }
          }
        }
      }
      if (s[s.size() - 1] == '?') {
        if (s[s.size() - 2] == 'a')
          s[s.size() - 1] = 'b';
        else if (s[s.size() - 2] == 'b')
          s[s.size() - 1] = 'c';
        else if (s[s.size() - 2] == 'c')
          s[s.size() - 1] = 'a';
      }
      for (int i = 1; i < s.size(); i++) {
        if (s[i] == s[i - 1]) m++;
      }
      if (m == 0) {
        for (int i = 0; i < s.size(); i++) cout << s[i];
        cout << endl;
      } else
        cout << "-1" << endl;
    }
  }
}
