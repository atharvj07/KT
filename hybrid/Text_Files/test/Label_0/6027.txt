#include <bits/stdc++.h>
using namespace std;
const int INF = 1 << 29;
const double PI = acos(-1);
const double EPS = 1e-8;
double x[500000], v[500000];
int main() {
  int n;
  scanf("%d", &n);
  for (int i = 0; i < (int)(n); ++i) scanf("%lf%lf", x + i, v + i);
  double low = 0, high = 1e12;
  for (int iter = 0; iter < (int)(100); ++iter) {
    double t = (low + high) / 2;
    double ma = -1e50;
    bool f = 0;
    for (int i = 0; i < (int)(n); ++i) {
      if (v[i] > 0) {
        ma = max(ma, x[i] + v[i] * t);
      } else {
        double tx = x[i] + v[i] * t;
        if (tx <= ma) {
          f = 1;
          break;
        }
      }
    }
    if (f)
      high = t;
    else
      low = t;
  }
  if (low >= 1e11)
    puts("-1");
  else
    printf("%.10f\n", low);
}
