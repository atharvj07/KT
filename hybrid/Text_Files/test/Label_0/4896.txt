int n,ans,l[N],r[N],dp[N][2];
int main()
{
	int x;
	file();
	read(n);
	For(i,1,n)
	{
		read(x);
		l[i]=B2(x);
		r[i]=l[i]+((1<<l[i])!=x);
		//printf("%d: %d l:%d r:%d\n",i,x,l[i],r[i]);
		ans+=r[i];
		dp[i][0]=max(dp[i-1][0]+min(r[i-1],l[i]),dp[i-1][1]+min(l[i-1],l[i]));
		dp[i][1]=max(dp[i-1][0]+min(r[i-1],r[i]),dp[i-1][1]+min(l[i-1],r[i]));
	}
	printf("%d\n",ans-max(dp[n][0],dp[n][1]));
