struct Edge{
    int from,to;
    Edge(int from,int to)
        : from(from),to(to){};
};
typedef vector<vector<Edge>> Graph;

int depth(Graph &T, vector<int> &p, vector<int> &dep,vector<vector<int>> &dtov,int root){
    int ret =0;
    queue<int> Q;
    Q.push(root);
    p[root]=root;
    dtov[0].push_back(root);
    while(!Q.empty()){
        int nex=Q.front();
        Q.pop();
        for(auto x: T[nex]){
            if(p[x.to]==-1){
                p[x.to]=x.from;
                Q.push(x.to);
                dep[x.to]=dep[x.from]+1;
                ret=max(ret,dep[x.to]);
                dtov[dep[x.to]].push_back(x.to);
            }
        }
    }
    return ret;
}

int main(){
    cin.tie(0);
    ios::sync_with_stdio(false);
    int n;cin >> n;
    vector<int>par(n,-1),dep(n),subleaf(n),subleng(n);
    Graph T(n);
    vector<vector<int>> d(n);
    REP(i,n-1){
        int a,b;
        cin >> a >>b;
        a--;b--;
        T[a].PB(Edge(a,b));
        T[b].PB(Edge(b,a));
    }
    int r,leaf=0;
    REP(i,n)if(T[i].size()==1){
        r=i;
        leaf++;
    }
    int mdep=depth(T,par,dep,d,r);
    int ans=0,ans2=0;
    for(int i=mdep; i>=0;--i){
        for(auto x : d[i]){
            //vector<int> mainchain;
            if(T[x].size()==1){
                subleaf[x]=1;
            }else{
                int lev=0,len=0,len2=0;
                for(auto y: T[x]){
                    if(y.to!=par[y.from]){
                        lev+=subleaf[y.to];
                        len2=max(len2,subleng[y.to]);
                        if(len2>len)swap(len,len2);
                    }
                }
                if(T[x].size()==2)len++;
                subleaf[x]=lev;
                subleng[x]=len;
                ans=max(ans,leaf+len2+len);
            }
        }
    }
    for(auto x: T[r]){
        ans2+=subleaf[x.to]+subleng[x.to]+1;
    }
    cout << max(ans,ans2) <<endl;
