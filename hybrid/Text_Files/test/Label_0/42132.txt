g = lambda : stdin.readline().strip()

inf = float('inf')
(n, k) = gil()
a = gil()

def getMaxSetBit(x):
	if x == 0:
		return 0
	return int(math.log2(x)) + 1

def getMin(x):
	b = []
	bit = getMaxSetBit(x)
	for v in a:
		if v == x:
			b.append(0)
		elif v > x:
			steps = getMaxSetBit(v) - bit
			if v >> steps == x:
				b.append(steps)
	b.sort()
	b.append(inf)
	ans = sum(b[:k])
	return ans
ans = min(range(1 + max(a)), key=lambda x: getMin(x))
print(getMin(ans))
