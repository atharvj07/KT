#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, k;
    cin >> n >> m >> k;
    char arr[n][m];
    char chars[100];
    int idx = 0;
    for (int i = 0; i < 26; i++)
      chars[idx++] = (char)('a' + i), chars[idx++] = (char)('A' + i);
    for (int i = 0; i < 10; i++) chars[idx++] = (char)('0' + i);
    int rice = 0;
    for (int i = 0; i < n; i++)
      for (int j = 0; j < m; j++) cin >> arr[i][j], rice += arr[i][j] == 'R';
    idx = 0;
    int div = (rice + k - 1) / k;
    int cnt[k];
    for (int i = 0; i < k; i++) cnt[i] = div;
    int diff = div * k - rice;
    while (diff) {
      cnt[idx++]--;
      diff--;
      if (idx >= k) idx = 0;
    }
    int x = 0, y = 0, add = 1;
    idx = 0;
    while (rice) {
      while (cnt[idx]) {
        if (arr[x][y] == 'R') cnt[idx]--, rice--;
        arr[x][y] = chars[idx];
        y += add;
        if (y == m || y == -1) {
          x++;
          add *= -1;
          y += add;
        }
      }
      idx++;
    }
    idx--;
    while (x < n) {
      arr[x][y] = chars[idx];
      y += add;
      if (y == m || y == -1) {
        x++;
        add *= -1;
        y += add;
      }
    }
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) cout << arr[i][j];
      cout << endl;
    }
  }
}
