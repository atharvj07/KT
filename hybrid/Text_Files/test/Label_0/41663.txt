import math

def divisorGenerator(n):
	large_divisors = []
	for i in range(1, int(math.sqrt(n) + 1)):
		if n % i == 0:
			yield i
			if i * i != n:
				large_divisors.append(n / i)
	for divisor in reversed(large_divisors):
		yield divisor

def reversesub(s, begin, end):
	return s[0:begin] + s[begin:end][::-1] + s[end:]
n = int(input())
m = input()
a = list(divisorGenerator(n))
b = list(map(int, a))
c = len(b)
ans = reversesub(m, 0, b[0])
for i in range(1, c):
	ans = reversesub(ans, 0, b[i])
print(ans)
