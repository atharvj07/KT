#include <bits/stdc++.h>
using namespace std;
int n, tmp, cha[111111], vis[111111], id;
int main() {
  while (~scanf("%d", &n)) {
    id = 0;
    memset(cha, -1, sizeof(cha));
    for (int i = 0; i < n; i++) {
      scanf("%d", &tmp);
      if (cha[tmp] == -1) {
        cha[tmp] = 0;
        vis[tmp] = i;
      } else if (cha[tmp] == -2)
        continue;
      else if (cha[tmp] == 0) {
        cha[tmp] = i - vis[tmp];
        vis[tmp] = i;
      } else {
        if (i - vis[tmp] == cha[tmp])
          vis[tmp] = i;
        else
          cha[tmp] = -2;
      }
    }
    for (int i = 0; i < 101111; i++) {
      if (cha[i] >= 0) id++;
      ;
    }
    printf("%d\n", id);
    for (int i = 0; i < 101111; i++) {
      if (cha[i] >= 0) printf("%d %d\n", i, cha[i]);
    }
  }
  return 0;
}
