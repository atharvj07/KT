#include <bits/stdc++.h>
using namespace std;

int main() {
    //const int Inf = 1e9;
    int n;
    cin >> n;

    int dp[305][305];
    for(int i=0; i<n; i++){
        for(int j=0; j<n; j++){
            cin >> dp[i][j];
        }
    }
    long long ans =0;
    for(int i=0; i<n; i++){
        for(int j=0; j<n; j++){
            bool flag=true;
            for(int k=0; k<n; k++){
                if(k==i | k ==j) continue;
                if(dp[i][j]>dp[i][k]+dp[k][j]){
                    cout << -1 << endl;
                    return 0;
                } else if (dp[i][j]==dp[i][k]+dp[k][j]){
                    flag = false;
                }
            }
            if(flag) ans += dp[i][j];
        }
    }

    cout << ans/2 << endl;
    return 0;
}
