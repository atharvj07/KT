(n, p, m) = map(int, input().split())
lastd = 0
acc = 0
output = 0
neg = 0
for i in range(n):
	(d, t) = map(int, input().split())
	acc += (d - lastd - 1) * -p
	if acc < 0:
		output += acc // -p - neg
		neg = acc // -p
		if acc % p != 0:
			output += 1
			neg += 1
	acc += t - p
	if acc < 0:
		output += 1
		neg = acc // -p
		if acc % p != 0:
			neg += 1
	else:
		neg = 0
	lastd = d
if d < m:
	acc += (m - d) * -p
	if acc < 0:
		output += acc // -p - neg
		if acc % p != 0:
			output += 1
print(output)
