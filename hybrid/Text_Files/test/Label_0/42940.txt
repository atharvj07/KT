from itertools import accumulate
(H, n) = map(int, input().split())
d = [*map(int, input().split())]
k = [*accumulate(d)]
mink = min(k)
if mink + H <= 0:
	for (i, j) in enumerate(k):
		if j + H <= 0:
			print(i + 1)
			break
elif k[-1] >= 0:
	print(-1)
else:
	rounds = (H + mink + abs(k[-1]) - 1) // abs(k[-1])
	health = H + k[-1] * rounds
	for (i, j) in enumerate(k):
		if j + health <= 0:
			print(i + 1 + rounds * n)
			break
