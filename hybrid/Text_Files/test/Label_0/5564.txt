#include <bits/stdc++.h>
using namespace std;
int main() {
  int a1, a2, a3, a4, a5, a6, tri, total = 0;
  cin >> a1 >> a2 >> a3 >> a4 >> a5 >> a6;
  tri = a1 - 1;
  for (int i = 0; i <= a2; i++) {
    if (i <= a6) tri++;
    total += tri;
  }
  for (int i = 0; i < a3 - 1; i++) {
    if (a2 + i >= a6) tri--;
    total += tri;
  }
  tri = a4 - 1;
  for (int i = 0; i <= a5; i++) {
    if (i <= a3) tri++;
    total += tri;
  }
  for (int i = 0; i < a6 - 1; i++) {
    if (a5 + i >= a3) tri--;
    total += tri;
  }
  cout << total << endl;
}
