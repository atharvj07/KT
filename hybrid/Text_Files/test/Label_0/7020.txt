#include <bits/stdc++.h>
using namespace std;
int main() {
  char x[100];
  cin >> x;
  int z = 1;
  for (int i = 0; i < strlen(x); i++) {
    if (z >= 7) {
      break;
    } else {
      if (x[i] == x[i + 1]) {
        z++;
      } else {
        z = 1;
      }
    }
  }
  if (z >= 7)
    cout << "YES";
  else
    cout << "NO";
  return 0;
}
