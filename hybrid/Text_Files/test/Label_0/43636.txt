def main():
	tests = int(input())
	for _ in range(tests):
		elount = int(input())
		array = [int(x) - 1 for x in input().split()]
		if len(set(array)) != elount:
			print('YES')
			continue
		print('YES' if can_be_trisorted(array) else 'NO')

def can_be_trisorted(arr):
	swaps = 0
	for i in range(len(arr)):
		while arr[i] != i:
			j = arr[i]
			(arr[i], arr[j]) = (arr[j], arr[i])
			swaps = 1 - swaps
	return swaps == 0
main()
