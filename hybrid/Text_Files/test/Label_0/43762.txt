import sys
input = sys.stdin.readline
n = int(input())
a = list(map(int, input().split()))
b = {}
for i in range(n):
	if a[i] not in b.keys():
		b[a[i]] = 0
	b[a[i]] += 1
c = []
for i in b:
	c.append(b[i])
e = sorted(c, reverse=True)
d = [e[0]]
a = e[0]
for i in range(len(c) - 1):
	a = min(e[i + 1], a // 2)
	d.append(a * (2 ** (i + 2) - 1))
	if a == 0:
		break
print(max(d))
