#include <bits/stdc++.h>
using namespace std;
long long t1, t2, x1, x2, t0, l, r, m;
double calc(long long trololoy, long long y2) {
  return (trololoy * t1 + y2 * t2) * 1.0 / (trololoy + y2);
}
double topt = 1e18, tc, tc2;
int trololoy, y2;
int main() {
  cin >> t1 >> t2 >> x1 >> x2 >> t0;
  if (t1 == t2) {
    cout << x1 << " " << x2;
    return 0;
  }
  tc = calc(x1, 0);
  tc2 = calc(0, x2);
  if (tc >= t0) topt = tc, trololoy = x1;
  if (tc2 >= t0 &&
      (tc2 < topt || fabs(tc2 - topt) < 1e-9 && x2 > trololoy + y2))
    topt = tc2, trololoy = 0, y2 = x2;
  if (tc2 >= t0 &&
      (tc2 < topt || fabs(tc2 - topt) < 1e-9 && x2 > trololoy + y2))
    topt = tc2, trololoy = 0, y2 = x2;
  for (long long i = 0; i <= x1; i++) {
    long long j = int(ceil((t0 - t1) * i * 1.0 / (t2 - t0)));
    if (j < 0 || j > x2) continue;
    tc = calc(i, j);
    if (tc >= t0 &&
        (tc < topt || fabs(tc - topt) < 1e-9 && trololoy + y2 < i + j)) {
      trololoy = i, y2 = j;
      topt = tc;
    }
  }
  cout << trololoy << " " << y2;
  return 0;
}
