#include <bits/stdc++.h>
using namespace std;
const long long maxn = 3e5 + 20, Mod = 1e9 + 7, lg = 20;
long long n, m, d, tmp1, tmp2, c[maxn], dpdown[maxn], dpup[maxn], ans;
vector<long long> adj[maxn];
void DFS1(long long u, long long p) {
  for (long long i = 0; i < adj[u].size(); i++) {
    long long v = adj[u][i];
    if (v != p) {
      DFS1(v, u);
      if (dpdown[v] != -1) dpdown[u] = max(dpdown[v] + 1, dpdown[u]);
    }
  }
}
void DFS2(long long u, long long p) {
  if (c[u] && dpup[u] == -1) dpup[u] = 0;
  long long tmp = 0, mx1 = -1, mx2 = -1;
  for (long long i = 0; i < adj[u].size(); i++) {
    long long v = adj[u][i];
    if (v != p) {
      if (dpdown[v] >= mx1) {
        mx2 = mx1;
        mx1 = dpdown[v];
      } else if (dpdown[v] > mx2) {
        mx2 = dpdown[v];
      }
    }
  }
  for (long long i = 0; i < adj[u].size(); i++) {
    long long v = adj[u][i];
    if (v != p) {
      if (dpup[u] >= 0) dpup[v] = dpup[u] + 1;
      if (dpdown[v] < mx1)
        dpup[v] = max(dpup[v], mx1 + 2);
      else if (mx2 >= 0)
        dpup[v] = max(dpup[v], mx2 + 2);
      DFS2(v, u);
    }
  }
}
int32_t main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> m >> d;
  memset(dpdown, -1, sizeof dpdown);
  memset(dpup, -1, sizeof dpup);
  for (long long i = 0; i < m; i++) {
    cin >> tmp1;
    c[tmp1] = 1;
    dpdown[tmp1] = 0;
  }
  for (long long i = 0; i < n - 1; i++) {
    cin >> tmp1 >> tmp2;
    adj[tmp1].push_back(tmp2);
    adj[tmp2].push_back(tmp1);
  }
  DFS1(1, 1);
  DFS2(1, 1);
  for (long long i = 1; i <= n; i++) {
    if (max(dpup[i], dpdown[i]) <= d) ans++;
  }
  cout << ans;
}
