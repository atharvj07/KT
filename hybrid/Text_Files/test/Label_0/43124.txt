def main():
	n = int(input())
	bigram = input().split()
	for i in range(n - 3):
		if bigram[i][1] == bigram[i + 1][0]:
			continue
		else:
			bigram.insert(i + 1, bigram[i][1] + bigram[i + 1][0])
			print(*[j[0] for j in bigram] + [bigram[-1][1]], sep='')
			return
	print(*[j[0] for j in bigram] + [bigram[-1][1] * 2], sep='')
for _ in range(int(input())):
	main()
