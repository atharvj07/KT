import sys
INF = float('inf')
mod = 10 ** 9 + 7
input = lambda : sys.stdin.readline().rstrip()

def li():
	return list(map(int, input().split()))
T = int(input())
for _ in range(T):
	(N, D) = li()
	A = li()
	if D == N:
		if 1 in A:
			print(-1)
		else:
			print(0)
		continue
	x = 0
	lst = [0]
	se = set([0])
	while True:
		x += D
		x %= N
		if x in se:
			break
		lst.append(x)
		se.add(x)
	ans = 0
	seen = set()
	for i in range(N):
		if i in seen:
			continue
		cnt = 0
		seen.add(i)
		for j in range(len(lst) * 2):
			a = (lst[j % len(lst)] + i) % N
			seen.add(a)
			if A[a] == 1:
				cnt += 1
				ans = max(ans, cnt)
			else:
				cnt = 0
	if ans > len(lst):
		ans = -1
	print(ans)
