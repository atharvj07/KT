#include <bits/stdc++.h>
using namespace std;
int a[7];
int main() {
  char s[100000];
  char t;
  int k = 0;
  for (int i = 0; cin >> s[i]; i++) {
    k++;
  }
  for (int i = 0; i < k; i++) {
    switch ((int)s[i]) {
      case (int)'B':
        a[0]++;
        break;
      case (int)'u':
        a[1]++;
        break;
      case (int)'l':
        a[2]++;
        break;
      case (int)'b':
        a[3]++;
        break;
      case (int)'a':
        a[4]++;
        break;
      case (int)'s':
        a[5]++;
        break;
      case (int)'r':
        a[6]++;
        break;
      default:
        break;
    }
  }
  a[1] /= 2;
  a[4] /= 2;
  int mm = a[0];
  for (int i = 0; i < 7; i++) {
    if (a[i] < mm) mm = a[i];
  }
  cout << mm;
  return 0;
}
