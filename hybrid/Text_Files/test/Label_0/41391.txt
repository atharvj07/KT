(n, m) = map(int, input().split())
ans = []
for i in range(n):
	z = input()
	ans.append(z)
t1 = []
t2 = []
flag = 0
dp = []
for i in range(m):
	if flag == 0:
		t1.append('B')
		t2.append('W')
		flag = 1
	else:
		t1.append('W')
		t2.append('B')
		flag = 0
for i in range(n):
	if i % 2 == 0:
		dp.append(t1.copy())
	else:
		dp.append(t2.copy())
for i in range(n):
	mn = ans[i]
	for j in range(m):
		if mn[j] == '-':
			dp[i][j] = '-'
for i in range(n):
	print(''.join(map(str, dp[i])))
