#include <bits/stdc++.h>
using namespace std;
signed main() {
  ios::sync_with_stdio(0);
  int n, k;
  cin >> n >> k;
  if (n < k) {
    cout << -1 << endl;
    exit(0);
  }
  if (k == 1) {
    if (n > 1) {
      cout << -1 << endl;
      exit(0);
    }
    for (int i = 0; i < n; ++i) {
      cout << "a";
    }
    cout << endl;
  } else {
    for (int i = 0; i < n - (k - 2); ++i) {
      cout << (char)('a' + (i & 1));
    }
    for (int i = 0; i < k - 2; ++i) {
      cout << (char)('c' + i);
    }
    cout << endl;
  }
  return 0;
}
