#include<bits/stdc++.h>

#define lowbit(x) (x&-x)
using namespace std;

map<pair<int,int>,int> mp;

int query(int l,int r){
    if(mp.count(make_pair(l,r))) return mp[make_pair(l,r)];
    cout<<"? "<<l<<" "<<r<<endl;
    int x;
    cin>>x;
    return mp[make_pair(l,r)]=r-l+1-x;
}

void dec(int k,int l,int r){
    mp[make_pair(l,r)]--;
    if(l==r)    return;
    int mid=(l+r)>>1;
    if(k<=mid)  dec(k,l,mid);
    else        dec(k,mid+1,r);
}

int n,t,k;

int main(){
    ios::sync_with_stdio(0);cin.tie(0);
    cin>>n>>t;
    while(t--){
        cin>>k;
        int now=k;
        int l=1,r=n;
        while(l<r){
            int mid=(l+r)>>1;
            if(query(l,mid) >= now){
                r=mid;
            }
            else{
                now-=query(l,mid);
                l=mid+1;
            }
        }
        cout<<"! "<<l<<endl;
        dec(l,1,n);
    }
    

    return 0;
}
