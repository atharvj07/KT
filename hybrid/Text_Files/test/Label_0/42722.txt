(n, p, q, r) = map(int, input().split())
a = list(map(int, input().split()))
(pre, suf) = ([p * a[0]], [r * a[n - 1]] * n)
for i in range(1, n):
	pre.append(max(pre[i - 1], p * a[i]))
	suf[n - i - 1] = max(suf[n - i], r * a[n - i - 1])
ans = -float('inf')
for i in range(n):
	ans = max(ans, pre[i] + suf[i] + a[i] * q)
print(ans)
