#include<bits/stdc++.h>
#define LL long long
#define Mod 1000000007
using namespace std;
int n,m; char str[5005];
int f[2][5005];
int main(){
	scanf("%d%s",&n,str);
	m=strlen(str);
	f[0][0]=1;
	for(int i=1;i<=n;++i){
		int I=i&1;
		f[I][0]=(f[I^1][0]+f[I^1][1])%Mod;
		for(int j=1;j<=n;++j)
			f[I][j]=((LL)f[I^1][j-1]+f[I^1][j-1]+f[I^1][j+1])%Mod;
	}
	int Ans=f[n&1][m];
	for(int i=1;i<=m;++i) Ans=(LL)Ans*500000004%Mod;
	printf("%d",Ans);
	return 0;
}
