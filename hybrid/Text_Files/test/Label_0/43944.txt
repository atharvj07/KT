(n, d) = map(int, input().split())
pessoas = [-1] * (n + 1)
count = 0

def find(x):
	if pessoas[x] < 0:
		return x
	else:
		return find(pessoas[x])
for i in range(d):
	(x, y) = map(int, input().split())
	if find(x) == find(y):
		count += 1
	else:
		pessoas[find(x)] += pessoas[find(y)]
		pessoas[find(y)] = find(x)
	resposta = 0
	lista = sorted(pessoas)
	for j in range(count + 1):
		if lista[j] < 0:
			resposta += lista[j]
	print(-1 - resposta)
