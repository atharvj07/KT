n = int(input())
tab = [int(x) for x in input().split()]
result = [2 * n for _ in range(n)]
last_zero = -1
for i in range(n):
	if tab[i] == 0:
		last_zero = i
	if last_zero != -1:
		result[i] = min([result[i], i - last_zero])
last_zero = -1
for i in range(n - 1, -1, -1):
	if tab[i] == 0:
		last_zero = i
	if last_zero != -1:
		result[i] = min([result[i], last_zero - i])
print(' '.join(map(str, result)))
