n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))
i = 0
j = 0
curra = a[i]
currb = b[j]
c = 0
if sum(a) != sum(b):
	print(-1)
else:
	while i < n and j < m:
		if curra == currb:
			i += 1
			j += 1
			c += 1
			if i < n and j < m:
				curra = a[i]
				currb = b[j]
		elif currb > curra:
			i += 1
			if i < n:
				curra += a[i]
		else:
			j += 1
			if j < m:
				currb += b[j]
	print(c)
