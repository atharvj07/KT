#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int a, b, c, x, y, z, ex = 0, f1 = 0, f2 = 0, f3 = 0;
  cin >> a >> b >> c >> x >> y >> z;
  if (a >= x) {
    a -= x;
    x = 0;
    ex += a / 2;
  }
  if (b >= y) {
    b -= y;
    y = 0;
    ex += b / 2;
  }
  if (c >= z) {
    c -= z;
    z = 0;
    ex += c / 2;
  }
  if (x == 0)
    f1 = 1;
  else {
    if (ex >= x - a) {
      ex -= (x - a);
      f1 = 1;
    }
  }
  if (y == 0)
    f2 = 1;
  else {
    if (ex >= y - b) {
      ex -= (y - b);
      f2 = 1;
    }
  }
  if (z == 0)
    f3 = 1;
  else {
    if (ex >= z - c) {
      ex -= (z - c);
      f3 = 1;
    }
  }
  if (f1 == 1 && f2 == 1 & f3 == 1) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
