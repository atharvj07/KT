import sys, io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
read = lambda : map(int, input().split())
out = []
for _ in range(int(input())):
	(n, m) = read()
	s = [2 * (i == 43) - 1 for i in input().strip()]
	pre = [0] * 3
	suf = [0] * 3
	pv = sv = 0
	for i in range(n):
		pv += s[i]
		sv -= s[-1 - i]
		if pv < pre[-3]:
			pre.append(pv)
		else:
			pre.append(pre[-3])
		if pv > pre[-3]:
			pre.append(pv)
		else:
			pre.append(pre[-3])
		pre.append(pv)
		if sv < suf[-3]:
			suf.append(sv)
		else:
			suf.append(suf[-3])
		if sv > suf[-3]:
			suf.append(sv)
		else:
			suf.append(suf[-3])
		suf.append(sv)
	for i in range(m):
		(l, r) = read()
		(ll, lh, lv) = pre[3 * l - 3:3 * l]
		(rl, rh, rv) = suf[3 * (n - r):3 * (n - r) + 3]
		rh = rh - rv + lv
		rl = rl - rv + lv
		if rl <= ll <= rh or rl <= lh <= rh or ll <= rl <= lh or (ll <= rh <= lh):
			out.append(str(max(ll, lh, rl, rh) - min(ll, lh, rl, rh) + 1))
		else:
			out.append(str(rh - rl + lh - ll + 2))
print('\n'.join(out))
