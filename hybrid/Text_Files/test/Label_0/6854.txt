using namespace std;

#define mod 1000000007


int main()
{
	int n;
	cin >> n;
	vector<int> x, y;
	for(int i = 0; i < n; i++){
		if(i % 6 == 1){
			for(int j = 0; j < n; j += 2){
				x.push_back(i);
				y.push_back(j);
			}
		} else if(i % 6 == 4){
			for(int j = 1; j < n; j += 2){
				x.push_back(i);
				y.push_back(j);
			}
		} else if(i % 6 == 3 || i % 6 == 5){
			x.push_back(i);
			y.push_back(0);
		}
	}
	if((n - 1) % 6 == 0){
		for(int j = 1; j < n; j += 2){
			x.push_back(n - 1);
			y.push_back(j);
		}
	} else if((n - 1) % 6 == 3){
		for(int j = 2; j < n; j += 2){
			x.push_back(n - 1);
			y.push_back(j);
		}
	}
	cout << x.size() << endl;
	for(int i = 0; i < x.size(); i++){
		cout << x[i] << " " << y[i] << endl;
	}
	return 0;
}
