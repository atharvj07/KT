#include <bits/stdc++.h>
using namespace std;
struct Node {
  int index;
  int num;
  bool operator<(Node o) const { return num > o.num; }
};
Node node[1005];
int out[1005];
int main() {
  int n, m;
  while (scanf("%d%d", &n, &m) != EOF) {
    for (int i = 0; i < m; i++) {
      scanf("%d", &node[i].num);
      node[i].index = i + 1;
    }
    sort(node, node + m);
    bool flag = true;
    int cur = 0;
    memset(out, 0, sizeof(out));
    out[n] = node[0].index;
    for (int i = 0; i < n; i += 2) {
      if (node[cur].num > 0 && node[cur].index != out[i - 1] &&
          node[cur].index != out[i + 1]) {
        node[cur].num--;
        out[i] = node[cur].index;
      } else {
        cur++;
        if (cur == m) {
          flag = false;
          break;
        }
        node[cur].num--;
        out[i] = node[cur].index;
      }
    }
    if (flag == false) {
      puts("-1");
      continue;
    }
    for (int i = 1; i < n; i += 2) {
      if (node[cur].num > 0 && node[cur].index != out[i - 1] &&
          node[cur].index != out[i + 1]) {
        node[cur].num--;
        out[i] = node[cur].index;
      } else {
        cur++;
        if (cur == m) {
          flag = false;
          break;
        }
        node[cur].num--;
        out[i] = node[cur].index;
      }
    }
    if (flag == false) {
      puts("-1");
      continue;
    }
    for (int i = 0; i < n; i++) {
      if (i != 0) {
        putchar(' ');
      }
      printf("%d", out[i]);
    }
    puts("");
  }
  return 0;
}
