#include <bits/stdc++.h>
using namespace std;
const int N = 1048576, INF = N;
int n;
int a[N];
int d[N];
int q[N], hd, tl;
void Push(int i, int c) {
  if (d[i] != -1) return;
  d[i] = c;
  q[tl++] = i;
}
int main() {
  cin >> n;
  for (int i = (0); i < (int)(n); i++) cin >> a[i];
  copy(a, a + n, a + n);
  long long ans = 0LL;
  for (int i = (0); i < (int)(n); i++) {
    int e = i + n - 1;
    hd = tl = 0;
    for (int k = (0); k < (int)(n); k++) d[i + k] = -1;
    Push(e, 0);
    while (hd != tl && d[i] == -1) {
      int x = q[hd], c = d[x];
      int off = 0;
      while (hd + off != tl && d[q[hd + off]] == c) {
        int x = q[hd + off];
        ++off;
        for (int u = 1; x + u <= e && d[x + u] == -1; Push(x + u, c), u++)
          ;
      }
      for (int u = (0); u < (int)(off); u++) {
        int x = q[hd + u];
        int y = max(i, x - a[x]);
        Push(y, c + 1);
      }
      hd += off;
    }
    ans += d[i];
  }
  cout << ans << endl;
  return 0;
}
