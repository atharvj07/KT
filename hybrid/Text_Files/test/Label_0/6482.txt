#include <bits/stdc++.h>
using namespace std;
const int N = 8e5 + 100;
long long BIT[N];
inline long long getPrefix(int i) {
  ++i;
  long long ret = 0;
  while (i) ret += BIT[i], i -= i & -i;
  return ret;
}
inline void add(int i, long long v) {
  ++i;
  while (i < N) BIT[i] += v, i += i & -i;
}
inline long long getRange(int l, int r) {
  return getPrefix(r) - getPrefix(l - 1);
}
pair<int, int> arr[N];
int n;
map<int, int> yidx;
vector<vector<int>> vec;
struct query {
  int idx, i, y;
  query(int a, int b, int c) { idx = a, i = b, y = c; }
  bool operator<(const query& b) const { return y < b.y; }
};
vector<query> queries;
vector<pair<int, int>> mxyi;
vector<int> uncom;
int com(int x) {
  return lower_bound(uncom.begin(), uncom.end(), x) - uncom.begin();
}
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) {
    scanf("%d", &arr[i].first), scanf("%d", &arr[i].second);
    uncom.push_back(arr[i].first);
    uncom.push_back(arr[i].second);
  }
  sort(uncom.begin(), uncom.end());
  uncom.erase(unique(uncom.begin(), uncom.end()), uncom.end());
  for (int i = 0; i < n; ++i) arr[i].first = com(arr[i].first);
  for (int i = 0; i < n; ++i) arr[i].second = com(arr[i].second);
  sort(arr, arr + n);
  for (int i = 0; i < n;) {
    int j = i;
    vec.push_back(vector<int>());
    while (j < n && arr[j].first == arr[i].first) {
      vec.back().push_back(arr[j].second);
      j++;
    }
    i = j;
  }
  int sz = vec.size();
  for (int i = 0; i < sz; i++) {
    for (auto v : vec[i]) {
      int idx = 0;
      if (yidx.count(v)) idx = yidx[v] + 1;
      yidx[v] = i;
      queries.push_back(query(idx, i, v));
    }
    mxyi.push_back(pair<int, int>(vec[i].back(), i));
  }
  sort(queries.begin(), queries.end());
  sort(mxyi.begin(), mxyi.end());
  for (int i = 0; i < sz; ++i) add(i, 1);
  int j = 0;
  long long out = 0;
  for (auto qr : queries) {
    while (j < sz && mxyi[j].first < qr.y) {
      add(mxyi[j].second, -1);
      j++;
    }
    long long cnt1 = getRange(qr.idx, qr.i);
    long long cnt2 = getRange(qr.i, sz - 1);
    out += cnt1 * cnt2;
  }
  cout << out;
}
