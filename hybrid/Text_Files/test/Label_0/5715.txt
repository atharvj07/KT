#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 10;
const int inf = 0x3f3f3f3f;
int fa[maxn][20], cou[maxn][20], dep[maxn], ans[maxn], crr[maxn], drr[maxn];
bool visit[maxn];
vector<int> v[maxn];
int n, m;
struct Edge {
  int from, to, id;
} edge[maxn << 1];
void dfs(int now, int step) {
  visit[now] = 1;
  dep[now] = step;
  for (int i = 0; i < v[now].size(); ++i) {
    int to = v[now][i];
    if (visit[to] == 0) {
      fa[to][0] = now;
      dfs(to, step + 1);
    }
  }
}
void update(int x, int y, int val) {
  if (dep[x] < dep[y]) swap(x, y);
  for (int i = 18; i >= 0; --i) {
    if (dep[fa[x][i]] >= dep[y]) {
      cou[x][i] += val;
      x = fa[x][i];
    }
  }
  if (x == y) return;
  for (int i = 18; i >= 0; --i) {
    if (fa[x][i] != fa[y][i]) {
      cou[x][i] += val;
      cou[y][i] += val;
      x = fa[x][i];
      y = fa[y][i];
    }
  }
  cou[x][0] += val;
  cou[y][0] += val;
  return;
}
int main(int argc, char const *argv[]) {
  scanf("%d", &m);
  for (int i = 1; i <= m - 1; ++i) {
    scanf("%d %d", &edge[i].from, &edge[i].to);
    edge[i].id = i;
    v[edge[i].from].push_back(edge[i].to);
    v[edge[i].to].push_back(edge[i].from);
  }
  memset(visit, 0, sizeof visit);
  dep[0] = -1;
  dfs(1, 0);
  for (int j = 1; j <= 18; ++j) {
    for (int i = 1; i <= m; ++i) {
      fa[i][j] = fa[fa[i][j - 1]][j - 1];
    }
  }
  int q;
  scanf("%d", &q);
  for (int i = 1; i <= q; ++i) {
    scanf("%d %d", &crr[i], &drr[i]);
  }
  memset(ans, inf, sizeof ans);
  memset(cou, 0, sizeof cou);
  for (int i = 1; i <= q; ++i) {
    update(crr[i], drr[i], 1);
  }
  for (int j = 17; j >= 0; --j) {
    for (int i = 1; i <= m; ++i) {
      cou[i][j] += cou[i][j + 1];
      cou[fa[i][j]][j] += cou[i][j + 1];
    }
  }
  for (int i = 1; i <= m - 1; ++i) {
    if (fa[edge[i].to][0] == edge[i].from)
      ans[edge[i].id] = cou[edge[i].to][0];
    else
      ans[edge[i].id] = cou[edge[i].from][0];
  }
  for (int i = 1; i <= m - 1; ++i) {
    printf("%d ", ans[i]);
  }
  return 0;
}
