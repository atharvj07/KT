s = input().split(' ')
m = int(s[0])
n = int(s[1])
a = input().split(' ')
b = input().split(' ')
for i in range(m):
	a[i] = int(a[i])
for i in range(0, n):
	b[i] = int(b[i])
e1 = 0
o1 = 0
e2 = 0
o2 = 0
for i in a:
	if i % 2 == 0:
		e1 = e1 + 1
	else:
		o1 = o1 + 1
for i in b:
	if i % 2 == 0:
		e2 = e2 + 1
	else:
		o2 = o2 + 1
print(min(e1, o2) + min(o1, e2))
