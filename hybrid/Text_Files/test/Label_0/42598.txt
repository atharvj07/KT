def Solve(n, k):
	if k > n:
		return 0
	elif k == 1:
		return n
	else:
		a = combination(n - 1, k - 1, 998244353)
		for i in range(2, n):
			if i * k > n:
				break
			el = n // i
			a += combination(el - 1, k - 1, 998244353)
			a %= 998244353
		return a

def combination(n, k, p):
	if k > n:
		return 0
	num = 1
	for i in range(n, n - k, -1):
		num = num * i % p
	denom = 1
	for i in range(1, k + 1):
		denom = denom * i % p
	return num * pow(denom, p - 2, p) % p
(n, k) = map(int, input().split())
print(Solve(n, k))
