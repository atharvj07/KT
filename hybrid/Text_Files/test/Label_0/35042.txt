import java.util.*;
import java.io.*;

public class B {
  static List<Long> read(BufferedReader in) throws Exception {
    List<Long> A = new ArrayList<Long>();
    String[] arr = in.readLine().split(" ");
    for(int i=0; i<arr.length; i++) {
      A.add(Long.parseLong(arr[i]));
    }
    return A;
  }
  static Comparator<Long> rev = new Comparator<Long>() {
    public int compare(Long A, Long B) {
      return B.compareTo(A);
    }
  };
  public static void main(String[] args) throws Exception {
    BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    String[] arr = in.readLine().split(" ");
    int n = Integer.parseInt(arr[0]);
    int m = Integer.parseInt(arr[1]);
    List<Long> A = read(in);
    List<Long> B = read(in);
    Collections.sort(A, rev);
    Collections.sort(B, rev);

    long asum = 0;
    long bsum = 0;
    for(Long x : A) {
      asum += x;
    }
    for(Long x : B) {
      bsum += x;
    }

    long ans = asum + bsum;

    long anum = 0;
    long asum2 = asum;
    for(Long a : A) {
      asum2 -= a;
      anum++;
      if(bsum <= (Long.MAX_VALUE/10)/anum) {
        long score = bsum*anum + asum2;
        ans = Math.min(ans, score);
      }
    }

    long bnum = 0;
    long bsum2 = bsum;
    for(Long b : B) {
      bsum2 -= b;
      bnum++;
      if(asum <= (Long.MAX_VALUE/10)/bnum) {
        ans = Math.min(ans, asum*bnum + bsum2);
      }
    }

    System.out.println(ans);
  }
}
