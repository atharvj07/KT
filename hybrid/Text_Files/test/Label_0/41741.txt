t = int(input())
while t:

	def fun(num, n, m, li):
		arr = [False] * m
		flag = 0
		for i in range(n):
			c = 0
			for j in range(m):
				if li[i][j] >= num:
					arr[j] = True
					c += 1
			if c > 1:
				flag = 1
		if not flag and m > 1:
			return False
		ans = True
		for i in arr:
			ans = ans and i
		return ans
	space = input()
	(n, m) = map(int, input().split())
	li = []
	for i in range(n):
		a = list(map(int, input().split()))
		li.append(a)
	low = 1
	high = 2
	while fun(high, n, m, li):
		high *= 2
	while high - low > 1:
		mid = (low + high) // 2
		if fun(mid, n, m, li):
			low = mid
		else:
			high = mid
	print(low)
	t -= 1
