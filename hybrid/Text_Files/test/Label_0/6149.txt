#include <bits/stdc++.h>
long long combimod(int n, int k) {
  if (k > n - k) k = n - k;
  long long min, combi[101] = {0};
  combi[0] = 1;
  for (long long i = 1; i <= n; i++) {
    min = (i < k) ? i : k;
    for (int j = min; j > 0; j--) {
      combi[j] = (combi[j] + combi[j - 1]);
    }
  }
  return combi[k];
}
int main() {
  int n;
  scanf("%d", &n);
  long long a1, a2;
  a1 = combimod(n + 2, 3);
  a2 = combimod(n + 4, 5);
  printf("%lld", a1 * a2);
  return 0;
}
