#include <bits/stdc++.h>
#define int long long
using namespace std;

signed main(){
	int n,a,b,c,nh,ah,z;
	z=0;
	bool bou=true;
	cin>>n;
	vector<int>x(n),y(n),h(n);
	for(int i=0;i<n;i++){
		cin>>x[i]>>y[i]>>h[i];
	}
	for(int nx=0;nx<=100;nx++){
		for(int ny=0;ny<=100;ny++){
			for(int st=0;st<n;st++){
				if(h[st]!=0){
					ah=h[st]+abs(nx-x[st])+abs(ny-y[st]);
					break;
				}
			}
			bou=true;
			for(int data=0;data<x.size();data++){
				nh=max(z,ah-abs(nx-x[data])-abs(ny-y[data]));
				if(nh!=h[data]){
					bou=false;
					break;
				}
			}
			if(bou){
				cout<<nx<<" "<<ny<<" "<<ah<<endl;
				return 0;
			}
		}
	}
}
