#include <bits/stdc++.h>
using namespace std;
int main() {
  {
    ios ::sync_with_stdio(false);
    cin.tie(0);
  }
  int n;
  cin >> n;
  vector<pair<int, int> > vp;
  vector<pair<int, int> > copy;
  for (int i = 0; i < n; ++i) {
    int a, b;
    cin >> a >> b;
    vp.push_back(make_pair(a, b));
    copy.push_back(make_pair(a, b));
  }
  sort(vp.begin(), vp.end());
  for (int i = 0; i < n - 1; ++i) {
    if (vp[i].first == vp[i + 1].first) {
      swap(vp[i].second, vp[i + 1].second);
      break;
    }
  }
  int r_max = 0;
  int flag = 0;
  int temp = 0;
  for (int i = 0; i < n; ++i) {
    if (r_max >= vp[i].second) {
      for (int j = 0; j < n; ++j) {
        if (vp[i].first == copy[j].first && vp[i].second == copy[j].second) {
          temp = j;
          cout << j + 1 << ' ';
          break;
        }
      }
      flag = 1;
      break;
    }
    r_max = max(r_max, vp[i].second);
  }
  if (flag)
    for (int i = 0; i < n; ++i) {
      if (r_max == vp[i].second) {
        for (int j = 0; j < n; ++j) {
          if (vp[i].first == copy[j].first && vp[i].second == copy[j].second &&
              j != temp) {
            cout << j + 1 << ' ';
            goto l1;
          }
        }
      }
    }
l1:;
  if (!flag) {
    cout << -1 << ' ' << -1 << endl;
  }
}
