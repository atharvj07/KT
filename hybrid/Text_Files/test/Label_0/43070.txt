import sys
readline = sys.stdin.readline

def RLC(S):
	pre = None
	sp = []
	num = []
	cnt = 0
	for s in S + [None]:
		if pre != s:
			sp.append(pre)
			num.append(cnt)
			cnt = 1
		else:
			cnt += 1
		pre = s
	return (sp[1:], num[1:])
T = int(readline())
Ans = [None] * T
for qu in range(T):
	S = list(map(ord, readline().strip()))
	(sp, num) = RLC(S)
	An = set()
	L = len(sp)
	for i in range(L):
		if num[i] % 2 == 1:
			An.add(sp[i])
	Ans[qu] = An
for a in Ans:
	if a:
		a = list(a)
		a.sort()
	sys.stdout.write(''.join(list(map(chr, a)) + ['\n']))
