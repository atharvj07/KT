#include <bits/stdc++.h>
using namespace std;
const int inf = 1000000000;
int main() {
  int n;
  cin >> n;
  int f = 0, i, a, c1 = 0, c2 = 0, c3 = 0;
  for (i = 0; i < n; i++) {
    cin >> a;
    if (f == 0) {
      if (a == 25) {
        c1++;
      } else if (a == 50) {
        c2++;
        c1--;
        if (c1 < 0) {
          f = 1;
        }
      } else {
        c3++;
        if (c2 >= 1 && c1 >= 1) {
          c2--;
          c1--;
        } else if (c1 >= 3) {
          c1 -= 3;
        } else
          f = 1;
      }
    }
  }
  if (f == 1)
    cout << "NO";
  else
    cout << "YES";
  return 0;
}
