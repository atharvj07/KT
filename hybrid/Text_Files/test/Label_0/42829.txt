t = int(input())
for i in range(t):
	n = int(input())
	arr = list(map(int, input().split(' ')))
	if n == 1:
		print('YES')
		print(arr[0])
		continue
	avg = 0
	for j in arr:
		avg += j
	avg = 2 * avg / (n * (n + 1))
	for j in range(len(arr)):
		arr[j] = arr[j] - avg
	a = []
	e = True
	x1 = arr[0] - arr[n - 1]
	x1 -= avg
	x1 = x1 / n
	x1 = -x1
	if x1 - int(x1) != 0 or x1 <= 0:
		e = False
	a += [int(x1)]
	for j in range(1, n):
		x = arr[j] - arr[j - 1]
		x -= avg
		x = x / n
		x = -x
		if x - int(x) != 0 or x <= 0:
			e = False
			break
		a += [int(x)]
	if e:
		print('YES')
		for q in a:
			print(q, end=' ')
		print('')
	else:
		print('NO')
