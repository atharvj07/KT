#include <bits/stdc++.h>
using namespace std;
void low(string& q) {
  for (int i = 0; i < q.length(); i++)
    if (q[i] < 'a') q[i] += 'a' - 'A';
  return;
}
int main() {
  int n;
  cin >> n;
  string* A = new string[n];
  string W;
  char ch;
  for (int i = 0; i < n; i++) {
    cin >> A[i];
    low(A[i]);
  }
  cin >> W;
  cin >> ch;
  bool* mask = new bool[W.length()];
  bool* nch = new bool[W.length()];
  for (int i = 0; i < W.length(); i++) nch[i] = false;
  for (int i = 0; i < W.length(); i++) mask[i] = (W[i] < 'a');
  low(W);
  for (int i = 0; i < n; i++) {
    int s = -1;
    while ((s = W.find(A[i], s + 1)) != string::npos)
      for (int j = s; j <= s + A[i].length() - 1; j++) nch[j] = true;
  }
  for (int i = 0; i < W.length(); i++)
    if (nch[i])
      if (W[i] != ch)
        W[i] = ch;
      else if (ch != 'a')
        W[i] = 'a';
      else
        W[i] = 'b';
  for (int i = 0; i < W.length(); i++)
    if (mask[i]) W[i] += 'A' - 'a';
  cout << W;
  return 0;
}
