s = input()
n = len(s)
c = s.count('c')
a = s.count('a')
b = s.count('b')
if c not in [a, b] or a * b * c == 0:
	print('NO')
elif c == a:
	if s[:a] == a * 'a' and s[n - c:] == c * 'c':
		print('YES')
	else:
		print('NO')
elif c == b:
	if s[a:n - c] == b * 'b' and s[n - c:] == c * 'c':
		print('YES')
	else:
		print('NO')
else:
	print('NO')
