n = int(input())
nums = list(map(int, input().split()))
left = [0] * n
right = [0] * n
for i in range(1, n):
	if nums[i] > nums[i - 1]:
		left[i] = left[i - 1] + 1
for i in range(n - 2, -1, -1):
	if nums[i] > nums[i + 1]:
		right[i] = right[i + 1] + 1

def main():
	ml = max(left)
	mr = max(right)
	if ml != mr:
		return print(0)
	(mlc, mrc) = (left.count(ml), right.count(mr))
	if mlc + mrc == 2:
		for p in range(n):
			if right[p] == mr and left[p] == ml and (not ml % 2):
				return print(1)
	return print(0)
main()
