import sys
input = sys.stdin.readline
n = int(input())
a = [0]
a.extend(list(map(int, input().split())))
s = sorted(a)
for i in range(1, n + 1):
	a[i] += a[i - 1]
	s[i] += s[i - 1]
for i in range(int(input())):
	(x, l, r) = map(int, input().split())
	if x == 1:
		print(a[r] - a[l - 1])
	else:
		print(s[r] - s[l - 1])
