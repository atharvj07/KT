import java.util.*;
import java.io.*;
public class Main{
    public static void main(String args[]){
        Scanner sc=new Scanner(System.in);
        int t=sc.nextInt();
        for(int i=0;i<t;i++){
            int n=sc.nextInt();
            int []arr=new int[n];
            for(int j=0;j<n;j++){
                arr[j]=sc.nextInt();
            }
            int p=1;
            for(int j=1;j<n;j++){
                if(arr[j-1]==1 && p==1){
                    p=2;
                }
                else{
                    if(arr[j-1]==1 && p==2){
                        p=1;
                    }
                    else{
                        break;
                    }
                }
            }
            if(p==1){
                System.out.println("First");
            }
            else{
                 System.out.println("Second");
            }
        }
    }
}
