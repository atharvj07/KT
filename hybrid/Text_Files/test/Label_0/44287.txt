import sys
input = sys.stdin.readline

def solve():
	(n, m) = map(int, input().split())
	orientation = ['R', 'W']
	fixed = False
	flag = True
	for i in range(n):
		row = input()
		for j in range(m):
			if row[j] == '.':
				continue
			if not fixed:
				if orientation[(i + j) % 2] != row[j]:
					orientation.reverse()
				fixed = True
			elif orientation[(i + j) % 2] != row[j]:
				flag = False
	if not flag:
		print('NO')
		return
	print('YES')
	for i in range(n):
		for j in range(m):
			print(end=orientation[(i + j) % 2])
		print()
for _ in range(int(input())):
	solve()
