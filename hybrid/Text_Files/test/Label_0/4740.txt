#include<bits/stdc++.h>
#define pb push_back
#define int long long
using namespace std;

void solve ()
{
    int n;
    cin >> n;
    string s;
    cin >> s;
    string a = s , b = s;
    if ( s [ 0 ] == '0' || s . back () == '0' )
    {
        cout << "NO" << endl;
        return;
    }
    int zero = 0;
    for ( auto i : s )
    {
        if ( i == '0' )
            zero++;
    }

    if ( zero & 1 )
    {
        cout << "NO" << endl;
        return;
    }
    int ones = n - zero;
    int need = ones / 2;
    for ( int i = 0 ; i < n && need ; i++ )
    {
        if ( s [ i ] == '1' )
        {
            a [ i ] = b [ i ] = '(';
            need--;
        }
    }

    need = ones / 2;
    for ( int i = n - 1 ; i >= 0 && need ; i-- )
    {
       if ( s [ i ] == '1' )
        {
            a [ i ] = b [ i ] = ')';
            need--;
        }
    }

    int open = 1;
    for ( int i = 0 ; i < n ; i++ )
    {
        if ( s [ i ] == '0' )
        {
            if ( open )
            {
                a [ i ] = '(';
                b [ i ] = ')';
                open = 1 - open;
            }
            else
            {
                a [ i ] = ')';
                b [ i ] = '(';
                open = 1 - open ;
            }
        }
    }

    cout << "YES" << endl;
    cout << a << endl << b << endl;
}


int32_t main()
{
        ios::sync_with_stdio(false);cin.tie(0);cout.tie(0);

        int t = 1;
        cin >> t;

        while ( t-- )
            solve ();

        return 0;
}
