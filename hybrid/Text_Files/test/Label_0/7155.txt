#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, m;
  cin >> n >> m;
  if (m % n != 0) {
    cout << "-1" << endl;
    return 0;
  }
  long long buf = (m / n);
  int ans = 0;
  while (buf % 2 == 0) {
    buf /= 2;
    ans++;
  }
  while (buf % 3 == 0) {
    buf /= 3;
    ans++;
  }
  if (buf == 1)
    cout << ans << endl;
  else
    cout << "-1" << endl;
  return 0;
}
