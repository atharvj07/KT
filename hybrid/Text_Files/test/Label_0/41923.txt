from sys import stdin
from math import fabs
t = int(stdin.readline())
for _ in range(t):
	(n, k, d1, d2) = map(int, stdin.readline().split())
	flg = False
	w = k + 2 * d1 + d2
	a = w // 3
	if w >= 0 and w % 3 == 0 and (a >= d1 + d2) and (2 * d1 + d2 <= n - k) and ((n - k - 2 * d1 - d2) % 3 == 0):
		flg = True
	w = k + 2 * d1 - d2
	a = w // 3
	if w >= 0 and w % 3 == 0 and (a >= d1) and (a >= d1 - d2):
		req = 0
		if d2 < d1:
			req = 2 * d1 - d2
		else:
			req = 2 * d2 - d1
		if n - k >= req and (n - k - req) % 3 == 0:
			flg = True
	w = k - 2 * d1 + d2
	a = w // 3
	if w >= 0 and w % 3 == 0 and (a >= d2 - d1) and (d1 + d2 <= n - k) and ((n - k - d1 - d2) % 3 == 0):
		flg = True
	w = k - 2 * d1 - d2
	if w >= 0 and w % 3 == 0 and (2 * d2 + d1 <= n - k) and ((n - k - 2 * d2 - d1) % 3 == 0):
		flg = True
	if n % 3:
		flg = False
	if flg:
		print('yes')
	else:
		print('no')
