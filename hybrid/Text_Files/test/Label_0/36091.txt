

import java.io.*;
import java.math.*;
import java.util.*;

public class Main{
    public static void main(String[] args){
        InputStream inputStream = System.in;//new FileInputStream("C:\\Users\\xxx\\Downloads\\test.in");
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        Task solver = new Task();
        solver.solve(in, out);
        out.close();
    }
    static class Task {
        static int MAXN =(int)(2e5+10);
        static  int a[]=new int[MAXN];
        static  long sum[]=new long[MAXN];
        static int nxt[]=new int[MAXN];
        public void solve(InputReader in, PrintWriter out) {
            int n=in.nextInt();
            BigInteger k=in.nextBigInteger();
            for(int i=1;i<=n;i++){
                a[i]=in.nextInt();
                sum[i]=sum[i-1]+a[i];
            }
            int now=0;
            for(int i=n;i>=1;i--){
                nxt[i]=now;
                if(a[i]!=1)now=i;
            }
            long ans=0;
            for(int i=1;i<=n;i++){
                BigInteger son=BigInteger.valueOf(a[i]);
                int _now=i;
                for(int j=1;j<=60;j++){
                    int bb=nxt[_now];
                    if(bb==0){
                        if(son.remainder(k).compareTo(BigInteger.ZERO)==0&&k.multiply(BigInteger.valueOf(sum[_now]-sum[i-1])).compareTo(son)<=0&&k.multiply(BigInteger.valueOf(sum[n]-sum[i-1])).compareTo(son)>=0){
                            ans++;
                        }
                        break;
                    }else{
                        if(son.remainder(k).compareTo(BigInteger.ZERO)==0&&k.multiply(BigInteger.valueOf(sum[_now]-sum[i-1])).compareTo(son)<=0&&k.multiply(BigInteger.valueOf(sum[bb-1]-sum[i-1])).compareTo(son)>=0) {
                            ans++;
                        }
                        son=son.multiply(BigInteger.valueOf(a[bb]));
                        if(son.compareTo(k.multiply(BigInteger.valueOf(sum[n])))>=0){
                            break;
                        }
                        _now=bb;
                    }
                   // out.println(ans);
                }
            }
            out.println(ans);
        }
    }
    static class InputReader {
        public BufferedReader reader;
        public StringTokenizer tokenizer;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream), 32768);
            tokenizer = null;
        }

        public String next() {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

        public long nextLong() {
            return Long.parseLong(next());
        }

        public double nextDouble() {
            return Double.parseDouble(next());
        }

        public char[] nextCharArray() {
            return next().toCharArray();
        }

        public boolean hasNext() {
            try {
                String string = reader.readLine();
                if (string == null) {
                    return false;
                }
                tokenizer = new StringTokenizer(string);
                return tokenizer.hasMoreTokens();
            } catch(IOException e) {
                return false;
            }
        }
        public BigInteger nextBigInteger() {
            return new BigInteger(next());
        }

        public BigDecimal nextBigDecinal() {
            return new BigDecimal(next());
        }
    }
}
