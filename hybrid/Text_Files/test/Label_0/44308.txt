def dense_array(list):
	count = 0
	for x in range(len(list) - 1):
		minimo = min(list[x], list[x + 1])
		maximo = max(list[x], list[x + 1])
		while maximo / minimo > 2:
			minimo = minimo * 2
			count += 1
	return count

	lista = [int(x) for x in input().split()]
	print(dense_array(lista))
