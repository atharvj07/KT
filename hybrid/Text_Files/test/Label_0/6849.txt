template <size_t N>
string to_string(bitset<N> &b, int x1 = 0, int x2 = 1e9) {
  string t = "";
  for (int __iii__ = min(x1, SIZE(b)), __jjj__ = min(x2, SIZE(b) - 1);
       __iii__ <= __jjj__; ++__iii__) {
    t += b[__iii__] + '0';
  }
  return '"' + t + '"';
}

int main() {

  cin.tie(NULL);
  cout.tie(NULL);
  long long t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    map<int, int> cnt;
    for (int i = 0, a; i < n; ++i) {
      cin >> a;
      ++cnt[a];
    }
    vector<int> v;
    for (auto &x : cnt) {
      v.push_back(x.second);
    }
    sort(v.begin(), v.end(), greater<int>());
    long long ans = 0;
    for (int i = 0; i < v.size(); ++i) {
      if (i) {
        if (v[i] >= v[i - 1]) {
          v[i] = v[i - 1] - 1;
          if (!v[i]) {
            break;
          }
        }
      }
      ans += v[i];
    }
    cout << ans << "\n";
  }
}
