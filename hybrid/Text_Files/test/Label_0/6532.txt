#include<iostream>
using namespace std;
int main(){
	int n,a[3];
	cin >> n >> a[0] >> a[1] >> a[2];
	int l[n];
	for(int i=0;i<n;i++) cin >> l[i];

	int ans=1e9;
	for(int i=0;i<(1<<2*n);i++){
		int tmp=0;
		int length[4]={};
		for(int j=0;j<n;j++){
			int index=(i>>2*j)&3;
			if(index!=3&&length[index]>0) tmp+=10;
			length[index]+=l[j];
		}
		bool flag=false;
		for(int j=0;j<3;j++){
			if(length[j]==0){
				flag=true;
				break;
			}
			tmp+=abs(length[j]-a[j]);
		}
		if(flag) continue;
		ans=min(ans,tmp);
	}
	cout << ans << endl;
	return 0;
}
