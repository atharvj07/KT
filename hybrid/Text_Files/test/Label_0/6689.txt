#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using Line = pair<int, ll>;
const int N = 2e5 + 5;
const ll INF = 0x3f3f3f3f3f3f3f3fLL;
int n, a[N], second[N], rt[N], tot;
ll f[N];
struct Tree {
  int l, r;
  Line s;
} t[N * 25];
basic_string<int> G[N];
ll Y(Line l, int x) { return 1LL * l.first * x + l.second; }
void upd(int &rt, int l, int r, Line v) {
  if (!rt) {
    t[rt = ++tot].s = v;
    return;
  }
  int mid = l + r >> 1;
  if (Y(t[rt].s, mid) > Y(v, mid)) swap(t[rt].s, v);
  if (l == r) return;
  if (t[rt].s.first < v.first)
    upd(t[rt].l, l, mid, v);
  else
    upd(t[rt].r, mid + 1, r, v);
}
int merge(int x, int y, int l, int r) {
  if (!x || !y) return x | y;
  int mid = l + r >> 1;
  if (l == r) {
    if (Y(t[x].s, l) > Y(t[y].s, l)) t[x].s = t[y].s;
    return x;
  }
  t[x].l = merge(t[x].l, t[y].l, l, mid),
  t[x].r = merge(t[x].r, t[y].r, mid + 1, r);
  upd(x, l, r, t[y].s);
  return x;
}
ll qry(int rt, int l, int r, int p) {
  if (!rt) return INF;
  if (l == r) return Y(t[rt].s, p);
  int mid = l + r >> 1;
  if (p <= mid)
    return min(Y(t[rt].s, p), qry(t[rt].l, l, mid, p));
  else
    return min(Y(t[rt].s, p), qry(t[rt].r, mid + 1, r, p));
}
void dfs(int u, int fa) {
  bool fl = 1;
  for (int v : G[u])
    if (v != fa) dfs(v, u), rt[u] = merge(rt[u], rt[v], -N, N), fl = 0;
  if (!fl) f[u] = qry(rt[u], -N, N, a[u]);
  upd(rt[u], -N, N, (Line){second[u], f[u]});
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; ++i) scanf("%d", a + i);
  for (int i = 1; i <= n; ++i) scanf("%d", second + i);
  for (int i = 1, x, y; i < n; ++i) scanf("%d%d", &x, &y), G[x] += y, G[y] += x;
  dfs(1, 0);
  for (int i = 1; i <= n; ++i) printf("%lld ", f[i]);
  return 0;
}
