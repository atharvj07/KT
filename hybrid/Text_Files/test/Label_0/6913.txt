#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  string p = "CODEFORCES";
  int ft = 0;
  while (ft < s.size() && ft < p.size() && p[ft] == s[ft]) ft++;
  int lt = 0;
  while (lt < s.size() && lt < p.size() &&
         p[p.size() - lt - 1] == s[s.size() - lt - 1])
    lt++;
  if (ft + lt >= p.size())
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
