#include <iostream>
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <algorithm>
#include <vector>
#include <queue>
#include <stack>
#include <cmath>

using namespace std;

const int MAXN = 3000 + 100;
const int MOD = 1000000000 + 7;

int n, m;
char s[MAXN];
int sum[MAXN];
int f[MAXN][MAXN], rig[MAXN];

void add(int &x, int y)
{
	x = (x + y) % MOD;
}

int main()
{
	ios::sync_with_stdio(false);
	// freopen("1.in", "r", stdin);
	// freopen("1.out", "w", stdout);

	cin >> n >> m;
	cin >> (s + 1);
	for(int i = 1; i <= n; i++)
		sum[i] = sum[i - 1] + (s[i] == '1');
	for(int i = 1; i <= n; i++)
		rig[i] = i;
	for(int i = 1; i <= m; i++)
	{
		int l, r;
		cin >> l >> r;
		rig[l] = max(rig[l], r);
	}
	for(int i = 1; i <= n; i++)
		rig[i] = max(rig[i], rig[i - 1]);

	// for(int i = 1; i <= n; i++)
	// 	cerr << i << ' ' << rig[i] << endl;

	f[0][0] = 1;
	for(int i = 0; i <= n - 1; i++)
		for(int j = 0; j <= rig[i] - i + 1; j++)
			if(f[i][j])
			{
				int a = j + sum[rig[i + 1]] - sum[rig[i]];
				int b = rig[i + 1] - (i + 1) + 1 - a;
				if(a >= 1)
					add(f[i + 1][a - 1], f[i][j]);
				if(b >= 1)
					add(f[i + 1][a], f[i][j]);
			}
	int ans = f[n][0];
	cout << ans << endl;

	return 0;
}
