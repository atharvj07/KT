const int inf=1e9;
const int64_t inf64=1e18;
const double eps=1e-9;

template <typename T> ostream &operator<<(ostream &os, const vector<T> &vec){
    os << "[";
    for (const auto &v : vec) {
    	os << v << ",";
    }
    os << "]";
    return os;
}

void solve(){
    int h,w,n;
    cin >> h >> w >> n;
    map<int,set<pair<int,int>>> erased1,erased2;
    rep(i,0,n){
        int a,b;
        cin >> a >> b;
        --a;
        --b;
        erased1[b-a].insert(make_pair(a,b));
        erased2[a+b+1].insert(make_pair(a,b+1));
    }

    rep(i,0,w){
        int x=i,y=0;
        while(y<=h-1){
            if(x%2==y%2){
                if(x==w-1) ++y;
                else{
                    int d=min(w-x-1,h-y);
                    int nx=x+d,ny=y+d;
                    auto it=erased1[x-y].lower_bound(make_pair(y,-inf));
                    if(it!=erased1[x-y].end()){
                        int bx=it->second,by=it->first;
                        nx=bx;
                        ny=by+1;
                    }
                    x=nx;
                    y=ny;
                }
            }else{
                if(x==0) ++y;
                else{
                    int d=min(x,h-y);
                    int nx=x-d,ny=y+d;
                    auto it=erased2[x+y].lower_bound(make_pair(y,-inf));
                    if(it!=erased2[x+y].end()){
                        int bx=it->second,by=it->first;
                        nx=bx;
                        ny=by+1;
                    }
                    x=nx;
                    y=ny;
                }
            }
        }
        cout << x+1 << endl;
    }
}

int main(){
    std::cin.tie(0);
    std::ios::sync_with_stdio(false);
    cout.setf(ios::fixed);
    cout.precision(10);
    solve();
    return 0;
}
