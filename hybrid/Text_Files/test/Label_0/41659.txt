from sys import stdin

def solve(tc):
	(x, y) = map(int, stdin.readline().split())
	(a, b) = map(int, stdin.readline().split())
	diff = abs(x - y)
	mini = min(x, y)
	ans = diff * a
	if mini * 2 * a < mini * b:
		ans += mini * 2 * a
	else:
		ans += mini * b
	print(ans)
tcs = int(stdin.readline().strip())
tc = 1
while tc <= tcs:
	solve(tc)
	tc += 1
