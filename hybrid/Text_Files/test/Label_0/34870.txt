class Flow{
	List<Edge>[] edges; 
	int src, sink,n;
	int[] level;

	Flow(int n,int src,int sink){
		this.n=n;
		this.src=src;
		this.sink=sink;
		edges=new ArrayList[n+1];
		for (int i = 0; i < edges.length; i++) {
			edges[i]=new ArrayList<>();
		}
		level=new int[n+1];
	}
	
	public void addEdge(int f,int to, long val) {
		Edge forward=new Edge(f, to, val);
		Edge backward=new Edge(to, f, 0L);
		forward.residual=backward;
		backward.residual=forward;
		edges[f].add(forward);
		edges[to].add(backward);
	}
	
	public long getMinCut() {
		long inf=Long.MAX_VALUE/2;
		long ans=0;
		while(bfs()) {
			int[] start = new int[n+1];
			long flow = dfs(src,inf,start);
//			System.out.println(flow);
			while(flow>0) {
				ans+=flow;
				flow=dfs(src,inf,start);
//				System.out.println(flow);
			}
		}
		return ans;
	}
 
	public long dfs(int curr, long flow, int[] start) {
		if(curr == sink)
            return flow;
        for(; start[curr] < edges[curr].size(); start[curr]++)
        {
            Edge e = edges[curr].get(start[curr]);
            if(e.capacityLeft() > 0 && level[e.to]-level[curr] == 1)
            {
                long val = dfs(e.to, Math.min(flow, e.capacityLeft()), start);
                if(val > 0)
                {
                    e.augment(val);
                    return val;
                }
            }
        }
        return 0L;  
	}

	public boolean bfs() {
		int def=-100;
		Arrays.fill(level, def);
		Queue<Integer> q = new LinkedList<Integer>();
		q.add(src);
		level[src]=0;
		while(!q.isEmpty()) {
			int curr=q.remove();
			for(Edge ne:edges[curr]) {
				if(ne.capacityLeft()>0&&level[ne.to]==def) {
					q.add(ne.to);
					level[ne.to]=level[curr]+1;
				}
			}	
		}
		return level[sink]!=def;
	}
	
	class Edge{
		public int from, to;
        public long flow, capacity;
        public Edge residual;
 
        public Edge(int f, int t, long cap)
        {
            from = f;
            to = t;
            capacity = cap;
            flow=0;
        }
        public long capacityLeft()
        {
            return capacity-flow;
        }
        public void augment(long val)
        {
            flow += val;
            residual.flow -= val;
        }
	}
	
}

class FastReader {
	BufferedReader br;
	StringTokenizer st;

	public FastReader() throws FileNotFoundException {
		try{
			File file = new File("src\\input.txt");
			br = new BufferedReader(new FileReader(file));
		} catch(Exception e) {
			br = new BufferedReader(new InputStreamReader(System.in));
		}

	}

	String next() {
		while (st == null || !st.hasMoreElements()) {
			try {
				st = new StringTokenizer(br.readLine());
			} catch (IOException e) {
				e.printStackTrace();
			}
		}
		return st.nextToken();
	}

	int nextInt() {
		return Integer.parseInt(next());
	}

	long nextLong() {
		return Long.parseLong(next());
	}

	double nextDouble() {
		return Double.parseDouble(next());
	}
	
	int[] nextInts(int N) {
        int[] res = new int[N];
        for (int i = 0; i < N; i++) {
            res[i] = (int) nextLong();
        }
        return res;
    }
 
    long[] nextLongs(int N) {
        long[] res = new long[N];
        for (int i = 0; i < N; i++) {
            res[i] = nextLong();
        }
        return res;
    }
    
    public double[] nextDoubles(int N) {
        double[] res = new double[N];
        for (int i = 0; i < N; i++) {
            res[i] = nextDouble();
        }
        return res;
    }
    
	String nextLine() {
		String str = "";
		try {
			str = br.readLine();
		} catch (IOException e) {
			e.printStackTrace();
		}
		return str;
	}
}
