#include <bits/stdc++.h>
using namespace std;
const auto INF = (long long)1e18;
const int N = 2e5 + 5;
struct node {
  int L, R, id, Len;
  int C;
} maps[N];
bool cmp(const struct node &u, const struct node &v) {
  if (u.L == v.L) return u.id > v.id;
  return u.L < v.L;
}
vector<node> ans;
map<int, int> E;
int main() {
  int n, x;
  scanf("%d%d", &n, &x);
  for (int i = 1; i <= n; i++) {
    scanf("%d%d%d", &maps[i].L, &maps[i].R, &maps[i].C);
    struct node temp;
    temp.L = maps[i].L, temp.id = 1;
    temp.Len = maps[i].R - maps[i].L + 1;
    temp.C = maps[i].C;
    ans.push_back(temp);
    temp.L = maps[i].R, temp.id = -1;
    ans.push_back(temp);
  }
  sort(ans.begin(), ans.end(), cmp);
  long long answer = INF;
  for (int i = 0; i < (int)ans.size(); i++) {
    struct node temp = ans[i];
    if (temp.id == -1) {
      if (E.find(temp.Len) == E.end()) {
        E[temp.Len] = temp.C;
        continue;
      }
      E[temp.Len] = min(E[temp.Len], temp.C);
    } else {
      if (x <= temp.Len) continue;
      if (E.find(x - temp.Len) == E.end()) continue;
      answer = min(answer, (long long)E[x - temp.Len] + (long long)temp.C);
    }
  }
  if (answer == INF) answer = -1;
  printf("%lld\n", answer);
}
