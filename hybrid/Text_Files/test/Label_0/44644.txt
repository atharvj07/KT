w = input()
weights = []
for i in range(10):
	if w[i] == '1':
		weights.append(i + 1)
import sys
sys.setrecursionlimit(100000)
m = int(input())
stack = [0] * (m + 1)

def dfs(i, j, k):
	global m
	if k == m:
		return True
	for weight in weights:
		if weight == j or weight <= abs(i):
			continue
		else:
			if (k + 1) % 2 == 0:
				result = dfs(i - weight, weight, k + 1)
			else:
				result = dfs(i + weight, weight, k + 1)
			if result:
				stack[k + 1] = weight
				return True
	return False
result = dfs(0, 0, 0)
if result:
	print('YES')
	print(*stack[1:])
else:
	print('NO')
