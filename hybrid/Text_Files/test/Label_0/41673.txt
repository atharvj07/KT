import sys

def get_array():
	return list(map(int, sys.stdin.readline().strip().split()))

def get_ints():
	return map(int, sys.stdin.readline().strip().split())

def input():
	return sys.stdin.readline().strip()
(n, m) = get_ints()
mat = []
for i in range(n):
	mat.append(list(input()))
for i in range(n):
	for j in range(m):
		if mat[i][j] == '.':
			if i + j & 1:
				mat[i][j] = 'W'
			else:
				mat[i][j] = 'B'
for i in mat:
	print(''.join(i))
