t = int(input())
for _ in range(t):
	n = int(input())
	on = n
	a = []
	for i in range(2, 1 + int(n ** 0.5)):
		if not n % i:
			a.append(i)
			n //= i
		if len(a) == 2:
			break
	if len(a) < 2 or on // (a[0] * a[1]) == 1 or on // (a[0] * a[1]) in a:
		print('NO')
	else:
		print('YES')
		a.append(on // (a[0] * a[1]))
		print(*a)
