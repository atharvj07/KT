[n, k] = [int(item) for item in input().split(' ')]
increasing = range(1, n + 1)
decreasing = reversed(increasing)
index = 0
seq = []
for item in decreasing:
	if index >= k:
		break
	index += 1
	seq.append(item)
for item in increasing:
	if index >= n:
		break
	index += 1
	seq.append(item)
print(' '.join(map(str, seq)))
