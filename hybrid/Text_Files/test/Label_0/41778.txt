def f(a, n):
	i = 1
	while i < n and a[i] >= a[i - 1]:
		i += 1
	if i == n:
		return 0
	else:
		(st1, en1) = (i, n - 1)
		(st2, en2) = (0, i - 1)
		for j in range(st1, en1):
			if a[j] <= a[j + 1]:
				continue
			else:
				return -1
		for j in range(st2, en2):
			if a[j] <= a[j + 1]:
				continue
			else:
				return -1
		if a[0] >= a[-1]:
			return n - i
		else:
			return -1
n = int(input())
a = list(map(int, input().split()))
print(f(a, n))
