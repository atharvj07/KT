#include <bits/stdc++.h>
using namespace std;
int main() {
  long long t;
  cin >> t;
  while (t--) {
    long long n;
    cin >> n;
    long long a[n], ev = 0, odd = 0;
    for (long long i = 0; i < n; i++) {
      cin >> a[i];
      if (a[i] % 2 != 0)
        odd++;
      else
        ev++;
    }
    if (odd % 2 != 0)
      cout << "YES" << endl;
    else if (odd % 2 == 0 && odd != 0) {
      if (ev > 0)
        cout << "yes" << endl;
      else
        cout << "NO" << endl;
    } else
      cout << "NO" << endl;
  }
}
