public class Main {
	public static void main(String[] args) {
		InputStream inputStream = System.in;
		OutputStream outputStream = System.out;
		PandaScanner in = new PandaScanner(inputStream);
		PrintWriter out = new PrintWriter(outputStream);
		C solver = new C();
		solver.solve(1, in, out);
		out.close();
	}
}

class C {
    String[] words = new String[6];
    char[] front = new char[6];
    char[] back = new char[6];
    int[] length = new int[6];
    int[] order = new int[6];
    String[] solution;
    public void solve(int testNumber, PandaScanner in, PrintWriter out) {
        for (int i = 0; i < 6; i++) {
            words[i] = in.next();
            length[i] = words[i].length();
            front[i] = words[i].charAt(0);
            back[i] = words[i].charAt(length[i] - 1);
        }
        search(0, 0);
        if (solution == null) {
            out.println("Impossible");
        }
        else {
            for (String line: solution) {
                out.println(line);
            }
        }
    }
    void search(int idx, int mask) {
        if (idx == 6) {
            char[][] newBoard = new char[length[order[2]]][length[order[3]]];
            for (int i = 0; i < length[order[2]]; i++) {
                Arrays.fill(newBoard[i], '.');
            }
            for (int i = 0; i < length[order[0]]; i++) {
                newBoard[0][i] = words[order[0]].charAt(i);
            }
            for (int i = 0; i < length[order[1]]; i++) {
                newBoard[i][0] = words[order[1]].charAt(i);
            }
            for (int i = 0; i < length[order[2]]; i++) {
                newBoard[i][length[order[0]] - 1] = words[order[2]].charAt(i);
            }
            for (int i = 0; i < length[order[3]]; i++) {
                newBoard[length[order[1]] - 1][i] = words[order[3]].charAt(i);
            }
            for (int i = 0; i < length[order[4]]; i++) {
                newBoard[length[order[1]] + i - 1][length[order[3]] - 1] = words[order[4]].charAt(i);
            }
            for (int i = 0; i < length[order[5]]; i++) {
                newBoard[length[order[2]] - 1][length[order[0]] + i - 1] = words[order[5]].charAt(i);
            }
            String[] newSolution = new String[length[order[2]]];
            for (int i = 0; i < length[order[2]]; i++) {
                newSolution[i] = new String(newBoard[i]);
            }
            solution = min(solution, newSolution);
            return;
        }
        for (int i = 0; i < 6; i++) {
            if ((mask & (1 << i)) == 0) {
                order[idx] = i;
                if (idx == 1 && front[order[0]] != front[i]) {
                    continue;
                }
                if (idx == 2 && (length[i] < length[order[1]] + 2 || back[order[0]] != front[i])) {
                    continue;
                }
                if (idx == 3 && (length[i] < length[order[0]] + 2 || back[order[1]] != front[i] || words[order[2]].charAt(length[order[1]] - 1) != words[i].charAt(length[order[0]] - 1))) {
                    continue;
                }
                if (idx == 4 && (back[order[3]] != front[i] || length[order[1]] + length[i] != length[order[2]] + 1)) {
                    continue;
                }
                if (idx == 5 && (back[order[2]] != front[i] || back[i] != back[order[4]] || length[order[0]] + length[i] != length[order[3]] + 1)) {
                    continue;
                }
                search(idx + 1, mask | (1 << i));
            }
        }
    }
    public String[] min(String[] a, String[] b) {
        if (a == null) {
            return b;
        }
        for (int i = 0; i < Math.min(a.length, b.length); i++) {
            if (a[i].compareTo(b[i]) != 0) {
                return a[i].compareTo(b[i]) < 0 ? a : b;
            }
        }
        return a.length - b.length < 0 ? a : b;
    }
}

class PandaScanner {
    public BufferedReader br;
    public StringTokenizer st;
    public InputStream in;

    public PandaScanner(InputStream in) {
        br = new BufferedReader(new InputStreamReader(this.in = in));
    }

    public String nextLine() {
        try {
            return br.readLine();
        }
        catch (Exception e) {
            return null;
        }
    }

    public String next() {
        if (st == null || !st.hasMoreTokens()) {
            st = new StringTokenizer(nextLine().trim());
            return next();
        }
        return st.nextToken();
    }

    }
