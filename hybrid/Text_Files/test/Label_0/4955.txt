#include <bits/stdc++.h>
using namespace std;
mt19937_64 mt(time(0));
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cout.precision(10);
  cout << fixed;
  int t;
  cin >> t;
  while (t--) {
    int n, x, val;
    cin >> n >> x;
    map<int, int> table;
    for (int i = 0; i < n; i++) {
      cin >> val;
      table[val] = 1;
    }
    int ret = 0;
    for (int i = 1;; i++) {
      if (table[i] == 0) x--;
      if (x == -1) {
        ret = i - 1;
        break;
      }
    }
    cout << ret << endl;
  }
  return 0;
}
