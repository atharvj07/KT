import java.util.Scanner;

public class Main {
    private static int n, k, l, r, sa, sk;
    private static Scanner entrada;

    public static void main(String[] args) {
        entrada = new Scanner(System.in);
        int values[];
        
        while (1 == 1) {
            try {
            
                int aux = 0, value;
                values = new int[1004];
                
                n = entrada.nextInt();
                k = entrada.nextInt();
                l = entrada.nextInt();
                r = entrada.nextInt();
                sa = entrada.nextInt();
                sk = entrada.nextInt();
                
                value = sk/k;
                
                for (int i = 0; i < sk - k*value; i++){
                    values[aux++] = value + 1;
                }
                    
                for (int i = 0; i < k - (sk - k*value); i++){
                    values[aux++] = value;
                }
                
                if (n - k>0){
                    value = (sa - sk) / (n - k);
                    for (int i = 0; i < (sa - sk) - value*(n - k); i++){
                        values[aux++] = value + 1;
                    }
                    while (aux < n){
                        values[aux++] = value;
                    }
                }
                System.out.print(values[0]);
                for (int i = 1; i < n; i++) {
                    System.out.print(" "+ values[i]);
                }
                System.out.println();
                
            } catch (Exception e) {
                break;
            }

        }

    }

}
