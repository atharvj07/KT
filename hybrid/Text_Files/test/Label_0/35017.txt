import java.io.*;
import java.util.*;

public class Main extends Thread {
    public Main() {
        try {
            this.input = new BufferedReader(new InputStreamReader(System.in));
            this.output = new PrintWriter(System.out);
            this.setPriority(Thread.MAX_PRIORITY);
        } catch (Exception e) {
            System.exit(666);
        }
    }


    private void solve() throws Throwable {
        int a = nextInt(), b = nextInt(), c = nextInt(), d = nextInt(), n = nextInt();
        int cnt[] = new int[n];
        for (int i = 0; i < n; ++i) {
            cnt[i] = nextInt();
        }
        boolean reversed = false;
        if (b < d) {
            reversed = true;
            int t = a;
            a = c;
            c = t;
            t = b;
            b = d;
            d = t;
        }
        char [][]grid = new char[b][a + c];
        for (int i = d; i < b; ++i) {
            for (int j = a; j < a + c; ++j) {
                grid[i][j] = '.';
            }
        }
        int to = d % 2 == 1 ? -1 : 1;
        int lastX = 0, lastY = to == 1 ? 0 : a + c - 1;
        for (int WHERE = 0; WHERE < n; ++WHERE) {
            while (cnt[WHERE] != 0) {
                grid[lastX][lastY] = (char)('a'+ WHERE);
                --cnt[WHERE];
                if (to == 1) {
                    int limit = lastX < d ? a + c : a;
                    if (lastY + to < limit) {
                        ++lastY;
                    } else {
                        to = -to;
                        ++lastX;
                    }
                } else {
                    if (lastY + to >= 0) {
                        --lastY;
                    } else {
                        to  = - to;
                        ++lastX;
                    }
                }
            }
        }
        output.println("YES");
        for (int i = 0; i < b; ++i) {
            if (reversed) {
                for (int j = a + c - 1; j >= 0; --j) {
                    output.print(grid[i][j]);
                }
            } else {
                for (int j = 0; j < a + c; ++j) {
                    output.print(grid[i][j]);
                }
            }
            output.println();
        }
    }


    public void run() {
        try {
            solve();
        } catch (Throwable e) {
            System.err.println(e.getMessage());
            e.printStackTrace();
            System.exit(666);
        } finally {
            output.flush();
            output.close();
        }
    }

    public static void main(String[] args) {
        new Main().start();
    }

    private int nextInt() throws IOException {
        return Integer.parseInt(nextToken());
    }

    private long nextLong() throws IOException {
        return Long.parseLong(nextToken());
    }

    private double nextDouble() throws IOException {
        return Double.parseDouble(nextToken());
    }

    private String nextToken() throws IOException {
        while (tokens == null || !tokens.hasMoreTokens()) {
            tokens = new StringTokenizer(input.readLine());
        }
        return tokens.nextToken();
    }

    private BufferedReader input;
    private PrintWriter output;
    private StringTokenizer tokens = null;
}
