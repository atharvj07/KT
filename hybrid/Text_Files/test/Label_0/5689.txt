#include <bits/stdc++.h>
using namespace std;
int n;
set<int> Hash;
int gcd(int x, int y) { return y ? gcd(y, x % y) : x; }
int main() {
  int a, b;
  scanf("%d%d", &a, &b);
  int g = gcd(a, b);
  scanf("%d", &n);
  Hash.clear();
  for (int i = 1; i * i <= g; i++)
    if (g % i == 0) {
      Hash.insert(i);
      Hash.insert(g / i);
    }
  for (int i = 0; i < n; i++) {
    int x, y;
    scanf("%d%d", &x, &y);
    int ret = -1;
    for (set<int>::iterator it = Hash.begin(); it != Hash.end(); it++)
      if (x <= *it && *it <= y && *it > ret) ret = *it;
    printf("%d\n", ret);
  }
}
