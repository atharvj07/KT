#include <bits/stdc++.h>
using namespace std;
int get01ValAtPos(int val, int pos) {
  if ((val >> pos) & 1 != 0) {
    return 1;
  } else {
    return 0;
  }
}
int getMaxExp(int val) {
  for (int curExp = 0;; curExp++) {
    if (val / (1 << curExp) == 0) {
      return curExp - 1;
    }
  }
}
int main() {
  int inputLen = 0;
  vector<int> *vecInput = new vector<int>;
  vector<int> *vecRes = new vector<int>;
  while (scanf("%d", &inputLen) != -1) {
    vecInput->clear();
    for (int i = 0; i < inputLen; i++) {
      int oneInput;
      scanf("%d", &oneInput);
      vecInput->push_back(oneInput);
    }
    int maxExp = getMaxExp(vecInput->back());
    for (int curExp = maxExp; curExp >= 0; curExp--) {
      vecRes->clear();
      int andRes = 0x7fffffff;
      for (vector<int>::iterator iter = vecInput->begin();
           iter != vecInput->end(); iter++) {
        if (get01ValAtPos(*iter, curExp) == 1) {
          andRes &= *iter;
          vecRes->push_back(*iter);
        }
      }
      if (andRes % (1 << curExp) == 0) {
        printf("%d\n", vecRes->size());
        for (vector<int>::iterator iter = vecRes->begin();
             iter != vecRes->end(); iter++) {
          if (iter == vecRes->begin()) {
            printf("%d", *iter);
          } else {
            printf(" %d", *iter);
          }
        }
        printf("\n");
        break;
      }
    }
  }
  return 0;
}
