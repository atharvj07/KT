import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.HashMap;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.StringTokenizer;
import java.util.Map.Entry;
import java.io.BufferedReader;
import java.util.Collections;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskD solver = new TaskD();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskD {
        final int HOR = 2;

        public void solve(int testNumber, InputReader in, PrintWriter out) {
            int n = in.nextInt();
            int w = in.nextInt();
            int h = in.nextInt();
            int[] g = new int[n];
            int[] p = new int[n];
            int[] t = new int[n];
            HashMap<Integer, ArrayList<Dancer>> horMap = new HashMap<>();
            HashMap<Integer, ArrayList<Dancer>> verMap = new HashMap<>();

            for (int i = 0; i < n; i++) {
                g[i] = in.nextInt();
                p[i] = in.nextInt();
                t[i] = in.nextInt();
                int diff = t[i] - p[i];
                ArrayList<Dancer> list;
                if (g[i] == HOR) {
                    if (!horMap.containsKey(diff))
                        horMap.put(diff, new ArrayList<>());
                    list = horMap.get(diff);
                } else {
                    if (!verMap.containsKey(diff))
                        verMap.put(diff, new ArrayList<>());
                    list = verMap.get(diff);
                }
                list.add(new Dancer(i, diff, g[i], t[i]));

            }

            for (ArrayList<Dancer> list : horMap.values()) {
                Collections.sort(list);
            }

            for (ArrayList<Dancer> list : verMap.values()) {
                Collections.sort(list);
            }

            HashMap<Integer, Integer> ans = new HashMap<>();
            for (Entry<Integer, ArrayList<Dancer>> kv : horMap.entrySet()) {
                ArrayList<Dancer> horList = kv.getValue();
                ArrayList<Dancer> verList = verMap.getOrDefault(kv.getKey(), null);
                if (verList == null) {
                    for (Dancer d : horList)
                        ans.put(d.num, d.num);
                    continue;
                }

                int hori = horList.size() - 1;
                int veri = 0;
                while (hori >= 0 && veri < verList.size()) {
                    ans.put(horList.get(hori).num, verList.get(veri).num);
                    hori--;
                    veri++;
                }

                if (horList.size() > verList.size()) {
                    int shift = verList.size();
                    hori = 0;
                    while (!ans.containsKey(horList.get(hori).num)) {
                        ans.put(horList.get(hori).num, horList.get(hori + shift).num);
                        hori++;
                    }
                }
            }
            for (Entry<Integer, ArrayList<Dancer>> kv : verMap.entrySet()) {
                ArrayList<Dancer> verList = kv.getValue();
                ArrayList<Dancer> horList = horMap.getOrDefault(kv.getKey(), null);
                if (horList == null) {
                    for (Dancer d : verList)
                        ans.put(d.num, d.num);
                    continue;
                }

                int veri = verList.size() - 1;
                int hori = 0;
                while (veri >= 0 && hori < horList.size()) {
                    ans.put(verList.get(veri).num, horList.get(hori).num);
                    hori++;
                    veri--;
                }

                if (verList.size() > horList.size()) {
                    int shift = horList.size();
                    veri = 0;
                    while (!ans.containsKey(verList.get(veri).num)) {
                        ans.put(verList.get(veri).num, verList.get(veri + shift).num);
                        veri++;
                    }
                }
            }


            for (int i = 0; i < n; i++) {
                int j = ans.get(i);
                if (g[j] == HOR) {
                    out.println(w + " " + p[j]);
                } else {
                    out.println(p[j] + " " + h);
                }
            }
        }

        class Dancer implements Comparable<Dancer> {
            public int num;
            public int diff;
            public int group;
            public int time;

            public Dancer(int num, int diff, int g, int time) {
                this.num = num;
                this.diff = diff;
                this.group = g;
                this.time = time;
            }


            public int compareTo(Dancer that) {
                if (this.group == that.group) {
                    return this.time - that.time;
                }
                return that.group - this.group;
            }

        }

    }

    static class InputReader {
        StringTokenizer st;
        BufferedReader br;

        public InputReader(InputStream is) {
            BufferedReader br = new BufferedReader(new InputStreamReader(is));
            this.br = br;
        }

        public String next() {
            if (st == null || !st.hasMoreTokens()) {
                String nextLine = null;
                try {
                    nextLine = br.readLine();
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
                if (nextLine == null)
                    return null;
                st = new StringTokenizer(nextLine);
            }
            return st.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
}
