import java.util.*;
import java.io.*;

public class E {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int nTest = scanner.nextInt();
        for(int iTest = 0; iTest < nTest; iTest++) {
            int m = scanner.nextInt();
            int d = scanner.nextInt();
            int w = scanner.nextInt();
            Solution solution = new Solution(m, d, w);
            solution.solve();
            System.out.println(solution.getOutputString());
        }
    }
}

class Solution {
    public Solution(int m, int d, int w) {
        this.m = m;
        this.d = d;
        this.w = w;
    }

    public void solve() {
        int x = w / gcd(w, d - 1);
        int y = Math.min(m, d);
        int div1 = (y + x - 1) / x;
        int quo1 = y % x;
        int div2 = y / x;
        int quo2 = x - y % x;
        ans = (long)quo1 * div1 * (div1 - 1) / 2 + (long)quo2 * div2 * (div2 - 1) / 2;
    }

    public String getOutputString() {
        return String.valueOf(ans);
    }

    private int gcd(int x, int y) {
        while(y > 0) {
            int tmp = x % y;
            x = y;
            y = tmp;
        }
        return x;
    }

    private int m;
    private int d;
    private int w;
    private long ans;
}
