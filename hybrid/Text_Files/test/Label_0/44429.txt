from sys import stdin
read = stdin.readline
from collections import Counter
for _ in range(int(read())):
	n = int(read())
	arr = sorted(list(map(int, read().split())))
	avg = 2 * (sum(arr) / len(arr))
	arr = Counter(arr)
	ans = 0
	for num in arr:
		if num == avg - num:
			ans += arr[num] * (arr[num] - 1) // 2
			arr[num] = 0
		elif avg - num in arr:
			ans += arr[num] * arr[avg - num]
			arr[avg - num] = 0
	print(ans)
