def isValidMatrix(matrix):
	for row in range(len(matrix)):
		for col in range(len(matrix[0])):
			if matrix[row][col] != 0:
				if not (rowOfOnes(matrix, row) or colOfOnes(matrix, col)):
					return False
	return True

def rowOfOnes(matrix, row):
	for value in matrix[row]:
		if int(value) == 0:
			return False
	return True

def colOfOnes(matrix, col):
	for row in range(len(matrix)):
		if matrix[row][col] == 0:
			return False
	return True

def constructEmptyMatrix(rows, cols):
	matrix = []
	for row in range(rows):
		matrix += [[0] * cols]
	return matrix

def constructMatrixA(matrixA, matrixB):
	global rows
	global cols
	for row in range(rows):
		for col in range(cols):
			if rowOfOnes(matrixB, row) and colOfOnes(matrixB, col):
				matrixA[row][col] = 1
	return matrixA

def printMatrixA(matrixA):
	(rows, cols) = (len(matrixA), len(matrixA[0]))
	for row in range(rows):
		rowString = ''
		for col in range(cols):
			rowString += str(matrixA[row][col])
			rowString += ' '
		rowString = rowString[:-1]
		print(rowString)
