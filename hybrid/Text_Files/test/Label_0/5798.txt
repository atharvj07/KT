#include<iostream>
#include<vector>
using namespace std;
vector<vector<int>>V; int H, W, cnt, vx, vy;
void solve(vector<vector<int>> &p, int x, int y, int cr) {
	int dx[4] = { 1,0,-1,0 };
	int dy[4] = { 0,1,0,-1 };
	int cx = x, cy = y;
	if (cr == 0) { if (x == vx || y == vy)cnt++; return; }
	for (int i = 0; i < 4; i++) {
		cx = x; cy = y;
		while (true) {
			if (cx < 0 || cx >= H || cy < 0 || cy >= W)break;
			if (p[cx][cy] == 3) {
				p[cx][cy] = 1;
				solve(p, cx, cy, cr - 1);
				p[cx][cy] = 3;
				break;
			}
			cx += dx[i]; cy += dy[i];
		}
	}
}
int main() {
	while (true) {
		cin >> W >> H; if (H == 0)break; cnt = 0; int vr = 0;
		vector<vector<int>> a(H, vector<int>(W));
		for (int i = 0; i < H; i++) {
			for (int j = 0; j < W; j++) {
				cin >> a[i][j];
				if (a[i][j] == 2) { vx = i; vy = j; }
				if (a[i][j] == 1) { a[i][j] = 3; vr++; }
			}
		}
		solve(a, vx, vy, vr);
		cout << cnt << endl;
	}
	return 0;
}
