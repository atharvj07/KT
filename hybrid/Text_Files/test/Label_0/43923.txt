(n, t) = map(int, input().split())
s = set('qwertyuioplkjhgfdsazxcvbnm')
s1 = input()
s2 = input()
k1 = k2 = 0
for i in range(n):
	if s1[i] == s2[i]:
		k1 += 1
	else:
		k2 += 1
if t < k2 // 2 + k2 % 2:
	print(-1)
else:
	t -= k2 % 2
	k2 //= 2
	while t > k2 and k2 > 0:
		t -= 2
		k2 -= 1
	if k2 == 0 and t > k1:
		print(-1)
	else:
		if k2 == 0:
			k1 = t
		else:
			k1 = 0
		k3 = k2
		for i in range(n):
			if s1[i] == s2[i]:
				ss = s - {s1[i]}
				if k1 > 0:
					print(ss.pop(), end='')
					k1 -= 1
				else:
					print(s1[i], end='')
			else:
				ss = s - {s1[i], s2[i]}
				if k3 > k2:
					print(s1[i], end='')
					k3 -= 1
				elif k2 > 0:
					print(s2[i], end='')
					k2 -= 1
				else:
					print(ss.pop(), end='')
