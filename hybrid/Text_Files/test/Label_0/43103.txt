def gcd(a, b):
	while b != 0:
		t = int(a)
		a = int(b)
		b = t % a
	return int(a)

def lcm(a, b):
	return int(a * b / gcd(a, b))
global ct, lo, l1, x, y
ct = 0
import math

def can(n):
	if n >= lo and n <= l1:
		return True
	return False
(lo, l1, x, y) = map(int, input().split())

def divisors(n):
	global ct
	i = 1
	while i <= math.sqrt(n):
		if n % i == 0:
			if n / i == i:
				if can(x * y / i) and can(i) and (lcm(i, x * y / i) == y) and (gcd(x * y / i, i) == x):
					ct += 1
			else:
				f = n / i
				if can(x * y / i) and can(i) and (lcm(i, x * y / i) == y) and (gcd(x * y / i, i) == x):
					ct += 1
				if can(x * y / f) and can(f) and (lcm(f, x * y / f) == y) and (gcd(x * y / f, f) == x):
					ct += 1
		i = i + 1
divisors(y)
print(ct)
