n = int(input())
pedras = input().split()
pedras_ordenadas = [int(x) for x in pedras]
pedras_ordenadas.sort()
m = int(input())
saida = []
soma_acumulada = []
soma_acumulada_ord = []
soma = 0
soma_ord = 0
for s in pedras:
	soma += int(s)
	soma_acumulada.append(soma)
for s_ord in pedras_ordenadas:
	soma_ord += int(s_ord)
	soma_acumulada_ord.append(soma_ord)
for i in range(m):
	entrada = input().split()
	q = entrada[0]
	l = int(entrada[1])
	r = int(entrada[2])
	if q == '1':
		if l > 1:
			saida.append(soma_acumulada[r - 1] - soma_acumulada[l - 2])
		else:
			saida.append(soma_acumulada[r - 1])
	elif l > 1:
		saida.append(soma_acumulada_ord[r - 1] - soma_acumulada_ord[l - 2])
	else:
		saida.append(soma_acumulada_ord[r - 1])
for k in saida:
	print(k)
