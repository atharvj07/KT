            boolean[][] can = new boolean[1 << n][sum + 1];

            }
            int[][] viaMu = new int[1 << n][sum + 1];

                    int mv = m ^ mu;
                    for (int u = 0; u <= sum; u++) {
                        for (int v = 0; u + v <= sum; v++) {
                            if (can[mu][u] && can[mv][v]) {

            List<String> ans = new ArrayList<>();
            reconstructAnsSlow(ans, viaMu, viaU, viaV, (1 << n) - 1, 1);

        private void reconstructAnsSlow(List<String> ans, int[][] viaMu, int[][] viaU, int[][] viaV, int m, int s) {
            if (Integer.bitCount(m) == 1) {
                return;
            }
            int mu = viaMu[m][s];
            int mv = m ^ mu;
            int u = viaU[m][s];
            int v = viaV[m][s];
            ans.add(u + " " + v);
            reconstructAnsSlow(ans, viaMu, viaU, viaV, mu, u);
            reconstructAnsSlow(ans, viaMu, viaU, viaV, mv, v);
        }

        class Pair {
            int a;
            int b;

            Pair(int a, int b) {
                this.a = a;
                this.b = b;
            }

        }

        class BitSet {
            final int Q = 31;
            int n;
            int[] a;

            BitSet(int n) {
                this.n = n;
                a = new int[(n - 1) / Q + 1];
            }

            BitSet shiftedLeft(int by) {
                BitSet res = new BitSet(n);
                for (int i = 0; i < a.length; i++) {
                    int j = i + by / Q;
//                long full = (long)a[i] << (by % Q);
                    if (j < a.length) {
                        int low = (a[i] << (by % Q)) & ((1 << Q) - 1);
//                    int low1 = (int)(full % (1L << Q));
                        res.a[j] |= low;
                    }
                    if (j + 1 < a.length) {
                        int high = a[i] >>> (Q - (by % Q));
//                    int high1 = (int)(full >>> Q);
                        res.a[j + 1] |= high;
                    }
                }
                return res;
            }

            void setBit(int i) {
                a[i / Q] |= 1 << (i % Q);
            }

            boolean testBit(int i) {
                return ((a[i / Q] >>> (i % Q)) & 1) != 0;
            }

            void or(BitSet o) {
                for (int i = 0; i < a.length; i++) {
                    a[i] |= o.a[i];
                }
            }

        }

    static class FastScanner {
        private BufferedReader in;
        private StringTokenizer st;

        public FastScanner(InputStream stream) {
            in = new BufferedReader(new InputStreamReader(stream));
        }

        public String next() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(in.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return st.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
