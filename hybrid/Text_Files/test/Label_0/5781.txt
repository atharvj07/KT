#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    double x;
    int n, m;
    scanf("%lf%d%d", &x, &n, &m);
    for (int i = 1; i <= n; i++) {
      if (x / 2.0 + 10 < x) {
        x = x / 2.0 + 10;
        x = (int)(x);
      } else
        break;
    }
    if (x <= m * 10)
      printf("YES\n");
    else
      printf("NO\n");
  }
}
