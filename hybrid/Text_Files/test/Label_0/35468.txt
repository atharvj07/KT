import java.util.*;
public class CFC219{
    public static void main(String args[]){
        Scanner ob=new Scanner(System.in);
        //System.out.println("SAKJJ");
        int n=ob.nextInt();
        int m=ob.nextInt();
        int a[]=new int[n];
        for(int i=0;i<n;i++) a[i]=ob.nextInt();
        int sum=0;
        while(m-->0){
            int b=ob.nextInt();
            int c=ob.nextInt();
            if(a[b-1]>a[c-1])
            sum+=a[c-1];
            else sum+=a[b-1];
        }
        System.out.println(sum);
    }
}
