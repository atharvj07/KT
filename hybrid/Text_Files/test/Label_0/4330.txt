#include <bits/stdc++.h>
using namespace std;
long long k, n, m;
long long A[200006];
vector<pair<long long, long long>> ad[200006];
pair<long long, long long> mx[200006];
vector<pair<long double, long long>> mul;
long long typ[200006];
void solve() {
  cin >> k >> n >> m;
  for (long long i = (1), iend = (k); i <= iend; ++i) scanf("%lld", A + i);
  long long op, t, b;
  for (long long i = (1), iend = (n); i <= iend; ++i) {
    scanf("%lld%lld%lld", &op, &t, &b);
    typ[i] = op;
    if (op == 1)
      mx[t] = max(mx[t], make_pair(b, i));
    else if (op == 2)
      ad[t].emplace_back(b, i);
    else if (op == 3)
      mul.emplace_back(b, i);
  }
  for (long long i = (1), iend = (k); i <= iend; ++i)
    if (mx[i].first > A[i])
      ad[i].emplace_back(mx[i].first - A[i], mx[i].second);
  for (long long i = (1), iend = (k); i <= iend; ++i) {
    sort(ad[i].rbegin(), ad[i].rend());
    long long lst = A[i];
    for (auto x : ad[i]) {
      mul.emplace_back(1.0L * (lst + x.first) / lst, x.second), lst += x.first;
    }
  }
  sort(mul.rbegin(), mul.rend());
  long long as = min((long long)mul.size(), m);
  printf("%lld\n", as);
  for (long long i = 0; i < as; ++i)
    if (typ[mul[i].second] == 1) printf("%lld ", mul[i].second);
  for (long long i = 0; i < as; ++i)
    if (typ[mul[i].second] == 2) printf("%lld ", mul[i].second);
  for (long long i = 0; i < as; ++i)
    if (typ[mul[i].second] == 3) printf("%lld ", mul[i].second);
}
signed main() { solve(); }
