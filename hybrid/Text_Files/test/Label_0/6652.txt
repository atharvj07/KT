#include <bits/stdc++.h>
long long f[101][101];
long long m[101][101];
using namespace std;
int main() {
  int n, k, d;
  cin >> n >> k >> d;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      f[i][j] = 0;
      m[i][j] = 0;
    }
  }
  for (int i = 0; i <= n; i++) f[i][0] = 0;
  for (int i = 0; i <= k; i++) f[0][i] = 1;
  f[1][1] = 1;
  for (int i = 2; i <= n; i++) {
    int x = min(i, k);
    for (int j = 1; j <= x; j++) {
      if (i != j) {
        for (int l = 0; l <= k; l++) {
          f[i][j] = (f[i][j] + f[i - j][l]) % 1000000007;
        }
      }
    }
    if (i <= k) f[i][i] = 1;
  }
  for (int i = 0; i <= n; i++) m[i][0] = 0;
  for (int i = 0; i <= d; i++) m[0][i] = 1;
  m[1][1] = 1;
  d--;
  for (int i = 2; i <= n; i++) {
    int x = min(i, d);
    for (int j = 1; j <= x; j++) {
      if (i == j) {
        ;
      } else {
        for (int l = 0; l <= d; l++) {
          m[i][j] = (m[i][j] + m[i - j][l]) % 1000000007;
        }
      }
    }
    if (i <= d) m[i][i] = 1;
  }
  long long ans = 0;
  for (int i = 1; i <= k; i++) {
    ans = (ans + f[n][i]) % 1000000007;
  }
  for (int i = 1; i <= d; i++) {
    ans = (ans - m[n][i]) % 1000000007;
  }
  ans = (ans + 1000000007) % 1000000007;
  cout << ans << endl;
  return 0;
}
