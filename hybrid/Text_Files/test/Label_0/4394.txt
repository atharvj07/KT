#include<iostream>
#include<string>

using namespace std;

int main(){

	int m;
	string map[5];

	cin>>m;

	for(int i=0;i<5;i++) map[i]="*****";

	for(int i=0;i<m;i++){

		for(int j=0;j<5;j++){
			for(int k=0;k<5;k++){
				cin>>map[j][k];
			}
		}

		int ans=0;

		for(int y=0;y<5;y++){
			for(int x=0;x<5;x++){
				for(int sy=1;y+sy-1<5;sy++){
					for(int sx=1;x+sx-1<5;sx++){
						bool f=true;
						for(int j=0;j<sy;j++){
							for(int k=0;k<sx;k++){
								if(map[y+j][x+k]=='0') f=false;
							}
						}
						if(f&&ans<sy*sx) ans=sx*sy;
					}
				}
			}
		}

		cout<<ans<<endl;
	}
}
