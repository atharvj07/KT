class Grade implements Comparable<Grade>
{
	long gr ,es;
	public int compareTo(Grade g)
	{
		if(this.es>g.es)
			return 1;
		else if(this.es<g.es)
			return -1;
		return 0;
	}
	public Grade(long g,long e)
	{this.gr=g;
		this.es=e;
	}

}

class Sorting
{
	private static Comparable[] aux; // auxiliary array for merges
	public static Comparable[] merge(Comparable[] a, int lo, int mid, int hi)
{ // Merge a[lo..mid] with a[mid+1..hi].
int i = lo, j = mid+1;
for (int k = lo; k <= hi; k++) // Copy a[lo..hi] to aux[lo..hi].
aux[k] = a[k];
for (int k = lo; k <= hi; k++) // Merge back to a[lo..hi].
if (i > mid) a[k] = aux[j++];
else if (j > hi ) a[k] = aux[i++];
else if (less(aux[j], aux[i])) a[k] = aux[j++];
else a[k] = aux[i++];
return a;
}


public static Comparable[] mergeSort(Comparable[] a)
{//client will call this method
aux = new Comparable[a.length]; // Allocate space just once.
mergeSort(a, 0, a.length - 1);
return a;
}
private static Comparable[] mergeSort(Comparable[] a, int lo, int hi)
{ // Sort a[lo..hi].
if (hi <= lo) return a;
int mid = lo + (hi - lo)/2;
mergeSort(a, lo, mid); // Sort left half.
mergeSort(a, mid+1, hi); // Sort right half.
merge(a, lo, mid, hi); // Merge results (code on page 271).
return a;
}
private static boolean less(Comparable v, Comparable w)
{ return v.compareTo(w) < 0; }
private static void exch(Comparable[] a, int i, int j)
{ Comparable t = a[i]; a[i] = a[j]; a[j] = t; }

}
