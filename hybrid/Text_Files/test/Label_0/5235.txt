#include <iostream>
#include <set>
#include <string>
#include <vector>
#define llint long long
#define inf 1e18

using namespace std;

llint n;
llint a[5005];
vector<llint> G[5005];
llint sum[5005];
llint dp[5005][5005][2];
llint dp2[5005][5005][2];

llint dfs(llint v, llint p)
{
	sum[v] = 1;
	for(int i = 0; i < G[v].size(); i++){
		if(G[v][i] == p) continue;
		sum[v] += dfs(G[v][i], v);
	}
	
	int m = G[v].size();
	for(int i = 0; i <= m; i++){
		for(int j = 0; j <= sum[v]; j++){
			for(int k = 0; k < 2; k++){
				dp2[i][j][k] = inf;
			}
		}
	}
	if(a[v] > 0) dp2[0][0][0] = a[v];
	else dp2[0][0][1] = a[v];
	
	llint psum = 1;
	for(int i = 0; i < m; i++){
		llint u = G[v][i];
		if(u == p){
			for(int j = 0; j <= psum; j++){
				for(int k = 0; k < 2; k++){
					dp2[i+1][j][k] = dp2[i][j][k];
				}
			}
			continue;
		}
		for(int j = 0; j <= psum; j++){
			for(int k = 0; k < 2; k++){
				for(int l = 0; l <= sum[u]; l++){
					for(int s = 0; s < 2; s++){
						dp2[i+1][j+l][min(1,k+s)] = min(dp2[i+1][j+l][min(1,k+s)], dp2[i][j][k] + dp[u][l][s]);
						if(j+l+1 <= psum+sum[u]){
							if((s == 0 && dp[u][l][s] < inf/2) || (s == 1 && dp[u][l][s] < 0)){
								dp2[i+1][j+l+1][k] = min(dp2[i+1][j+l+1][k], dp2[i][j][k]);
							}
						}
					}
				}
			}
		}
		psum += sum[u];
	}
	for(int j = 0; j <= psum; j++){
		for(int k = 0; k < 2; k++){
			dp[v][j][k] = dp2[m][j][k];
		}
	}
	return sum[v];
}

int main(void)
{
	cin >> n;
	for(int i = 1; i <= n; i++) cin >> a[i];
	llint u, v;
	for(int i = 1; i <= n-1; i++){
		cin >> u >> v;
		G[u].push_back(v);
		G[v].push_back(u);
	}
	dfs(1, -1);
	
	llint ans = inf;
	for(llint i = 0; i <= n-1; i++){
		if(dp[1][i][0] < inf/2) ans = min(ans, i);
		if(dp[1][i][1] < 0) ans = min(ans, i);
	}
	cout << ans << endl;
	
	return 0;
}
