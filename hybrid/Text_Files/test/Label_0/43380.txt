from sys import stdin, stdout
(n, k) = map(int, stdin.readline().split())
first = list(map(int, stdin.readline().split()))
second = sorted(list(map(int, stdin.readline().split())))
for i in range(n):
	if not first[i]:
		first[i] = second.pop()
if first == sorted(first):
	stdout.write('No')
else:
	stdout.write('Yes')
