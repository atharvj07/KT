import java.util.*;

public class A_237 {

    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int lines = sc.nextInt();
        char pos = ' ', neg = ' ';
        boolean error=false;
        for(int i=0;i<lines;++i){
            String line = sc.next();
            if(i==0){
                pos = line.charAt(0);
                neg = line.charAt(1);
                if(pos == neg){
                    error = true;
                    break;
                }
            }
            for(int j=0;j<lines;++j){
                char c = line.charAt(j);
                char cmp;
                if(j == i || j == lines - i - 1){
                    cmp = pos;
                }else{
                    cmp = neg;
                }
                if(cmp != c){
                    error=true;
                    break;
                }
            }
            if(error)break;
        }
        if(error){
            System.out.println("NO");
        }else{
            System.out.println("YES");
        }

    }
}
