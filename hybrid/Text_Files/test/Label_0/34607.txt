import java.util.Scanner;
public class Lcm {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();
        while(T-- > 0) {
            int l = sc.nextInt();
            int r = sc.nextInt();
            if (2*l > r) System.out.println("-1 -1");
            else System.out.println(l + " " + 2*l);
        }
        sc.close();
    }
}
