import sys, math, queue, bisect
MOD = 998244353
sys.setrecursionlimit(1000000)
n = int(input())
a = list(map(int, input().split()))
for i in range(1, n):
	if a[i] == a[i - 1]:
		print('NO')
		exit()
y = None
for i in range(1, n):
	if abs(a[i] - a[i - 1]) == 1:
		continue
	if y is None:
		y = abs(a[i] - a[i - 1])
	elif y != abs(a[i] - a[i - 1]):
		print('NO')
		exit()
if y is None:
	y = max(a)
for i in range(1, n):
	if abs(a[i] - a[i - 1]) != 1:
		continue
	if a[i] % y == 0:
		if a[i - 1] != a[i] - 1 and a[i - 1] != a[i] + y:
			print('NO')
			exit()
	elif a[i] % y == 1:
		if a[i - 1] != a[i] - y and a[i - 1] != a[i] + 1:
			print('NO')
			exit()
x = math.ceil(max(a) / y)
print('YES')
print(x, y)
