#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <cmath>

int main() {
    int t;
    std::cin >> t;

    while (t--) {
        int n, k;
        std::cin >> n >> k;

        std::cout << 0 << std::endl;
        std::cout.flush();

        int r = 0;
        std::cin >> r;
        if (r == 1) {
            continue;
        }
        int i = 1;
        for (int j = 1; j <= n; j++) {
            std::cout << i << std::endl;
            std::cout.flush();
            std::cin >> r;
            if (r != 0) {
                break;
            }
            i = j ^ (j + 1);
        }


    }
    return 0;
}
