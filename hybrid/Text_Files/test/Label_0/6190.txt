#include <bits/stdc++.h>
using namespace std;
const int mod = 998244353;
int l, n, k;
int ans = 0;
namespace usual {
int fac[4005], ifac[4005], pow2[2005];
int add(int ta, int tb) { return ta + tb >= mod ? ta + tb - mod : ta + tb; }
int sub(int ta, int tb) { return ta < tb ? ta - tb + mod : ta - tb; }
int ksm(long long ta, int tp) {
  int s = 1;
  for (; tp; ta = ta * ta % mod, tp >>= 1)
    if (tp & 1) s = ta * s % mod;
  return s;
}
void make_fac() {
  fac[0] = 1;
  for (int i = 1; i <= 2 * n + 1; i++) fac[i] = 1ll * fac[i - 1] * i % mod;
  ifac[2 * n + 1] = ksm(fac[2 * n + 1], mod - 2);
  for (int i = 2 * n + 1; i; i--) ifac[i - 1] = 1ll * ifac[i] * i % mod;
}
void make_pow2() {
  pow2[0] = 1;
  for (int i = 1; i <= n; i++) pow2[i] = (pow2[i - 1] << 1) % mod;
}
int C(int tn, int tm) {
  return tn < tm ? 0 : 1ll * fac[tn] * ifac[tm] % mod * ifac[tn - tm] % mod;
}
}  // namespace usual
using namespace usual;
int betaf(int x, int y) {
  return 1ll * ifac[x + y - 1] * fac[x - 1] % mod * fac[y - 1] % mod;
}
int main() {
  scanf("%d %d %d", &n, &k, &l);
  make_fac();
  make_pow2();
  for (int i = k; i <= n; i++) {
    int ts = 0;
    for (int j = 0; j <= n - i; j++) {
      if (j & 1)
        ts = sub(ts, 1ll * C(n - i, j) * pow2[i + j] % mod *
                         betaf(i + j + 1, i + j + 1) % mod);
      else
        ts = add(ts, 1ll * C(n - i, j) * pow2[i + j] % mod *
                         betaf(i + j + 1, i + j + 1) % mod);
    }
    ans = add(ans, 1ll * ts * C(n, i) % mod);
  }
  printf("%lld\n", 1ll * ans * l % mod);
  return 0;
}
