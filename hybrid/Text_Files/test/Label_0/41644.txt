ch = 0
m = input()
a = int(m[0:len(m) - 1:1])
b = m[len(m) - 1:len(m):1]
if a % 4 == 1 or a % 4 == 3:
	ch = a // 4 * 14 + 'fedabc'.find(b) + 1 + a // 4 * 2
elif a % 4 == 0:
	ch = a // 4 * 14 + 'fedabc'.find(b) - 6 + (a // 4 - 1) * 2
else:
	ch = a // 4 * 14 + 'fedabc'.find(b) + 8 + a // 4 * 2
print(ch)
