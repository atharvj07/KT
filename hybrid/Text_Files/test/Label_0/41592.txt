n = int(input())
a = [int(i) for i in input().split()]
j = -1
mn = [1000000000] * n
for i in range(n):
	if a[i] == 0:
		mn[i] = 0
		continue
	if j == -1 or a[i - 1] == 0:
		j = i
		while j < n and a[j] != 0:
			j += 1
		if j == n:
			break
		mn[i] = j - i
	else:
		mn[i] = j - i
a = a[::-1]
j = -1
for i in range(n):
	if a[i] == 0:
		continue
	if j == -1 or a[i - 1] == 0:
		j = i
		while j < n and a[j] != 0:
			j += 1
		if j == n:
			break
		mn[n - i - 1] = min(mn[n - i - 1], j - i)
	else:
		mn[n - i - 1] = min(mn[n - i - 1], j - i)
print(*mn)
