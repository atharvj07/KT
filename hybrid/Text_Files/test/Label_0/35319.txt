import java.io.*;
import java.util.*;
public class r196d {
    private LinkedList<Integer> adj[];
    
    public r196d(int n)
    {
        adj=new LinkedList[n+1];
        for(int i=0;i<=n;i++)
            adj[i]=new LinkedList<Integer>();
    }
    void setEdge(int u,int v)
    {
        adj[u].add(v);
    }
    void dfs(int src,int dist[],boolean visited[])
    {
        visited[src]=true;
        Iterator<Integer> it=adj[src].iterator();
        while(it.hasNext())
        {
            int x=it.next();
            if(!visited[x])
            {
                dist[x]=dist[src]+1;
                dfs(x,dist,visited);
            }
        }
    }
    public static void main(String args[])
    {
        fastio in=new fastio(System.in);
        PrintWriter pw=new PrintWriter(System.out);

        int n=in.nextInt();
        int m=in.nextInt();
        int d=in.nextInt();
        
        int ar[]=new int[m];
        boolean isbook[]=new boolean[n+1];
        for(int i=0;i<m;i++)
        {
            ar[i]=in.nextInt();
            isbook[ar[i]]=true;
        }
        
        r196d ob=new r196d(n);
        for(int i=0;i<n-1;i++)
        {
            int u=in.nextInt();
            int v=in.nextInt();
            ob.setEdge(u, v);
            ob.setEdge(v, u);
        }
        
        if(m==1)
        {
            boolean visited[]=new boolean[n+1];
            int dist[]=new int[n+1];
            ob.dfs(ar[0], dist, visited);
            
            int ans=0;
            for(int i=1;i<=n;i++)
                if(dist[i]<=d)
                    ans++;
            
            pw.println(ans);
        }
        else
        {
            boolean visited[]=new boolean[n+1];
            int dist[]=new int[n+1];
            ob.dfs(ar[0], dist, visited);
            
            int farthest=-1,pos=-1;
            for(int i=1;i<=n;i++)
                if(dist[i]>pos&&isbook[i])
                {
                    pos=dist[i];
                    farthest=i;
                }
            
            boolean visited1[]=new boolean[n+1];
            int dist1[]=new int[n+1];
            ob.dfs(farthest, dist1, visited1);
            
            int farthest1=-1,pos1=-1;
            for(int i=1;i<=n;i++)
                if(dist1[i]>pos1&&isbook[i])
                {
                    pos1=dist1[i];
                    farthest1=i;
                }
            
            boolean visited2[]=new boolean[n+1];
            int dist2[]=new int[n+1];
            ob.dfs(farthest1, dist2, visited2);
            
            int ans=0;
            for(int i=1;i<=n;i++)
                if(dist1[i]<=d&&dist2[i]<=d)
                    ans++;
            
            pw.println(ans);
        }
        
        pw.close();
    }

    static class fastio {
     
            private final InputStream stream;
            private final byte[] buf = new byte[8192];
            private int cchar, snchar;
            private SpaceCharFilter filter;
     
            public fastio(InputStream stream) {
                this.stream = stream;
            }
     
            public int nxt() {
                if (snchar == -1)
                    throw new InputMismatchException();
                if (cchar >= snchar) {
                    cchar = 0;
                    try {
                        snchar = stream.read(buf);
                    } catch (IOException e) {
                        throw new InputMismatchException();
                    }
                    if (snchar <= 0)
                        return -1;
                }
                return buf[cchar++];
            }
     
            public int nextInt() {
                int c = nxt();
                while (isSpaceChar(c)) {
                    c = nxt();
                }
                int sgn = 1;
                if (c == '-') {
                    sgn = -1;
                    c = nxt();
                }
                int res = 0;
                do {
                    if (c < '0' || c > '9')
                        throw new InputMismatchException();
                    res *= 10;
                    res += c - '0';
                    c = nxt();
                } while (!isSpaceChar(c));
                return res * sgn;
            }
     
            public long nextLong() {
                int c = nxt();
                while (isSpaceChar(c)) {
                    c = nxt();
                }
                int sgn = 1;
                if (c == '-') {
                    sgn = -1;
                    c = nxt();
                }
                long res = 0;
                do {
                    if (c < '0' || c > '9')
                        throw new InputMismatchException();
                    res *= 10;
                    res += c - '0';
                    c = nxt();
                } while (!isSpaceChar(c));
                return res * sgn;
            }
     
            public int[] nextIntArray(int n) {
                int a[] = new int[n];
                for (int i = 0; i < n; i++) {
                    a[i] = nextInt();
                }
                return a;
            }
     
            public String readString() {
                int c = nxt();
                while (isSpaceChar(c)) {
                    c = nxt();
                }
                StringBuilder res = new StringBuilder();
                do {
                    res.appendCodePoint(c);
                    c = nxt();
                } while (!isSpaceChar(c));
                return res.toString();
            }
     
            public String nextLine() {
                int c = nxt();
                while (isSpaceChar(c))
                    c = nxt();
                StringBuilder res = new StringBuilder();
                do {
                    res.appendCodePoint(c);
                    c = nxt();
                } while (!isEndOfLine(c));
                return res.toString();
            }
     
            public boolean isSpaceChar(int c) {
                if (filter != null)
                    return filter.isSpaceChar(c);
                return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
            }
     
            private boolean isEndOfLine(int c) {
                return c == '\n' || c == '\r' || c == -1;
            }
     
            public interface SpaceCharFilter {
                public boolean isSpaceChar(int ch);
            }
        }
    

}
