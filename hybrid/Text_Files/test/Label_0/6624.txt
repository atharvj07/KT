#include <bits/stdc++.h>
char s[200001];
char pick(char a, char b) {
  char c;
  for (c = 'a'; c == a || c == b; c++)
    ;
  return c;
}
char *handle(char *s) {
  char *ps = s;
  for (s += 1; *s; s++) {
    if (*s == *(s - 1)) *s = pick(*(s - 1), *(s + 1));
  }
  return ps;
}
int main() {
  scanf("%s", s);
  printf("%s\n", handle(s));
  return 0;
}
