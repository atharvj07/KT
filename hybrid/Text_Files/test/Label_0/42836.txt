T = int(input())
for _ in range(T):
	n = int(input())
	A = []
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	for i in range(n):
		A.append([a[i], b[i]])
	f1 = min(a)
	f2 = min(b)
	ans = 0
	for i in range(n):
		d1 = A[i][0] - f1
		d2 = A[i][1] - f2
		diff = min(d1, d2)
		ans += diff
		A[i][0] -= diff
		A[i][1] -= diff
		ans += A[i][0] - f1 + A[i][1] - f2
	print(ans)
