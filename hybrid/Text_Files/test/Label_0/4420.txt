#include <iostream>
#include <list>
#include <vector>
#include <string>
#include <algorithm>
#include <climits>
using namespace std;

using ll = long long int;

long long int N, M;

int main() {

	cin >> N >> M;
	vector<int>ans;
	while (N) {
		vector<int>hu(N);
		for (int i = 0; i < N; i++) {
			cin >> hu[i];
		}
		vector<int>de(M);
		vector<vector<pair<int, int>>>dp(1 << N, vector<pair<int, int>>(M));
		vector<int>sum(1 << N, INT_MAX);
		sum[0] = 0;
		for (int i = 0; i < M; i++) {
			cin >> de[i];
			dp[0][i].second = 1;
			sum[0] += de[i];
		}
		for (int i = 0; i < 1 << N; i++) {
			for (int j = 0; j < N; j++) {
				if ((i >> j)&1)continue;
				int box = 0;
				vector<pair<int, int>>v(M);
				for (int k = 0; k < M; k++) {
					if (!dp[i][k].second) {
						box += abs(de[k] - dp[i][k].first);
						v[k].first = dp[i][k].first;
					}
					else {
						if (abs(de[k] - dp[i][k].first)>abs(de[k] - (dp[i][k].first + hu[j]))) {
							v[k].first = dp[i][k].first + hu[j];
							v[k].second = 1;
							box += abs(de[k] - v[k].first);
						}
						else {
							v[k].first = dp[i][k].first;
							box += abs(de[k] - v[k].first);
						}
					}
				}
				if (sum[i + (1 << j)] > box) {
					sum[i + (1 << j)] = box;
					dp[i + (1 << j)] = v;
				}
			}
		}
		ans.push_back(sum[(1 << N) - 1]);
		cin >> N >> M;
	}
	for (auto i : ans) {
		cout << i << endl;
	}
	return 0;
}
