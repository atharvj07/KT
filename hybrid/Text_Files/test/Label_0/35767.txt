import java.io.*;
public class Main{
	static int m;
    public static void main(String[] args) throws IOException{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] dp = new int[n+1];
        if(n>5)
        	dp[6]=1;
        for(int i=7 ; i<=n ; i++) {
        	dp[i] = dp[i-1];
        	m = log(i);
        	int[] c = binary(i);
        	for(int j=1 ; j<pow(m-1) ; j++) {
        		int num = decimal(XOR(binary(j), c));
        		if(num<i && num+j != i)
        			dp[i]++;
        	}
        }
        System.out.print(dp[n]);
    }
    static int pow(int n) {
    	int mul=1;
    	for(int i=0 ; i<n ; i++)
    		mul *= 2;
    	return mul;
    }
    static int log(int n) {
    	int count=0;
    	while(n != 0) {
    		n /= 2;
    		count++;
    	}
    	return count;
    }
    static int[] binary(int n) {
    	int[] arr = new int[m];
    	for(int j=0, k=n ; j<m ; j++, k/=2)
    		arr[j] = k%2;
    	return arr;
    }
    static int decimal(int[] b) {
    	int sum=0, mul=1;
    	for(int i=0 ; i<m ; i++, mul*=2)
    		sum += mul*b[i];
    	return sum;
    }
    static int[] XOR(int[] a, int[] c) {
    	int[] arr = new int[m];
    	for(int i=0 ; i<m ; i++)
    		arr[i] = (a[i]+c[i])%2;
    	return arr;
    }
}
