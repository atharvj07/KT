from math import ceil
from collections import defaultdict
from sys import stdin
input = stdin.readline
for _ in range(int(input())):
	n = int(input())
	d = defaultdict(list)
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	for i in range(n):
		d[a[i]].append(b[i])
	for i in d:
		d[i].sort(reverse=True)
	for i in d:
		for j in range(1, len(d[i])):
			d[i][j] += d[i][j - 1]
	ans = [0] * n
	for j in range(n):
		div = j + 1
		l = []
		for i in d:
			an = len(d[i])
			if div > an:
				l.append(i)
				continue
			ans[j] += d[i][an - an % div - 1]
		for i in l:
			del d[i]
	print(*ans)
