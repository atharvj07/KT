import java.util.*;
import java.util.function.Function;
import java.util.stream.Collectors;

public class ComparingStrings {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String line = scanner.nextLine();
        String line2 = scanner.nextLine();

        if (line.length() != line2.length()) {
            System.out.println("NO");
            return;
        }
        List<Integer> differentIndex = new ArrayList<>();
        for (int i = 0; i < line.length(); i++) {
            if (line.charAt(i) != line2.charAt(i)) {
                differentIndex.add(i);
            }
        }

        if (differentIndex.size() == 0) {
            Map<String, Long> collect = line.chars().map(e -> (char) e).mapToObj(e -> String.valueOf(e)).collect(Collectors.groupingBy(Function.identity(), Collectors.counting()));
            Optional<Map.Entry<String, Long>> any = collect.entrySet().stream().filter(e -> e.getValue() > 1).findAny();
            System.out.println(any.isPresent() ? "YES" : "NO");
        } else if (differentIndex.size() == 2) {
            int ind1 = differentIndex.get(0);
            int ind2 = differentIndex.get(1);
            if (line.charAt(ind1) == line2.charAt(ind2) && line.charAt(ind2) == line2.charAt(ind1)) {
                System.out.println("YES");
            }else{
                System.out.println("NO");
            }
        }else{
            System.out.println("NO");
        }
    }
}
