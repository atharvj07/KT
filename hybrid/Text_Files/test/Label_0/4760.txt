#include<iostream>

#include<fstream>
#include<algorithm>
#include<vector>
#include<map>
#include<set>
#include<queue>
#include<cmath>
#include<cstring>
#include<cstdlib>
using namespace std;
typedef long long LL;
typedef double DB;
const int N = 22;
const int M = 1<<N;
const int mod = 1e9+7;
int n,m,k,b[N],c[N];
int v[M][N],f[M],dp[M];
int main()
{
	int i,j,x,y,z;
	cin>>n>>k>>m;
	n--;
	while(m--){
		cin>>x>>y>>z;
		x--,y--;
		b[x]|=(1<<y);
		if(z)
			c[x]|=(1<<y);
	}
	m=1<<n;
	for(j=0;j<m;j=j+1){
		z=0;
		for(i=0;i<n;i=i+1)
			v[j][i]=-1;
		for(i=1;i<n;i=i+1){
			x=1<<i,y=1<<(i-1);
			if(j&x){
				if(!(j&y))
					v[j][z]=j^x^y;
				z++;
			}
		}
		f[j]=m;
		for(i=0;i<n;i=i+1)
			if(!(j&(1<<i)))
				f[j]=j^(1<<i);
	}
	dp[0]=1;
	for(z=0;z<k;z=z+1){
		for(j=0;j<m;j=j+1)
			dp[f[j]]=(dp[f[j]]+dp[j])%mod;
		for(i=0;i<n;i=i+1)
			for(j=m-1;j>=0;j=j-1)
				if(v[j][i]>=0)
					dp[v[j][i]]=(dp[v[j][i]]+dp[j])%mod;
		for(j=0;j<m;j=j+1)
			if((j&b[z])!=c[z])
				dp[j]=0;
	}
	z=0;
	for(j=0;j<m;j=j+1)
		z=(z+dp[j])%mod;
	cout<<z<<endl;
	return 0;
}
