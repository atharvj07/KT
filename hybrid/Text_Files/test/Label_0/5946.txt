#include <bits/stdc++.h>
using namespace std;
int main() {
  int k, n, b, v, nsec, nbox = 0;
  cin >> k >> n >> b >> v;
  while (n > 0) {
    if (b + 1 >= k) {
      n = n - k * v;
      b = b - k + 1;
      nbox++;
    } else {
      n = n - (b + 1) * v;
      if (n < 0) {
        nbox++;
        break;
      } else {
        nbox++;
        if (n % v == 0)
          nbox = nbox + (n / v);
        else
          nbox = nbox + (n / v) + 1;
        break;
      }
    }
  }
  cout << nbox << endl;
  return 0;
}
