vi dp,visit;

void dfs(int a){
	if(visit[a])return;
	if(tG[a].size()==0){
		dp[a]=true;
		return;
	}
	visit[a]=true;
	rep(i,tG[a].size())dfs(tG[a][i]);
	return;
}
