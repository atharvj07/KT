template <class T>
inline bool chmax(T &a, T b)
{
    if (a < b)
    {
        a = b;
        return 1;
    }
    return 0;
}
template <class T>
inline bool chmin(T &a, T b)
{
    if (a > b)
    {
        a = b;
        return 1;
    }
    return 0;
}
typedef long long int ll;

#define ALL(v) (v).begin(), (v).end()
#define RALL(v) (v).rbegin(), (v).rend()
#define endl "\n"
const double EPS = 1e-7;
const int INF = 1 << 30;
const ll LLINF = 1LL << 60;
const double PI = acos(-1);
const int MOD = 1000000007;

//-------------------------------------

using P = pair<double, double>; //x, y

P makeVec(P prev, P next)
{
    double x = next.first - prev.first;
    double y = next.second - prev.second;
    return P(x, y);
}

double calc(P prev, P next)
{
    double x1 = prev.first;
    double y1 = prev.second;
    double x2 = next.first;
    double y2 = next.second;
    return fabs((atan2(x1 * y2 - x2 * y1, x1 * x2 + y1 * y2) / PI) * 180.0);
}

int main()
{
    cin.tie(0);
    ios::sync_with_stdio(false);
    int n;
    cin >> n;
    vector<P> point(n);
    for (int i = 0; i < n; i++)
    {
        cin >> point[i].first >> point[i].second;
    }
    double ans = 1000000;
    sort(ALL(point));
    do
    {
        // cout << "dbg:";
        // for(int i=0;i<n;i++){
        //     if(i == n - 1){

        //     }else

        //         cout << "(" << point[i].first << ", " << point[i].second << ")" << " ";
        //     }
        // }
        double tmp = 0;
        vector<P> vec;
        vec.push_back(P(1, 0));
        vec.push_back(makeVec(P(0, 0), point[0]));
        for (int i = 0; i < n; i++)
        {
            if (i == n - 1)
            {
                vec.push_back(makeVec(point[i], P(0, 0)));
            }
            else
            {
                vec.push_back(makeVec(point[i], point[i + 1]));
            }
        }
        for (int i = 0; i < vec.size() - 1; i++)
        {
            tmp += calc(vec[i], vec[i + 1]);
        }
        chmin(ans, tmp);
    } while (next_permutation(ALL(point)));
    cout << fixed;
    cout << setprecision(15) << ans << endl;
}
