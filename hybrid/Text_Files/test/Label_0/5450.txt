#include <bits/stdc++.h>
using namespace std;
const int N = 55;
const int D = 10100;
char dp[N * D];
int n, d;
int A[N];
void do_package() {
  memset(dp, 0, sizeof(dp));
  dp[0] = 1;
  for (int i = 0; i < n; i++) {
    for (int j = N * D - 1; j >= 0; j--) {
      if (dp[j] && j + A[i] < N * D) {
        dp[j + A[i]] = 1;
      }
    }
  }
}
void solve(int &val, int &day) {
  val = day = 0;
  while (1) {
    bool flag = false;
    for (int i = val + d; i > val; i--) {
      if (dp[i]) {
        val = i;
        day++;
        flag = true;
        break;
      }
    }
    if (!flag) break;
  }
}
int main() {
  while (cin >> n >> d) {
    for (int i = 0; i < n; i++) {
      cin >> A[i];
    }
    do_package();
    int a, b;
    solve(a, b);
    cout << a << ' ' << b << endl;
  }
  return 0;
}
