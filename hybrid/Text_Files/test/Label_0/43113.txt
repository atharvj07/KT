def mices_saved(n, k, mices):
	mices.sort()
	cat = 0
	pos = len(mices) - 1
	result = 0
	while pos > -1 and cat < mices[pos]:
		cat += n - mices[pos]
		pos -= 1
		result += 1
	return result
test_cases = int(input())
for i in range(test_cases):
	(n, k) = list(map(int, input().split()))
	mices = list(map(int, input().split()))
	print(mices_saved(n, k, mices))
