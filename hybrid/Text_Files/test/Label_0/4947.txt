#include <bits/stdc++.h>
using namespace std;
const int INF = 0x3c3c3c3c;
long long f(long long n) {
  long long ans = 0;
  for (long long i = 2; i * i * i <= n; i++) {
    ans += n / (i * i * i);
  }
  return ans;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  long long m;
  cin >> m;
  long long l = 1;
  long long r = 5e16;
  while (r - l > 1) {
    long long n = (r + l) / 2;
    if (f(n) < m) {
      l = n;
    } else {
      r = n;
    }
  }
  if (f(r) != m) {
    r = -1;
  }
  cout << r << endl;
  return 0;
}
