def f1(n):
	if n // 2 % 2:
		print('NO')
		return False
	arr = []
	for i in range(n // 2):
		arr.append(i * 2 + 2)
	for i in range(n // 2):
		arr.append(i * 2 + 1)
	print('YES')
	arr[-1] += n // 2
	for m in arr:
		print(m, end=' ')
	return True
T = int(input())
for i in range(T):
	n = int(input())
	f1(n)
