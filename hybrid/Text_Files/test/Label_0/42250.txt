import sys
input = sys.stdin.readline
mod = 998244353
n = int(input())
p = list(map(int, input().split()))
a = pow(100, n, mod)
b = pow(100, mod - 2, mod)
x = a
y = 1
for i in range(n - 1):
	a = a * b % mod
	y = y * p[i] % mod
	x = (x + a * y) % mod
y = y * p[n - 1] % mod
y_inv = pow(y, mod - 2, mod)
print(x * y_inv % mod)
