mod = 10 ** 9 + 7

def func(a, b):
	result = 1
	for i in range(0, b):
		result = result * a % mod
	return result
t = int(input())
while t:
	(n, k) = map(int, input().split())
	k = bin(k)
	k = str(k)
	res = 0
	k = k[2:]
	g = len(k)
	for i in range(g - 1, -1, -1):
		if k[i] == '0':
			continue
		else:
			res = (res + func(n, g - i - 1)) % mod
	print(res)
	t -= 1
