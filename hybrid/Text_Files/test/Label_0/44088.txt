import io, os
input = io.StringIO(os.read(0, os.fstat(0).st_size).decode()).readline
ii = lambda : int(input())
mi = lambda : map(int, input().split())
li = lambda : list(mi())
MOD = 998244353
(n, k) = mi()
a = li()
grps = []
ok = 1
for i in range(1, n - 1):
	if a[i - 1] == a[i + 1] and a[i - 1] != -1:
		ok = 0
		break
lt = sz = None
for i in range(0, n, 2):
	if a[i] == -1:
		if lt is None:
			if i - 2 < 0:
				lt = -1
			else:
				lt = a[i - 2]
			sz = 0
		sz += 1
		if i + 2 >= n:
			grps.append((lt, -1, sz))
			lt = None
		elif a[i + 2] != -1:
			grps.append((lt, a[i + 2], sz))
			lt = None
lt = sz = None
for i in range(1, n, 2):
	if a[i] == -1:
		if lt is None:
			if i - 2 < 0:
				lt = -1
			else:
				lt = a[i - 2]
			sz = 0
		sz += 1
		if i + 2 >= n:
			grps.append((lt, -1, sz))
			lt = None
		elif a[i + 2] != -1:
			grps.append((lt, a[i + 2], sz))
			lt = None
ans = ok
for (lt, rt, sz) in grps:
	(lt, rt) = (min(lt, rt), max(lt, rt))
	if sz == 1:
		if lt == -1:
			if rt == -1:
				cur = k
			else:
				cur = k - 1
		elif lt == rt:
			cur = k - 1
		else:
			cur = k - 2
	else:
		if lt == -1:
			(eq, neq) = (1, 1)
		else:
			(eq, neq) = (0, 1)
		for _ in range(sz - 1):
			eq2 = neq * (k - 1)
			neq2 = neq * (k - 2) + eq
			(eq, neq) = (eq2 % MOD, neq2 % MOD)
		if rt == -1:
			cur = neq * (k - 1) + eq
		elif lt == rt:
			cur = neq * (k - 1)
		else:
			cur = neq * (k - 2) + eq
	ans = ans * cur % MOD
print(ans)
