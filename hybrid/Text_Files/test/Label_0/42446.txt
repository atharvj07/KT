def car():
	for i in range(int(input())):
		(n, k) = map(int, input().split())
		s = input().strip()
		c1 = s.count('1')
		ans = n
		for i in range(k):
			su = 0
			for j in range(i, n, k):
				if s[j] == '1':
					su += 1
				else:
					su -= 1
					su = max(0, su)
				ans = min(ans, c1 - su)
		print(ans)
car()
