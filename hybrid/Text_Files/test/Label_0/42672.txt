t = int(input())

def prime(n, i):
	j = int(n / i)
	if i < j and j <= n:
		return True
	else:
		return False
res = []
while t:
	t -= 1
	n = int(input())
	for i in range(2, n):
		if prime(n, i):
			print(i, int(n / i))
			break
		else:
			x = n % i
			j = i + 1
			while j <= n:
				if n % j == x:
					print(i, j)
					break
				else:
					j += 1
