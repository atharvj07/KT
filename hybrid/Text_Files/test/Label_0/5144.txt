#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 7, M = 1e7, OO = 0x3f3f3f3f;
const long long int MOD = 2e9 + 7;
int main() {
  int n, s[101], r[101], h[101], c[101];
  int Min = N, cnt = 0;
  scanf("%d", &n);
  for (int i = 0; i < int(n); ++i) cin >> s[i] >> r[i] >> h[i] >> c[i];
  for (int i = 0; i < int(n); ++i) {
    for (int j = 0; j < int(n); ++j) {
      if (s[i] < s[j] && r[i] < r[j] && h[i] < h[j]) c[i] = Min;
    }
  }
  for (int i = 0; i < n; i++)
    if (Min > c[i]) {
      Min = c[i];
      cnt = i;
    }
  cout << cnt + 1;
  return 0;
}
