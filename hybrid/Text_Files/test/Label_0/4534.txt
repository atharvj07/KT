#include <bits/stdc++.h>
using namespace std;

int main() {
	int t;
	cin>>t;
	while(t--)
	{
	    string s;
	    cin>>s;
	    int X , Y , x , y , i , j , n = s.size();
	    
	    for(i=0;i<n;i++)
	    {
	        x = y = 0;
	        for(j=0;j<=i;j++)
	        {
	            if(s[j] == 'R') x++;
	            if(s[j] == 'L') x--;
	            if(s[j] == 'U') y++;
	            if(s[j] == 'D') y--;
	        }
	        X = x;
	        Y = y;
	        
	        x = y = 0;
	        for(j=0;j<n;j++)
	        {
	            if(s[j] == 'R')
	            {
	                x++;
	                if(X == x && Y == y)    x--;
	            }
	            if(s[j] == 'L')
	            {
	                x--;
	                if(X == x && Y == y)    x++;
	            }
	            if(s[j] == 'U')
	            {
	                y++;
	                if(X == x && Y == y)    y--;
	            }
	            if(s[j] == 'D')
	            {
	                y--;
	                if(X == x && Y == y)    y++;
	            }
	        }
	        
	        if(x==0 && y==0)    
	            break;
	    }
	    
	    if(i==n)    cout<<"0 0"<<endl;
	    else        cout<<X<<" "<<Y<<endl;
	}
	return 0;
}
