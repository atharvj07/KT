import java.util.Scanner;
import java.util.Arrays;
public class Main {
    public static void main(String args[]) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int arr[] = new int[n];
        boolean visited[] = new boolean[n];
        for(int i = 0; i < n; i++){
            arr[i] = s.nextInt();
        }
        Arrays.sort(arr);
        int count = 0,i = 0;
        while(i < n){
            if(!visited[i]){
                visited[i] = true;
                int start = i + 1,prev = arr[i];
                while(start < n){
                    if(arr[start] > prev && !visited[start]){
                        count++;
                        prev = arr[start];
                        visited[start] = true;
                    }
                    start++;
                }
            }
            i++;
        }
        System.out.println(count);
    }
}
