def geto(a, b):
	return max(0, min(a[1], b[1]) - max(a[0], b[0]) + 1)
(n, m, s, f) = map(int, input().split())
di = {}
for _ in range(m):
	(t, l, r) = map(int, input().split())
	di[t] = [l, r]
t = 1
ans = []
while s != f:
	if f > s:
		inte = [s, s + 1]
		if t in di and geto(inte, di[t]):
			ans += ['X']
		else:
			ans += ['R']
			s += 1
	else:
		inte = [s - 1, s]
		if t in di and geto(inte, di[t]):
			ans += ['X']
		else:
			ans += ['L']
			s -= 1
	t += 1
print(''.join(ans))
