import sys
input = sys.stdin.readline
output = sys.stdout.write

def main():
	tests = int(input().rstrip())
	for i in range(tests):
		length_ = int(input().rstrip())
		nums_ = list(map(int, input().rstrip().split()))
		summ = sum(nums_)
		if summ % length_ == 0:
			res = int(summ / length_)
			ans = 0
			for _ in nums_:
				ans += _ > res
			output(str(ans))
		else:
			output('-1')
		output('\n')
main()
