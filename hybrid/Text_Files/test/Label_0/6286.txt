#include <iostream>
#include <vector>
#include <stack>
#include <queue>
#include <map>
#include <bitset>
#include <algorithm>
#include <cstdio>
#include <cstdlib>
#include <cmath>

using namespace std;

int main(){
	int n = 0;
	cin >> n;
	while(n--){
		int m = 0;
		cin >> m;
		vector<int> data(m, 0);
		int max = 0, min = 0;
		for(int i = 0; i < m; i++){
			cin >> data[i];
			if(i){
				if(min > data[i]-data[i-1]) min = data[i]-data[i-1];
				if(max < data[i]-data[i-1]) max = data[i]-data[i-1];
			}
		}

		if(min < 0) min *= -1;

		cout << max << " " << min << endl;
	}

	return 0;
}
