import sys, math

def input():
	return sys.stdin.readline()[:-1]

def gcd(a, b):
	if b == 0:
		return a
	return gcd(b, a % b)

def lcm(a, b):
	return a * b // gcd(a, b)

def main():
	n = int(input())
	for _ in range(n):
		(r, b, k) = map(int, input().split())
		M = lcm(r, b)
		R = (M - 1) // r
		B = (M - 1) // b
		if max(math.ceil(R / (B + 1)), math.ceil(B / (R + 1))) >= k:
			print('REBEL')
		else:
			print('OBEY')
main()
