#include <bits/stdc++.h>
using namespace std;
vector<int> x, y;
int main() {
  long long n, j, ax, ay;
  cin >> n >> j >> ax >> ay;
  for (int i = 0; i < n; i++) {
    int bx, by;
    cin >> bx >> by;
    x.push_back(bx);
    y.push_back(by);
  }
  j %= 2 * n;
  for (int i = 0; i < j; i++) {
    ax = (x[i % n] - ax) * 2 + ax;
    ay = (y[i % n] - ay) * 2 + ay;
  }
  cout << ax << ' ' << ay;
}
