#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll ans=0;
const ll mod=1e9+7;
ll ksm(ll a,ll n)
{
    a%=mod;
    ll ans=1,base=a;
    while(n)
    {
        if(n&1)
            ans=(ans*base)%mod;
        base=(base*base)%mod;
        n>>=1;
    }
    return ans%mod;
}
ll fa(int x)
{
    ll sum=1;
    for(int i=2;i<=x;i++)
        sum=(sum*i)%mod;
    return sum;
}
ll C(ll a,ll b)
{
    return fa(b)*ksm(fa(a)*fa(b-a),mod-2)%mod;
}
int main()
{
    int x,y;
    scanf("%d%d",&x,&y);
    int a=(2*y-x)/3;
    int b=(2*x-y)/3;
    if((x+y)%3==0&&x*2>=y&&y*2>=x)
        printf("%lld\n",C(a,a+b));
    else printf("0\n");
    return 0;
}
