#include <bits/stdc++.h>
using namespace std;
inline void readINFile();
void solve() {
  long long x, y, k;
  cin >> x >> y >> k;
  long long s = y * k + k - 1;
  cout << s / (x - 1) + k + (s % (x - 1) ? 1 : 0) << '\n';
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  readINFile();
  long long tc = 1;
  cin >> tc;
  while (tc--) solve();
}
