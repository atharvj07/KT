void init() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cout.precision(16);
}
void solve() {
  int n, k;
  string s;
  cin >> n >> k >> s;
  if (k > 0 && n == 1) {
    cout << "0\n";
    return;
  }
  for (int i = 0; i < n; ++i) {
    if (k == 0) break;
    if (i == 0 && s[i] != '1') {
      s[i] = '1';
      k--;
    }
    if (i > 0 && s[i] != '0') {
      s[i] = '0';
      k--;
    }
  }
  cout << s << '\n';
}
bool multitest = false;
int main() {
  init();
  int t = 1;
  if (multitest) cin >> t;
  for (int i = 0; i < t; ++i) solve();
  return 0;
}
