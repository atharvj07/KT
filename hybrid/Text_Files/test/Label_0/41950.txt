def main():
	string = str(input())
	arr = [int(x) for x in string]
	mx = max(arr)
	print(mx)
	for _ in range(mx):
		res = ''
		for i in range(len(arr)):
			if arr[i] > 0:
				res += '1'
				arr[i] -= 1
			else:
				res += '0'
		for i in range(len(res)):
			if res[i] == '1':
				break
		print(res[i:], end=' ')
main()
