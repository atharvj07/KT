public class D {
	public static void gen() {
		int n = 100, r = 62;
		System.out.println(n + " " + r);
		int[] ls = new int[n], ts = new int[n];
		for(int i=0;i<n;i++) {
			ts[i] = (int)(Math.random()*200);
			ls[i] = (int)(Math.random()*ts[i]); // T >= L, L <= T
		}
		for(int i=0;i<n;i++)
			System.out.print(ls[i] + " ");
		System.out.println();
		for(int i=0;i<100;i++)
			System.out.print(ts[i] + " ");
		System.out.println();
	}
	public static void main(String[] args) throws IOException {
//		gen();
		FastScanner in = new FastScanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		int n = in.nextInt();
		long r = in.nextLong();
		long[] ls = new long[n], ts = new long[n];
		for (int i = 0; i < n; i++) {
			ls[i] = in.nextLong();
		}
		for (int i = 0; i < n; i++) {
			ts[i] = in.nextLong();
		}

		long k = 0;
		long leftoverBoost = 0;
		long time = 0;
		ArrayList<Long> ans = new ArrayList<>();
		for (int i = 0; i < n; i++) {
			// System.out.println(i + ": t=" + time);
			long T = ts[i], L = ls[i];
			if (T < L) {
				k = -1;
				break;
			}

			// long boostsNeeded = ls[i]-leftoverBoost)
			long b = Math.min(L, leftoverBoost);
			T -= b;
			L -= b;
			time += b;
			leftoverBoost -= b;
			if (leftoverBoost == 0) {
				long boostTimeNeeded = 2 * L - T; // 2*L-T <= B
//				System.out.println(i + "th bridge has length " + L + ", max time " + T);
				if (boostTimeNeeded < 0)
					boostTimeNeeded = 0;
				T -= boostTimeNeeded;
				L -= boostTimeNeeded;
				time += 2 * L;
				// System.out.println("using " + ((boostTimeNeeded+r-1)/r) + "
				// boosts at t=" + time);
				long count = (boostTimeNeeded + r - 1) / r;
				if (k + count <= 100000) {
					long left = boostTimeNeeded;
//					System.out.println("boosting for " + left + " seconds, walking for " + walkTime + " seconds gives " + (left+2*walkTime));
					for (int iter = 0; iter < count; iter++) {
						long boost = Math.min(left, r);
						ans.add(time);
						left -= boost;
						// System.out.println("add " + boost);
						time += boost;
						k++;
					}
				} else {
					time += boostTimeNeeded;
					k += count;
				}
				long last = boostTimeNeeded % r;
				leftoverBoost = (r - last) % r;
			}
		}
		out.println(k);
		if (k <= 100000 && k >= 0) {
			for (int i = 0; i < ans.size(); i++) {
				if (i != 0)
					out.print(' ');
				out.print(ans.get(i));
			}
			out.println();
		}
		out.close();

	}

	static class FastScanner {
		BufferedReader br;
		StringTokenizer st;

		public FastScanner(InputStream in) {
			br = new BufferedReader(new InputStreamReader(in));
			st = new StringTokenizer("");
		}

		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		public long nextLong() throws IOException {
			return Long.parseLong(next());
		}

		public String next() throws IOException {
			while (!st.hasMoreTokens()) {
				st = new StringTokenizer(br.readLine());
			}
			return st.nextToken();
		}
	}
}
