import java.util.*;
import java.io.*;

public class F1430 {

	static int n, k, l[], r[], a[];
	static long ps[], dp[][];

	public static long getPS(int l, int r) {
		return (r == -1 ? 0 : ps[r]) - (l == 0 ? 0 : ps[l - 1]);
	}

	public static long dp(int idx, int lastReload) {
		if (dp[idx][lastReload] != -1) {
			return dp[idx][lastReload];
		}
		int len = r[idx] - l[idx];
		long bulletsNow = lastReload == idx ? k : (k - getPS(lastReload, idx - 1) % k) % k;
		long ans = (long) 1e18;
		if (idx == n - 1) {
			if (bulletsNow + 1l * len * k >= a[idx])
				return dp[idx][lastReload] = 0;
			else
				return dp[idx][lastReload] = (long) 1e18;
		}
		if (r[idx] == l[idx + 1]) {
			if (bulletsNow + 1l * (len - 1) * k >= a[idx]) {
				ans = Math.min(ans, dp(idx + 1, lastReload));
				ans = Math.min(ans, (bulletsNow + 1l * (len - 1) * k - a[idx]) % k + dp(idx + 1, idx + 1));
			} else if (bulletsNow + 1l * (len) * k >= a[idx]) {
				ans = Math.min(ans, dp(idx + 1, lastReload));
			}
		} else {
			if (bulletsNow + 1l * len * k >= a[idx]) {
				ans = Math.min(ans, dp(idx + 1, lastReload));
				ans = Math.min(ans, (bulletsNow + 1l * len * k - a[idx]) % k + dp(idx + 1, idx + 1));
			}
		}
		return dp[idx][lastReload] = ans;
	}

	public static void main(String[] args) throws IOException {
		Scanner sc = new Scanner(System.in);
		PrintWriter pw = new PrintWriter(System.out);
		n = sc.nextInt();
		k = sc.nextInt();
		l = new int[n];
		r = new int[n];
		a = new int[n];
		ps = new long[n];
		for (int i = 0; i < n; i++) {
			l[i] = sc.nextInt();
			r[i] = sc.nextInt();
			a[i] = sc.nextInt();
			if (i == 0) {
				ps[i] = a[i];
			} else {
				ps[i] = a[i] + ps[i - 1];
			}
		}
		dp = new long[n][n];
		for (long[] x : dp) {
			Arrays.fill(x, -1);
		}
		long wasted = dp(0, 0);
		if (wasted == (long) 1e18) {
			pw.println(-1);
		} else {
			pw.println(wasted + ps[n - 1]);
		}
		pw.close();
	}

	static class Scanner {
		BufferedReader br;
		StringTokenizer st;

		public Scanner(InputStream s) {
			br = new BufferedReader(new InputStreamReader(s));
		}

		public Scanner(FileReader f) {
			br = new BufferedReader(f);
		}

		public String next() throws IOException {
			while (st == null || !st.hasMoreTokens())
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		public long nextLong() throws IOException {
			return Long.parseLong(next());
		}

		public double nextDouble() throws IOException {
			return Double.parseDouble(next());
		}

		public int[] nextIntArr(int n) throws IOException {
			int[] arr = new int[n];
			for (int i = 0; i < n; i++) {
				arr[i] = Integer.parseInt(next());
			}
			return arr;
		}

	}

}
