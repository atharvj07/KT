#include <bits/stdc++.h>
using namespace std;
int solution(int n, int m) {
  if (m <= n)
    return n - m;
  else if (m % 2 == 0)
    return solution(n, m / 2) + 1;
  else
    return solution(n, m + 1) + 1;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, m;
  cin >> n >> m;
  cout << solution(n, m);
}
