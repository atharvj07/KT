#include <bits/stdc++.h>
using namespace std;
int md = 1e9 + 7;
int __gcd(int a, int b) {
  if (b == 0) return a;
  return __gcd(b, a % b);
}
int poww(int a, int b) {
  int res = 1;
  while (b) {
    if (b & 1) {
      res *= a;
      res %= md;
    }
    a *= a;
    a %= md;
    b >>= 1;
  }
  return res;
}
void ainp(int arr[], int n) {
  for (int i = 1; i <= n; i++) cin >> arr[i];
}

  cin >> n >> s;
  long long int lo = 1, hi = n;
  while (lo <= hi) {
    long long int mid = (hi + lo) / 2;
    long long int temp = mid, dig = 0;
    while (temp) {
      dig += temp % 10;
      temp /= 10;
    }
    if (mid - dig < s)
      lo = mid + 1;
    else
      hi = mid - 1;
  }
