#include <iostream>
#include <queue>
#include <cstring>
using namespace std;

int dx[]={0,-1,0,1};
int dy[]={-1,0,1,0};

bool f[50][50];
bool h[50][50][50][50];

class State
{
public:
	int lx,ly,rx,ry,c;
	State(int lx, int ly, int rx, int ry, int c)
	:lx(lx),ly(ly),rx(rx),ry(ry),c(c)
	{}
};
int main()
{
	int W,H;
	while(cin >> W >> H, (W||H))
	{
		memset(h,0,sizeof(h));

		int lx,ly,rx,ry;
		cin >> lx >> ly >> rx >> ry;

		lx--; ly--;
		rx--; ry--;

		for(int i=0; i<H; i++)
		for(int j=0; j<W; j++)
			cin >> f[j][i];

		queue<State> q;
		q.push(State(lx,ly,rx,ry,0));

		bool g=false;
		while(!q.empty())
		{
			State s = q.front(); q.pop();
			if(s.c >= 100) break;

			if(s.lx==s.rx&&s.ly==s.ry)
			{
				g=true;
				cout << s.c << endl;
				break;
			}

			for(int i=0; i<4; i++)
			{
				int tlx=s.lx+dx[i], tly=s.ly+dy[i];
				int trx=s.rx+dx[(i+2)%4], ttry=s.ry+dy[(i+2)%4];

				if(tlx<0||tlx>=W||tly<0||tly>=H||f[tlx][tly])
				{
					tlx = s.lx;
					tly = s.ly;
				}
				if(trx<0||trx>=W||ttry<0||ttry>=H||f[trx][ttry])
				{
					trx = s.rx;
					ttry = s.ry;
				}

				if(h[tlx][tly][trx][ttry]) continue;
				h[tlx][tly][trx][ttry]=1;
				
				q.push(State(tlx,tly,trx,ttry,s.c+1));

			}
		}

		if(!g) cout << "NA" << endl;

	}
}
