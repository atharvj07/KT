import java.util.Scanner;

public class Question {
	
	public static void main(String[] args) {
		
	        int n,k;
	        Scanner sc = new Scanner(System.in);
	        n = sc.nextInt();
	        k = sc.nextInt();
	        int arr[] = new int[n];
	        int b = n/k;
	        int globalcount1 = 0,globalcount2 = 0;
	        int max_result = 0;
	        for(int i = 0;i<n;i++){
	            arr[i] = sc.nextInt();
	            if(arr[i]==1)
	            		globalcount1++;
	            else
	            		globalcount2++;
	        }
        for(int i = 1;i<=k;i++){
	            int count1 = 0,count2 = 0;
	           for(int j = i-1;j<n;j=j+k){
	                  if(arr[j]==1)
	                      count1++;
	                  else
	                      count2++;
	           }
	           int res = Math.abs((globalcount1-count1)-(globalcount2-count2));
	           if(res > max_result)
	                max_result = res;
	        }
	        System.out.println(max_result);

	    
		
	}

}
