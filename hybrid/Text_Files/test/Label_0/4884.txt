#include "bits/stdc++.h"

using namespace std;

using ll = long long;
using ld = long double;
using P = pair<int, int>;
constexpr ld EPS = 1e-12;
constexpr int INF = numeric_limits<int>::max() / 2;
constexpr int MOD = 1e9 + 7;

template <typename T>
void printv(const vector<T> &v)
{
    int sz = v.size();
    for (int i = 0; i < sz; i++)
    {
        cout << v[i] << " \n"[i == sz - 1];
    }
}

struct State
{
    int x, y, dist;
};

bool operator<(const State &e, const State &f)
{
    return e.dist > f.dist;
}

int main()
{
    cin.tie(0);
    ios::sync_with_stdio(false);

    int r, c;
    cin >> r >> c;
    vector<string> s(r);
    int sx, sy, gx, gy;
    for (int i = 0; i < r; i++)
    {
        cin >> s[i];
        for (int j = 0; j < c; j++)
        {
            if (s[i][j] == 'S')
            {
                sx = i;
                sy = j;
                s[i][j] = '.';
            }
            if (s[i][j] == 'G')
            {
                gx = i;
                gy = j;
                s[i][j] = '.';
            }
        }
    }
    vector<vector<int>> d(r, vector<int>(c, INF));
    int dx[4] = {1, 0, -1, 0}, dy[4] = {0, 1, 0, -1};
    priority_queue<State> que;
    d[sx][sy] = 0;
    que.push(State{sx, sy, 0});
    while (!que.empty())
    {
        auto t = que.top();
        que.pop();
        int x = t.x, y = t.y, dist = t.dist;
        if (d[x][y] < dist)
            continue;
        // 通常移動
        for (int i = 0; i < 4; i++)
        {
            int nx = x + dx[i], ny = y + dy[i];
            if (nx < 0 || r <= nx || ny < 0 || c <= ny)
                continue;
            if (s[nx][ny] == '#')
                continue;
            if (s[nx][ny] == 'x')
                continue;
            int cost = (s[nx][ny] == 'o' ? 1 : 2);
            if (d[nx][ny] <= d[x][y] + cost)
                continue;
            d[nx][ny] = d[x][y] + cost;
            que.push(State{nx, ny, d[nx][ny]});
        }
        // 塗り+移動
        for (int i = 0; i < 4; i++)
        {
            for (int j = 1; j <= 3; j++)
            {
                int nx = x + j * dx[i], ny = y + j * dy[i];
                if (nx < 0 || r <= nx || ny < 0 || c <= ny)
                    continue;
                if (s[nx][ny] == '#')
                    break;
                int cost = 2 + j;
                if (d[nx][ny] <= d[x][y] + cost)
                    continue;
                d[nx][ny] = d[x][y] + cost;
                que.push(State{nx, ny, d[nx][ny]});
            }
        }
    }
    cout << d[gx][gy] << endl;
}
