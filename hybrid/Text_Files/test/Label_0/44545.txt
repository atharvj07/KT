(a, b) = map(int, input().split())
val = a // 5
an = a % 5
aa = []
for i in range(5):
	aa.append(val + (1 if an >= i + 1 else 0))
val = b // 5
an = b % 5
bb = []
for i in range(5):
	bb.append(val + (1 if an >= i + 1 else 0))
ans = 0
for i in range(4):
	ans += aa[i] * bb[3 - i]
ans += aa[4] * bb[4]
print(ans)
