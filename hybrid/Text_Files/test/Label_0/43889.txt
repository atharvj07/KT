pri = pow(10, 9) + 7
import sys
input = sys.stdin.readline
a = int(input())
for i in range(a):
	(n, k) = map(int, input().split())
	g1 = pow(2, n, pri)
	g2 = pow(2, n, pri)
	if k == 0:
		print(1)
		continue
	ans = [0 for i in range(k)]
	if n % 2 == 0:
		ways = pow(2, n - 1, pri)
	else:
		ways = pow(2, n - 1, pri)
		ways += 1
	ans[0] = ways
	for i in range(1, k):
		if n % 2 == 0:
			ways = pow(2, n - 1, pri)
		else:
			ways = pow(2, n - 1, pri)
			ways += 1
		ways %= pri
		if n % 2 == 0:
			ans[i] = ans[i - 1] * (ways - 1) % pri
			ans[i] += g2
			ans[i] %= pri
			g2 = g2 * g1
			g2 %= pri
		else:
			ans[i] = ans[i - 1] * ways
			ans[i] %= pri
	print(ans[-1])
