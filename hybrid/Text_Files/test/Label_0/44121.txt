while True:
	try:

		def read():
			(n, m, k) = map(int, input().split())
			mtrx = []
			for i in range(n):
				mtrx.append(list(map(int, input().split())))
			solve(n, m, k, mtrx)

		def solve(n, m, k, mtrx):
			ans = rplc = cunt = 0
			(i, j) = (0, 0)
			while i < m:
				j = 0
				s = 0
				(pos, rplc) = (0, 0)
				while j < n:
					if mtrx[j][i] == 1:
						(z, t) = (j, 0)
						while z < j + k and z < n:
							t += mtrx[z][i]
							z += 1
						if t > s:
							s = t
							rplc = pos
						pos += 1
					j += 1
				i += 1
				ans += s
				cunt += rplc
			print(ans, cunt)
		if __name__ == '__main__':
			read()
	except EOFError:
		break
