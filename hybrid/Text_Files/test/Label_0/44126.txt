def func(x, d, n, k):
	ans = str(x)
	for i in range(d):
		ans = '9' + ans
	for i in range(k + 1):
		y = (x + i) % 10
		if y >= x:
			n -= d * 9 + y
		else:
			n -= 1 + y
		if n < 0:
			return -1
	if n % (k + 1) != 0:
		return -1
	p = int(n / (k + 1))
	if p >= 8:
		ans = '8' + ans
		n -= (k + 1) * 8
		n = int(n / (k + 1))
		v = int(n / 9)
		for i in range(v):
			ans = '9' + ans
		if n - 9 * v != 0:
			ans = str(n - 9 * v) + ans
	else:
		ans = str(p) + ans
	return int(ans)
t = int(input())
while t:
	t -= 1
	(n, k) = list(map(int, input().split()))
	ans = -1
	for i in range(10):
		for j in range(18):
			res = func(i, j, n, k)
			if res != -1 and ans == -1:
				ans = res
			elif res != -1:
				ans = min(ans, res)
	print(ans)
