#include <bits/stdc++.h>
int i, n, x, s, k, q;
float avg;
int main() {
  scanf("%d%d", &n, &k);
  s = 0;
  for (i = 1; i <= n; i++) {
    scanf("%d", &x);
    s += x;
  }
  q = 0;
  while (1) {
    avg = float(float(s) / float(n));
    if (avg >= float(float(k) - 0.5)) {
      printf("%d\n", q);
      break;
    } else {
      s += k;
      n++;
      q++;
    }
  }
  return 0;
}
