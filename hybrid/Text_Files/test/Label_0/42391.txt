for case in range(int(input())):
	n = int(input())
	series = [int(x) for x in input().split()]
	no_of_ways = 0
	set_a = set()
	set_b = set(series)
	letter_freq = {}
	for l in series:
		try:
			letter_freq[l] += 1
		except KeyError:
			letter_freq[l] = 1
	answers = []
	indices_to_end_a = set()
	visited = {}
	curr_max = -1
	for i in range(1, n + 1):
		num = series[i - 1]
		if num in visited:
			break
		visited[num] = True
		curr_max = max(curr_max, num)
		if curr_max == i:
			indices_to_end_a.add(i)
	visited = {}
	curr_max = -1
	series2 = series[::-1]
	indices_to_end_b = set()
	for i in range(1, n + 1):
		num = series2[i - 1]
		if num in visited:
			break
		visited[num] = True
		curr_max = max(curr_max, num)
		if curr_max == i:
			indices_to_end_b.add(n - i)
	for i in indices_to_end_a:
		if i in indices_to_end_b:
			answers.append([i, n - i])
	print(len(answers))
	for a in answers:
		print(a[0], a[1])
