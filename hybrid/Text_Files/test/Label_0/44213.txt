t = int(input())
q = []
for _ in range(t):
	n = int(input())
	a = [0] + list(map(int, input().split()))
	f = [0] * (n + 1)
	avail = [1] * (n + 1)
	t = [0] * (n + 1)

	def dfs(s):
		global q
		avail[s] = 0
		q.append(s)
		if avail[a[s]] == 1:
			dfs(a[s])
		else:
			res = len(q)
			for i in q:
				f[i] = res
			q = []
	for i in range(1, n + 1):
		if avail[i] == 1:
			dfs(i)
	for i in range(1, n + 1):
		print(f[i], end=' ')
	print()
