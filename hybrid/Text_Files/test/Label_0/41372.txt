from operator import itemgetter
t = int(input())
for _ in range(t):
	n = int(input())
	arr = [int(i) for i in input().split()]
	if len(set(arr)) < n:
		print('YES')
	else:
		s = list(map(itemgetter(0), sorted(list(enumerate(arr)), key=itemgetter(1))))
		d = set(range(n))
		ans = 0
		while d:
			z = d.pop()
			while s[z] in d:
				ans += 1
				z = s[z]
				d.remove(z)
		print('YES' if ans % 2 == 0 else 'NO')
