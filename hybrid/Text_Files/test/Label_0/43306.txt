import sys
import math
mod = 10 ** 9 + 7
LI = lambda : [int(k) for k in input().split()]
input = lambda : sys.stdin.readline().rstrip()
IN = lambda : int(input())
S = lambda : input()
r = range
for t in r(IN()):
	n = IN()
	li = LI()
	g = 0
	for i in range(n):
		g = math.gcd(g, li[i])
	for i in range(n):
		li[i] //= g
	ans = 0
	for i in range(n):
		j = i
		num1 = li[i]
		num2 = li[(i + 1) % n]
		while num1 != num2:
			j += 1
			num1 = math.gcd(num1, li[j % n])
			num2 = math.gcd(num2, li[(j + 1) % n])
		ans = max(ans, j - i)
	print(ans)
