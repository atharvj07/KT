#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long n, m, k;
  cin >> n >> m >> k;
  if (4 * m * n - 2 * n - 2 * m < k) {
    cout << "NO";
    return 0;
  }
  vector<pair<long long, string> > moves, ans;
  for (int i = 0; i < n - 1; i++) {
    if (m > 1) moves.push_back({(m - 1), "RDU"});
    if ((m - 1) / 4 != 0) {
      moves.push_back({(m - 1) / 4, "LLLL"});
    }
    string move;
    for (int j = 0; j < (m - 1) % 4; j++) {
      move += 'L';
    }
    if (!move.empty()) moves.push_back({1, move});
    moves.push_back({1, "D"});
  }
  if ((m - 1) / 4 != 0) {
    moves.push_back({(m - 1) / 4, "RRRR"});
  }
  string t1, t2, t3;
  for (int j = 0; j < (m - 1) % 4; j++) {
    t1 += 'R';
  }
  if (!t1.empty()) moves.push_back({1, t1});
  if ((m - 1) / 4 != 0) {
    moves.push_back({(m - 1) / 4, "LLLL"});
  }
  for (int j = 0; j < (m - 1) % 4; j++) {
    t2 += 'L';
  }
  if (!t2.empty()) moves.push_back({1, t2});
  if ((n - 1) / 4 != 0) {
    moves.push_back({(n - 1) / 4, "UUUU"});
  }
  for (int j = 0; j < (n - 1) % 4; j++) {
    t3 += 'U';
  }
  if (!t3.empty()) moves.push_back({1, t3});
  long long cnt = 0;
  for (int i = 0; i < moves.size(); i++) {
    if (cnt + moves[i].first * moves[i].second.size() < k) {
      cnt += moves[i].first * moves[i].second.size();
      ans.push_back(moves[i]);
    } else if (cnt + moves[i].first * moves[i].second.size() == k) {
      cnt += moves[i].first * moves[i].second.size();
      ans.push_back(moves[i]);
      break;
    } else {
      for (int j = 1; j <= moves[i].first; j++) {
        for (int x = 0; x < moves[i].second.size(); x++) {
          if ((j - 1) * moves[i].second.size() + x + 1 + cnt == k) {
            cnt += (j - 1) * moves[i].second.size() + x + 1;
            if (j > 1) ans.push_back({j - 1, moves[i].second});
            ans.push_back({1, moves[i].second.substr(0, x + 1)});
            goto done;
          }
        }
      }
    done:
      break;
    }
  }
  cout << "YES\n";
  cout << ans.size() << "\n";
  for (int i = 0; i < ans.size(); i++) {
    cout << ans[i].first << " " << ans[i].second << "\n";
  }
}
