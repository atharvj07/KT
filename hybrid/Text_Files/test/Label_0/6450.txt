#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;

const long double PI = acos(-1);
constexpr long double EPS = 1e-15;
constexpr int inf = 2e9;
constexpr ll INF = 2e18;
constexpr ll MOD = 1e9+7;
constexpr ll MOD1 = 998244353;
typedef pair<ll,ll> P;

#define all(v) (v).begin(), (v).end()
#define rep(i,a,b) for (int i = (a); i < (b); i++)
#define REP(i,n) rep(i,0,n)
#define sz(s) (s).size()
#define pb push_back
#define fi first
#define se second
//#define mp make_pair

int ans[110][110];
set<int> st[10010];

int main(){
    int n;
    cin >> n;
    int k = 2 * n * n - 2 * n;
    REP(i,k) {
        int a,b;
        cin >> a >> b;
        st[a].insert(b);
        st[b].insert(a);
    }
    int beg = 0;
    REP(i,n*n) {
        if (sz(st[i+1]) == 2) {
            beg = i+1;
            break;
        }
    }
    ans[0][0] = beg;
    ans[0][1] = *st[beg].begin();
    int right = ans[0][1];
    st[beg].erase(right);
    st[right].erase(beg);
    ans[1][0] = *st[beg].begin();
    int down = ans[1][0];
    st[beg].erase(down);
    st[down].erase(beg);
    int left;
    int up;
    for (int i = 0; i < n - 1; i++) {
        if (i != 0) {
            if (ans[i+1][0] == 0)
                ans[i+1][0] = *st[ans[i][0]].begin();
            up = ans[i][0];
            down = ans[i+1][0];
            st[up].erase(down);
            st[down].erase(up);
        }
        for (int j = 1; j < n; j++) {
            if (i == 0 && j == 0) continue;
            //下
            up = ans[i][j];
            left = ans[i+1][j-1];
            set<int> tmp;
            set_intersection(all(st[up]), all(st[left]), inserter(tmp, tmp.end()));
            if (ans[i+1][j] == 0)
                ans[i+1][j] = *tmp.begin();
            down = ans[i+1][j];
            st[down].erase(up);
            st[down].erase(left);
            st[up].erase(down);
            st[left].erase(down);

            if (j != n-1) {
                //右
                left = ans[i][j];
                if (ans[i][j+1] == 0)
                    ans[i][j+1] = *st[left].begin();
                int now = ans[i][j+1];
                st[now].erase(left);
                st[left].erase(now);
                if (i != 0) {
                    up = ans[i-1][j+1];
                    st[up].erase(now);
                    st[now].erase(up);
                }
            }
        }
    }
    if (ans[n-1][n-1] == 0)
        ans[n-1][n-1] = *st[ans[n-1][n-2]].begin();
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cout << ans[i][j] << " \n"[j == n-1];
        }
    }
    return 0;
}
