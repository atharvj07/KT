(n, m) = map(int, input().split())
l = list(map(int, input().split()))
p = []
paint_remaining = sum(l)
if paint_remaining < n or len(l) + l[len(l) - 1] - 1 > n:
	print(-1)
else:
	firstindex = 1
	p.append(1)
	paint_remaining -= l[0]
	failed = False
	for paint_length in l[1:]:
		if n - firstindex > paint_remaining:
			firstindex = n - paint_remaining + 1
		else:
			firstindex += 1
			if firstindex + paint_length - 1 > n:
				print(-1)
				failed = True
				break
		paint_remaining -= paint_length
		p.append(firstindex)
	if not failed:
		print(*p)
