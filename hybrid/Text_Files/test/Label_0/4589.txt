#include <bits/stdc++.h>
using namespace std;
long long n, k;
vector<pair<long long, long long> > g[100005];
bool vis[100005];
long long d[100005];
long long deg[100005];
long long td[100005];
bool ok(long long M) {
  queue<long long> q;
  long long t = 0;
  for (long long i = 1; i <= n; i++) {
    vis[i] = false;
    d[i] = 0;
    td[i] = deg[i];
  }
  for (long long i = 1; i <= n; i++) {
    if (deg[i] == 1) {
      q.push(i);
      d[i] = 0;
    } else if (deg[i] >= 3) {
      t++;
    }
  }
  long long verts = n;
  while (q.empty() == false) {
    long long u = q.front();
    q.pop();
    if (vis[u]) continue;
    pair<long long, long long> X = {-1, -1};
    for (auto i : g[u]) {
      long long to = i.first;
      long long w = i.second;
      if (vis[to] == false) {
        X = i;
        break;
      }
    }
    if (X.first == -1 || X.second == -1) continue;
    long long to = X.first;
    long long w = X.second;
    long long Y = max(d[to], d[u] + w);
    if (Y > M) continue;
    d[to] = max(d[to], d[u] + w);
    vis[u] = true;
    if (td[to] == 3) {
      t--;
    }
    td[to]--;
    verts--;
    if (td[to] == 1) {
      q.push(to);
    }
  }
  if (verts <= k && t == 0) return true;
  return false;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  cin >> n >> k;
  for (long long i = 1; i < n; i++) {
    long long a, b, c;
    cin >> a >> b >> c;
    g[a].push_back(make_pair(b, c));
    g[b].push_back(make_pair(a, c));
    deg[a]++;
    deg[b]++;
  }
  long long lo = 0LL;
  long long hi = 1e9 + 10;
  while (hi > lo) {
    long long mid = lo + (hi - lo) / 2;
    if (ok(mid)) {
      hi = mid;
    } else
      lo = mid + 1;
  }
  cout << lo << endl;
  return 0;
}
