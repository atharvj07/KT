    void solve(int TC)throws Exception{
        int n = ni();
        long[][] p = new long[n][];
        
        for(int i = 0; i< n; i++)p[i] = new long[]{i, nl(), nl()};
        Arrays.sort(p, (long[] l1, long[] l2) -> Long.compare(l1[1], l2[1]));
        int m = ni();
        long[][] a = new long[m][];
        for(int i = 0; i< m; i++)a[i] = new long[]{i, nl(), nl(), nl()};
        Arrays.sort(a, (long[] l1, long[] l2) -> Long.compare(l1[1], l2[1]));
        TreeMap<Long, ArrayList<long[]>> mp = new TreeMap<>();
        
        int[] ans = new int[n];
        int ptr = 0;
        boolean f = true;
        for(int i = 0; i< n; i++){
            while(ptr < m && a[ptr][1] <= p[i][1]){
                if(!mp.containsKey(a[ptr][2]))mp.put(a[ptr][2], new ArrayList<>());
                ArrayList<long[]> lst = mp.get(a[ptr][2]);
                lst.add(new long[]{a[ptr][0], a[ptr][3]});
                mp.put(a[ptr][2], lst);
                ptr++;
            }
            Long x = mp.ceilingKey(p[i][2]);
            if(x == null){
                f = false;
                break;
            }
            ArrayList<long[]> lst = mp.get(x);
            long[] pp = lst.get(lst.size()-1);
            lst.remove(lst.size()-1);
            ans[(int)p[i][0]] = (int)pp[0];
            pp[1]--;
            if(pp[1] > 0)lst.add(pp);
            
            if(lst.size() > 0)mp.put(x, lst);
            else mp.remove(x);
        }
        if(!f)pn("NO");
        else{
            pn("YES");
            for(int i = 0; i< n; i++)p(1+ans[i]+" ");pn("");
        }
    }

    class FastReader{
        BufferedReader br;
        StringTokenizer st;
        public FastReader(){
            br = new BufferedReader(new InputStreamReader(System.in));
        }
 
        public FastReader(String s) throws Exception{
            br = new BufferedReader(new FileReader(s));
        }
 
        String next() throws Exception{
            while (st == null || !st.hasMoreElements()){
                try{
                    st = new StringTokenizer(br.readLine());
                }catch (IOException  e){
                    throw new Exception(e.toString());
                }
            }
            return st.nextToken();
        }
 
        String nextLine() throws Exception{
            String str = "";
            try{   
                str = br.readLine();	
            }catch (IOException e){
                throw new Exception(e.toString());
            }  
            return str;
        }
    }
