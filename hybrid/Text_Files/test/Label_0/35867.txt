public class Task5 {
    public static void main(String[] args) throws IOException {

        //Scanner scanner = new Scanner(new FileInputStream("file.in"));
        Scanner scanner = new Scanner(System.in);
        char[] all = scanner.next().toCharArray();
        char[] sub = scanner.next().toCharArray();
        String k = getk(all.length);
        int[] counts = new int[10];
        for (int i = 0; i < all.length; ++ i){
            counts[all[i] - '0']++;
        }

        for (int i = 0; i < sub.length; ++ i){
            counts[sub[i] - '0']--;
        }

        for (int i =0 ;i < k.length(); ++ i){
            counts[k.charAt(i) - '0']--;
        }

        int subFirst = sub[0] - '0';

        StringBuilder res = new StringBuilder();
        int min = 0;
        for (int i = 1; i < 10; ++ i){
            if (counts[i] > 0){
                min = i;
                break;
            }
        }

        boolean subUse = false;
        if (subFirst > 0){
            if (min == 0 || compareFirst(sub, min, counts) <= 0 ){
                res.append(sub);
                subUse = true;
            }else{
                res.append(min);
                counts[min] --;
            }
        }else if (counts[min] > 0){

            res.append(min);
            counts[min] --;
        }



        for (int i = 0 ; i < 10; ++ i){
            if (subUse == false){
                if (compare(sub, i, counts) <= -1){
                    res.append(sub);
                    subUse = true;
                    --i;
                }else {
                    for (int j = 0; j < counts[i]; ++ j){
                        res.append(i);
                    }
                }
            }else{
                for (int j = 0; j < counts[i]; ++ j){
                    res.append(i);
                }
            }
        }

        if (subUse == false){
            res.append(sub);
        }

        System.out.println(res);

        scanner.close();
    }

    public static int compare(char[] sub, int d, int[] counts){
        int res = 0;
        int dcounter = 0;
        for (int i = 0; i < sub.length; ++ i){
            if (dcounter == counts[d]){
                return 0;
            }
            if (sub[i] - '0' > d){
                res = 1;
                break;
            }else if (sub[i] - '0' < d){
                res = -1;
                break;
            }
            dcounter++;
        }
        return res;
    }

    public static int compareFirst(char[] sub, int d, int[] temp){
        int[] counts = Arrays.copyOf(temp, temp.length);
        int res = 0;
        boolean fl = false;
        for (int i = 0; i < sub.length; ++ i){
            if (counts[d] == 0){
                if (fl == false) {
                    for (int j = d + 1; j < 10; ++j) {
                        if (counts[j] > 0) {
                            d = j;
                            break;
                        }
                    }
                    if (counts[d] == 0) return 0;
                    fl=true;
                }else {
                    return 0;
                }
            }
            if (sub[i] - '0' > d){
                res = 1;
                break;
            }else if (sub[i] - '0' < d){
                res = -1;
                break;
            }
            counts[d] --;
            if (!fl){
                d = 0;
            }
        }
        return res;
    }

    public static String getk(int len){
        String l = len + "";
        int check = len - l.length();
        int checkLength;
        while (true){
            checkLength = ((len - l.length()) + "").length();
            if (checkLength + check == len){
                break;
            }
            check++;
        }
        return check + "";
    }


}
