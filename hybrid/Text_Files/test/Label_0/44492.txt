import sys
input = sys.stdin.readline
from collections import Counter
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	w = list(map(int, input().split()))
	if k == 1:
		print(n - max(Counter(w).values()))
	else:
		(d, m) = (set(w), 10 ** 5)
		for i in d:
			(a, b) = (0, 0)
			while a < n:
				if w[a] != i:
					a += k
					b += 1
				else:
					a += 1
			m = min(m, b)
		print(m)
