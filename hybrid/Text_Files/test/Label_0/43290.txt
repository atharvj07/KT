t = int(input())
while t > 0:
	t -= 1
	(a, b, c, d) = map(int, input().split(' '))
	l1 = ['Tidak'] * 4
	if (a + b) % 2 != 0:
		if a + d > 0:
			l1[0] = 'Ya'
		if b + c > 0:
			l1[1] = 'Ya'
	else:
		if a + d > 0:
			l1[3] = 'Ya'
		if b + c > 0:
			l1[2] = 'Ya'
	for i in l1:
		print(i, end=' ')
	print()
