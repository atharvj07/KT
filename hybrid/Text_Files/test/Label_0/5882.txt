#include <bits/stdc++.h>
using namespace std;
long long value(string a) {
  long long ans = 0;
  for (int i = 0; i < a.size(); i++) {
    ans *= 10;
    ans += a[i] - '0';
  }
  return ans;
}
int main() {
  string points;
  cin >> points;
  long long ans = -1;
  for (int i = 1; i < points.size() - 1; i++) {
    for (int j = i + 1; j < points.size(); j++) {
      if (j > i + 7) continue;
      if (i > 7) continue;
      if (points.size() - 1 - j > 7) continue;
      if (points[0] == '0' && i > 1) break;
      if (points[i] == '0' && j - i > 1) break;
      if (points[j] == '0' && points.size() - 1 - j > 0) continue;
      long long val1, val2, val3;
      string one = points.substr(0, i);
      string two = points.substr(i, j - i);
      string three = points.substr(j, points.size() - j);
      val1 = value(one);
      val2 = value(two);
      val3 = value(three);
      if (val1 > 1000000 || val2 > 1000000 || val3 > 1000000) continue;
      long long temp = val1 + val2 + val3;
      if (temp > ans) ans = temp;
    }
  }
  cout << ans;
  return 0;
}
