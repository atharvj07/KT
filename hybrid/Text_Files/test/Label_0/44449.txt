import sys, os, io
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
q = int(input())
s = []
l = 0
sl = 0
for _ in range(q):
	a = [int(i) for i in input().split()]
	if len(a) == 2:
		s.append(a[1])
	res = (sl + s[-1]) / (l + 1)
	while l < len(s) and (sl + s[l] + s[-1]) / (l + 2) < res:
		sl += s[l]
		l += 1
		res = (sl + s[-1]) / (l + 1)
	if len(a) == 1:
		print('{0:.8f}'.format(s[-1] - res))
