t = int(input())
for ti in range(t):
	(n,) = map(int, input().strip().split(' '))
	b = list(map(int, input().strip().split(' ')))
	a = []
	ptr = 0
	for i in range(1, n * 2 + 1):
		if ptr < n and i == b[ptr]:
			ptr += 1
		else:
			a.append(i)
	ptr = 0
	mmi = 0
	for i in range(n):
		while ptr < n and a[ptr] < b[i]:
			ptr += 1
		if ptr >= n:
			break
		else:
			mmi += 1
			ptr += 1
	ptr = n - 1
	mmx = 0
	for i in range(n):
		i = n - 1 - i
		while ptr >= 0 and a[ptr] > b[i]:
			ptr -= 1
		if ptr < 0:
			break
		else:
			mmx += 1
			ptr -= 1
	ans = mmi + mmx - n + 1
	print(ans)
