import java.util.*;
import java.lang.*;
import java.io.*;

/* Name of the class has to be "Main" only if the class is public. */
public class Ideone
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner in = new Scanner(System.in); 
		int n = in.nextInt(); 
		int s = in.nextInt();
		int d = in.nextInt();
		int total=0,ans =0; 
		int currentOr=0;
		for(int i=0;i<n;i++){
			currentOr = in.nextInt(); 
		if(currentOr > s){
			continue;
		}else{
			total+=currentOr; 
			
			if(total > d){
				ans++; 
				total=0;
			}
		}
		}
		
	System.out.println(ans);
	}
}
