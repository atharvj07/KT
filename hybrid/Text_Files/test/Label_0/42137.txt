import sys
from collections import deque
input = sys.stdin.buffer.readline
(n, h, m, k) = map(int, input().split())
tram = []
dic = {}
for i in range(n):
	(H, M) = map(int, input().split())
	r = M % (m // 2)
	tram.append(r)
	if r not in dic:
		dic[r] = 0
	dic[r] += 1
data = [(i, dic[i]) for i in dic] + [(i + m // 2, dic[i]) for i in dic]
data.sort()
k -= 1
add = deque([])
erase = deque([])
for (i, val) in data:
	add.append((i - k + 1, val))
	erase.append((i + 1, -val))
S = 0
while add and add[0][0] <= 0:
	(t, val) = add.popleft()
	S += val
while erase and erase[0][0] <= 0:
	(t, val) = erase.popleft()
	S += val
ans = S
res = 0
while add or erase:
	time = 10 ** 15
	if add:
		time = min(time, add[0][0])
	if erase:
		time = min(time, erase[0][0])
	if time >= m // 2:
		break
	while add and add[0][0] == time:
		(time, val) = add.popleft()
		S += val
	while erase and erase[0][0] == time:
		(time, val) = erase.popleft()
		S += val
	if ans > S:
		ans = S
		res = time
Ans = ans
print(Ans, (res + k) % (m // 2))
ans = []
for i in range(n):
	L = res
	R = res + k - 1
	if R >= tram[i] >= L or R >= tram[i] + m // 2 >= L:
		ans.append(i + 1)
print(*ans)
