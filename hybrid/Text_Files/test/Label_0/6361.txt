#include <bits/stdc++.h>
using namespace std;
const long long MOD = 1e9 + 7;
long long a[202000];
long long sum[202000];
int main() {
  ios::sync_with_stdio(0);
  long long n;
  cin >> n;
  long long T;
  cin >> T;
  long long Min = 0x3f3f3f3f;
  for (int i = 1; i <= n; ++i) cin >> a[i];
  for (int i = 1; i <= n; ++i) {
    sum[i] = sum[i - 1] + a[i];
  }
  if (T % sum[n] == 0) {
    cout << T / sum[n] * n << endl;
  } else {
    long long ans = T / sum[n] * n;
    T = T % sum[n];
    vector<long long> v, temp;
    long long tot = 0;
    for (int i = 1; i <= n; ++i) {
      if (a[i] <= T) {
        v.push_back(a[i]);
        Min = min(Min, a[i]);
        tot += a[i];
      }
    }
    int index = 0;
    while (T >= Min && v.size()) {
      index %= v.size();
      if (index == 0) {
        temp.clear();
        if (v.size() == 0) {
        } else if (T % tot == 0) {
          ans += T / tot * v.size();
          T %= tot;
        } else {
          ans += T / tot * v.size();
          T %= tot;
        }
        tot = 0;
        for (int i = 0; i < v.size(); i++) {
          if (v[i] <= T) {
            temp.push_back(v[i]);
            tot += v[i];
          }
        }
        v = temp;
      }
      if (T >= v[index]) {
        T -= v[index];
        ans++;
      }
      index++;
    }
    cout << ans << endl;
  }
  return 0;
}
