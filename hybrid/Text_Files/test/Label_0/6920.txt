li fastpow(li base,li exp,li M) 
{
    li res=1;
    while(exp>0) {
        if(exp&1)
            res=(res*base)%M;

        base=(base*base)%M;
        exp>>=1;
    }
    return res;
}

bool solve()
{
    string s;

    cin>>s;

    li n=s.length();

    li i=0,j=n-1;

    string tp="";

    char last = ('a'+ n);

    while(i<=j)
    {
        if(s[i]==(last-1))
        {
            last=s[i];
            i++;
            continue;
        }

        if(s[j]==(last-1))
        {
            last=s[j];
            j--;
            continue;
        }

        break;
    }

    if(last=='a')
    {
        return true;
    }

    return false;
}
