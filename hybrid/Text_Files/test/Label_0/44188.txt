n = int(input())
arr = list(map(int, input().split()))

def findCost(t):
	ans = 0
	for x in arr:
		li = []
		li.append(abs(t - x))
		li.append(abs(t + 1 - x))
		li.append(abs(t - 1 - x))
		ans += min(li)
	return ans
costs = []
for i in range(1, 100):
	costs.append(findCost(i))
print(costs.index(min(costs)) + 1, min(costs))
