import sys
input = sys.stdin.readline

class UnionFindAlgo:

	def __init__(self, n):
		self.par = [i for i in range(n)]

	def find(self, node):
		if self.par[node] == node:
			return node
		self.par[node] = self.find(self.par[node])
		return self.par[node]

	def union(self, u, v):
		a = self.find(u)
		b = self.find(v)
		if a == b:
			return
		self.par[b] = a
t = int(input())
for _ in range(t):
	(n, m) = map(int, input().split())
	p = list(map(int, input().split()))
	cnt = [0] * n
	for i in range(n):
		cnt[(i + 1 - p[i] + n) % n] += 1
	ans = []
	for i in range(n):
		if cnt[i] + 2 * m >= n:
			obj = UnionFindAlgo(n)
			curr = 0
			for j in range(i, n):
				obj.union(curr, p[j] - 1)
				curr += 1
			for j in range(0, i):
				obj.union(curr, p[j] - 1)
				curr += 1
			connectedRoot = set()
			for j in range(n):
				connectedRoot.add(obj.find(j))
			if n - len(connectedRoot) <= m:
				ans.append(i)
	print(len(ans), *ans)
