import java.util.Scanner;
public class t{
    public static void main(String args[]){
        Scanner in=new Scanner(System.in);
		String s=in.next();
		int a=0,b=0;
		int ii=s.indexOf("2");
		
		for(int i=0;i<s.length();i++){
			char ch=s.charAt(i);
			if(ch=='2'){break;}
			else if(ch=='0')a++;
		}
		for(int i=0;i<s.length();i++){
			char ch=s.charAt(i);
			if(ch=='1')b++;
		}
		for(int i=0;i<a;i++)System.out.print('0');
		for(int i=0;i<b;i++)System.out.print('1');
		if(ii!=-1)
			for(int i=ii;i<s.length();i++){
			char ch=s.charAt(i);
			if(ch!='1')System.out.print(ch);
		}
		System.out.println();
    }
}
