#include <bits/stdc++.h>
using namespace std;
string s;
int r = 1;
int flag = -1;
long long ans = 0;
int main() {
  cin >> s;
  int len = s.size();
  for (int l = 0; l < len && r < len; ++l) {
    while (1) {
      if (flag >= l) break;
      ++r;
      if (r == len) {
        cout << ans;
        return 0;
      }
      for (int k = 1;; ++k) {
        if (r < l + 2 * k) break;
        if (s[r] == s[r - k] && s[r] == s[r - 2 * k]) {
          flag = r - 2 * k;
          break;
        }
      }
    }
    int cha = len - r;
    ans += cha;
  }
  cout << ans;
  return 0;
}
