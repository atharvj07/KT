#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> g(n + 1);
  vector<int> a(n);
  for (int i = 0; i < m; i++) {
    int maxj = 0;
    for (int j = 0; j < n; j++) {
      cin >> a[j];
      if (a[j] > a[maxj])
        maxj = j;
      else if (a[j] == a[maxj]) {
        if (j < maxj) maxj = j;
      }
    }
    g[maxj + 1]++;
    a.resize(n);
  }
  int ma = 0;
  for (int i = 1; i <= n; i++) {
    if (g[i] > g[ma])
      ma = i;
    else if (g[i] == g[ma]) {
      if (i < ma) ma = i;
    }
  }
  cout << ma;
  return 0;
}
