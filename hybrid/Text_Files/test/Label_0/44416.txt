import math
import os
import random
import re
from sys import stdin, stdout
from collections import Counter
from itertools import permutations

def si():
	return str(stdin.readline())

def ii():
	return int(stdin.readline())

def mi():
	return map(int, stdin.readline().split())

def li():
	return list(mi())

def debug(x):
	return stdout.write(str(x))

def gcd(a, b):
	if b == 0:
		return a
	return gcd(b, a % b)
(l, r, x, y) = mi()
if y % x != 0:
	debug(0)
else:
	res = 0
	p = y // x
	t = 1
	while t * t <= p:
		if p % t == 0:
			a = x * t
			b = x * y / a
			if l <= a <= r and l <= b <= r and (gcd(a, b) == x):
				if b != a:
					res += 2
				else:
					res += 1
		t += 1
	debug(res)
