
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.List;
import java.util.Set;
import java.util.TreeSet;


public class CoderStrikeC {
    public static void main(String [] argv) throws Exception{
        
        new   CoderStrikeC().run(); 
    }
    
    int[] getArray(String line){
        String[] array=line.split(" ");
        int[] res=new int[array.length];
        for(int i=0;i<res.length;i++)
            res[i]=Integer.parseInt(array[i]);
        return res;
    }
     
     void getArrays(int n, BufferedReader reader, int[]... ar) throws Exception{
           for(int i=0;i<n;i++){
               String[] ss=reader.readLine().split(" ");
               for(int j=0;j<ss.length;j++)
                   ar[j][i]=Integer.parseInt(ss[j]);
           }
        }
     
    
        void run() throws Exception{
            BufferedReader reader=new BufferedReader(new InputStreamReader(System.in));
            int[] res=getArray(reader.readLine());
            int n=res[0];
            int m=res[1];
            boolean[] canBe=new boolean[n];
            Arrays.fill(canBe, true);
            
            int[] change=new int[m];
            int[] changeArt=new int[m];
            for(int i=0;i<m;i++){
                String[] ss=reader.readLine().split(" ");
                if(ss[0].equals("+"))
                    changeArt[i]=1;
                else
                    changeArt[i]=2;
                change[i]=Integer.parseInt(ss[1])-1;
            }
            
            //backwarts:
            Set<Integer> in=new TreeSet<Integer>();
            Set<Integer> used=new TreeSet<Integer>();
            for(int i=m-1;i>=0;i--){
                used.add(change[i]);
                if(changeArt[i]==2)
                    in.add(change[i]);
                else
                    in.remove(change[i]);
            }
            
            //forwarts:
            for(int i=0;i<m;i++){
                if(changeArt[i]==2){
                    if(in.size()>1) 
                        canBe[change[i]]=false;
                    in.remove(change[i]);
                    if(in.size()==0){
                        for(int u:used)
                            if(u!=change[i])
                                canBe[u]=false;
                        used.clear();
                        used.add(change[i]);
                    }
                }
                else{
                    if(in.size()!=0)
                        canBe[change[i]]=false;
                    
                    in.add(change[i]);
                    if(in.size()==1){
                        for(int u:used)
                            if(u!=change[i])
                                canBe[u]=false;
                        
                        used.clear();
                        used.add(change[i]);
                    }
                }
            }
            List<Integer> list=new LinkedList<Integer>();
            
            for(int i=0;i<n;i++)
                if(canBe[i])
                    list.add(i+1);
            
            int cnt=0;
            System.out.println(list.size());
            StringBuilder sb=new StringBuilder();
            if(list.size()!=0)
                for(int pos: list){
                    if(cnt!=0)
                        sb.append(" ");
                    sb.append(pos);
                    cnt++;
                }
                System.out.println(sb.toString());
                reader.close();
        }

}
