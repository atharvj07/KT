n = int(input())
a = list(map(int, input().strip().split()))[:n]
b = list(map(int, input().strip().split()))[:n]
db = {}
for i in range(len(a)):
	db[b[i]] = i
res = []
for i in range(len(a)):
	k = db[a[i]]
	if k >= i:
		res.append(k - i)
	else:
		res.append(len(a) - (i - k))
r = [0] * len(a)
for i in range(len(res)):
	r[res[i]] += 1
print(max(r))
