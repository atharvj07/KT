import sys
input = sys.stdin.readline
for _ in range(int(input())):
	n = int(input())
	s = input()[:-1]
	if n == 1:
		print(s * 2)
	elif s[0] == s[1]:
		print(s[0] * 2)
	else:
		for i in range(n - 1):
			if s[i + 1] > s[i]:
				print(s[:i + 1] + s[:i + 1][::-1])
				break
		else:
			print(s + s[::-1])
