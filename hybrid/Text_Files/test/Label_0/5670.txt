#include "bits/stdc++.h"
using namespace std;
typedef long long ll; 
const ll INF = 1LL << 60;
const ll MOD = 998244353;
typedef pair<int, int> P;
int A[100001], B[100001], C[100001];
ll dp[100001][1 << 4];
map<P, int> mp;
ll mod_pow(ll a, ll n) {
	if (n == 0) return 1;
	if (n % 2 == 0) {
		ll tmp = mod_pow(a, n / 2);
		return (tmp * tmp) % MOD;
	}
	return (a * mod_pow(a, n - 1)) % MOD;
}
int out(int y, int x, int val) {
	if (mp.count(P(y, x)) == 0) return 0;
	if (mp[P(y, x)] == (val & 1)) return 0;
	return 1;
}
int popcount(int x) {
	int ret = 0;
	while (x > 0) {
		if (x & 1) ret++;
		x >>= 1;
	}
	return ret;
}
int main() {
	int N, M;
	cin >> N >> M;
	for (int i = 0; i < M; i++) {
		cin >> A[i] >> B[i] >> C[i];
		A[i]--; B[i]--;
		mp[P(A[i], B[i])] = C[i];
		if (abs(A[i] - B[i]) >= 3) {
			if (mp.count(P(B[i], A[i])) && mp[P(B[i], A[i])] != C[i]) {
				cout << 0 << endl;
				return 0;
			}
			mp[P(B[i], A[i])] = C[i];
		}
	}
	ll fr = 0;
	for (int i = 0; i < N; i++) fr += N - (min(N - 1, i + 2) - max(0, i - 2) + 1);
	for (auto p : mp) if (abs(p.first.first - p.first.second) > 2) fr--;
	fr /= 2;
	ll t = mod_pow(2, fr);
	for (int mask = 0; mask < (1 << 4); mask++) {
		if (popcount(mask) % 2) continue;
		int NG = 0;
		for (int y = 0; y < 2; y++) for (int x = 0; x < 2; x++) {
			NG |= out(y, x, mask >> (y * 2 + x));
		}
		if (NG == 0) dp[1][mask]++;
	}
	for (int i = 2; i < N; i++) {
		for (int mask = 0; mask < (1 << 5); mask++) {
			if (popcount(mask) % 2) continue;
			if (out(i, i, mask >> 4)) continue;
			if (out(i - 1, i, mask >> 3)) continue;
			if (out(i - 2, i, mask >> 2)) continue;
			if (out(i, i - 1, mask >> 1)) continue;
			if (out(i, i - 2, mask >> 0)) continue;
			for (int mask2 = 0; mask2 < (1 << 4); mask2++) {
				if (popcount(mask2) % 2) continue;
				int nmask = (mask2 >> 3) & 1;
				nmask |= ((mask >> 1) & 1) << 1;
				nmask |= ((mask >> 3) & 1) << 2;
				nmask |= ((mask >> 4) & 1) << 3;
				if (popcount(nmask) % 2 == 0) (dp[i][nmask] += dp[i - 1][mask2]) %= MOD;
			}
		}
	}
	ll ans = 0;
	for (int i = 0; i < (1 << 4); i++) (ans += dp[N - 1][i]) %= MOD;
	(ans *= t) %= MOD;
	cout << ans << endl;
 }
