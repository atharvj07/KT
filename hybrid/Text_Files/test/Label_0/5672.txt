typedef long long ll;
ll p[64];
ll num(ll x,int j){
	return x/p[j+1]*p[j]+max(0LL,x%p[j+1]-p[j]);
}
void solve(int n){
	int nc=n;
	ll k[64],kc[64];
	rep(i,64) k[i]=-1;
	rep(i,n) cin>>k[i];
	rep(i,n) kc[i]=k[i];
//	while(n>0&&k[n-1]==0) n--;
//	if(n==-1){}
	rep(i,n){
		if(k[i]>=p[60]){
			puts("None");
			return;
		}
	}
	ll ansa=-1,ansb=-1;
	for(ll mo=kc[0]*2-1;mo<=kc[0]*2+1;mo++){
		ll a=0,b=0;
		ll m=mo;
		n=nc;
		rep(i,n) k[i]=kc[i];
		for(int i=n-1;i>=0;i--){
			if(k[i]==0) continue;
			if(k[i]==m){
				k[i]=0;
				a+=p[i],b+=p[i];
				continue;
			}
			if(k[i]>m){
				break;
			}
			rep(j,i) k[j]-=num(k[i],j);
			m-=k[i];
			b+=p[i]+k[i]-1;
			a+=p[i]-m;
			k[i]=0;
//			rep(j,n) show(k[j]);
//			show(m);
			rep(j,i) k[j]-=(m-num(m,j));
//			rep(j,n) show(k[j]);
			break;
		}
		bool can=true;
		rep(i,nc) if(k[i]!=0) can=false;
		if(1>a||a>b||b>1e18) can=false;
		if(can){
			if(ansa==-1){
				ansa=a,ansb=b;
			}else{
				puts("Many");
				return;
			}
		}
	}
	if(ansa==-1){
		puts("None");
		return;
	}
	cout<<ansa<<" "<<ansb<<endl;
}
int main(){
	p[0]=1;
	rep(i,63) p[i+1]=p[i]*2;
	while(true){
		int n;
		cin>>n;
		if(n==0) break;
		solve(n);
	}
}
