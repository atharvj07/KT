#include <bits/stdc++.h>
using namespace std;
map<pair<string, int>, vector<pair<string, int>>> g;
vector<pair<string, int>> bfs(vector<pair<string, int>> a) {
  vector<pair<string, int>> b;
  for (int i = 0; i < a.size(); ++i) {
    for (int j = 0; j < g[a[i]].size(); ++j) {
      b.push_back(g[a[i]][j]);
    }
  }
  return b;
}
int main() {
  int n;
  pair<string, int> pol;
  cin >> n;
  for (int i = 0; i < n; ++i) {
    int m;
    pair<string, int> aa;
    cin >> aa.first >> aa.second;
    if (i == 0) {
      pol = aa;
    }
    cin >> m;
    g[aa].resize(m);
    for (int j = 0; j < m; ++j) {
      cin >> g[aa][j].first >> g[aa][j].second;
    }
  }
  vector<pair<string, int>> data;
  data.resize(1);
  data[0] = pol;
  map<string, int> version;
  map<string, int> visit;
  vector<pair<string, int>> ans;
  visit[pol.first]++;
  while (!data.empty()) {
    data = bfs(data);
    version.clear();
    for (int i = 0; i < data.size(); ++i) {
      version[data[i].first] = max(version[data[i].first], data[i].second);
    }
    vector<pair<string, int>> n_data;
    for (int i = 0; i < data.size(); ++i) {
      if (visit.count(data[i].first) == 0) {
        n_data.push_back(make_pair(data[i].first, version[data[i].first]));
        visit[data[i].first]++;
      }
    }
    for (int i = 0; i < n_data.size(); ++i) {
      ans.push_back(n_data[i]);
    }
    data = n_data;
  }
  sort(ans.begin(), ans.end());
  cout << ans.size() << endl;
  for (int i = 0; i < ans.size(); ++i) {
    cout << ans[i].first << " " << ans[i].second << endl;
  }
}
