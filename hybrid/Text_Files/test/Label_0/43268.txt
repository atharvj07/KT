from math import log2
for _ in range(int(input())):
	(n, m) = map(int, input().split())
	bit = [0] * 61
	sum = 0
	for x in map(int, input().split()):
		bit[int(log2(x))] += 1
		sum += x
	if sum < n:
		print(-1)
		continue
	else:
		(index, res) = (0, 0)
		while index < 60:
			if 1 << index & n != 0:
				if bit[index] > 0:
					bit[index] -= 1
				else:
					while index < 60 and bit[index] == 0:
						index += 1
						res += 1
					bit[index] -= 1
					continue
			bit[index + 1] += bit[index] // 2
			index += 1
	print(res)
