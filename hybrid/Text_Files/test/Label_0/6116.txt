vector<int> dp[51],cdp[51],ldp[51];

int n,m;

void calcInv()
{
    vector<int>v(1200,0);
    v[0]=1;
    dp[1]=v;
    for(int i=2;i<=50;i++)
    {
        vector<int>csum(1200,0);
        for(int j=0;j<1198;j++)
        {
            int t=min(1199,j+i-1);
            csum[t]=(csum[t]+v[j])%m;
            csum[j]=((csum[j]-v[j])%m+m)%m;
        }
        int sum=0;
        for(int j=1199;j>-1;j--)
        {
            sum=(sum+csum[j])%m;
            v[j]=(v[j]+sum)%m;
        }
        dp[i]=v;
        cdp[i]=vector<int>(1200,0);
        sum=0;
        for(int j=1199;j>-1;j--)
        {
            sum=(sum+v[j])%m;
            cdp[i][j]=sum;
        }
    }
}

void calcLead()
{
    for(int i=2;i<=50;i++)
    {   
        ldp[i]=vector<int>(1200,0);
        for(int j=1;j<=1000;j++)
        {
            int sum=0;
            for(int k=0;k<1200;k++)
            {
                if(k+j+1>=1200)
                    break;
                sum=(sum+(dp[i][k]*1LL*cdp[i][k+j+1])%m)%m;
            }
            ldp[i][j]=sum;
        }
    }
}

int comb[1000][1000];



int main()
{
    cin >> n >> m;
    int fac[1024];
    if(n<3 or m==1)
        cout << "0\n";
    else
    {
        fac[0]=1;
        for(int i=1;i<1024;i++)
            fac[i]=fac[i-1]*1LL*i%m;
        comb[0][0]=1;
        comb[1][0]=comb[1][1]=1;
        for(int i=2;i<1000;i++)
        {
            for(int j=0;j<=i;j++)
            {
                if(j==0 or j==i)
                {
                    comb[i][j]=1;
                    continue;
                }
                comb[i][j]=(comb[i-1][j]+comb[i-1][j-1])%m;
            }
        }
        calcInv();
        calcLead();
        int ans=0;
        for(int i=1;i<=n-2;i++)
        {
            int t=comb[n][i-1],t1=0;
            t=(t*1LL*fac[i-1])%m;
            int rem=n-i;
            for(int k=2;k<=n-i+1;k++)
            {
                for(int j=1;j<k;j++)
                {

                    t1=(t1+ldp[rem][k-j])%m;
                }
            }
            t=(t*1LL*t1)%m;
            ans=(ans+t)%m;
        }
        cout << ans << '\n';
    }
}
