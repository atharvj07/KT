import sys
from math import *

def minp():
	return sys.stdin.readline().strip()

def mint():
	return int(minp())

def mints():
	return map(int, minp().split())
(n, k) = mints()
a = list(mints())
b = list(mints())
d = [0] * (n + 1)
for i in range(n - 1, -1, -1):
	d[i] = d[i + 1] + a[i] if b[i] == 1 else d[i + 1]
c = 0
for i in range(k):
	c += a[i]
r = c + d[k]
s = 0
for i in range(k, n):
	c += a[i]
	c -= a[i - k]
	if b[i - k]:
		s += a[i - k]
	r = max(r, s + c + d[i + 1])
print(r)
