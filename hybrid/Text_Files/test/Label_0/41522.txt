def twoDifferentDividers(n, start):
	i = start if start >= 2 else 2
	while i * i <= n:
		if n % i == 0:
			return (i, n // i)
		i += 1
	return (0, 0)
t = int(input())
anss = []
for _ in range(t):
	P = int(input())
	for q in range(0, P):
		PP = P - q
		candidates = twoDifferentDividers(PP, q)
		if candidates != (0, 0):
			if candidates[0] == candidates[1]:
				anss.append((candidates[0], PP))
			else:
				anss.append(candidates)
			break
for ans in anss:
	print(*ans)
