n = int(input())
sections = sorted((tuple(map(int, [i + 1] + input().split())) for i in range(n)), key=lambda x: (x[1], -x[2]))
max_end = (sections[0][0], sections[0][2])
for s in sections[1:]:
	if s[2] <= max_end[1]:
		print('{} {}'.format(s[0], max_end[0]))
		break
	else:
		max_end = (s[0], s[2])
else:
	print('-1 -1')
