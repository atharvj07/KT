void solve() {

	int n;
	scanf("%d",&n);

	vector<ll> a(n),b(n);

	rep(i,0,n) scanf("%lld",&a[i]);
	rep(i,0,n) scanf("%lld",&b[i]);

	ll sum=0;

	rep(i,0,n) sum += a[i]*b[i];

	ll ans = sum;

	rep(i, 0, n) {

		ll cur = sum;
		for(int x=i-1,y=i+1;x>=0&&y<n;x--,y++) {
			cur -= a[x]*b[x] + a[y]*b[y];
			cur += a[x]*b[y] + a[y]*b[x];
			ans = max(ans, cur);
		}
	}

	rep(i,0,n-1) {
		ll cur = sum;
		for(int x=i,y=i+1;x>=0&&y<n;x--,y++) {
			cur -= a[x]*b[x] + a[y]*b[y];
			cur += a[x]*b[y] + a[y]*b[x];
			ans = max(ans, cur);	
		}
	}

	printf("%lld\n",ans);
	return;
}

int main() {

	int t;
	t=1;
	// scanf("%d",&t);
	while(t--) {
		solve();
	}

	return 0;
}
