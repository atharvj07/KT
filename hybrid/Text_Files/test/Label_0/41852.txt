T = int(input())
for t_itr in range(T):
	(N, M, A, B) = map(int, input().rstrip().split())
	result = [['0' for _ in range(M)] for _ in range(N)]
	if M * B != N * A:
		print('No')
	else:
		for i in range(N):
			for j in range(i * A, A + i * A):
				result[i][j % M] = '1'
		print('Yes')
		for r in result:
			print(''.join(r))
