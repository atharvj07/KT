import sys
from array import array
from typing import List, Tuple, TypeVar, Generic, Sequence, Union

def input():
	return sys.stdin.buffer.readline().decode('utf-8')

def main():
	(n, m) = map(int, input().split())
	a = list(map(int, input().split()))
	if sum(a) < n or m - 1 + a[-1] > n:
		print(-1)
		exit()
	pos = list(range(1, m)) + [n - a[-1] + 1]
	for i in range(m - 2, -1, -1):
		pos[i] = min(pos[i], n - a[i] + 1, pos[i + 1] - 1)
		if pos[i] + a[i] < pos[i + 1]:
			pos[i] = min(n - a[i] + 1, pos[i + 1] - a[i])
	if min(pos) < 1:
		print(-1)
	else:
		print(*pos)
main()
