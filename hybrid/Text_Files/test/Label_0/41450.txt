def answer(n, A, B):
	count = 0
	min1 = min(A)
	min2 = min(B)
	for i in range(n):
		k1 = A[i] - min1
		k2 = B[i] - min2
		kmin = min(k1, k2)
		count += k1 + k2 - kmin
	return count
t = int(input())
for i in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	print(answer(n, a, b))
