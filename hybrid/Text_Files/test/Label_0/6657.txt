#include <bits/stdc++.h>
using namespace std;
int n, k, x, s;
vector<int> p;
long long first;
void check() {
  for (int i = 0; i < k; i++) {
    x = i;
    s = 0;
    while ((s < 8) && (x != 0)) {
      x = p[x];
      s++;
    }
    if (s == 8) return;
  }
  first++;
}
void put(int d) {
  if (d == k) {
    check();
  } else {
    for (p[d] = 0; p[d] < k; p[d]++) put(d + 1);
  }
}
int main() {
  cin >> n >> k;
  p.resize(k);
  put(0);
  for (long long i = 1; i <= n - k; i++) first = (first * (n - k)) % 1000000007;
  cout << first << endl;
  return 0;
}
