t = int(input())
for _ in range(t):
	n = int(input())
	numbers = list(map(int, input().split()))
	old_sum = sum(numbers)
	if old_sum * (n - 2) % n != 0:
		print(0)
		continue
	new_sum = old_sum * (n - 2) // n
	diff = old_sum - new_sum
	amount = {}
	for x in numbers:
		amount[x] = amount.get(x, 0) + 1
	pairs = 0
	for x in numbers:
		if diff - x in amount:
			pairs += amount[diff - x] - int(diff == 2 * x)
	print(pairs // 2)
