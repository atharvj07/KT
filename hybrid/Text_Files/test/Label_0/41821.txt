for _ in range(int(input())):
	(a, b, c, d) = [int(x) for x in input().split()]
	result = ['Ya' if a + d > 0 else 'Tidak', 'Ya' if b + c > 0 else 'Tidak']
	if (a + b) % 2 == 0:
		result = result[::-1]
		result[0:0] = ['Tidak', 'Tidak']
	else:
		result.extend(['Tidak', 'Tidak'])
	print(' '.join(result))
