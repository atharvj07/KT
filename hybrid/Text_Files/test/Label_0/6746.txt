#include<bits/stdc++.h>
using namespace std;

int N;

struct Trie {
  struct node {
    node(){
      memset(v,-1,sizeof(v));
      ef = false;
    }
    node(char c):c(c){
      memset(v,-1,sizeof(v));
      ef = false;
    }
    ~node(){}
    char c;
    int v[28];
    bool ef;
  };


  node* root;
  void init(){
    vs.clear();
    root = new node();
    vs.emplace_back( root );
  }

  void push(const string& s, node* v, int id){
    if((int)s.size() == id) {
      v->ef = true; return;
    }
    int nx = s[id]-'a';
    if( v->v[nx]==-1 ){
      vs.emplace_back( new node(s[id]) );
      v->v[nx] = vs.size()-1;
    }
    push( s, vs[v->v[nx]], id+1 );
  }
  void push(const string& s){    
    push( s, root, 0 );
  }  
  
  vector<node*> vs;
  node* operator[](size_t id) {
    return vs[id];
  }
};

string nexs[] = {"",
                 "",
                 "abc",
                 "def",
                 "ghi",
                 "jkl",
                 "mno",
                 "pqrs",
                 "tuv",
                 "wxyz"};

Trie T;
string S;
bool dp[333][5111];
vector<string> res;
vector<char> tms;

void init(){
  T.init();
  for(int i=0;i<=300;i++)
    for(int j=0;j<=5001;j++)
      dp[i][j] = true;
  res.clear();
  tms.clear();
}

bool solve(int id,int v){
  //cout << id <<  " " << v << " " << dp[id][v] << endl;
  if( !dp[id][v] ) return dp[id][v];
  bool rf = false;
  auto nv = T[v];
  if( id == (int)S.size() ){
    if( nv->ef ) {
      string ns = "";
      for( char c : tms )
        ns += c;
      ns += ".";
      res.emplace_back( ns );
      rf = true;
    }
    return dp[id][v] = rf;
  }
  int p = S[id]-'0';
  if( nv->ef ) {
    tms.emplace_back( ' ' );
    rf |= solve(id,0);
    tms.pop_back();
  }
  for(char c : nexs[p]){
    int to = c-'a';
    if( nv->v[to] == -1 ) continue;
    tms.emplace_back( c );
    rf |= solve( id+1, nv->v[to] );
    tms.pop_back();
  }
  return dp[id][v] = rf;
}

int main(){
  cin.tie(0);
  ios::sync_with_stdio(false);

  while( cin >> N && N ){
    init();
    for(int i=0;i<N;i++){      
      string w; cin >> w;
      T.push(w);
    }
    cin >> S;
    solve(0,0);
    
    for(string s:res) cout << s << endl;
    cout << "--" << endl;
  }
  
}
