#include <bits/stdc++.h>
int inp() {
  char c = getchar();
  while (c < '0' || c > '9') c = getchar();
  int sum = 0;
  while (c >= '0' && c <= '9') {
    sum = sum * 10 + c - '0';
    c = getchar();
  }
  return sum;
}
int n, m;
int num[12][100010];
struct SEG {
  int l;
  int r;
  int ls[12];
  int rs[12];
  int sum;
  int cnt;
} nul;
int fa[42];
int used[42];
int find(int x) { return fa[x] == x ? x : fa[x] = find(fa[x]); }
SEG operator+(SEG a, SEG b) {
  if (a.l == -1) return b;
  if (b.l == -1) return a;
  SEG ret;
  ret.l = a.l;
  ret.r = b.r;
  ret.sum = a.sum + b.sum;
  for (int i = 1; i <= a.cnt + b.cnt; i++) fa[i] = i;
  memset(used, 0, sizeof(used));
  for (int i = 1; i <= n; i++) {
    b.ls[i] += a.cnt;
    b.rs[i] += a.cnt;
  }
  for (int i = 1; i <= n; i++) {
    if (num[i][a.r] == num[i][b.l]) {
      int f1 = find(a.rs[i]);
      int f2 = find(b.ls[i]);
      if (f1 != f2) {
        ret.sum--;
        fa[f1] = f2;
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    ret.ls[i] = find(a.ls[i]);
    ret.rs[i] = find(b.rs[i]);
  }
  int cnt = 0;
  for (int i = 1; i <= n; i++) {
    if (!used[ret.ls[i]]) used[ret.ls[i]] = ++cnt;
    ret.ls[i] = used[ret.ls[i]];
    if (!used[ret.rs[i]]) used[ret.rs[i]] = ++cnt;
    ret.rs[i] = used[ret.rs[i]];
  }
  ret.cnt = cnt;
  return ret;
}
struct SEG_Tree {
  SEG t[400010];
  void push_up(int cur) { t[cur] = t[((cur) << 1)] + t[((cur) << 1 | 1)]; }
  void build(int cur, int l, int r) {
    t[cur].l = l;
    t[cur].r = r;
    if (l == r) {
      int cnt = 0;
      for (int i = 1; i <= n; i++) {
        if (num[i][l] != num[i - 1][l]) {
          t[cur].sum++;
          t[cur].ls[i] = t[cur].rs[i] = ++cnt;
        } else
          t[cur].ls[i] = t[cur].rs[i] = t[cur].ls[i - 1];
      }
      t[cur].cnt = cnt;
      return;
    }
    int mid = (l + r) >> 1;
    build(((cur) << 1), l, mid);
    build(((cur) << 1 | 1), mid + 1, r);
    push_up(cur);
  }
  SEG query(int cur, int l, int r) {
    if (t[cur].l > r || t[cur].r < l) return nul;
    if (t[cur].l >= l && t[cur].r <= r) return t[cur];
    return query(((cur) << 1), l, r) + query(((cur) << 1 | 1), l, r);
  }
} t;
int main() {
  nul.l = -1;
  n = inp();
  m = inp();
  int q = inp();
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) num[i][j] = inp();
  t.build(1, 1, m);
  while (q--) {
    int l = inp();
    int r = inp();
    printf("%d\n", t.query(1, l, r).sum);
  }
}
