def zamien(S):
	A = 26 * [0]
	for znak in S:
		A[ord(znak) - ord('a')] += 1
	return ''.join([str(i % 2) for i in A])

def mainn():
	A = {}
	wynik = 0
	for i in range(int(input())):
		s = zamien(input())
		wynik += A.get(s, 0)
		for i in range(26):
			wynik += A.get(s[:i] + str((int(s[i]) + 1) % 2) + s[i + 1:], 0)
		A[s] = A.get(s, 0) + 1
	return wynik
print(mainn())
