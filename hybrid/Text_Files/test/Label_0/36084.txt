

import java.io.IOException;
import java.util.Scanner;


public class Main {

    
    
    public static void main(String[] args) throws NumberFormatException, IOException
    {
        
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        
        if(n%2==1)
            System.out.print(-1);
        else
        {
            for(int i = 1; i <= n; i += 2)
                System.out.print((i==1?"":" ")+(i+1)+" "+i);
            System.out.println();
        }
    }


}
