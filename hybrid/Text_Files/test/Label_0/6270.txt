#include <bits/stdc++.h>
using namespace std;
int a[50][50];
int rows0[50];
int rows1[50];
int cols0[50];
int cols1[50];
long long tp[51];
int main() {
  tp[0] = 1;
  for (int i = 1; i <= 50; i++) tp[i] = tp[i - 1] * 2;
  memset(rows0, 0, sizeof rows0);
  memset(cols0, 0, sizeof cols0);
  memset(rows1, 0, sizeof rows1);
  memset(cols1, 0, sizeof cols1);
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) cin >> a[i][j];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (a[i][j] == 0) {
        rows0[i]++;
        cols0[j]++;
      } else {
        rows1[i]++;
        cols1[j]++;
      }
    }
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) ans += tp[rows0[i]] + tp[rows1[i]] - m - 2;
  for (int i = 0; i < m; i++) ans += tp[cols0[i]] + tp[cols1[i]] - n - 2;
  ans += n * m;
  cout << ans << endl;
  return 0;
}
