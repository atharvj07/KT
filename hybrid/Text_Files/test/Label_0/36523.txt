public class Main {
	static long mod = 1000000007;

	public static void main(String[] args) throws IOException, InterruptedException {
		PrintWriter pw = new PrintWriter(System.out);
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt(), m = sc.nextInt();
		DSU dsu = new DSU(n);
		boolean poss = true;
		for (int i = 0; i < m; i++) {
			int u = sc.nextInt()-1, v = sc.nextInt()-1, c = sc.nextInt();
			if (c == 1) {
				poss &= dsu.love(u, v);
			} else
				poss &= dsu.hate(u, v);
		}
		if(!poss) {pw.println(0); pw.close();}
		
		boolean[] visited = new boolean[n];
		int counter = 0;
		for (int i = 0; i < n; i++) {
			if (visited[i] == false && i == dsu.p[i]) {
				counter++;
			//	pw.println(i+" "  +counter);
				if (dsu.enemy[i] != -1)
					visited[dsu.enemy[i]] = true;
			}
		}
		long ans = pow(2, counter - 1);
		pw.println(ans);
		pw.flush();
		pw.close();
	}

	private static long pow(long a, long e) {
		a %= mod;
		long res = 1;
		while (e > 0) {
			if ((e & 1) == 1)
				res = (res * a) % mod;
			a = (a * a) % mod;
			e >>= 1;
		}
		return res;

	}

	static class DSU {
		int[] p, rank, size, enemy;

		public DSU(int N) {
			p = new int[N + 1];
			rank = new int[N + 1];
			size = new int[N + 1];
			enemy = new int[N + 1];
			for (int i = 0; i < N; i++) {
				p[i] = i;
				size[i] = 1;
				enemy[i] = -1;
			}
		}

		public int findSet(int i) {
			return p[i] == i ? i : (p[i] = findSet(p[i]));
		}

		public boolean isSameSet(int i, int j) {
			return findSet(i) == findSet(j);
		}

		public int union(int x, int y) {
			if (x == -1)
				return y;
			if (y == -1)
				return x;

			if (size[x] >= size[y]) {
				size[x] += size[y];
				p[y] = x;
				return x;
			} else {
				p[x] = y;
				size[y] += size[x];
				return y;
			}

		}

		public boolean love(int i, int j) {
			i= findSet(i);
			j= findSet(j);
			if (i==j)
				return true;
			if (i == enemy[j] || j == enemy[i])
				return false;
			int p1 = union(i, j);
			int p2 = union(enemy[i], enemy[j]);
			enemy[p1] = p2;
			if (p2 != -1)
				enemy[p2] = p1;

			return true;
		}

		public boolean hate(int i, int j) {
			i= findSet(i);
			j= findSet(j);
			if (i==j)
				return false;
			if (enemy[i] == enemy[j] && enemy[i] != -1)
				return true;

			int p1 = union(i, enemy[j]);
			int p2 = union(j, enemy[i]);
			enemy[p1] = p2;
			enemy[p2] = p1;

			return true;
		}
	}

	static class Scanner {
		StringTokenizer st;
		BufferedReader br;

		public Scanner(InputStream s) {
			br = new BufferedReader(new InputStreamReader(s));
		}

		public Scanner(String file) throws FileNotFoundException {
			br = new BufferedReader(new FileReader(file));
		}

		public String next() throws IOException {
			while (st == null || !st.hasMoreTokens())
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		public long nextLong() throws IOException {
			return Long.parseLong(next());
		}

		public String nextLine() throws IOException {
			return br.readLine();
		}

		public double nextDouble() throws IOException {
			return Double.parseDouble(next());
		}

		public boolean ready() throws IOException {
			return br.ready();
		}

	}
}
