def dsu(par, i):
	if par[i] == i:
		return i
	return dsu(par, par[i])

def solve():
	(n, d) = map(int, input().split())
	par = [0] * (n + 1)
	ans = [1] * (n + 1)
	ct = 0
	for i in range(1, n + 1):
		par[i] = i
	for i in range(d):
		(x, y) = map(int, input().split())
		p1 = dsu(par, x)
		p2 = dsu(par, y)
		if p1 != p2:
			par[p2] = p1
			ans[p1] += ans[p2]
		else:
			ct += 1
		b = []
		for j in range(1, n + 1):
			if par[j] == j:
				b.append(ans[j])
		b.sort()
		b.reverse()
		s = 0
		for j in range(min(ct + 1, len(b))):
			s += b[j]
		print(s - 1)
	return 0
t = 1
while t > 0:
	t -= 1
	solve()
