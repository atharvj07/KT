import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main{
	
	public static void main(String[] args) throws IOException
	{
		new Main().run();		
	}
	
	int[] starts;
	int[] ends;
	long[] downStarts;
	long[] downEnds;
	long[] intervalLengths;
	long[] suffixSum;
	long H;
	
	public void run() throws IOException
	{
		BufferedReader file = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(file.readLine());
		//always start at beginning of up flow
		//find x coordinate where (non up flow) - upFlow = 0
		//altitude when reaching 
		//end right after up flow
		//consider only intervals with no upflow
		int N = Integer.parseInt(st.nextToken());
		H = Long.parseLong(st.nextToken());
		starts = new int[N];
		ends = new int[N];
		for(int i = 0;i<N;i++)
		{
			st = new StringTokenizer(file.readLine());
			starts[i] = Integer.parseInt(st.nextToken());
			ends[i] = Integer.parseInt(st.nextToken());
		}
		downStarts = new long[N+1];
		downEnds = new long[N+1];
		downStarts[0] = -1000000001l;
		downEnds[0] = starts[0];
		for(int i = 0;i<starts.length - 1;i++)
		{
			downStarts[i+1] = ends[i];
			downEnds[i+1] = starts[i+1];
		}
		downStarts[N] = ends[N-1];
		downEnds[N] = downStarts[N] + 1000000001l;
		long[] intervalLengths = new long[N+1];
		for(int i = 0;i<intervalLengths.length;i++)
			intervalLengths[i] = downEnds[i] - downStarts[i];
		suffixSum = new long[intervalLengths.length+1];
		suffixSum[suffixSum.length - 1] = 0;
		for(int i = suffixSum.length - 2;i>=0;i--)
			suffixSum[i] = intervalLengths[i] + suffixSum[i+1];
		long maxDist = H;
		//System.out.println(Arrays.toString(downEnds));
		//System.out.println(Arrays.toString(intervalLengths));
		for(int i = 0;i<downEnds.length - 1;i++)
		{
			long initialPos = getInitialXPos(i);
			long distanceFlown = (downStarts[i+1] - initialPos + 1);
			//System.out.println("distace flown is "+ distanceFlown);
			maxDist = Math.max(maxDist,distanceFlown);
			//System.out.println();
		}
		System.out.println(maxDist);
	}
	
	public long intervalSum(int start, int end)
	{
		return suffixSum[start] - suffixSum[end+1];
	}
	
	public long getInitialXPos(int intervalEnd)
	{
		//1: find the interval the point is in
		//2: 
		//System.out.println("ending point is " +(downStarts[intervalEnd+1]+1));
		int R = intervalEnd+1;
		int L = -1;
		int M = (L+R)/2;
		while(R-L > 1)
		{
			M = (L+R)/2;
			long sum = intervalSum(M, intervalEnd);
			//System.out.println(M+" "+sum);
			if(sum > H-1)
			{
				L = M;
			}
			else {
				R = M;
			}
		}
		int start = R;//maybe needs changing;
		//System.out.println("interval is "+start+" "+intervalEnd);
		long distBefore = Math.max(0,intervalSum(start,intervalEnd));
		long distLeft = H-1-distBefore;
		//System.out.println("distBefore is "+ distBefore);
		return downEnds[start-1] - distLeft;
	}
	
	public int getIntervalIntersecting(int X)
	{
		int index = Arrays.binarySearch(starts, X);
		if(index < 0)
			index = -index+1;
		return index;
	}
	
}
