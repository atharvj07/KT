
import java.util.Scanner;


public class Solution {

   
    
    public static void main(String[] args) {
       Scanner sc = new Scanner(System.in);
       int n = sc.nextInt();
       int[][] students= new int[3][n];
       int[] indexes = {0,0,0}; 
       int subject ;
       for(int i = 1; i<=n; i++){
           subject = sc.nextInt();
           students[subject - 1][indexes[subject - 1]++] += i;
       }
      int n_teams = Math.min(indexes[0] , Math.min(indexes[1] , indexes[2] ));
      System.out.println(n_teams);
      for(int i = 0; i<n_teams; i++){
          System.out.println(students[0][i] + " " + students[1][i] + " " + students[2][i]);
      }
    }
    
}
