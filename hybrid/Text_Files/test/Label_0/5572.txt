#include <bits/stdc++.h>
using namespace std;
const int N = 2010;
static const int seed = 31;
class strHash {
 private:
  unsigned long long hash[N];
  unsigned long long st[N];

 public:
  strHash(string str) {
    int len = str.size();
    st[0] = 1, hash[0] = 0;
    for (int i = 1; i <= len; i++)
      st[i] = st[i - 1] * seed,
      hash[i] = hash[i - 1] * seed + (str[i - 1] - 'a' + 1);
  }
  strHash(char *str) {
    int len = strlen(str);
    st[0] = 1, hash[0] = 0;
    for (int i = 1; i <= len; i++)
      st[i] = st[i - 1] * seed,
      hash[i] = hash[i - 1] * seed + (str[i - 1] - 'a' + 1);
  }
  unsigned long long Hash(int bg, int end) {
    return hash[end] - hash[bg - 1] * st[end - bg + 1];
  }
  unsigned long long Hash2(int end, int len) {
    return hash[end] - hash[end - len] * st[len];
  }
};
char str[N], s[N], e[N];
unsigned long long ss[N * N];
int main() {
  cin >> str >> s >> e;
  int len = strlen(str);
  strHash A(str);
  int len1 = strlen(s);
  int len2 = strlen(e);
  unsigned long long bg = 0, end = 0;
  for (int i = 0; i < len1; i++) bg = bg * seed + (s[i] - 'a' + 1);
  for (int i = 0; i < len2; i++) end = end * seed + (e[i] - 'a' + 1);
  int cnt = 0;
  for (int i = len1; i <= len; i++) {
    if (A.Hash2(i, len1) != bg) continue;
    for (int j = max(i, len2 - len1 + i); j <= len; j++) {
      if (A.Hash2(j, len2) != end) continue;
      ss[cnt++] = A.Hash(i - len1 + 1, j);
    }
  }
  sort(ss, ss + cnt);
  cnt = unique(ss, ss + cnt) - ss;
  cout << cnt << endl;
  return 0;
}
