import java.util.Scanner;
public class ayhbl 
{
	public static void main(String[] args)
	{
		Scanner in=new Scanner(System.in);
		int n=in.nextInt();
		int a=in.nextInt();
		int[] x=new int[n];
		int i=a-2;
		int j=a,c=0;
		for(int k=0;k<n;k++)
		{
			x[k]=in.nextInt();
		}
		if(x[a-1]==1) c++;
		while(i>=0||j<n)
		{
			if (i>=0 &&j<n&&x[i]==x[j]&&x[i]==1)
			{
					c+=2;i--;j++;
			}
			else if(i>=0&&x[i]==1&&j>=n) {i--;c++;}
			else if(j<n&&x[j]==1&&i<0) {c++;j++;}
			else {i--;j++;}
		}
		System.out.print(c);
	}

}
