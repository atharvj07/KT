import java.util.Scanner;

public class Main {
	private static int[][] ans = new int[100005][105];
	private static final int MOD = 1000000007;

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt(), m = sc.nextInt();
		long[] array = new long[n + 1];
		for (int i = 1; i <= n; i++)
			array[i] = sc.nextInt();

		int[][] comb = new int[100105][105];
		for (int i = 0; i < 100105; i++) {
			comb[i][0] = 1;
			for (int j = 1; j <= Math.min(i, 104); j++) {
				comb[i][j] = (comb[i - 1][j - 1] + comb[i - 1][j]);
				if (comb[i][j] >= MOD)
					comb[i][j] -= MOD;
			}
		}

		for (int i = 0; i < m; i++) {
			int li = sc.nextInt(), ri = sc.nextInt(), ki = sc.nextInt();
			ans[li][ki]++;
			for (int j = ki; j >= 0; j--) {
				ans[ri + 1][j] += MOD - comb[ri - li + ki - j][ki - j];
				if (ans[ri + 1][j] >= MOD)
					ans[ri + 1][j] -= MOD;
			}
		}
		sc.close();

		for (int i = 100; i >= 0; i--)
			for (int j = 1; j <= n; j++) {
				ans[j][i] += ans[j][i + 1];
				if (ans[j][i] >= MOD)
					ans[j][i] -= MOD;
				ans[j][i] += ans[j - 1][i];
				if (ans[j][i] >= MOD)
					ans[j][i] -= MOD;
			}

		for (int i = 1; i <= n; i++) {
			array[i] = (array[i] + ans[i][0]) % MOD;
			System.out.print(array[i] + " ");
		}
	}

}
