x = int(input())
if x < 0:
	y = list(str(x))
	v1 = y[:]
	v1.pop(0)
	v1.pop(len(v1) - 1)
	v2 = y[:]
	v2.pop(0)
	v2.pop(len(v2) - 2)
	if int(''.join(v1)) > int(''.join(v2)):
		if int(''.join(v2)) == 0:
			print(int(''.join(v2)))
		else:
			print('-' + str(''.join(v2)))
	elif int(''.join(v2)) == 0:
		print(int(''.join(v2)))
	else:
		print('-' + str(''.join(v1)))
else:
	print(x)
