import java.util.Scanner;
import java.io.PrintStream;

public class TaskD {

	public static Scanner in = new Scanner(System.in);
	public static PrintStream out = System.out;

	public static void main(String[] args) {
		int n = in.nextInt();
		in.nextLine();
		String[] dash = new String[n];
		for (int i = 0; i < n; i++) {
			dash[i] = in.nextLine();
		}
		char[][] poss = new char[2 * n - 1][2 * n - 1];
		for (int i = 0; i < 2 * n - 1; i++) {
			for (int j = 0; j < 2 * n - 1; j++) {
				if (i == n - 1 && j == n - 1) {
					poss[i][j] = 'o';
				}
				else {
					poss[i][j] = 'x';
				}
			}
		}

		for (int i = 0; i < n; i++) {
			for (int j = 0; j < n; j++) {
				if (dash[i].charAt(j) == 'o') {
					for (int k = 0; k < n; k++) {
						for (int l = 0; l < n; l++) {
							if (dash[k].charAt(l) == '.') {
								poss[n - 1 + k - i][n - 1 + l - j] = '.';
							}
						}
					}
				}
			}
		}

		char[][] newDash = new char[n][n];
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < n; j++) {
				newDash[i][j] = dash[i].charAt(j);
			}
		}

		for (int i = 0; i < n; i++) {
			for (int j = 0; j < n; j++) {
				if (newDash[i][j] == 'o') {
					for (int k = 0; k < n; k++) {
						for (int l = 0; l < n; l++) {
							if (newDash[k][l] == 'x' && poss[n - 1 + k - i][n - 1 + l - j] == 'x') {
								newDash[k][l] = '.';
							}
						}
					}
				}
			}
		}

		boolean flag = true;
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < n; j++) {
				if (newDash[i][j] == 'x') {
					flag = false;
				}
			}
		}

		if (flag) {
			System.out.println("YES");
			for (int i = 0; i < 2 * n - 1; i++) {
				for (int j = 0; j < 2 * n - 1; j++) {
					System.out.print(poss[i][j]);
				}
				System.out.println();
			}
		}
		else {
			System.out.println("NO");
		}
	}

}
