from math import ceil, floor
t = int(input())
for _ in range(t):
	n = int(input())
	c = list(map(int, input().split()))
	x = 0
	ans = c[0] * n + c[1] * n
	i = 2
	a = c[0]
	b = c[1]
	sa = c[0]
	sb = c[1]
	aa = bb = 1
	cura = c[0] * n
	curb = c[1] * n
	for i in range(2, n):
		if x:
			if c[i] < b:
				curb = sb + c[i] * (n - bb)
				ans = min(ans, cura + curb)
				b = c[i]
			else:
				curb = curb - b + c[i]
			bb += 1
			sb += c[i]
		else:
			if c[i] < a:
				cura = sa + c[i] * (n - aa)
				ans = min(ans, cura + curb)
				a = c[i]
			else:
				cura = cura - a + c[i]
			aa += 1
			sa += c[i]
		x ^= 1
	print(ans)
