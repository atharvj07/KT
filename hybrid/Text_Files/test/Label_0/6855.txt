#include <bits/stdc++.h>
using namespace std;
constexpr long long mod = 1000000007;
void IOS() {
  ios::sync_with_stdio(0);
  cin.tie(0);
}
int main() {
  IOS();
  long long n, a, b;
  cin >> n >> a >> b;
  b *= 5;
  long long ans = n;
  for (long long i = n; i >= 0; i -= b) {
    ans = min(ans, i % a);
  }
  cout << ans << '\n';
}
