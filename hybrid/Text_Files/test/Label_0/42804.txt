from math import inf

def diff(s, t):
	ans = 0
	for i in range(len(s)):
		if s[i] != t[i]:
			ans += 1
	return ans
(n, m) = map(int, input().split())
s = input()
t = input()
ans = inf
index = 0
selected_string = t[0:n]
for i in range(m - n + 1):
	sub = t[i:i + n]
	calc_diff = diff(sub, s)
	if calc_diff < ans:
		ans = calc_diff
		index = i
		selected_string = sub
print(ans)
for i in range(n):
	if s[i] != selected_string[i]:
		print(i + 1, end=' ')
print()
