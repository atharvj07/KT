#include <bits/stdc++.h>
using namespace std;

int main()
{
	int N;
	long long C, K;
	cin >> N >> C >> K;
	vector<long long> T( N );
	for( int i = 0; i < N; i++ ) {
		cin >> T[i];
	}

	sort( T.begin(), T.end() );
	int ans = 0;
	int i = 0;
	while( i < N ) {
		long long cur = T[i];
		int j = 1;
		while( j < C && i + j < N && T[i + j] <= cur + K ) j++;
		ans++;
		i += j;
	}

	cout << ans << endl;
}
