#include <bits/stdc++.h>
using namespace std;

map<int,int>P{
{ 2,0},{ 3,0},{ 5,0},{ 7,0},{11,0},
{13,0},{17,0},{19,0},{23,0},{29,0},
{31,0},{37,0},{41,0},{43,0},{47,0},
{53,0},{59,0},{61,0},{67,0},{71,0},
{73,0},{79,0},{83,0},{89,0},{97,0},
};

int f(int x){
	x--;
	int c = 0;
	for(auto e:P) if(e.second>=x) c++;
	return c;
}

int main(){
	int N; cin>>N;

	for(int i=2; i<=N; ++i)
		for(auto e:P){
			int p = e.first;
			if(p > i) break;
			for(int n=i; n%p==0; n/=p) P[p]++;
		}

	int ans = f(75); // {f(75)}_C_{1}
	ans += f(25) * (f(3)-1); // {f(25)}_C_{1} * {f(3)-1}_C_{1}
	ans += f(15) * (f(5)-1); // {f(15)}_C_{1} * {f(5)-1}_C_{1}
	ans += f(5)*(f(5)-1)/2 * (f(3)-2); // {f(5)}_C_{2} * {f(3)-2}_C_{1}
	cout << ans <<endl;
}
