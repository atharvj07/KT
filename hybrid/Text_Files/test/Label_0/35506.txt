boolean dfs(int parent, int i, int root, ArrayList< ArrayList<Integer> > a, int[][] d){
	if ( d[root][parent] + d[parent][i] != d[root][i] ) return false;
	for(Integer k: a.get(i)){
		int j = k.intValue();
		if (j!=parent && j!=i)
			if (!dfs(i,j,root,a,d)) return false;
	}
	return true;
	

}

private boolean solve(){
	int n = in.nextInt();
	int[][] d = new int[n][n];
	for(int i = 0; i < n; ++i)
	for(int j = 0; j < n; ++j) 
		d[i][j] = in.nextInt();
	
	for(int i = 0; i < n; ++i)
	for(int j = 0; j < n; ++j){
		if (i==j && d[i][j]!=0) return false;
		if (i!=j && d[i][j]==0) return false;
		if (d[i][j] != d[j][i]) return false;
	}
	
	//BuildTree
	int[] dist = new int[n], v = new int[n];
	for(int i = 0; i < n; ++i){
		dist[i] = d[0][i];
	}
	
	ArrayList<ArrayList<Integer>> a = new ArrayList< ArrayList<Integer> >();
	for(int i = 0; i < n; ++i) a.add(new ArrayList<Integer>());
	while (true){
		int mark = -1;
		for(int i = 0; i < n; ++i)
			if (v[i]>=0 && (mark<0 || dist[i] < dist[mark]))
				mark = i;
		
		if (mark == -1) break;
		a.get(mark).add( v[mark] );
		a.get(v[mark]).add(mark);
		v[mark] = -1;
		for(int i = 0; i < n; ++i)
			if (v[i] >= 0 && dist[i] >= dist[mark] + d[mark][i]){
				dist[i] = dist[mark] + d[mark][i];
				v[i] = mark;
			}
	}
	
	for(int i = 0; i < n; ++i)
		if (!dfs(i, i, i, a, d)) return false;
	return true;
	
	
}
	
public static void main (String[] args) throws java.lang.Exception {		
	long startTime = System.currentTimeMillis();

	out = new PrintWriter(System.out);
	boolean result = new Main().solve();	
	if (result) out.println("YES");
	else out.println("NO");
	//out.println((String.format("%.2f",(double)(System.currentTimeMillis()-startTime)/1000)));
	out.close();
}
static PrintWriter out;
static class in {
	static BufferedReader reader = new BufferedReader( new InputStreamReader(System.in) ) ;
	static StringTokenizer tokenizer = new StringTokenizer("");				
	static String next() {		
		while ( !tokenizer.hasMoreTokens() )
			try { tokenizer = new StringTokenizer( reader.readLine() ); }
			catch (IOException e){
				throw new RuntimeException(e);			
			}		
		return tokenizer.nextToken();
	}
	static int nextInt() { return Integer.parseInt( next() ); }			
}
