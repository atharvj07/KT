#include <bits/stdc++.h>
using namespace std;
const int N = 3005;
const long long mod = 1e9 + 7;
using namespace std;
int n;
char novi[N];
char s[N], t[N];
vector<int> v;
void op(char *s, int n, int pos) {
  for (int i = pos; i <= n; i++) novi[n - i + 1] = s[i];
  for (int i = 1; i < pos; i++) novi[i + (n - pos + 1)] = s[i];
  for (int i = 1; i <= n; i++) s[i] = novi[i];
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr), cout.tie(nullptr);
  cin >> n >> (s + 1) >> (t + 1);
  for (auto(i) = (1); (i) <= (n); (i)++) {
    int ok = 0;
    for (auto(j) = (1); (j) <= (n - i + 1); (j)++)
      if (t[i] == s[j]) {
        ok = j;
        break;
      }
    if (!ok) {
      return cout << -1 << endl, 0;
    }
    if (ok + 1 <= n) {
      v.push_back(n - ok);
      op(s, n, ok + 1);
    }
    v.push_back(1);
    op(s, n, n);
    v.push_back(n);
    op(s, n, 1);
  }
  cout << v.size() << endl;
  for (auto i : v) cout << i << " ";
  return cout << endl, 0;
}
