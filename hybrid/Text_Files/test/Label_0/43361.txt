n = int(input())
lighthouses = [0] * (1000000 + 2)
count = [0] * (1000000 + 2)
for i in range(n):
	(a, b) = map(int, input().split())
	lighthouses[a] = b
if lighthouses[0] > 0:
	count[0] = 1
for i in range(1, 1000000 + 1):
	if lighthouses[i] == 0:
		count[i] = count[i - 1]
	elif lighthouses[i] >= i:
		count[i] = 1
	else:
		count[i] = count[i - lighthouses[i] - 1] + 1
print(n - max(count))
