#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  int n;
  cin >> n;
  long long a[n], b[n];
  int c(0);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    b[i] = a[i];
  }
  sort(b, b + n);
  for (int i = 0; i < n; i++) {
    if (a[i] != b[i]) c++;
  }
  if (c == 0 || c == 2)
    cout << "YES";
  else
    cout << "NO";
  return 0;
}
