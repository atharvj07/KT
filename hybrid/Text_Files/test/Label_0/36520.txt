import java.util.Scanner;

public class taxi {
	private static Scanner scan;
	
	public static void main(String[] args) {
		scan = new Scanner(System.in);
		int n = scan.nextInt();
		int four = 0;
		int three = 0;
		int two = 0;
		int one = 0;
		int ans = 0;
		for(int i = 0; i < n; i++){
			int num = scan.nextInt();
			if(num == 4)
				four++;
			else if(num == 3)
				three++;
			else if(num == 2)
				two++;
			else if(num == 1)
				one++;
		}
		
		ans += four;
		if(three > one){
			ans += three;
			if(two % 2 == 0){
				ans += two / 2;
			}else{
				ans += two / 2 + 1;
			}
		}else if(three < one){
			ans += three;
			one -= three;
			if(two % 2 == 0){
				ans += two / 2;
				if(one % 4 == 0){
					ans += one / 4;
				}else{
					ans += one / 4 + 1;
				}
			}else{
				ans += two / 2 + 1;
				if(one > 2){
					one -= 2;
					if(one % 4 == 0){
						ans += one / 4;
					}else{
						ans += one / 4 + 1;
					}		
				}
			}
		}else{
			ans += three;
			if(two % 2 == 0){
				ans += two / 2;
			}else{
				ans += two / 2 + 1;
			}
		}
		System.out.print(ans);
	}
	
}
