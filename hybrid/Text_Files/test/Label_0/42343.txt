n = int(input())
L = [list(map(int, input().split())) for _ in range(n)]

def trouver(L):
	n = len(L)
	for i in range(n):
		for j in range(n):
			if L[i][j] == 0:
				return (i, j)
(x, y) = trouver(L)
if n == 1:
	print(42)
else:
	ligne = sum(L[(x + 1) % n])
	L[x][y] = ligne - sum(L[x])
	somme = sum(L[x])
	ok = 1
	for i in range(n):
		if sum(L[i]) != somme:
			ok = 0
			break
	for j in range(n):
		if sum([L[i][j] for i in range(n)]) != somme:
			ok = 0
			break
	if sum([L[i][i] for i in range(n)]) != somme:
		ok = 0
	if sum([L[i][n - 1 - i] for i in range(n)]) != somme:
		ok = 0
	if ok and L[x][y] > 0:
		print(L[x][y])
	else:
		print(-1)
