import java.util.*;
import java.lang.Long;

public class main {

	public static void main(String[] args) {

		Scanner input = new Scanner(System.in);

		long lower = (-1) * 1000000010L, upper = 1000000010, n, num;
		String ans;
		String equation;

		n = input.nextInt();
		input.nextLine();

		for (int i = 0; i < n; i++) {
			equation = input.next();
			num = input.nextInt();
			ans = input.next();
			input.nextLine();

			if (equation.equals(">")) {
				if (ans.equals("Y")) {
					if (num >= lower) {
						lower = num + 1;
					}

				} else {
					if (num < upper) {
						upper = num;
					}

				}

			} else if (equation.equals("<")) {
				if (ans.equals("Y")) {
					if (num <= upper) {
						upper = num - 1;
					}

				} else {
					if (num > lower) {
						lower = num;
					}

				}
			} else if (equation.equals(">=")) {
				if (ans.equals("Y")) {
					if (num > lower) {
						lower = num;
					}

				} else {
					if (num < upper) {
						upper = num - 1;
					}

				}
			} else if (equation.equals("<=")) {
				if (ans.equals("Y")) {
					if (num < upper) {
						upper = num;
					}

				} else {
					if (num > lower) {
						lower = num + 1;
					}
				}
			}

		}
		if (upper < lower) {
			System.out.print("Impossible");
		} else {
			System.out.print((lower + upper) / 2);
		}

		input.close();
	}

}
