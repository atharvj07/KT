max2 = lambda x, y: x if x > y else y
min2 = lambda x, y: x if x < y else y

def bisect_rev(a, v, l):
	r = len(a)
	while l + 1 < r:
		m = (l + r) // 2
		if a[m] <= v:
			r = m
		else:
			l = m
	return l
import sys
from bisect import bisect
read = sys.stdin.buffer.read
readline = sys.stdin.buffer.readline

def solve(A):
	N = len(A)
	pairs = []
	stack_val = [10 ** 9 + 1, A[0]]
	stack_ind = [-1, 0]
	l = 1
	for (i, a) in enumerate(A + A):
		if a * 2 < stack_val[l]:
			k = bisect_rev(stack_val, a * 2, l)
			j = stack_ind[k]
			if j >= N:
				break
			l = k + 1
			pairs.append((j, i))
		while stack_val[-1] <= a:
			stack_val.pop()
			stack_ind.pop()
		stack_val.append(a)
		stack_ind.append(i)
		l = min2(l, len(stack_ind) - 1)
	if not pairs:
		return [-1] * N
	pairs.append((pairs[0][0] + N, pairs[0][1] + N))
	res = [0] * N
	pi = 0
	(ns, nt) = pairs[pi]
	for i in range(N):
		res[i] = nt - i
		if i == ns:
			pi += 1
			(ns, nt) = pairs[pi]
	return res

def naive(A):
	N = len(A)
	res = [0] * N
	for i in range(N):
		m = A[i]
		for k in range(N * 2):
			if 2 * A[(i + k) % N] < m:
				res[i] = k
				break
			m = max2(m, A[(i + k) % N])
		else:
			res[i] = -1
	return res
