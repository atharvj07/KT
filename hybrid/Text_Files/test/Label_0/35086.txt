    void solve(int TC) throws Exception{

        int N = ni(), K = ni();
        int[] A = new int[2+N], B = new int[2+K];
        for(int i = 1; i<= N; i++)A[i] = ni();
        for(int i = 1; i<= K; i++)B[i] = ni();
        A[0] = -((int)1e9+N+1);
        A[N+1] = ((int)1e9+N+1);
        B[0] = 0;
        B[K+1] = N+1;
        Arrays.sort(B);
        boolean good = true;
        for(int i = 1; i<= K+1; i++)
            good &= A[B[i]]-A[B[i-1]]-1 >= B[i]-B[i-1]-1;
        if(good){
            int ans = 0;
            for(int i = 1; i<= K+1; i++)ans += B[i]-B[i-1]+1 - lis(A, B[i-1], B[i]);
            pn(ans);
        }else pn(-1);
    }
    int lis(int[] A, int l, int r){
        for(int i = l; i<= r; i++)A[i] -= i;
        ConvexSet set = new ConvexSet();
        int max = 0;
        for(int i = l; i <= r; i++){
            Long key = set.floorKey(A[i]);
            long length = 1;
            if(key != null)length = 1+set.get(key);
            else if(i > l)continue;
            set.put(A[i], length);
            max = Math.max(max, (int)length);
        }
        int ans = (int)((long)set.get(A[r]));
        for(int i = l; i<= r; i++)A[i] += i;
        return ans;
    }
    class ConvexSet{
        TreeMap<Long, Long> map;
        public ConvexSet(){
            map = new TreeMap<>();
        }
        void put(long key, long val){
            Map.Entry<Long, Long> e;
            if((e = map.floorEntry(key))!=null && e.getValue()>=val)return;
            while(!map.isEmpty() && (e=  map.ceilingEntry(key))!=null && e.getValue()<=val)map.remove(e.getKey());
            map.put(key, val);
        }
        Long get(long k){return map.get(k);}
        Long floorKey(long k){return map.floorKey(k);}
    }

    int[][] make(int n, int e, int[] from, int[] to, boolean f){
        int[][] g = new int[n][];int[]cnt = new int[n];
        for(int i = 0; i< e; i++){
            cnt[from[i]]++;
            if(f)cnt[to[i]]++;
        }
        for(int i = 0; i< n; i++)g[i] = new int[cnt[i]];
        for(int i = 0; i< e; i++){
            g[from[i]][--cnt[from[i]]] = to[i];
            if(f)g[to[i]][--cnt[to[i]]] = from[i];
        }
        return g;
    }

    class FastReader{
        BufferedReader br;
        StringTokenizer st;
        public FastReader(){
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        public FastReader(String s) throws Exception{
            br = new BufferedReader(new FileReader(s));
        }

        String next() throws Exception{
            while (st == null || !st.hasMoreElements()){
                try{
                    st = new StringTokenizer(br.readLine());
                }catch (IOException  e){
                    throw new Exception(e.toString());
                }
            }
            return st.nextToken();
        }

        String nextLine() throws Exception{
            String str;
            try{
                str = br.readLine();
            }catch (IOException e){
                throw new Exception(e.toString());
            }
            return str;
        }
    }
