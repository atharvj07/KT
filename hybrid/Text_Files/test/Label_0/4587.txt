#include <bits/stdc++.h>
using namespace std;
int n, m, k, x, y, ln, cnt, flag, a[500006], l[500006], f[500006],
    to[2 * 500006], ne[2 * 500006], h[500006], d[500006], p[500006][21];
void add(int x, int y) {
  to[++cnt] = y;
  ne[cnt] = h[x];
  h[x] = cnt;
}
void dfs(int x, int fa) {
  p[x][0] = fa;
  d[x] = d[fa] + 1;
  f[x] = 1;
  if (d[x] * k >= n) flag = x;
  int t = 1;
  for (int i = 1; i <= 20; i++) p[x][i] = p[p[x][i - 1]][i - 1];
  for (int i = h[x]; i; i = ne[i])
    if (!f[to[i]]) t = 0, dfs(to[i], x);
  if (t) l[++ln] = x;
}
int lca(int x, int y) {
  if (d[x] < d[y]) swap(x, y);
  for (int i = 20; i >= 0; i--)
    if (d[p[x][i]] >= d[y]) x = p[x][i];
  if (x == y) return x;
  for (int i = 20; i >= 0; i--)
    if (d[p[x][i]] != d[p[y][i]]) x = p[x][i], y = p[y][i];
  return p[x][0];
}
int dis(int x, int y) { return d[x] + d[y] - 2 * d[lca(x, y)]; }
int main() {
  scanf("%d%d%d", &n, &m, &k);
  for (int i = 1; i <= m; i++) scanf("%d%d", &x, &y), add(x, y), add(y, x);
  d[0] = -1;
  dfs(1, 0);
  if (flag) {
    puts("PATH");
    printf("%d\n", 1 + d[flag]);
    while (flag != 1) printf("%d ", flag), flag = p[flag][0];
    puts("1");
    return 0;
  } else {
    puts("CYCLES");
    for (int i = 1; i <= ln; i++) {
      if (k == 0) return 0;
      int w = l[i], r[2], rn = 0;
      for (int j = h[w]; j; j = ne[j])
        if (to[j] != p[w][0] && rn < 2) r[rn++] = to[j];
      if ((dis(r[1], w) + 1) % 3 != 0) {
        k--;
        int q = 0;
        printf("%d\n", dis(r[1], w) + 1);
        while (w != lca(r[1], w)) a[++q] = w, w = p[w][0];
        a[++q] = w;
        int v = r[1], t = q;
        while (v != w) a[++q] = v, v = p[v][0];
        for (int j = 1; j <= t; j++) printf("%d ", a[j]);
        for (int j = q; j > t; j--) printf("%d ", a[j]);
        puts("");
        continue;
      }
      if ((dis(r[0], w) + 1) % 3 != 0) {
        k--;
        int q = 0;
        printf("%d\n", dis(r[0], w) + 1);
        while (w != lca(r[0], w)) a[++q] = w, w = p[w][0];
        a[++q] = w;
        int v = r[0], t = q;
        while (v != w) a[++q] = v, v = p[v][0];
        for (int j = 1; j <= t; j++) printf("%d ", a[j]);
        for (int j = q; j > t; j--) printf("%d ", a[j]);
        puts("");
        continue;
      }
      if ((dis(r[1], r[0]) + 2) % 3 != 0) {
        k--;
        int q = 0;
        printf("%d\n", dis(r[1], r[0]) + 2);
        a[q = 1] = w;
        int u = r[1], v = r[0];
        while (u != lca(u, v)) a[++q] = u, u = p[u][0];
        a[++q] = u;
        int t = q;
        while (v != u) a[++q] = v, v = p[v][0];
        for (int j = 1; j <= t; j++) printf("%d ", a[j]);
        for (int j = q; j > t; j--) printf("%d ", a[j]);
        puts("");
      }
    }
  }
}
