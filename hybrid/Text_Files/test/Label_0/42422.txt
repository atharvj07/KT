(n, m, h) = list(map(int, input().split()))
front = list(map(int, input().split()))
left = list(map(int, input().split()))
top = [list(map(int, input().split())) for i in range(0, n)]
for i in range(0, n):
	for j in range(0, m):
		if top[i][j]:
			top[i][j] = min(left[i], front[j])
		print(top[i][j], end=' ')
	print()
