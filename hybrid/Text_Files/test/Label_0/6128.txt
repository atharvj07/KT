#include <bits/stdc++.h>
using namespace std;
const long long N = 4e5 + 7;
const long long Mod = 1e9 + 7;
const long long Inf = 1e18 + 7;
void Kinishinaide() {
  ios_base::sync_with_stdio(0);
  cin.tie(0), cout.tie(0);
}
int M, q, top[N], A[N];
vector<int> g[N];
int timer = false;
int tin[N], tout[N];
long long t[N * 4];
long long Add[N * 4];
void Dfs(int v, int p) {
  tin[v] = ++timer;
  top[timer] = v;
  for (int to : g[v]) {
    if (to == p) continue;
    Dfs(to, v);
  }
  tout[v] = timer;
}
void Build(int v = 1, int tl = 1, int tr = M) {
  if (tl == tr) {
    t[v] = (1LL << A[top[tl]]);
    return;
  }
  int tm = (tl + tr) >> 1;
  Build(v + v, tl, tm);
  Build(v + v + 1, tm + 1, tr);
  t[v] = (t[v + v] | t[v + v + 1]);
}
void Push(int v, int tl, int tr) {
  if (Add[v]) {
    t[v] = (1LL << Add[v]);
    if (tl != tr) Add[v + v] = Add[v + v + 1] = Add[v];
    Add[v] = false;
  }
}
void Upd(int L, int R, int val, int v = 1, int tl = 1, int tr = M) {
  Push(v, tl, tr);
  if (L <= tl && tr <= R) {
    Add[v] = val;
    Push(v, tl, tr);
    return;
  }
  if (tr < L || tl > R) return;
  int tm = (tl + tr) >> 1;
  Upd(L, R, val, v + v, tl, tm);
  Upd(L, R, val, v + v + 1, tm + 1, tr);
  t[v] = (t[v + v] | t[v + v + 1]);
}
long long Get(int L, int R, int v = 1, int tl = 1, int tr = M) {
  Push(v, tl, tr);
  if (L <= tl && tr <= R) return t[v];
  if (tr < L || tl > R) return 0LL;
  int tm = (tl + tr) >> 1;
  return (Get(L, R, v + v, tl, tm) | Get(L, R, v + v + 1, tm + 1, tr));
}
int main() {
  Kinishinaide();
  cin >> M >> q;
  for (int i = 1; i <= M; i++) {
    cin >> A[i];
  }
  for (int i = 1, u, v; i < M; i++) {
    cin >> u >> v;
    g[u].push_back(v);
    g[v].push_back(u);
  }
  Dfs(1, -1);
  Build();
  while (q--) {
    int type;
    cin >> type;
    if (type == 1) {
      int v, c;
      cin >> v >> c;
      Upd(tin[v], tout[v], c);
    } else {
      int v;
      cin >> v;
      cout << __builtin_popcountll(Get(tin[v], tout[v])) << '\n';
    }
  }
}
