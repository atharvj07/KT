#include <bits/stdc++.h>
using namespace std;
long long n, k, d;
queue<long long> que;
int a[400001];
int main() {
  cin >> n >> d >> k;
  if (d > n - 1)
    puts("NO");
  else {
    long long t = 0, tt = 1;
    int dp;
    for (int i = 2; i <= (d + 1) / 2; i++) {
      dp = i - 1;
      tt *= k - 1;
      t += tt - 1;
      if (t > n) break;
    }
    t *= 2;
    if (d % 2 == 0) t += tt * (k - 1) - 1;
    t += d + 1;
    if (t < n)
      puts("NO");
    else {
      puts("YES");
      for (int i = 1; i < d + 1; i++) printf("%d %d\n", i, i + 1);
      int t = d + 2;
      if (n) {
        for (int i = 2; i <= (d + 1) / 2; i++) {
          while (que.size()) que.pop();
          int dp = i - 1, f = 0;
          while (dp--) {
            int j;
            if (f) {
              int nn = que.size();
              while (nn--) {
                int tt = que.front();
                que.pop();
                for (j = t; j < t + k - 1; j++) {
                  if (j > n) break;
                  printf("%d %d\n", tt, j);
                  que.push(j);
                }
                t = j;
                if (t > n) break;
              }
              continue;
            }
            if (!f)
              for (j = t; j < t + k - 2; j++) {
                if (j > n) break;
                f = 1;
                que.push(j);
                printf("%d %d\n", i, j);
              }
            t = j;
            if (t > n) break;
          }
        }
        if (t <= n) {
          for (int i = d; i > (d + 1) / 2; i--) {
            while (que.size()) que.pop();
            int dp = d - i + 1, f = 0;
            while (dp--) {
              int j;
              if (f) {
                int nn = que.size();
                while (nn--) {
                  int tt = que.front();
                  que.pop();
                  for (j = t; j < t + k - 1; j++) {
                    if (j > n) break;
                    que.push(j);
                    printf("%d %d\n", tt, j);
                  }
                  t = j;
                  if (t > n) break;
                }
              }
              if (!f)
                for (j = t; j < t + k - 2; j++) {
                  if (j > n) break;
                  f = 1;
                  que.push(j);
                  printf("%d %d\n", i, j);
                }
              t = j;
              if (t > n) break;
            }
          }
        }
      }
    }
  }
  return 0;
}
