#include <bits/stdc++.h>
using namespace std;
const int N = 1000005;
int mn[N], v[5], top, x;
long long qq[105], ans, n, m;
pair<int, int> q[105];
void init() {
  for (int i = (int)(2); i <= (int)(N - 1); i++)
    if (!mn[i])
      for (int j = (int)(1); j <= (int)((N - 1) / i); j++)
        if (!mn[i * j]) mn[i * j] = i;
}
void Inputv(long long &n) {
  int v1, v2, v3;
  scanf("%d%d%d", &v1, &v2, &v3);
  n = 1ll * v1 * v2 * v3;
}
void Inputvv() {
  top = 0;
  for (int T = (int)(1); T <= (int)(3); T++) {
    int v;
    scanf("%d", &v);
    for (; v != 1;) {
      q[++top] = pair<int, int>(mn[v], 1);
      v /= mn[v];
    }
  }
  q[++top] = pair<int, int>(2, 1);
  sort(q + 1, q + top + 1);
  int tmp = top;
  top = 1;
  for (int i = (int)(2); i <= (int)(tmp); i++)
    if (q[i].first != q[i - 1].first)
      q[++top] = q[i];
    else
      ++q[top].second;
}
void count1(int d, long long v) {
  if (d == top + 1) {
    ans += (v <= n);
    return;
  }
  for (int i = (int)(0); i <= (int)(q[d].second); i++) {
    count1(d + 1, v);
    v *= q[d].first;
  }
}
void count2(int d, long long v, long long fl) {
  if (d == *qq + 1) {
    ans += fl * v;
    return;
  }
  count2(d + 1, v, fl);
  count2(d + 1, v / qq[d], -fl);
}
long long gcd(long long x, long long y) { return y ? gcd(y, x % y) : x; }
void solve() {
  ans = 0;
  *qq = 0;
  n = 1;
  m = 1;
  for (int i = (int)(1); i <= (int)(3); i++) scanf("%d", &v[i]), n *= v[i];
  for (int i = (int)(1); i <= (int)(3); i++) scanf("%d", &x), m *= x;
  Inputvv();
  for (int i = (int)(1); i <= (int)(top); i++) {
    long long rem = 1;
    for (int j = (int)(1); j <= (int)(q[i].second + 1); j++) rem *= q[i].first;
    for (int j = (int)(1); j <= (int)(3); j++) {
      long long G = gcd(v[j], rem);
      v[j] /= G;
      rem /= G;
    }
    if (rem == 1) {
      qq[++*qq] = 1;
      for (int j = (int)(1); j <= (int)(q[i].second + 1); j++)
        qq[*qq] *= q[i].first;
      for (int j = (int)(1); j <= (int)(3); j++)
        for (; v[j] % q[i].first == 0; v[j] /= q[i].first)
          ;
    }
  }
  for (int i = (int)(1); i <= (int)(3); i++)
    for (; v[i] != 1; v[i] /= mn[v[i]]) qq[++*qq] = mn[v[i]];
  sort(qq + 1, qq + *qq + 1);
  *qq = unique(qq + 1, qq + *qq + 1) - qq - 1;
  count1(1, 1);
  count2(1, m, 1);
  printf("%lld\n", ans);
}
int main() {
  init();
  int T;
  scanf("%d", &T);
  while (T--) solve();
}
