n = int(input())
color = ['purple', 'green', 'blue', 'orange', 'red', 'yellow']
gem = ['Power', 'Time', 'Space', 'Soul', 'Reality', 'Mind']
for i in range(n):
	s = input()
	if s == 'purple':
		gem.remove('Power')
	elif s == 'green':
		gem.remove('Time')
	elif s == 'blue':
		gem.remove('Space')
	elif s == 'orange':
		gem.remove('Soul')
	elif s == 'red':
		gem.remove('Reality')
	elif s == 'yellow':
		gem.remove('Mind')
x = len(gem)
print(x)
for i in range(x):
	print(gem[i])
