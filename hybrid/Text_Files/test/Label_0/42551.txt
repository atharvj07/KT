import sys
import math

def input():
	return sys.stdin.readline().strip()

def iinput():
	return [int(x) for x in sys.stdin.readline().split()]

def main():
	(n, m) = iinput()
	s = dict()
	z = []
	for i in range(n):
		a = input()
		b = a[::-1]
		if a not in s and b not in s:
			s[a] = 1
		elif a in s:
			s[a] += 1
		elif b in s:
			s[b] += 1
		if b == a:
			z += [a]
	count = 0
	data = list(s.keys())
	ans = ''
	for i in data:
		c = s[i] // 2
		ans += i * c
		count += c * 2
		s[i] %= 2
	if count == 0 and len(z) == 0:
		print(0)
	else:
		ans1 = ans[::-1]
		for i in range(len(z)):
			if s[z[i]] >= 1:
				ans += z[i]
				count += 1
				break
		print(m * count, ans + ans1, sep='\n')
	return
for ______ in range(1):
	main()
