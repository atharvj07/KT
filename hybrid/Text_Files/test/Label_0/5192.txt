#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:256000000")
using namespace std;
const double PI = acos(-1.0);
const int INF = 1000000000;
const int MOD = 1000000007;
const int M = INF;
int main() {
  double TIME_START = clock();
  int n, m, k;
  cin >> m >> n >> k;
  vector<vector<char> > field(m + 2, vector<char>(n + 2, '#'));
  int cx[30], cy[30];
  for (int i = 1; i <= m; i++)
    for (int j = 1; j <= n; j++) {
      cin >> field[i][j];
      if (!isdigit(field[i][j])) {
        cx[field[i][j] - 'a'] = i;
        cy[field[i][j] - 'a'] = j;
      }
    }
  int is, js, ie, je;
  string s;
  cin >> is >> js >> s >> ie >> je;
  int fi = is, fj = js;
  s += '1';
  for (int i = 0; i < s.size(); i++) {
    int ti, tj;
    if (s[i] != '1') {
      ti = cx[s[i] - 'a'];
      tj = cy[s[i] - 'a'];
    } else {
      ti = ie;
      tj = je;
    }
    while (fi != ti || fj != tj) {
      if (isdigit(field[fi][fj]))
        k -= (field[fi][fj] - '0');
      else
        k--;
      if (k < 0 || (s[i] == '1' && fi == ie && fj == je)) {
        cout << fi << " " << fj << endl;
        return 0;
      }
      if (fi < ti)
        fi++;
      else if (fi > ti)
        fi--;
      if (fj < tj)
        fj++;
      else if (fj > tj)
        fj--;
    }
    if (s[i] == '1' && s[i] == '1' && fi == ie && fj == je) {
      cout << fi << " " << fj << endl;
      return 0;
    }
  }
  fprintf(stderr, "\n\n%.15lf\n\n",
          (double)(clock() - TIME_START) / CLOCKS_PER_SEC);
  return 0;
}
