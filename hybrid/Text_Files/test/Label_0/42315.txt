for t in range(int(input())):
	(N, M) = tuple(map(int, input().split()))
	AA = [input() for n in range(N)]
	t = None
	flag = False
	for n in range(N):
		for m in range(M):
			if AA[n][m] == '.':
				continue
			p = ((AA[n][m] == 'W') - n - m) % 2
			if t == None:
				t = p
			elif t != p:
				flag = True
				break
		if flag:
			break
	if flag:
		print('NO')
	else:
		print('YES')
		if t == None:
			t = 0
		for n in range(N):
			R = ''
			for m in range(M):
				if (t + n + m) % 2:
					R += 'W'
				else:
					R += 'R'
			print(R)
