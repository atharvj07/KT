MODULE = 10 ** 9 + 7

def r(n, k):
	a = [1 for _ in range(n + 1)]
	new_a = [0 for _ in range(n + 1)]
	cur_k = 1
	while cur_k < k:
		new_a[0] = 1
		for i in range(1, n + 1):
			new_a[i] = (new_a[i - 1] + a[n - i]) % MODULE
		cur_k += 1
		(a, new_a) = (new_a, a)
	return a[-1]

def main():
	t = int(input())
	for _ in range(t):
		(n, k) = map(int, input().split())
		print(r(n, k))
main()
