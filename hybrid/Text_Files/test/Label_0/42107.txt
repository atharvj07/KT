from collections import deque
n = int(input())
values = []
for i in range(n):
	(a, b) = list(map(int, input().split()))
	values.append([a, b])
values.sort(key=lambda x: x[1], reverse=True)
values = deque(values)
total = 0
cost = 0
while len(values) > 1:
	if total >= values[-1][1]:
		total += values[-1][0]
		cost += values[-1][0]
		values.pop()
	elif values[0][0] > values[-1][1] - total:
		values[0][0] -= values[-1][1] - total
		cost = cost + (values[-1][1] - total) * 2
		total += values[-1][1] - total
	elif values[0][0] < values[-1][1] - total:
		cost = cost + values[0][0] * 2
		total += values[0][0]
		values.popleft()
	elif values[0][0] == values[-1][1] - total:
		cost = cost + values[0][0] * 2
		total += values[0][0]
		values.popleft()
if total + values[0][0] < values[0][1]:
	cost += values[0][0] * 2
elif total > values[0][1]:
	cost += values[0][0]
else:
	cost = cost + (values[0][1] - total) * 2 + values[0][0] - (values[0][1] - total)
print(cost)
