#include<bits/stdc++.h>

#define REP(i,s,n) for(int i=s;i<n;++i)
#define rep(i,n) REP(i,0,n)

using namespace std;

#define MAX_V 100000

typedef long long ll;

int V,E,K,S;
ll P,Q,mini[MAX_V];
bool zombi[MAX_V];
vector<int> G[MAX_V];
bool danger[MAX_V];

struct Data2 {
  int cur, TL;
  bool operator < ( const Data2 &data ) const {
    if( TL != data.TL ) return TL < data.TL;
    return cur < data.cur;
  }
};

struct Data {
  int cur;
  ll cost;
  bool operator < ( const Data &data ) const {
    if( cost != data.cost ) return cost > data.cost;
    return cur < data.cur;
  }
};

void preprocess() {
  priority_queue<Data2> Q;
  rep(i,V) if( zombi[i] ) {
    Q.push((Data2){i,S});
    danger[i] = true;
  }
  while( !Q.empty() ) {
    Data2 data = Q.top(); Q.pop();
    if( data.TL <= 0 ) continue;
    rep(i,(int)G[data.cur].size()) {
      int next = G[data.cur][i];
      if( danger[next] ) continue;
      danger[next] = true;
      Q.push((Data2){next,data.TL-1});
    }
  }
  zombi[0] = zombi[V-1] = false;
}

void compute() {
  preprocess();
  rep(i,V) mini[i] = LLONG_MAX;
  mini[0] = 0;
  priority_queue<Data> pq;
  pq.push((Data){0,0});
  while( !pq.empty() ) {
    Data data = pq.top(); pq.pop();
    if( data.cur == V-1 ) {
      cout << data.cost << endl;
      return;
    }
    rep(i,(int)G[data.cur].size()) {
      int next = G[data.cur][i];
      if( zombi[next] ) continue;
      ll cost =  data.cost + (danger[next]?Q:P);
      if( next == V-1 ) cost = data.cost;
      if( mini[next] > cost ) {
	mini[next] = cost;
	pq.push((Data){next,mini[next]});
      }
    }
  }
  assert(false);
}

int main() {
  scanf("%d %d %d %d",&V,&E,&K,&S);
  cin >> P >> Q;
  int tmp,tmp2;
  rep(i,K) {
    scanf("%d",&tmp);
    --tmp;
    zombi[tmp] = true;
  }
  rep(i,E) {
    scanf("%d %d",&tmp,&tmp2);
    --tmp, --tmp2;
    G[tmp].push_back(tmp2);
    G[tmp2].push_back(tmp);
  }
  compute();
  return 0;
}
