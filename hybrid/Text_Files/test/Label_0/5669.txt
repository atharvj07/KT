#include <bits/stdc++.h>
using namespace std;
long long year[30];
long long poww(long long b) {
  long long ans = 1;
  for (long long i = 0; i < b; i++) ans *= 10;
  return ans;
}
void compute(char *x, long long pos) {
  char buf[30];
  strcpy(buf, x + pos);
  long long y = year[pos + 1];
  long long dig = atoi(buf);
  if (strlen(buf) == 1) {
    if (dig == 9)
      year[pos] = 1989;
    else
      year[pos] = 1990 + dig;
  } else {
    long long b = poww(strlen(buf));
    while (dig <= y) {
      dig += b;
    }
    year[pos] = dig;
  }
}
int main() {
  long long n;
  scanf("%I64d", &n);
  for (long long i = 0; i < n; i++) {
    char next[30];
    scanf("%s", next);
    long long len = strlen(next);
    for (long long i = 0; i < len - 4; i++) compute(next, len - 1 - i);
    printf("%I64d\n", year[4]);
  }
}
