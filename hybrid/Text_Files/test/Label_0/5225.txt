#include <iostream>
#include <algorithm>

using namespace std;
int main(){
  int n,r;
  while(cin >> n >> r,n!=0){
    int s[n];
    for(int i=0;i<n;i++){
      s[i] = n -i;
    } 
    int p,c;
    for(int i=0;i<r;i++){
      cin >> p >> c;
      int tmp[c];
      for(int i=0;i<c;i++){
	tmp[i] = s[p+i-1];
      }
      for(int i=0;i<p-1;i++){
	s[c+p-2-i] = s[p-2-i];
      }
      for(int i=0;i<c;i++){
	s[i] = tmp[i];
      }
    }
    cout << s[0] << endl;
  }
}
