from math import ceil, log2
t = int(input())
for _ in range(t):
	res = 0
	n = int(input())
	li = list(map(int, input().split(' ')))
	for i in range(len(li) - 1):
		x1 = max([li[i], li[i + 1]])
		x2 = min([li[i], li[i + 1]])
		if x1 / x2 <= 2:
			continue
		else:
			res += ceil(log2(x1 / x2)) - 1
	print(res)
