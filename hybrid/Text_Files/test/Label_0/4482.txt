#include <bits/stdc++.h>
using namespace std;

int main(){
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    long long int t, n, a, b, odp, c;
    cin >> t;
    for (int x=0;x<t;x++){
        cin >> n >> a >> b;
        if (a == 1){
            if ((n - 1) % b == 0){
                cout << "Yes\n";
            }
            else{
                cout << "No\n";
            }
            continue;
        }
        odp = 0;
        c = 1;
        while (c <= n){
            if (c%b == n%b){
                odp = 1;
            }
            c = c * a;
        }
        if (odp){
            cout << "Yes\n";
        }
        else{
            cout << "No\n";
        }
    }
}
