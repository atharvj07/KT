#include <bits/stdc++.h>
using namespace std;
int main() {
  int testcases;
  cin >> testcases;
  for (int test = 0; test < testcases; test++) {
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    if (a + b > c + d) {
      cout << a + b << endl;
    } else {
      cout << c + d << endl;
    }
  }
  return 0;
}
