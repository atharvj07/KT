#include <bits/stdc++.h>
using namespace std;
struct trie {
  int val, en;
  trie* ch;
  trie* nxt[26];
  trie() {
    val = 0;
    en = 0;
    ch = nullptr;
    for (int i = 0; i < 26; i += 1) {
      nxt[i] = nullptr;
    }
  }
};
string s;
int main() {
  trie* root = new trie();
  int ans = 0, tc = 0;
  while (getline(cin, s)) {
    tc += 1;
    if (s == "]") {
      break;
    }
    ans += 1;
    for (int i = 0; i < s.size(); i += 1) {
      if (!(s[i] >= 'a' && s[i] <= 'z')) {
        ans += 1;
        continue;
      }
      string f = "";
      int pos = i;
      while (pos < s.size() && s[pos] >= 'a' && s[pos] <= 'z') {
        f += s[pos];
        pos += 1;
      }
      int val = f.size(), j;
      trie* curr = root;
      for (j = 0; j < f.size(); j += 1) {
        if (curr->en == 1 && curr->val == 1) {
          int sum = 0;
          trie* sp = curr;
          while (sp->val == 1) {
            sp = sp->ch;
            sum += 1;
          }
          val -= max(0, sum - 2);
        }
        if (!curr->nxt[f[j] - 'a']) {
          curr->nxt[f[j] - 'a'] = new trie();
          curr->nxt[f[j] - 'a']->ch = curr;
        }
        curr = curr->nxt[f[j] - 'a'];
      }
      if (curr->en == 0) {
        curr->en = 1;
        while (curr->ch) {
          curr->val += 1;
          curr = curr->ch;
        }
        ans += val;
      } else {
        int sum = 0;
        while (curr->val == 1) {
          curr = curr->ch;
          sum += 1;
        }
        ans += val - max(0, sum - 2);
      }
      i = pos - 1;
    }
  }
  cout << ans << endl;
}
