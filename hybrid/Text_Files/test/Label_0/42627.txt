import math
import sys
from bisect import bisect_left, bisect_right
input = sys.stdin.readline

def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	s = input().strip()
	return list(s[:len(s)])

def invr():
	return map(int, input().split())

def recur(n):
	if n > 10 ** 6:
		return
	s.add(n)
	recur(n * 10)
	recur(n * 10 + 1)
s = set()
recur(1)
l = list(s)
l.sort()
n = inp()
k = bisect_right(l, n)
l = l[:k]
dp = [sys.maxsize for i in range(n + 1)]
dp[0] = 0
num = n
v = [0 for i in range(n + 1)]
for i in range(1, n + 1):
	for each in l:
		if each <= i:
			if dp[i - each] + 1 < dp[i]:
				dp[i] = dp[i - each] + 1
				v[i] = each
print(dp[n])
while num > 0:
	print(v[num], end=' ')
	num -= v[num]
