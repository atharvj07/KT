#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  scanf("%d", &n);
  int a[n];
  for (int i = 0; i < n; i++) scanf("%d", &a[i]);
  int m = 0, ans = 1, counter;
  for (int i = 0; i < n; i++) m = max(m, a[i]);
  for (int i = 0; i < n; i++) {
    if (a[i] == m) {
      counter = 1;
      for (int j = i + 1; j < n && a[j] == a[j - 1]; j++) counter++;
      i = i + counter - 1;
      ans = max(ans, counter);
    }
  }
  printf("%d\n", ans);
}
