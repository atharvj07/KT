string S;

struct info{
    int val,nes,al;
    info(int val,int nes,int al):val(val),nes(nes),al(al){}
    info(){}
};

info solve(int l,int d,int u){
    if(S[l]=='0'){
        return info(0,l+1,l+1);
    }
    if(isdigit(S[l])){
        info t;
        t.val=S[l]-'0';
        t.nes=l+2;
        if(isdigit(S[l+1])){
            t.val=t.val*10+S[l+1]-'0';
            t.al=l+2;
        }
        else t.al=l+1;

        d++;
        u--;
        if(d/10>S[l]-'0'){
            t.nes=l+1;
        }
        if(u/10<S[l]-'0'&&u<S[l]-'0')t.nes=l+1;
        return t;
    }

    if(S[l]=='^'){
        l+=2;
        info t=solve(l,d,u);
        if(t.val>=u){
            info s=solve(t.al+1,d,u);
            return info(u,t.nes,s.al+1);
        }
        info s=solve(t.al+1,max(d,t.val),u);
        return info(max(t.val,s.val),s.nes,s.al+1);
    }

    l+=2;
    info t=solve(l,d,u);
    if(t.val<=d){
        info s=solve(t.al+1,d,u);
        return info(d,t.nes,s.al+1);
    }
    info s=solve(t.al+1,d,min(u,t.val));
    return info(min(t.val,s.val),s.nes,s.al+1);
}

signed main(){
    int Q;
    cin>>Q;
    while(Q--){
        cin>>S;
        info t=solve(0,0,99);
        cout<<t.val<<" "<<t.nes<<endl;
    }
    return 0;
}
