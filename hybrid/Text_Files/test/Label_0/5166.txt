#include <bits/stdc++.h>
using namespace std;
int main() {
  long i, t, n, p, k;
  cin >> t;
  while (t--) {
    cin >> n >> p >> k;
    int b[n], sum[n];
    for (i = 0; i < n; i++) cin >> b[i];
    sort(b, b + n);
    for (i = 0; i < n; i++) {
      if (i > 1)
        sum[i] = b[i] + sum[i - 2];
      else
        sum[i] = b[i];
      if (sum[i] > p) break;
    }
    cout << i << endl;
  }
  return 0;
}
