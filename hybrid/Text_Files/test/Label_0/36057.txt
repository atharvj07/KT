import java.util.Scanner;


public class cfA424 {

    /**
     * @param args
     */
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n=sc.nextInt();
        String s=sc.next();
        int stay = 0,sit=0,counter =0;   
        for (int i = 0; i < n; i++) {
            if(s.charAt(i)=='X')stay++;
            if(s.charAt(i)=='x')sit++;
        }
        int dis=Math.abs((stay-sit)/2);
        char[] c= s.toCharArray();
        if(stay>sit)
        {           
            while(stay>sit) {
                if(c[counter]!='x')
                {
                c[counter]='x';             
                stay--;
                sit++;              
                }       
                counter++;
            }
            System.out.println(dis);
        }
        if(sit>stay)
        {
            while(stay<sit) {
                if(c[counter]!='X')
                {
                c[counter]='X';             
                sit--;
                stay++;             
                }
                counter++;
            }
            System.out.println(dis);
        }
        if(sit==stay&&counter==0) System.out.println(0);
        System.out.println(c);
    }       

}
