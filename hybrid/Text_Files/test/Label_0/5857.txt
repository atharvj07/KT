void solve() {
    int n, k = 0;
    cin >> n >> k;
    vector <int> a(n);
    vector <int> right(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    unordered_map<int, int> last;
    for (int i = n - 1; i >= 0; --i) {
        int now = a[i];
        int help = 1;
        for (int j = 2; j * j <= now; ++j) {
            int cnt = 0;
            while (now % j == 0) {
                cnt++;
                now /= j;
            }
            if (cnt % 2 == 1)
                help *= j;
        }
        help *= now;
        right[i] = last[help] - 1;
        if (right[i] == -1)
            right[i] = n;
        last[help] = i + 1;
    }
    int ans = 0;
    int r = right[0];
    for (int i = 0; i <= n; i++) {
        if (i == r) {
            ans++;
            r = right[i];
        }
        if (i == n) break;
        r = min(r, right[i]);
    }
    cout << ans << "\n";
}
signed main() {
    fast_stream();
    ll t;
    cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
