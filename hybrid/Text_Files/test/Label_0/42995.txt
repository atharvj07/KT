t = int(input())
nums = [None] * t
ans = [False] * t
for i in range(t):
	nums[i] = input()
	if '2' not in nums[i] and '4' not in nums[i] and ('6' not in nums[i]) and ('8' not in nums[i]):
		ans[i] = -1
	elif int(nums[i]) % 2 == 0:
		ans[i] = 0
	elif int(nums[i][0]) % 2 == 0:
		ans[i] = 1
	else:
		ans[i] = 2
for i in ans:
	print(i)
