#include <bits/stdc++.h>
using namespace std;
using pi = pair<int, int>;
using lint = long long;
const int mod = 1e9 + 7;
const int inv2 = (mod + 1) / 2;
const int MAXN = 100005;
char s[MAXN];
int main() {
  scanf("%s", s);
  int n = strlen(s);
  lint r = 0, k = 0, p = 1;
  for (int i = 1; i <= n; ++i) {
    lint l = n - i;
    r = (r + lint(s[n - i] - '0') *
                 (k + p * l % mod * (l + 1) % mod * inv2 % mod)) %
        mod;
    k = (k + p * i) % mod;
    p = p * 10 % mod;
  }
  printf("%lld\n", r);
}
