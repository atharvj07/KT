#include <bits/stdc++.h>
using namespace std;
const long long mod = 1e9 + 7;
long long ksm(long long a, long long b, long long p) {
  long long res = 1;
  while (b) {
    if (b & 1) res = res * a % p;
    b >>= 1;
    a = a * a % p;
  }
  return res;
}
int main() {
  long long a, b, n, x, s;
  scanf("%lld%lld%lld%lld", &a, &b, &n, &x);
  if (a == 1) {
    s = (x + n % mod * b % mod) % mod;
    printf("%lld", s);
    return 0;
  }
  if (n == 1) {
    s = (a * x % mod + b) % mod;
    printf("%lld", s);
    return 0;
  }
  if (n == 0) {
    s = x % mod;
    printf("%lld", s);
    return 0;
  }
  s = ksm(a, n, mod) * x % mod;
  long long p = (1 - ksm(a, n, mod)) % mod + mod;
  long long q = ksm((1 - a), mod - 2, mod) % mod + mod;
  s = (s + p * q % mod * b % mod) % mod;
  printf("%lld", s);
  return 0;
}
