(n, mxH, k) = map(int, input().split())
arr = list(map(int, input().split()))
(currT, currH, res) = (0, 0, 0)
i = 0
while i < len(arr):
	while i < len(arr) and currH + arr[i] <= mxH:
		currH += arr[i]
		i += 1
	if i < len(arr):
		currT = (currH + arr[i] - mxH + k - 1) // k
	else:
		currT = (currH + k - 1) // k
	currH = max(currH - currT * k, 0)
	res += currT
print(res)
