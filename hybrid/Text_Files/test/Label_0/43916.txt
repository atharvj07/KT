from sys import stdin, stdout
import math
for _ in range(int(stdin.readline())):
	(n, k) = map(int, stdin.readline().split())
	a = list(map(int, stdin.readline().split()))
	a1 = []
	a2 = []
	for i in a:
		if i < 0:
			a1.append(i * -1)
		elif i > 0:
			a2.append(i)
	a1.sort()
	a2.sort()
	ans = 0
	n1 = len(a1)
	n2 = len(a2)
	for i in range(n2 - 1 - k, -1, -k):
		ans += 2 * a2[i]
	for i in range(n1 - 1 - k, -1, -k):
		ans += 2 * a1[i]
	if n1 > 0 and n2 > 0:
		if a1[-1] > a2[-1]:
			ans += a1[-1]
			ans += 2 * a2[-1]
		else:
			ans += a2[-1]
			ans += 2 * a1[-1]
	elif n1 > 0:
		ans += a1[-1]
	elif n2 > 0:
		ans += a2[-1]
	print(ans)
