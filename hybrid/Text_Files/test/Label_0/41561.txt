for _ in range(int(input())):
	(n, k) = list(map(int, input().split()))
	a = list(map(int, input().split()))
	t = 0
	size = 2
	res_t = 0
	res_l = 0
	for r in range(2, n):
		size += 1
		l = max(0, r - k + 1)
		if a[r - 2] < a[r - 1] and a[r - 1] > a[r]:
			t += 1
		if size > k:
			if a[l - 1] < a[l] and a[l] > a[l + 1]:
				t -= 1
			size -= 1
		if t > res_t:
			res_t = t
			res_l = l
	print(res_t + 1, res_l + 1)
