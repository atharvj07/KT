t = int(input())
for i in range(t):
	n = int(input())
	mass = [int(x) for x in input().split()]
	answer = 0
	q = []
	Max = 0
	ind = 0
	for j in range(n):
		if Max < mass[j]:
			Max = mass[j]
	ind = Max - 1
	a1 = mass[:n - ind - 1]
	a2 = mass[n - ind - 1:]
	a3 = mass[:ind + 1]
	a4 = mass[ind + 1:]
	a1.sort()
	a2.sort()
	a3.sort()
	a4.sort()
	f1 = True
	f2 = True
	for i in range(n - ind - 1):
		if a1[i] != i + 1:
			f1 = False
	if f1:
		for i in range(ind + 1):
			if a2[i] != i + 1:
				f1 = False
	if f1:
		answer += 1
		q.append((n - ind - 1, ind + 1))
	for i in range(ind + 1):
		if a3[i] != i + 1:
			f2 = False
	if f2:
		for i in range(n - ind - 1):
			if a4[i] != i + 1:
				f2 = False
	if f2 and n - ind - 1 != ind + 1:
		answer += 1
		q.append((ind + 1, n - ind - 1))
	print(answer)
	for l in q:
		print(*l)
