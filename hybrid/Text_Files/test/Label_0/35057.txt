import java.io.BufferedOutputStream;
import java.io.DataInputStream;
import java.io.File;
import java.io.FileInputStream;

import java.io.FileOutputStream;
import java.io.IOException;
import java.io.OutputStream;
import java.io.PrintStream;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.Map;
import java.util.Queue;
import java.util.stream.Collector;
import java.util.stream.Collectors;
import java.util.stream.IntStream;

public class NpHardProblem {
    static Output out;
    static Reader in;
    static Exception exception;

    static int pariCount = 0;
    static ArrayList<Node> paris = new ArrayList<>();

    static int ariCount = 0;
    static ArrayList<Node> aris = new ArrayList<>();


    public static void operate() throws IOException {
        int n = in.nextInt();
        int m = in.nextInt();

        Map<Integer, Node> map = new HashMap<>();

        IntStream.rangeClosed(1, n).forEach(i -> {
            map.put(i, new Node(i));
        });

        IntStream.rangeClosed(1, m).forEach(i -> {
            int node1 = in.nextInt();
            int node2 = in.nextInt();

            map.get(node1).siblings.add(map.get(node2));
            map.get(node2).siblings.add(map.get(node1));

        });

        map.forEach((index, node) -> {
            if (node.marked) {
                return;
            }
            Queue<Node> queue = new LinkedList<>();
            queue.add(
                    map.get(index)
                            .setColor(Color.Pari)
                            .setMarked(true)
            );

            while (!queue.isEmpty()) {
                Node poll = queue.poll();

                ArrayList<Node> siblings = poll.siblings;
                for (Node sibling : siblings) {
                    if (!sibling.marked) {
                        queue.add(sibling);
                        sibling
                                .setColor(poll.color == Color.Aria ? Color.Pari : Color.Aria)
                                .setMarked(true);
                    }
                }
            }
        });

        boolean collision = map.entrySet().stream().anyMatch(entry -> {
            if (map.get(entry.getKey()).siblings.isEmpty()) {
                return false;
            }
            return map.get(entry.getKey()).siblings.stream().anyMatch(node -> {
                return map.get(entry.getKey()).color == node.color;
            });
        });

        if (collision) {
            out.println(-1);
        } else {
            out.println(pariCount);
            out.println(
                    paris.stream()
                            .map(node -> node.nodeId + "")
                            .collect(Collectors.joining(" "))
            );


            out.println(ariCount);
            out.println(
                    aris.stream()
                            .map(node -> node.nodeId + "")
                            .collect(Collectors.joining(" "))
            );
        }


    }

    static class Node {
        ArrayList<Node> siblings = new ArrayList<>();
        boolean marked = false;
        Color color;
        Integer nodeId;

        public Node(Integer nodeId) {
            this.nodeId = nodeId;
        }

        public Node setColor(Color color) {
            switch (color) {
                case Aria:
                    ariCount++;
                    aris.add(this);
                    break;
                case Pari:
                    pariCount++;
                    paris.add(this);
                    break;
            }
            this.color = color;
            return this;
        }

        public Node setMarked(boolean marked) {
            this.marked = marked;
            return this;
        }

        public Node setNodeId(Integer nodeId) {
            this.nodeId = nodeId;
            return this;
        }

        public Node setSiblings(ArrayList<Node> siblings) {
            this.siblings = siblings;
            return this;
        }

        @Override
        public String toString() {
            return "Node{" +
                    "nodeId=" + nodeId +
                    ", color=" + color +
                    ", siblings=[" + siblings.stream().map(s -> s.nodeId + "").collect(Collectors.joining(";")) + "]" +
                    ", marked=" + marked +
                    '}';
        }

    }

    static enum Color {
        Pari, Aria
    }

    private static void initReaderWriter() throws Exception {
        in = new Reader();
        out = new Output(System.out);

//            in = new Reader("input.txt");

    }

    public static void main(String[] args) throws Exception {
        Runnable run = () -> {
            try {
                initReaderWriter();
                operate();
                out.close();
            } catch (Exception ex) {
                exception = ex;
            }
        };
        Thread thread = new Thread(null, run, "", 1 << 26);
        thread.start();
        thread.join();

        if (exception != null) {
            throw exception;
        }
    }

    static class Output extends PrintStream {

        public Output(OutputStream out) {
            super(new BufferedOutputStream(out));
        }

        public Output(File file) throws FileNotFoundException {
            super(new BufferedOutputStream(new FileOutputStream(file)));
        }
    }

    static class Reader {
        final private int BUFFER_SIZE = 1 << 16;
        private DataInputStream din;
        private byte[] buffer;
        private int bufferPointer, bytesRead;

        public Reader() {
            din = new DataInputStream(System.in);
            buffer = new byte[BUFFER_SIZE];
            bufferPointer = bytesRead = 0;
        }

        public Reader(String file_name) throws IOException {
            din = new DataInputStream(new FileInputStream(file_name));
            buffer = new byte[BUFFER_SIZE];
            bufferPointer = bytesRead = 0;
        }

        public String readLine() throws IOException {
            byte[] buf = new byte[64]; // line length
            int cnt = 0, c;
            while ((c = read()) != -1) {
                if (c == '\n')
                    break;
                buf[cnt++] = (byte) c;
            }
            return new String(buf, 0, cnt);
        }

        public int nextInt() {
            try {
                int ret = 0;
                byte c = read();
                while (c <= ' ')
                    c = read();
                boolean neg = (c == '-');
                if (neg)
                    c = read();
                do {
                    ret = ret * 10 + c - '0';
                } while ((c = read()) >= '0' && c <= '9');

                if (neg)
                    return -ret;
                return ret;
            } catch (Exception ex) {
                throw new RuntimeException(ex);
            }

        }

        public long nextLong() {
            try {
                long ret = 0;
                byte c = read();
                while (c <= ' ')
                    c = read();
                boolean neg = (c == '-');
                if (neg)
                    c = read();
                do {
                    ret = ret * 10 + c - '0';
                }
                while ((c = read()) >= '0' && c <= '9');
                if (neg)
                    return -ret;
                return ret;
            } catch (IOException ex) {
                throw new RuntimeException(ex);
            }
        }

        public double nextDouble() {
            try {
                double ret = 0, div = 1;
                byte c = read();
                while (c <= ' ')
                    c = read();
                boolean neg = (c == '-');
                if (neg)
                    c = read();

                do {
                    ret = ret * 10 + c - '0';
                }
                while ((c = read()) >= '0' && c <= '9');

                if (c == '.') {
                    while ((c = read()) >= '0' && c <= '9') {
                        ret += (c - '0') / (div *= 10);
                    }
                }

                if (neg)
                    return -ret;
                return ret;
            } catch (IOException ex) {
                throw new RuntimeException(ex);
            }
        }

        private void fillBuffer() throws IOException {
            bytesRead = din.read(buffer, bufferPointer = 0, BUFFER_SIZE);
            if (bytesRead == -1)
                buffer[0] = -1;
        }

        private byte read() throws IOException {
            if (bufferPointer == bytesRead)
                fillBuffer();
            return buffer[bufferPointer++];
        }

        public void close() throws IOException {
            if (din == null)
                return;
            din.close();
        }
    }
}
