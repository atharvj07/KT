(n, m) = [int(x) for x in input().strip().split()]

def encode(x):
	accum = 1
	res = 0
	base_A = ord('A')
	base_Z = ord('Z')
	for idx in range(m, 0, -1):
		c = x[idx - 1]
		if idx % 2 != 0:
			res += (ord(c) - base_A) * accum
		else:
			res += (base_Z - ord(c)) * accum
		accum *= 26
	return res
strings = [[0, 0] for _ in range(n)]
for i in range(n):
	x = input().rstrip()
	strings[i] = (i + 1, encode(x))
strings.sort(key=lambda x: x[1])
print(' '.join([str(x[0]) for x in strings]))
