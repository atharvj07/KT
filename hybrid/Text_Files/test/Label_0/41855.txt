import collections as cc
import math as mt
import sys
I = lambda : list(map(int, input().split()))
(n, m) = I()
a = [0] * (n + 1)
for i in range(n):
	a[i] = list(input()) + ['.']
a[n] = ['.'] * m
used = [[False] * m for i in range(n)]
ans = []
for i in range(n):
	for j in range(m):
		if a[i][j] == '.':
			continue
		k = 1
		while a[i - k][j] == '*' and a[i + k][j] == '*' and (a[i][j - k] == '*') and (a[i][j + k] == '*'):
			used[i - k][j] = used[i + k][j] = used[i][j - k] = used[i][j + k] = True
			k += 1
		if k > 1:
			used[i][j] = True
			ans += [[i + 1, j + 1, k - 1]]
for i in range(n):
	for j in range(m):
		if a[i][j] == '*' and (not used[i][j]):
			print(-1)
			sys.exit()
print(len(ans))
for i in ans:
	print(*i)
