#include <bits/stdc++.h>
using namespace std;
int n, ar[220020], fr[220020];
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    int ok = 0;
    for (int i = 0; i < n; i++) cin >> ar[i];
    for (int i = 1; i < n; i++) {
      if (abs(ar[i] - ar[i - 1]) == 1 || (ar[i] == n && ar[i - 1] == 1) ||
          ar[i] == 1 && ar[i - 1] == n)
        continue;
      else {
        ok = 1;
        break;
      }
    }
    if (ok)
      cout << "NO" << endl;
    else
      cout << "YES" << endl;
  }
  return 0;
}
