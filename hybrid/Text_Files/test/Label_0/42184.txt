for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	mx = max(a)
	p1 = a[:mx]
	p2 = a[mx:]
	test = 0
	test1 = 0
	if len(set(p1)) == mx and max(p1) == mx and (len(set(p2)) == n - mx) and (max(p2) == n - mx):
		test = 1
	p1 = a[:n - mx]
	p2 = a[n - mx:]
	if len(set(p1)) == n - mx and max(p1) == n - mx and (len(set(p2)) == mx) and (max(p2) == mx):
		test1 = 1
	if mx == n // 2 and test:
		print(1)
		print(mx, n - mx)
	elif test and test1:
		print(2)
		print(mx, n - mx)
		print(n - mx, mx)
	elif test:
		print(1)
		print(mx, n - mx)
	elif test1:
		print(1)
		print(n - mx, mx)
	else:
		print(0)
