#include <bits/stdc++.h>
using namespace std;
int n, taken[1000006], cst[1000006];
vector<pair<int, string> > v;
vector<int> id;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n;
  for (int i = 0; i < n; i++) {
    string tmp;
    int a;
    cin >> tmp >> a;
    v.push_back({a, tmp});
  }
  sort(v.begin(), v.end());
  int now = 0;
  for (int i = 0; i < v.size(); i++) {
    int val = v[i].first;
    string name = v[i].second;
    if (val > i) {
      cout << -1 << endl;
      return 0;
    }
    if (val != now) {
      int diff = val - now;
      for (int j = i - 1; j >= 0 && diff > 0; j--) {
        if (taken[j] == 0) {
          diff--;
          taken[j] = 1;
          id.push_back(j);
        }
      }
      now = val;
    }
    for (int j = 0; j < id.size(); j++) {
      cst[id[j]]++;
    }
    while (i < v.size() && val == v[i].first) {
      i++;
    }
    i--;
    id.push_back(i);
    taken[i] = 1;
    now++;
  }
  for (int i = 0; i < n; i++) {
    cout << v[i].second << " " << 1 + cst[i] << endl;
  }
  return 0;
}
