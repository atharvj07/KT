T = int(input())
for x in range(T):
	N = int(input())
	A = [int(d) for d in input().split()]
	used = [0] * (N + 1)
	pt = 1
	A.sort()
	ans = 0
	done = False
	for a in A:
		while used[pt] == 1:
			pt += 1
		if a <= N and used[a] == 0:
			used[a] = 1
		elif int((a - 1) / 2) >= pt:
			used[pt] = 1
			ans += 1
		else:
			print('-1')
			done = True
			break
	if not done:
		print(ans)
