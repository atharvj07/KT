#include<bits/stdc++.h>
using namespace std;

int n,k,a[100100];

int gcd(int a,int b){
	if (a%b==0) return b;
	else return gcd(b,a%b);
}

int main(){
	cin>>n>>k;
	cin>>a[1];
	int g=a[1],m=a[1];
	for (int i=2;i<=n;i++){
		cin>>a[i];
		g=gcd(max(g,a[i]),min(g,a[i]));
		m=max(m,a[i]);
	}
	if (k<=m && k%g==0) cout<<"POSSIBLE"<<endl;
	else cout<<"IMPOSSIBLE"<<endl;
	return 0;
}
