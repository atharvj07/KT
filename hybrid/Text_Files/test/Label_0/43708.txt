for _ in range(int(input())):
	n = int(input())
	if n == 1:
		print(9)
	elif n == 2:
		print(98)
	elif n == 3:
		print(989)
	else:
		print(989, end='')
		if n <= 13:
			for i in range(n - 3):
				print(i, end='')
			print()
		else:
			s = '0123456789'
			p = (n - 3) // 10
			l = (n - 3) % 10
			print('0123456789' * p + s[:l], end='')
			print()
