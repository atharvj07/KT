#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef unsigned long long hxtype;
#define rep(i,l,r) for(i=(l);i<=(r);++i)
#define per(i,l,r) for(i=(l);i>=(r);--i)
#define REP(i,l,r) for(i=(l);i< (r);++i)
#define PER(i,l,r) for(i=(l);i> (r);--i)
template<class IT>inline void cmax(IT &a,IT b){if(a<b)a=b;}
template<class IT>inline void cmin(IT &a,IT b){if(b<a)a=b;}
const ll p=998244353ll;
const int N=1000005;
struct duck{
	int b,c,h,u;
	inline duck(){}
	inline duck(int b,int c,int h):b(b),c(c),h(h),u(!(b&&c)){}
}x[N];
char s[N];
int a[N],b[N];
int main(){
	int n=0,m,i,c[3]={0,0,0},he=3,t=0,c1=0,c2=0,ans,B,C,H;
	int x1=0,x2=0,u1=0,u2=0;
	scanf("%s",s+1);
	m=strlen(s+1);
	rep(i,1,m)if(s[i]!=s[i-1])++c[a[++n]=s[i]-65];
	if(!(c[0]&&c[1]&&c[2]))return 0;
	m=0;
	if(c[m]>c[1])m=1;
	if(c[m]>c[2])m=2;
	ans=c[m];
	if(m)rep(i,1,n){
		if(a[i]){
			if(a[i]==m)a[i]=0;
		}else a[i]=m;
	}
//	rep(i,1,n)printf("%d ",a[i]);printf("\n");
	rep(i,1,n){
		if(a[i]){
			if(he==3)he=a[i]-1;
			if(a[i]==1)++c1;else ++c2;
		}else{
			x[++t]=duck(c1,c2,he);
			c1=c2=0;he=3;
		}
	}
	x[++t]=duck(c1,c2,he);
	x[1].u=x[t].u=0;
//	rep(i,1,t)printf("(b=%d,c=%d,h=%d,u=%d)\n",x[i].b,x[i].c,x[i].h,x[i].u);
	rep(i,1,t){
		if(x[i].b){
			++x1;
			if(x[i].u)++u1;
		}
		if(x[i].c){
			++x2;
			if(x[i].u)++u2;
		}
	}
//	printf("x1=%d x2=%d u1=%d u2=%d\n",x1,x2,u1,u2);
	rep(i,1,t)if(u1>x2&&x[i].h!=2&&x[i].b&&x[i].u){
		--ans;
		--u1;
		--x1;
		x[i].h=2;
	}
	rep(i,1,t)if(u2>x1&&x[i].h!=2&&x[i].c&&x[i].u){
		--ans;
		--u2;
		--x2;
		x[i].h=2;
	}
//	rep(i,1,t)printf("(b=%d,c=%d,h=%d,u=%d)\n",x[i].b,x[i].c,x[i].h,x[i].u);
	u1=u2=0;
	rep(i,1,t)if(x[i].h!=2){
		u1+=x[i].b;
		u2+=x[i].c;
	}
//	printf("t=%d ans=%d u1=%d u2=%d\n",t,ans,u1,u2);
	rep(i,1,t)if(x[i].h!=2){
		if(u1>u2&&!x[i].u&&x[i].b> x[i].c){--x[i].b;--u1;if(!x[i].c)x[i].h=3;}
		if(u1>u2&& x[i].c&&x[i].b==x[i].c){--x[i].b;--u1;           x[i].h=1;}
	}
	rep(i,1,t)if(x[i].h!=2){
		if(u2>u1&&!x[i].u&&x[i].c> x[i].b){--x[i].c;--u2;if(!x[i].b)x[i].h=3;}
		if(u2>u1&& x[i].b&&x[i].c==x[i].b){--x[i].c;--u2;           x[i].h=0;}
	}
//	printf("t=%d u1=%d u2=%d\n",t,u1,u2);
//	rep(i,1,t)printf("(b=%d,c=%d,h=%d,u=%d)\n",x[i].b,x[i].c,x[i].h,x[i].u);
	c1=c2=0;
	rep(i,1,t)if(x[i].h!=2){
		if(c2)b[++c1]=0;else c2=1;
		if((H=x[i].h)!=3){
			B=x[i].b;
			C=x[i].c;
			while(B&&C&&u1>ans){
				--u1;--B;--C;
				if(!(B||C)&&i!=1&&i!=t){++u1;B=C=1;break;}
			}
			while(B||C){
				if(H){
					--C;
					b[++c1]=2;
					H=0;
				}else{
					--B;
					b[++c1]=1;
					H=1;
				}
			}
		}
	}
	if(m)rep(i,1,c1){
		if(b[i]){
			if(b[i]==m)b[i]=0;
		}else b[i]=m;
	}
	rep(i,1,c1)putchar(b[i]+65);
	return 0;
}
