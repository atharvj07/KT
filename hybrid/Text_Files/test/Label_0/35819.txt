public class LipshitzSequence {

    public static void main ( String [] args ) throws IOException {

        BufferedReader reader = new BufferedReader ( new InputStreamReader( System.in ) );
        String [] aux = reader.readLine().split( " " );
        int n = Integer.valueOf( aux[0] );
        int m = Integer.valueOf( aux[1] );
        int [] data = new int[n];
        aux = reader.readLine().split( " " );
        for ( int i=0; i<n; i++ ){
            data[i] = Integer.valueOf( aux[i] );
        }
        int [] delta = new int[n-1];
        // delta[i] = (data[i+1]-data[i])/1

        for ( int i=0; i<n-1; i++ ){
            delta[i] = Math.abs( data[i+1] - data[i] );
        }

        Map<String,Long> sols = new HashMap<>();

        PrintWriter output = new PrintWriter( System.out );
        for ( int i=0; i<m; i++ ){
            aux = reader.readLine().split( " " );
            SumStack sum = new SumStack();
            int a = Integer.valueOf( aux[0] ) - 1;
            int b = Integer.valueOf( aux[1] ) - 1;

            String key = a + " " + b;

            if ( sols.get( key ) == null ){

                long sol = 0;
                for ( int j=a; j<b; j++ ){
                    sum.add( delta[j] );
                    sol += sum.sum;
                }
                sols.put( key, sol );
            }
            output.println( sols.get( key ) );
        }
        output.flush();;

    }

    static class Pair {
        long value;
        long count;
    }

    static class SumStack {
        long sum = 0;
        Stack<Pair> stack = new Stack<Pair>();

        public void add( int value ){
            Pair p = new Pair();
            p.value = value;
            p.count = 1;
            sum += value;
            while ( stack.size() > 0 && stack.peek().value <= p.value  ){
                Pair pop = stack.pop();
                sum -= pop.value * pop.count;
                sum += p.value * pop.count;
                p.count += pop.count;
            }
            stack.add( p );
        }
    }

   static class BinaryIndexedTree {

        long [] data;

        public BinaryIndexedTree( int n ){
            data = new long[n+1];
        }

        public void update( int index, int value ){
            while ( index < data.length ){
                data[ index ] += value;
                index = index + ( index & -index );
            }
        }

        // suma de la 1 pana la index
        public long sum( int index ){
            long ret = 0;
            while ( index > 0 ){
                ret += data[ index ];
                index = index - ( index & -index );
            }
            return ret;
        }

    }

}
