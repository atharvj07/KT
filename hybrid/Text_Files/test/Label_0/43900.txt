import sys
data = sys.stdin.read().splitlines()
s = data[0]
n = int(data[1])
letters = snows = candys = 0
for ch in s:
	if ch == '?':
		candys += 1
	elif ch == '*':
		snows += 1
	else:
		letters += 1
mn = letters - snows - candys
mx = float('inf') if snows else letters
if n < mn or n > mx:
	print('Impossible')
elif snows == 0:
	rem = n - mn
	lst = []
	for ch in s:
		lst.append(ch)
		if ch == '?':
			lst.pop()
			if rem:
				rem -= 1
			else:
				lst.pop()
	print(''.join(lst))
else:
	lst = []
	rem = n - mn
	for ch in s:
		lst.append(ch)
		if ch == '?':
			lst.pop()
			lst.pop()
		elif ch == '*':
			lst.pop()
			c = lst.pop()
			if rem:
				lst.append(c * rem)
				rem = 0
	print(''.join(lst))
