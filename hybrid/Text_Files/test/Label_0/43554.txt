e = int(input())
l = []
y = []
A = []
for i in range(e):
	[a, b] = input().split(' ')
	n = int(a)
	m = int(b)
	A.append((n, m))
	x = []
	z = input()
	y.append(z)
	for j in range(m):
		t = input().split(' ')
		for i in range(2):
			t[i] = int(t[i])
		x.append(t)
	l.append(x)
for i in range(e):
	(n, m) = A[i]
	x = 0
	L = [0]
	M = [(0, 0)]
	mi = 0
	ma = 0
	for j in range(n):
		if y[i][j] == '-':
			x = x - 1
		else:
			x = x + 1
		mi = min(mi, x)
		ma = max(ma, x)
		M.append((mi, ma))
		L.append(x)
	N = [(L[-1], L[-1])]
	(mi, ma) = (L[-1], L[-1])
	for j in reversed(range(n)):
		mi = min(mi, L[j])
		ma = max(ma, L[j])
		N.append((mi, ma))
	for j in range(m):
		[a, b] = l[i][j]
		(X, Y) = M[a - 1]
		(K, P) = N[n - b]
		if b < n:
			X = min(K - (L[b] - L[a - 1]), X)
			Y = max(P - (L[b] - L[a - 1]), Y)
		print(Y - X + 1)
