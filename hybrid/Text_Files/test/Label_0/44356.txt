import sys
(s, left_index, sub_sum, subset) = ([], -1, 0, 1)
input()
for line in sys.stdin:
	inp = list(map(int, line.split()))
	if inp[0] == 2:
		sub_sum += s[-1]
		for j in range(left_index + 1, len(s)):
			if s[j] <= sub_sum / subset:
				sub_sum += s[j]
				subset += 1
				left_index += 1
			else:
				break
		print(s[-1] - sub_sum / subset)
		sub_sum -= s[-1]
	else:
		s.append(inp[1])
