#include <bits/stdc++.h>
using namespace std;
const int N = 100500;
int n, a, b, p[N], deg[N];
set<int> cnt;
map<int, int> type;
void dfs(int cur, int pr = -1) {
  int a_pair = a - cur;
  int b_pair = b - cur;
  if (cnt.count(a_pair) && a_pair != pr) {
    dfs(a_pair, cur);
    if (cnt.count(cur) && cnt.count(a_pair)) {
      type[cur] = 0;
      type[a_pair] = 0;
      cnt.erase(cur);
      cnt.erase(a_pair);
    }
  } else if (cnt.count(b_pair) && b_pair != pr) {
    dfs(b_pair, cur);
    if (cnt.count(cur) && cnt.count(b_pair)) {
      type[cur] = 1;
      type[b_pair] = 1;
      cnt.erase(cur);
      cnt.erase(b_pair);
    }
  }
}
int main() {
  cin >> n >> a >> b;
  for (int i = 1; i <= n; i++) {
    cin >> p[i];
    cnt.insert(p[i]);
  }
  if (a == b) {
    for (int i = 1; i <= n; i++)
      if (cnt.count(p[i]) && cnt.count(a - p[i])) {
        cnt.erase(p[i]);
        cnt.erase(a - p[i]);
      }
    if (cnt.size()) {
      puts("NO");
      return 0;
    }
    puts("YES");
    for (int i = 1; i <= n; i++) cout << "0 ";
    return 0;
  }
  for (int i = 1; i <= n; i++) {
    if (cnt.count(a - p[i])) deg[i]++;
    if (cnt.count(b - p[i])) deg[i]++;
    if (deg[i] == 0) {
      puts("NO");
      return 0;
    }
  }
  for (int i = 1; i <= n; i++)
    if (deg[i] == 1) dfs(p[i]);
  if (cnt.size()) {
    puts("NO");
    return 0;
  }
  puts("YES");
  for (int i = 1; i <= n; i++) cout << type[p[i]] << " ";
  cout << endl;
  return 0;
}
