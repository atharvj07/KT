n = int(input())
s = list(input())
a = 'RGB'
if n == 1:
	print(0)
	print(s[0])
	from sys import exit
	exit()
cost = 0
for i in range(1, n):
	if s[i] == s[i - 1]:
		for c in a:
			if s[i - 1] != c and (i < n - 1 and c != s[i + 1]):
				s[i] = c
				cost += 1
				break
if s[-1] == s[-2]:
	for c in a:
		if c != s[-2]:
			s[-1] = c
			cost += 1
			break
print(cost)
print(''.join(s))
