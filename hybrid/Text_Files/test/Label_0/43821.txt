(a, b) = map(int, input().split())
s = [[0 for n in range(a)] for n in range(a)]
x = a * a
res = 0
for n in range(a):
	for k in range(a - b + 1):
		s[n][k] = x
		x -= 1
	res += s[n][a - b]
	s[n] = s[n][::-1]
x = 1
for n in range(a):
	for k in range(b - 1):
		s[n][k] = x
		x += 1
print(res)
for n in s:
	for k in n:
		print(k, end=' ')
	print()
