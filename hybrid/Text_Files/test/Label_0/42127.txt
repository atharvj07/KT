from sys import stdin
input = stdin.readline
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	a = input()
	b = input()
	l1 = [0] * 26
	l2 = [0] * 26
	f = True
	for i in range(n):
		x = ord(a[i]) - ord('a')
		l1[x] += 1
	for i in range(n):
		x = ord(b[i]) - ord('a')
		l2[x] += 1
	for i in range(26):
		if l1[i] < l2[i] or (l1[i] - l2[i]) % k != 0:
			f = False
			break
		elif i < 25:
			l1[i + 1] += l1[i] - l2[i]
	if f:
		print('Yes')
	else:
		print('No')
