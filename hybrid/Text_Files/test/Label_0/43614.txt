from math import *
(n, k1, k2) = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
m = 0
c = []
s = k1 + k2
for i in range(n):
	c.append((a[i] - b[i]) * (a[i] - b[i]))
m = sum(c)
while s:
	s -= 1
	ma = c.index(max(c))
	c[ma] = (sqrt(c[ma]) - 1) * (sqrt(c[ma]) - 1)
print(int(sum(c)))
