import java.util.Scanner;

public class taskB {
    static int maxSum;
    static long bestNum;

    public static void main(String[] args) {
        String x = new Scanner(System.in).next();
        maxSum = summing(Long.valueOf(x));
        bestNum = Long.valueOf(x);
//        отброс маленьких чисел
        if(Long.valueOf(x)<10){
            System.out.println(summing(Long.valueOf(x)));
            return;
        }
//        обработка рез-тов
        int p;
        for(int i=x.length()-1;i>0;i--){
            if(x.charAt(i)=='9'){
                check(x);
                continue;
            }
            if(x.charAt(i-1)=='0'){
                x = x.substring(0,i)+"9"+x.substring(i+1);
                continue;
            }
            p=Integer.valueOf(Character.toString(x.charAt(i-1)))-1;
            x = x.substring(0,i-1)+p+"9"+x.substring(i+1);
            check(x);
        }
//      подсчет максимального числа меньшего порядка
        String min="";
        for(int i=0;i<x.length()-1;i++){
            min+="9";
        }
        check(min);
//        вывод
        System.out.println(bestNum);

    }

    private static void check(String x){
        long l = Long.valueOf(x);
        if(summing(l)>maxSum){
            maxSum = summing(l);
            bestNum = l;
        }
    }

    private static int summing(long x){
        int sum=0;
        while (x!=0){
            sum+=x%10;
            x/=10;
        }
        return sum;
    }
}
