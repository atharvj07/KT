[n, m] = [int(x) for x in input().split()]
s = input()
t = input()
(bst, idx) = (0, 0)
for i in range(m - n + 1):
	ac = 0
	for j in range(n):
		ac += int(s[j] == t[i + j])
	if ac > bst:
		bst = ac
		idx = i
ans = [str(k + 1) for (k, x) in enumerate(s) if s[k] != t[idx + k]]
print('\n'.join([str(len(ans)), ' '.join(ans)]))
