def f():
	(n, m, k) = [int(s) for s in input().split()]
	A = [int(s) for s in input().split()]
	maxA = 1
	for a in A:
		maxA = max(maxA, a)
	limit = min(m - 1, k)
	free = m - 1 - limit
	ans = 1
	for a0 in range(limit + 1):
		minX = maxA
		for a in range(a0, a0 + free + 1):
			b = m - 1 - a
			x = max(A[a], A[-b - 1])
			minX = min(minX, x)
		ans = max(ans, minX)
	return ans
T = int(input())
for t in range(T):
	print(f())
