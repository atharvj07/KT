#include <bits/stdc++.h>
using namespace std;
long long ara[100011], cnt[100010];
map<long long, long long> mymp;
map<long long, long long> arek;
int main() {
  long long n;
  scanf("%lld", &n);
  for (long long i = 1; i <= n; i++) scanf("%lld", &ara[i]);
  for (long long i = n; i > 0; i--) {
    cnt[i] = cnt[i + 1];
    long long tm = ara[i];
    if (mymp[tm] == 0) {
      cnt[i]++;
      mymp[tm] = 1;
    }
  }
  long long ans = 0;
  for (long long i = 1; i <= n; i++) {
    long long tm = ara[i];
    if (arek[tm] == 1) continue;
    arek[tm] = 1;
    ans += cnt[i + 1];
  }
  cout << ans << endl;
}
