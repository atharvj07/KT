#include <bits/stdc++.h>
using namespace std;
int a[7];
int main() {
  for (int i = 1; i <= 6; i++) cin >> a[i];
  if (a[4] == 0 || (a[2] == 0 && a[3] > 0)) {
    cout << "Hermione";
    return 0;
  }
  float res;
  if (a[2] == 0)
    res = a[1];
  else
    res = float(a[1]) / float(a[2]);
  res *= (float(a[3]) / float(a[4]));
  if (a[5] == 0 && a[6] == 0) {
    if (res < 1)
      cout << "Ron";
    else
      cout << "Hermione";
    return 0;
  }
  float m;
  if (a[6] == 0)
    m = a[5];
  else
    m = float(a[5]) / float(a[6]);
  if (m < 1 / res) {
    cout << "Ron";
    return 0;
  }
  cout << "Hermione";
  return 0;
}
