from collections import defaultdict
q = int(input())
for _ in range(q):
	n = int(input())
	l = [0] + list(map(int, input().split()))
	boo = defaultdict(bool)
	ans = [0] * (n + 1)
	for i in range(1, n + 1):
		z = i
		seti = set()
		while boo[z] == False:
			seti.add(z)
			boo[z] = True
			z = l[z]
		for j in seti:
			ans[j] = len(seti)
	print(*ans[1:])
