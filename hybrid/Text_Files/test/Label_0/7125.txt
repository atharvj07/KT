#include <bits/stdc++.h>
using namespace std;
string solve(int n, string s) {
  string w = "";
  for (int i = 0; i < n; i++) {
    string sub = s.substr(i, n);
    w = w + sub[i];
  }
  return w;
}
int main() {
  int t;
  cin >> t;
  for (int i = 0; i < t; i++) {
    int n;
    string s;
    cin >> n;
    cin >> s;
    string ans = solve(n, s);
    cout << ans << endl;
  }
  return 0;
}
