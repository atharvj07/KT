#include <bits/stdc++.h>
using namespace std;
const int MX = 1e5, MOD = 1e9+7;
typedef long long ll;
#define int long long
int32_t main() {
	ios::sync_with_stdio(0);cin.tie(0);
	int n, k;
	cin >> n >> k;
	vector<int> terms;
	int t;
	for(t=1;t*t<=n; t++) {
		terms.push_back(n/t);
	}
	t--;
	vector<int> arr;
	for(int i=1; i<=t; i++) {
		arr.push_back(1);
	}
	if(terms.back()-t > 0)
		arr.push_back(terms.back()-t);
	for(int i=(int)terms.size()-1; i>0; i--) {
		arr.push_back(terms[i-1]-terms[i]);
	}
	vector<vector<int>> dp(k, vector<int>(arr.size()));
	dp[0] = arr;
	int m = arr.size();
	for(int i=1; i<k; i++) {
		int acc = 0;
		for(int j=0; j<m; j++) {
			acc += dp[i-1][j];
			acc %= MOD;
			dp[i][m-1-j] = (acc*arr[m-1-j])%MOD;
		}
	}
	cout << accumulate(dp[k-1].begin(), dp[k-1].end(), 0ll) % MOD;
}
