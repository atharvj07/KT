import java.io.*;
import java.util.*;
public class j1to10 {
	public static void main(String args[]) {
		Scanner sc=new Scanner(System.in);
		long n=sc.nextLong();
		while(n-->0) {
				long a=sc.nextLong();
				long b=sc.nextLong();
				long d=Math.abs(a-b);
				System.out.println((d+9)/10);
		}
				sc.close();
	}
}
