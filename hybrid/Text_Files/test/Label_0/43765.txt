(n, m) = map(int, input().split())
v = []
for i in range(n):
	v.append(input())
dele = list(map(int, input().split()))
result = 'Yes'
choose = []
seen = [0] * n
Len = len(v[dele[0] - 1])
pattenr = list(v[dele[0] - 1])
seen[dele[0] - 1] = 1
for i in dele[1:]:
	if len(v[i - 1]) != Len:
		result = 'No'
		break
	seen[i - 1] = 1
	for (cu, j) in enumerate(v[i - 1]):
		if j != pattenr[cu]:
			pattenr[cu] = '?'
for i in range(n):
	yy = 0
	if seen[i] == 0 and len(v[i]) == Len:
		for (cu, j) in enumerate(v[i]):
			if pattenr[cu] != '?' and pattenr[cu] != j:
				break
			yy += 1
		if yy == Len:
			result = 'No'
			break
print(result)
if result != 'No':
	print(''.join(pattenr))
