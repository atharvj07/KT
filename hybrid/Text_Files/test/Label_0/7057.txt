#include <bits/stdc++.h>
using namespace std;
const int mxn = 505;
int n, cnt;
long long k, maxi, tmp, a[mxn];
;
int main() {
  cin >> n >> k;
  cin >> a[1];
  maxi = a[1];
  for (int i = 2; i <= n; i++) {
    cin >> a[i];
    if (maxi > a[i])
      cnt++;
    else {
      if (cnt >= k) {
        cout << maxi << endl;
        return 0;
      }
      maxi = a[i];
      cnt = 1;
    }
  }
  cout << maxi << endl;
}
