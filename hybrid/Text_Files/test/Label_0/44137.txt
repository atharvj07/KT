f = lambda : map(int, input().split())
(r, c, n, k) = f()
t = [[0] * (c + 1) for i in range(r + 1)]
for j in range(n):
	(y, x) = f()
	t[y][x] = 1
for y in range(r):
	p = t[y + 1]
	for x in range(c):
		p[x + 1] += p[x]
	t[y + 1] = [a + b for (a, b) in zip(p, t[y])]
s = (c * c + c) * (r * r + r) >> 2
for y in range(r):
	for x in range(c):
		for a in range(1, c - x + 1):
			for b in range(1, r - y + 1):
				if t[y + b][x + a] + t[y][x] - t[y + b][x] - t[y][x + a] >= k:
					break
				s -= 1
print(s)
