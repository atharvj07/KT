is_debug = False
from collections import deque
(n, m) = map(int, input().split())
q = deque()
q.append(m)
d = [-1] * 1000000
d[m] = 0
c = 0
while True:
	v = q.popleft()
	print(f'v={v}, q={q}') if is_debug else ''
	if v == n:
		c = d[v]
		break
	elif v < 0:
		continue
	v1 = v // 2
	v2 = v + 1
	if v1 < 1000000 and d[v1] == -1 and (v1 * 2 == v):
		d[v1] = d[v] + 1
		q.append(v1)
	if v2 < 1000000 and d[v2] == -1:
		d[v2] = d[v] + 1
		q.append(v2)
print(f'{c}')
