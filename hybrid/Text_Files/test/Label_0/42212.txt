def funcao(lista):
	tam = len(lista)
	maior = 1
	i = 0
	j = i + 1
	while i < tam and j < tam:
		if int(lista[j]) - int(lista[i]) > 5:
			i += 1
		else:
			j += 1
			maior = max(maior, j - i)
	return maior
input()
lista = input().split()
lista = list(map(int, lista))
lista.sort()
print(funcao(lista))
