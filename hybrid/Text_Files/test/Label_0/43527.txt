entrada = input().split(' ')
n = int(entrada[0])
m = int(entrada[1])
k = int(entrada[2])
lista = []
nodes = list(map(int, input().split(' ')))
for i in range(n):
	lista.append([])
for i in range(m):
	valores = input().split(' ')
	u = int(valores[0])
	v = int(valores[1])
	lista[u - 1].append(v - 1)
	lista[v - 1].append(u - 1)
usado = [False] * n
parc1 = 0
parc2 = -1

def calc(valor):
	global x, y
	x += 1
	usado[valor] = True
	y += len(lista[valor])
	for i in lista[valor]:
		if not usado[i]:
			calc(i)
for i in range(k):
	x = 0
	y = 0
	calc(nodes[i] - 1)
	parc1 += (x * (x - 1) - y) // 2
	parc2 = max(parc2, x)
	n -= x
aux3 = 0
lenght = len(lista)
for i in range(lenght):
	if not usado[i]:
		aux3 += len(lista[i])
parc1 += ((parc2 + n) * (n + parc2 - 1) - parc2 * (parc2 - 1)) // 2
maximumNumber = parc1 - aux3 // 2
print(maximumNumber)
