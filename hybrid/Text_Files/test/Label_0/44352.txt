import math
letters = ['a', 'b', 'c', 'd', 'e', 'f', 'g', 'h', 'i', 'j', 'k', 'l', 'm', 'n', 'o', 'p', 'q', 'r', 's', 't', 'u', 'v', 'w', 'x', 'y', 'z']
t = int(input())
for _ in range(t):
	(p, a, b, c) = map(int, input().split())
	r = []
	for item in [a, b, c]:
		if p > item:
			if p % item == 0:
				r.append(0)
			else:
				r.append(item - p % item)
		elif p < item:
			r.append(item - p)
		else:
			r.append(0)
	print(min(r))
