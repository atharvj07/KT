#include<iostream>
using namespace std;
string s[55];
int n,m;
bool vis[4][55][55];
int dx[4]={0,-1,0,1},dy[4]={1,0,-1,0};
int main()
{
	cin>>n>>m;
	int x,y,r;
	for(int i=0;i<n;i++)
	{
		cin>>s[i];
		for(int j=0;j<m;j++)
		{
			if(s[i][j]=='^')
			{
				x=i,y=j,r=1;
			}
			else if(s[i][j]=='<')
			{
				x=i,y=j,r=2;
			}
			else if(s[i][j]=='v')
			{
				x=i,y=j,r=3;
			}
			else if(s[i][j]=='>')
			{
				x=i,y=j,r=0;
			}
		}
	}
	do{
		vis[r][x][y]=1;
		for(int dr=3;dr<7;dr++)
		{
			int nr=(r+dr)%4;
			int tx=x+dx[nr],ty=y+dy[nr];
			if(tx<0||ty<0||tx>=n||ty>=m||s[tx][ty]=='#')continue;
			r=nr;
			x=tx;
			y=ty;
			break;
		}
	}while(s[x][y]!='G'&&!vis[r][x][y]);
	if(s[x][y]!='G')
	{
		cout<<-1<<endl;
	}
	else
	{
		int cnt=0;
		for(int i=0;i<n;i++)for(int j=0;j<m;j++)
		{
			bool flag=0;
			for(int r=0;r<4;r++)flag|=vis[r][i][j];
			cnt+=flag;
		}
		cout<<cnt+1<<endl;
	}
}
