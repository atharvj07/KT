def main():
	input()
	(info, dir_c) = (0, -1)
	a = list(map(int, input().split()))
	while a:
		temp = []
		dir_c += 1
		for aa in a:
			if aa <= info:
				info += 1
			else:
				temp.insert(0, aa)
		a = temp
	print(dir_c)

def main2():
	input()
	(kn, res) = (0, -1)
	aa = list(map(int, input().split()))[::-1]
	while aa:
		bb = []
		res += 1
		for a in reversed(aa):
			if a <= kn:
				kn += 1
			else:
				bb.append(a)
		aa = bb
	print(res)
main()
