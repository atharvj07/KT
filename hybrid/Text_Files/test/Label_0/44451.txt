testCases = int(input())
for _ in range(testCases):
	(a1, b1) = map(int, input().split())
	(a2, b2) = map(int, input().split())
	if a1 > b1:
		if a1 == b1 + a2 and a1 == b2 or (a1 == b1 + b2 and a1 == a2):
			print('YES')
		else:
			print('NO')
	elif b1 == a1 + a2 and b1 == b2 or (b1 == a1 + b2 and b1 == a2):
		print('YES')
	else:
		print('NO')
