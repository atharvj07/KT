#include <bits/stdc++.h>
using namespace std;
int Read() {
  char c;
  while (c = getchar(), (c != '-') && (c < '0' || c > '9'))
    ;
  bool neg = (c == '-');
  int ret = (neg ? 0 : c - 48);
  while (c = getchar(), c >= '0' && c <= '9') ret = ret * 10 + c - 48;
  return neg ? -ret : ret;
}
const int MAXN = 20005;
const long double eps = 1e-10, pi = acosl(-1);
struct Event {
  int tp;
  long double rad;
} b[1000005];
int N, D, n, ans;
int sign(double x) { return x > eps ? 1 : (x < -eps ? -1 : 0); }
bool cmp(Event x, Event y) {
  if (sign(x.rad - y.rad) == -1) return 1;
  return sign(x.rad - y.rad) == 0 && x.tp > y.tp;
}
void init() { scanf("%d%d", &N, &D); }
void work() {
  int sum = 0;
  for (int i = 1; i <= N; i++) {
    int x = Read(), y = Read(), r = Read();
    long double dist = sqrtl(((x) * (x)) + ((y) * (y)));
    int bg = 1 + (dist - r - eps) / D, en = (dist + r) / D;
    long double cr = atan2(y, x);
    if (cr < -eps) cr += 2 * pi;
    for (int kc = bg; kc <= en; kc++) {
      int cd = kc * D, cs = ((cd) * (cd));
      long double wr =
          (((x) * (x)) + ((y) * (y)) + cs - ((r) * (r))) * 0.5 / dist / cd;
      wr = acos(wr);
      b[++n].rad = cr - wr, b[n].tp = 1;
      b[++n].rad = cr + wr, b[n].tp = -1;
    }
  }
  sort(b + 1, b + n + 1, cmp);
  for (int i = 1; i <= n; i++) {
    sum += b[i].tp;
    ans = max(ans, sum);
  }
  printf("%d\n", ans);
}
int main() {
  init();
  work();
  return 0;
}
