#include <bits/stdc++.h>
using namespace std;
const int siz = 1e5 + 7;
const int mod = 1e9 + 7;
int dp[siz];
int a[siz];
int n;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> n;
  dp[0] = 1;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    vector<int> v;
    for (int k = 1; k <= sqrt(a[i]); k++) {
      if (a[i] % k == 0) {
        v.push_back(k);
        if (a[i] / k != k) v.push_back(a[i] / k);
      }
    }
    sort(v.begin(), v.end(), greater<int>());
    for (int k = 0; k < int(v.size()); k++) {
      if (v[k] <= n) {
        dp[v[k]] += dp[v[k] - 1];
        dp[v[k]] %= mod;
      }
    }
  }
  long long res = 0;
  for (int i = 1; i <= n; i++) {
    res += dp[i];
    res %= mod;
  }
  cout << res << endl;
}
