#include <bits/stdc++.h>
using namespace std;
const int inf = 0x3f3f3f3f, mod = 1e9 + 7;
const int maxn = 2e5 + 100;
const long double PI = acos(-1.0);
int gcd(int a, int b) { return (a ? gcd(b % a, a) : b); }
long long power(long long a, long long n) {
  long long p = 1;
  while (n > 0) {
    if (n % 2) {
      p = p * a;
    }
    n >>= 1;
    a *= a;
  }
  return p;
}
long long power(long long a, long long n, long long mod) {
  long long p = 1;
  while (n > 0) {
    if (n % 2) {
      p = p * a;
      p %= mod;
    }
    n >>= 1;
    a *= a;
    a %= mod;
  }
  return p % mod;
}
int n, m, k;
long long b, c;
long long a[maxn];
int main() {
  while (cin >> n >> k >> b >> c) {
    for (int i = 1; i <= n; i++) scanf("%lld", &a[i]);
    if (c * 5 <= b) b = c * 5;
    sort(a + 1, a + 1 + n);
    long long minn = 1e18;
    for (int j = 0; j < 5; j++) {
      queue<int> qu[10];
      long long sum = 0;
      for (int i = 1; i <= k; i++) {
        qu[(a[i] % 5 + 5) % 5].push(a[i]);
        int d = a[k] + (5 + j - a[k] % 5) % 5;
        sum += (d - a[i]) / 5 * b + (d - a[i]) % 5 * c;
      }
      minn = min(minn, sum);
      for (int i = k + 1; i <= n; i++) {
        long long be = a[i - 1] + (5 + j - a[i - 1] % 5) % 5,
                  now = a[i] + (5 + j - a[i] % 5) % 5;
        long long d = (now - be) / 5 * b;
        sum += d * k + (5 + j - a[i] % 5) % 5 * c;
        long long mm = sum;
        int kk;
        for (int s = 0; s < 5; s++)
          if (!qu[s].empty()) {
            int g = qu[s].front();
            if (mm > sum - (now - g) / 5 * b - (now - g) % 5 * c) {
              mm = sum - (now - g) / 5 * b - (now - g) % 5 * c;
              kk = s;
            }
          }
        sum = mm;
        qu[kk].pop();
        qu[(a[i] % 5 + 5) % 5].push(a[i]);
        minn = min(minn, sum);
      }
    }
    cout << minn << endl;
  }
  return 0;
}
