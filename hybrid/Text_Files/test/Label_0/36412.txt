import java.io.*;
import java.util.*;

public class P215C
{
    public static void main(String[] args)
    {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int s = sc.nextInt();
        for (int i = 0; i < memo.length; i++) Arrays.fill(memo[i], -1);
        if (s % 2 == 0)
        {
            System.out.println("0");
            return;
        }
        long sum = 0;
        for (int i = 1; i <= n; i+=2)
        {
            for (int j = 1; j <= m; j+=2)
            {
                if (i * j < s) continue;
                int diff = (i * j) - s;
                if (diff % 4 != 0) continue;
                long vol = (n-i+1) * (m-j+1);
                long con = 0;
                if (diff == 0)
                {
                    long cd = ((i/2+1) * (j/2+1) * 2) - 1;
                    con = cd * vol;
                }
                else
                {
                    int quart = diff / 4;
                    long cd = count(quart, i/2, j/2);
                    con = cd * vol * 2;
                }
                sum += con;
                //System.out.printf("Contribution to %dx%d is %d\n", i, j, con);
            }
        }
        System.out.println(sum);
    }
    
    static long[][] memo = new long[501][501];

    public static long countInside(int h, int w)
    {
        if (memo[h][w] != -1) return memo[h][w];
        if (memo[w][h] != -1) return memo[w][h];
        long sum = 0;
        for (int i = 1; i <= h; i+=2)
        {
            for (int j = 1; j <= w; j+=2)
            {
                
            }
        }
        return sum;
    }
    
    public static long count(int size, int h, int w)
    {
        if (size == 0 || h == 0 || w == 0) return 0;
        if (h * w < size) return 0;
        int max = (int)Math.floor(Math.sqrt(size));
        int mind = Math.min(h, w);
        int maxd = Math.max(h, w);
        max = Math.min(max, maxd);
        int count = 0;
        for (int i = 1; i <= max; i++)
        {
            if (size % i != 0) continue;
            int k = size / i;
            if (Math.max(i, k) > maxd || Math.min(i, k) > mind) continue;
            if (i == k)
            {
                count++;
            }
            else if (Math.max(i, k) <= mind)
            {
                count+=2;
            }
            else
            {
                count++;
            }
        }
        //System.out.printf("In %dx%d space there are %d of size %d\n", h, w, count, size);
        return count;
    }
}
