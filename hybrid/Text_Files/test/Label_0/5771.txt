#include <bits/stdc++.h>
int arba[4 * 200000], arbb[4 * 200000];
int n, k, a, b;
inline int min2(int a, int b) { return a < b ? a : b; }
void upd(int p, int st, int dr, int x, int y) {
  if (st == dr) {
    arba[p] += y;
    arba[p] = min2(arba[p], a);
    arbb[p] += y;
    arbb[p] = min2(arbb[p], b);
  } else {
    int m = (st + dr) / 2;
    if (x <= m)
      upd(2 * p + 1, st, m, x, y);
    else
      upd(2 * p + 2, m + 1, dr, x, y);
    arba[p] = arba[2 * p + 1] + arba[2 * p + 2];
    arbb[p] = arbb[2 * p + 1] + arbb[2 * p + 2];
  }
}
int suma(int *v, int p, int st, int dr, int x, int y) {
  if (st >= x && dr <= y)
    return v[p];
  else {
    int rez = 0, m = (st + dr) / 2;
    if (m >= x) rez += suma(v, 2 * p + 1, st, m, x, y);
    if (m < y) rez += suma(v, 2 * p + 2, m + 1, dr, x, y);
    return rez;
  }
}
int main() {
  int q, i, t, x, y, rez;
  scanf("%d%d%d%d%d", &n, &k, &a, &b, &q);
  for (i = 0; i < q; i++) {
    scanf("%d", &t);
    if (t == 1) {
      scanf("%d%d", &x, &y);
      x--;
      upd(0, 0, n - 1, x, y);
    } else {
      scanf("%d", &x);
      x--;
      rez = 0;
      if (x != 0) rez += suma(arbb, 0, 0, n - 1, 0, x - 1);
      if (x + k < n) rez += suma(arba, 0, 0, n - 1, x + k, n - 1);
      printf("%d\n", rez);
    }
  }
  return 0;
}
