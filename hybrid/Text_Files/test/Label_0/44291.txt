entrada = input()
iguais = []
x = 0
for s in range(len(entrada) - 1):
	iguais.append(x)
	if entrada[s] == entrada[s + 1]:
		x += 1
iguais.append(x)
queries = int(input())
final = []
for q in range(queries):
	res = 0
	(a, b) = map(int, input().split())
	res = iguais[b - 1] - iguais[a - 1]
	final.append(res)
for r in final:
	print(r)
