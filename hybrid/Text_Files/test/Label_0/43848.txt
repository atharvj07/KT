for t in range(int(input())):
	n = int(input())
	l = list(map(int, input().split()))
	oc = n
	ec = n
	min_o = float('inf')
	min_e = float('inf')
	res = 0
	ans = float('inf')
	for i in range(n):
		if i % 2 == 0:
			res += l[i]
			ec -= 1
			min_e = min(min_e, l[i])
			ans = min(ans, res + ec * min_e + oc * min_o)
		else:
			res += l[i]
			oc -= 1
			min_o = min(min_o, l[i])
			ans = min(ans, res + ec * min_e + oc * min_o)
	print(ans)
