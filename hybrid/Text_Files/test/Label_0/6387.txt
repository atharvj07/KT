#include <bits/stdc++.h>
using namespace std;
int p[1005];
int main() {
  int T, n, k, h, i;
  scanf("%d", &T);
  while (T--) {
    scanf("%d%d%d", &h, &k, &n);
    for (i = 0; i < n; i++) {
      scanf("%d", &p[i]);
    }
    int e = 0;
    while (1) {
      if (k + e <= h) {
        int q = k + e;
        for (i = 0; i < n; i++) {
          if (p[i] == q) break;
        }
        if (i == n) {
          break;
        }
      }
      if (e != 0) {
        if (k - e >= 1) {
          int q = k - e;
          for (i = 0; i < n; i++) {
            if (p[i] == q) break;
          }
          if (i == n) {
            break;
          }
        }
      }
      e++;
    }
    printf("%d\n", e);
  }
  return 0;
}
