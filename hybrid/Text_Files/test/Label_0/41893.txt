t = int(input())
for _ in range(t):
	s = input()
	q = s.count('?')
	mn = 9
	for i in range(2 ** q):
		temp = []
		for j in range(q):
			if 2 ** j & i != 0:
				temp += ['1']
			else:
				temp += ['0']
		x = list(map(lambda x: temp.pop(0) if x == '?' else x, s))
		team = [0, 0]
		for j in range(10):
			if x[j] == '1':
				team[j % 2] += 1
			if j % 2 == 0:
				if team[0] > team[1] + 5 - j // 2 or team[1] > team[0] + 5 - (j // 2 + 1):
					mn = min(mn, j)
					break
			elif team[1] > team[0] + 5 - (j // 2 + 1) or team[0] > team[1] + 5 - (j // 2 + 1):
				mn = min(mn, j)
				break
	print(mn + 1)
