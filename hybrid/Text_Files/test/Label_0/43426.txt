for i in range(int(input())):
	(W, H) = list(map(int, input().split()))
	(x1, y1, x2, y2) = list(map(int, input().split()))
	(w, h) = list(map(int, input().split()))
	if w + (x2 - x1) > W and h + (y2 - y1) > H:
		print(-1)
	elif x1 >= w or W - x2 >= w or y1 >= h or (H - y2 >= h):
		print(0)
	else:
		print(min(w - x1 if w + (x2 - x1) <= W else 1000000000, w - W + x2 if w + (x2 - x1) <= W else 1000000000, h - y1 if h + (y2 - y1) <= H else 1000000000, h - H + y2 if h + (y2 - y1) <= H else 1000000000))
