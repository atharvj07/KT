#include <bits/stdc++.h>

#define fi first
#define se second
#define mp make_pair
#define pb push_back

using namespace std;
typedef long long ll;

const int maxn = 2e5 + 5;

array<int, 3> edge[maxn];
int N, lab[maxn]; ll res[maxn];
int info[maxn], par[maxn], sz[maxn];

int finds(int u)
{
  if (lab[u] == u) return u;
  return lab[u] = finds(lab[u]);
}

signed main(void)
{
  ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
  if (fopen("A.INP", "r")){
    freopen("A.INP", "r", stdin);
    freopen("A.OUT", "w", stdout);
  }
  cin >> N;
  for (int i = 1; i < N; ++i){
    cin >> edge[i][1] >> edge[i][2] >> edge[i][0];
  }
  sort(edge + 1, edge + N);
  for (int i = 1; i <= N; ++i){
    lab[i] = info[i] = i; sz[i] = 1;
  }
  int cnt = N + 1;
  for (int i = N - 1; i >= 1; --i){
    int u = edge[i][1], v = edge[i][2], w = edge[i][0];
    u = finds(u); v = finds(v);
    par[info[u]] = cnt;
    par[info[v]] = cnt;
    res[info[u]] = 1ll * sz[v] * w;
    res[info[v]] = 1ll * sz[u] * w;
    info[v] = cnt;
    sz[v] += sz[u];
    lab[u] = v;
    ++cnt;
  }
  for (int i = cnt; i >= 1; --i){
    res[i] += res[par[i]];
  }
  for (int i = 1; i <= N; ++i){
    cout << res[i] << '\n';
  }
}
