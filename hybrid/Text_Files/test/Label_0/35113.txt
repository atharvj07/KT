import java.util.*;

public class StrangePartition{
  public static void main(String[] args){
    Scanner gd = new Scanner(System.in);
    int tt = IO.ri();
    int n;
    long x;
    long[] arr;
    long min,max;
    long sum;
    long val;
    while(tt>0){
      n = IO.ri();
      x = IO.rl();
      sum = 0;
      min = 0;
      max = 0;
      for(int i=0;i<n;i++){
        val = IO.rl();
        sum+=val;
        min+=(long)Math.ceil(val/(x*1.0));
      }
      max = (long)Math.ceil(sum/(x*1.0));
      if(min<max){
        IO.pf("%d %d\n",min,max);
      }
      else{
        IO.pf("%d %d\n",max,min);
      }
      tt--;
    }
  }
}

class UT{
  public static long fact(long k){
    if(k==1){
      return 1;
    }
    else{
      return k*fact(k-1);
    }
  }
  public static HashMap<Integer,Integer> gen_primes(int upper_bound){
    int[] arr = new int[upper_bound];
    int ptr = 0;
    boolean flag;
    int next_ptr=1;
    for(int i=0;i<upper_bound;i++){
      arr[i] = i+2;
    }
    while(next_ptr!=0){
      flag = true;
      next_ptr = 0;
      for(int j=ptr+1;j<arr.length;j++){
        if(((arr[ptr]*arr[ptr])>upper_bound)){
          break;
        }
        if(arr[j]!=0&&flag==true){
          flag = false;
          next_ptr = j;
        }
        if(arr[j]%arr[ptr]==0){
          arr[j] = 0;
        }
      }
      ptr = next_ptr;
    }
    HashMap<Integer,Integer> prime_nums = new HashMap<Integer,Integer>();
    for(int p=0;p<arr.length;p++){
      if(arr[p]!=0){
        prime_nums.put(arr[p],arr[p]);
      }
    }
    return prime_nums;
  }
}
