from math import *
n = int(input())
a = []
for i in range(n):
	a.append(input().split()[1])
x = a.count('soft')
y = a.count('hard')
if (abs(x - y) == 1 or x == y) and int(sqrt(n) + 0.5) ** 2 == n:
	print(int(sqrt(n)))
else:
	ma = max(x, y)
	t = ma * 2
	if t - 1 < n:
		print(ceil(sqrt(t)))
	else:
		print(min(ceil(sqrt(t)), ceil(sqrt(t - 1))))
