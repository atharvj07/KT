#include<map>
#include<cstdio>
#include<string>
#include<vector>

#define	mp		make_pair
#define	pb		push_back

using namespace std;

typedef	pair<int,int>	pii;

const int inf=1<<29;

int main(){
	for(int n;scanf("%d",&n),n;){
		int unum=0;
		pair<pii,int> edge[100];
		map<string,int> name;
		for(int i=0;i<n;i++){
			int pw;
			char u1[17],u2[17];
			scanf("%*d %s %*c %*d%*c%d %s",u1,&pw,u2);

			int id1,id2;
			if(name.count(u1)==0)	name[u1]=unum++;
			id1=name[u1];
			if(name.count(u2)==0)	name[u2]=unum++;
			id2=name[u2];

			edge[i]=mp(mp(id1,id2),pw);
		}

		bool ok=true;
		static int wf[200][200];
		for(int i=0;i<unum;i++)for(int j=0;j<unum;j++)	wf[i][j]=(i==j?0:inf);
		for(int k=0;k<n;k++){
			int from=edge[k].first.first,to=edge[k].first.second,dis=edge[k].second;
			if(wf[from][to]==inf){
				wf[from][to]=dis;
				wf[to][from]=-dis;
				for(int i=0;i<unum;i++)for(int j=0;j<unum;j++){
					if(wf[i][from]==inf || wf[to][j]==inf)	continue;
					wf[i][j]=min(wf[i][j],wf[i][from]+dis+wf[to][j]);
					wf[j][i]=-wf[i][j];
				}
			}
			else{
				if(wf[from][to]!=dis){
					ok=false;
					break;
				}
			}
		}
		puts(ok?"Yes":"No");
	}

	return 0;
}
