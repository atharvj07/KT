LL gcd(LL a, LL b) {
	if (b == 0 || a == 0) return a + b;
	if (a % b == 0) return b;
	return gcd(b, a % b);
}

LL w, h, k;
LL Calc(LL w, LL h) {
	LL ans = 0;
	for (LL x = 1; x < w; x++) {
		for (LL y1 = 1; y1 < h; y1++) {
			LL R = 2 * k - 2 - y1 * (w - x) + gcd(x, y1);
			if (R + w * 2 < 0) break;
			// xy2 - g(w - x, y2) - g(w, y2 - y1) <= R
			LL mn = R / x;
			mn = MAX(mn, 0);
			LL mx = (R + 2 * w) / x;
			mx = MIN(mx, h - 1);
			ans += MIN(mn, h - 1);
			for (LL y2 = mn + 1; y2 <= mx; y2++) {
				LL L = x * y2 - gcd(w - x, y2) - gcd(w, ABS(y2 - y1));
				ans += (L <= R);
			}
		}
	}
	return ans;
}
