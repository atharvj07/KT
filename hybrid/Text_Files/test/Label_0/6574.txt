#include <bits/stdc++.h>
using namespace std;
int n;
string s;
string full(string s) {
  string res;
  bool ok = true;
  while (ok) {
    ok = false;
    int pp = -1;
    for (int i = 0; i <= s.size(); i++) {
      if (i == s.size() || s[i] == ':') {
        if (i - pp < 5 && i - pp != 1) {
          s.insert(s.begin() + (pp + 1), '0');
          ok = true;
          break;
        }
        pp = i;
      }
    }
  }
  for (int i = 0; i < s.size() - 1; i++) {
    if (s[i] == ':' && s[i + 1] == ':') {
      i++;
      s.insert(s.begin() + i, '0');
      s.insert(s.begin() + i, '0');
      s.insert(s.begin() + i, '0');
      s.insert(s.begin() + i, '0');
      while (s.size() < 39) {
        s.insert(s.begin() + i, ':');
        s.insert(s.begin() + i, '0');
        s.insert(s.begin() + i, '0');
        s.insert(s.begin() + i, '0');
        s.insert(s.begin() + i, '0');
      }
      break;
    }
  }
  if (s[0] == ':') s.erase(s.begin());
  if (s[s.size() - 1] == ':') s.erase(s.end() - 1);
  return s;
}
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> s;
    cout << full(s) << endl;
  }
  return 0;
}
