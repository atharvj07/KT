#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  string s, t;
  cin >> s >> t;
  int n = s.size();
  sort(s.begin(), s.end());
  sort(t.rbegin(), t.rend());
  s = s.substr(0, (n + 1) / 2);
  t = t.substr(0, n / 2);
  int sb = 0, tb = 0, ub = 0, sf = s.size() - 1, tf = t.size() - 1, uf = n - 1;
  string u(n, '*');
  for (int i = 1; i <= n; ++i) {
    if (i % 2) {
      if (s[sb] < t[tb]) {
        u[ub++] = s[sb++];
      } else {
        u[uf--] = s[sf--];
      }
    } else {
      if (t[tb] > s[sb]) {
        u[ub++] = t[tb++];
      } else {
        u[uf--] = t[tf--];
      }
    }
  }
  cout << u << endl;
  return 0;
}
