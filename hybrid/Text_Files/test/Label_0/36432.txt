import java.util.*;
import java.io.*;
import java.lang.*;

public class Sketch {

    public static void main(String[] args) {

        Scanner scan = new Scanner(new BufferedInputStream(System.in));
        int t = scan.nextInt();
        while (t-->0) {
            int n = scan.nextInt();
            char[] a = scan.next().toCharArray();
            char[] b = scan.next().toCharArray();

            Integer[] ab = new Integer[n];
            for (int i=0; i<n; ++i) ab[i] = i;
            Arrays.sort(ab, (x, y) -> (a[x] - a[y]) == 0 ? b[x] - b[y] : a[x] - a[y]);

            int[] parent = new int[20];
            for (int i=0; i<20; ++i) {
                parent[i] = i;
            }

            int ans = 0;
            for (int i=0; i<n; ++i) {
                int j = ab[i];
                if (a[j] > b[j]) {
                    ans = -1;
                    break;
                } else if (a[j] < b[j]) {
                    int x = a[j]-'a';
                    int y = b[j]-'a';
                    if (union(x, y, parent)) {
                        ++ans;
                    }
                }
            }
            System.out.println(ans);
        }
    }


    private static int find(int x, int[] parent) {
        if (x != parent[x]) {
            parent[x] = find(parent[x], parent);
        }
        return parent[x];
    }


    private static boolean union(int x, int y, int[] parent) {

        int xroot = find(x, parent);
        int yroot = find(y, parent);
        if (xroot != yroot) {
            parent[yroot] = xroot;
            return true;
        }
        return false;
    }
}
