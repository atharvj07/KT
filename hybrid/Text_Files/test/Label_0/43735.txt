for _ in range(int(input())):
	n = int(input())
	seq = [int(x) for x in input().split()]
	sign = seq[0] > 0
	mmax = 0
	mmin = float('-inf')
	ans = 0
	for val in seq:
		if val > 0 and sign:
			mmax = max(mmax, val)
		elif val < 0 and (not sign):
			mmin = max(mmin, val)
		elif val > 0 and (not sign):
			ans += mmin
			mmin = float('-inf')
			sign = True
			mmax = val
		elif val < 0 and sign:
			ans += mmax
			mmax = 0
			sign = not True
			mmin = val
	if seq[-1] > 0:
		ans += mmax
	else:
		ans += mmin
	print(ans)
