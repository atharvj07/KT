import itertools

def main(inp):
	(n, m, k) = split_inp_int(inp)
	grid = {}
	for i in range(n):
		row = inp()
		for j in range(m):
			grid[i, j] = row[j]
	water = {(i, j) for (i, j) in itertools.product(range(n), range(m)) if grid[i, j] == '.'}

	def is_connected(grid):
		water_left = {e for e in water}
		first_water = water_left.pop()
		connected = {first_water}
		queue = {first_water}
		while queue:
			(vx, vy) = queue.pop()
			if len(connected) < len(water) - k:
				connected.add((vx, vy))
			else:
				break
			for (dx, dy) in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
				consider = (vx + dx, vy + dy)
				if consider in water_left:
					queue.add(consider)
					water_left.remove(consider)
		return connected
	connected = is_connected(grid)
	for i in range(n):
		for j in range(m):
			if grid[i, j] == '.' and (i, j) not in connected:
				grid[i, j] = 'X'
	[print(''.join([grid[i, j] for j in range(m)])) for i in range(n)]

def split_inp_int(inp):
	return list(map(int, inp().split()))

def use_fast_io():
	import sys

	class InputStorage:

		def __init__(self, lines):
			lines.reverse()
			self.lines = lines

		def input_func(self):
			if self.lines:
				return self.lines.pop()
			else:
				return ''
	input_storage_obj = InputStorage(sys.stdin.readlines())
	return input_storage_obj.input_func

	return reduce(operator.mul, arr_, 1)
main(use_fast_io())
