from sys import stdin
from heapq import *
for _ in range(int(input())):
	n = int(input())
	heap = []
	for (j, i) in enumerate(map(int, input().split())):
		if i != 0:
			heap.append([-i, j + 1])
	heapify(heap)
	j = n - 1
	ans = []
	while heap:
		curr = heappop(heap)
		while curr[0] and heap:
			nxt = heappop(heap)
			nxt[0] += 1
			curr[0] += 1
			ans.append(f'{nxt[1]} {curr[1]}')
			if nxt[0]:
				heappush(heap, nxt)
	print(len(ans))
	print(*ans, sep='\n')
