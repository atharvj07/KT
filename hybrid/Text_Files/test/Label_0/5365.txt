#include <bits/stdc++.h>
using namespace std;
int n, k;
double s[12][12];
int main() {
  int n, t, i, j, k, ans = 0;
  scanf("%d%d", &n, &t);
  for (i = 1; i <= t; i++) {
    s[1][1] += 1.0;
    for (j = 1; j <= n; j++)
      for (k = 1; k <= j; k++)
        if (s[j][k] > 1.0) {
          s[j + 1][k] += (s[j][k] - 1.0) / 2;
          s[j + 1][k + 1] += (s[j][k] - 1.0) / 2;
          s[j][k] = 1.0;
        }
  }
  for (j = 1; j <= n; j++) {
    for (k = 1; k <= j; k++) {
      if (s[j][k] == 1.0) ans++;
    }
  }
  printf("%d\n", ans);
  return 0;
}
