def check(midd, i):
	if midd * n + i >= arr[i]:
		return True
	return False
n = int(input())
arr = list(map(int, input().split()))
pre_cnt = 10 ** 18
ans = 10 ** 18
for i in range(n):
	l = 0
	r = 10 ** 9
	cnt = 10 ** 18
	while l <= r:
		mid = (l + r) // 2
		if check(mid, i):
			cnt = mid
			r = mid - 1
		else:
			l = mid + 1
	if cnt < pre_cnt:
		pre_cnt = cnt
		ans = i
print(ans + 1)
