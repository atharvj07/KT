t = int(input())
for _ in range(t):
	(n, m) = map(int, input().split())
	p = list(map(int, input().split()))
	c = [0] * n
	for i in range(n):
		c[(i + 1 - p[i]) % n] += 1
	res = []
	for i in range(n):
		if c[i] >= n - 2 * m:
			swaps = 0
			norm = [0] * n
			rev_map = [0] * n
			for j in range(n):
				norm[j] = (p[j] + i - 1) % n
				rev_map[norm[j]] = j
			for j in range(n):
				if norm[j] != j:
					k = norm[j]
					ind_j = rev_map[j]
					(norm[ind_j], norm[j]) = (k, j)
					(rev_map[k], rev_map[j]) = (ind_j, j)
					swaps += 1
			if swaps <= m:
				res.append(i)
	print(len(res), *res)
