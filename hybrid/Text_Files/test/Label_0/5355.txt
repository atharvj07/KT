#include <bits/stdc++.h>
int main() {
  int T, N;
  std::vector<int> A;
  std::cin >> T;
  for (int t = 0; t < T; t++) {
    std::cin >> N;
    A.resize(N);
    for (int i = 0; i < N; i++) std::cin >> A[i];
    int zeros = 0, sum = 0;
    for (int i = 0; i < N; i++) {
      if (A[i] == 0) zeros++;
      sum += A[i];
    }
    int moves = zeros;
    sum += moves;
    if (sum == 0) moves++;
    std::cout << moves << "\n";
  }
}
