t = int(input())
for _ in range(t):
	n = int(input())
	flag = False
	cnt = 0
	if n == 1:
		print('0')
		continue
	while n != 1:
		if n % 6 == 0:
			n = n // 6
			cnt = cnt + 1
		else:
			n = n * 2
			if n % 6 != 0:
				flag = True
				break
			cnt = cnt + 1
	if flag == True:
		print('-1')
	else:
		print(cnt)
