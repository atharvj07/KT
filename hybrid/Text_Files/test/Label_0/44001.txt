ni = list(map(int, input().strip().split(' ')))
n = ni[0]
f = ni[1]
s = 0
li = []
for i in range(n):
	arr = list(map(int, input().strip().split(' ')))
	k = arr[0]
	l = arr[1]
	s = s + min(l, k)
	if l > k:
		li.append(min(k, l - k))
li.sort(reverse=True)
if f >= len(li):
	s = s + sum(li)
else:
	s = s + sum(li[:f])
print(s)
