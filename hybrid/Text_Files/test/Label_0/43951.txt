from collections import deque
import sys
input = sys.stdin.readline

def bfs():
	q = deque()
	q.append(1)
	p0 = []
	while q:
		i = q.popleft()
		p0.append(i)
		for j in G[i]:
			q.append(j)
	childcnt = [1] * (n + 1)
	while p0:
		i = p0.pop()
		if i == 1:
			break
		childcnt[p[i - 2]] += childcnt[i]
	return childcnt

def euler_tour():
	s = []
	s.append(1)
	now = [0] * (n + 1)
	t = 1
	(t1, t2) = ([-1] * (n + 1), [-1] * (n + 1))
	v = []
	while s:
		i = s[-1]
		if t1[i] == -1:
			v.append(i)
			t1[i] = t
			t += 1
		if now[i] == len(G[i]):
			t2[i] = t
			t += 1
			s.pop()
		else:
			s.append(G[i][now[i]])
			now[i] += 1
	return (t1, t2, v)
(n, q) = map(int, input().split())
p = list(map(int, input().split()))
G = [[] for _ in range(n + 1)]
for i in range(n - 1):
	G[p[i]].append(i + 2)
childcnt = bfs()
(t1, t2, v) = euler_tour()
r = dict()
for i in range(n):
	r[v[i]] = i
for _ in range(q):
	(u, k) = map(int, input().split())
	ans = -1 if childcnt[u] < k else v[r[u] + k - 1]
	print(ans)
