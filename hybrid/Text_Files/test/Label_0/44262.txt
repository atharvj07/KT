T = int(input())
for _ in range(T):
	w = input()
	x = int(input())
	n = len(w)
	l = ['1' for k in range(n)]
	for i in range(n):
		if w[i] == '0':
			if i - x >= 0:
				l[i - x] = '0'
			if i + x < n:
				l[i + x] = '0'
	flag = 0
	for i in range(n):
		if i + x < n and l[i + x] == '1':
			if w[i] == '0':
				flag += 1
		elif i - x >= 0 and l[i - x] == '1':
			if w[i] == '0':
				flag += 1
		elif w[i] == '1':
			flag += 1
	if flag != 0:
		print(-1)
	else:
		print(''.join(l))
