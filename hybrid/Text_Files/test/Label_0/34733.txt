// upsolve with rainboy
import java.io.*;
import java.util.*;

public class CF962E {
	static int[] xx;
	static char[] cc;
	static long[] gap;
	static long span(int n, char c) {
		int l;
		for (l = 0; l < n; l++)
			if (cc[l] == c)
				break;
		if (l == n)
			return 0;
		int r;
		for (r = n - 1; r >= 0; r--)
			if (cc[r] == c)
				break;
		return xx[r] - xx[l];
	}
	static long solve(int n, char ignore) {
		int l;
		for (l = 0; l < n; l++)
			if (cc[l] != ignore)
				break;
		if (l == n)
			return 0;
		int r;
		for (r = n - 1; r >= 0; r--)
			if (cc[r] != ignore)
				break;
		long ans = 0;
		for (int i = l; i <= r; i++)
			if (cc[i] == 'P') {
				ans += xx[i] - xx[l];
				l = i;
				break;
			}
		for (int i = r; i >= l; i--)
			if (cc[i] == 'P') {
				ans += xx[r] - xx[i];
				r = i;
				break;
			}
		int g = 0;
		for (int h = l, i = l; i <= r; i++) {
			if (cc[i] == ignore)
				continue;
			g = Math.max(g, xx[i] - xx[h]);
			h = i;
			if (cc[i] == 'P') {
				gap[i] += g;
				g = 0;
			}
		}
		return ans;
	}
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		int n = Integer.parseInt(br.readLine());
		xx = new int[n];
		cc = new char[n];
		for (int i = 0; i < n; i++) {
			StringTokenizer st = new StringTokenizer(br.readLine());
			xx[i] = Integer.parseInt(st.nextToken());
			cc[i] = st.nextToken().charAt(0);
		}
		gap = new long[n];
		boolean hasP = false;
		for (int i = 0; i < n; i++)
			if (cc[i] == 'P') {
				hasP = true;
				break;
			}
		if (!hasP) {
			System.out.println(span(n, 'B') + span(n, 'R'));
			return;
		}
		long ans = solve(n, 'B') + solve(n, 'R');
		for (int h = -1, i = 0; i < n; i++)
			if (cc[i] == 'P') {
				if (h != -1) {
					long len = xx[i] - xx[h];
					ans += Math.min(len * 2, len * 3 - gap[i]);
				}
				h = i;
			}
		System.out.println(ans);
	}
}
