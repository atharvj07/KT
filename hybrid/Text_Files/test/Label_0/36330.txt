public class x1519E
{
    static ArrayDeque<Edge>[] edges;
    public static void main(String hi[]) throws Exception
    {
        FastScanner infile = new FastScanner();
        int N = infile.nextInt();
        HashMap<String, Integer> map = new HashMap<String, Integer>();
        ArrayDeque<Integer> input = new ArrayDeque<Integer>();
        int boof = 0;
        for(int i=0; i < N; i++)
        {
            int a = infile.nextInt();
            int b = infile.nextInt();
            int c = infile.nextInt();
            int d = infile.nextInt();
            int gcd = gcd(a, b);
            a /= gcd;   b /= gcd;
            gcd = gcd(c, d);
            c /= gcd;   d /= gcd;
            String key = form(a+b, b, c, d);
            if(!map.containsKey(key))
                map.put(key, boof++);
            input.add(map.get(key));
            key = form(a, b, c+d, d);
            if(!map.containsKey(key))
                map.put(key, boof++);
            input.add(map.get(key));
        }
        N = boof;   //number of nodes in graph
        edges = new ArrayDeque[N];
        for(int v=0; v < N; v++)
            edges[v] = new ArrayDeque<Edge>();
        int inputid = 1;
        while(input.size() > 0)
        {
            int a = input.poll();
            int b = input.poll();
            edges[a].add(new Edge(b, inputid));
            edges[b].add(new Edge(a, inputid));
            inputid++;
        }
        res = new ArrayList<Integer>();
        removedEdges = new boolean[inputid+1];
        seen = new int[N];
        for(int v=0; v < N; v++)
            if(seen[v] == 0)
                dfs(v, -1);
        StringBuilder sb = new StringBuilder(res.size()/2+"\n");
        for(int i=0; i < res.size(); i+=2)
            sb.append(res.get(i)+" "+res.get(i+1)+"\n");
        System.out.print(sb);
    }
    static ArrayList<Integer> res;
    static boolean[] removedEdges;
    static int[] seen;
    public static void dfs(int curr, int par)
    {
        seen[curr] = 1;
        int leftover = -1;
        int parid = -1;
        for(Edge e: edges[curr])
        {
            if(removedEdges[e.id] || e.to == par)
            {
                if(e.to == par && !removedEdges[e.id])
                    parid = e.id;
                continue;
            }
            if((seen[e.to]&1) == 0)
            {
                if(seen[e.to] == 0)
                    dfs(e.to, curr);
                if(!removedEdges[e.id])
                {
                    if(leftover == -1)
                        leftover = e.id;
                    else
                    {
                        res.add(leftover);
                        res.add(e.id);
                        removedEdges[leftover] = removedEdges[e.id] = true;
                        leftover = -1;
                    }
                }
            }
        }
        if(leftover != -1 && parid != -1)
        {
            res.add(leftover);
            res.add(parid);
            removedEdges[leftover] = removedEdges[parid] = true;
        }
        seen[curr] = 2;
    }
    public static int gcd(int a, int b)
    {
        if(a > b)
            a = (a+b)-(b=a);
        if(a == 0)
            return b;
        return gcd(b%a, a);
    }
    public static long gcd(long a, long b)
    {
        if(a > b)
            a = (a+b)-(b=a);
        if(a == 0L)
            return b;
        return gcd(b%a, a);
    }
    public static String form(int a, int b, int c, int d)
    {
        int g = gcd(a, b);
        a /= g; b /= g;
        g = gcd(c, d);
        c /= g; d /= g;
        //convert
        long num = (long)c*b;
        long denom = (long)d*a;
        long gg = gcd(num, denom);
        num /= gg;  denom /= gg;
        return num+"/"+denom;
    }
}
class Edge
{
    public int to;
    public int id;

    public Edge(int a, int b)
    {
        to = a;
        id = b;
    }
}

class FastScanner
{
    private int BS = 1 << 16;
    private char NC = (char) 0;
    private byte[] buf = new byte[BS];
    private int bId = 0, size = 0;
    private char c = NC;
    private double cnt = 1;
    private BufferedInputStream in;

    public FastScanner() {
        in = new BufferedInputStream(System.in, BS);
    }

    public FastScanner(String s) {
        try {
            in = new BufferedInputStream(new FileInputStream(new File(s)), BS);
        } catch (Exception e) {
            in = new BufferedInputStream(System.in, BS);
        }
    }

    private char getChar() {
        while (bId == size) {
            try {
                size = in.read(buf);
            } catch (Exception e) {
                return NC;
            }
            if (size == -1) return NC;
            bId = 0;
        }
        return (char) buf[bId++];
    }

    public int nextInt() {
        return (int) nextLong();
    }

    public int[] nextInts(int N) {
        int[] res = new int[N];
        for (int i = 0; i < N; i++) {
            res[i] = (int) nextLong();
        }
        return res;
    }

    public long[] nextLongs(int N) {
        long[] res = new long[N];
        for (int i = 0; i < N; i++) {
            res[i] = nextLong();
        }
        return res;
    }

    public long nextLong() {
        cnt = 1;
        boolean neg = false;
        if (c == NC) c = getChar();
        for (; (c < '0' || c > '9'); c = getChar()) {
            if (c == '-') neg = true;
        }
        long res = 0;
        for (; c >= '0' && c <= '9'; c = getChar()) {
            res = (res << 3) + (res << 1) + c - '0';
            cnt *= 10;
        }
        return neg ? -res : res;
    }

    public double nextDouble() {
        double cur = nextLong();
        return c != '.' ? cur : cur + nextLong() / cnt;
    }

    public double[] nextDoubles(int N) {
        double[] res = new double[N];
        for (int i = 0; i < N; i++) {
            res[i] = nextDouble();
        }
        return res;
    }

    public String next() {
        StringBuilder res = new StringBuilder();
        while (c <= 32) c = getChar();
        while (c > 32) {
            res.append(c);
            c = getChar();
        }
        return res.toString();
    }

    public String nextLine() {
        StringBuilder res = new StringBuilder();
        while (c <= 32) c = getChar();
        while (c != '\n') {
            res.append(c);
            c = getChar();
        }
        return res.toString();
    }

    public boolean hasNext() {
        if (c > 32) return true;
        while (true) {
            c = getChar();
            if (c == NC) return false;
            else if (c > 32) return true;
        }
    }
}
