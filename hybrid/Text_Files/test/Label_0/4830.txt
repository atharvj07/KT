long long power(long long a, long long p, long long M) {
  long long r = 1;
  a = a % M;
  while (p > 0) {
    if (p % 2) r = (r * a) % M;
    a = (a * a) % M;
    p /= 2;
  }
  return r;
}
long long gcd(long long a, long long b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}
long long n;
long long arr[2005];
vector<pair<long long, long long> > one, zero;
int dp[2005][5][2005][2];
bool check[2005][5][2005][2];
long long solve(int ind, int c, long long co, bool last) {
  if (c == 4) return co - 1;
  if (ind >= n) return co;
  if (check[ind][c][co][last]) return dp[ind][c][co][last];
  check[ind][c][co][last] = 1;
  if (arr[ind] != last)
    return dp[ind][c][co][last] = max(solve(ind + 1, c + 1, co + 1, !last),
                                      solve(ind + 1, c, co, last));
  return dp[ind][c][co][last] =
             max(solve(ind + 1, c, co + 1, last), solve(ind + 1, c, co, last));
}
void solve() {
  cin >> n;
  for (long long i = 0; i < n; i++) {
    long long x;
    scanf("%lld", &x);
    arr[i] = x - 1;
  }
  cout << solve(0, 0, 0, 0) << endl;
}
