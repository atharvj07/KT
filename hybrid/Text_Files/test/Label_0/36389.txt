import java.util.*;
import java.io.*;
import static java.util.Arrays.*;
import static java.lang.Math.*;
import java.math.*;

public class Main {
    int[] dx = {1, 1, 1, 0, 0, -1, -1, -1};
    int[] dy = {1, 0, -1, 1, -1, 1, 0, -1};
    void run() throws IOException {
        String[] s = new String[4];
        for (int i = 0; i < 4; i++) {
            s[i] = nToken();
        }
        boolean[][] beat = new boolean[10][10];
        int h1 = (int)(s[0].charAt(0) - 'a' + 1), v1 = (int)(s[0].charAt(1) - '0');
        int h2 = (int)(s[1].charAt(0) - 'a' + 1), v2 = (int)(s[1].charAt(1) - '0');
        for (int i = 1; i <= 8; i++) {
            for (int j = 1; j <= 8; j++) {
                beat[i][j] = (i == h1 || i == h2 || j == v1 || j == v2);
            }
        }
        if (!(h1 == h2 || v1 == v2)) {
            beat[h1][v1] = beat[h2][v2] = false;
        }
        int h = (int)(s[2].charAt(0) - 'a' + 1), v = (int)(s[2].charAt(1) - '0');
        if (h == h1 && h != h2) {
            if (v > v1) {
                for (int i = v + 2; i <= 8; i++) {
                    beat[h][i] = false;
                }
            } else {
                for (int i = v - 2; i >= 1; i--) {
                    beat[h][i] = false;
                }
            }
        }
        if (h == h2 && h != h1) {
            if (v > v2) {
                for (int i = v + 2; i <= 8; i++) {
                    beat[h][i] = false;
                }
            } else {
                for (int i = v - 2; i >= 1; i--) {
                    beat[h][i] = false;
                }
            }
        }
        if (v == v1 && v != v2) {
            if (h > h1) {
                for (int i = h + 2; i <= 8; i++) {
                    beat[i][v] = false;
                }
            } else {
                for (int i = h - 2; i >= 1; i--) {
                    beat[i][v] = false;
                }
            }
        }
        if (v == v2 && v != v1) {
            if (h > h2) {
                for (int i = h + 2; i <= 8; i++) {
                    beat[i][v] = false;
                }
            } else {
                for (int i = h - 2; i >= 1; i--) {
                    beat[i][v] = false;
                }
            }
        }
        for (int i = 0; i < dx.length; i++) {
            beat[h + dx[i]][v + dy[i]] = true;
        }
        for (int i = 0; i < 10; i++) {
            beat[0][i] = beat[9][i] = beat[i][0] = beat[i][9] = true;
        }
        h = (int)(s[3].charAt(0) - 'a' + 1);
        v = (int)(s[3].charAt(1) - '0');
        if (!beat[h][v]) {
            out.println("OTHER");
        } else {
            for (int i = 0; i < dx.length; i++) {
                if (!beat[h + dx[i]][v + dy[i]]) {
                    out.println("OTHER");
                    return;
                }
            }
            out.println("CHECKMATE");
        }
    }

    static StringTokenizer tokenizer;
    static BufferedReader in;
    static PrintWriter out;
    //static Scanner in;

    public static void main(String[] args) throws IOException {
        //final String FILENAME = "inner";
        tokenizer = null;
        //in = new BufferedReader(new FileReader(FILENAME + ".in")); out = new PrintWriter (new File (FILENAME + ".out"));
        in = new BufferedReader (new InputStreamReader(System.in)); out = new PrintWriter (System.out);
        new Main().run();
        out.close();
    }

    int nInt() throws IOException {
        return Integer.parseInt(nToken());
    }

    long nLong() throws IOException {
        return Long.parseLong(nToken());
    }

    double nDouble() throws IOException {
        return Double.parseDouble(nToken());
    }

    String nToken() throws IOException {
        while (tokenizer == null || !tokenizer.hasMoreTokens()) {
            tokenizer = new StringTokenizer(in.readLine());
        }
        return tokenizer.nextToken();
    }

}
