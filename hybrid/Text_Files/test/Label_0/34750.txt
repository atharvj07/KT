import java.util.*;

public class Main {
    static int[][] a = new int[102][102];
    static char[][] b = new char[102][102];
    public static void main(String[] args) {
        Scanner reader = new Scanner(System.in);
        int n, m, i, j, p, q, s, t;
        n = reader.nextInt();
        m = reader.nextInt();
        s = 0;
        for(i = 0; i < n; i++){
            String u = reader.next();
            for(j = 0; j < m; j++){
                b[i][j] = u.charAt(j);
            }
        }
        for(p = n - 1; p >= 0; p--){
            for(q = m - 1; q >= 0; q--){
                if(b[p][q] == 'W')
                    t = 1;
                else
                    t = -1;
                if(a[p][q] != t){
                    for(i = 0; i <= p; i++){
                        for(j = 0; j <= q; j++){
                            a[i][j] += t - a[p][q];
                        }
                    }
                    s++;
                }
            }
        }
        System.out.println(s);
    }
}
