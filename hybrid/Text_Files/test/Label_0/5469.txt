#include <bits/stdc++.h>
using namespace std;
int main() {
  int a, b, c, max, min, avg, num;
  cin >> num;
  while (num >= 1) {
    cin >> a >> b >> c;
    if (a >= b && a >= c) {
      max = a;
      if (b >= c) {
        avg = b;
        min = c;
      } else {
        avg = c;
        min = b;
      }
    } else if (b > a && b >= c) {
      max = b;
      if (a >= c) {
        avg = a;
        min = c;
      } else {
        avg = c;
        min = a;
      }
    } else {
      max = c;
      if (a >= b) {
        avg = a;
        min = b;
      } else {
        avg = b;
        min = a;
      }
    }
    if (min + avg + 1 >= max)
      cout << "Yes" << endl;
    else
      cout << "No" << endl;
    num--;
  }
}
