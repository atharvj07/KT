#include <bits/stdc++.h>
using namespace std;
const int inf = 0x3f3f3f3f;
const long long infll = 0x3f3f3f3f3f3f3f3f;
const long long mod = 1e9 + 7;
const int maxn = 1e5 + 10;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(n);
  for (int &i : a) cin >> i;
  vector<int> b(m);
  for (int &i : b) cin >> i;
  map<int, int> mp1, mp2;
  for (int i = 0; i < n; ++i) {
    int cnt = 0;
    while (i < n and a[i] == 1) cnt++, i++;
    for (int j = 1; cnt >= 1; ++j, --cnt) mp1[cnt] += j;
  }
  for (int i = 0; i < m; ++i) {
    int cnt = 0;
    while (i < m and b[i] == 1) cnt++, i++;
    for (int j = 1; cnt >= 1; ++j, --cnt) mp2[cnt] += j;
  }
  long long ans = 0;
  for (auto i : mp1)
    if (k % i.first == 0) ans += 1ll * i.second * mp2[k / i.first];
  cout << ans << endl;
  return 0;
}
