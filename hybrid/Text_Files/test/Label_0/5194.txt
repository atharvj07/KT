#include <bits/stdc++.h>
#define rep(i,n) for (long i=0;i<n;i++)
using namespace std;

int main() {
	string s, t;
	long long k, c = 0;
	cin >> s >> k;
	t = s;
	long long S = s.size();
	rep(i, S) {
		if (i != 0 && s[i] == s[i - 1]) {
			s[i] = '.';
			c++;
		}
	}
	long long C = 0;
	rep(i, S) {
		if (i == 0) {
			if (t[0] == s[S - 1]) {
				t[0] = '.';
				C++;
			}
		}
		else {
			if (t[i] == t[i - 1]) {
				t[i] = '.';
				C++;
			}
		}
	}
	long long ans;
	if (t[S - 1] == s[S - 1]) ans = c + C * (k - 1);
	else ans = S * k / 2;
	cout << ans << endl;
}
