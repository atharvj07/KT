s = input()
si = len(s)
res = 1
seen = {}
Len = 0
if '0' <= s[0] <= '9':
	pass
elif s[0] == '?':
	res *= 9
else:
	res *= 9
	seen[s[0]] = 1
	Len += 1
tmp = ''
for i in range(1, si):
	if '0' <= s[i] <= '9':
		pass
	elif s[i] != '?':
		if seen.get(s[i], 0) == 0:
			res *= 10 - Len
			seen[s[i]] = 1
			Len += 1
		else:
			pass
	else:
		tmp += '0'
print(res, tmp, sep='')
