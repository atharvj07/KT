from math import ceil
for _ in range(int(input())):
	n = int(input())
	(a, b, c) = [int(i) for i in input().split()]
	w = input()
	s = []
	for i in range(n):
		elem = w[i]
		if elem == 'R' and b > 0:
			s.append('P')
			b -= 1
		elif elem == 'S' and a > 0:
			s.append('R')
			a -= 1
		elif elem == 'P' and c > 0:
			s.append('S')
			c -= 1
		else:
			s.append(0)
	if s.count(0) > n - ceil(n / 2):
		print('NO')
	else:
		print('YES')
		for elem in s:
			if elem != 0:
				print(elem, end='')
			elif elem == 0:
				if a > 0:
					a -= 1
					print('R', end='')
				elif b > 0:
					b -= 1
					print('P', end='')
				else:
					print('S', end='')
		print()
