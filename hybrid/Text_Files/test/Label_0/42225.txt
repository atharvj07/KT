n = int(input())
a = list(map(int, input().split()))
from math import ceil
s = set()
s0 = None
my = 0
for i in range(1, n):
	if a[i] == a[i - 1]:
		print('NO')
		exit()
	if abs(a[i] - a[i - 1]) != 1:
		s.add(abs(a[i] - a[i - 1]))
		s0 = abs(a[i] - a[i - 1])
	if len(s) > 1:
		print('NO')
		exit()
	elif len(s) == 1:
		my = max(ceil(max(a[i], a[i - 1]) / s0), my)
if len(s) == 0:
	x = 1
	print('YES')
	print(x, max(a))
	exit()
for i in range(1, n):
	if abs(a[i] - a[i - 1]) == 1:
		if ceil(a[i] / s0) != ceil(a[i - 1] / s0):
			print('NO')
			exit()
print('YES')
print(my, s0)
