#include <bits/stdc++.h>
#pragma comment(linker, "/stack:200000000")
#pragma GCC optimize("Ofast")
#pragma GCC target("sse,sse2,sse3,ssse3,sse4,popcnt,abm,mmx,avx,tune=native")
#pragma GCC optimize("unroll-loops")
#pragma GCC optimize("O3")
using namespace std;
const long long N = 2e5 + 5;
const long long M = 3e5 + 5;
const long long B = 2100;
const long long sum = 2e9;
const long long MOD = 1e9 + 7;
const long long MOD1 = 1e9 + 7;
const long long rx[4] = {0, 1, -1, 0};
const long long ry[4] = {1, 0, 0, -1};
const char rr[5] = {'5', 'N', 'E', 'S', 'W'};
mt19937 gen(chrono::high_resolution_clock::now().time_since_epoch().count());
long long a[N];
long long p[N];
vector<long long> g[N];
int32_t main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  srand(time(0));
  long long n;
  cin >> n;
  vector<pair<long long, long long> > all;
  for (long long i = 1; i <= n; i++) {
    long long x;
    cin >> x;
    a[i * 2] = x;
    a[i * 2 - 1] = x;
    all.push_back({x, i * 2 - 1});
  }
  sort(all.rbegin(), all.rend());
  vector<long long> way;
  long long r = 0;
  long long last = -1;
  for (auto to : all) {
    p[r] = to.second;
    r++;
    way.push_back(to.second);
    if (last == -1) {
      last = way.back();
      continue;
    }
    g[last].push_back(way.back());
    g[way.back()].push_back(last);
    last = way.back();
  }
  for (long long i = 0; i < way.size(); i++) way[i]++;
  long long add = 0;
  for (long long to : way) {
    long long x = a[to];
    x--;
    x += add;
    add++;
    long long v = p[x];
    if (0) cerr << to << ' ' << x << ' ' << v << '\n';
    if (p[x + 1] == 0) {
      p[x + 1] = to;
    }
    g[to].push_back(v);
    g[v].push_back(to);
  }
  for (long long i = 1; i <= n * 2; i++) {
    for (long long to : g[i]) {
      if (i < to) cout << i << ' ' << to << '\n';
    }
  }
  return 0;
}
