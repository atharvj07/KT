template<class T> inline void chmax(T& a, const T& b) { if(b > a) a = b; }
template<class T> inline void chmin(T& a, const T& b) { if(b < a) a = b; }

template<typename T, typename U>
ostream& operator<<(ostream& os, const pair<T, U>& p) {
	os << '(' << p.first << ", " << p.second << ')';
	return os;
}

template<class Tuple, unsigned Index>
void print_tuple(ostream& os, const Tuple& t) {}

template<class Tuple, unsigned Index, class Type, class... Types>
void print_tuple(ostream& os, const Tuple& t) {
	if(Index > 0) os << ", ";
	os << get<Index>(t);
	print_tuple<Tuple, Index + 1, Types...>(os, t);
}

template<class... Types>
ostream& operator<<(ostream& os, const tuple<Types...>& t) {
	os << '(';
	print_tuple<tuple<Types...>, 0, Types...>(os, t);
	return os << ')';
}

template<class Iterator>
ostream& dump_range(ostream&, Iterator, const Iterator&);

template<typename T>
ostream& operator<<(ostream& os, vector<T> c) {
	return dump_range(os, c.cbegin(), c.cend());
}

template<class Iterator>
ostream& dump_range(ostream& os, Iterator first, const Iterator& last) {
	os << '[';
	for(int i = 0; first != last; ++i, ++first) {
		if(i) os << ", ";
		os << *first;
	}
	return os << ']';
}

unordered_map<string, int> en;
unordered_map<int, string> de;
int encode(const string& s) {
	if(!en.count(s)) {
		en.insert(make_pair(s, en.size()));
		de[en[s]] = s;
	}
	return en[s];
}

string decode(int x) {
	return de[x];
}

struct button {
	int x1, y1, x2, y2;
	string to;
	button(int x1, int y1, int x2, int y2, string to):x1(x1), y1(y1), x2(x2), y2(y2), to(to){}
	bool hit(int x, int y) {
		return x1 <= x && x <= x2 && y1 <= y && y <= y2;
	}
};

int main() {
	cin.tie(0);
	ios::sync_with_stdio(false);

	for(int n; cin >> n && n;) {
		en.clear();

		int w, h;
		cin >> w >> h;

		vector<vector<button>> links(n);
		for(int i = 0; i < n; ++i) {
			string name;
			int number_of_links;
			cin >> name >> number_of_links;

			const int idx = encode(name);
			for(int j = 0; j < number_of_links; ++j) {
				int x1, y1, x2, y2;
				string to;
				cin >> x1 >> y1 >> x2 >> y2 >> to;
				links[idx].emplace_back(x1, y1, x2, y2, to);
			}
		}

		stack<int> undo, redo;
		int current = 0;

		int q;
		cin >> q;

		while(q--) {
			string operation;
			cin >> operation;

			switch(operation[0]) {
			case 's':
			{
				cout << decode(current) << endl;
				break;
			}
			case 'c':
			{
				int x, y;
				cin >> x >> y;

				for(int i = 0; i < static_cast<int>(links[current].size()); ++i) {
					if(links[current][i].hit(x, y)) {
						undo.push(current);
						current = encode(links[current][i].to);
						while(redo.size()) redo.pop();
						break;
					}
				}
				break;
			}
			case 'b':
			{
				if(!undo.empty()) {
					redo.push(current);
					current = undo.top();
					undo.pop();
				}
				break;
			}
			case 'f':
			{
				if(!redo.empty()) {
					undo.push(current);
					current = redo.top();
					redo.pop();
				}
				break;
			}
			default:
				dump(operation);
				assert(false);
			}
		}
	}

	return EXIT_SUCCESS;
}
