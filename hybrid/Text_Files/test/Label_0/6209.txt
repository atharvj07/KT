#include <bits/stdc++.h>
using namespace std;
vector<pair<int, int>> road[1005], route;
set<pair<int, int>> st;
int dist[1005][1005], n, m, k;
void dijkstra(int src) {
  int i;
  for (i = 1; i < n + 1; i++) dist[src][i] = 1e9;
  dist[src][src] = 0;
  st.insert({0, src});
  while (!st.empty()) {
    auto u = (*st.begin()).second;
    st.erase(st.begin());
    for (auto path : road[u]) {
      int v = path.first;
      int wt = path.second;
      if (dist[src][v] > dist[src][u] + wt) {
        if (dist[src][v] != 1e9) st.erase({dist[src][v], v});
        dist[src][v] = dist[v][src] = dist[src][u] + wt;
        st.insert({dist[src][v], v});
      }
    }
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(NULL);
  cout.tie(NULL);
  int x, y, w, i, j, mn = 1e9, temp;
  cin >> n >> m >> k;
  for (i = 0; i < m; i++)
    cin >> x >> y >> w, road[x].push_back({y, w}), road[y].push_back({x, w});
  for (i = 0; i < k; i++) cin >> x >> y, route.push_back({x, y});
  for (i = 1; i < n + 1; i++) dijkstra(i);
  for (x = 1; x < n + 1; x++) {
    for (auto el : road[x]) {
      y = el.first;
      temp = 0;
      for (j = 0; j < k; j++)
        temp += min({dist[route[j].first][route[j].second],
                     dist[route[j].first][x] + dist[y][route[j].second],
                     dist[route[j].first][y] + dist[x][route[j].second]});
      mn = min(mn, temp);
    }
  }
  cout << mn;
}
