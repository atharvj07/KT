import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Scanner;

public class prb327c {
	int n;
	short[] a;
	short[] a2;

	private void readInput() throws NumberFormatException, IOException {
		BufferedReader bi = new BufferedReader(new InputStreamReader(System.in));
		n = Integer.parseInt(bi.readLine());
		String ss = bi.readLine();
		a = new short[n];
		a2 = new short[n];
		for(int i=0; i<n; i++)
			a[i] = (short)(ss.charAt(2*i)-'0');		
	}
	void solve(){
		int maxlen = 0;
		int len = 0;
		for(int i=0; i<n; i++)
			a2[i] = a[i];
		for(int i=1; i<n-1; i++){			
			if (a[i]!=a[i-1] && a[i]!=a[i+1])
				len++;
			else{
				if (len>maxlen)
					maxlen = len;
				for(int j=0; j<len; j++)
					if (j<len/2)
						a2[i-len+j] = a2[i-len-1];
					else
						a2[i-len+j] = a2[i+1];
				len = 0;
			}
		}
		if (len>0){
			if (len>maxlen)
				maxlen = len;
			for(int j=0; j<len; j++)
				if (j<len/2)
					a2[n-1-len+j] = a2[n-1-len-1];
				else
					a2[n-1-len+j] = a2[n-1];
		}			
		System.out.println((maxlen+1)/2);
		for(int i=0; i<n; i++)
			System.out.print(a2[i]+" ");
	}
	public static void main(String[] args) throws NumberFormatException, IOException {
		prb327c p = new prb327c();
		p.readInput();
		p.solve();
	}

	

}
