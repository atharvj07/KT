#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  long long l = s.size();
  for (long long i = 1; i < l - 1; i++) {
    if (s[i] == s[i - 1]) {
      if (s[i - 1] != 'a' && s[i + 1] != 'a')
        s[i] = 'a';
      else if (s[i - 1] != 'b' && s[i + 1] != 'b')
        s[i] = 'b';
      else
        s[i] = 'c';
    }
  }
  if (s[l - 1] == s[l - 2]) {
    if (s[l - 2] != 'a')
      s[l - 1] = 'a';
    else if (s[l - 2] != 'b')
      s[l - 1] = 'b';
    else
      s[l - 1] = 'c';
  }
  cout << s;
  return 0;
}
