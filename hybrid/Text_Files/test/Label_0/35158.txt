import java.io.*;
import java.math.BigInteger;
import java.util.*;

public class Main {

    static FastReader in;
    static PrintWriter out;
    static Random rand = new Random();
    static final int INF = (int) (1e9 + 10), MOD = (int) (998244353), LOGN = 60;
    static final long IINF = (long) (3e18 + 10);
    static final int N = (int) (1e5 + 5);

    static void shuffle(int[] a) {
        for (int i = 0; i < a.length; i++) {
            int j = i + rand.nextInt(a.length - i);
            int temp = a[i];
            a[i] = a[j];
            a[j] = temp;
        }
    }

    static int lower(int[] arr, int x) {
        int l = 0, r = arr.length;
        while (l < r) {
            int m = (l + r) / 2;
            if (arr[m] < x) {
                l = m + 1;
            } else {
                r = m;
            }
        }
        return l;
    }

    static int upper(int[] arr, int x) {
        if (arr[0] > x) {
            return -1;
        }
        int l = 0, r = arr.length - 1;
        while (l < r) {
            int m = (l + r + 1) / 2;
            if (arr[m] > x) {
                r = m - 1;
            } else {
                l = m;
            }
        }
        return l;
    }

    static long get(long x, long y, long z) {
        return (x - y) * (x - y) + (y - z) * (y - z) + (z - x) * (z - x);
    }

    static void solve() {
        int nr = in.nextInt();
        int ng = in.nextInt();
        int nb = in.nextInt();

        int[] r = new int[nr];
        for (int i = 0; i < nr; i++) {
            r[i] = in.nextInt();
        }
        shuffle(r);
        Arrays.sort(r);

        int[] g = new int[ng];
        for (int i = 0; i < ng; i++) {
            g[i] = in.nextInt();
        }
        shuffle(g);
        Arrays.sort(g);

        int[] b = new int[nb];
        for (int i = 0; i < nb; i++) {
            b[i] = in.nextInt();
        }
        shuffle(b);
        Arrays.sort(b);

        long ans = IINF;
        for (int i = 0; i < nr; i++) {
            int rr = r[i];
            {
                int i1 = lower(g, rr);
                int i2 = upper(b, rr);
                if (i1 != ng && i2 != -1)
                    ans = Math.min(ans, get(rr, g[i1], b[i2]));
            }
            {
                int i1 = upper(g, rr);
                int i2 = lower(b, rr);
                if (i1 != -1 && i2 != nb)
                    ans = Math.min(ans, get(rr, g[i1], b[i2]));
            }
        }
        for (int i = 0; i < ng; i++) {
            int gg = g[i];
            {
                int i1 = lower(r, gg);
                int i2 = upper(b, gg);
                if (i1 != nr && i2 != -1)
                    ans = Math.min(ans, get(gg, r[i1], b[i2]));
            }
            {
                int i1 = upper(r, gg);
                int i2 = lower(b, gg);
                if (i1 != -1 && i2 != nb)
                    ans = Math.min(ans, get(gg, r[i1], b[i2]));
            }
        }
        for (int i = 0; i < nb; i++) {
            int bb = b[i];
            {
                int i1 = lower(r, bb);
                int i2 = upper(g, bb);
                if (i1 != nr && i2 != -1)
                    ans = Math.min(ans, get(bb, r[i1], g[i2]));
            }
            {
                int i1 = upper(r, bb);
                int i2 = lower(g, bb);
                if (i1 != -1 && i2 != ng)
                    ans = Math.min(ans, get(bb, r[i1], g[i2]));
            }
        }

        out.println(ans);
    }

    public static void main(String[] args) throws FileNotFoundException, InterruptedException {
        in = new FastReader(System.in);
//        in = new FastReader(new FileInputStream("input.txt"));
        out = new PrintWriter(System.out);
//        out = new PrintWriter(new FileOutputStream("output.txt"));

        Thread thread = new Thread(null, () -> {
            int tests = 1;
            tests = in.nextInt();

            while (tests-- > 0) {
                solve();
            }
        }, "Go", 1 << 28);
        thread.start();
        thread.join();

//        out.flush();
        out.close();
    }

    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        FastReader(InputStream is) {
            br = new BufferedReader(new InputStreamReader(is));
        }

        Integer nextInt() {
            return Integer.parseInt(next());
        }

        Long nextLong() {
            return Long.parseLong(next());
        }

        Double nextDouble() {
            return Double.parseDouble(next());
        }

        String next() {
            while (st == null || !st.hasMoreTokens()) {
                st = new StringTokenizer(nextLine());
            }
            return st.nextToken();
        }

        String nextLine() {
            String s = "";
            try {
                s = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return s;
        }
    }
}
