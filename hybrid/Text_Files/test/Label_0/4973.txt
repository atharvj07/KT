#include <bits/stdc++.h>
using namespace std;
const int maxn = 100001;
int a[maxn];
int c[maxn];
int n, m;
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
  }
  for (int i = 1; i <= n; i++) {
    scanf("%d", &c[i]);
  }
  priority_queue<pair<int, int> > q;
  for (int i = 1; i <= n; i++) {
    q.push(make_pair(-c[i], -i));
  }
  long long cnt = 0;
  for (int i = 1; i <= n; i++) {
    cnt += a[i];
  }
  for (int i = 1; i <= m; i++) {
    long long cur = 0;
    int t, d;
    scanf("%d%d", &t, &d);
    while (cnt && d) {
      if (a[t]) {
        if (a[t] >= d) {
          cur += (long long)c[t] * d;
          a[t] -= d;
          cnt -= d;
          d = 0;
        } else {
          cur += (long long)c[t] * a[t];
          d -= a[t];
          cnt -= a[t];
          a[t] = 0;
        }
      } else {
        int t1 = -q.top().second;
        if (a[t1] >= d) {
          cur += (long long)c[t1] * d;
          a[t1] -= d;
          cnt -= d;
          d = 0;
        } else {
          cur += (long long)c[t1] * a[t1];
          d -= a[t1];
          cnt -= a[t1];
          a[t1] = 0;
        }
        if (!a[t1]) q.pop();
      }
    }
    if (d) {
      printf("0\n");
    } else {
      printf("%lld\n", cur);
    }
  }
  return 0;
}
