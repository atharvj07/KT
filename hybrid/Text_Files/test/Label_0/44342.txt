import math
from collections import Counter, defaultdict
from sys import stdin
input = stdin.readline
for _ in range(int(input())):
	n = int(input())
	u = [int(i) for i in input().split()]
	s = [int(i) for i in input().split()]
	st = set(u)
	d = defaultdict(list)
	for i in range(n):
		d[u[i]].append(s[i])
	l = [0] * (n + 1)
	sm = [0] * (n + 1)
	for i in st:
		d[i].sort()
		l[i] = len(d[i])
	a = [0] * n
	for i in st:
		ps = [0] * (l[i] + 1)
		for j in range(l[i]):
			ps[j + 1] = ps[j] + d[i][j]
		for k in range(1, l[i] + 1):
			s = 0
			rem = l[i] % k
			s = ps[-1] - ps[rem]
			a[k - 1] += s
	print(*a)
