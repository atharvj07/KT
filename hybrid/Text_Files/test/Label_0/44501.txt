from sys import stdin
n = stdin.readline().strip().lower()
a = list('abcdefghijklmnopqrstuvwxyz')
b = list('zyxwvutsrqponmlkjihgfedcba')
cont = 0
m = 'a' + n
list(m)
for i in range(len(m) - 1):
	co = []
	lis = abs(a.index(m[i]) - a.index(m[i + 1]))
	lis1 = b.index(m[i]) + a.index(m[i + 1]) + 1
	lis2 = a.index(m[i]) + b.index(m[i + 1]) + 1
	co = [lis, lis1, lis2]
	cont += min(co)
print(cont)
