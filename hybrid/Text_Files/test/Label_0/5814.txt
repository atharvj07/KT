#include <bits/stdc++.h>
using namespace std;
const int maxn = 3005;
int p[maxn], dp[maxn][maxn], bak[maxn];
int main() {
  int n, k;
  while (scanf("%d%d", &n, &k) != EOF) {
    memset(dp, 63, sizeof(dp));
    for (int i = 1; i <= n; i++) scanf("%d", &p[i]);
    for (int i = n; i >= 1; i--)
      if (i == n)
        bak[i] = p[i];
      else
        bak[i] = max(bak[i + 1], p[i]);
    dp[1][2] = 0;
    for (int i = 1; i <= n; i++)
      for (int j = i + 1; j <= n; j++) {
        if (bak[j] > 0 && p[i] < 100)
          dp[j][j + 1] = min(dp[j][j + 1], dp[i][j] + 1);
        if (bak[j] > 0 && p[i] > 0)
          dp[j + 1][j + 2] = min(dp[j + 1][j + 2], dp[i][j] + 1);
        if (bak[j] < 100 && p[i] > 0)
          dp[i][j + 1] = min(dp[i][j + 1], dp[i][j] + 1);
      }
    int output = 0;
    for (int i = 1; i <= n; i++)
      for (int j = i + 1; j <= n + 1; j++)
        if (dp[i][j] <= k) output++;
    if (dp[n + 1][n + 2] <= k) output++;
    printf("%d\n", output);
  }
  return 0;
}
