import java.io.*;
import java.util.*;
import java.math.*;
import java.awt.geom.*;

import static java.lang.Math.*;

public class Solution implements Runnable {
	
	class Edge {
		int to, cost;
		
		public Edge(int to, int cost) {
			this.to = to;
			this.cost = cost;
		}
	}
	
	ArrayList<Integer> g [];
	
	public void solve () throws Exception {
		int n = nextInt();
		int m = nextInt();
		int k = nextInt();
		int a [] = new int [k];
		for (int i = 0; i < k; i++) {
			a [i] = nextInt() - 1;
		}
		
		g = new ArrayList [n];
		for (int i = 0; i < n; i++) {
			g [i] = new ArrayList<Integer>();
		}
		
		for (int i = 0; i < m; i++) {
			int v1 = nextInt() - 1;
			int v2 = nextInt() - 1;
			g[v1].add(v2);
			g[v2].add(v1);
		}
		
		int s = nextInt() - 1;
		int f = nextInt() - 1;
		
		boolean flag = false;
		
		int dist [] = new int [n];
		
		Arrays.fill (dist, Integer.MAX_VALUE / 2);
		LinkedList<Integer> q = new LinkedList<Integer>();
		
		for (int i = 0; i < k; i++) {
			if (a [i] == f) flag = true;
			q.add (a[i]);
			dist [a[i]] = 0;
		}
		
		if (!flag) {
			q.add (f);
			dist [f] = 0;
		}
		
		while (!q.isEmpty()) {
			int v = q.pollFirst();
			for (int to : g [v]) {
				if (dist[to] > dist[v] + 1) {
					dist[to] = dist[v] + 1;
					q.addLast(to);
				}
			}
		}
		
		ArrayList<Edge> gr [] = new ArrayList [n];
		
		for (int i = 0; i < n; i++) {
			gr [i] = new ArrayList<Solution.Edge>();
		}
		
		for (int v = 0; v < n; v++) {
			if (dist [v] == Integer.MAX_VALUE / 2) continue;
			for (int to : g [v]) {
				gr[v].add(new Edge (to, dist[v] + dist[to] + 1));
			}
		}
		
		int l = 0, r = n + 1;
		
		while (r - l > 1) {
			int mid = (r + l) >> 1;
			Arrays.fill(dist, Integer.MAX_VALUE / 2);
			q.add(s);
			dist [s] = 0;
			while (!q.isEmpty()) {
				int v = q.pollFirst();
				for (Edge e : gr [v]) {
					if (e.cost <= mid) {
						if (dist[e.to] > dist[v]) {
							dist[e.to] = dist[v];
							q.addLast(e.to);
						}
					}
				}
			}
			
			if (dist[f] == Integer.MAX_VALUE / 2) {
				l = mid;
			} else {
				r = mid;
			}
			
		}
		
		if (r == n + 1) r = -1;
		out.println(r);
		
	}
	
	static final String fname = "";
	static long stime = 0;
	
	BufferedReader in;
	PrintWriter out;
	StringTokenizer st;
	
	private String nextToken () throws Exception {
		while (st == null || !st.hasMoreTokens()) {
			st = new StringTokenizer(in.readLine());
		}
		return st.nextToken();
	}
	
	private int nextInt () throws Exception {
		return Integer.parseInt(nextToken());
	}
	
	private long nextLong () throws Exception {
		return Long.parseLong(nextToken());
	}
	
	private double nextDouble () throws Exception {
		return Double.parseDouble(nextToken());
	}
	
	@Override
	public void run() {
		try {
			in = new BufferedReader(new InputStreamReader(System.in));
			out = new PrintWriter(System.out);
			solve ();
		} catch (Exception e) {
			throw new RuntimeException(e);
		} finally {
			out.close();
		}
	}
	
	public static void main(String[] args) {
		new Thread(null, new Solution(), "", 1<<26).start();
	}

}
