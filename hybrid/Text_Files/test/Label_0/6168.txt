#include <bits/stdc++.h>
using namespace std;
int arr[2000];
int perc[2000];
pair<int, int> drr[2000];
void func(int moment, unordered_set<int>& good, int n) {
  int completed = 0;
  for (int i = 0; i < n; i++) {
    if (moment < drr[i].first) continue;
    int q = moment - drr[i].first + 1;
    if (q > arr[i]) completed++;
  }
  for (int i = 0; i < n; i++) {
    if (moment < drr[i].first) continue;
    int q = moment - drr[i].first + 1;
    if (q > arr[i]) continue;
    if (q == perc[completed]) good.insert(i);
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, k;
  cin >> n >> k;
  int sum = 0;
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
    sum += arr[i];
  }
  for (int i = 0; i < n + 1; i++) {
    perc[i] = round(((double)100) * ((double)i / (double)n));
  }
  vector<pair<int, int> > minai;
  for (int i = 0; i < min(n, k); i++) {
    drr[i].first = 1;
    drr[i].second = i;
    minai.push_back(make_pair(arr[i], i));
  }
  sort(minai.begin(), minai.end(), greater<pair<int, int> >());
  int p = min(n, k);
  pair<int, int> temp;
  vector<pair<int, int> >::iterator it;
  while (p != n) {
    temp = minai.back();
    minai.pop_back();
    drr[p].first = temp.first + 1;
    drr[p].second = temp.second;
    temp = make_pair(drr[p].first + arr[p] - 1, drr[p].second);
    it = lower_bound(minai.begin(), minai.end(), temp,
                     greater<pair<int, int> >());
    minai.insert(it, temp);
    p++;
  }
  unordered_set<int> good;
  for (int i = 0; i < sum + 2; i++) {
    func(i, good, n);
  }
  cout << good.size() << endl;
  return 0;
}
