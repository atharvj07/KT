#include <bits/stdc++.h>
using namespace std;
long long n, cnt, ans[555555], g0[555555][20], g[20][555555];
bool K[20][20];
map<vector<long long>, long long> vis;
void FWT(long long *f, long long _) {
  if (!_) return;
  long long mid = 1 << _ - 1;
  for (long long i = 0; i < mid; i++) {
    f[i + mid] += f[i];
  }
  FWT(f, _ - 1);
  FWT(f + mid, _ - 1);
}
void IFWT(long long *f, long long _) {
  if (!_) return;
  long long mid = 1 << _ - 1;
  IFWT(f, _ - 1);
  IFWT(f + mid, _ - 1);
  for (long long i = 0; i < mid; i++) {
    f[i] -= f[i + mid];
  }
}
signed main() {
  cin >> n;
  for (long long i = 0; i < n; i++) {
    for (long long j = 0; j < n; j++) {
      scanf("%1d", &K[i][j]);
    }
  }
  for (long long i = 0; i < n; i++) g0[1 << i][i] = 1;
  for (long long S = 0; S < 1 << n; S++) {
    for (long long j = 0; j < n; j++) {
      if (!(S & (1 << j))) {
        for (long long i = 0; i < n; i++) {
          if (S & (1 << i) && K[i][j]) {
            g0[S | (1 << j)][j] += g0[S][i];
          }
        }
      }
    }
  }
  for (long long S = 0; S < 1 << n; S++) {
    long long _S_ = __builtin_popcount(S);
    for (long long i = 0; i < n; i++) {
      g[_S_][S] += g0[S][i];
    }
  }
  for (long long l = 0; l <= n; FWT(g[l++], n))
    ;
  for (long long S = 0; S < 1 << n - 1; S++) {
    long long cnt = 0;
    vector<long long> G;
    for (long long i = 0; i < n; i++) {
      cnt++;
      if (!(S & (1 << i))) {
        G.push_back(cnt);
        cnt = 0;
      }
    }
    if (cnt) G.push_back(cnt);
    sort(G.begin(), G.end());
    if (!vis.count(G)) {
      long long res = 0;
      for (long long T = 1; T < 1 << n; T++) {
        long long _ = 1;
        for (long long a : G) _ *= g[a][T];
        res += (n - __builtin_popcount(T) & 1) ? -_ : _;
      }
      vis[G] = res;
    }
    ans[S] = vis[G];
  }
  IFWT(ans, n - 1);
  for (long long S = 0; S < (1 << n - 1); S++) {
    cout << ans[S] << " ";
  }
  return 0;
}
