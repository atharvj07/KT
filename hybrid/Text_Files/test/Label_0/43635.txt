def getFib():
	ans = [1, 2]
	a = 1
	b = 2
	while ans[-1] < 1000:
		ans.append(a + b)
		a = b
		b = ans[-1]
	return ans

def main():
	fib = getFib()
	n = I()
	j = 0
	for i in range(n):
		if i + 1 == fib[j]:
			print('O', end='')
			j += 1
		else:
			print('o', end='')
tc = 1
for _ in range(tc):
	main()
