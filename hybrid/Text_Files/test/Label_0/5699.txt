#include <bits/stdc++.h>
using namespace std;
const int inf = 1e9;
const long long inf64 = 1e18;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  long long n, k;
  int m;
  cin >> n >> m >> k;
  set<long long> q;
  for (int i = 0; i < m; i++) {
    long long x;
    cin >> x;
    q.insert(x - 1);
  }
  int res = 0;
  long long sh = 0;
  while (!q.empty()) {
    long long b = (*q.begin() - sh) / k;
    long long tmp = 0;
    while (!q.empty() && (*q.begin() - sh) / k == b) {
      q.erase(q.begin());
      tmp++;
    }
    res++;
    sh += tmp;
  }
  cout << res << "\n";
  return 0;
}
