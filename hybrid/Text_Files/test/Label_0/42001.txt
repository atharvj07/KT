from collections import defaultdict
d = {}
(left, right) = map(int, input().split())
N = max(left, right) + 1
stack = [1]
d[1] = 1
while stack:
	p = stack.pop()
	for i in (2 * p, 3 * p):
		if i < N:
			if not i in d:
				d[i] = 1
				stack.append(i)
ans = 0
for a in d:
	if left <= a <= right:
		ans += 1
print(ans)
