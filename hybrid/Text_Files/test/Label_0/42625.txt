def move_up(l, d, a):
	if a > 0:
		in_icon = l[a]
		swap_icon = l[a - 1]
		d[in_icon] = a - 1
		d[swap_icon] = a
		(l[a], l[a - 1]) = (l[a - 1], l[a])
line1 = input()
line2 = input()
line3 = input()
(n, m, k) = (int(t) for t in line1.split())
position_list = [int(t) for t in line2.split()]
position_dict = {}
for i in range(len(position_list)):
	position_dict[position_list[i]] = i
desired_list = [int(t) for t in line3.split()]
gestures = 0
for desired_id in desired_list:
	icon_index = position_dict[desired_id]
	if icon_index < k:
		gestures += 1
	else:
		gestures += icon_index // k + 1
	move_up(position_list, position_dict, icon_index)
print(gestures)
