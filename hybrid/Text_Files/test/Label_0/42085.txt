from collections import defaultdict
n = int(input())
a = list(map(int, input().split()))
d = defaultdict(list)
for i in range(2 * n):
	d[a[i]].append(i + 1)
a = 0
b = 0
ans = 0
for i in range(n):
	t1 = abs(d[i + 1][0] - a) + abs(d[i + 1][1] - b)
	t2 = abs(d[i + 1][0] - b) + abs(d[i + 1][1] - a)
	if t1 > t2:
		ans += t2
		a = d[i + 1][1]
		b = d[i + 1][0]
	else:
		ans += t1
		a = d[i + 1][0]
		b = d[i + 1][1]
print(ans - 2)
