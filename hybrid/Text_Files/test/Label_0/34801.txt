import java.util.*;
import java.io.*;

public class c {

    public static void main(String[] args) throws Exception {

        BufferedReader stdin = new BufferedReader(new InputStreamReader(System.in));
        long[] aPos = new long[2];
        long[] bPos = new long[2];
        long[] bin = new long[2];
        StringTokenizer tok = new StringTokenizer(stdin.readLine());
        aPos[0] = Long.parseLong(tok.nextToken());
        aPos[1] = Long.parseLong(tok.nextToken());
        bPos[0] = Long.parseLong(tok.nextToken());
        bPos[1] = Long.parseLong(tok.nextToken());
        bin[0] = Long.parseLong(tok.nextToken());
        bin[1] = Long.parseLong(tok.nextToken());

        int n = Integer.parseInt(stdin.readLine().trim());
        long[][] pts = new long[n][2];

        double twoBin = 0;
        double[] binD = new double[n];

        for (int i=0; i<n; i++) {
            tok = new StringTokenizer(stdin.readLine());
            pts[i][0] = Long.parseLong(tok.nextToken());
            pts[i][1] = Long.parseLong(tok.nextToken());
            binD[i] = dist(bin, pts[i]);
            twoBin += (2*binD[i]);
        }

        double myDiff = 2000000000;
        int myBest = -1;
        double hisDiff = 2000000000;
        int hisBest = -1;
        for (int i=0; i<n; i++) {

            double myD = dist(aPos, pts[i]) - binD[i];
            double hisD = dist(bPos, pts[i]) - binD[i];

            if (myD < myDiff) {
                myBest = i;
                myDiff = myD;
            }

            if (hisD < hisDiff) {
                hisBest = i;
                hisDiff = hisD;
            }
        }

        // One person should be used in these cases.
        if (myDiff > 0 || hisDiff > 0)
            twoBin += Math.min(myDiff, hisDiff);

        // Both should help.
        else {

            // Easy case, no conflict on best bottle for both to get.
            if (myBest != hisBest)
                twoBin += (myDiff+hisDiff);

            else {

                // Try both options.
                double optA = twoBin + myDiff;
                double diffB = 2000000000;
                for (int i=0; i<n; i++) {
                    if (i == myBest) continue;
                    double hisD = dist(bPos, pts[i]) - binD[i];
                    diffB = Math.min(diffB, hisD);
                }
                if (diffB < 0) optA += diffB;


                double optB = twoBin + hisDiff;
                double diffA = 2000000000;
                for (int i=0; i<n; i++) {
                    if (i == hisBest) continue;
                    double myD = dist(aPos, pts[i]) - binD[i];
                    diffA = Math.min(diffA, myD);
                }
                if (diffA < 0) optB += diffA;

                double res = Math.min(optA, optB);
                twoBin = res;
            }
        }

        System.out.println(twoBin);
    }

    public static double dist(long[] a, long[] b) {
        return Math.sqrt( (a[0]-b[0])*(a[0]-b[0]) +  (a[1]-b[1])*(a[1]-b[1]) );
    }
}
