#include <bits/stdc++.h>
using namespace std;
int main() {
  int i, j, n, m;
  cin >> n;
  vector<int> v(n);
  for (i = 0; i < n; i++) cin >> v[i];
  int count = 0;
  for (i = 0; i < n; i++) {
    if (v[i] == 1) count++;
  }
  for (i = 1; i < n - 1; i++) {
    if (v[i] == 0 && v[i - 1] == 1 && v[i + 1] == 1) count++;
  }
  cout << count << '\n';
  return 0;
}
