from collections import Counter, defaultdict
from functools import lru_cache, reduce
import operator
t = int(input())
x = []
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	h1 = Counter(a)
	h2 = Counter()
	res = 0
	for i in range(n):
		h1[a[i]] -= 1
		s = 0
		for j in range(i + 1, n):
			s -= h1[a[j]] * h2[a[j]]
			h1[a[j]] -= 1
			if j > i + 1:
				if a[j] != a[j - 1]:
					s -= h1[a[j - 1]] * h2[a[j - 1]]
				h2[a[j - 1]] += 1
				if a[j] != a[j - 1]:
					s += h1[a[j - 1]] * h2[a[j - 1]]
			s += h1[a[j]] * h2[a[j]]
			if a[i] == a[j]:
				res += s
		h2[a[n - 1]] += 1
		(h1, h2) = (h2, h1)
	x.append(res)
print(*x, sep='\n')
