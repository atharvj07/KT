import java.util.*;
import java.io.*;
public class Planets
{
	public static ArrayList<Stargate>[] adj;
	public static long[] dist;
	public static ArrayList<Integer>[] closed;
	
	public static final long INFINITY = 10000000000000L;
	
	public static void main(String[] args)
	{
		Scanner in = new Scanner(System.in);
		
		int n = in.nextInt();
		int m = in.nextInt();
		
		adj = new ArrayList[n];
		for(int i = 0; i < adj.length; i++)
		{
			adj[i] = new ArrayList<Stargate>();
		}
		
		for(int x = 0; x < m; x++)
		{
			int a = in.nextInt() - 1;
			int b = in.nextInt() - 1;
			int c = in.nextInt();
			
			adj[a].add(new Stargate(a, b, c));
			adj[b].add(new Stargate(b, a, c));
		}
		
		closed = new ArrayList[n];
		for(int y = 0; y < n; y++)
		{
			int k = in.nextInt();
			
			closed[y] = new ArrayList<Integer>();
			
			for(int z = 0; z < k; z++)
			{
				closed[y].add(in.nextInt());
			}
		}
		
		dijkstra(0);
		
		if(dist[n - 1] == INFINITY)
		{
			System.out.println("-1");
		}
		else
		{
			System.out.println(dist[n - 1]);
		}
	}
	
	public static void dijkstra(int start)
	{
		dist = new long[adj.length];
		Arrays.fill(dist, INFINITY);
		dist[start] = 0;
		
		PriorityQueue<Planet> unused = new PriorityQueue<Planet>();
		for(int i = 0; i < adj.length; i++)
		{
			Planet current = new Planet(i, INFINITY);
			
			if(i == start)
			{
				current.dist = 0;
			}
			
			unused.add(current);
		}
		
		while(unused.size() > 0)
		{
			Planet node = unused.poll();
			
			if(dist[node.index] == INFINITY)
			{
				break;
			}
			
			int wait = 0;
			for(int j = 0; j < closed[node.index].size(); j++)
			{
				if(closed[node.index].get(j) > dist[node.index] + wait)
				{
					break;
				}
				else if(closed[node.index].get(j) == dist[node.index] + wait)
				{
					wait++;
				}
			}
			
			for(Stargate e : adj[node.index])
			{
				long newDist = dist[node.index] + wait + e.cost;
				if(newDist < dist[e.end])
				{
					dist[e.end] = newDist;
					
					unused.add(new Planet(e.end, newDist));
				}
			}
		}
	}
}
 
class Stargate
{
	int start;
	int end;
	int cost;
	
	public Stargate(int s, int e, int c)
	{
		start = s;
		end = e;
		cost = c;
	}
}
 
class Planet implements Comparable<Planet>
{
	int index;
	long dist;
	
	public Planet(int i, long d)
	{
		index = i;
		dist = d;
	}
	
	public int compareTo(Planet o)
	{
		if(dist != o.dist)
		{
			if(dist < o.dist)
			{
				return -1;
			}
			else
			{
				return 1;
			}
		}
		else
		{
			return index - o.index;
		}
	}
}
