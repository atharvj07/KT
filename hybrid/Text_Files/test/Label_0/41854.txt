from sys import stdin
input = stdin.readline
q = int(input())
for rwer in range(q):
	n = int(input())
	dasie = True
	pary = [list(map(int, input().split())) for i in range(n)]
	for i in range(1, n):
		if pary[i][0] - pary[i - 1][0] < pary[i][1] - pary[i - 1][1] or pary[i][0] < pary[i - 1][0] or pary[i][1] < pary[i - 1][1]:
			dasie = False
			break
	for i in range(n):
		if pary[i][0] < 0 or pary[i][1] < 0:
			dasie = False
		if pary[i][0] < pary[i][1]:
			dasie = False
	if dasie:
		print('YES')
	else:
		print('NO')
