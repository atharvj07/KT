// practice with rainboy
import java.io.*;
import java.util.*;

public class CF1236E extends PrintWriter {
	CF1236E() { super(System.out, true); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF1236E o = new CF1236E(); o.main(); o.flush();
	}

	void main() {
		int n = sc.nextInt();
		if (n == 1) {
			println(0);
			return;
		}
		int m = sc.nextInt();
		int[] aa = new int[m + 1];
		for (int j = 1; j <= m; j++)
			aa[j] = sc.nextInt();
		int[] dd = new int[m + 2];
		int[] jj = new int[n + m + 2];
		Arrays.fill(jj, m + 1);
		for (int j = m; j >= 1; j--) {
			int x = aa[j] - j + m;
			dd[j] = jj[x - 1] == m + 1 ? 1 : dd[jj[x - 1]] + 1;
			jj[x] = j;
		}
		int[] pp = new int[n + 1];
		for (int i = 1; i <= n; i++) {
			int x = i - 0 + m;
			pp[i] = Math.min(i + (m + 1 - dd[jj[x]]), n);
		}
		Arrays.fill(jj, m + 1);
		for (int j = m; j >= 1; j--) {
			int x = aa[j] + j;
			dd[j] = jj[x + 1] == m + 1 ? 1 : dd[jj[x + 1]] + 1;
			jj[x] = j;
		}
		int[] qq = new int[n + 1];
		for (int i = 1; i <= n; i++) {
			int x = i + 0;
			qq[i] = Math.max(i - (m + 1 - dd[jj[x]]), 1);
		}
		long ans = 0;
		for (int i = 1; i <= n; i++)
			ans += pp[i] - qq[i] + 1;
		println(ans);
	}
}
