#include <bits/stdc++.h>
using namespace std;
const int maxn = 222222;
int main() {
  int a, b, n;
  cin >> n >> a >> b;
  int cnt = 0;
  int minn = min(a, b);
  int sum = a + b;
  bool flag = false;
  while (n >= minn) {
    if (n % a == 0 || n % b == 0 || n == a + b) {
      flag = true;
      cout << "YES" << endl;
      if (n == a + b)
        cout << "1 1" << endl;
      else if (n % a == 0)
        cout << n / a + cnt << " " << cnt << endl;
      else
        cout << cnt << " " << cnt + n / b << endl;
      break;
    }
    n -= sum;
    cnt++;
  }
  if (!flag) cout << "NO" << endl;
  return 0;
}
