#include <bits/stdc++.h>
using namespace std;
int N, ans, idl, idr;
int cub[300005];
deque<int> q;
int main() {
  ans = -1;
  cin.tie(0);
  ios_base::sync_with_stdio(0);
  cin >> N;
  for (int i = 1; i <= N; i++) {
    int x;
    cin >> x;
    cub[x]++;
  }
  int l = 1;
  while (l <= 200000) {
    if (cub[l] < 2)
      l++;
    else {
      int r = l;
      int sum = cub[l];
      while (cub[r + 1] >= 2) {
        r++;
        sum += cub[r];
      }
      if (cub[l - 1] == 1) sum++;
      if (cub[r + 1] == 1) sum++;
      if (sum > ans) {
        ans = sum;
        idl = l;
        idr = r;
      }
      l = r + 1;
    }
  }
  if (ans == -1) {
    for (int i = 1; i <= 200000; i++)
      if (cub[i] && cub[i + 1]) {
        cout << 2 << "\n";
        cout << i << " " << i + 1 << "\n";
        exit(0);
      }
    cout << "1\n";
    for (int i = 1; i <= 200000; i++)
      if (cub[i]) {
        cout << i << "\n";
        exit(0);
      }
  }
  cout << ans << "\n";
  while (cub[idr + 1]--) q.push_back(idr + 1);
  for (int i = idr; i >= idl; i--) {
    q.push_front(i);
    cub[i]--;
    while (cub[i]--) q.push_back(i);
  }
  while (cub[idl - 1]--) q.push_back(idl - 1);
  for (auto v : q) cout << v << " ";
  return 0;
}
