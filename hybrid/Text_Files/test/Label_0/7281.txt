#include<bits/stdc++.h>
using namespace std;
long double h[160];
double dp[160][305];

double f(double a, double b, double height) {
    double da = abs(a - height);
    double db = abs(b - height);
    if((a - height) * (b-height) >= 0) {
        return (da + db) / 2.0;
    }
    double xa = da / (da + db);
    double xb = db / (da + db);
    return (xa * da + xb * db) / 2.0;
}

int main() {
    int n;
    cin >> n;
    for(int i = 1;i <= n; i++) {
        cin >> h[i];
    }
    for(int i = 0; i <= 300; i++) {
        dp[0][i] = 1e9;
    }
    dp[0][1] = 0;
    for(int i = 1; i <= n; i++) {
        for(int after = 1; after <= 300; after++) {
            dp[i][after] = 1e9;
        }
        for(int before = 1; before <= 300; before++) {
            for(int after = 1; after <= 300; after++) {
                //cout << before << " " << after << " " << h[i] << " " << f(before, after, h[i]) << endl;
                dp[i][after] = min(dp[i][after], dp[i-1][before] + f(before, after, h[i]));
            }
        }
    }
    double ans = 1e9;
    for(int i = 1; i <= 300; i++) {
        ans = min(ans, dp[n][i]);
    }
    cout << ans << endl;
    return 0;
}
