import math
(n, k) = map(int, input().split())
if k < n // 2:
	x = math.ceil(n / (2 * k + 1)) * (2 * k + 1) - n
	a = min(n, k)
	b = x - a
	total = (n - 2 * (2 * k + 1) + a + b) // (2 * k + 1)
	ans = []
	ans.append(k - a + 1)
	for i in range(total):
		ans.append(ans[-1] + 2 * k + 1)
	ans.append(ans[-1] + 2 * k + 1)
	print(len(ans))
	if b < 0:
		for i in range(len(ans)):
			ans[i] += -b
	print(*ans)
else:
	print(1)
	print(n // 2 + 1)
