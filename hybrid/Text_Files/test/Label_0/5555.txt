#include <bits/stdc++.h>
using namespace std;
const int INF = 0x3f3f3f3f;
const int maxn = 500 + 10;
int t[maxn][maxn];
int n, m, k, q;
bool check(int mid) {
  int cnt[maxn][maxn];
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (t[i][j] <= mid)
        cnt[i][j] = 1;
      else
        cnt[i][j] = 0;
    }
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      cnt[i][j] += cnt[i][j - 1] + cnt[i - 1][j] - cnt[i - 1][j - 1];
    }
  }
  for (int i = k; i <= n; i++) {
    for (int j = k; j <= m; j++) {
      if (cnt[i][j] - cnt[i - k][j] - cnt[i][j - k] + cnt[i - k][j - k] ==
          k * k)
        return true;
    }
  }
  return false;
}
int main() {
  cin >> n >> m >> k >> q;
  memset(t, 0x3f, sizeof(t));
  for (int i = 0; i < q; i++) {
    int x, y, v;
    scanf("%d %d %d", &x, &y, &v);
    t[x][y] = v;
  }
  int l = 0, r = 1e9 + 1, ans = -1;
  while (r >= l) {
    int mid = (l + r) / 2;
    if (check(mid)) {
      ans = mid;
      r = mid - 1;
    } else
      l = mid + 1;
  }
  cout << ans << endl;
  return 0;
}
