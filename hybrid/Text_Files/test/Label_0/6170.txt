#include <bits/stdc++.h>
using namespace std;
template <typename P, unsigned long long int mod>
class Mod {
 public:
  Mod() : v_() {}

 public:
  template <typename V>
  P& operator=(const V&& rv) {
    static_assert(std::is_integral<V>::value, "should be integral type");
    v_ = rv;
    return v_;
  }
  template <typename V>
  P& operator+=(const V&& rv) {
    static_assert(std::is_integral<V>::value, "should be integral type");
    v_ += rv;
    return v_;
  }

 private:
  P v_;
};
const int sz = 1e5 + 1;
int n;
struct mice {
  int axis[2];
  double v[2];
};
pair<double, double> False(0, 0);
pair<double, double> Mt(0, numeric_limits<double>::max());
const int shift = 100000;
pair<double, double> findinterval(mice& r, double x1, double x2, int idx) {
  if (x1 > x2) {
    swap(x1, x2);
  }
  if (r.v[idx] < 0) {
    swap(x1, x2);
  }
  double s, t;
  if (r.v[idx] == 0) {
    return (x1 < r.axis[idx] && x2 > r.axis[idx]) ? Mt : False;
  }
  s = (x1 - r.axis[idx]) * shift / r.v[idx];
  t = (x2 - r.axis[idx]) * shift / r.v[idx];
  s = s < 0 ? 0 : s;
  t = t < 0 ? 0 : t;
  if (t == 0 || s == t) {
    return False;
  }
  constexpr double fix = 1e-6;
  return make_pair(s + fix, t - fix);
}
int main() {
  ios::sync_with_stdio(false);
  cin >> n;
  pair<double, double> x, y;
  cin >> x.first >> x.second >> y.first >> y.second;
  vector<mice> mouse(n);
  for (int i = 0; i < n; ++i) {
    cin >> mouse[i].axis[0] >> mouse[i].axis[1] >> mouse[i].v[0] >>
        mouse[i].v[1];
  }
  vector<pair<double, double> > at;
  for (int i = 0; i < n; ++i) {
    pair<double, double> local = findinterval(mouse[i], x.first, y.first, 0);
    at.push_back(local);
    local = findinterval(mouse[i], x.second, y.second, 1);
    at.push_back(local);
  }
  pair<double, double> ans = at[0];
  for (pair<double, double> x : at) {
    if (x == False) {
      cout << -1 << endl;
      return 0;
    }
    if (ans > x) {
      swap(ans, x);
    }
    ans = make_pair(x.first, min(x.second, ans.second));
    if (ans.first > ans.second) {
      cout << -1 << endl;
      return 0;
    }
  }
  cout << fixed << setprecision(10) << ans.first / shift << endl;
}
