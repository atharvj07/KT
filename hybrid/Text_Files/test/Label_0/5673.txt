#define REP(i,n) for(int i=0;i<(int)n;++i)

const int INF = 1<<29;
const double PI = acos(-1);
const double EPS = 1e-8;

typedef long long ll;

int N;
int c[8][8];

struct P {
  int v[8]; 
  int dist;
  P(int a[], int dist) : dist(dist) {
    REP(i,N)v[i]=a[i];
  }
  P() {}
};

struct P2 {
  int v[8];
  P2(int a[]) {
    REP(i, N) v[i] = a[i];
  }
  P2() {}
};

bool operator<(const P &a, const P &b) {
  return a.dist > b.dist;
}

bool operator<(const P2 &a, const P2 &b) {
  REP(i, N) if (a.v[i]!=b.v[i]) return a.v[i]<b.v[i];
  return 0;
}
				 

int main() {
  cin >> N;
  REP(i, N) {
    REP(j, N) {
      cin >> c[i][j];
    }
  }
  priority_queue<P> Q;
  int ini[8];
  REP(i,N) ini[i] = i;
  Q.push(P(ini, 0));
  map<P2, int> dist;
  dist[P2(ini)] = 0;
  int ans = 0;
  while(!Q.empty()) {
    P p = Q.top(); Q.pop();
    if (p.dist > dist[P2(p.v)]) continue; 
    ans = max(ans, p.dist);
//     REP(i, N) cout << p.v[i] << " ";cout <<endl;
//     cout << p.dist << " " << dist[P2(p.v)] << endl;
    REP(i, N) {
      REP(j, i) {
	int tmp[N];
	REP(k, N) tmp[k] = p.v[k];
	swap(tmp[i], tmp[j]);
	int d = p.dist + c[p.v[i]][p.v[j]];
	if (dist.count(P2(tmp)) == 0 || dist[P2(tmp)] > d) {
	  dist[P2(tmp)] = d;
	  Q.push(P(tmp, d));
	}
      }
    }
  }
  cout << ans << endl;
}
