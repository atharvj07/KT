#include<bits/stdc++.h>
#include<algorithm>
using namespace std;
typedef long long ll;
ll rc[3001][3001];
int r,c,k;
ll dp[3001][3001][4]; 
int main(){
	 scanf("%d%d%d",&r,&c,&k);
	 for(int i=1;i<=k;i++){
	 	int x,y,w;
	 	scanf("%d%d%d",&x,&y,&w);
	 	rc[x][y]=w;
	 }	 
	 for(int i=1;i<=r;i++){
	 	for(int j=1;j<=c;j++){
	 		dp[i][j][0]=max(max(dp[i-1][j][0],dp[i-1][j][1]),dp[i-1][j][2])+rc[i][j];
			 for(int k=0;k<3;k++){
			 	dp[i][j][k]=max(dp[i][j-1][k],dp[i][j][k]);
	 			if(dp[i][j-1][k]==0){
	 				dp[i][j][k]=max(dp[i][j][k],rc[i][j]);
	 				break;
				 }else{
				 	dp[i][j][k+1]=max(dp[i][j-1][k]+rc[i][j],dp[i][j][k+1]);
				 }
			 }
		 }
	 }
	 ll res=0;
	 for(int i=0;i<3;i++)
	 res=max(res,dp[r][c][i]);
	 printf("%lld\n",res);
}
