#include <iostream>
#include <array>
#include <algorithm>
#include <limits>

using namespace std;

struct Item {
    int value;
    int weight;
};

int main()
{
    int n, w;
    cin >> n >> w;

    array<int, 10001> value_to_weight;
    value_to_weight.fill(numeric_limits<int>::max());
    value_to_weight[0] = 0;
    
    for (int i = 0; i < n; ++i) {
        int value, weight;
        cin >> value >> weight;
        for (int total_value = 10000; total_value >= value; --total_value) {
            if (value_to_weight[total_value - value] == numeric_limits<int>::max())
                continue;
            value_to_weight[total_value] =
                min(value_to_weight[total_value],
                    weight + value_to_weight[total_value - value]);
        }
    }
    int max_value = 0;

    for (int total_value = 0; total_value <= 10000; ++total_value) {
        if (value_to_weight[total_value] <= w &&
            total_value > max_value) max_value = total_value;
    }
    cout << max_value << endl;
}
