import java.io.PrintWriter;
import java.util.Arrays;
import java.util.Scanner;

public class Problem2 implements Runnable {
    public void run() {
        Scanner scanner = new Scanner(System.in);
        PrintWriter writer = new PrintWriter(System.out);
        final long N_MAX = 1000000007L;
        scanner.nextLine();
        String lineOne = scanner.nextLine();
        String lineTwo = scanner.nextLine();

        int firstMore = 0;
        int secondMore = 0;
        int count = 0;
        for (int i = 0; i < lineOne.length(); i++) {
            if (lineOne.charAt(i) == '?') {
                count++;
            }
            if (lineTwo.charAt(i) == '?') {
                count++;
            }
        }
        for (int i = 0; i < lineOne.length(); i++) {
            if ((lineOne.charAt(i) != '?') && (lineTwo.charAt(i) != '?')) {
                if (lineOne.charAt(i) > lineTwo.charAt(i)) {
                    firstMore++;
                }
                if (lineOne.charAt(i) < lineTwo.charAt(i)) {
                    secondMore++;
                }
            }
        }
        long answer = 1L;
        for (int i = 0; i < count; i++) {
            answer *= 10L;
            answer %= N_MAX;
        }

        if ((firstMore > 0) && (secondMore > 0)) {
            writer.println(answer);
            writer.close();
            return;
        }

        if (firstMore > 0 && secondMore == 0) {
            long subAnswer = 1;
            for (int i = 0; i < lineOne.length(); i++) {
                if ((lineOne.charAt(i) == '?') && (lineTwo.charAt(i)) == '?') {
                    subAnswer *= 55L;
                    subAnswer %= N_MAX;
                }
                if ((lineOne.charAt(i)) == '?' && (lineTwo.charAt(i) != '?')) {
                    subAnswer *= (9 - (lineTwo.charAt(i) - '0') + 1);
                    subAnswer %= N_MAX;
                }
                if ((lineOne.charAt(i)) != '?' && (lineTwo.charAt(i) == '?')) {
                    subAnswer *= ((lineOne.charAt(i) - '0') + 1);
                    subAnswer %= N_MAX;
                }
            }
            answer -= subAnswer;
            answer %= N_MAX;
            if (answer < 0) {
                answer = N_MAX + answer;
            }
            writer.println(answer);
            writer.close();
            return;
        }
        if (firstMore == 0 && secondMore > 0) {
            String buf = lineOne;
            lineOne = lineTwo;
            lineTwo = buf;
            long subAnswer = 1;
            for (int i = 0; i < lineOne.length(); i++) {
                if ((lineOne.charAt(i) == '?') && (lineTwo.charAt(i)) == '?') {
                    subAnswer *= 55L;
                    subAnswer %= N_MAX;
                }
                if ((lineOne.charAt(i)) == '?' && (lineTwo.charAt(i) != '?')) {
                    subAnswer *= (9 - (lineTwo.charAt(i) - '0') + 1);
                    subAnswer %= N_MAX;
                }
                if ((lineOne.charAt(i)) != '?' && (lineTwo.charAt(i) == '?')) {
                    subAnswer *= ((lineOne.charAt(i) - '0') + 1);
                    subAnswer %= N_MAX;
                }
            }
            answer -= subAnswer;
            answer %= N_MAX;
            if (answer < 0) {
                answer = N_MAX + answer;
            }
            writer.println(answer);
            writer.close();
            return;
        }
        long subAnswer = 1;
        long count1 = 1;
        for (int i = 0; i < lineOne.length(); i++) {
            if ((lineOne.charAt(i) == '?') && (lineTwo.charAt(i)) == '?') {
                subAnswer *= 55L;
                subAnswer %= N_MAX;
                count1 *= 10;
                count1 %= N_MAX;
            }
            if ((lineOne.charAt(i)) == '?' && (lineTwo.charAt(i) != '?')) {
                subAnswer *= (9 - (lineTwo.charAt(i) - '0') + 1);

                subAnswer %= N_MAX;

                count1 %= N_MAX;
            }
            if ((lineOne.charAt(i)) != '?' && (lineTwo.charAt(i) == '?')) {
                subAnswer *= ((lineOne.charAt(i) - '0') + 1);
                subAnswer %= N_MAX;

                count1 %= N_MAX;
            }
        }
        long sub = subAnswer;

        String buf = lineOne;
        lineOne = lineTwo;
        lineTwo = buf;
        subAnswer = 1;
        for (int i = 0; i < lineOne.length(); i++) {
            if ((lineOne.charAt(i) == '?') && (lineTwo.charAt(i)) == '?') {
                subAnswer *= 55L;

                subAnswer %= N_MAX;
            }
            if ((lineOne.charAt(i)) == '?' && (lineTwo.charAt(i) != '?')) {
                subAnswer *= (9 - (lineTwo.charAt(i) - '0') + 1);
                subAnswer %= N_MAX;
            }
            if ((lineOne.charAt(i)) != '?' && (lineTwo.charAt(i) == '?')) {
                subAnswer *= ((lineOne.charAt(i) - '0') + 1);
                subAnswer %= N_MAX;
            }
        }
        answer -= ((subAnswer + (sub) - count1) % N_MAX);
        answer %= N_MAX;
        if (answer < 0) {
            answer = N_MAX - answer;
        }
        writer.println(answer);

        writer.close();
    }

    public static void main(String[] args) {
        new Problem2().run();
    }
}
