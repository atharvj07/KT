listA = list(input())
done = []
store = [[], []]
even = 0
odd = 0
ans = ['First', 'Second']
idex = 0
for c in listA:
	if c not in done:
		done.append(c)
		v = listA.count(c)
		if v & 1:
			store[1].append(v)
			odd += 1
		else:
			store[0].append(v)
			even += 1
while 1:
	if odd == 0 or odd == 1:
		break
	else:
		if even > 0:
			next = store[0].pop(0)
			next -= 1
			if next > 0:
				store[1].append(next)
				odd += 1
			even -= 1
		else:
			next = store[1].pop(0)
			next -= 1
			if next > 0:
				store[0].append(next)
				even += 1
			odd -= 1
		idex ^= 1
print(ans[idex])
