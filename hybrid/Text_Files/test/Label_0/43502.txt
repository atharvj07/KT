n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = list(map(int, input().split()))
i = 0
j = 0
x = 0
y = 0
f = 0
ans = 0
if sum(a) == sum(b):
	while i < n and j < m:
		if f == 0:
			x += a[i]
			y += b[j]
		elif f == 1:
			y += b[j]
		elif f == 2:
			x += a[i]
		if x == y:
			i += 1
			j += 1
			x = 0
			y = 0
			ans += 1
			f = 0
		elif x > y:
			j += 1
			f = 1
		elif y > x:
			i += 1
			f = 2
	print(ans)
else:
	print(-1)
