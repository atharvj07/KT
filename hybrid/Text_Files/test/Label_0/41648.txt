import sys

def input():
	return sys.stdin.readline()[:-1]
t = int(input())
for _ in range(t):
	n = int(input())
	s = input()
	lis = [(s, 1)]
	for i in range(2, n):
		if (n - i) % 2 == 1:
			lis.append((s[i - 1:] + s[:i - 1], i))
		else:
			lis.append((s[i - 1:] + s[:i - 1][::-1], i))
	lis.append((s[::-1], n))
	lis.sort()
	print(lis[0][0])
	print(lis[0][1])
