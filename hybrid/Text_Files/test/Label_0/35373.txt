import java.util.*;
public class SoroBan {

	public static void main(String[] args) {
		Scanner scn = new Scanner(System.in);
		int num=scn.nextInt();
		String str="";
		int num2=1;
		while(num2!=0) {
			int rem=num%10;
			if(rem-5>=0)
			{	str=str+"-O|";
			int i=1;
				for(;i<=(rem-5);i++) {
					str=str+"O";
				}
				str=str+"-";
				for(int j=1;j+i-1<=4;j++)
					str=str+"O";
			}
			else {
				str=str+"O-|";
				for(int i=1;i<=(rem);i++) {
					str=str+"O";
				}
				str=str+"-";
				for(int j=1;j<=4-rem;j++)
					str=str+"O";
			}
			System.out.println(str);
			str="";
			num=num/10;
			num2=num;
		}
	}

}
