#pragma GCC optimize("O3")
#include <bits/stdc++.h>
using namespace std;
#define all(arr) (arr).begin(), (arr).end()
#define ll long long
#define ld long double
#define pb push_back
#define sz(x) int((x).size())
#define fi first
#define se second
#define endl '\n'

const int N = 22;
const ld eps = 1e-12;

ld c, m, p, v;
ld ans[N];

bool zero(ld a){
  return abs(a) < eps;
}

void proc(int iter, ld prob, ld cpr, ld mpr){
  ans[iter] += prob * (1 - cpr - mpr);
  if (zero(cpr) && zero(mpr)) return;
  if (!zero(cpr)){
    ld ncpr = max((ld)0, cpr - v), nmpr = mpr;
    if (!zero(mpr)) nmpr += min(cpr, v) / 2;
    proc(iter + 1, prob * cpr, ncpr, nmpr);
  }
  if (!zero(mpr)){
    ld nmpr = max((ld)0, mpr - v), ncpr = cpr;
    if (!zero(cpr)) ncpr += min(mpr, v) / 2;
    proc(iter + 1, prob * mpr, ncpr, nmpr);
  }
}

void solve(){
  for (int i = 0; i < N; i++) ans[i] = 0;
  cin >> c >> m >> p >> v;
  proc(1, 1, c, m);
  ld ev = 0;
  for (int i = 1; i < N; i++){
    ev += i * ans[i];
  }
  cout << ev << endl;
}

int main(){
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.precision(20);
  int t;
  cin >> t;
  while (t--){
    solve();
  }
}
