#include <bits/stdc++.h>
using namespace std;
using ll = long long;
template <class T>
void print(vector<T>& v, bool spaced = true) {
  int n = v.size();
  string space = (spaced ? " " : "");
  for (int i = 0; i < n; ++i) {
    cout << v[i] << space;
  }
  cout << '\n';
}
template <class T>
void printendl(vector<T>& v) {
  int n = v.size();
  for (int i = 0; i < n; ++i) {
    cout << v[i] << '\n';
  }
}
template <class T>
T gcd(T a, T b) {
  while (b) {
    a %= b;
    swap(a, b);
  }
  return a;
}
template <class T>
T lcm(T a, T b) {
  return a / gcd(a, b) * b;
}
template <class T>
T extended_gcd(T a, T b, T& x, T& y) {
  if (b == 0) {
    x = 1;
    y = 0;
    return a;
  }
  T x1, y1;
  T g = extended_gcd(b, a % b, x1, y1);
  x = y1;
  y = x1 - (a / b) * y1;
  return g;
}
template <class T>
T diophantine_eq(T a, T b, T c, T& x, T& y, T& g) {
  g = extended_gcd(abs(a), abs(b), x, y);
  if (c % g) return false;
  x *= (c / g);
  y *= (c / g);
  if (a < 0) x *= -1;
  if (b < 0) y *= -1;
  return true;
}
template <class T>
T euler_phi(T n) {
  T result = n;
  for (int i = 2; i * i <= n; ++i) {
    if (n % i == 0) {
      while (n % i == 0) {
        n /= i;
      }
      result -= result / i;
    }
  }
  if (n > 1) result -= result / n;
  return result;
}
vector<ll> ranged_euler_phi(ll n) {
  vector<ll> phi(n + 1);
  phi[0] = 0;
  phi[1] = 1;
  for (int i = 2; i <= n; ++i) phi[i] = i;
  for (int i = 2; i <= n; ++i) {
    if (phi[i] == i) {
      for (int j = i; j <= n; j += i) {
        phi[j] -= phi[j] / i;
      }
    }
  }
  return phi;
}
bool check(int n) {
  int count = 0;
  for (int i = 2; i * i <= n; ++i) {
    if (n % i == 0) {
      count++;
      while (n % i == 0) {
        n /= i;
      }
    }
  }
  if (n > 1) count++;
  return count == 2;
}
