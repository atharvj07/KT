#include <bits/stdc++.h>
using namespace std;
struct node {
  int v, nex;
} e[1000005];
int son[1000005], n, q, ans[1000005], cnt = 0, head[1000000], a, fa[1000005],
                                      s[1000005], p[1000008];
void add(int u, int v) { e[++cnt].v = v, e[cnt].nex = head[u], head[u] = cnt; }
void dfs(int x) {
  ans[x] = x;
  s[x] = 1;
  int maxn = 0, t = 0;
  for (int i = head[x]; i; i = e[i].nex) {
    int v = e[i].v;
    dfs(v);
    s[x] += s[v];
    if (s[v] > maxn) maxn = s[v], t = v;
  }
  p[x] = maxn;
  if (p[x] * 2 < s[x])
    ans[x] = x;
  else {
    int k = ans[t];
    while (fa[k] && max(p[k], s[x] - s[k]) > max(p[fa[k]], s[x] - s[fa[k]]))
      k = fa[k];
    ans[x] = k;
  }
}
int main() {
  cin >> n >> q;
  for (int i = 2; i <= n; i++) scanf("%d", &a), add(a, i), fa[i] = a;
  dfs(1);
  for (int i = 1; i <= q; i++) scanf("%d", &a), printf("%d\n", ans[a]);
}
