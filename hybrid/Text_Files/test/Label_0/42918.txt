(n, k, a, b) = map(int, input().split())
if a == b:
	print('GB' * (n // 2))
else:
	(maxx, minn) = (max(a, b), min(a, b))
	if maxx / (minn + 1) > k:
		print('NO')
	else:
		x = maxx // (minn + 1)
		y = maxx - (minn + 1) * x
		if a > b:
			x1 = 'G'
			x2 = 'B'
		else:
			x1 = 'B'
			x2 = 'G'
		arr = [x1 * x for i in range(minn + 1)]
		for i in range(y):
			arr[i] += x1
		print(x2.join(arr))
