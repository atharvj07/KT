public class asfa {

	static int find1(int[] a,int x) {
		int l=0;
		int idx =-1;
		int m=a.length-1;
		int mid=0;
		while(l<=m) {
			mid =l+m >>1;

			if(a[mid]>=x) {
				m=mid-1;
			}
			else {
				l=mid+1;
				idx=mid;
			}
		}
		return idx;
	}
	static int find2(int[] a,int x) {
		int l=0;
		int idx =-1;
		int m=a.length-1;
		int mid=0;
		while(l<=m) {
			mid =l+m >>1;
			if(a[mid]<=x) {
				l=mid+1;
			}
			else {
				m=mid-1;
				idx=mid;
			}
		}
		return idx;
	}
	public static boolean disdig(int x) {
		int[] a =new int[4];
		int i=-1;
		while(x!=0) {
			a[++i]=x%10;
			x/=10;
		}
		for(i=0;i<4;i++) {
			for(int j=i;j<3;j++) {
				if(a[i]==a[j+1])
					return false;
			}
		}
		return true;
	}
	public static boolean lucky(long x) {
		if(x==0)
			return false;
		while(x!=0) {
			if(x%10!=7 && x%10!=4)
				return false;
			x/=10;
		}
		return true;
	}
	static int countdig(long x) {
		int c=0;
		while(x!=0) {
			if(x%10==7 || x%10==4)
				c++;
			x/=10;
		}
		return c;
	}
	static boolean isprime (int x) {
		int n =(int)(Math.sqrt(x)+1);
		for(int i=2;i<=n;i++) {
			if(x%i==0)
				return false;
		}
		return true;
	}

				System.out.println("+");
			else
				System.out.println("?");
		}
		else if(n<m) {
			if(m-n>k)
				System.out.println("-");
			else
				System.out.println("?");
		}
		else {
			if(k==0)
				System.out.println(0);
			else

		}

	}
