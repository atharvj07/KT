m = 998244353
(n, k) = [int(d) for d in input().split()]
a = [int(d) for d in input().split()]
d = {}
for i in range(n):
	d[a[i]] = i
e = 0
d[n] = -1
for i in range(n - 1):
	if d[a[i] + 1] < d[a[i + 1] + 1]:
		e = e + 1
if k + e < n:
	print(0)
else:
	num = 1
	den = 1
	for i in range(1, n + 1):
		num = num * (k + e - n + i) % m
		den = den * i % m
	x = pow(den, m - 2, m)
	print(num * x % m)
