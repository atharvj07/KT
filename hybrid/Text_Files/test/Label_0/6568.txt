#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

int main(){
  ll n;
  while(cin>>n&&n){
    ll ans=0;
    for(ans=100000;;ans--){
      // (a+(a+ans-1) )/ 2 * ans == n
      // ( a*2+ans-1 ) * ans == n*2
      // a*2*ans + ans*ans - ans == n*2
      // a*2*ans = (n*2 - ans*ans + ans)
      ll a=(n*2-ans*ans+ans)/2/ans;

      if(0<a&& (a*2+ans-1)*ans==n*2 ){
        cout<<a<<' '<<ans<<endl;
        break;
      }
    }
  }
  return 0;
}
