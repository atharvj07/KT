t = int(input())

def solver(arr, n):
	s = sum(arr)
	count = 0
	if s % n != 0:
		return -1
	else:
		thresh = sum(arr) // n
		for i in range(n):
			if arr[i] > thresh:
				count += 1
	return count
res = []
for i in range(t):
	n = int(input())
	arr = [int(item) for item in input().split()]
	res.append(solver(arr, n))
for r in res:
	print(r)
