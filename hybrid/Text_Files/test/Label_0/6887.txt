#include <bits/stdc++.h>
using namespace std;
using ll=long long;

int main() {
    ll N;
    cin >> N;
    ll t=1;
    while(t*(t-1) < 2*N){
        t++;
    }
    if(t*(t-1) != 2*N){
        cout <<"No" << endl;
        return 0;
    }
    cout <<"Yes" << endl;
    cout <<t <<endl;
    vector<vector<ll>> ans(t);
    ll num =1;
    for(int i=0;i<t;i++){
        for(int j=(i+1);j<t;j++){
            ans[i].push_back(num);
            ans[j].push_back(num);
            num++;
        }
    }
    for(int i=0;i<t;i++){
        cout << ans[i].size() <<" ";
        for(int a:ans[i]){
            cout << a <<" ";
        }
        cout <<endl;
    }

    return 0;
}
