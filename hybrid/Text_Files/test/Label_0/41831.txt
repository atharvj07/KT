def check(n, k, d1, d2):
	if (k - d1 - d2) % 3 != 0:
		return False
	y = (k - d1 - d2) // 3
	return min(y + d1, y, y + d2) >= 0 and max(y + d1, y, y + d2) <= n // 3
t = int(input())
all_res = []
for _ in range(t):
	(n, k, d1, d2) = map(int, input().split())
	if n % 3 != 0:
		all_res += ['no']
	elif check(n, k, d1, d2) or check(n, k, -d1, d2) or check(n, k, d1, -d2) or check(n, k, -d1, -d2):
		all_res += ['yes']
	else:
		all_res += ['no']
print('\n'.join(all_res))
