#include<bits/stdc++.h>
typedef long long ll;
using namespace std;
ll dp[402][402];
ll c[402][402];
ll kp[404];
int main(void)
{
	ll n,m;
	cin>>n>>m;
	ll i,j;
	for(i=0;i<=401;i++)
	{
		c[i][0]=1;
		for(j=1;j<=i;j++)
		{
			c[i][j]=c[i-1][j-1]+c[i-1][j];
			c[i][j]%=m;
		}
	}
	kp[1]=2;kp[0]=1;
	for(i=2;i<=400;i++)kp[i]=kp[i-1]*2%m;
	dp[1][1]=1;
	ll h;
	for(i=2;i<=n;i++)
	{
		for(j=1;j<=i;j++)
		{
			for(h=2;h<=i-1;h++)
			{
				ll len=i-h;
				if(len<=j)dp[i][j]+=dp[h-1][j-len]*c[j][len]%m*kp[len-1]%m;
				dp[i][j]%=m;
			}
		}
		dp[i][i]=(dp[i][i]+kp[i-1])%m;
	}
	ll ans=0;
	for(i=1;i<=n;i++)ans+=dp[n][i];
	cout<<ans%m<<endl;
}
