from collections import Counter

def solve(t, m, b):
	c = Counter(t)
	letters = list(sorted(c))
	ix = len(letters) - 1
	res = [None] * m
	solved = 0
	while solved < m:
		zeros = [i for (i, v) in enumerate(b) if not res[i] and (not v)]
		lz = len(zeros)
		while c[letters[ix]] < lz:
			ix -= 1
		ll = letters[ix]
		ix -= 1
		for z in zeros:
			res[z] = ll
		solved += lz
		for i in range(m):
			if not res[i]:
				for z in zeros:
					b[i] -= abs(z - i)
	return ''.join(res)
q = int(input())
for i_q in range(q):
	t = input()
	m = int(input())
	b = list(map(int, input().split(' ')))
	print(solve(t, m, b))
