#include <bits/stdc++.h>
using namespace std;
const int mod = 998244353;
const int max_n = 100005;
int n, k;
int c[3], w[3];
int val[1 << 2];
int A[1 << 2][1 << 17], B[1 << 2], C[1 << 17];
int all = 0;
int quick_pow(int a, int b) {
  int res = 1;
  while (b) {
    if (b & 1) res = 1ll * res * a % mod;
    b >>= 1;
    a = 1ll * a * a % mod;
  }
  return res;
}
void fwt_xor(int* a, int l, int opt) {
  for (int i = 1; i < l; i <<= 1) {
    for (int j = 0; j < l; j += (i << 1)) {
      for (int k = 0; k < i; k++) {
        int tmp = a[j + k];
        a[j + k] = (tmp + a[j + k + i]) % mod;
        a[j + k + i] = (tmp + mod - a[j + k + i]) % mod;
      }
    }
  }
  if (opt == -1) {
    int inv = quick_pow(l, mod - 2);
    for (int i = 0; i < l; i++) a[i] = 1ll * a[i] * inv % mod;
  }
}
int main() {
  scanf("%d%d", &n, &k);
  for (int i = 0; i < 3; i++) scanf("%d", c + i);
  for (int i = 0; i < (1 << 2); i++) {
    val[i] = c[0];
    for (int j = 0; j < 2; j++) {
      if (i >> j & 1)
        val[i] += mod - c[j + 1], val[i] %= mod;
      else
        val[i] += c[j + 1], val[i] %= mod;
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 3; j++) scanf("%d", w + j);
    all ^= w[0];
    for (int j = 1; j < 3; j++) w[j] ^= w[0];
    for (int j = 0; j < (1 << 2); j++) {
      int wei = 0;
      for (int l = 0; l < 2; l++)
        if (j >> l & 1) wei ^= w[l + 1];
      A[j][wei]++;
    }
  }
  for (int i = 0; i < (1 << 2); i++) fwt_xor(A[i], 1 << k, 1);
  for (int i = 0; i < (1 << k); i++) {
    for (int j = 0; j < (1 << 2); j++) B[j] = A[j][i];
    fwt_xor(B, 1 << 2, -1);
    int res = 1;
    for (int j = 0; j < (1 << 2); j++)
      res = 1ll * res * quick_pow(val[j], B[j]) % mod;
    C[i] = res;
  }
  fwt_xor(C, 1 << k, -1);
  for (int i = 0; i < (1 << k); i++) printf("%d ", C[all ^ i]);
  printf("\n");
  return 0;
}
