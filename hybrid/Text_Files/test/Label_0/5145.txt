using namespace std;
typedef long long LL;
typedef pair<int,int> par;
const int N=100010;
set <int> s[N],s2[N];
set <int> :: iterator it;
int n,t,fa[N];
par q[N<<4];
int get(int x){return fa[x]?fa[x]=get(fa[x]):x;}
int merge(int u,int v){
	if ((u=get(u))==(v=get(v))) return 0;
	if (s[u].size()+s2[u].size()<s[v].size()+s2[v].size()) swap(u,v);
	fa[v]=u;
	s[u].erase(s[u].find(v));
	s2[u].erase(s2[u].find(v));
	for (it=s[v].begin();it!=s[v].end();++it){
		if (*it==u) continue;
		int x=*it;
		s[u].insert(x);
		s[x].erase(s[x].find(v));
		s[x].insert(u);
		if (s2[x].find(u)!=s2[x].end()) q[++t]=(par){x,u};
	}
	for (it=s2[v].begin();it!=s2[v].end();++it){
		if (*it==u) continue;
		int x=*it;
		s2[u].insert(x);
		s2[x].erase(s2[x].find(v));
		s2[x].insert(u);
		if (s[x].find(u)!=s[x].end()) q[++t]=(par){x,u};
	}
	return 1;
}
int main(){
	scanf("%d",&n);
	rep(i,2,n){
		int x,y;
		scanf("%d%d",&x,&y);
		s[x].insert(y);
		s[y].insert(x);
	}
	rep(i,2,n){
		int x,y;
		scanf("%d%d",&x,&y);
		s2[x].insert(y);
		s2[y].insert(x);
		if (s[x].find(y)!=s[x].end()) q[++t]=(par){x,y};
	}
	int tot=n;
	rep(h,1,t) tot-=merge(q[h].fi,q[h].se);
	if (tot==1) puts("YES"); else puts("NO");
	return 0;
}
