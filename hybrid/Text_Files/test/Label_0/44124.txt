from sys import stdin

def main():
	t = int(stdin.readline())
	for _ in range(t):
		(n, p) = map(int, stdin.readline().split())
		for i in range(n):
			print(i + 1, (i + 1) % n + 1)
		count = n + p
		ok = False
		for i in range(n):
			for j in range(i + 1, n):
				if (i - 1) % n != j and (i + 1) % n != j:
					print(i + 1, j + 1)
					count -= 1
					if count == 0:
						ok = True
						break
			if ok:
				break
main()
