import math, sys

def round_slightly(x):
	return x + 2 ** 20 - 2 ** 20
points = []
lines = []
n = int(input())
for i in range(n):
	(x, y) = map(float, input().split())
	points.append((x, y))
grad_freqs = {}
for i in range(len(points)):
	for j in range(i + 1, len(points)):
		((x, y), (a, b)) = (points[i], points[j])
		y2_y1 = b - y
		x2_x1 = a - x
		if x == a:
			gradient = float('Inf')
			intercept = x
		else:
			gradient = y2_y1 / x2_x1
			intercept = y - gradient * x
		gradient = round_slightly(gradient)
		intercept = round_slightly(intercept)
		if gradient not in grad_freqs:
			grad_freqs[gradient] = set()
		grad_freqs[gradient].add(intercept)
total_lines = 0
for key in grad_freqs.keys():
	grad_freqs[key] = len(grad_freqs[key])
	total_lines += grad_freqs[key]
res = 0
values = grad_freqs.values()
for (idx, item) in enumerate(values):
	res += (total_lines - item) * item
print(res // 2)
