string S;
struct State {
    int i;
    char check() {
        if (i == S.size()) return '$';
        return S[i];
    }
    char next() {
        char ret = check();
        i++;
        return ret;
    }
};

int or_exp(State &st, int p);
int and_exp(State &st, int p);
int term(State &st, int p);

pii memo_or[200005][2];
pii memo_and[200005][2];

int or_exp(State &st, int p) {
    auto &MEMO = memo_or[st.i][p];
    if (MEMO.first != INF) {
        st.i = MEMO.second;
        return MEMO.first;
    }

    if (p) {
        // 1
        State bkup = st;
        int zero = and_exp(st, 0);
        st = bkup;
        int one = and_exp(st, 1);
        
        int ret = one;
        while (st.check() == '|') {
            st.next();
            bkup = st;
            int tmp_z = and_exp(st, 0);
            st = bkup;
            int tmp_o = and_exp(st, 1);

            chmin(ret, zero + tmp_o);
            zero += tmp_z;
        }
        MEMO.second = st.i;
        return MEMO.first = ret;
    } else {
        // 0
        int ret = and_exp(st, 0);
        while (st.check() == '|') {
            st.next();
            ret += and_exp(st, 0);
        }
        MEMO.second = st.i;
        return MEMO.first = ret;
    }
}
int and_exp(State &st, int p) {
    auto &MEMO = memo_and[st.i][p];
    if (MEMO.first != INF) {
        st.i = MEMO.second;
        return MEMO.first;
    }

    if (p) {
        // 1
        int ret = term(st, 1);
        while (st.check() == '&') {
            st.next();
            ret += term(st, 1);
        }
        MEMO.second = st.i;
        return MEMO.first = ret;
    } else {
        // 0
        State bkup = st;
        int zero = term(st, 0);
        st = bkup;
        int one = term(st, 1);
        
        int ret = zero;
        while (st.check() == '&') {
            st.next();
            bkup = st;
            int tmp_z = term(st, 0);
            st = bkup;
            int tmp_o = term(st, 1);

            chmin(ret, one + tmp_z);
            one += tmp_o;
        }
        MEMO.second = st.i;
        return MEMO.first = ret;
    }
}

int term(State &st, int p) {
    if (st.check() == '(') {
        st.next();
        int ret = or_exp(st, p);
        assert(st.check() == ')');
        st.next();
        return ret;
    } else {
        assert(st.check() == '?');
        st.next();
        return 1;
    }
}
