t = int(input())
for _ in range(t):
	n = int(input())
	l1 = [int(x) for x in input().split()]
	l2 = [int(x) for x in input().split()]
	min1 = min(l1)
	min2 = min(l2)
	ans = 0
	for x in range(len(l1)):
		if l1[x] - min1 <= l2[x] - min2:
			first = l1[x] - min1
			second = l2[x] - first - min2
		else:
			first = l2[x] - min2
			second = l1[x] - first - min1
		ans += first + second
	print(ans)
