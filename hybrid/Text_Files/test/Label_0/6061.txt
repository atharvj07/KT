#include <bits/stdc++.h>
using namespace std;
const int LOGNMAX = 30;
const int NMAX = 2e5 + 7;
const int INF = 2e6 + 7;
int n, m, Q;
int p[NMAX], a[NMAX];
int vidjen[NMAX];
int skoci[LOGNMAX][NMAX];
int sled[NMAX];
int treba[NMAX];
struct segTree {
  int t[4 * NMAX];
  void update(int node, int st, int en, int idx, int val) {
    if (st == en) {
      t[node] = val;
      return;
    }
    int mid = (st + en) / 2;
    if (idx <= mid)
      update(2 * node, st, mid, idx, val);
    else
      update(2 * node + 1, mid + 1, en, idx, val);
    t[node] = min(t[2 * node], t[2 * node + 1]);
  }
  int get(int node, int st, int en, int l, int r) {
    if (r < st || l > en || st > en) return INF;
    if (l <= st && en <= r) return t[node];
    int mid = (st + en) / 2;
    int m1 = get(2 * node, st, mid, l, r);
    int m2 = get(2 * node + 1, mid + 1, en, l, r);
    return min(m1, m2);
  }
} T;
int main() {
  ios_base::sync_with_stdio(false);
  cin >> n >> m >> Q;
  for (int i = 1; i <= n; i++) cin >> p[i];
  for (int i = 1; i <= m; i++) cin >> a[i];
  for (int i = 1; i <= n - 1; i++) sled[p[i]] = p[i + 1];
  sled[p[n]] = p[1];
  for (int i = 1; i <= n; i++) vidjen[i] = m + 1;
  skoci[0][m + 1] = m + 1;
  for (int i = m; i; i--) {
    skoci[0][i] = vidjen[sled[a[i]]];
    vidjen[a[i]] = i;
  }
  for (int i = 1; i < LOGNMAX; i++)
    for (int j = 1; j <= m + 1; j++)
      skoci[i][j] = skoci[i - 1][skoci[i - 1][j]];
  for (int i = 1; i <= m; i++) {
    int idi = i;
    for (int j = 0; j < 22; j++)
      if ((n - 1) & (1 << j)) idi = skoci[j][idi];
    treba[i] = idi;
  }
  for (int i = 1; i <= m; i++) T.update(1, 1, m, i, treba[i]);
  while (Q--) {
    int l, r;
    cin >> l >> r;
    int odg = T.get(1, 1, m, l, r) <= r;
    cout << odg;
  }
  return 0;
}
