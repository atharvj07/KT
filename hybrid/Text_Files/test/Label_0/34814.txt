import java.util.*;
public class main {

	/**
	 * @param args
	 */
	
	static int steps = 0;

	static LinkedList<Integer>[] adj;
	static int[] colour;
	static boolean[] visited;
	static void dfs(int s,int c,boolean[] visited){
		visited[s] = true;
		int len = adj[s].size();
		
		for(int i=0;i<len;i++){
			if(!visited[adj[s].get(i)]){
				
				if(c==colour[adj[s].get(i)]){
					dfs(adj[s].get(i),c,visited);
				}
				else{
					steps++;
					dfs(adj[s].get(i),colour[adj[s].get(i)],visited);
				}
				
			}
		}
		
	}
	
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		
		int n = sc.nextInt();
		adj = new LinkedList[n];
		for(int i=0;i<n;i++){
			adj[i] = new LinkedList();
		}
		colour = new int[n];
		visited = new boolean[n];
		Arrays.fill(visited, false);
		for(int i=1;i<=n-1;i++){
			int p  = sc.nextInt();
			adj[i].add(p-1);
			adj[p-1].add(i);
		}
		
		for(int i=0;i<n;i++){
			colour[i] = sc.nextInt();
		}
		
		dfs(0,colour[0],visited);
		System.out.println(steps+1);
	}

}
