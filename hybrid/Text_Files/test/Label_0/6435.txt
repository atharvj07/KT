#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, t, c1, c2, a, in1, in2;
  c1 = c2 = 0;
  cin >> n >> t;
  vector<int> v;
  for (int i = 0; i < n; i++) {
    cin >> a;
    v.push_back(a);
  }
  for (int j = 0; j < n; j++) {
    if (v[j] == 1)
      c1++;
    else
      c2++;
  }
  for (int i = 0; i < t; i++) {
    cin >> in1 >> in2;
    if ((in2 - in1 + 1) % 2 != 0) {
      cout << 0 << endl;
    } else {
      if (c1 >= (in2 - in1 + 1) / 2 && c2 >= (in2 - in1 + 1) / 2) {
        cout << 1 << endl;
      } else {
        cout << 0 << endl;
      }
    }
  }
  return 0;
}
