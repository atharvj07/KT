#include <bits/stdc++.h>
using namespace std;
char arr[51][51];
bool v[51][51];
bool dp[51][51];
long long n, m;
long long d[4][2] = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
void ip(long long i, long long j) {
  if (arr[i][j] == '#' || arr[i][j] == 'B') return;
  if (dp[i][j]) return;
  dp[i][j] = true;
  v[i][j] = true;
  if (i - 1 >= 0 && !v[i - 1][j]) ip(i - 1, j);
  if (i + 1 < n && !v[i + 1][j]) ip(i + 1, j);
  if (j - 1 >= 0 && !v[i][j - 1]) ip(i, j - 1);
  if (j + 1 < m && !v[i][j + 1]) ip(i, j + 1);
  v[i][j] = false;
}
signed main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  long long t;
  cin >> t;
  while (t--) {
    cin >> n >> m;
    long long gc = 0, bc = 0;
    for (long long i = 0; i < n; i++) {
      for (long long j = 0; j < m; j++) {
        v[i][j] = false;
        dp[i][j] = false;
        cin >> arr[i][j];
        if (arr[i][j] == 'G')
          gc++;
        else if (arr[i][j] == 'B')
          bc++;
      }
    }
    if (!gc && !bc) {
      cout << "Yes" << '\n';
      continue;
    }
    long long flag = 1;
    for (long long i = 0; i < n; i++) {
      for (long long j = 0; j < m; j++) {
        if (arr[i][j] == 'B') {
          for (long long k = 0; k < 4; k++) {
            long long i1 = i + d[k][0];
            long long j1 = j + d[k][1];
            if (i1 >= 0 && i1 < n && j1 >= 0 && j1 < m) {
              if (arr[i1][j1] == '.')
                arr[i1][j1] = '#';
              else if (arr[i1][j1] == 'G')
                flag = 0;
            }
            if (!flag) break;
          }
        }
        if (!flag) break;
      }
      if (!flag) break;
    }
    if (arr[n - 1][m - 1] == 'B') flag = 0;
    if (!flag) {
      cout << "No" << '\n';
      continue;
    }
    ip(n - 1, m - 1);
    for (long long i = 0; i < n; i++) {
      for (long long j = 0; j < m; j++) {
        if (arr[i][j] == 'G') {
          if (!dp[i][j]) flag = 0;
          if (!flag) break;
        }
      }
      if (!flag) break;
    }
    if (flag)
      cout << "Yes" << '\n';
    else
      cout << "No" << '\n';
  }
}
