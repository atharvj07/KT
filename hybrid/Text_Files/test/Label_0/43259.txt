n = int(input())
s = list(map(int, input().split()))
d25 = 0
d50 = 0
answer = 'YES'
for i in range(n):
	if s[i] == 25:
		d25 += 1
		continue
	elif s[i] == 50:
		d50 += 1
		d25 -= 1
		if d25 < 0:
			answer = 'NO'
			break
	elif d50 >= 1 and d25 >= 1:
		d50 -= 1
		d25 -= 1
		continue
	elif d50 < 1 and d25 >= 3:
		d25 -= 3
		continue
	else:
		answer = 'NO'
		break
print(answer)
