const int N=1e6+7;
const int maxn=1e6+7; long long dep[N],f1[N],f2[N],sum[N];
int n,m,fa[N],ft[N],xa[N],ya[N],za[N],FF[N]; vector<int>v[N],w[N],sq[N],v1[N],v2[N]; bool fl[N],fs[N];
inline long long gcd(long long a,long long b){
	if(!b) return a; return gcd(b,a%b);
}

}edge[N];
queue <int> q;
vector <int> cb[N];
vector <int> rdr[N];
int ans[N],totq,x,y,rd[maxn],u,vis[maxn],dis_[maxn],dis[maxn];
int dfn[N],low[maxn],f[maxn],times,cntqq;
int stack_[N],heads[N],visit[N],cnt,tot,index_;
void tarjan(int x)
{
    dfn[x]=low[x]=++times;
    stack_[++index_]=x;	
    visit[x]=1;
   	for(int i=0;i<v[x].size();i++)
    {
    	int u=v[x][i];
        if(!dfn[u])
        {
           	tarjan(u);
            low[x]=min(low[x],low[u]);
       	}
       	else 
        if(visit[u])
        low[x]=min(low[x],dfn[u]);
    }
    if(low[x]==dfn[x])
   	{
   		tot++;//强连通分量编号 
   		while(1)
       	{
       		vis[stack_[index_]]=tot;
       		dis_[tot]+=dis[stack_[index_]];	
       		visit[stack_[index_]]=0;index_--;
       		if(x==stack_[index_+1])break;
       	}
    }
}
inline void dfs(int u){
	fl[u]=1;
	for(int i=0;i<v[u].size();i++){
		int x=v[u][i]; if(vis[u]!=vis[x]||fl[x]) continue;
		fa[x]=u,ft[x]=w[u][i],dep[x]=dep[u]+ft[x],dfs(x);
		FF[sq[u][i]]=1;
	}
}
inline void dfs2(int u){
	fl[u]=1;
	for(int i=0;i<v1[u].size();i++){
		int x=v1[u][i]; if(vis[u]!=vis[x]) continue;
//		cout<<u<<","<<x<<endl;
		if(!fl[x]){
			dep[x]=dep[u]+v2[u][i]; dfs2(x);
		}
		else{
			sum[vis[u]]=gcd(sum[vis[u]],v2[u][i]+dep[u]-dep[x]);
		}
	}
}
int main(){
	scanf("%d%d",&n,&m); int a,b,c;
	for(int i=1;i<=m;i++) scanf("%d%d%d",&a,&b,&c),v[a].push_back(b),w[a].push_back(c),sq[a].push_back(i),xa[i]=a,ya[i]=b,za[i]=c;
	for(int i=1;i<=n;i++) if(!dfn[i])tarjan(i);
	for(int i=1;i<=n;i++) if(!fl[i]) dfs(i),fs[i]=1;
	for(int i=1;i<=m;i++){
		a=xa[i],b=ya[i],c=za[i];
		v1[a].push_back(b),v2[a].push_back(c);
		v1[b].push_back(a),v2[b].push_back(-c);
	}
	memset(fl,0,sizeof(fl));
	for(int i=1;i<=n;i++) if(!fl[i]) dfs2(i);
	int q; cin>>q;
//	cout<<sum[vis[10]]<<endl;
	while(q--){
		scanf("%d%d%d",&a,&b,&c); int P=gcd(b,c);
		a=vis[a];
		if(b==0) puts("YES");
		else if(sum[a]==0) puts("NO");
		else if(P%gcd(c,sum[a])==0)  puts("YES");
		else puts("NO");
	}
}
