#include <bits/stdc++.h>
using namespace std;
template <typename TH>
void _dbg(const char* sdbg, TH h) {
  cerr << sdbg << " = " << h << "\n";
}
template <typename TH, typename... TA>
void _dbg(const char* sdbg, TH h, TA... t) {
  while (*sdbg != ',') cerr << *sdbg++;
  cerr << " = " << h << ",";
  _dbg(sdbg + 1, t...);
}
const int N = 1e5 + 3;
int64_t f[N], s[N];
int NumberOfDigit(int n) {
  int res = 0;
  while (n) {
    res++;
    n /= 10;
  }
  return res;
}
void Init() {
  for (int i = (1), _b = (N - 1); i <= _b; ++i) {
    f[i] = f[i - 1] + NumberOfDigit(i);
    s[i] = s[i - 1] + f[i];
  }
}
int Calc(int k) {
  int pos = 0;
  for (int i = (N - 1), _b = (0); i >= _b; --i) {
    if (s[i] < k) {
      k -= s[i];
      pos = i + 1;
      break;
    }
  }
  int num = 0;
  for (int i = (pos), _b = (0); i >= _b; --i) {
    if (f[i] < k) {
      k -= f[i];
      num = i + 1;
      break;
    }
  }
  vector<int> digit;
  while (num) {
    digit.push_back(num % 10);
    num /= 10;
  }
  reverse(digit.begin(), digit.end());
  return digit[k - 1];
}
void Solve() {
  int q;
  cin >> q;
  while (q--) {
    int k;
    cin >> k;
    cout << Calc(k) << '\n';
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  Init();
  Solve();
  return 0;
}
