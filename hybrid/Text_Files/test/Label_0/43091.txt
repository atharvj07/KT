import collections
t = int(input())
for _ in range(t):
	s = input()
	n = int(input())
	nums = list(map(int, input().split()))
	cnt = list(collections.Counter(s).items())
	cnt.sort()
	(count, res, k) = (0, [0] * n, len(cnt) - 1)
	while count < n:
		que = []
		for i in range(n):
			if nums[i] == 0:
				que.append(i)
				nums[i] = -1
				count += 1
		for i in range(n):
			if nums[i] > 0:
				nums[i] -= sum((abs(i - j) for j in que))
		while cnt[k][1] < len(que):
			k -= 1
		for i in que:
			res[i] = cnt[k][0]
		k -= 1
	print(''.join(res))
