from sys import stdin

def get_ints():
	return list(map(int, stdin.readline().strip().split()))
for _ in range(int(input())):
	n = int(input())
	ar = input()
	cs = [0] * n
	cs[0] = 1 if ar[0] == '*' else 0
	for i in range(1, n):
		cs[i] = cs[i - 1] + (1 if ar[i] == '*' else 0)
	ans = 0
	for i in range(n):
		if ar[i] == '.':
			ans += min(cs[i], cs[-1] - cs[i])
	print(ans)
