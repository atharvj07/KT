def make_grid(r, c, row, col):
	mod = 10 ** 9 + 7
	grid = [[-1 for i in range(c)] for _ in range(r)]
	for i in range(r):
		k = row[i]
		for j in range(k):
			grid[i][j] = 'b'
		if k < c:
			grid[i][k] = 's'
	for i in range(c):
		k = col[i]
		for j in range(k):
			if grid[j][i] == 's':
				return 0
			grid[j][i] = 'b'
		if k < r:
			if grid[k][i] == 'b':
				return 0
			grid[k][i] = 's'
	count = 0
	ans = 1
	for i in range(r):
		for j in range(c):
			if grid[i][j] == -1:
				count += 1
				ans = ans * 2
				ans = ans % mod
	return ans % mod
(r, c) = map(int, input().split())
row = list(map(int, input().split()))
col = list(map(int, input().split()))
print(make_grid(r, c, row, col))
