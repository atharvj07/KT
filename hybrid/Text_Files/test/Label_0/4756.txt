#include <bits/stdc++.h>
using namespace std;
int main() {
  long long t, s, x;
  cin >> t >> s >> x;
  if (x < t) {
    cout << "NO\n";
    return 0;
  } else if (x == t) {
    cout << "YES\n";
    return 0;
  }
  if ((x - t) % s == 0 && (x - t) >= s) {
    cout << "YES\n";
    return 0;
  } else if ((x - t - 1) % s == 0 && (x - t - 1) >= s) {
    cout << "YES\n";
    return 0;
  } else
    cout << "NO\n";
  return 0;
}
