def ub(arr, lo, hi, val):
	while lo < hi:
		mid = (lo + hi) // 2
		if arr[mid] > val:
			hi = mid - 1
		elif arr[mid] <= val:
			lo = mid + 1
	if lo == len(arr) - 1 and arr[-1] <= val:
		lo = len(arr)
	return lo

def lb(arr, lo, hi, val):
	while lo < hi:
		mid = (lo + hi) // 2
		if arr[mid] < val:
			lo = mid + 1
		elif arr[mid] >= val:
			hi = mid - 1
	if lo == len(arr) - 1 and arr[-1] < val:
		lo = len(arr)
	return lo
tc = 1
tc = int(input())
for _ in range(tc):
	n = int(input())
	a = list(map(int, input().split()))
	mx = max(a)
	mn = min(a)
	(mxs, mns) = (-1, -1)
	for i in range(n):
		if a[i] == mx and mxs == -1:
			mxs = i
		if a[i] == mn and mns == -1:
			mns = i
	(mxe, mne) = (-1, -1)
	for i in range(n - 1, -1, -1):
		if a[i] == mx and mxe == -1:
			mxe = i
		if a[i] == mn and mne == -1:
			mne = i
	ans = min(max(mxs, mns) + 1, max(n - 1 - mxe, n - 1 - mne) + 1, 2 + mxs + n - 1 - mne, 2 + mns + n - 1 - mxe)
	print(ans)
