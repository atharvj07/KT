
import java.io.*;
import java.util.*;
public class A1214 {
    public static void main(String args[])throws IOException
        {
            Scanner sc=new Scanner(System.in);
            int n=sc.nextInt();
            int d=sc.nextInt();
            int e=sc.nextInt();
            e=e*5;int min=n;
            for(int i=0;i*d <=n;i++)
            {
                int m=(n- i*d)%e;
                min=Math.min(min, m);
            }
            System.out.println(min);
        }

}
