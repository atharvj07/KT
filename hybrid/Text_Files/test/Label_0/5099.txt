#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long a, b, c, count = 0;
  scanf("%lld %lld", &a, &b);
  while (a--) {
    scanf("%lld", &c);
    if (c <= b)
      count++;
    else
      count += 2;
  }
  printf("%lld", count);
}
