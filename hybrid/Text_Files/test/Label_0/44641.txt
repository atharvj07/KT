from math import sqrt
t = int(input())

def factor(n):
	factors = []
	div = int(sqrt(n))
	i = 2
	count = 1
	aux_n = n
	while n > 1 and i <= div:
		while n % i == 0:
			factors.append(i)
			n = n // i
			count *= i
		i += 1
	if n > 1:
		factors.append(aux_n // count)
	return factors
for _ in range(t):
	n = int(input())
	primes = factor(n)
	ans = []
	i = 0
	while i < len(primes):
		count = primes[i]
		if count not in ans:
			ans.append(count)
			i += 1
		else:
			i += 1
			while i < len(primes) and count in ans:
				count *= primes[i]
				i += 1
			ans.append(count)
	if len(ans) > 2:
		last = 1
		for i in range(2, len(ans)):
			last *= ans[i]
		ans = [ans[0], ans[1], last]
		if len(set(ans)) == 3:
			print('YES')
			print(ans[0], ans[1], last)
		else:
			print('NO')
	else:
		print('NO')
