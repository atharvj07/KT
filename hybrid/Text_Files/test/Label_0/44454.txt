(n, m) = map(int, input().split())
c = [[] for i in range(m)]
for i in range(n):
	a = list(map(int, input().split()))
	for j in range(m):
		c[j] += [a[j]]
total = 0

def match(arr, start, m):
	c = 0
	for i in range(len(arr)):
		if arr[i] != start + i * m:
			c += 1
	return c
t = 0
for i in range(m):
	col = c[i]
	mina = [n + i for i in range(n)]
	for j in range(n):
		col[j] = (col[j] - i - 1) // m + 1 if (col[j] - i - 1) % m == 0 else -1
		if col[j] > n * m:
			col[j] = -1
	for j in range(n):
		if col[j] != -1:
			mina[(n - (col[j] - j - 1)) % n] -= 1
	t += min(mina)
print(t)
