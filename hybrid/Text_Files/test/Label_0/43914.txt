import sys
t_str = sys.stdin.readline().rstrip()
if t_str != '':
	t = int(t_str)
	while t > 0:
		(a, b, c, m) = [int(num) for num in sys.stdin.readline().rstrip().split(' ')]
		if a - 1 > b + c + m or b - 1 > a + c + m or c - 1 > b + a + m:
			print('NO')
		elif a + b + c - 3 >= m:
			print('YES')
		else:
			print('NO')
		t -= 1
