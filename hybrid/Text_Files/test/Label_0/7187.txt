#include<iostream>
#include<bits/stdc++.h>
using namespace std;

int main(){
  int n;
  cin>>n;
  int sa=0,sb=0,sc=0;
  for(int day=1;day<=n;day++){
      int a,b,c,ta,tb,tc;
      cin>>a>>b>>c;
      ta = max(sb+a,sc+a);
      tb = max(sa+b,sc+b);
      tc = max(sa+c,sb+c);
      sa=ta; sb=tb; sc=tc;
  }
  cout<<max({sa,sb,sc})<<endl;
}
