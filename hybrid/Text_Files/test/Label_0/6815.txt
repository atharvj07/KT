#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:102400000,102400000")
using namespace std;
const int maxn = 1e5 + 3000;
int f[maxn], b[maxn], a[maxn], f2[maxn];
int n, m;
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    scanf("%d", &f[i]);
    if (f2[f[i]])
      f2[f[i]] = -1;
    else
      f2[f[i]] = i;
  }
  for (int i = 1; i <= m; i++) {
    scanf("%d", &b[i]);
  }
  for (int i = 1; i <= m; i++) {
    if (!f2[b[i]]) {
      puts("Impossible");
      return 0;
    }
  }
  for (int i = 1; i <= m; i++) {
    if (f2[b[i]] == -1) {
      puts("Ambiguity");
      return 0;
    }
  }
  puts("Possible");
  for (int i = 1; i <= m; i++) {
    printf("%d ", f2[b[i]]);
  }
  return 0;
}
