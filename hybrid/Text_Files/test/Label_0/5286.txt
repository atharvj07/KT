char s[1111];
int dp[1111][1111][2][2],n;

int dfs(int i,int j,int x,int y)
{
    if(dp[i][j][x][y]!=intINF) return dp[i][j][x][y];
    if(j&&x==0) dp[i][j][x][y]=min(dp[i][j][x][y],-dfs(i,j,1,y));
    if(i) dp[i][j][x][y]=min(dp[i][j][x][y],1-dfs(i-1,j+1,0,y));
    if(j) dp[i][j][x][y]=min(dp[i][j][x][y],1-dfs(i,j-1,0,y));
    if(!y) dp[i][j][x][y]=min(dp[i][j][x][y],1-dfs(i,j,0,1));
    return dp[i][j][x][y];
}
void predo()
{
    int n=1000;
    rep(i,0,n)
     rep(j,0,n)
      rep(k,0,1) dp[i][j][k][0]=dp[i][j][k][1]=intINF;
    dp[0][0][0][1]=dp[0][0][1][1]=0;

    rep(i,0,n)
     rep(j,0,n)
      rep(k,0,1)
      {
          dfs(i,j,k,0);
          dfs(i,j,k,1);
      }
}

void init()
{
    n=read();
    scanf("%s",s);
}

void solve()
{
    int sum=0;
    int zero=0;
    rep(i,0,n-1-i-1)
    {
        if(s[i]!=s[n-1-i]) sum++;
        if(s[i]=='0'&&s[n-1-i]=='0') zero++;
    }
    int flag=(!((n&1)&&s[n/2]=='0'));
    int res=dp[zero][sum][0][flag];
    if(res==0) printf("DRAW\n");
     else if(res<0) printf("ALICE\n");
      else printf("BOB\n");
}

int main()
{
   //freopen("1.in","r",stdin);
   //freopen("1.out","w",stdout);
   predo();
   int cas=read();
   //int cas=1;
   while(cas--)
   {
       //printf("Case #%d: ",v);
       init();
       solve();
   }
   return 0;
}
