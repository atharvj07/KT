#include <bits/stdc++.h>
using namespace std;
int a[100000];
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a, a + n);
  int cnt = 0;
  if (a[0] != 0) cnt = 1;
  for (int i = 1; i < n; i++) {
    if (a[i] != 0) {
      if (a[i] != a[i - 1]) cnt++;
    }
  }
  cout << cnt << endl;
  return 0;
}
