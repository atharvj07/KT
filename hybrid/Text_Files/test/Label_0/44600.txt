for t in range(int(input())):
	n = int(input())
	ar = list(map(int, input().split()))
	mx = max(ar)
	d = []
	for c in range(2):
		if set(ar[:mx]) == set(range(1, mx + 1)) and set(ar[mx:]) == set(range(1, n - mx + 1)):
			d.append(mx)
		mx = n - mx
	p = set(d)
	print(len(p))
	for x in p:
		print(x, n - x)
