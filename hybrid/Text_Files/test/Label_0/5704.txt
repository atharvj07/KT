#include <bits/stdc++.h>
using namespace std;
const int nmax = 4000100;
const int inf = 1e9 + 7;
int n, m;
int u[2010][2010], d[2010][2010];
pair<int, int> q[nmax];
char s[2010][2010];
int l = 0, r = 0;
map<pair<int, int>, int> M;
pair<int, int> fwejiofw(int x) {
  return make_pair(x / m + (x % m > 0), x % m + m * (x % m == 0));
}
int get(int x, int y) { return ((x - 1) * m + y); }
bool ch(int x, int y) { return !(x < 1 || x > n || y < 1 || y > m); }
void add(int x, int y) {
  if (u[x][y] == 1) return;
  q[r++] = make_pair(x, y);
}
void rm(int x, int y) {
  d[x][y]--;
  for (int i = -1; i <= 1; i++)
    for (int j = -1; j <= 1; j++) {
      if (i * j != 0) continue;
      if (i == 0 && j == 0) continue;
      int nx = x + i, ny = y + j;
      if (!ch(nx, ny)) continue;
      if (s[nx][ny] == '*') continue;
      d[nx][ny]--;
      if (d[nx][ny] == 1) add(nx, ny);
    }
}
int restore(int x, int y, int x2, int y2) {
  if (x == x2 && y > y2) swap(y, y2);
  if (y == y2 && x > x2) swap(x, x2);
  if (x == x2)
    s[x][y] = '<', s[x2][y2] = '>';
  else
    s[x][y] = '^', s[x2][y2] = 'v';
}
int main() {
  scanf("%d %d\n", &n, &m);
  for (int i = 1; i <= n; i++) scanf("%s", s[i] + 1);
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) {
      if (s[i][j] == '*') continue;
      if (ch(i - 1, j) && s[i - 1][j] == '.') {
        d[i - 1][j]++;
        d[i][j]++;
      }
      if (ch(i, j - 1) && s[i][j - 1] == '.') {
        d[i][j - 1]++;
        d[i][j]++;
      }
    }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++)
      if (d[i][j] == 1) add(i, j);
  while (l < r) {
    int x = q[l].first, y = q[l].second;
    l++;
    if (u[x][y]) continue;
    u[x][y] = 1;
    for (int i = -1; i <= 1; i++)
      for (int j = -1; j <= 1; j++) {
        if (i * j != 0) continue;
        if (i == 0 && j == 0) continue;
        int nx = x + i, ny = y + j;
        if (!ch(nx, ny)) continue;
        if (s[nx][ny] == '*') continue;
        if (u[nx][ny] == 1) continue;
        restore(x, y, nx, ny);
        rm(x, y);
        rm(nx, ny);
        d[nx][ny]++;
        d[x][y]++;
        u[nx][ny] = 1;
        i = j = 1000000;
      }
  }
  for (int i = 1; i <= n; i++)
    if (count(s[i] + 1, s[i] + 1 + m, '.') > 0) return printf("Not unique") * 0;
  for (int i = 1; i <= n; i++) printf("%s\n", s[i] + 1);
}
