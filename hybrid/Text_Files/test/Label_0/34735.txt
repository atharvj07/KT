import java.io.*;
import java.util.*;

public class wordcut {
  public static void main(String[] args) throws Exception {
    BufferedReader r = new BufferedReader(new InputStreamReader(System.in));
    BufferedWriter w = new BufferedWriter(new OutputStreamWriter(System.out));
    String start = r.readLine();
    String end = r.readLine();
    int n = Integer.parseInt(r.readLine());
    long[] dp1 = new long[n + 1];
    long[] dp2 = new long[n + 1];
    String temp = start + start;
    int good = 0;
    int bad = 0;
    for (int i = 0; i < start.length(); i++) {
      if (temp.substring(i, i + start.length()).equals(end)) {
        good++;
      } else {
        bad++;
      }
    }
    if (start.equals(end)) {
      dp1[0] = 1;
      dp2[0] = 0;
    } else {
      dp1[0] = 0;
      dp2[0] = 1;
    }
    for (int i = 1; i < n + 1; i++) {
        dp1[i] = (good - 1) * dp1[i - 1] + good * dp2[i - 1];
        dp1[i] %= MOD;
        dp2[i] = bad * dp1[i - 1] + (bad - 1) * dp2[i - 1];
        dp2[i] %= MOD;
    }
    w.write("" + dp1[n] + "\n");
    w.close();
    System.exit(0);
  }
  static int MOD = 1000000007;
}
