#include <bits/stdc++.h>
using namespace std;
long long cnt1[300001], cnt2[300001];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  long long n;
  cin >> n;
  string s[n];
  for (long long i = 0; i < n; i++) {
    cin >> s[i];
  }
  stack<char> st[n];
  for (long long j = 0; j < n; j++) {
    long long k = s[j].length();
    for (long long i = 0; i < k; i++) {
      if (s[j][i] == '(') {
        st[j].push('(');
      } else {
        if (!st[j].empty() && st[j].top() == '(') {
          st[j].pop();
        } else
          st[j].push(')');
      }
    }
  }
  long long cnt3 = 0;
  for (long long i = 0; i < n; i++) {
    if (st[i].empty()) {
      cnt3++;
    } else if (st[i].top() == ')') {
      long long ans = 0, sz = 0;
      while (!st[i].empty()) {
        if (st[i].top() == '(') {
          ans = 1;
        }
        st[i].pop();
        sz++;
      }
      if (ans == 0) {
        cnt1[sz]++;
      }
    } else if (st[i].top() == '(') {
      long long ans = 0, sz = 0;
      while (!st[i].empty()) {
        if (st[i].top() == ')') {
          ans = 1;
        }
        st[i].pop();
        sz++;
      }
      if (ans == 0) {
        cnt2[sz]++;
      }
    }
  }
  long long sum = 0;
  for (int i = 0; i <= 300000; i++) {
    sum += cnt1[i] * cnt2[i];
  }
  cout << cnt3 * cnt3 + sum;
}
