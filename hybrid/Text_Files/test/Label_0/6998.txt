#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long int n, k, mod = 0;
    cin >> n >> k;
    long long int cnt = 0;
    while (n > 0) {
      if (n % k == 0) {
        n /= k;
        cnt++;
      } else {
        mod = n % k;
        n -= mod;
        cnt += mod;
      }
    }
    cout << cnt << endl;
  }
  return 0;
}
