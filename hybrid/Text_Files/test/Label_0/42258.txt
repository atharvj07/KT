from math import inf
for _ in range(int(input())):
	n = int(input())
	a = [int(X) for X in input().split()]
	x = [0 for i in range(n + 1)]
	vis = [0 for i in range(n + 1)]
	for i in range(n):
		x[a[i]] = max(x[a[i]], i + 1 - vis[a[i]])
		vis[a[i]] = i + 1
	for i in range(n):
		x[a[i]] = max(x[a[i]], n + 1 - vis[a[i]])
	an = [inf for i in range(n + 1)]
	for i in range(1, n + 1):
		an[x[i]] = min(i, an[x[i]])
	p = -1
	for i in range(1, n + 1):
		if an[i] != inf and p == -1:
			p = 10 ** 6
		else:
			an[i] = min(p, an[i - 1], an[i])
	print(*an[1:])
