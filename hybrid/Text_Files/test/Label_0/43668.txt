t = int(input())
for i in range(t):
	n = int(input())
	c = list(map(int, input().strip().split()))
	a = list(map(int, input().strip().split()))
	b = list(map(int, input().strip().split()))
	len = [0 for i in range(n)]
	for i in range(1, n):
		d = abs(b[i] - a[i])
		len[i] = c[i] + 1 + int(d != 0) * max(d, len[i - 1] - d)
	print(max(len))
