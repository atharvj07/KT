#include <bits/stdc++.h>
using namespace std;
int cnt[1005], head[1005], par[1005];
struct node {
  int next, to, w;
} edge[1005];
int main() {
  int n;
  memset(head, 0, sizeof(head));
  memset(par, 0, sizeof(par));
  while (~scanf("%d", &n)) {
    int num = 0;
    for (int i = 2; i <= n; i++) {
      int x;
      scanf("%d", &x);
      edge[++num].next = head[x];
      head[x] = num;
      par[i] = x;
    }
    for (int i = 2; i <= n; i++) {
      if (!head[i]) ++cnt[par[i]];
    }
    int flag = 0;
    for (int i = 1; i <= n; i++) {
      if (head[i] && cnt[i] < 3) {
        flag = 1;
        break;
      }
    }
    if (flag == 1) {
      printf("No\n");
    } else {
      printf("Yes\n");
    }
  }
}
