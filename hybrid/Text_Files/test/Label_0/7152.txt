#include <bits/stdc++.h>
using namespace std;
const int N = 1005, M = 50, mod = 1e9 + 7;
int mul(int a, int b) { return ((long long)a * b) % mod; }
int add(int a, int b) { return (a + b) % mod; }
int dp[N][N][M], dp2[N][N], combi[N][N], per[N];
void init() {
  per[0] = 1;
  for (int i = 1; i < N; ++i) per[i] = mul(per[i - 1], i);
  for (int i = 0; i < N; ++i) {
    combi[i][i] = combi[i][0] = 1;
    for (int j = 1; j < i; ++j)
      combi[i][j] = add(combi[i - 1][j], combi[i - 1][j - 1]);
  }
  dp[0][0][0] = 1;
  for (int sum = 0; sum < N - 1; ++sum) {
    for (int maxi = 0; maxi < N - 1; ++maxi)
      for (int k = 0; k < M - 1; ++k) {
        if (sum + maxi < N - 1)
          dp[sum + maxi][maxi + 1][k + 1] =
              add(dp[sum + maxi][maxi + 1][k + 1], dp[sum][maxi][k]);
        dp[sum][maxi + 1][k] = add(dp[sum][maxi + 1][k], dp[sum][maxi][k]);
      }
  }
  for (int i = 1; i < N; ++i) {
    dp2[i][i] = 1;
    for (int j = i + 1; j < N; ++j)
      dp2[j][i] = add(combi[j - 1][i - 1], dp2[j - 1][i]);
  }
}
int main() {
  init();
  int q, n, k;
  cin >> q;
  while (q--) {
    scanf("%d%d", &n, &k);
    int ret = 0;
    if (k >= M) {
      printf("0\n");
      continue;
    }
    for (int i = k; i <= n; ++i) {
      ret = add(ret, mul(per[k], mul(dp[n - i][N - 1][k], dp2[i][k])));
    }
    printf("%d\n", ret);
  }
  return 0;
}
