 * Built using CHelper plug-in
 * Actual solution is at the top

public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskK solver = new TaskK();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskK {
        final int maxN = 1 << 20;

        public void solve(int testNumber, InputReader in, PrintWriter out) {
            int q = in.nextInt();
            segmentTree tree = new segmentTree(maxN);
            //for (int i = 0; i < maxN; i++) {
            //    tree.update(i, -1);
            //}
            int[] knight = new int[q];
            int[] duration = new int[q + 1];
            for (int query = 0; query < q; query++) {
                char type = in.nextChar();
                if (type == '+') {
                    int t = in.nextInt();
                    int d = in.nextInt();
                    tree.update(t, d);
                    knight[query] = t;
                    duration[query] = d;
                } else if (type == '-') {
                    int i = in.nextInt() - 1;
                    tree.update(knight[i], 0);
                } else {
                    assert (type == '?');
                    int t = in.nextInt();
                    long ans = tree.query(0, t + 1);
                    out.println(Math.max(0, ans - t));
                }
            }
        }

        class segmentTree {
            int log;
            int n;
            long[] max;
            long[] sum;

            segmentTree(int _n) {
                this.n = 1;
                this.log = 1;
                while (this.n < _n) {
                    this.n *= 2;
                    log++;
                }
                max = new long[n * 2];
                sum = new long[n * 2];
            }

            void update(int node, long val) {
                node += this.n;
                sum[node] = val;
                max[node] = val == 0 ? 0 : sum[node] + (node - this.n);
                while (node > 0) {
                    node >>= 1;
                    sum[node] = sum[node * 2] + sum[node * 2 + 1];
                    max[node] = Math.max(max[node * 2] + sum[node * 2 + 1], max[node * 2 + 1]);
                }
            }

            long query(int l, int r) {
                l += this.n;
                r += this.n;
                long ans = max[r - 1];
                long suffixSum = 0;
                while (l < r) {
                    ans = Math.max(ans, suffixSum + max[r - 1]);
                    if ((r & 1) == 1) {
                        suffixSum += sum[--r];
                    }
                    if ((l & 1) == 1) l++;
                    l >>= 1;
                    r >>= 1;
                }
                return ans;
            }

        }

    }

    static class InputReader {
        private byte[] buf = new byte[1024];
        private int curChar;
        private int numChars;
        private InputStream stream;

        public InputReader(InputStream stream) {
            this.stream = stream;
        }

        private boolean isWhitespace(int c) {
            return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
        }

        private int read() {
            if (numChars == -1)
                throw new InputMismatchException();
            if (curChar >= numChars) {
                curChar = 0;
                try {
                    numChars = stream.read(buf);
                } catch (IOException e) {
                    throw new InputMismatchException();
                }
                if (numChars <= 0)
                    return -1;
            }
            return buf[curChar++];
        }

        public int nextInt() {
            int c = read();
            while (isWhitespace(c))
                c = read();
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = read();
            }
            int res = 0;
            do {
                if (c < '0' || c > '9')
                    throw new InputMismatchException();
                res *= 10;
                res += c - '0';
                c = read();
            } while (!isWhitespace(c));
            return res * sgn;
        }

        public char nextChar() {
            int c = read();
            while (isWhitespace(c))
                c = read();
            char c1 = (char) c;
            while (!isWhitespace(c))
                c = read();
            return c1;
        }

    }
}
