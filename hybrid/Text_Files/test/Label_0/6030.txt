#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, x[5001], m, y[5001], A, B, k, z[5001];
  cin >> n;
  for (int i = 0; i < n; i++) cin >> x[i];
  cin >> m;
  for (int i = 0; i < m; i++) cin >> y[i];
  cin >> k;
  for (int i = 0; i < k; i++) cin >> z[i];
  cin >> A >> B;
  int p2 = *min_element(z, z + k);
  int r1 = *max_element(x, x + n);
  double ans = 0.00;
  for (int i = 0; i < m; i++) {
    double r2 = r1 * sqrt(((double)B * y[i]) / (A * p2 + B * y[i]));
    ans = max(ans, r2);
  }
  cout << setprecision(12) << ans << endl;
  return 0;
}
