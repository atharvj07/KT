(n, m) = map(int, input().split())
tests = list(map(int, input().split()))
flag = 0
test_counters = [0] * (n + 1)
for test in tests:
	test_counters[test] += 1
	if test_counters[test] == 1:
		flag += 1
	if flag == n:
		print(1, end='')
		for j in range(1, n + 1):
			test_counters[j] -= 1
			if test_counters[j] == 0:
				flag -= 1
	else:
		print(0, end='')
