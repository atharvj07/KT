import java.util.*;

public class Main{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        int t=sc.nextInt();
        while(t-->0){
            int n=sc.nextInt(),m=sc.nextInt(),x=sc.nextInt(),y=sc.nextInt();
            int cnt2=0,cnt1=0;
            String[] s=new String[n];
            boolean[][] vis=new boolean[n][m];
            for(int i=0;i<n;i++){
                s[i]=sc.next();
                for(int j=0;j<m;j++){
                    if(s[i].charAt(j)=='.'){
                        cnt1++;
                        if(j>0 && s[i].charAt(j-1)=='.'){
                            if(vis[i][j-1]==false){
                                cnt2++;
                                vis[i][j-1]=vis[i][j]=true;
                            }
                        }
                    }
                }
            }
            int ans=0;
            if(2*x<=y){
                ans=cnt1*x;
            }
            else{
                ans=cnt2*y;
                ans+=(cnt1-2*cnt2)*x;
            }
            System.out.println(ans);
        }
    }
}
