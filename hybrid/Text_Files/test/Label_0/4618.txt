#include <bits/stdc++.h>
using namespace std;
const int N = 100005;
int ask(char ch, int num) {
  static int cnt = 0;
  if (++cnt > 60) return -1;
  cout << ch << ' ' << num << endl;
  cout.flush();
  int tmp;
  cin >> tmp;
  return tmp;
}
int gcd(int a, int b) { return b ? gcd(b, a % b) : a; }
int main() {
  ios::sync_with_stdio(false);
  int n;
  cin >> n;
  int l = 0, r = 1000000000, last = -1;
  while (l < r) {
    int mid = (l + r) >> 1;
    if (ask('>', mid))
      last = l = mid + 1;
    else
      r = mid;
  }
  vector<int> a;
  default_random_engine e;
  e.seed(time(0));
  uniform_int_distribution<int> u(1, n);
  while (a.empty() || a.back() != -1) a.push_back(ask('?', u(e)));
  a.pop_back();
  assert(a.size() > 1);
  sort(a.begin(), a.end());
  int d = a[1] - a[0];
  for (int i = 1; i < a.size() - 1; i++) d = gcd(d, a[i + 1] - a[i]);
  cout << "! " << last - (n - 1) * d << ' ' << d;
  return 0;
}
