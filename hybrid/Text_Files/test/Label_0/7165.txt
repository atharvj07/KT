#include <bits/stdc++.h>
using namespace std;
map<string, int> mp;
int cost[605][605];
int u[605], v[605];
bool cmp(int a, int b) { return a > b; }
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int n, a, b, k, f;
  cin >> n >> a >> b >> k >> f;
  mp.clear();
  memset(cost, 0, sizeof(cost));
  int ans = 0;
  string s1, s2;
  for (int i = 1; i <= n; i++) {
    cin >> s1 >> s2;
    if (mp[s1] == 0) {
      ans++;
      mp[s1] = ans;
    }
    if (mp[s2] == 0) {
      ans++;
      mp[s2] = ans;
    }
    u[i] = mp[s1];
    v[i] = mp[s2];
  }
  for (int i = 1; i <= n; i++) {
    int d = min(u[i], v[i]);
    int t = max(u[i], v[i]);
    if (u[i] == v[i - 1]) {
      cost[d][t] += b;
    } else {
      cost[d][t] += a;
    }
  }
  vector<int> G;
  G.clear();
  int sum = 0;
  for (int i = 1; i <= ans; i++) {
    for (int j = i + 1; j <= ans; j++) {
      if (cost[i][j] != 0) {
        sum += cost[i][j];
        G.push_back(cost[i][j]);
      }
    }
  }
  int d = G.size();
  sort(G.begin(), G.end(), cmp);
  k = min(d, k);
  for (int i = 0; i < k; i++) {
    if (G[i] > f) {
      sum = sum - (G[i] - f);
    } else
      break;
  }
  cout << sum << endl;
  return 0;
}
