#include <bits/stdc++.h>
using namespace std;
vector<int> letters;
int main() {
  int i, sum = 0, j;
  string str, ans;
  cin >> str;
  letters.resize(30);
  fill_n(letters.begin(), 30, 0);
  for (i = 0; i < (int)str.length(); i++) {
    letters[str[i] - 'a']++;
  };
  for (i = 0; i < 28; i++) {
    sum += letters[i] % 2;
  };
  sum /= 2;
  j = 28;
  for (i = 0; i < 28; i++) {
    if (letters[i] % 2) {
      for (; j >= 0; j--) {
        if (letters[j] % 2) {
          if (sum) {
            letters[i]++;
            letters[j]--;
            sum--;
            break;
          };
        };
      };
    };
  };
  for (i = 0; i < 28; i++) {
    for (j = 0; j < letters[i] / 2; j++) {
      ans.push_back(i + 'a');
    };
    letters[i] -= 2 * j;
  };
  for (i = 0; i < 28; i++) {
    for (j = 0; j < letters[i]; j++) ans.push_back(i + 'a');
  };
  for (i = 0; i < (int)ans.length(); i++) {
    printf("%c", ans[i]);
  };
  for (i = (int)ans.length() - 1 - ((int)str.length() % 2); i >= 0; i--)
    printf("%c", ans[i]);
  printf("\n");
  return 0;
}
