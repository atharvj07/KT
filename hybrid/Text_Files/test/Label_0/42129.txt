for i in range(int(input())):
	b = int(input())
	a = list(map(int, input().split(' ')))
	if sum(a) == 0:
		print('NO')
	elif sum(a) < 0:
		print('YES')
		e = sorted(a)
		for k in e:
			print(k, end=' ')
		print()
	else:
		print('YES')
		e = reversed(sorted(a))
		for q in e:
			print(q, end=' ')
		print()
