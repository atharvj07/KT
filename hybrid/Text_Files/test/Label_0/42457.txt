wires = input()
w = []
w = input().split(' ')
birds_Wires = w[:int(wires)]
shots = input()
opration = []
for i in range(0, int(shots)):
	x_y = input().split(' ')
	opration.append(tuple(x_y))
for shot in opration:
	x = int(shot[0])
	y = int(shot[1])
	birds = int(birds_Wires[x - 1])
	birds_Wires[x - 1] = 0
	left = birds - (birds - y + 1)
	right = birds - y
	if x - 1 + 1 == len(birds_Wires):
		right = 0
	if x - 1 - 1 == -1:
		left = 0
	if x - 1 - 1 != -1:
		birds_Wires[x - 1 - 1] = int(birds_Wires[x - 1 - 1]) + left
	if x - 1 + 1 != len(birds_Wires):
		birds_Wires[x - 1 + 1] = int(birds_Wires[x - 1 + 1]) + right
for i in birds_Wires:
	print(i)
