#include <bits/stdc++.h>
using namespace std;
int main() {
  int N, M, R;
  cin >> N >> M >> R;
  vector<int> s(N), b(M);
  for (int i = 0; i < N; ++i) cin >> s[i];
  for (int i = 0; i < M; ++i) cin >> b[i];
  sort(s.begin(), s.end());
  sort(b.rbegin(), b.rend());
  int buy = R / s[0];
  int ans = R + b[0] * buy - s[0] * buy;
  if (b[0] < s[0]) {
    cout << R << '\n';
    goto done;
  }
  R -= s[0] * buy;
  if (R && s.size() > 1 && s[1] < b[0])
    ans += b[0] * (R / s[1]) - s[1] * (R / s[1]);
  cout << ans << '\n';
done:
  return 0;
}
