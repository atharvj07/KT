#include <bits/stdc++.h>
using namespace std;
inline int re_ad() {
  int x = 0, f = 1;
  char ch = getchar();
  while (ch > '9' || ch < '0') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') x = x * 10 + (ch ^ 48), ch = getchar();
  return x * f;
}
int n, k;
int a[5010], sum;
int p[5010], q[5010];
bool fla[5010];
int pla[5010];
int main() {
  register int i, x, y, j, z, ii;
  bool fl = false;
  srand(time(0));
  k = re_ad();
  n = 1 << k;
  for (i = 0; i < n; ++i) {
    a[i] = re_ad();
    sum ^= a[i];
  }
  if (sum) {
    puts("Fou");
    return 0;
  }
  memset(p, -1, sizeof(p));
  memset(q, -1, sizeof(q));
  memset(pla, -1, sizeof(p));
  i = 0;
  while (i < n) {
    x = rand() & (n - 1);
    while (p[x] >= 0) x = rand() & (n - 1);
    fl = false;
    for (j = 0; j < n; ++j) {
      if (!fla[j] && pla[j ^ a[x]] == -1) {
        p[x] = j;
        q[x] = j ^ a[x];
        fl = true;
        fla[j] = true;
        pla[j ^ a[x]] = x;
        break;
      }
    }
    if (fl) {
      ++i;
      continue;
    }
    y = rand() & (n - 1);
    while (fla[y]) y = rand() & (n - 1);
    z = pla[y ^ a[x]];
    p[x] = y;
    q[x] = y ^ a[x];
    fla[y] = true;
    pla[y ^ a[x]] = x;
    fla[p[z]] = false;
    q[z] = p[z] = -1;
  }
  puts("Shi");
  for (i = 0; i < n; ++i) printf("%d ", p[i]);
  puts("");
  for (i = 0; i < n; ++i) printf("%d ", q[i]);
  return 0;
}
