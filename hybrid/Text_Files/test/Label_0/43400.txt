m = int(input())
b = list(map(int, input().strip().split()))
s = [0 for i in range(500)]
for i in a:
	s[i] += 1
for j in b:
	s[j] += 1
done = False
for i in a:
	for j in b:
		if s[i + j] == 0:
			print(i, j)
			done = True
		if done == True:
			break
	if done == True:
		break
