from sys import stdin
for _ in range(int(input())):
	n = int(input())
	s = list(stdin.readline().rstrip())
	m = s
	c = 1
	for i in range(2, n + 1):
		if (n - i + 1) % 2 == 0:
			z = s[i - 1:] + s[:i - 1]
		else:
			z = s[i - 1:] + s[:i - 1][::-1]
		if z < m:
			m = z
			c = i
	print(''.join(m))
	print(c)
