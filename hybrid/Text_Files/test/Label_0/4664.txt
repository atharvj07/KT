#define mod 1000000007 //1e9+7(prime number)
#define INF 1000000000 //1e9
#define LLINF 2000000000000000000LL //2e18
#define SIZE 100010

int sign(int x){
  return (x>0)-(x<0);
}



int main(){
  int n;
  int a[SIZE];
  
  scanf("%d",&n);  
  
  for(int i=0;i<n;i++){
    scanf("%d",a+i);
  }

  int t = 0;
  int ans = 0;

  while(t+1 < n && a[t] == a[t+1]) t++;
  
  bool isup = (a[t+1] - a[t]) > 0;
  int counter = 0;
  
  for(;t+1<n;t++){
    while(t+1 < n && a[t] == a[t+1]) t++;

    if(t+1 >= n) break;
    
    if(a[t+1] - a[t] > 0 && isup){
      counter++;
    }else if(a[t+1] - a[t] < 0 && !isup){
      counter++;
    }else{
      ans = max(ans, counter);
      counter = 1;
      while(t+1 < n && a[t] == a[t+1]) t++;
      if(t+1 >= n) break;
      isup = (a[t+1] - a[t]) > 0;
    }
  }

  ans = max(ans, counter);

  printf("%d\n",ans + 1);
  
  return 0;
}
