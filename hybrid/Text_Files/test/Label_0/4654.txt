#include <iostream>
#include <cstdio>
#include <complex>

using namespace std;
typedef long long ll;

typedef double R;
typedef complex<R> P;

double cross(const P &a, const P &b) { return imag(conj(a)*b); }
double dot(const P &a, const P &b) { return real(conj(a)*b); }

int n;
P p[100];
R solve() {
    R res = 0;
    for (int i = 2; i < n; i++) {
        res += cross(p[i-1]-p[0], p[i]-p[0]) / 2.0;
    }
    return abs(res);
}


int main() {
    int t = 1;
    while (true) {
        scanf("%d", &n);
        if (!n) break;
        for (int i = 0; i < n; i++) {
            R x, y;
            scanf("%lf %lf", &x, &y);
            p[i] = P(x, y);
        }
        printf("%d %.1f\n", t, solve());
        t++;
    }
    return 0;
}
