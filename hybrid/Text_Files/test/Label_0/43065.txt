from sys import stdin
input = lambda : stdin.readline().strip()
from math import ceil, sqrt, factorial, gcd
from collections import deque
s = input()
m = int(input())
z = []
for i in range(10):
	if s[i] == '1':
		z.append(i + 1)
stack = [[1, 0, 0, []]]
while stack:
	(x, y, p, ans) = stack.pop()
	if x > m:
		print('YES')
		print(*ans)
		exit()
	for i in z:
		if i > y and i != p:
			stack.append([x + 1, i - y, i, ans + [i]])
print('NO')
