#include<iostream>
#include<queue>
#include<algorithm>
#include<stack>
#include<map>
#include<set>
using namespace std;
typedef long long ll;
struct E{
  ll p,t;
};
bool operator<(E a,E b){
  return a.t>b.t;
}
ll L,n;
set<ll> node;
map<ll,vector<E> > edge;

void input(){
  cin>>L>>n;
  node.insert(0);
  node.insert(L);
  for(ll i=0;i<n;i++){
    ll p,d,t;
    cin>>p>>d>>t;
    edge[p].push_back((E){d,t});
    node.insert(p);
    node.insert(p+d);
  }
  for(auto i:node){
    //cout<<i<<endl;
  }
}

ll solve(){
  priority_queue<E> q;
  q.push((E){0,0});
  map<ll,int> dp;
  for(auto i:node){
    dp[i]=-1;
  }
  while(!q.empty()){
    E tmp = q.top();q.pop();
    if(tmp.p == L)return tmp.t;
    if(dp[tmp.p]!=-1&& tmp.t>=dp[tmp.p])continue;
    dp[tmp.p]=tmp.t;
    //cout<<tmp.p<<" "<<tmp.t<<endl;
    const vector<E> e = edge[tmp.p];
    for(int i=0;i<e.size();i++){
      q.push((E){tmp.p+e[i].p,tmp.t+e[i].t});
    }
    for(auto i:node){
      if(i>tmp.p){
        q.push((E){i,tmp.t + (i-tmp.p)});
      }
    }
  }
  return 0;
}


int main(){
  input();
  cout<<solve()<<endl;
}
