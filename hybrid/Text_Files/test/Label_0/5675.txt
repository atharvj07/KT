#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  long long n;
  cin >> n;
  cout << '?' << " " << 1 << " " << 2 << '\n' << flush;
  long long x1, x2, x3;
  cin >> x1;
  cout << '?' << " " << 2 << " " << 3 << '\n' << flush;
  cin >> x2;
  cout << '?' << " " << 1 << " " << 3 << '\n' << flush;
  cin >> x3;
  vector<long long> sol(n);
  sol[0] = (x1 + x2 + x3) / 2 - x2;
  sol[1] = (x1 + x2 + x3) / 2 - x3;
  sol[2] = (x1 + x2 + x3) / 2 - x1;
  for (int i = 0; i < n - 3; ++i) {
    cout << '?' << " " << i + 4 << " " << 1 << '\n' << flush;
    long long a;
    cin >> a;
    sol[i + 3] = a - sol[0];
  }
  cout << '!' << " ";
  for (int i = 0; i < n; ++i) {
    cout << sol[i] << " ";
  }
  cout << '\n' << flush;
  return 0;
}
