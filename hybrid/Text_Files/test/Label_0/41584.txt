import sys
import math
import bisect

def solve(A, m):
	n = len(A)
	total = sum(A)
	cnt = 0
	cur = 0
	ans = 0
	while cnt < total:
		cur = 0
		ans += 1
		for i in range(n):
			if cur + A[i] <= m:
				cur += A[i]
				cnt += A[i]
				A[i] = 0
			else:
				break
	return ans

def main():
	(n, m) = map(int, input().split())
	A = list(map(int, input().split()))
	ans = solve(A, m)
	print(ans)
main()
