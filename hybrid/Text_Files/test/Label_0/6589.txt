#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, a[12323], h;
  cin >> n >> m >> h;
  int s = 0;
  for (int i = 1; i <= m; i++) cin >> a[i], s += a[i];
  if (s < n) {
    puts("-1.0");
  } else {
    n--;
    s--;
    a[h]--;
    int f = s - n;
    double da = 1;
    for (int i = 0; i < a[h]; i++) da *= (f - i + 0.0) / (s - i);
    printf("%.11lf\n", 1 - da);
  }
  cin >> n;
}
