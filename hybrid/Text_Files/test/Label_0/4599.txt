int N;
int A[111];

void solve(){
    rep(i,N){
        char c;cin>>c;A[i]=c-'A';
    }

    int S[111]={};
    rep(i,N){
        S[A[i]]++;

        rep(j,26){
            bool ok=true;
            rep(k,26)if(k!=j&&S[k]+N-1-i>=S[j])ok=false;
            if(!ok)continue;
            cout<<(char)(j+'A')<<" "<<i+1<<endl;
            return;
        }
    }
    cout<<"TIE"<<endl;
}
