#include <bits/stdc++.h>
using namespace std;
int n, c[205], du[205], degree[205];
bool vis[205];
vector<int> a[205];
int work(int ci) {
  int tot = 0, sum = 0;
  memset(vis, 0, sizeof(vis));
  while (tot < n) {
    while (1) {
      int flag = 1;
      for (int i = 1; i <= n; i++) {
        if (!du[i] && c[i] == ci && !vis[i]) {
          vis[i] = 1;
          flag = 0;
          tot++;
          for (int j = 0; j < a[i].size(); j++) {
            int t = a[i][j];
            du[t]--;
          }
          sum++;
        }
      }
      if (flag) break;
    }
    ci = (ci % 3) + 1;
    sum++;
  }
  return sum;
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d", &c[i]);
  for (int i = 1; i <= n; i++) {
    int k;
    scanf("%d", &k);
    degree[i] = k;
    for (int j = 1; j <= k; j++) {
      int t;
      scanf("%d", &t);
      a[t].push_back(i);
    }
  }
  int ans = 0x7fffffff;
  for (int i = 1; i <= n; i++) du[i] = degree[i];
  ans = min(ans, work(1));
  for (int i = 1; i <= n; i++) du[i] = degree[i];
  ans = min(ans, work(2));
  for (int i = 1; i <= n; i++) du[i] = degree[i];
  ans = min(ans, work(3));
  printf("%d\n", ans - 1);
  return 0;
}
