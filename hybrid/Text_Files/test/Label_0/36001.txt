	static class SuffixAutomata{
		int trie[][];
		int suffixLink[];
		int next[][];
		int N=0;

		SuffixAutomata(){
			trie=new int[1000000][26];
			next=new int[1000000][26];
			suffixLink=new int[1000000];
		}

		int getNewNode(){
			return ++N;
		}

		int insert(String c){
			char s[]=c.toCharArray();
			int n=s.length;
			int node=0;
			for(int i=0;i<n;i++){
				if(trie[node][s[i]-'a']==0)
					trie[node][s[i]-'a']=getNewNode();
				node=trie[node][s[i]-'a'];
			}
			return node;
		}

		void createSuffixLinks(){
			next[0]=trie[0];
			ArrayDeque<Integer> queue = new ArrayDeque<>();
			for(int i=0;i<26;i++)
				if(trie[0][i]!=0)
					queue.add(trie[0][i]);

			while(!queue.isEmpty()){
				int node=queue.poll();
				//set the next values;
				for(int i=0;i<26;i++){
					int nextNode=trie[node][i];
					if(nextNode==0){
						next[node][i]=next[suffixLink[node]][i];
					}
					else{
						next[node][i]=nextNode;
						suffixLink[nextNode]=next[suffixLink[node]][i];
						queue.add(nextNode);
					}
				}
			}
		}
	}

	static class SegmentTree
	{
		int n;
		long tree[];
		public SegmentTree(int len)
		{
			n=len+31;
			tree=new long[n+1];
		}

		void incrementRange(int l,int r,long value)
		{
			add(l, value);
			add(r+1, -value);
		}
		void add(int i,long value)
		{
			for(;i<n;i+=i&-i)
				tree[i]+=value;
		}
		long query(int i)
		{
			long ans=0;
			for(;i>0;i-=i&-i)
				ans+=tree[i];
			return ans;
		}
	}

	static class Trie {
		int trie[][];
		int N = 0;

		Trie() {
			trie = new int[1000000][26];
		}

		int getNewNode(){
			return ++N;
		}

		int insert(int node, String c){
			char s[]=c.toCharArray();
			int n=s.length;
			for(int i=0;i<n;i++){
				if(trie[node][s[i]-'a']==0)
					trie[node][s[i]-'a']=getNewNode();
				node=trie[node][s[i]-'a'];
			}
			return node;
		}

		int insert(String c){
			return insert(0, c);
		}

	}
	static class Query{

		public Query(int queryNode) {
			this.queryNode = queryNode;
		}

		int queryNode;
		long ans;

	}

	static void addQuery(int queryIndex, int queryNode, int treeNode){
		q[queryIndex]=new Query(queryNode);
		if(!queries.containsKey(treeNode))
			queries.put(treeNode, new ArrayList<>());
		queries.get(treeNode).add(queryIndex);
	}
	static class Tree{
		ArrayList<Integer> g[];
		int time;
		int position[];
		int chainHead[];
		int subtreeSize[];
		int parent[];
		SegmentTree segmentTree;
		Tree(ArrayList<Integer> g[]){
			this.g=g;
			segmentTree=new SegmentTree(g.length);
			time=1;
			position=new int[g.length];
			chainHead=new int[g.length];
			subtreeSize=new int[g.length];
			parent=new int[g.length];
			calculateSubtreeSize(0);
			HLD(0,0);
		}
		int calculateSubtreeSize(int node){
			int ans=1;
			for(int x:g[node]){
				ans+=calculateSubtreeSize(x);
			}
			return subtreeSize[node]=ans;
		}
		void HLD(int node, int ch){
			position[node]=time++;
			chainHead[node]=ch;
			int specialChild=-1, maxSubtreeSize=0;
			for(int i:g[node]){
				parent[i]=node;
				if(subtreeSize[i] > maxSubtreeSize){
					maxSubtreeSize=subtreeSize[i];
					specialChild=i;
				}
			}
			if(specialChild!=-1)
				HLD(specialChild, ch);
			for(int i:g[node]){
				if(i==specialChild)
					continue;
				HLD(i,i);
			}
		}
		void addOnRootToNodePath(int node, long value){
			while(true){
				segmentTree.incrementRange(position[chainHead[node]], position[node], value);
				if(chainHead[node]==0)
					break;
				node=parent[chainHead[node]];
			}
		}

		int getNodeValue(int node){
			return (int)segmentTree.query(position[node]);
		}
	}
	static HashMap<Integer, Integer> map;

	static void sol(int node, int  matchNode){
		tree.addOnRootToNodePath(matchNode,1);

		//answer queries
		if(queries.containsKey(node)){
			for(int qi:queries.get(node)){
				q[qi].ans=tree.getNodeValue(q[qi].queryNode);
			}
		}

		for(int i=0;i<26;i++){
			if(trie.trie[node][i]!=0){
				sol(trie.trie[node][i], sa.next[matchNode][i]);
			}
		}

		tree.addOnRootToNodePath(matchNode,-1);

	}

		Scanner in = new Scanner(System.in);

			}
		}
		int m=in.nextInt();
		queries=new HashMap<>();
		q=new Query[m];
		for(int i=0;i<m;i++){
			int treeNode = map.get(in.nextInt());
			int queryNode = sa.insert(in.next());
			addQuery(i, queryNode, treeNode);
		}
		sa.createSuffixLinks();
		g=new ArrayList[sa.N+1];
		for(int i=0;i<=sa.N;i++){
			g[i]=new ArrayList<>();
		}
		for(int i=1;i<=sa.N;i++)
			g[sa.suffixLink[i]].add(i);
		tree=new Tree(g);
		ArrayList<Integer> st=new ArrayList<>(), en=new ArrayList<>();
		sol(0,0);
		for(int i=0;i<m;i++)
			out.println(q[i].ans);
		out.close();
	}
