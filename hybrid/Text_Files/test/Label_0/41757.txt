def solve():
	(n, _, a, b) = [int(x) for x in input().strip().split(' ')]
	cost = [int(x) for x in input().strip().split(' ')]
	cost.sort()
	lb = 0
	ub = 0
	if a < b:
		lb = a
		ub = b - 1
	else:
		lb = n - a + 1
		ub = n - b
	p = lb
	q = 0
	res = 0
	while p < ub and q < len(cost):
		if p >= cost[q]:
			p += 1
			q += 1
			res += 1
		elif cost[q] >= ub:
			break
		else:
			p = cost[q] + 1
			res += 1
			q += 1
	return res
t = int(input().strip())
for _ in range(t):
	print(solve())
