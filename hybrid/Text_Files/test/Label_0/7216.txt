#include <bits/stdc++.h>
using namespace std;
#define dbg(x) cerr << #x << " = " << x << endl
#define rep(i, a, b) for(int i = (a); i <= (b); ++ i)
#define per(i, a, b) for(int i = (a); i >= (b); -- i)
#define MP make_pair
#define pb push_back
#define SIZ(x) ((int)(x).size())
#define fi first
#define se second
#define fill(a, b) memset(a, b, sizeof(a))
typedef long long LL;
typedef unsigned long long u64;
typedef unsigned int u32;
template <typename Tp> void read(Tp &x) {
	x = 0; int op = 1; char ch = getchar();
	while(!isdigit(ch)) { if(ch == '-') op = -1; ch = getchar(); }
	while(isdigit(ch)) { x = x*10+ch-'0'; ch = getchar(); } x *= op;
}
template <typename Tp> void CMax(Tp &x, Tp y) { if(y > x) x = y; }
template <typename Tp> void CMin(Tp &x, Tp y) { if(y < x) x = y; }

typedef vector<int> vi;
typedef pair<int, int> pii;
const int N = 2e5 + 10;
const int Inf = 0x3f3f3f3f;

int n, Q, s, d;
int a[N];

int par[N];
vi Set[N];
void init() {
	rep(i, 1, n) {
		par[i] = i;
		Set[i].pb(i);
	}
}
int find(int x) {
	if(par[x] == x) return x;
	return par[x] = find(par[x]);
}
void Merge(int x, int y) {
	int fx = find(x), fy = find(y);
	if(SIZ(Set[fx]) < SIZ(Set[fy])) swap(fx, fy);
	for(auto i : Set[fy]) Set[fx].pb(i);
	Set[fy].clear(); par[fy] = fx;
}

set<pii> st;
pii Best[N];
int Val(int i, int j) { return abs(d - abs(a[i] - a[j])); }
int Val(pii p) { return Val(p.fi, p.se); }

int head[N], nxt[N << 1], to[N << 1], val[N << 1], tot;
void add_edge(int x, int y, int z) {
	to[++ tot] = y; val[tot] = z;
	nxt[tot] = head[x]; head[x] = tot;
}
void Adde(int x, int y, int z) {
	add_edge(x, y, z);
	add_edge(y, x, z);
}

int Mn[N];
void dfs(int x, int fa, int mn) {
	Mn[x] = mn;
	for(int i = head[x]; i; i = nxt[i]) {
		int y = to[i], z = val[i]; if(y == fa) continue;
		dfs(y, x, max(mn, z));
	}
}

int main()
{
	read(n); read(Q); read(s); read(d);
	rep(i, 1, n) read(a[i]), st.insert(MP(a[i], i));
	init();
	int cnt = n;
	while(cnt > 1) {
		vector<int> ver; ver.clear();
		rep(i, 1, n) if(find(i) == i) {
			ver.pb(i);
			for(auto j : Set[i]) st.erase(MP(a[j], j));
			Best[i] = MP(0, 0);
			for(auto j : Set[i]) {
				set<pii>::iterator it = st.lower_bound(MP(a[j] - d, 0));
				if(it != st.end()) {
					if(!Best[i].fi || Val(j, (*it).se) < Val(Best[i])) Best[i] = MP(j, (*it).se);
				}
				if(it != st.begin()) {
					-- it;
					if(!Best[i].fi || Val(j, (*it).se) < Val(Best[i])) Best[i] = MP(j, (*it).se);
				}
				it = st.lower_bound(MP(a[j] + d, 0));
				if(it != st.end()) {
					if(!Best[i].fi || Val(j, (*it).se) < Val(Best[i])) Best[i] = MP(j, (*it).se);
				}
				if(it != st.begin()) {
					-- it;
					if(!Best[i].fi || Val(j, (*it).se) < Val(Best[i])) Best[i] = MP(j, (*it).se);
				}
			}
			for(auto j : Set[i]) st.insert(MP(a[j], j));
		}
		for(auto i : ver) if(find(Best[i].fi) != find(Best[i].se)) {
			cnt --; Merge(Best[i].fi, Best[i].se);
			Adde(Best[i].fi, Best[i].se, Val(Best[i]));
		}
	}
	dfs(s, 0, 0);
	int t, k;
	while(Q -- ) {
		read(t); read(k);
		if(Mn[t] <= k) puts("Yes");
		else puts("No");
	}
	return 0;
}
