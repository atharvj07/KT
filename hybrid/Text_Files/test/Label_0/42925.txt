from sys import stdin, stdout

def inp():
	return stdin.readline().strip()

def iinp():
	return int(inp())

def out(var, end='\n'):
	stdout.write(str(var) + '\n')

def mp():
	return map(int, inp().split())
n = iinp()
ml = []
for i in range(n):
	(x, y) = mp()
	ml.append((x, y))
ml.sort(key=lambda x: abs(x[0]) + abs(x[1]))
ansl = []
for (x, y) in ml:
	if x > 0:
		ansl.append('1 ' + str(x) + ' R')
	if x < 0:
		ansl.append('1 ' + str(-x) + ' L')
	if y > 0:
		ansl.append('1 ' + str(y) + ' U')
	if y < 0:
		ansl.append('1 ' + str(-y) + ' D')
	ansl.append('2')
	if x > 0:
		ansl.append('1 ' + str(x) + ' L')
	if x < 0:
		ansl.append('1 ' + str(-x) + ' R')
	if y > 0:
		ansl.append('1 ' + str(y) + ' D')
	if y < 0:
		ansl.append('1 ' + str(-y) + ' U')
	ansl.append('3')
out(len(ansl))
out('\n'.join(ansl))
