import java.util.*;
import java.io.*;
public class CaseOfATopSecret
{
	public static void main(String[] args) throws Exception
	{
		FastScanner in = new FastScanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		
		int n = in.nextInt();
		int m = in.nextInt();
		
		int[] xs = new int[n];
		for(int x = 0; x < n; x++)
		{
			xs[x] = in.nextInt();
		}
		
		Peg[] pegs = new Peg[n];
		for(int y = 0; y < xs.length; y++)
		{
			pegs[y] = new Peg(y, xs[y]);
		}
		
		Arrays.sort(pegs);
		
		int[] array = new int[n];
		for(int i = 0; i < pegs.length; i++)
		{
			array[i] = pegs[i].pos;
		}
		
		for(int z = 0; z < m; z++)
		{
			int a = in.nextInt() - 1;
			int l = in.nextInt();
			
			int index = Arrays.binarySearch(array, xs[a]);
			int dir = 1;
			boolean first = true;
			while(true)
			{
//				out.println(array[index] + " " + dir + " " + l);
				
				int next = Arrays.binarySearch(array, array[index] + dir * l);
				if(next < 0)
				{
					next = -(next + 1);
					if(dir == 1)
					{
						next--;
					}
				}
				
				if(first)
				{
					l -= Math.abs(array[index] - array[next]);
					index = next;
					dir *= -1;
				}
				else
				{
					if(index == next)
					{
						break;
					}
					
					int t = l / Math.abs(array[index] - array[next]);
					l %= Math.abs(array[index] - array[next]);
					
					if(t % 2 == 1)
					{
						index = next;
						dir *= -1;
					}
				}
				
				first = false;
			}
			
			out.println(pegs[index].index + 1);
		}
		
		out.close();
	}
	
	static class Peg implements Comparable<Peg>
	{
		int index;
		int pos;
		
		public Peg(int i, int p)
		{
			index = i;
			pos = p;
		}
		
		public int compareTo(Peg o)
		{
			return pos - o.pos;
		}
	}
	
	static class FastScanner
	{
		BufferedReader br;
		StringTokenizer st;
		
		public FastScanner(InputStream input)
		{
			br = new BufferedReader(new InputStreamReader(input));
			st = new StringTokenizer("");
		}
		
		public String next() throws IOException
		{
			if(st.hasMoreTokens())
			{
				return st.nextToken();
			}
			else
			{
				st = new StringTokenizer(br.readLine());
				return next();
			}
		}
		
		public int nextInt() throws IOException
		{
			return Integer.parseInt(next());
		}
	}
}
