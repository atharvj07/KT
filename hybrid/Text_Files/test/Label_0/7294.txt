#define REP(i,s,n) for(int i=s;i<n;i++)
#define rep(i,n) REP(i,0,n)
#define inf (1<<28)
#define EPS 1e-10
#define equals(a,b) (fabs((a)-(b)) < EPS) 
#define MAX 5010
using namespace std;
/*
dp[今どこの切断可能な点をみているのか][自分のとったお菓子の数][１つ前の切断可能な点を切断したかしないか（0->切断していない,1->切断した）] := その時の最小コスト

切断する -> 切断した場所のお菓子を自分のものとする
 */
int dp[2][MAX][2];

/*
cost[0]とcost[N]は端の部分
 */
int cost[MAX*2];
int N;


int main()
{
  cin >> N;
  REP(i,1,N)cin >> cost[i];
  rep(i,2)rep(j,MAX)rep(k,2)dp[i][j][k] = inf;
  dp[0][0][0] = 0;

  rep(i,N+1)
    {
      rep(j,min((int)(N/2),i)+1)
	{
	  dp[(i+1)&1][j][0]   = min(dp[i&1][j][0],dp[i&1][j][1]);
	  dp[(i+1)&1][j+1][1] = min(dp[i&1][j][0]+cost[i]+cost[i+1],dp[i&1][j][1]-cost[i]+cost[i+1]);
	}
    }
  cout << min(dp[N%2][N/2][0],dp[N%2][N/2][1]) << endl;
  return 0;
}
