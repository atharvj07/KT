import itertools
s = input()
e = len(s)
while len(s) < 10:
	s = '0' + s
m = 0
for i in itertools.product(range(2), repeat=10):
	loc = ''
	for j in range(10):
		if i[j]:
			loc += s[j]
	if not loc:
		continue
	if loc and loc[0] == '0':
		continue
	k = int(loc) ** 0.5
	if int(k) != k:
		continue
	if len(loc) > m:
		m = len(loc)
print(e - m if m != 0 else -1)
