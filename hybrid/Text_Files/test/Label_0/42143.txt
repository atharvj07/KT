import sys
input = sys.stdin.readline
t = int(input())
for i in range(t):
	n = int(input())
	stri = input()
	stri = stri[0:n]
	z = 'z' * 50002
	if n == 1:
		print(stri)
		print(1)
	else:
		count = 0
		if n % 2 == 1:
			for i in range(1, n + 1):
				if i % 2 == 0:
					s = stri[i - 1:] + stri[:i - 1]
					if s < z:
						z = s
						count = i
				else:
					s = stri[i - 1:n] + stri[0:i - 1][::-1]
					if s < z:
						z = s
						count = i
		else:
			for i in range(1, n + 1):
				if i % 2 == 1:
					s = stri[i - 1:] + stri[:i - 1]
					if s < z:
						z = s
						count = i
				else:
					s = stri[i - 1:n] + stri[0:i - 1][::-1]
					if s < z:
						z = s
						count = i
		print(''.join(z))
		print(count)
