#include <bits/stdc++.h>
using namespace std;
struct node {
  int u, d, l, r;
  int x;
} p[2000200];
int n, m, q, tot, row[1010], col[1010];
int main() {
  scanf("%d%d%d", &n, &m, &q);
  for (int i = 1; i <= n; i++) {
    tot++;
    row[i] = tot;
    tot++;
  }
  for (int i = 1; i <= m; i++) {
    tot++;
    col[i] = tot;
    tot++;
  }
  for (int i = 1; i <= n; i++) {
    p[row[i]].r = tot + 1;
    p[row[i] + 1].l = tot + m;
    for (int j = 1; j <= m; j++) {
      tot++;
      scanf("%d", &p[tot].x);
      if (j < m)
        p[tot].r = tot + 1;
      else
        p[tot].r = row[i] + 1;
      if (j > 1)
        p[tot].l = tot - 1;
      else
        p[tot].l = row[i];
      if (i == 1) {
        p[tot].u = col[j];
        p[col[j]].d = tot;
      } else
        p[tot].u = tot - m;
      if (i == n) {
        p[tot].d = col[j] + 1;
        p[col[j] + 1].u = tot;
      } else
        p[tot].d = tot + m;
    }
  }
  for (int i = 1; i <= q; i++) {
    int a, b, c, d, h, w;
    scanf("%d%d%d%d%d%d", &a, &b, &c, &d, &h, &w);
    int k1 = row[a], k2 = row[c];
    for (int j = 1; j <= b; j++) k1 = p[k1].r;
    for (int j = 1; j <= d; j++) k2 = p[k2].r;
    swap(p[k1].u, p[k2].u);
    p[p[k1].u].d = k1;
    p[p[k2].u].d = k2;
    for (int j = 1; j < w; j++) {
      k1 = p[k1].r;
      k2 = p[k2].r;
      swap(p[k1].u, p[k2].u);
      p[p[k1].u].d = k1;
      p[p[k2].u].d = k2;
    }
    swap(p[k1].r, p[k2].r);
    p[p[k1].r].l = k1;
    p[p[k2].r].l = k2;
    for (int j = 1; j < h; j++) {
      k1 = p[k1].d;
      k2 = p[k2].d;
      swap(p[k1].r, p[k2].r);
      p[p[k1].r].l = k1;
      p[p[k2].r].l = k2;
    }
    swap(p[k1].d, p[k2].d);
    p[p[k1].d].u = k1;
    p[p[k2].d].u = k2;
    for (int j = 1; j < w; j++) {
      k1 = p[k1].l;
      k2 = p[k2].l;
      swap(p[k1].d, p[k2].d);
      p[p[k1].d].u = k1;
      p[p[k2].d].u = k2;
    }
    swap(p[k1].l, p[k2].l);
    p[p[k1].l].r = k1;
    p[p[k2].l].r = k2;
    for (int j = 1; j < h; j++) {
      k1 = p[k1].u;
      k2 = p[k2].u;
      swap(p[k1].l, p[k2].l);
      p[p[k1].l].r = k1;
      p[p[k2].l].r = k2;
    }
  }
  for (int i = 1; i <= n; i++) {
    int k = row[i];
    for (int j = 1; j <= m; j++) {
      k = p[k].r;
      printf("%d ", p[k].x);
    }
    printf("\n");
  }
  return 0;
}
