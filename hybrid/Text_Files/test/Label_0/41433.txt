import sys
(n, m) = map(int, input().split())
lis_2d = [[int(i) - 1 for i in input().split()] for i in range(n)]
final = 0
for j in range(m):
	count = [0] * n
	for i in range(n):
		if lis_2d[i][j] % m == j:
			pos = lis_2d[i][j] // m
			if pos < n:
				k = (i - pos + n) % n
				count[k] += 1
	ans1 = sys.maxsize
	for i in range(len(count)):
		ans = n - count[i] + i
		if ans < ans1:
			ans1 = ans
	final += ans1
print(final)
