import java.io.*;
import java.util.*;

import static java.lang.Math.*;
import static java.util.Arrays.*;

public class ladders115a {
    static BufferedReader __in;
    static PrintWriter __out;
    static StringTokenizer input;

    public static void main(String[] args) throws IOException {
        __in = new BufferedReader(new InputStreamReader(System.in));
        __out = new PrintWriter(new OutputStreamWriter(System.out));

        int n = readInt();
        int ans = 0;
        Node[] g = new Node[n];
        for(int i = 0; i < n; ++i) {
            g[i] = new Node();
        }
        for(int i = 0; i < n; ++i) {
            int p = readInt();
            if(p > 0) {
                g[i].next = g[p - 1];
            }
        }
        for(int i = 0; i < n; ++i) {
            ans = max(ans, g[i].d());
        }
        println(ans);

        close();
    }

    static class Node {
        Node next;
        int d = -1;

        int d() {
            if(d > 0) {
                return d;
            }
            if(next == null) {
                return 1;
            }
            return d = (1 + next.d());
        }
    }
