import sys
input = sys.stdin.readline
I = lambda : list(map(int, input().split()))
(t,) = I()
for _ in range(t):
	(n, k) = I()
	if k == 0:
		a = '9' * (n // 9) + (str(n % 9) if n % 9 else '')
		print(a[::-1])
		continue
	mi = (k + 1) * k // 2
	if n < mi:
		print(-1)
	else:
		an = -1
		for i in range(10):
			mi = 0
			for j in range(k + 1):
				mi += sum((int(p) for p in str(i + j)))
			if n < mi or (n - mi) % (k + 1):
				continue
			a = str(i)
			r = (n - mi) // (k + 1)
			if i + k >= 10:
				a += str(min(8, r))
				r -= min(8, r)
			while r:
				a += str(min(9, r))
				r -= min(9, r)
			if an == -1:
				an = int(a[::-1])
			else:
				an = min(an, int(a[::-1]))
		print(an)
