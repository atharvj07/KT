#include "bits/stdc++.h"
#include<unordered_map>
#include<unordered_set>
#pragma warning(disable:4996)
using namespace std;
using ld = long double;
template<class T>
using Table = vector<vector<T>>;


int main() {
	int N, M; cin >> N >> M;
	vector<vector<ld>>pers(M +1,vector<ld>(M +1));
	pers[0][0] = 1;
	for (int i = 1; i <= M; ++i) {
		for (int j = 0; j < M; ++j) {
			pers[i][j] += pers[i - 1][j] *2 / 3;
			pers[i][j + 1]+= pers[i - 1][j] / 3;
		}
		
	}
	for (int i = 1; i <= M; ++i) {
		for (int j = M; j >0; --j) {
			pers[i][j - 1] += pers[i][j];
		}
	}
	
	vector<int>pluspoints{ 5,3,2 };
	vector<int>mys(3);
	vector<vector<int>>ops(3);
	vector<vector<ld>>kitaitis(3,vector<ld>(M+1));
	{
		cin >> mys[0] >> mys[1] >> mys[2];
	}
	for (int  i = 1; i < N; i++)
	{
		int vi, da, vo; cin >> vi >> da >> vo;
		ops[0].push_back(vi);
		ops[1].push_back(da);
		ops[2].push_back(vo);
	}
	for (int perform = 0; perform < 3; ++perform) {
		for (int chal = 0; chal <= M; ++chal) {
			ld kitai = 0;
			int mypoint = chal*mys[perform];
			vector<ld>losepers(N - 1);
			for (int op = 0; op < N - 1; ++op) {
				int needchal = (mypoint - 1) / ops[perform][op] + 1;
				if (mypoint == 0)needchal = 0;
				if (needchal > M) {
					losepers[op] = 0;
				}
				else {
					losepers[op] = pers[M][needchal];
				}
			}
			vector<vector<ld>>windp(N, vector<ld>(4));
			windp[0][0] = 1;
			for (int op = 0; op < N-1; ++op) {
				for (int j = 0; j < 3; ++j) {
					windp[op + 1][j] += windp[op][j] * (1 - losepers[op]);
					windp[op + 1][j + 1] += windp[op][j] * losepers[op];
				}
			}
			kitai += (windp[N - 1][0] + windp[N - 1][1] + windp[N - 1][2])*pluspoints[perform];

			ld biri = 1;
			for (int op = 0; op < N - 1; ++op) {
				biri *= losepers[op];
			}
			kitai -= biri;
			kitaitis[perform][chal] = kitai;
		}
	}
	ld ans = -1e18;
	for (int a = 0; a <= M; ++a) {
		for (int b = 0; b <= M - a; ++b) {
			int c = M - a - b;
			if (ans<kitaitis[0][a] + kitaitis[1][b] + kitaitis[2][c])
			{
				ans = max(ans, kitaitis[0][a] + kitaitis[1][b] + kitaitis[2][c]);
			}

		}
	}
	cout << fixed<<setprecision(22)<<ans << endl;


	return 0;
}
