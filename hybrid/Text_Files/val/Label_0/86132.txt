#include <bits/stdc++.h>
using namespace std;
const int MAXN = 4000;
const int INF = 1000000010;
const long long int MOD = 1000000007;
int a, ta, b, tb;
int main() {
  int lastime = 24 * 60 - 5 * 60;
  scanf("%d %d", &a, &ta);
  scanf("%d %d", &b, &tb);
  int h, m;
  scanf("%d:%d", &h, &m);
  int tm = h * 60 + m - 5 * 60;
  int ttm = tm + ta;
  int ans = 0;
  for (int i = 0; i < lastime; i += b) {
    if (i + tb > tm && i < ttm) ans++;
  }
  printf("%d", ans);
  return 0;
}
