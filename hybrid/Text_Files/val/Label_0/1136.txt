(n, m, k) = map(int, input().split())
if n == 1:
	print(m + m)
	print('L' * m + 'R' * m)
	exit()
if m == 1:
	print(n + n)
	print('U' * n + 'D' * n)
	exit()
if m == n == 2:
	print('5\nULRDL')
	exit()
print(n + m + n * m + n - 1)
ans = 'L' * m + 'U' * n
for i in range((n - n % 2) // 2):
	ans += 'R' * m + 'D' + 'L' * m + 'D'
if n % 2 == 1:
	ans += 'R' * m + 'D'
ans = ans[:-1]
print(ans)
