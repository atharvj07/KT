import java.io.*;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.StringTokenizer;

public class mainE {
    public static PrintWriter out = new PrintWriter(System.out);
    public static FastScanner enter = new FastScanner(System.in);

    public static void main(String[] args) throws IOException {
        solve();
        out.close();
    }
    static LinkedList<Integer>[] vert;
    static long[] kras;
    static long ans=0;
    static long mod=(long)1e9+7;
    private static void solve() throws IOException {
        int n=enter.nextInt();
        vert=new LinkedList[n+1];
        hm=new HashMap[n+1];
        kras=new long[n+1];
        for (int i = 0; i < n; i++) {
            kras[i+1]=enter.nextLong();
        }
        for (int i = 0; i < n + 1; i++) {
            vert[i]=new LinkedList<>();
            hm[i]=new HashMap<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int a=enter.nextInt();
            int b=enter.nextInt();
            vert[a].addLast(b);
            vert[b].addLast(a);
        }
        dsu(1, -1, new boolean[n+1]);
        out.println(ans);
    }
    static HashMap<Long, Integer>[] hm;
    private static void dsu(int v, int p, boolean[] was) {
        was[v]=true;
        if(hm[v].get(kras[v])!=null)hm[v].put(kras[v], hm[v].get(kras[v])+1);
        else hm[v].put(kras[v],1);
        if(p!=-1){
            for (Long val:hm[p].keySet()) {
                long res=gcd(val, kras[v]);
                if(hm[v].get(res)!=null)hm[v].put(res, hm[v].get(res)+hm[p].get(val));
                else hm[v].put(res,hm[p].get(val));
            }
        }
        for (int son:vert[v]) {
            if(!was[son]) dsu(son, v, was);
        }
        for (long val:hm[v].keySet()) {
            ans+=(val%mod*hm[v].get(val)%mod)%mod;
            ans%=mod;
        }
    }
    public static long gcd(long a, long b){
        while(b!=0){
            a%=b;
            a^=(b^=a);
            b^=a;
        }
        return a;
    }


    static class FastScanner {
        BufferedReader br;
        StringTokenizer stok;

        FastScanner(InputStream is) {
            br = new BufferedReader(new InputStreamReader(is));
        }

        String next() throws IOException {
            while (stok == null || !stok.hasMoreTokens()) {
                String s = br.readLine();
                if (s == null) {
                    return null;
                }
                stok = new StringTokenizer(s);
            }
            return stok.nextToken();
        }

        int nextInt() throws IOException {
            return Integer.parseInt(next());
        }

        long nextLong() throws IOException {
            return Long.parseLong(next());
        }

        double nextDouble() throws IOException {
            return Double.parseDouble(next());
        }

        char nextChar() throws IOException {
            return (char) (br.read());
        }

        String nextLine() throws IOException {
            return br.readLine();
        }
    }
}
