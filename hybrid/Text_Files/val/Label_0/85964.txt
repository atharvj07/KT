#include <bits/stdc++.h>
using namespace std;
int n, m, t, o, l[35], L, R;
int main() {
  scanf("%d%d", &m, &n);
  t = m;
  for (int i = 1; i <= n; i++) {
    printf("%d\n", t);
    fflush(stdout);
    scanf("%d", &o);
    if (o == 0)
      exit(0);
    else if (o == 1)
      l[i] = -1;
    else
      l[i] = 1;
  }
  L = 1, R = m;
  int f = 0;
  while (L < R) {
    printf("%d\n", (L + R >> 1));
    fflush(stdout);
    scanf("%d", &o);
    ++f;
    if (f > n) f = 1;
    if (o == 0)
      exit(0);
    else if (o * l[f] == 1)
      L = (L + R >> 1) + 1;
    else
      R = (L + R >> 1) - 1;
  }
  printf("%d\n", (L + R >> 1));
  fflush(stdout);
  scanf("%d", &o);
  return 0;
}
