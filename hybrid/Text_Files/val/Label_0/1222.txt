n = int(input())
F = [int(i) for i in input().split()]
F.sort()
if n % 2 != 0:
	A = F[:int((n - 1) / 2)]
	B = F[int((n - 1) / 2):]
	for i in range(int((n + 1) / 2)):
		F[2 * i] = B[i]
	for i in range(int((n - 1) / 2)):
		F[2 * i + 1] = A[i]
else:
	A1 = F[:int(n / 2)]
	B1 = F[int(n / 2):]
	for i in range(int(n / 2)):
		F[2 * i] = B1[i]
	for i in range(int(n / 2)):
		F[2 * i + 1] = A1[i]
count = 0
for i in range(1, n - 1):
	if F[i] < F[i - 1] and F[i] < F[i + 1]:
		count += 1
print(count)
print(*F)
