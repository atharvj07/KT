(n, m) = map(int, input().split())
a = [[] for i in range(201)]
for i in range(n):
	(b, c) = map(int, input().split())
	a[b].append([c, i + 1])
l = 0
d = 0
stack = []
count = 0
for i in range(1, 201):
	a[i].sort()
	l = len(a[i])
	if l > m:
		d = l - m
		for j in range(d):
			stack.append(a[i][-1][1])
			a[i].pop(-1)
			count += 1
	while len(a[i]) > 0 and a[i][0][0] == i:
		a[i].pop(0)
	if len(a[i]) > 0:
		a[i + 1] = a[i + 1] + a[i]
print(count)
for i in range(count):
	print(stack[i], end=' ')
