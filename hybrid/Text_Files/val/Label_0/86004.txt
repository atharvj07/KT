#include <bits/stdc++.h>
using namespace std;
vector<int> di[200111], si[200111];
int val[200111], n;
long long st[4 * 200111];
int lz[4 * 200111], u[4 * 200111], v[4 * 200111];
vector<long long> f, fval;
long long init_st(int i, int x, int y) {
  u[i] = x;
  v[i] = y;
  lz[i] = 0;
  if (x == y) return st[i] = x;
  int mid = (x + y) >> 1;
  return st[i] = init_st(i << 1, x, mid) + init_st(i << 1 | 1, mid + 1, y);
}
long long upd_st(int i, int x, int y, int k, int kall) {
  if (kall != 0) {
    lz[i] = kall;
    st[i] = lz[i] * (long long)(v[i] - u[i] + 1);
  }
  if (y < u[i] || v[i] < x) {
    return st[i];
  }
  if (x <= u[i] && v[i] <= y) {
    lz[i] = k;
    return st[i] = lz[i] * (long long)(v[i] - u[i] + 1);
  }
  kall = lz[i];
  lz[i] = 0;
  return st[i] =
             upd_st(i << 1, x, y, k, kall) + upd_st(i << 1 | 1, x, y, k, kall);
}
long long get_st(int i, int x, int y, int kall) {
  if (kall != 0) {
    lz[i] = kall;
    st[i] = lz[i] * (long long)(v[i] - u[i] + 1);
  }
  if (y < u[i] || v[i] < x) {
    return 0;
  }
  if (x <= u[i] && v[i] <= y) {
    return st[i];
  }
  kall = lz[i];
  lz[i] = 0;
  return get_st(i << 1, x, y, kall) + get_st(i << 1 | 1, x, y, kall);
}
void update(int x, int y, int k) {
  int mid, res = -1, u = x;
  while (x <= y) {
    mid = (x + y) >> 1;
    if (get_st(1, mid, mid, 0) < k) {
      res = mid;
      x = mid + 1;
    } else
      y = mid - 1;
  }
  if (res != -1) upd_st(1, u, res, k, 0);
}
int main() {
  for (int i = 1; i <= 200001; i++) {
    for (int j = i; j <= 200001; j += i) di[j].push_back(i);
  }
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &val[i]);
    for (int j = 0; j < di[val[i]].size(); j++) si[di[val[i]][j]].push_back(i);
  }
  init_st(1, 1, n);
  f.push_back(1ll * n * (n + 1) / 2);
  for (int i = 200000; i >= 1; i--) {
    if (si[i].size() > 1) {
      int a1 = si[i][0];
      int a2 = si[i][1];
      int a3 = si[i][si[i].size() - 2];
      int a4 = si[i].back();
      update(1, a1, a3);
      update(a1 + 1, a2, a4);
      update(a2 + 1, n, n + 1);
      f.push_back(get_st(1, 1, n, 0));
      fval.push_back(i);
    }
  }
  long long res = 0;
  for (int i = 0; i < fval.size(); i++) {
    res += (f[i + 1] - f[i]) * fval[i];
  }
  cout << res << endl;
  return 0;
}
