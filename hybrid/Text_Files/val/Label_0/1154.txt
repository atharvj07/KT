def solve(x):
	ones = x.count(1)
	if ones == 1:
		return 0
	mini = 0
	z = []
	for i in range(len(x)):
		if x[i] == 1:
			z.append(i)
	for i in range(1, len(z)):
		mini += z[i] - z[i - 1] - 1
	return mini
for _ in range(int(input())):
	x = int(input())
	l = list(map(int, input().split()))
	print(solve(l))
