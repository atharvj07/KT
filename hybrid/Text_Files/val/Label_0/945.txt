r = int(input())
for i in range(r):
	a = 0
	n = int(input())
	num = input()
	list = [int(x) for x in num.split()]
	if n == 1 and list[0] % 2 != 0:
		print(-1)
	else:
		for i in range(n):
			if list[i] % 2 == 0:
				print(1)
				print(i + 1)
				break
			else:
				a += 1
			if a == 2:
				print(2)
				print(1, 2, end=' ')
				print()
				break
