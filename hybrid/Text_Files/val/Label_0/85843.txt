#include <bits/stdc++.h>
using namespace std;
int main() {
  string s1, s2;
  cin >> s1 >> s2;
  vector<set<int> > d(256);
  for (int i = 0; i < (int)(s1.size()); ++i) {
    d[s1[i]].insert(i);
  }
  int curr = -1, res = 1;
  for (int i = 0; i < (int)(s2.size()); ++i) {
    char c = s2[i];
    if (d[c].empty()) {
      cout << -1 << endl;
      return 0;
    }
    set<int>::iterator it = d[c].upper_bound(curr);
    if (it == d[c].end()) {
      curr = *d[c].begin();
      res++;
    } else {
      curr = *it;
    }
  }
  cout << res << endl;
}
