def fun(lst, d):
	flag = 1
	for el in lst:
		if el > d:
			flag = 0
	if flag == 1:
		return 'YES'
	for i in range(len(lst)):
		for j in range(i + 1, len(lst)):
			if lst[i] + lst[j] <= d:
				flag = 1
				break
	if flag == 1:
		return 'YES'
	else:
		return 'NO'
for _ in range(int(input())):
	(n, d) = map(int, input().split())
	lst = [int(i) for i in input().split()]
	print(fun(lst, d))
