(n, m) = [int(x) for x in input().split()]
p = []
for x in range(m):
	t = sorted([int(x) for x in input().split()])
	p.append((t[0], t[1]))
if n < 7:
	print(m)
	exit(0)
res = 0
for i in range(1, 8):
	for j in range(1, 8):
		if i == j:
			continue
		p1 = []
		for (u, v) in p:
			t = [u, v]
			if u == i:
				t = sorted([j, v])
			elif v == i:
				t = sorted([j, u])
			p1.append((t[0], t[1]))
		res = max(res, len(set(p1)))
print(res)
