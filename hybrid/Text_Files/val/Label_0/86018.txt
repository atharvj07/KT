#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  string s;
  cin >> s;
  string::iterator eit = s.end();
  --eit;
  list<int> l;
  l.push_back(1);
  list<int>::iterator it = l.begin();
  int n = 2;
  for (string::iterator i = s.begin(); i != eit; i++) {
    if (*i == 'l')
      it = l.insert(it, n);
    else
      it = l.insert(++it, n);
    ++n;
  }
  for (int i : l) cout << i << '\n';
  return 0;
}
