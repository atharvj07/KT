import java.util.Map.Entry;
import java.util.Scanner;
import java.util.TreeMap;
public class JeffAndPeriodOrg {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        TreeMap<Integer,Integer> diff=new TreeMap<>();
        TreeMap<Integer,Integer> pos=new TreeMap<>();
        
        
        int n=sc.nextInt();
        int a[]=new int[n+1];
        
        
        for(int i=1;i<=n;i++)
        {
            int x=sc.nextInt();
            
            if(!diff.containsKey(x))
            {
                diff.put(x, 0);
                pos.put(x, i);
            }
            else
            {
                int d=i-pos.get(x);
                if(diff.get(x)==0)
                {
                    diff.put(x, d);
                    pos.put(x, i);
                }
                else
                {
                    int s=i-pos.get(x);
                    if(s!=diff.get(x))
                    {
                        diff.put(x,-1);
                        // pos.put(x,i);
                    }
                    else
                    {
                        diff.put(x, s);
                        pos.put(x,i);
                    }
                    
                }
            }
        }
        StringBuilder sb=new StringBuilder();
        int c=0;
        for(Entry<Integer,Integer> en:diff.entrySet())
        {
            if(en.getValue()==-1)
            {
                continue;
            }
            c++;
            
            {
                sb.append(en.getKey()+" "+en.getValue()+"\n");
            }
        }
        System.out.println(c);
        System.out.println(sb);
    }
}
