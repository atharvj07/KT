import sys
import bisect
import math
from collections import deque
input = lambda : sys.stdin.readline().strip()
mod = 10 ** 9 + 9

def solve():
	n = int(input())
	b = list(map(int, input().split()))
	a = []
	for i in range(n):
		a.append([b[i], i + 1])
	queue = deque()
	queue.append(0)
	a.sort(reverse=True)
	for i in range(0, n, 2):
		queue.append(a[i][1])
	for i in range(1, n, 2):
		queue.appendleft(a[i][1])
	a = [x for x in queue]
	arr = [0] * (n + 1)
	for i in range(n + 1):
		arr[a[i]] = i
	ans = 0
	for i in range(1, n + 1):
		ans += 2 * abs(arr[i] - arr[0]) * b[i - 1]
	return str(ans) + '\n' + ' '.join((str(x) for x in arr))
t = 1
t = int(input())
for _ in range(t):
	print(solve())
