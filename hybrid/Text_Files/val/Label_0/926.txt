(n, k) = [int(x) for x in input().split()]
if 2 ** min(n, 62) < k:
	print('1 1')
	exit(0)
mod = int(1000003)
x = n
mult = 2
while mult <= k - 1:
	x += (k - 1) // mult
	mult *= 2

def modpow(x, n, m):
	mult = x % m
	a = 1
	ans = 1
	while a <= n:
		if a & n:
			ans *= mult
			ans %= m
		mult *= mult
		mult %= m
		a *= 2
	return ans
denom = modpow(2, n * k - x, mod)
num = modpow(2, n * k - x, mod)
if k < mod:
	s = 1
	for i in range(1, k):
		val = i
		y = 0
		while not val % 2:
			val //= 2
			y += 1
		s *= modpow(2, (n - y) % (mod - 1), mod) - val
		s %= mod
	num -= s
	num %= mod
	if num < 0:
		s += mod
print(num, denom)
