#include "iostream"
#include "math.h"
using namespace std;

int N;
int num;

int main() {
	cin >> N;
	if (N == 2) {
		cout << "-1\n";
		return 0;
	}
	if (N % 2 == 1) {
		for (int i = 1; i <= N; i++) {
			for (int j = 0; j < N-1; j++) {
				num = i + j+1;
				if (num > N) {
					num %= N;
				}
				cout << num << " ";
			}
			cout << "\n";
		}
	}
	else {
		for (int i = 1; i <= N; i++) {
			for (int j = 0; j < N - 1; j++) {
				if (i *2<=N) {
					if ((j * 2 + 2) == N) {
						num = i + N - 1;
						if (num > N) {
							num %= N;
						}
						cout << num << " ";
					}
					else if (j == N - 2) {
						num = i + N / 2;
						if (num > N) {
							num %= N;
						}
						cout << num << " ";
					}
					else {
						num = i + j + 1;
						if (num > N) {
							num %= N;
						}
						cout << num << " ";
					}
				}
				else {
					num = i + j + 1;
					if (num > N) {
						num %= N;
					}
					cout << num << " ";
				}
			}
			cout << "\n";
		}
	}
	return 0;
}
