#include<bits/stdc++.h>
using namespace std;
#define int long long
#define fast ios_base::sync_with_stdio(false);cin.tie(NULL);

void sol(){
    int n,u,v;
    int f=0;
    cin>>n>>u>>v;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        cin>>a[i];
        if(i){
            f=max(f,abs(a[i]-a[i-1]));
        }
    }
    if(f>1)cout<<0;
    else if(f==0)   cout<<min(2*v,u+v);
    else    cout<<min(u,v);
}
signed main(){
    //fast;
    int t=1;    cin>>t;
    while(t--){
        sol();
        cout<<"\n";
    }
}
