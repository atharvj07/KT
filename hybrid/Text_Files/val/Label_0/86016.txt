void add(vector<int>& a) {
  int inc = 1;
  for (int i = a.size() - 1; i >= 0; i--) {
    int temp = a[i];
    a[i] = (temp + inc) % 10;
    inc = (temp + inc) / 10;
  }
}
int compare(vector<int>& a, vector<int>& b, int k) {
  for (int i = 0; i < a.size(); i++) {
    if (b[i + k] > a[i]) return -1;
    if (b[i + k] < a[i]) return 1;
  }
  return 0;
}
int main() {
  long long n, k;
  cin >> n >> k;
  vector<int> x;
  long long N = n;
  if (n % k != 0) N = k * (n / k + 1);
  x.resize(N, 0);
  for (int i = 0; i < n; i++) {
    char a;
    cin >> a;
    x[i] = a - 48;
  }
  vector<int> b;
  for (int i = 0; i < k; i++) {
    b.push_back(x[i]);
  }
  for (int i = k; i < N; i += k) {
    if (compare(b, x, i) == -1) {
      add(b);
      break;
    }
    if (compare(b, x, i) == 1) {
      break;
    }
  }
  cout << n << endl;
  long long temp = 0;
  for (int i = 0; i < n; i++) {
    cout << b[temp];
    temp++;
    if (temp == k) temp = 0;
  }
  return 0;
}
