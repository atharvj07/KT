from math import *
(n, k) = list(map(int, input().split()))
l = list(map(int, input().split()))
s = []
for i in range(len(l)):
	s.append([int(str(l[i])[-1]), i, l[i]])
s.sort(key=lambda s: s[0])
s = s[-1::-1]
i = 0
SUM = 0
while k > 0 and i <= n - 1:
	if k >= 10 - s[i][0] and s[i][0] != 0:
		s[i][2] = s[i][2] + (10 - s[i][0])
		k -= 10 - s[i][0]
		i += 1
	else:
		break
LL = []
for i in range(len(s)):
	LL.append(s[i][2])
LL = sorted(LL)
if k != 0:
	i = 0
	while k > 0 and i <= len(LL) - 1:
		old = LL[i]
		LL[i] = min(100, LL[i] + k)
		k -= LL[i] - old
		i += 1
for i in range(len(LL)):
	SUM += floor(LL[i] / 10)
print(SUM)
