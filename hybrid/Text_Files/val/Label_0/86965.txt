import java.util.*;
import java.io.*;
public class TernaryXOR{
    public static void main(String args[]){
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        try{
            int t=Integer.parseInt(br.readLine());
            while(t-- > 0){
                int n=Integer.parseInt(br.readLine());
                String q=br.readLine();
                String resA="1",resB="1";
                boolean equal=true;
                for (int i = 1; i < n; i++) {
                    char x=q.charAt(i);
                    if(equal==true){
                        if(x=='0'){
                            resA+='0';
                            resB+='0';
                        }
                        else if(x=='2'){
                            resA+='1';
                            resB+='1';
                        }
                        else{
                            equal=false;
                            resA+='1';
                            resB+='0';
                        }
                    }
                    else{
                        if(x=='0'){
                            resA+='0';
                            resB+='0';
                        }
                        else if(x=='2'){
                            resA+='0';
                            resB+='2';
                        }
                        else{
                            equal=false;
                            resA+='0';
                            resB+='1';
                        }
                    }
                }
                System.out.println(resA+"\n"+resB);
            }
            br.close();
        }
        catch(IOException e){}
    }
}
