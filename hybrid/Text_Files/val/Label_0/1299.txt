from sys import stdin
input = stdin.readline
(n, q) = [int(i) for i in input().split()]
s = list(input())
m = ''.join(s)
c = m.count('abc')
for i in range(q):
	(a, b) = input().split()
	a = int(a) - 1
	if s[a] == b:
		print(c)
		continue
	if s[a] == 'a':
		if a < n - 2 and s[a + 1] == 'b' and (s[a + 2] == 'c'):
			c -= 1
	elif s[a] == 'c':
		if a > 1 and s[a - 2] == 'a' and (s[a - 1] == 'b'):
			c -= 1
	elif s[a] == 'b':
		if a < n - 1 and a > 0 and (s[a - 1] == 'a') and (s[a + 1] == 'c'):
			c -= 1
	s[a] = b
	if b == 'a':
		if a < n - 2 and s[a + 1] == 'b' and (s[a + 2] == 'c'):
			c += 1
	elif b == 'c':
		if a > 1 and s[a - 2] == 'a' and (s[a - 1] == 'b'):
			c += 1
	elif a < n - 1 and a > 0 and (s[a - 1] == 'a') and (s[a + 1] == 'c'):
		c += 1
	print(c)
