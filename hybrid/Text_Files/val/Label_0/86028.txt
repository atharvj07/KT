#include<bits/stdc++.h>
using namespace std;
int T,n,a[1000005];
long long k,x;
void solve(int now,long long rk,int num,int len) {
	if(now==n+1)return;
	long long sum=1;
	int p=now;
	if(len>=63) {
		a[now]=1+num;
		solve(now+1,rk,num+1,len-1);
	} else {
		for(int i=1; i<=len-2; i++)sum*=2;
		while(sum) {
			if(rk<=sum)break;
			rk-=sum;
			p++;
			sum/=2;
		}
		a[now]=1+num+p-now;
		for(int i=now+1; i<=p; i++)a[i]=a[i-1]-1;
		solve(p+1,rk,num+p-now+1,len-p+now-1);
	}
}
int main() {
	cin>>T;
	while(T--) {
		cin>>n>>k;
		x=1;
		if(n<=62) {
			for(int i=1; i<n; i++)x*=2;
			if(k>x) {
				puts("-1");
				continue;
			}
		}
		solve(1,k,0,n);
		for(int i=1; i<=n; i++)cout<<a[i]<<' ';
		cout<<endl;
	}
}
