#include <bits/stdc++.h>
struct San : std::vector<int> {
  void prepare() {
    std::sort(begin(), end());
    erase(std::unique(begin(), end()), end());
  }
  int get(int x) { return std::lower_bound(begin(), end(), x) - begin(); }
} ss;
const int N = 100000 + 5;
int A[N], B[N];
int permutation[N + N];
int C[N + N];
int n;
void modify(int p, int dt) {
  for (int i = p; i < ss.size(); i += ~i & i + 1) C[i] += dt;
}
int query(int p) {
  int ret = 0;
  for (int i = p; i >= 0; i -= ~i & i + 1) ret += C[i];
  return ret;
}
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) {
    scanf("%d%d", A + i, B + i);
    ss.push_back(A[i]);
    ss.push_back(B[i]);
  }
  ss.prepare();
  for (int i = 0; i < ss.size(); ++i) {
    permutation[i] = ss[i];
  }
  for (int i = 0; i < n; ++i) {
    std::swap(permutation[ss.get(A[i])], permutation[ss.get(B[i])]);
  }
  long long ret = 0;
  for (int i = 0; i < ss.size(); ++i) {
    ret += i - query(ss.get(permutation[i]));
    modify(ss.get(permutation[i]), 1);
    if (permutation[i] < ss[i]) {
      ret += ss[i] - 1 - permutation[i] - (i - ss.get(permutation[i]) - 1);
    } else if (permutation[i] > ss[i]) {
      ret += permutation[i] - 1 - ss[i] - (ss.get(permutation[i]) - i - 1);
    }
  }
  printf("%I64d\n", ret);
}
