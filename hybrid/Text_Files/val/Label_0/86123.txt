#include <iostream>

using namespace std;
int c[26],s[400][30];
int main(){
    int i,j,d; cin >> d;
    for(i=0;i<26;i++){
        cin >> c[i];
    }
    for(i=0;i<d;i++){
        for(j=0;j<26;j++){
            cin >> s[i][j];
        }
    }
    for(i=0;i<d;i++){
        cout << i%26 + 1 << endl;
    }
}
