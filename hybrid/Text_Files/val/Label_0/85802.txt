#include <bits/stdc++.h>
using namespace std;
long long int brute(vector<pair<long long int, long long int> > v1,
                    vector<pair<long long int, long long int> > v2) {
  long long int n = v1.size(), m = v2.size();
  long long int ans = 0, ind = 0;
  set<long long int> val;
  long long int h[10] = {0};
  for (long long int i = 0; i < n; i++) {
    long long int cnt = 0;
    val.clear();
    long long int p = v1[i].first, q = v1[i].second;
    if (p > q) swap(p, q);
    for (long long int j = 0; j < m; j++) {
      long long int r = v2[j].first, s = v2[j].second;
      if (r > s) swap(r, s);
      if (p == r && q != s)
        cnt++, val.insert(p);
      else if (p == s && q != r)
        cnt++, val.insert(p);
      else if (q == r && p != s)
        cnt++, val.insert(q);
      else if (q == s && p != r)
        cnt++, val.insert(q);
    }
    if (val.size() >= 2) return -1;
    if (val.size()) ans = *val.begin(), h[ans] = 1;
  }
  for (long long int i = 0; i < m; i++) {
    long long int cnt = 0;
    val.clear();
    long long int p = v2[i].first, q = v2[i].second;
    if (p > q) swap(p, q);
    for (long long int j = 0; j < n; j++) {
      long long int r = v1[j].first, s = v1[j].second;
      if (r > s) swap(r, s);
      if (p == r && q != s)
        cnt++, val.insert(p);
      else if (p == s && q != r)
        cnt++, val.insert(p);
      else if (q == r && p != s)
        cnt++, val.insert(q);
      else if (q == s && p != r)
        cnt++, val.insert(q);
    }
    if (val.size() >= 2) return -1;
    if (val.size()) ans = *val.begin(), h[ans] = 1;
  }
  for (long long int i = 1; i < 10; i++) ind += h[i];
  if (ind == 1) return ans;
  return 0;
}
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long int n, m;
  cin >> n >> m;
  vector<pair<long long int, long long int> > v1, v2;
  for (long long int i = 0; i < n; i++) {
    long long int x, y;
    cin >> x >> y;
    v1.push_back({x, y});
  }
  for (long long int i = 0; i < m; i++) {
    long long int x, y;
    cin >> x >> y;
    v2.push_back({x, y});
  }
  cout << brute(v1, v2);
}
