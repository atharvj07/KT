#include <bits/stdc++.h>
using namespace std;
template <class T>
void vp(T &a, int p) {
  for (int i = 0; i < p; i++) cout << a[i] << " ";
  cout << endl;
}
const int N = 100001;
long long mul[100];
int getDiff(long long a, long long b) {
  int cnt = 0;
  while (a || b) {
    if (a % 10 != b % 10) cnt++;
    a /= 10;
    b /= 10;
  }
  return cnt;
}
void decre(long long &h, long long &b, long long m) {
  if (b == 0)
    h--, b = m - 1;
  else
    b--;
}
void incre(long long &h, long long &b, long long m) {
  if (b == m - 1)
    h++, b = 0;
  else
    b++;
}
long long getVal(long long h, long long m, int k, const long long M) {
  if (h == 0 && m == 0) return 0;
  long long ret = 0;
  if (h > 0) {
    ret += (h) * ((M - 1) / mul[k]);
  }
  if (m > 0) {
    ret += m / mul[k];
  }
  int atl = k - getDiff(0, M - 1);
  if (atl <= 0) {
    ret += h;
  } else {
    ret += h / mul[atl];
  }
  return ret;
}
long long compute(long long h1, long long m1, long long h2, long long m2, int k,
                  long long M) {
  long long a = getVal(h1, m1, k, M), b = getVal(h2, m2, k, M);
  return b - a;
}
int main() {
  mul[0] = 1;
  mul[1] = 1;
  for (int i = 2; i < 100; i++) mul[i] = 10 * mul[i - 1];
  long long h, m;
  int k;
  while (cin >> h >> m >> k) {
    long long h1, h2, m1, m2;
    cin >> h1 >> m1 >> h2 >> m2;
    if (h1 == h2 && m1 == m2) {
      cout << 0 << endl;
      goto end;
    }
    if (h1 < h2 || (h1 == h2 && m1 < m2)) {
      cout << compute(h1, m1, h2, m2, k, m) << endl;
    } else {
      long long th[2][2], tm[2][2];
      long long val[2];
      th[0][0] = 0;
      tm[0][0] = 0;
      th[0][1] = h2;
      tm[0][1] = m2;
      th[1][0] = h1;
      tm[1][0] = m1;
      th[1][1] = h - 1;
      tm[1][1] = m - 1;
      long long ec = getDiff(m - 1, 0) + getDiff(h - 1, 0) >= k ? 1 : 0;
      for (int i = 0; i < 2; i++)
        val[i] = compute(th[i][0], tm[i][0], th[i][1], tm[i][1], k, m);
      cout << val[0] + val[1] + ec << endl;
    }
  end:;
  }
  return false;
}
