import java.util.*;
public class ndz {

 public static void main(String[] args) {
   Scanner no=new Scanner(System.in);
   //1234567
   //12345678
   int n=no.nextInt();
   int m=no.nextInt();
   if(n%2==1){
     if(n==1){
       System.out.println(1);
     }
    else{
     int r=(n+1)/2;
     if(m>r){
       System.out.println(m-1);
     }
     else if(m<r){
       System.out.println(m+1);
     }
     else{
       System.out.println(r-1);
     }
    }
   }
   else{
     int r=n/2;
     if(m>r){
       System.out.println(m-1);
     }
     else if(m<r){
       System.out.println(m+1);
     }
     else{
       System.out.println(r+1);
     }
   }
  }

}
