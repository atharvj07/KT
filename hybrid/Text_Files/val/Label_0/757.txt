import sys, bisect
(n, m) = map(int, sys.stdin.readline().split())
dorm = list(map(int, sys.stdin.readline().split()))
psum = [dorm[0]]
for i in range(1, n):
	psum.append(psum[i - 1] + dorm[i])
now = 0
arr = list(map(int, sys.stdin.readline().split()))
i = 0
for j in range(m):
	if psum[i] >= arr[j]:
		print(i + 1, arr[j] - now)
	else:
		while psum[i] < arr[j]:
			i += 1
		now = psum[i - 1]
		print(i + 1, arr[j] - now)
