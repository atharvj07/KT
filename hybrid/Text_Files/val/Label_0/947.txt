n = int(input())
while n != 0:
	l = int(input())
	ar = input()
	p = 0
	arr = ar.split(' ')
	ans = []
	sum = 0
	for i in range(l):
		if int(arr[i]) % 2 == 0:
			print(1)
			p = 1
			print(i + 1)
			break
		else:
			ans.append(i + 1)
			sum += int(arr[i])
			if sum % 2 == 0:
				p = 1
				print(2)
				print(ans[0], ans[1])
				break
	if p == 0:
		print(-1)
	n -= 1
