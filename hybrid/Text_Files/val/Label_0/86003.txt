const int dd = 4e5 + 7;
long long S[4 * dd], P[4 * dd], MN[4 * dd], MX[4 * dd];
long long G[dd];
void push(int v, int tl, int tr) {
  if (P[v]) {
    P[v * 2] = P[v * 2 + 1] = P[v];
    MN[v] = MX[v] = P[v];
    S[v] = (tr - tl + 1) * P[v];
    P[v] = 0;
  }
}
long long get(int v, int tl, int tr) {
  if (P[v]) {
    return (tr - tl + 1) * P[v];
  } else {
    return S[v];
  }
}
void upd(int v, int tl, int tr, int l, int r, int val) {
  if (MN[v] >= val || l > r) return;
  push(v, tl, tr);
  if (tl == l && tr == r && MX[v] <= val) {
    P[v] = val;
  } else {
    int tm = (tl + tr) / 2;
    if (r <= tm)
      upd(v * 2, tl, tm, l, r, val);
    else if (l > tm)
      upd(v * 2 + 1, tm + 1, tr, l, r, val);
    else {
      upd(v * 2, tl, tm, l, tm, val);
      upd(v * 2 + 1, tm + 1, tr, tm + 1, r, val);
    }
    push(v * 2, tl, tm);
    push(v * 2 + 1, tm + 1, tr);
    S[v] = S[v * 2] + S[v * 2 + 1];
    MN[v] = min(MN[v * 2], MN[v * 2 + 1]);
    MX[v] = max(MX[v * 2], MX[v * 2 + 1]);
  }
}
vector<int> V[dd];
long long H[dd];
int solve() {
  int n;
  cin >> n;
  for (int j = 1; j < (long long)n + 1; j++) {
    scanf("%lld", G + j);
    for (int i = 1; i * i <= G[j]; i++) {
      if (G[j] % i == 0) {
        V[i].push_back(j);
        if (i * i != G[j]) {
          V[G[j] / i].push_back(j);
        }
      }
    }
  }
  for (int i = 1; i < (long long)n + 1; i++) {
    upd(1, 1, n, i, i, i);
  }
  for (int i = 2e5 + 1; i >= 1; i--) {
    if (V[i].size() > 1) {
      upd(1, 1, n, V[i][0] + 1, V[i][1], V[i].back());
      upd(1, 1, n, V[i][1] + 1, n, n + 1);
      upd(1, 1, n, 1, V[i][0], V[i][V[i].size() - 2]);
    }
    H[i] = n * 1ll * (n + 1) - get(1, 1, n);
  }
  long long ans = 0;
  for (int i = 1; i <= 2e5; i++) {
    ans += (H[i + 1] - H[i]) * i;
  }
  printf("%lld", ans);
  return 0;
}
