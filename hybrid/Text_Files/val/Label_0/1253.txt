n = int(input())
orig = list(map(int, input().split()))
second = list(map(int, input().split()))
third = list(map(int, input().split()))
orig.sort()
second.sort()
third.sort()
ans1 = orig[-1]
for i in range(len(second)):
	if orig[i] != second[i]:
		ans1 = orig[i]
		break
print(ans1)
ans2 = second[-1]
for i in range(len(third)):
	if second[i] != third[i]:
		ans2 = second[i]
		break
print(ans2)
