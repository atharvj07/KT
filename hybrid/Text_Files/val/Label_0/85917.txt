#include<bits/stdc++.h>
using namespace std;

using ll=long long;
#define REP(i,n) for(ll i=0;i<ll(n);i++)
#define ALL(v) v.begin(),v.end()


const ll MOD=(int)(1e9)+7;
vector<ll> fact;
vector<ll> invfact;

const int SIZE=4*(int)(1e6);

ll powm(ll x,ll k){
  ll res=1;
  while(k){
    if(k&1) res*=x;
    res%=MOD;
    x*=x;
    x%=MOD;
    k>>=1;
  }
  return res;
}

ll mod_inverse(ll x){
  return powm(x,MOD-2);
}

void init(){
  fact.assign(SIZE,0);
  fact[0]=1;
  for(int i=0;i+1<SIZE;i++){
    fact[i+1]=(i+1)*fact[i];
    fact[i+1]%=MOD;
  }
  invfact.assign(SIZE,0);
  for(int i=0;i<SIZE;i++){
    invfact[i]=mod_inverse(fact[i]);
  }
  return;
}


ll comb(ll n,ll k){
  return (((fact[n]*invfact[n-k])%MOD)*invfact[k])%MOD;
}
ll bunpai(ll x,ll n){
  if(n==0){
    assert(x==0);
    return 1;
  }
  return comb(x+n-1,n-1);
}

ll solve(ll a,ll b,ll c,ll sx,ll sy){
  ll n=a+b+c;
  ll res=0;

  for(ll buf=0;buf<=min(sx-a,sy-b);buf++){
    ll bufcomb=bunpai(buf,n);
    if(a==0 && sx-buf-a>0) continue;
    ll acomb=bunpai(sx-buf-a,a);
    if(b==0 && sy-buf-b>0) continue;
    ll bcomb=bunpai(sy-buf-b,b);
    res+=(((bufcomb*acomb)%MOD)*bcomb)%MOD;
    res%=MOD;
  }

  ll ord=fact[n];
  ord*=invfact[a];
  ord%=MOD;
  ord*=invfact[b];
  ord%=MOD;
  ord*=invfact[c];
  ord%=MOD;
  res*=ord;
  res%=MOD;
  return res;

}

int main(){
  init();
  ll a,b,c,sx,sy;
  while(cin>>a>>b>>c>>sx>>sy,!(a==0 && b==0 && c==0 )){
    cout<<solve(a,b,c,sx,sy)<<endl;
  }

  return 0;
}
