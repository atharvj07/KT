def main():
	n = int(input())
	a = list(map(int, input().split()))
	d = {x: 0 for x in range(20)}
	for num in a:
		if num == 0:
			continue
		b = bin(num)
		for j in range(len(b) - 2):
			if b[-j - 1] == '1':
				d[j] += 1
	for xx in range(20):
		if d[xx] == 0:
			del d[xx]
	l = sorted(d.keys(), reverse=True)
	total = 0
	while l:
		new_num = 0
		z = 0
		while z < len(l):
			new_num += 2 ** l[z]
			d[l[z]] -= 1
			if d[l[z]] == 0:
				l.pop(z)
			else:
				z += 1
		total += new_num ** 2
	print(total)
main()
