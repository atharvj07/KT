import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.Arrays;
import java.util.StringTokenizer;

public class I {
	static int inf = (int)1e8;
	static int n;
	static int p;
	static int s;
	static Pair[] arr;
	static int[] sum;
	static int[][] dp;
	static StringBuilder sb1;
	static StringBuilder sb2;
	static int cnt1 = 1;
	static int cnt2 = 1;
	public static void main(String[] args) throws IOException {
		BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
		PrintWriter pw=new PrintWriter(System.out,true);
		
		StringTokenizer st = new StringTokenizer(br.readLine());
		n =Integer.parseInt(st.nextToken());
		p = Integer.parseInt(st.nextToken());
		s = Integer.parseInt(st.nextToken());
		
		dp = new int[n+5][p+5];
		for (int i = 0; i < dp.length; i++) {
			Arrays.fill(dp[i], -1);
		}
		st = new StringTokenizer(br.readLine());
		int[] prog = new int[n];
		for (int i = 0; i < prog.length; i++) 
			prog[i] = Integer.parseInt(st.nextToken());
		
		st = new StringTokenizer(br.readLine());
		int[] spor = new int[n];
		for (int i = 0; i < spor.length; i++)
			spor[i] = Integer.parseInt(st.nextToken());
			
		arr = new Pair[n];
		for (int i = 0; i < spor.length; i++)
			arr[i] = new Pair(i+1, prog[i], spor[i]);
		
		Arrays.sort(arr);
		sum = new int[n+1];
		for (int i = 1; i <= arr.length; i++) {
			sum[i] = sum[i-1] + arr[i-1].b;
		}
		
		sb1 = new StringBuilder("");
		sb2 = new StringBuilder("");
		pw.println(solve(0, 0));
		printSol(0, 0);
		pw.println(sb1);
		pw.println(sb2);
		pw.flush();
		pw.close();
	}
	
	private static void printSol(int ind, int num) 
	{
		if(num == p)
		{
			int have = ind-num;
			int rem = Math.max(0, s-have);
			for (int i = 0; i < rem; i++) {
				sb2.append(arr[i+ind].id);
				if(i!=rem)
					sb2.append(" ");
			}
			return;
		}
		if(ind >= n)
			return;
		
		if(dp[ind][num] == arr[ind].a + dp[ind+1][num+1])
		{
			sb1.append(arr[ind].id);
			if(cnt1++ != p)
				sb1.append(" ");
			printSol(ind+1, num+1);
		}
		else 
			if(dp[ind][num] == arr[ind].b + dp[ind+1][num])
			{
				sb2.append(arr[ind].id);
				if(cnt2++ != s)
					sb2.append(" ");
				printSol(ind+1, num);
			}
			else
			{
				printSol(ind+1, num);
			}
			
		
//		return dp[ind][num] = Math.max(sol1, sol2);
	}

	public static int solve(int ind, int num)
	{
		if(num == p)
		{
			int have = ind-num;
			int rem = Math.max(0, s-have);
			return dp[ind][num] = sum[ind+rem]-sum[ind];
		}
		if(ind >= n)
			return dp[ind][num] = -inf;
		
		if(dp[ind][num] != -1)
			return dp[ind][num];
		int sol1 = arr[ind].a + solve(ind+1, num+1);
		int sol2 = 0;
		if(ind-num < s)
			sol2 = arr[ind].b + solve(ind+1, num);
		else
			sol2 = solve(ind+1, num);
		
		return dp[ind][num] = Math.max(sol1, sol2);
	}
	
	static class Pair implements Comparable<Pair>
	{
		int id, a, b;
		public Pair(int id, int a, int b)
		{
			this.id = id;
			this.a = a;
			this.b = b;
		}
		@Override
		public int compareTo(Pair arg0) {
			return arg0.b - this.b;
		}
		
		public String toString()
		{
			return  a + " " + b;
		}
	}
}
