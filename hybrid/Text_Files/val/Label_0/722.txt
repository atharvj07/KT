def move(ct, cf, dt, df):
	global a, b
	if ct == dt:
		if cf == df:
			return (ct, cf, 0)
		else:
			move = df - cf
			cf = df
	elif cf >= a and cf <= b:
		move = dt - ct
		ct = dt
	elif cf < a:
		move = a - cf
		cf = a
	elif cf > b:
		move = b - cf
		cf = b
	return (ct, cf, abs(move))
(n, h, a, b, k) = map(int, input().split())
for i in range(k):
	counter = 0
	(ta, fa, tb, fb) = map(int, input().split())
	while True:
		(ta, fa, nm) = move(ta, fa, tb, fb)
		counter += nm
		if nm == 0:
			break
	print(counter)
