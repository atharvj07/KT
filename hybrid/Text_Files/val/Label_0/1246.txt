n = int(input())
a = input().split()
b = input().split()
if n == 2:
	print('YES')
else:
	max_cap = max2_cap = 0
	tot_cola = 0
	for i in range(n):
		ai = int(a[i])
		bi = int(b[i])
		if bi > max_cap:
			if max_cap > max2_cap:
				max2_cap = max_cap
			max_cap = bi
		elif bi > max2_cap:
			max2_cap = bi
		tot_cola += ai
	print('YES' if tot_cola <= max_cap + max2_cap else 'NO')
