#include <bits/stdc++.h>
using namespace std;
long i, n;
int main() {
  scanf("%ld", &n);
  if (n % 2 != 0) {
    for (i = 0; i < n - 1; i++) printf("%ld ", i);
    printf("%ld\n", n - 1);
    for (i = 1; i < n; i++) printf("%ld ", i);
    printf("0\n");
    printf("%ld", 1 % n);
    for (i = 1; i < n; i++) printf(" %ld", (2 * i + 1) % n);
    printf("\n");
  } else
    printf("-1\n");
  return 0;
}
