k = int(input())
factor = False
vowels = ['a', 'e', 'i', 'o', 'u']
r = 0
c = 0
answer = list()
for r in range(5, 101):
	c = k / r
	if c.is_integer() and c >= 5:
		c = k // r
		factor = True
		break
if not factor:
	print(-1)
	quit()
for shift in range(r):
	for x in range(c):
		answer.append(vowels[(x + shift) % 5])
print(''.join(answer))
