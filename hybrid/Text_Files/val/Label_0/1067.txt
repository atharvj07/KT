import sys
input = sys.stdin.readline
from collections import *
(n, m, k) = map(int, input().split())
a = list(map(int, input().split()))
t = [0] * (10 ** 6 + 100)
for ai in a:
	t[ai] = 1
now = 0
ans = 0
for i in range(10 ** 6 + 100):
	now += t[i]
	if i >= m:
		now -= t[i - m]
	if now == k:
		t[i] = 0
		now -= 1
		ans += 1
print(ans)
