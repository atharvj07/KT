import java.io.*;
import java.util.*;
import java.math.*;

public class Main {
    private BufferedReader in;
    private PrintWriter out;
    private StringTokenizer st;
    private Random rnd;
    
    boolean[][] g;
    int n;
    int[] status;
    int[] paths;
    boolean finded = false;
    
    public void dfs(int u) {
        if(finded) return;
        
        status[u] = 1;
        
        for(int v = 0; v < n && !finded; v++) {
            if(!g[u][v]) continue;
            
            if(status[v] == 0) {
                paths[v] = u;
                dfs(v);
            } else if(status[v] == 1) {
                out.println((u + 1) + " " + (v + 1) + " " + (paths[u] + 1));
                finded = true;
            }
        }
        
        status[u] = 2;
    }
    
    public void solve() throws IOException {
        n = nextInt();
        status = new int[n];
        g = new boolean[n][n];
        paths = new int[n];
        
        for(int i = 0; i < n; i++) {
            String s = nextToken();
            
            for(int j = 0; j < n; j++) {
                g[i][j] = (s.charAt(j) == '1');
            }
        }
        
        for(int i = 0; i < n; i++) {
            if(status[i] == 0) {
                dfs(i);
            }
        }
        
        if(!finded) out.println(-1);
    }
        
    public static void main(String[] args) {
        new Main().run();
    } 
    
    public void run() {
        try {
            //in = new BufferedReader(new FileReader("balls.in"));
            //out = new PrintWriter(new FileWriter("balls.out"));
            
            in = new BufferedReader(new InputStreamReader((System.in)));
            out = new PrintWriter(System.out);
            
            st = null;
            rnd = new Random();
            
            solve();
            
            out.close();
        } catch(IOException e) {
            e.printStackTrace();
        }
    }
    
    private String nextToken() throws IOException, NullPointerException {
        while(st == null || !st.hasMoreTokens()) {
            st = new StringTokenizer(in.readLine());
        }
        
        return st.nextToken();
    }
    
    private int nextInt() throws IOException {
        return Integer.parseInt(nextToken());
    }
    
    private long nextLong() throws IOException {
        return Long.parseLong(nextToken());
    }
    
    private double nextDouble() throws IOException {
        return Double.parseDouble(nextToken());
    }

}
