#include <bits/stdc++.h>
using namespace std;
long long N, A, Cf, Cm, m;
vector<pair<long long, long long> > V;
long long acm[100005];
long long need[100005];
long long rpta[100005];
long long f(long long pos, long long able) {
  if (pos == N) return A;
  able -= need[pos];
  able /= (N - pos);
  return min(A, V[pos].first + able);
}
int main() {
  cin >> N >> A >> Cf >> Cm >> m;
  V.resize(N);
  for (int i = (0); i < (N); i++) {
    cin >> V[i].first;
    V[i].second = i;
  }
  sort((V).rbegin(), (V).rend());
  acm[N] = 0;
  long long L = -1, R;
  for (int i = N - 1; i >= 0; i--) {
    acm[i] = acm[i + 1] + V[i].first;
    need[i] = (N - i) * V[i].first - acm[i];
    if (need[i] <= m) R = i;
  }
  for (int i = (0); i < (N); i++)
    if (V[i].first == A) L = i;
  pair<long long, pair<long long, long long> > ans =
      make_pair((L + 1) * Cf + f(R, m) * Cm, make_pair(L, f(R, m)));
  long long M = m;
  for (int i = (0); i < (N); i++) {
    L = i;
    M -= (A - V[i].first);
    if (M < 0) break;
    R = max(R, L + 1);
    while (R < N and need[R] > M) R++;
    ans =
        max(ans, make_pair((L + 1) * Cf + f(R, M) * Cm, make_pair(L, f(R, M))));
  }
  L = ans.second.first;
  R = ans.second.second;
  for (int i = (0); i < (N); i++) {
    if (i <= L)
      rpta[V[i].second] = A;
    else
      rpta[V[i].second] = max(V[i].first, R);
  }
  cout << ans.first << endl;
  for (int i = (0); i < (N); i++) cout << rpta[i] << " ";
}
