t = int(input())
while t > 0:
	t = t - 1
	dict = {}
	n = int(input())
	L = input().split()
	Max = 0
	for x in L:
		if x not in dict.keys():
			dict[x] = 1
		else:
			dict[x] = dict[x] + 1
		if dict[x] > Max:
			Max = dict[x]
	if 2 * Max > n:
		print(2 * Max - n)
	elif n % 2 == 0:
		print(0)
	else:
		print(1)
