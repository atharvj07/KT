#include <bits/stdc++.h>
using namespace std;
int n, x = 0, y = 0;
char s[20][20], c[20][20];
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) {
      cin >> s[i][j];
      if (s[i][j] == 'X') x++;
    }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) {
      cin >> c[i][j];
      if (c[i][j] == 'X') y++;
    }
  if (x != y) {
    cout << "No" << endl;
    return 0;
  }
  x = 0;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) {
      if (s[i][j] == c[i][j])
        x++;
      else
        break;
    }
  if (x == n * n) {
    cout << "Yes" << endl;
    return 0;
  } else {
    x = 0;
  }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) {
      if (s[i][j] == c[i][n - j + 1])
        x++;
      else
        break;
    }
  if (x == n * n) {
    cout << "Yes" << endl;
    return 0;
  } else {
    x = 0;
  }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) {
      if (s[i][j] == c[n - i + 1][j])
        x++;
      else
        break;
    }
  if (x == n * n) {
    cout << "Yes" << endl;
    return 0;
  } else {
    x = 0;
  }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) {
      if (s[i][j] == c[n - i + 1][n - j + 1])
        x++;
      else
        break;
    }
  if (x == n * n) {
    cout << "Yes" << endl;
    return 0;
  } else {
    x = 0;
  }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) {
      if (s[i][j] == c[j][i])
        x++;
      else
        break;
    }
  if (x == n * n) {
    cout << "Yes" << endl;
    return 0;
  } else {
    x = 0;
  }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) {
      if (s[i][j] == c[n - j + 1][i])
        x++;
      else
        break;
    }
  if (x == n * n) {
    cout << "Yes" << endl;
    return 0;
  } else {
    x = 0;
  }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) {
      if (s[i][j] == c[j][n - i + 1])
        x++;
      else
        break;
    }
  if (x == n * n) {
    cout << "Yes" << endl;
    return 0;
  } else {
    x = 0;
  }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++) {
      if (s[i][j] == c[n - j + 1][n - i + 1])
        x++;
      else
        break;
    }
  if (x == n * n) {
    cout << "Yes" << endl;
    return 0;
  } else {
    x = 0;
  }
  cout << "No" << endl;
  return 0;
}
