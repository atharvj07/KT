#include <bits/stdc++.h>
using namespace std;
int dp[505][505];
int n, m, k, i, j, arr[505];
int f(int bas, int son) {
  if (bas >= son) return 1;
  int &r = dp[bas][son];
  if (r != -1) return r;
  r = f(bas + 1, son);
  for (int i = bas; i < son; i++)
    if (arr[bas] < arr[i + 1])
      r = (r + (long long int)f(bas + 1, i) * f(i + 1, son) % 1000000007) %
          1000000007;
  return r;
}
int main() {
  memset(dp, -1, sizeof dp);
  cin >> n;
  for (i = 1; i <= n; i++) scanf("%d", arr + i);
  cout << f(2, n) << '\n';
}
