#include <bits/stdc++.h>
using namespace std;
void getint(int &v) {
  char ch, fu = 0;
  for (ch = '*'; (ch < '0' || ch > '9') && ch != '-'; ch = getchar())
    ;
  if (ch == '-') fu = 1, ch = getchar();
  for (v = 0; ch >= '0' && ch <= '9'; ch = getchar()) v = v * 10 + ch - '0';
  if (fu) v = -v;
}
int n;
char a[66][66], b[66][66], c[66][66], d[66][66];
bool same() {
  for (int i = 1; i <= n; ++i)
    for (int j = 1; j <= n; ++j)
      if (c[i][j] != b[i][j]) return 0;
  return 1;
}
bool ck() {
  for (int i = 1; i <= n; ++i)
    for (int j = 1; j <= n; ++j) c[i][j] = a[i][j];
  for (int D = 0; D <= 3; ++D) {
    if (same()) return 1;
    for (int i = 1; i <= n; ++i)
      for (int j = 1; j <= n; ++j) d[j][n - i + 1] = c[i][j];
    for (int i = 1; i <= n; ++i)
      for (int j = 1; j <= n; ++j) c[i][j] = d[i][j];
  }
  return 0;
}
void zy() {
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= n / 2; ++j) swap(a[i][j], a[i][n - j + 1]);
  }
}
void sx() {
  for (int j = 1; j <= n; ++j) {
    for (int i = 1; i <= n / 2; ++i) swap(a[i][j], a[n - i + 1][j]);
  }
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; ++i) scanf("%s", a[i] + 1);
  for (int i = 1; i <= n; ++i) scanf("%s", b[i] + 1);
  if (ck()) return puts("Yes"), 0;
  zy();
  if (ck()) return puts("Yes"), 0;
  sx();
  if (ck()) return puts("Yes"), 0;
  zy();
  if (ck()) return puts("Yes"), 0;
  puts("No");
  return 0;
}
