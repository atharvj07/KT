def solution_e1(l):
	result = []
	aux = [l[0]]
	tmpmin = l[0]
	for x in l[1:]:
		if x < tmpmin:
			tmpmin = x
			aux.append(x)
		else:
			result.append(x)
	return aux[::-1] + result
t = int(input())
for _ in range(t):
	try:
		ln = int(input())
		l = input()
		l = [int(x) for x in l.split(' ')]
		s = solution_e1(l)
		print(' '.join([str(c) for c in s]))
	except Exception as e:
		print(repr(e))
