#include<bits/stdc++.h>
using namespace std;
int main()
{
  int n,i,j,k;cin>>n>>k;
 	int A[n];
  vector<int> dp(n+k,1000000007);
  for( i=0;i<n;i++)
    cin>>A[i];
  dp[0]=0;
  for(i=0;i<n;i++)
  {	
    for(j=i+1;j<=i+k;j++)
    {
     	if(j<n)
		dp[j]=min(dp[j],dp[i]+abs(A[j]-A[i]));
    }
  }
  cout<<dp[n-1];
}
