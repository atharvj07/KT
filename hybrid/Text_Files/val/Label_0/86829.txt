import java.math.BigInteger;
import java.util.*;
import java.util.stream.Stream;

import static java.lang.Math.max;
import static java.lang.Math.min;

public class Main {
    private static Scanner in = new Scanner(System.in);

    public static void main(String[] args) {
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            toDo();
        }
    }



    private static void toDo() {
        int n = in.nextInt();
        int m = in.nextInt();
        int k = in.nextInt();
        int[] lst = new int[n];
        for (int o = 0; o < n; o++){
            lst[o] = in.nextInt();
        }
        int i = min(k, m - 1);
        int right1;
        int mn = Integer.MIN_VALUE;
        int cur_min;
        for (int left = 0; left <= i; left++){
            int cur_cur_mx =  max(lst[left], lst[lst.length - m + left]);
            cur_min = cur_cur_mx;
            for (int left1 = left; left1 <= left + m - 1 - i; left1++){
                right1 = lst.length - m + 1 + left1;
                cur_cur_mx = max(lst[left1], lst[right1 - 1]);
                cur_min = min(cur_cur_mx, cur_min);

            }

            mn = max(cur_min, mn);
        }
        System.out.println(mn);
    }
}
