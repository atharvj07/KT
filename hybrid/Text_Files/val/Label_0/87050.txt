        static class TaskE {
            int n,a[];
            int seg[][];
            void make(int n,int s,int e){
                if(s==e){
                    seg[n]=new int[1];
                    seg[n][0]=a[s];
                    return;
                }
                int m=(s+e)>>1;
                make(2*n+1,s,m);make(2*n+2,m+1,e);
                seg[n]=new int[e-s+1];
                merge(n,2*n+1,2*n+2);
            }
            void merge(int n,int n1,int n2){
                int l1=seg[n1].length,l2=seg[n2].length;
                int p1=0,p2=0;
                int p=0;
                while(p1<l1&&p2<l2){
                    if(seg[n1][p1]<seg[n2][p2]){
                        seg[n][p]=seg[n1][p1];
                        ++p;++p1;
                    }else{
                        seg[n][p]=seg[n2][p2];
                        ++p;++p2;
                    }
                }
                while(p1<l1){
                    seg[n][p]=seg[n1][p1];
                    ++p;++p1;
                }
                while(p2<l2){
                    seg[n][p]=seg[n2][p2];
                    ++p;++p2;
                }
            }
            int query(int n,int s,int e,int l,int r,int val){
                if(l>e||r<s)return 0;
                if(s>=l&&e<=r)return bs(n,val);
                int m=(s+e)>>1;
                return query(2*n+1,s,m,l,r,val)+query(2*n+2,m+1,e,l,r,val);
            }
            int bs(int n,int val){
                int ans=0;
                int l=0,r=seg[n].length-1,max=r;
                while(l<=r){
                    int m=(l+r)>>1;
                    if(seg[n][m]>=val){
                        ans=max-m+1;r=m-1;
                    }
                    else l=m+1;
                }return ans;
            }
        public void solve(int testNumber, InputReader in, PrintWriter out) {
            n=in.nextInt();
            a=new int[n];
            seg=new int[4*n][];
            for(int i=0;i<n;i++)a[i]=in.nextInt();
            make(0,0,n-1);
            long ans=0;
            for(int i=0;i<n-1;i++){
                int l=i+1,r=Math.min(n-1,a[i]-1);
                if(r<l)continue;
                ans+=query(0,0,n-1,l,r,i+1);
            }out.print(ans);
        }

    static class InputReader {
        BufferedReader br;
        StringTokenizer st;
        public InputReader(InputStream stream) {
            br = new BufferedReader(new InputStreamReader(stream));
            st = null;
        }

        String next() {
            while (st == null || !st.hasMoreTokens()) {
                String s = null;
                try {
                    s = br.readLine();
                } catch (IOException e) {
                    e.printStackTrace();
                }
                if (s == null)
                    return null;
                st = new StringTokenizer(s);
            }
            return st.nextToken();
        }

        boolean hasMoreTokens() {
            while (st == null || !st.hasMoreTokens()) {
                String s = null;
                try {
                    s = br.readLine();
                } catch (IOException e) {
                    e.printStackTrace();
                }
                if (s == null)
                    return false;
                st = new StringTokenizer(s);
            }
            return true;
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }
        public long nextLong() {
            return Long.parseLong(next());
        }
        public double nextDouble() {
            return Double.parseDouble(next());
        }
    }
