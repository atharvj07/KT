#include <bits/stdc++.h>
int main() {
  int a, b;
  while (scanf("%d%d", &a, &b) != EOF) {
    int t = a / b;
    int m = a;
    if (a % b == 0) {
      int i;
      for (i = 1; i < b; i++) printf("%d ", t);
      printf("%d\n", t);
      continue;
    }
    while (b) {
      b--;
      printf("%d ", t);
      m -= t;
      if (m % b == 0) break;
    }
    int i;
    for (i = 1; i < b; i++) printf("%d ", t + 1);
    printf("%d\n", t + 1);
  }
  return 0;
}
