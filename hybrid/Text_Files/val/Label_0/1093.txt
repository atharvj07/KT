def func(s):
	st = ''
	for i in s:
		if i == '1':
			st += '0'
		else:
			st += '1'
	return st[::-1]
t = int(input())
while t:
	n = int(input())
	s1 = input()
	s2 = input()
	i = n - 1
	l = []
	while i >= 0:
		if s1[i] != s2[i]:
			if s1[0] == s2[i]:
				l.append(1)
				s1 = func(s1[1:i + 1])
				l.append(i + 1)
			else:
				s1 = func(s1[:i + 1])
				l.append(i + 1)
		i -= 1
	print(len(l), *l)
	t -= 1
