import sys
import bisect
import math
from collections import deque
input = lambda : sys.stdin.readline().strip()
mod = 10 ** 9 + 7

def solve():
	(n, k) = map(int, input().split())
	bits = 0
	for _t in range(k):
		(l, r, x) = map(int, input().split())
		bits |= x
	return bits * pow(2, n - 1, mod) % mod
t = 1
t = int(input())
for _ in range(t):
	print(solve())
