#include <iostream>
#include<string>
#include<algorithm>
using namespace std;
int n, m, ans = 999999, w[55], b[55], r[55];
string k;
int main() {
	int i, j;
	cin >> n >> m;
	for (i = 1; i <= n; i++) {
		w[i] = w[i - 1];
		b[i] = b[i - 1];
		r[i] = r[i - 1];
		cin >> k;
		for (j = 0; j < m; j++) {
			if (k[j] != 'W') w[i]++;
			if (k[j] != 'B') b[i]++;
			if (k[j] != 'R') r[i]++;
		}
	}
	for (i = 1; i <= n - 2; i++) for (j = i + 1; j <= n - 1; j++) {
		ans = min(ans, w[i] + b[j] - b[i] + r[n] - r[j]);
	}
	cout << ans << endl;
}
