import sys
input = sys.stdin.readline

def I():
	return input().strip()

def II():
	return int(input().strip())

def LI():
	return [*map(int, input().strip().split())]
import string, math, time, functools, random, fractions
from heapq import heappush, heappop, heapify
from bisect import bisect_left, bisect_right
from collections import deque, defaultdict, Counter, OrderedDict
from itertools import permutations, combinations, groupby
from operator import itemgetter
(num, hour, left, right) = LI()
arr = LI()
presum = [0] * num
presum[0] = arr[0]
for i in range(1, num):
	presum[i] = arr[i] + presum[i - 1]
dp = [[0 for i in range(num + 1)] for j in range(num + 1)]
dp[0][0] = 0
for i in range(1, num + 1):
	dp[i][0] = dp[i - 1][0] + (1 if left <= presum[i - 1] % hour <= right else 0)
for i in range(1, num + 1):
	for j in range(1, i + 1):
		dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - 1])
		dp[i][j] += 1 if left <= (presum[i - 1] - j + hour) % hour <= right else 0
mex = 0
for i in range(num + 1):
	mex = max(mex, dp[-1][i])
print(mex)
