#include <bits/stdc++.h>
using namespace std;
double pi = acos(-1.);
int r, v, R;
double L;
double binary(double l, double r, double len) {
  int t = 70;
  while (t--) {
    double m = (l + r) / 2.;
    double y = R * (m - sin(m));
    double x = pi * R - y;
    if (x < len)
      r = m;
    else
      l = m;
  }
  return l;
}
void solve() {
  int n;
  double s, f, ans = 0;
  double len;
  cin >> s >> f;
  len = f - s;
  n = len / (L);
  ans = 1. * n * L / v;
  len -= n * L;
  len /= 2;
  double al = pi - binary(0, pi, len);
  ans = ans + 1.0 * al / pi * L / v;
  printf("%.7lf\n", ans);
}
int main() {
  int t;
  cin >> t;
  cin >> r >> v;
  R = r;
  L = 2. * pi * r;
  while (t--) {
    solve();
  }
  return 0;
}
