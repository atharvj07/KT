#include <bits/stdc++.h>
#define ll long long
#define INF 1000000005
#define MOD 1000000007
#define rep(i,n) for(int i=0;i<n;++i)

using namespace std;

typedef pair<ll,int>P;

struct edge{
	int to;   //?????????
	int cost; //?????????
};

const int MAX_N = 100005;

int n,m,c;
ll d[MAX_N];
vector<edge> G[MAX_N];
set<int> K;
multimap<ll,int> mp;

void dijkstra(int s)
{
	priority_queue<P,vector<P>,greater<P> > que;
	d[s] = 0LL;
	que.push(P(0LL,s));
	while(!que.empty()){
		P p = que.top();
		que.pop();
		int v = p.second;
		if(d[v] < p.first) continue;
		vector<edge>::iterator it = G[v].begin();
		while(it != G[v].end()){
			if(d[it->to] > d[v] + it->cost){
				d[it->to] = d[v] + it->cost;
				que.push(P(d[it->to],it->to));
			}
			it++;
		}
	}
}

int main()
{
	ll esum = 0LL;
	scanf("%d%d%d",&n,&m,&c);
	rep(i,m){
		int x,y,z;
		scanf("%d%d%d",&x,&y,&z);
		edge e1,e2;
		e1.to=y-1,e1.cost=z;
		e2.to=x-1,e2.cost=z;
		G[x-1].push_back(e1);
		G[y-1].push_back(e2);
		esum += z;
	}
	fill(d,d+n,(1LL << 60));
	dijkstra(0);
	rep(i,n){
		mp.emplace(d[i],i);
	}
	sort(d,d+n);
	int id=0;
	ll ans=(1LL << 60);
	while(id<n){
		auto p = mp.equal_range(d[id]);
		for(auto it=p.first;it!=p.second;++it){
			int u = it->second;
			rep(k,G[u].size()){
				if(K.find(G[u][k].to)!=K.end()){
					esum -= G[u][k].cost;
				}
			}
			K.insert(u);
		}
		ans = min(ans,c*d[id]+esum);
		id += (int)mp.count(d[id]);
	}
	printf("%lld\n",ans);
	return 0;
}
