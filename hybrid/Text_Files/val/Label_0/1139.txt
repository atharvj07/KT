import sys
import math
input = sys.stdin.readline
t = 1
for _ in range(t):
	(n, m, k) = map(int, input().split())
	for i in range(k):
		(sx, sy) = map(int, input().split())
	for i in range(k):
		(fx, fy) = map(int, input().split())
	s = ''
	for i in range(m - 1):
		s += 'L'
	for i in range(n - 1):
		s += 'U'
	cnt = 0
	ch = 0
	while cnt < n * m:
		if ch % 2 == 0:
			for i in range(1, m):
				s += 'R'
				sy += 1
				cnt += 1
		else:
			for i in range(1, m):
				sy -= 1
				s += 'L'
				cnt += 1
		s += 'D'
		sx += 1
		cnt += 1
		ch += 1
	print(len(s))
	print(s)
