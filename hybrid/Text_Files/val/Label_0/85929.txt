#include <bits/stdc++.h>
using namespace std;

const int maxn = 1e5 + 10;
vector<int> mp[maxn];
int n, sz[maxn], dep[maxn], fa[maxn];
int dfs(int u, int f, int d){
    dep[u] = d;
    fa[u] = f;
    sz[u] = 1;
    for(auto &v:mp[u])if(v != f){
        int ret = dfs(v, u, d + 1);
        sz[u] += ret;
    }
    return sz[u];
}
int main(){
    ios::sync_with_stdio(false);
    cin >> n;
    for(int i = 1; i < n; ++i){
        int a, b;
        cin >> a >> b;
        mp[a].push_back(b);
        mp[b].push_back(a);
    }
    dfs(1, 0, 1);
    int sum = 0, len = dep[n] / 2;
    for(int i = 1, p = n; i <= len; ++i){
        sum = sz[p];
        p = fa[p];
    }
    if(sum >= n - sum)
        cout << "Snuke" << endl;
    else
        cout << "Fennec" << endl;
    return 0;
}
