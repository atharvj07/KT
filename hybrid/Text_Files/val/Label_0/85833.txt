void addmod(int &x, int y, int mod = 1000000007) {
  assert(y >= 0);
  x += y;
  if (x >= mod) x -= mod;
  assert(x >= 0 && x < mod);
}

long long fastPow(long long x, long long y, int mod = 1000000007) {
  long long ans = 1;
  while (y > 0) {
    if (y & 1) ans = (x * ans) % mod;
    x = x * x % mod;
    y >>= 1;
  }
  return ans;
}

bool ckX(int vx, int T) {
  vector<pair<int, int> > vp;
  for (int(i) = 1; (i) <= (int)(k); (i)++) {
    if (x[i] - T <= vx && vx <= x[i] + T) vp.push_back({y[i] - T, y[i] + T});
  }
  sort(vp.begin(), vp.end());
  int pre = 0;
  for (int(i) = 0; (i) < (int)(vp.size()); (i)++) {
    if (vp[i].first > pre + 1) return 0;
    pre = max(pre, vp[i].second);
  }
  return pre >= m;
}
bool ckY(int vy, int T) {

bool ck(int T) {
  cntV = 0;
  cntH = 0;
  vc[++cntV] = 1;
  hc[++cntH] = 1;
  for (int(i) = 1; (i) <= (int)(k); (i)++) {
    vc[++cntV] = max(x[i] - T - 1, 1);
    vc[++cntV] = min(x[i] + T + 1, n);
    hc[++cntH] = max(y[i] - T - 1, 1);
    hc[++cntH] = min(y[i] + T + 1, m);
  }
  vc[++cntV] = n;
  hc[++cntH] = m;
  U = (1 << 30);
  L = (1 << 30);
  R = 0;
  D = 0;
  for (int(i) = 1; (i) <= (int)(cntV); (i)++) {
    if (ckX(vc[i], T)) continue;
    L = min(L, vc[i]);
    R = max(R, vc[i]);
  }
  for (int(i) = 1; (i) <= (int)(cntH); (i)++) {
    if (ckY(hc[i], T)) continue;
    U = min(U, hc[i]);
    D = max(D, hc[i]);
  }
  if (R == 0 && D == 0) return 1;
  if (R == 0) {
    L = 1;
    R = n;
  }
  if (D == 0) {
    U = 1;
    D = m;
  }
  if ((R - L + 1 <= T * 2 + 1) && (D - U + 1 <= T * 2 + 1)) return 1;
  return 0;
}
