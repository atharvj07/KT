(n, m) = map(int, input().split())
a = [[0 for i in range(m)] for j in range(n)]
sum1 = 0
fg = 0
for i in range(n):
	a[i] = list(map(int, input().split()))
for i in range(n - 1, 0, -1):
	for j in range(m - 1, 0, -1):
		if a[i][j] == 0:
			a[i][j] = min(a[i + 1][j], a[i][j + 1]) - 1
		if a[i][j] <= a[i - 1][j] or a[i][j] <= a[i][j - 1]:
			fg = 1
			break
if fg != 1:
	for i in range(n):
		for j in range(m):
			sum1 += a[i][j]
	print(sum1)
else:
	print(-1)
