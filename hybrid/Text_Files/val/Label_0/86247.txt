#include <bits/stdc++.h>
using namespace std;
int t;
int maxi = 0;
int blocked1 = -1, blocked2 = -1, blocked3 = -1, blocked4 = -1, blocked = 0;
int br[2000006];
int ch[2000006];
int expected(int a) {
  if (blocked == 0) return 4 * a;
  if (blocked == 1) return 2 * blocked1 + 2 * a + 1;
  return blocked1 + blocked2 + a + 1 + max(0, blocked2 - a + blocked1 + 1);
}
int a0;
bool check(int n, int m, int x, int y) {
  int nll = 0;
  for (int i = 0; i <= maxi; i++) ch[i] = br[i];
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      int p = abs(i - x) + abs(j - y);
      ch[p]--;
      if (ch[p] == 0) nll++;
      if (ch[p] < 0) return 0;
    }
  }
  if (nll == maxi + 1)
    return 1;
  else
    return 0;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> t;
  for (int i = 0; i < t; i++) {
    int a;
    cin >> a;
    if (i == 0) a0 = a;
    br[a]++;
    maxi = max(maxi, a);
  }
  int i = 1;
  while (br[i] == expected(i)) i++;
  int x = expected(i) - br[i];
  if (x == 1) {
    blocked = 1;
    blocked1 = i - 1;
  } else if (x == 3 || x == 4) {
    if (t % (i * 2 - 1) == 0 && check(i * 2 - 1, t / (i * 2 - 1), i, i))
      cout << i * 2 - 1 << ' ' << t / (i * 2 - 1) << '\n'
           << i << ' ' << i << '\n';
    else
      cout << "-1\n";
    return 0;
  } else if (x == 2) {
    blocked = 2;
    blocked1 = i - 1;
    blocked2 = i - 1;
    if (t % (i * 2 - 1) == 0 &&
        check(i * 2 - 1, t / (i * 2 - 1), i, maxi - i + 2)) {
      cout << i * 2 - 1 << ' ' << t / (i * 2 - 1) << '\n'
           << i << ' ' << maxi - i + 2 << '\n';
      return 0;
    }
  } else {
    cout << "-1\n";
    return 0;
  }
  while (br[i] == expected(i)) i++;
  x = expected(i) - br[i] + blocked;
  if (x == 2) {
    blocked = 2;
    blocked2 = i - 1;
    if (t % (blocked1 + blocked2 + 1) == 0 &&
        check(blocked1 + blocked2 + 1, t / (blocked1 + blocked2 + 1),
              blocked1 + 1, maxi - blocked2 + 1)) {
      cout << blocked1 + blocked2 + 1 << ' ' << t / (blocked1 + blocked2 + 1)
           << '\n'
           << blocked1 + 1 << ' ' << maxi - blocked2 + 1 << '\n';
      return 0;
    }
  }
  if (x == 3 || x == 4) {
    int y;
    if (blocked == 2)
      y = blocked1 + 1;
    else
      y = i;
    if (t % (blocked1 + i) == 0 &&
        check(blocked1 + i, t / (blocked1 + i), blocked1 + 1, y))
      cout << blocked1 + i << ' ' << t / (blocked1 + i) << '\n'
           << blocked1 + 1 << ' ' << y << '\n';
    else
      cout << "-1\n";
    return 0;
  }
  if (x > 4 || x < 2) {
    cout << "-1\n";
    return 0;
  }
  while (br[i] == expected(i)) i++;
  x = expected(i) - br[i] + blocked;
  if (x > 4 || x < 3) {
    cout << "-1\n";
    return 0;
  }
  if (t % (blocked1 + i) == 0 &&
      check(blocked1 + i, t / (blocked1 + i), blocked1 + 1, blocked2 + 1))
    cout << blocked1 + i << ' ' << t / (blocked1 + i) << '\n'
         << blocked1 + 1 << ' ' << blocked2 + 1 << '\n';
  else if (t % (blocked2 + i) == 0 &&
           check(blocked2 + i, t / (blocked2 + i), blocked2 + 1, blocked1 + 1))
    cout << blocked2 + i << ' ' << t / (blocked2 + i) << '\n'
         << blocked2 + 1 << ' ' << blocked1 + 1 << '\n';
  else
    cout << "-1\n";
  return 0;
}
