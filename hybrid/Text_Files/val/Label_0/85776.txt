#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main(int argc, char *argv[]) {
  ios::sync_with_stdio(false);
	cin.tie(nullptr);
	cout.tie(nullptr);

  int n,k;cin>>n>>k;
  vector<int> a(n);
  vector<ll> s(n + 1);
  for(int i=0;i<n;i++){
    cin>>a[i];
    s[i + 1] = (s[i] + a[i] - 1 + k) % k;
  }
  ll ans=0;
  map<ll,ll> mp;
  for(int i=0;i<=n;i++){
    if(i-k>=0)mp[s[i-k]]--;
    ans += mp[s[i]];
    mp[s[i]]++;
  }
  cout<<ans<<endl;

  return 0;
}
