#include<bits/stdc++.h>
typedef long long int ll;
typedef unsigned long long int ull;
#define BIG_NUM 2000000000
#define MOD 1000000007
#define EPS 0.000000001
using namespace std;


#define NUM 301

int M;
int table[6];
int start,goal,command[NUM];
int min_dist[NUM],to_dist[NUM];

bool rangeCheck(int loc){
	if(loc >= 0 && loc <= M-1){
		return true;
	}else{
		return false;
	}
}

int recursive(int loc,int sum_dist){

	if(min_dist[loc] != -1){

		if(min_dist[loc] == BIG_NUM){
			return BIG_NUM;
		}else{
			return min_dist[loc]+sum_dist;
		}
	}

	int ret = BIG_NUM,next_loc = 0;

	for(int i = 0; i < 6; i++){

		//右方向
		next_loc = loc+table[i];
		if(rangeCheck(next_loc) == true){
			next_loc += command[next_loc];
			if(to_dist[next_loc] > sum_dist+1){
				to_dist[next_loc] = sum_dist+1;
				ret = min(ret,recursive(next_loc,sum_dist+1));
			}
		}
		//左方向
		next_loc = loc-table[i];
		if(rangeCheck(next_loc) == true){
			next_loc += command[next_loc];
			if(to_dist[next_loc] > sum_dist+1){
				to_dist[next_loc] = sum_dist+1;
				ret = min(ret,recursive(next_loc,sum_dist+1));
			}
		}
	}

	return ret;
}


int main(){

	scanf("%d",&M);

	for(int i = 0; i < 6; i++)scanf("%d",&table[i]);

	scanf("%d %d",&start,&goal);
	start--;
	goal--;

	for(int i = 0; i < M; i++){
		scanf("%d",&command[i]);
	}

	for(int i = 0; i < M; i++){
		min_dist[i] = -1;
	}
	min_dist[goal] = 0;

	for(int i = 0; i < M; i++){
		if(i == goal)continue;
		for(int k = 0; k < M; k++)to_dist[k] = BIG_NUM; //そこへたどり着くまでのコスト

		to_dist[i] = 0;

		min_dist[i] = recursive(i,0);
	}

	int current_loc = start,dice;
	int right_loc,left_loc,right_dist,left_dist;

	while(true){

		scanf("%d",&dice);
		dice--;

		if(!rangeCheck(current_loc+table[dice])){
			right_dist = BIG_NUM;
		}else{
			right_loc = current_loc+table[dice];
			right_loc += command[right_loc];
			right_dist = min_dist[right_loc];
		}

		if(!rangeCheck(current_loc-table[dice])){
			left_dist = BIG_NUM;
		}else{
			left_loc = current_loc-table[dice];
			left_loc += command[left_loc];
			left_dist = min_dist[left_loc];
		}

		if(min_dist[current_loc] <= right_dist && min_dist[current_loc] <= left_dist){ //現在地が最適である場合

			printf("0\n");
			fflush(stdout);

		}else{

			if(left_dist < right_dist){
				printf("-1\n");
				fflush(stdout);
				current_loc = left_loc;
			}else{
				printf("1\n");
				fflush(stdout);
				current_loc = right_loc;
			}
		}

		if(current_loc == goal){
			return 0;
		}
	}

	return 0;
}
