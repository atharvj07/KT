public class f {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long k = in.nextInt();
        int l = in.nextInt();
        if(k*l > n) {
            System.out.println(0);
            return;
        }
        int[] is = new int[n];
        char[] arr = in.next().toCharArray();
        for(int i = 0; i < n; i++) {
            if(Character.isUpperCase(arr[i])) is[i] = 1;
        }
        long min = solve(is, n, k, l);
        for(int i = 0; i < n; i++) is[i] = 1-is[i];
        min = Math.min(min, solve(is, n, k, l));
        System.out.println(min);
    }
    
    static long solve(int[] is, int n, long k, int l) {
        int cost = -1;
        for(int pow = 19; pow >= 0; pow--) {
            int tmp = cost+(1<<pow);
            long[] dp = new long[n+1];
            int[] num = new int[n+1];
            for(int i = 1; i <= n; i++) {
                dp[i] = dp[i-1]+is[i-1];
                num[i] = num[i-1];
                if(i >= l && tmp+dp[i-l] < dp[i]) {
                    dp[i] = tmp+dp[i-l];
                    num[i] = num[i-l]+1;
                }
            }
            if(num[n] > k) {
                cost = tmp;
            }
        }
        cost++;
        long[] dp = new long[n+1];
        for(int i = 1; i <= n; i++) {
            dp[i] = dp[i-1]+is[i-1];
            if(i >= l && cost+dp[i-l] < dp[i]) {
                dp[i] = cost+dp[i-l];
            }
        }
        return dp[n]-k*cost;
    }
}
