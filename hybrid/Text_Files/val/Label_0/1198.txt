def gcdPrime(a, b):
	if b == 0 or a == 0:
		return 0
	if b == 1 or a == 1:
		return 1
	if b > a:
		return gcdPrime(a, b % a)
	return gcdPrime(b, a % b)

def gcdPrimeSet(set, a):
	result = []
	if len(set) >= 4:
		return [x for x in set]
	for i in set:
		if gcdPrime(i, a) == 0:
			result.append(i)
	return result
