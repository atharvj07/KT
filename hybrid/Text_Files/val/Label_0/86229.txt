#include<bits/stdc++.h>
using namespace std;
int N,L;
map<string,int> S;
vector<string> ans;

int main(){
  cin>>N>>L;
  for(int i=0;i<N;i++){
    string str;
    cin>>str;
    string tmp=str;
    reverse(tmp.begin(),tmp.end());
    if( S[tmp]>0 ){
      S[tmp]--;
      ans.push_back( min(tmp,str) );
    }else{
      S[str]++;
    }
  }

  string X="";
  map<string,int> :: iterator it;
  for(it=S.begin();it!=S.end();it++){
    if(it->second==0)continue;
    string str=it->first;
    string tmp=str;
    reverse(str.begin(),str.end());
    if(str==tmp && ( X=="" || X>str) ){
      X=str;
    }
  }
  sort(ans.begin(),ans.end());
  for(int i=0;i<ans.size();i++)cout<<ans[i];
  cout<<X;
  reverse(ans.begin(),ans.end());
  for(int i=0;i<ans.size();i++){
    reverse(ans[i].begin(),ans[i].end());
    cout<<ans[i];
  }
  cout<<endl;
  return 0;
}
