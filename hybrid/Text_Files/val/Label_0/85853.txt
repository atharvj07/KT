#include <bits/stdc++.h>
using namespace std;
const int MAX_N = 2e3 + 5;
int N, M, cnt[MAX_N], bel[MAX_N][MAX_N], del[MAX_N];
bitset<MAX_N> bs[MAX_N];
char str[MAX_N][MAX_N];
void solve() {
  scanf("%d %d", &N, &M);
  for (int i = 1; i <= N; i++) bs[i].reset();
  for (int i = 1; i <= M; i++) {
    scanf("%s", str[i] + 1), cnt[i] = 0;
    for (int j = 1; j <= N; j++)
      if (str[i][j] == '1') cnt[i]++;
    if (cnt[i] == 1) {
      cnt[i] = 1;
      continue;
    }
    for (int j = 1; j <= N; j++)
      if (str[i][j] == '1') bs[j][i] = 1;
  }
  vector<pair<int, int> > ans;
  queue<pair<int, int> > que;
  for (int i = 1; i <= N; i++)
    for (int j = 1; j <= N; j++)
      if (i != j && ((bs[i] & bs[j]) == bs[i]))
        bel[i][j] = 1, que.emplace(i, j);
      else
        bel[i][j] = 0;
  for (int i = 1; i <= N; i++) del[i] = 0;
  while (!que.empty() && (int)ans.size() < N - 1) {
    auto p = que.front();
    que.pop();
    int u = p.first, v = p.second;
    if (del[u] || del[v]) continue;
    del[u] = 1, ans.emplace_back(u, v);
    for (int i = 1; i <= M; i++)
      if (bs[u][i]) cnt[i]--;
    for (int i = 1; i <= M; i++) {
      if (cnt[i] != 1) continue;
      cnt[i]--;
      for (int j = 1; j <= N; j++) {
        if (str[i][j] == '0' || del[j]) continue;
        bs[j][i] = 0;
        for (int k = 1; k <= N; k++)
          if (j != k && !bel[j][k] && ((bs[j] & bs[k]) == bs[j]))
            bel[j][k] = 1, que.emplace(j, k);
      }
    }
  }
  if ((int)ans.size() < N - 1)
    puts("NO");
  else {
    puts("YES");
    for (auto it : ans) printf("%d %d\n", it.first, it.second);
  }
}
int main() {
  int T;
  scanf("%d", &T);
  while (T--) solve();
  return 0;
}
