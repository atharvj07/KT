#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for(int i = 0; i < (n); ++i)
#define all(x) x.begin(), x.end()
typedef long long ll;

const int MOD = 1e9 + 7;
int main() {
    int N; cin >> N;
    vector<int> C(N);
    for(int i = 0; i < N; ++i) cin >> C[i];
    sort(all(C));
    ll ret = 0;
    for(int i = 0; i < N; ++i) {
        ret = (ret + (ll)C[i] * (N - i + 1)) % MOD;
    }
    for(int i = 0; i < 2*N-2; ++i) {
        ret = (ret * 2) % MOD;
    }
    cout << ret << endl;
}
