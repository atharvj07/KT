import java.util.*;

public class Main {
    
    static class Bag implements Comparable<Bag> {
        public int index;
        public int value;
        public Bag(int index, int value) {
            this.index = index;
            this.value = value;
        }
        
        @Override
        public int compareTo(Bag o) {
            return o.value - value;
        }
        
        public String toString() {
            return index + " " + value;
        }
    }
    
    private static final long MOD = 1000000007;
    
    public static void main(String[] args) throws Exception {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int s = scan.nextInt();
        
        Bag[] original = new Bag[n];
        Bag[] sorted = new Bag[n];
        int maxValue = 0;
        for (int i = 0; i < n; i++) {
            original[i] = sorted[i] = new Bag(i, scan.nextInt());
            maxValue = Math.max(maxValue, original[i].value);
        }
        scan.close();
        
        Arrays.sort(sorted);
        
        s -= sorted[0].value;
        if (s < 0) {
            System.out.println(-1);
            return;
        }
        
        HashMap<Integer, ArrayList<Bag>> map = new HashMap<Integer, ArrayList<Bag>>();
        for (int i = 1; i <= maxValue; i++) {
            map.put(i, new ArrayList<Bag>());
        }
        for (int i = 1; i < n; i++) {
            map.get(sorted[i].value).add(sorted[i]);
        }
        
        HashSet<Integer> current = new HashSet<Integer>();
        HashSet<Integer> next = new HashSet<Integer>();
        current.add(0);
        int[] prev = new int[s + 1];
        Arrays.fill(prev, -1);
        prev[0] = sorted[0].index;
        for (int i = 1; i <= maxValue; i++) {
            ArrayList<Bag> bagList = map.get(i);
            if (bagList.isEmpty()) {
                continue;
            }
            for (int val : current) {
                next.add(val);
                for (int j = 0, sum = val + i; j < bagList.size() && sum <= s; j++, sum += i) {
                    if (prev[sum] == -1) {
                        prev[sum] = bagList.get(j).index;
                        next.add(sum);
                    } else {
                        break;
                    }
                }
            }
            if (next.contains(s)) {
                break;
            }
            HashSet<Integer> temp = current;
            current = next;
            next = temp;
            next.clear();
        }
        
        if (prev[s] == -1) {
            System.out.println(-1);
            return;
        }
        
        HashSet<Integer> root = new HashSet<Integer>();
        int search = s;
        while (search > 0) {
            if (root.contains(prev[search])) {
                throw new Error();
            }
            root.add(prev[search]);
            search -= original[prev[search]].value;
        }
        if (root.contains(prev[0])) {
            throw new Error();
        }
        root.add(prev[0]);
        
        int[] contain = new int[n];
        int last = -1;
        for (int i = n - 1; i > 0; i--) {
            int index = sorted[i].index;
            if (root.contains(index)) {
                contain[index] = -1;
            } else {
                contain[index] = last;
                last = index;
            }
        }
        contain[sorted[0].index] = last;
        
        for (int i = 0; i < n; i++) {
            if (contain[i] == -1) {
                System.out.println(String.format("%d 0", original[i].value));
            } else {
                System.out.println(String.format("%d 1 %d", original[i].value - original[contain[i]].value, contain[i] + 1));
            }
        }
    }
    
    private static void print(int... arr) {
        System.out.print(arr[0]);
        for (int i = 1; i < arr.length; i++) {
            System.out.print(" ");
            System.out.print(arr[i]);
        }
        System.out.println();
    }
    
    private static void print(long... arr) {
        System.out.print(arr[0]);
        for (int i = 1; i < arr.length; i++) {
            System.out.print(" ");
            System.out.print(arr[i]);
        }
        System.out.println();
    }
    
    private static <T> void print(T... arr) {
        System.out.print(arr[0]);
        for (int i = 1; i < arr.length; i++) {
            System.out.print(" ");
            System.out.print(arr[i]);
        }
        System.out.println();
    }
    
    private static void read(Scanner scan, int[]... arrs) {
        int len = arrs[0].length;
        for (int i = 0; i < len; i++) {
            for (int[] arr : arrs) {
                arr[i] = scan.nextInt();
            }
        }
    }
    
    private static void read(Scanner scan, long[]... arrs) {
        int len = arrs[0].length;
        for (int i = 0; i < len; i++) {
            for (long[] arr : arrs) {
                arr[i] = scan.nextLong();
            }
        }
    }
    
    private static void decreaseByOne(int[]... arrs) {
        for (int[] arr : arrs) {
            for (int i = 0; i < arr.length; i++) {
                arr[i] --;
            }
        }
    }
}
