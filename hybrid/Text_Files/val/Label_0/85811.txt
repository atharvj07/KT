#include <bits/stdc++.h>
using namespace std;
const long double pi = 4 * atan(1);
const long double eps = 1e-6;
inline int dcmp(long double x) {
  if (fabs(x) < eps)
    return 0;
  else
    return x < 0 ? -1 : 1;
}
long double fix_acute(long double th) {
  return th < -pi ? (th + 2 * pi) : th > pi ? (th - 2 * pi) : th;
}
inline long double getDistance(long double x, long double y) {
  return sqrt(x * x + y * y);
}
inline long double torad(long double deg) { return deg / 180 * pi; }
struct Point {
  long double x, y;
  Point(long double x = 0, long double y = 0) : x(x), y(y) {}
  void read() { scanf("%lf %lf", &x, &y); }
  bool operator==(const Point& u) const {
    return dcmp(x - u.x) == 0 && dcmp(y - u.y) == 0;
  }
  bool operator!=(const Point& u) const { return !(*this == u); }
  bool operator<(const Point& u) const {
    return dcmp(x - u.x) < 0 || (dcmp(x - u.x) == 0 && dcmp(y - u.y) < 0);
  }
  bool operator>(const Point& u) const { return u < *this; }
  bool operator<=(const Point& u) const { return *this < u || *this == u; }
  bool operator>=(const Point& u) const { return *this > u || *this == u; }
  Point operator+(const Point& u) { return Point(x + u.x, y + u.y); }
  Point operator-(const Point& u) { return Point(x - u.x, y - u.y); }
  Point operator*(const long double u) { return Point(x * u, y * u); }
  Point operator/(const long double u) { return Point(x / u, y / u); }
  long double operator*(const Point& u) { return x * u.y - y * u.x; }
};
namespace Punctual {
long double getDistance(Point a, Point b) {
  long double x = a.x - b.x, y = a.y - b.y;
  return sqrt(x * x + y * y);
}
};  // namespace Punctual
namespace Vectorial {
long double getDot(Point a, Point b) { return a.x * b.x + a.y * b.y; }
long double getCross(Point a, Point b) { return a.x * b.y - a.y * b.x; }
long double getLength(Point a) { return sqrt(getDot(a, a)); }
long double getPLength(Point a) { return getDot(a, a); }
long double getAngle(Point u) { return atan2(u.y, u.x); }
long double getSignedAngle(Point a, Point b) {
  return getAngle(b) - getAngle(a);
}
Point rotate(Point a, long double rad) {
  return Point(a.x * cos(rad) - a.y * sin(rad),
               a.x * sin(rad) + a.y * cos(rad));
}
Point getNormal(Point a) {
  long double l = getLength(a);
  return Point(-a.y / l, a.x / l);
}
};  // namespace Vectorial
namespace Linear {
using namespace Vectorial;
using namespace Punctual;
long double getDistanceToSegment(Point p, Point a, Point b) {
  if (a == b) return getLength(p - a);
  Point v1 = b - a, v2 = p - a, v3 = p - b;
  if (dcmp(getDot(v1, v2)) < 0)
    return getLength(v2);
  else if (dcmp(getDot(v1, v3)) > 0)
    return getLength(v3);
  else
    return fabs(getCross(v1, v2) / getLength(v1));
}
long double SegmentSegmentPointDIst(Point a1, Point a2, Point b1, Point b2) {
  long double lo = 0;
  long double hi = 1;
  long double mid1, mid2, val1, val2;
  for (int i = 1; i <= 30; i++) {
    mid1 = lo + (hi - lo) / 3;
    mid2 = hi - (hi - lo) / 3;
    val1 =
        getDistance(a1 * mid1 + a2 * (1 - mid1), b1 * mid1 + b2 * (1 - mid1));
    val2 =
        getDistance(a1 * mid2 + a2 * (1 - mid2), b1 * mid2 + b2 * (1 - mid2));
    if (val1 < val2)
      hi = mid2;
    else
      lo = mid1;
  }
  return val1;
}
}  // namespace Linear
Point A[100005];
Point B[100005];
int main() {
  using namespace Linear;
  int n, d1, d2;
  scanf("%d", &n);
  scanf("%d %d", &d1, &d2);
  int Count = 0;
  long double Min = 1e18;
  long double Max = 0;
  double a, b, c, d;
  for (int i = 1; i <= n; i++) {
    scanf("%lf %lf %lf %lf", &a, &b, &c, &d);
    A[i] = Point(a, b);
    B[i] = Point(c, d);
  }
  for (int i = 1; i < n; i++) {
    long double CurrMax = getDistance(A[i], B[i]);
    long double CurrMin =
        SegmentSegmentPointDIst(A[i], A[i + 1], B[i], B[i + 1]);
    Max = max(Max, CurrMax);
    if (Count == 0 || dcmp(Max - d2) > 0) Min = min(Min, CurrMin);
    if (dcmp(Min - d1) <= 0) {
      Count++;
      Min = 1e18;
      Max = 0;
    }
  }
  cout << Count << endl;
}
