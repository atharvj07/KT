int main(void) {
	int i,j,h,n,K;cin>>n>>K;
	llint dp[101][256]={0};
	dp[0][0]=1;
	for(i=0;i<n;i++){
		int x;cin>>x;
		for(j=i;j>=0;j--){
			for(h=0;h<256;h++){
				dp[j+1][h^x]+=dp[j][h];
				dp[j+1][h^x]%=mod;
			}
		}
	}
	llint ans=0,kai=1;
	for(i=0;i<=n;i++){
		ans+=dp[i][K]*kai;
		ans%=mod;
		kai*=(i+1);
		kai%=mod;
	}
	cout<<ans<<endl;
	
	return 0;
}
