for i in range(int(input())):
	s = input()
	l = len(s)
	aa = set(('a', 'b', 'c'))
	if l == 1:
		if s[0] == '?':
			print('a')
		else:
			print(s)
		continue
	s = list(s)
	for i in range(l):
		if s[i] == '?':
			if i == 0:
				if s[1] == '?':
					s[i] = 'a'
				elif s[1] == 'a':
					s[i] = 'b'
				else:
					s[i] = 'a'
			elif i == l - 1:
				if s[i - 1] == 'a':
					s[i] = 'b'
				else:
					s[i] = 'a'
			elif s[i + 1] != '?':
				s[i] = list(aa.difference(set((s[i - 1], s[i + 1]))))[0]
			else:
				s[i] = list(aa.difference(set(s[i - 1])))[0]
	t = True
	for i in range(l - 1):
		if s[i] == s[i + 1]:
			t = False
	print(''.join(s) if t else -1)
