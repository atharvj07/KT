using namespace std;
using ll = long long;
using ull = unsigned long long;

const ll INF = 1e16;
const ll MOD = 1e9 + 7;

#define REP(i, n) for(ll i = 0; i < n; i++)




ll dp[105][300][105];

int main() {
    ll n, k;
    cin >> n >> k;
    vector<ll> a(n);
    REP(i, n){
        cin >> a[i];
    }
    
    dp[0][0][0] = 1;
    for(ll i = 0; i < n; i++){
        for(ll j = 0; j < 260; j++){
            for(ll l = 0; l < n; l++){
                dp[i + 1][j ^ a[i]][l + 1] += dp[i][j][l];
                dp[i + 1][j ^ a[i]][l + 1] %= MOD;
                dp[i + 1][j][l] += dp[i][j][l];
                dp[i + 1][j][l] %= MOD;
            }
        }
    }
    
    vector<ll> r(105);
    r[1] = 1;
    for(ll i = 2; i <= 102; i++){
        r[i] = r[i - 1] * i;
        r[i] %= MOD;
    }
    
    ll ans = 0;
    for(ll i = 1; i <= n; i++){
        ans += (dp[n][k][i] * r[i]) % MOD;
        ans %= MOD; 
    }
    cout << ans << endl;
}
