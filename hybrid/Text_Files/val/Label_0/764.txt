from bisect import bisect_left as lower_bound, bisect_right as upper_bound
from sys import stdin, stdout

def main():
	t = int(stdin.readline())
	for _ in range(t):
		n = int(stdin.readline())
		a = list(map(int, stdin.readline().strip().split()))
		(pre, su) = ({i: [] for i in range(27)}, {i: [] for i in range(27)})
		for (i, num) in enumerate(a):
			su[num].append(i)
		ret = 1
		for (i, num) in enumerate(a):
			pre[num].append(i)
			su[num].pop(0)
			x = min(len(pre[num]), len(su[num]))
			if x == 0:
				continue
			(lo, hi) = (pre[num][x - 1], su[num][len(su[num]) - x])
			y = 0
			for j in range(27):
				p = lower_bound(su[j], hi)
				q = upper_bound(su[j], lo)
				y = max(y, p - q)
			ret = max(ret, 2 * x + y)
		print(ret)
main()
