n = int(input())
l = list(map(int, input().split()))
l = list(set(l))
le = len(l)
if le > 3:
	print(-1)
elif le == 3:
	l.sort()
	if l[2] - l[1] == l[1] - l[0]:
		print(l[1] - l[0])
	else:
		print(-1)
elif le == 2:
	if abs(l[-1] - l[0]) % 2 == 0:
		print(abs(l[-1] - l[0]) // 2)
	else:
		print(abs(l[-1] - l[0]))
else:
	print(0)
