a = int(input())
b = list(map(int, input().split()))
c = {}
for i in b:
	c.setdefault(i, 0)
if len(c) == 1:
	print(0)
elif len(c) == 2:
	if abs(list(c.keys())[1] - list(c.keys())[0]) % 2 == 0:
		print(abs(list(c.keys())[1] - list(c.keys())[0]) // 2)
	else:
		print(abs(list(c.keys())[1] - list(c.keys())[0]))
elif len(c) == 3:
	d = list(c.keys())
	d.sort()
	if (d[2] + d[0]) % 2 == 0 and (d[2] + d[0]) // 2 == d[1]:
		print((d[2] - d[0]) // 2)
	else:
		print(-1)
else:
	print(-1)
