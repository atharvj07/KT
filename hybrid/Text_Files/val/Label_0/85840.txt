int ccw(Point a, Point b, Point c)
{
  b -= a;
  c -= a;
  if(b.det(c) > 0.0)    return  1;         // counter clockwise
  if(b.det(c) < 0.0)    return -1;         // clockwise
  if(b.dot(c) < 0.0)    return  2;         // c--a--b on line
  if(norm(b) < norm(c)) return -2;         // a--b--c on line
  return 0;
}


struct Circle : Point
{
private:
  Point p;
  double r;

public:
  Circle() : p(Point(0.0, 0.0)), r(0.0) {}
  Circle(Point p, double r) : p(p), r(r) {}
  Circle(double x, double y, double r) : p(Point(x, y)), r(r) {}
  inline Point center(){ return this->p; }
  inline double radius(){ return this->r; }
};


vector<Point> intersectionOfCC(Circle A, Circle B)
{
  vector<Point> ret(2);
  double x1 = A.center().real(), y1 = A.center().imag(), r1 = A.radius();
  double x2 = B.center().real(), y2 = B.center().imag(), r2 = B.radius();
  double dx = x2 - x1;
  double dy = y2 - y1;
  double a = dx * dx + dy * dy;
  double b = ((r1 * r1 - r2 * r2) / a + 1) / 2;
  double d = r1 * r1 / a - b * b;
  double x3 = x1 + dx * b;
  double y3 = y1 + dy * b;
  double x4 = -dy * sqrt(d);
  double y4 =  dx * sqrt(d);
  ret[0] = Point(x3 - x4, y3 - y4);
  ret[1] = Point(x3 + x4, y3 + y4);
  return ret;
}


int N;
double X, R;
Circle* C;

bool check(double d)
{
  deque<int> deq;
  int pos = 0;
  deq.pb(pos++);
  while(!deq.empty()){
    int f = deq.front();
    Point c = C[f].center();
    double hx = c.real();
    double hr = C[f].radius();
    if(semile(hr, d / 2.0)){
      deq.pop_front();
      if(pos < N) deq.pb(pos++);
      continue;
    }
    double x = hx - sqrt(hr * hr - d * d / 4.0);
    Point np = Point(x + d, d / 2.0);
    if(!semige(abs(c - np), hr)) return true;
    int sz = deq.size();
    int t1 = deq.at(sz - 1);
    if(sz > 1){
      int t2 = deq.at(sz - 2);
      vector<Point> in = intersectionOfCC(C[t1], C[t2]);
      double y = max(in[0].imag(), in[1].imag());
      if(!semige(abs(C[t1].center() - np), C[t1].radius()))
        if(!semige(d / 2.0, y)) return true;
      if(semile(y, d / 2.0)){
        deq.pop_front();
        continue;
      }
    }
    if(pos >= N) deq.pop_front();
    if(pos < N) deq.pb(pos++);
  }
  return false;
}

int main(int argc, char *argv[])
{
  while(cin >> N, N){
    C = new Circle[N];

    rep(i, N){

      C[i] = Circle(X, 0.0, R);
      maxup(l, R);

    rep(_, 100){

      if(check(mid)) l = mid;

    }

    delete[] C;
  }
  
  return 0;
}
