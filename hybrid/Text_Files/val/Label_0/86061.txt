#include <bits/stdc++.h>
using namespace std;
long long dp[100005] = {};
int main() {
  int n, q;
  cin >> n >> q;
  vector<int> v(n), c(n);
  for (int i = 0; i < n; i++) cin >> v[i];
  for (int i = 0; i < n; i++) cin >> c[i];
  for (int r = 0; r < q; r++) {
    long long a, b;
    cin >> a >> b;
    memset(dp, 0, sizeof(dp));
    long long ans = 0;
    for (int i = 1; i <= n; i++) dp[i] = -9999999999999999LL;
    long long best1 = -9999999999999999LL, best2 = -9999999999999999LL;
    int pos1 = -1, pos2 = -1;
    for (int i = 1; i <= n; i++) {
      long long color = c[i - 1];
      long long value = v[i - 1];
      dp[color] = max(dp[color], dp[color] + value * a);
      dp[color] = max(dp[color], value * b);
      if (pos1 != -1) {
        if (pos1 != color)
          dp[color] = max(dp[color], best1 + value * b);
        else
          dp[color] = max(dp[color], best1 + value * a);
      }
      if (pos2 != -1) {
        if (pos2 != color)
          dp[color] = max(dp[color], best2 + value * b);
        else
          dp[color] = max(dp[color], best2 + value * a);
      }
      if (color == pos1) {
        best1 = max(best1, dp[color]);
      } else if (color == pos2) {
        best2 = max(best2, dp[color]);
        if (best2 > best1) {
          swap(best2, best1);
          swap(pos1, pos2);
        }
      } else if (dp[color] > best1) {
        best2 = best1;
        pos2 = pos1;
        best1 = dp[color];
        pos1 = color;
      } else if (dp[color] > best2) {
        best2 = dp[color];
        pos2 = color;
      }
      ans = max(ans, dp[color]);
    }
    cout << ans << endl;
  }
  return 0;
}
