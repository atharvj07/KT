numOfChar = int(input())
stringOfMov = input()
endpoint = int(numOfChar / 2)
maxlength = 0
if numOfChar > 1:
	while endpoint > 1:
		if stringOfMov[0:endpoint] == stringOfMov[endpoint:endpoint * 2]:
			maxlength = endpoint
			endpoint = 0
		else:
			endpoint -= 1
			maxlength = endpoint
	if numOfChar == 2:
		print(2)
	elif numOfChar == 3:
		print(3)
	else:
		print(numOfChar - maxlength + 1)
else:
	print(1)
