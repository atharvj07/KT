import java.util.*;
public class SocialNetwork1{
	public static void main(String arg[]){
	Scanner sc=new Scanner(System.in);
	int n=sc.nextInt();
	int m=sc.nextInt();
	HashSet<Integer> hs=new HashSet<>();
	LinkedList<Integer> ll=new LinkedList<>();
	for(int i=1;i<=n;i++){
		int a=sc.nextInt();
		if(!hs.contains(a)){
			hs.add(a);
			if(ll.size()==m){
				int b=ll.removeLast();
				hs.remove(b);
			}
			ll.addFirst(a);
		}
	}
	System.out.println(ll.size());
	for(int i:ll)
		System.out.print(i+" ");
	}
}
