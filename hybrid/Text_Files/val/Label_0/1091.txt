from sys import stdin
t = int(stdin.readline())
for _ in range(t):
	n = int(stdin.readline())
	a = stdin.readline().strip()
	b = stdin.readline().strip()
	res = []
	if a[0] != b[0]:
		res.append(1)
	for i in range(1, n):
		if a[i] != b[i]:
			res.append(i + 1)
			res.append(1)
			res.append(i + 1)
	print(len(res), *res)
