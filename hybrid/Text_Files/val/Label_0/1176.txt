import sys
for val in sys.stdin:
	(n, m) = val.split()
	n = int(n)
	m = int(m)
	(num1, num2) = ([], [])
	for val in map(int, input().split()):
		num1.append(val)
	for val in map(int, input().split()):
		num2.append(val)
	(cnt1, cnt2, sum1, sum2, total) = (-1, -1, 0, 0, -1)
	while True:
		if sum1 == sum2:
			total += 1
			sum1 = 0
			sum2 = 0
			cnt1 += 1
			cnt2 += 1
			if cnt1 == n and cnt2 == m:
				break
			sum1 += num1[cnt1]
			sum2 += num2[cnt2]
		if sum1 < sum2:
			cnt1 += 1
			sum1 += num1[cnt1]
		if sum1 > sum2:
			cnt2 += 1
			sum2 += num2[cnt2]
	print(total)
