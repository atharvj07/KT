#include <iostream>
#include <algorithm>

using namespace std;
typedef long long ll;
ll mod = 1000000007;
ll pw(ll a, ll x){
	ll ret = 1;
	while(x){
		while(!(x&1)){
			(a *= a) %= mod;
			x /= 2;
		}
		(ret *= a) %= mod;
		x--;
	}
	return ret;
}

int c[200010];
ll sum[200010];
int main(){
	ll i,n;
	cin >> n;
	for(i=0;i<n;i++){
		cin >> c[i];
	}
	sort(c,c + n);
	ll ans = 0;
	for(i=0;i<n;i++){
		(ans += pw(2,2*n - 2)*(n + 1 - i)%mod*c[i]%mod) %= mod;
	}
	cout << ans << endl;
}
