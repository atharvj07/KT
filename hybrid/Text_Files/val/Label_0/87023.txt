import java.io.PrintWriter;
import java.util.Arrays;
import java.util.Scanner;
/*
10
4 1 5 7 2 9 3 6 10 8
7
6 1 5 7 2 4 3
 */
public class lasers {

	static int N;
	static int[] mapTo,whatMapsHere;
	static boolean vis[];
	static char c[][];
	static int curRow;
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		N = in.nextInt();
		mapTo = new int[N];
		whatMapsHere = new int[N];
		boolean baseCase=true;
		for(int i = 0; i < N; i++) {
			mapTo[i]=in.nextInt()-1;
			whatMapsHere[mapTo[i]]=i;
			if(whatMapsHere[i]!=i)baseCase=false;
		}
		c = new char[N][N];
		for(char a[] : c)Arrays.fill(a, '.');
		if(baseCase) {
			out.println(N);
			for(int i = 0; i < N; i++)out.println(String.valueOf(c[i]));
			out.close();
			return;
		}
		
		vis = new boolean[N];
		curRow = 0;
		init(mapTo[N-1]);
//		print();
		for(int i = N-1; i >= 0; i--) {
			if(!vis[i]) go(mapTo[i],0);
//			print();
		}
		//System.out.println("Good: "+good(c,mapTo));

		out.println(N-1);
		for(int i = 0; i < N; i++) out.println(String.valueOf(c[i]));
		out.close();
		
	}
	static void go(int column, int dep) {

		vis[column]=true;
		int match = mapTo[column];
		boolean end = vis[match];
		//System.out.println("C:"+column+"  "+dep+"  "+mapTo[column]+"  e="+end+"  curRow="+curRow);
		//print();
		if(end) {
			if(match != column) {
				//System.out.println("  here");
				c[curRow][N-1]='/';
				c[curRow-dep][match]=c[curRow-dep][N-1]='\\';			
			}
			curRow++;
		}
		else {
			if(match < column) c[curRow+1][column]=c[curRow+1][match]='\\';	
			else if(match > column) c[curRow+1][column]=c[curRow+1][match]='/';
			curRow++;
			go(match, dep+1);
		}
	}
	static void init(int column) {
		vis[column]=true;
		if(column == N-1)return;
		int match = mapTo[column];
		if(match < column) c[curRow][column]=c[curRow][match]='\\';	
		else if(match > column) c[curRow][column]=c[curRow][match]='/';
		curRow++;
		init(match);
	}
	
	static boolean good(char b[][], int perm[]) {

		for(int i = 0; i < N-1; i++){
			int curR = N-1;
			int curC = i;
			int dR = -1;
			int dC = 0;
			System.out.println("i="+i+" map="+perm[i]);
			while(true) {
				System.out.println(curR+" "+curC+"  "+b[curR][curC]);
				if(b[curR][curC] == '/') {
					if(dR==-1) {
						dR=0;
						dC=1;
					}
					else if(dR==1) {
						dR=0;
						dC=-1;
					}
					else if(dC==-1) {
						dC=0;
						dR=1;
					}
					else if(dC==1) {
						dC=0;
						dR=-1;
					}
				}
				else if(b[curR][curC] == '\\') {
					if(dR==-1) {
						dR=0;
						dC=-1;
					}
					else if(dR==1) {
						dR=0;
						dC=1;
					}
					else if(dC==-1) {
						dC=0;
						dR=-1;
					}
					else if(dC==1) {
						dC=0;
						dR=1;
					}
				}
				curR += dR;
				curC += dC;
				if(curR < 0) {
					if(curC != perm[i])return false;
					break;
				}
				if(curR >= N || curC < 0 || curC >= N) return false;
			}		
		}
		
		return true;
	}
	
	
	static void print() {
		for(int i = 0; i < N; i++) System.out.println(String.valueOf(c[i]));
		System.out.println();
	}

}
