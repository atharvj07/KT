from collections import Counter
import math

def rsingle_int():
	return int(stdin.readline().rstrip())

def rmult_int():
	return [int(x) for x in stdin.readline().rstrip().split()]

def rmult_str():

def r_str():
	return stdin.readline().rstrip()

def rsingle_char():

def main():
	n = rsingle_int()
	parties = rmult_int()
	alice_p = parties[0]
	sum = alice_p
	coalition = [1]
	coalition_sum = alice_p
	for i in range(1, len(parties)):
		sum += parties[i]
		if int(alice_p / 2) >= parties[i]:
			coalition.append(i + 1)
			coalition_sum += parties[i]

		print(len(coalition))

main()
