t = int(input())
(line1s, line2s) = ([], [])
for i in range(t):
	line1s.append(input())
	line2s.append(input())

def solve(line1, line2):
	(n, x) = map(int, line1.split())
	a = list(map(int, line2.split()))
	a.sort(reverse=True)
	if x in a:
		print(1)
	elif x < a[0]:
		print(2)
	elif x % a[0] == 0:
		print(x // a[0])
	else:
		print(x // a[0] + 1)
for i in range(t):
	solve(line1s[i], line2s[i])
