#include <iostream>
#include <string>
using namespace std;

int main() {
	int n, kotae = 0, tdn = 0; string yj;
	cin >> n; cin >> yj;
	for(int i = 0; i < n; i++) {
		tdn = 0;
		string s; cin >> s;
		for(int j = 0; j < s.size(); j++) {
			for(int k = 1; k < s.size(); k++) {
				for(int l = 0; l < yj.size() && j + k * l < s.size(); l++) {
					if(s[j + k * l] != yj[l])
						break;
					if(l == yj.size() - 1) {
						kotae++;
						tdn = 1;
						break;
					}
				}
				if(tdn)
					break;
			}
			if(tdn)
				break;
		}
	}
	cout << kotae << endl;
	return 0;
}
