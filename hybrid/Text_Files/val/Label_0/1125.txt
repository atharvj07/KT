from bisect import bisect_right
for _ in range(int(input())):
	(n, q) = map(int, input().split())
	l = list(map(int, input().split()))
	l.sort()
	cumsum = l.copy()
	for i in range(1, n):
		cumsum[i] += cumsum[i - 1]
	possible_sums = set()

	def recurse(nums, sums):
		if not nums:
			return
		possible_sums.add(sums[-1] - (sums[0] - nums[0]))
		if len(nums) < 2 or nums[0] == nums[-1]:
			return
		mid = (nums[0] + nums[-1]) / 2
		mid_ind = bisect_right(nums, mid)
		if nums[mid_ind] == mid:
			mid_ind += 1
		recurse(nums[:mid_ind], sums[:mid_ind])
		recurse(nums[mid_ind:], sums[mid_ind:])
	recurse(l, cumsum)
	for _ in range(q):
		p = int(input())
		print('Yes' if p in possible_sums else 'No')
