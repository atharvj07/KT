import sys
from collections import defaultdict as dd
input = sys.stdin.readline
t = int(input())
for _ in range(t):
	arr = list(map(str, input().strip()))
	co = 0
	cz = 0
	mini = 99999999999
	ans = 0
	summ = 0
	l = [-1] * (len(arr) + 1)
	ind = 0
	for i in arr:
		ind += 1
		if i == '+':
			summ += 1
		else:
			summ -= 1
		if summ < 0 and l[summ * -1] == -1:
			l[summ * -1] = ind
			ans += ind
		mini = min(mini, summ)
	print(ans + len(arr))
