n = int(input())
a = list(map(int, input().split()))
b = a[::-1]
L1 = [1] * n
L2 = [1] * n
for i in range(n - 1):
	if a[i] < a[i + 1]:
		L1[i + 1] += L1[i]
for i in range(n - 1):
	if b[i] > b[i + 1]:
		L2[-i - 2] += L2[-i - 1]
ans = max(L1)
for i in range(n - 2):
	if a[i] < a[i + 2]:
		ans = max(ans, L1[i] + L2[i + 2])
print(ans)
