#include <iostream>
#include <cstdio>
using namespace std;

int main(){
    int m,n,i,j,k;
    for(int c=1;cin>>m,m;c++){
        cout << "Case " << c << ":\n";
        int mtx[10][10],dd=-1;
        i=0,j=0,n=1;
        for(k=0;k<=2*m-2;k++){
            while(true){
                mtx[i][j]=n;
                n++;
                if(((dd==-1)&&(i*(j-m+1)==0))||((dd==1)&&(j*(i-m+1)==0))) break;
                i+=dd;j-=dd;
            }
            dd*=-1;
            if(i==m-1)j++;
            else if(j==m-1)i++;
            else if(i) i++;
            else j++;
        }
        for(i=0;i<m;i++){
            for(j=0;j<m;j++){
                printf("%3d", mtx[i][j]);
            }
            cout<<'\n';
        }
    }
    return 0;
}
