def minExtraction(array):
	if len(array) == 1:
		return array[0]
	minel = min(array)
	array.sort()
	n = len(array)
	window = 0
	for x in range(1, n):
		if array[x] - array[x - 1] > window:
			window = array[x] - array[x - 1]
	return max(window, minel)
t = int(input())
final = []
for _ in range(t):
	n = int(input())
	array = list(map(int, input().split()))
	final.append(minExtraction(array))
for _ in range(t):
	print(final[_])
