def arr_inp():
	return [[int(x) for x in stdin.readline().split()] for i in range(4)]
from sys import stdin
(n, a) = (int(input()), arr_inp())
for i in range(4):
	if sum(a[i][0::2]) <= n:
		exit(print(i + 1, a[i][0], n - a[i][0]))
	if sum(a[i][1::2]) <= n:
		exit(print(i + 1, a[i][1], n - a[i][1]))
	if sum(a[i][1:3]) <= n:
		exit(print(i + 1, a[i][1], n - a[i][1]))
	if a[i][0] + a[i][3] <= n:
		exit(print(i + 1, a[i][0], n - a[i][0]))
print(-1)
