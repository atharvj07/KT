#include <bits/stdc++.h>
#define il inline
#define RG register
#define ll long long
#define rhl (1000000007)

using namespace std;

int f[2010][2010][2],n,k;

int main(){
  cin>>n>>k,f[n][0][0]=1;
  for (RG int i=n;i;--i)
    for (RG int j=n;j>=0;--j){
      (f[i][j][1]+=f[i][j+1][1])%=rhl;
      for (RG int p=0;p<2;++p){
	(f[i-1][j+1][0]+=f[i][j][p])%=rhl;
	(f[i-1][j][1]+=f[i][j][p])%=rhl;
      }
    }
  for (RG int i=n-k-1;i>0;--i) (f[0][n-k][1]<<=1)%=rhl;
  cout<<f[0][n-k][1]; return 0;
}
