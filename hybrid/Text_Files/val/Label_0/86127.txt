#include<bits/stdc++.h>
using ll = long long;
using namespace std;

signed main() {
	ll D, c, s; cin >> D;
	for (int h = 0; h < 26; h++) {
		cin >> c;
	}
	for (int i = 0; i < D; i++) {
		for (int h = 0; h < 26; h++) {
			cin >> s;
		}
	}
	for (int h = 0; h < D; h++) {
		cout << h % 26 + 1 << endl;
	}
	return 0;
}
