(n, k) = map(int, input().split())
point = list(map(int, input().split()))
MOD = [0] * 10
if k + sum(point) >= n * 100:
	print(n * 10)
else:
	ans = 0
	for i in range(n):
		ans += point[i] // 10
		MOD[point[i] % 10] += 1
	for j in range(9, -1, -1):
		if j == 0:
			ans += k // 10
			break
		elif k // (10 - j) >= MOD[j]:
			ans += MOD[j]
			k -= (10 - j) * MOD[j]
		else:
			ans += k // (10 - j)
			break
	print(ans)
