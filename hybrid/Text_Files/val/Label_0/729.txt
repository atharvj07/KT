import sys
cin = int(sys.stdin.readline().strip())
for testcase in range(cin):
	cin1 = int(sys.stdin.readline().strip())
	words = []
	for i in range(cin1):
		words.append(sys.stdin.readline().strip())
	(ans1, ans2, ans3, ans4, ans5) = (0, 0, 0, 0, 0)
	(x1, x2, x3, x4, x5) = (0, 0, 0, 0, 0)
	(la, lb, lc, ld, le) = ([], [], [], [], [])
	for i in words:
		d = {'a': 0, 'other': 0}
		for j in i:
			if j == 'a':
				d[j] += 1
			else:
				d['other'] += 1
		la.append(d['a'] - d['other'])
	la.sort(reverse=True)
	for i in la:
		if ans1 + i <= 0:
			break
		else:
			ans1 += i
			x1 += 1
	for i in words:
		d = {'b': 0, 'other': 0}
		for j in i:
			if j == 'b':
				d[j] += 1
			else:
				d['other'] += 1
		lb.append(d['b'] - d['other'])
	lb.sort(reverse=True)
	for i in lb:
		if ans2 + i <= 0:
			break
		else:
			ans2 += i
			x2 += 1
	for i in words:
		d = {'c': 0, 'other': 0}
		for j in i:
			if j == 'c':
				d[j] += 1
			else:
				d['other'] += 1
		lc.append(d['c'] - d['other'])
	lc.sort(reverse=True)
	for i in lc:
		if ans3 + i <= 0:
			break
		else:
			ans3 += i
			x3 += 1
	for i in words:
		d = {'d': 0, 'other': 0}
		for j in i:
			if j == 'd':
				d[j] += 1
			else:
				d['other'] += 1
		ld.append(d['d'] - d['other'])
	ld.sort(reverse=True)
	for i in ld:
		if ans4 + i <= 0:
			break
		else:
			ans4 += i
			x4 += 1
	for i in words:
		d = {'e': 0, 'other': 0}
		for j in i:
			if j == 'e':
				d[j] += 1
			else:
				d['other'] += 1
		le.append(d['e'] - d['other'])
	le.sort(reverse=True)
	for i in le:
		if ans5 + i <= 0:
			break
		else:
			ans5 += i
			x5 += 1
	print(max(x1, x2, x3, x4, x5))
