#define REP(i,b,n) for(int i=b;i<n;i++)
#define rep(ui,n)  REP(i,0,n)
#define double long double
const double r = 1./sqrt(3);
const double pi=acos(-1);
const double eps = 1e-10;

double CONVERT(double r){
  return r*pi/180.;
}

double solve(int n,double theta){
  double npolygon=0,minus=0;
  double l =2*r*sin(CONVERT(theta/2));
  npolygon=n*r*r*cos(CONVERT(theta/2))*sin(CONVERT(theta/2));
  minus=n*l*tan(CONVERT(60-theta/2))*l/4.;
  return npolygon-minus;
}

double calc(double a){
  while(a > 360-eps)a-=360;
  return a;
}
