#include <bits/stdc++.h>
using namespace std;
long long a[100005], b[100005];
long long sum[100005];
pair<long long, int> p[100005];
int main() {
  int n, A, cf, cm;
  long long m;
  scanf("%d%d%d%d%lld", &n, &A, &cf, &cm, &m);
  for (int i = 1; i <= n; i++) {
    scanf("%lld", &a[i]);
    p[i] = pair<long long, int>(a[i], i);
    b[i] = a[i];
  }
  sort(p + 1, p + 1 + n);
  for (int i = 1; i <= n; i++) {
    sum[i] = sum[i - 1] + (A - p[i].first);
  }
  int idx1 = n + 1, idx2 = 0, mn = 0;
  long long ans = 0;
  for (int i = n + 1; i >= 1; i--) {
    if (sum[n] - sum[i - 1] > m) break;
    int lo = p[1].first, hi = A, res = p[1].first, res2 = 1;
    while (lo <= hi) {
      int mid = (lo + hi) / 2;
      int lo2 = 1, hi2 = i - 1, idx = 0;
      while (lo2 <= hi2) {
        int mid2 = (lo2 + hi2) / 2;
        if (p[mid2].first <= mid)
          idx = mid2, lo2 = mid2 + 1;
        else
          hi2 = mid2 - 1;
      }
      long long result = sum[n] - sum[i - 1];
      result += sum[idx] - (A - mid) * 1ll * idx;
      if (result <= m)
        res = mid, res2 = idx, lo = mid + 1;
      else
        hi = mid - 1;
    }
    if (ans < (n - i + 1) * 1ll * cf + res * 1ll * cm) {
      ans = (n - i + 1) * 1ll * cf + res * 1ll * cm;
      idx1 = i, idx2 = res2;
      mn = res;
    }
  }
  for (int i = 1; i <= idx2; i++) {
    b[p[i].second] = mn;
  }
  for (int i = idx1; i <= n; i++) {
    b[p[i].second] = A;
  }
  printf("%lld\n", ans);
  for (int i = 1; i <= n; i++) printf("%lld ", b[i]);
}
