import sys
from collections import defaultdict
import heapq
(n, k) = map(int, sys.stdin.readline().split())
clouds = []
points = [0 for _ in range(201)]
for _ in range(n):
	(u, v) = map(int, sys.stdin.readline().split())
	clouds.append([u, v, _])
	for i in range(u, v + 1):
		points[i] += 1
clouds.sort()
(ans, i) = (0, 0)
heap = []
heapq.heapify(heap)
l = []
while i < n:
	start = clouds[i][0]
	while i < n:
		if clouds[i][0] == start:
			heapq.heappush(heap, [-clouds[i][1], clouds[i][2] + 1])
			i += 1
		else:
			break
	while points[start] > k:
		(right, ind) = heapq.heappop(heap)
		right = abs(right)
		l.append(ind)
		for j in range(start, right + 1):
			points[j] -= 1
		ans += 1
print(ans)
print(*l)
