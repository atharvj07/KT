#include<stdio.h>
#include<algorithm>
#include<math.h>
using namespace std;
double x[110000];
double r[110000];
pair<double,int>ev[110000];
double eps=1e-9;
int main(){
	int a;
	while(scanf("%d",&a),a){
		for(int i=0;i<a;i++)scanf("%lf%lf",x+i,r+i);
		double left=0;
		double right=1100000;
		for(int i=0;i<100;i++){
			double M=(left+right)/2;
			int sz=0;
			for(int j=0;j<a;j++){
				if(M+eps<r[j]){
					double t=sqrt(r[j]*r[j]-M*M);
					ev[sz++]=make_pair(x[j]-t,-1);
					ev[sz++]=make_pair(x[j]+t,1);
				}
			}
			std::sort(ev,ev+sz);
			int now=0;
			double last=0;
			bool ok=false;
			for(int j=0;j<sz;j++){
				if(!now)last=ev[j].first;
				now+=ev[j].second;
				if(!now){
					if(ev[j].first-last>2*M)ok=true;
				}
			}
			if(ok)left=M;
			else right=M;
		}
		printf("%.12f\n",left*2);
	}
}
