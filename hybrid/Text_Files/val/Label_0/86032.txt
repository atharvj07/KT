using LL = long long;
const int BIT = 62;

int n;
LL k;

bool is_ok(int m, LL k) {
  if (m >= BIT) return true;
  return (1LL << (m - 1)) >= k;
}

LL way(int A, int B) {
  CHECK(A > 0);
  LL ret = (1LL << (A - 1)) - 1;
  ret *= (1LL << B);
  return ret;
}

bool is_ok(int A, int B, LL k) {
  return way(A, B) < k;
}

bool solve() {
  if (!is_ok(n, k)) return false;
  std::vector<int> perm(n);
  for (int i = 0; i < n;) {
    if (is_ok(n - 1 - i, k)) {
      perm[i] = i;
      ++i;
      continue;
    }
    int seg = 1;
    while (i + seg < n && is_ok(seg + 1, n - i - seg - 1, k)) {
      ++seg;
    }
    k -= way(seg, n - i - seg);
    // DUMP(seg, "sub", way(seg, n - i - seg), seg, n - i - seg, i, n);
    for (int j = 0; j < seg; ++j) {
      perm[i + j] = i + seg - 1 - j;
    }
    i += seg;
  }
  for (int i = 0; i < n; ++i) {
    printf("%d%c", perm[i] + 1, " \n"[i + 1 == n]);
  }
  // DUMP(k);
  CHECK(k == 1);
  return true;
}

int main() {
  std::ios::sync_with_stdio(false);
  std::cin.tie(nullptr);
  std::istream& reader = std::cin;

  int cas;
  reader >> cas;
  while (cas--) {
    reader >> n >> k;
    if (!solve()) {
      puts("-1");
    }
  }
}
