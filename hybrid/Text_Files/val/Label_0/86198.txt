#include<iostream>
#include<cmath>
using namespace std; int n;
double SIN(double p) { return sin(p*acos(-1) / 180); }
int main() {
	while (cin >> n, n) {
		if (n % 3 == 0)n /= 3;
		double D = sqrt(3.0)*(SIN(60.0 + 60.0 / n) / (SIN(60.0 + 60.0 / n) + SIN(60.0 / n))) - sqrt(0.75);
		printf("%.99lf\n", sqrt(0.25 + D*D)*SIN(60.0 / n)*n);
	}
	return 0;
}
