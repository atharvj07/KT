#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, res;
    cin >> n;
    m = 2 * n;
    int a[m], d = 0, d1 = 0;
    bool check[m];
    for (int i = 1; i <= m; ++i) {
      cin >> a[i];
      if (a[i] % 2 == 0)
        d++;
      else
        d1++;
      check[i] = false;
    }
    if (d % 2 != 0 || d1 % 2 != 0) {
      for (int i = 1; i <= m; ++i) {
        if (a[i] % 2 == 0) {
          a[i] = -1;
          break;
        }
      }
      for (int i = 1; i <= m; ++i) {
        if (a[i] % 2 != 0) {
          a[i] = -1;
          break;
        }
      }
      for (int i = 1; i <= m - 1; ++i) {
        if (a[i] == -1) continue;
        for (int j = i + 1; j <= m; ++j) {
          if (a[j] == -1) continue;
          if ((a[i] % 2 == 0 && a[j] % 2 == 0) &&
              (check[i] == false && check[j] == false)) {
            cout << i << " " << j << "\n";
            check[i] = true;
            check[j] = true;
          }
          if ((a[i] % 2 != 0 && a[j] % 2 != 0) &&
              (check[i] == false && check[j] == false)) {
            cout << i << " " << j << "\n";
            check[i] = true;
            check[j] = true;
          }
        }
      }
    } else if (d % 2 == 0 || d1 % 2 == 0) {
      int tam = max(d, d1), dem = 0;
      if (tam == d) {
        for (int i = 1; i <= m; ++i) {
          if (dem == 2) break;
          if (a[i] % 2 == 0) {
            dem++;
            a[i] = -1;
          }
        }
      } else if (tam == d1) {
        for (int i = 1; i <= m; ++i) {
          if (dem == 2) break;
          if (a[i] % 2 != 0) {
            dem++;
            a[i] = -1;
          }
        }
      }
      for (int i = 1; i <= m - 1; ++i) {
        if (a[i] == -1) continue;
        for (int j = i + 1; j <= m; ++j) {
          if (a[j] == -1) continue;
          if ((a[i] % 2 == 0 && a[j] % 2 == 0) &&
              (check[i] == false && check[j] == false)) {
            cout << i << " " << j << "\n";
            check[i] = true;
            check[j] = true;
          }
          if ((a[i] % 2 != 0 && a[j] % 2 != 0) &&
              (check[i] == false && check[j] == false)) {
            cout << i << " " << j << "\n";
            check[i] = true;
            check[j] = true;
          }
        }
      }
    }
  }
}
