//package srms;

import java.util.Scanner;

public class theatre {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int t = sc.nextInt();
        long ans = 0;
        for (int man = t - 1; man >= 4; --man) {
            int woman = t - man;
            if (woman < 1) break;
            ans += C(n, man) * C(m, woman);
        }
        System.out.print(ans);
    }

    public static long C(int n, int k) {
        double res = 1;
        for (int i = 1; i <= k; ++i)
            res = res * (n - k + i) / i;
        return (long) (res + 0.01);
    }

}
