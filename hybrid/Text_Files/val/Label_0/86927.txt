import java.io.*;
import java.util.*;



public class Bridges {
	void solve() throws IOException {
		int platforms_number = nextInt();
		int[] bridges = new int[platforms_number - 1];
		
		for (int i = 0; i < platforms_number - 1; ++i) {
			bridges[i] = nextInt();
		}
		
		
		long result = 0;
		
		
		long[] ltransitions_number  = new long[platforms_number];
		long[] lrtransitions_number = new long[platforms_number];
		long[] rtransitions_number  = new long[platforms_number];
		long[] rrtransitions_number = new long[platforms_number];
		
		ltransitions_number[0]  = 0;
		lrtransitions_number[0] = 0;
		for (int i = 1; i < platforms_number; ++i) {
			int bi = i - 1;
			int b  = bridges[bi];
			
			if (b % 2 == 0) {
				ltransitions_number[i]  = Math.max((b - 1) + ltransitions_number[i - 1], b + lrtransitions_number[i - 1]);
				lrtransitions_number[i] = b + lrtransitions_number[i - 1];
			} else {
				ltransitions_number[i]  = b + ltransitions_number[i - 1];
				
				if (b != 1) {
					lrtransitions_number[i] = b - 1 + lrtransitions_number[i - 1];
				} else {
					lrtransitions_number[i] = 0;
				}
			}
		}
		
		rtransitions_number[platforms_number - 1]  = 0;
		rrtransitions_number[platforms_number - 1] = 0;
		
		result = ltransitions_number[platforms_number - 1];
		
		for (int i = platforms_number - 2; i >= 0; --i) {
			int b  = bridges[i];
			
			if (b % 2 == 0) {
				rtransitions_number[i]  = Math.max((b - 1) + rtransitions_number[i + 1], b + rrtransitions_number[i + 1]);
				rrtransitions_number[i] = b + rrtransitions_number[i + 1];
			} else {
				rtransitions_number[i]  = b + rtransitions_number[i + 1];
				
				if (b != 1) {
					rrtransitions_number[i] = b - 1 + rrtransitions_number[i + 1];
				} else {
					rrtransitions_number[i] = 0;
				}
			}
			
			result = Math.max(result, ltransitions_number[i] + rrtransitions_number[i]);
			result = Math.max(result, rtransitions_number[i] + lrtransitions_number[i]);
		}
		
		
		out.println(result);
	}



	void run() throws IOException {
		//reader = new BufferedReader(new FileReader("B-large.in"));
		reader = new BufferedReader(new InputStreamReader(System.in));
		tokenizer = null;
		//out = new PrintWriter(new FileWriter("B-large.out"));
		out = new PrintWriter(new OutputStreamWriter(System.out));
		solve();
		reader.close();
		out.flush();
	}

	public static void main(String[] args) throws IOException {
		new Bridges().run();
	}

	BufferedReader reader;
	StringTokenizer tokenizer;
	PrintWriter out;

	int nextInt() throws IOException {
		return Integer.parseInt(nextToken());
	}

	long nextLong() throws IOException {
		return Long.parseLong(nextToken());
	}

	double nextDouble() throws IOException {
		return Double.parseDouble(nextToken());
	}

	String nextToken() throws IOException {
		while (tokenizer == null || !tokenizer.hasMoreTokens()) {
			tokenizer = new StringTokenizer(reader.readLine());
		}
		return tokenizer.nextToken();
	}
}
