n = int(input())
arr = list(map(int, input().split()))
if n == 1:
	print('+')
elif n == 2:
	print('-+')
else:
	ans = ['+']
	cur = arr[-1]
	for i in range(n - 2, -1, -1):
		if cur > 0:
			cur -= arr[i]
			ans.append('-')
		else:
			cur += arr[i]
			ans.append('+')
	ans.reverse()
	if cur < 0:
		for i in range(n):
			if ans[i] == '-':
				ans[i] = '+'
			else:
				ans[i] = '-'
	print(''.join(ans))
