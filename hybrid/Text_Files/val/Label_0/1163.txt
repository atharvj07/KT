import sys
input = sys.stdin.readline

def I():
	return input().strip()

def II():
	return int(input().strip())

def LI():
	return [*map(int, input().strip().split())]
import copy
import re
import string, math, time, functools, random, fractions
from heapq import heappush, heappop, heapify

for _ in range(II()):
	n = II()
	a = LI()
	b = LI()
	inc = []
	dec = []
	if sum(a) != sum(b):
		print(-1)
		continue
	for i in range(n):
		if a[i] < b[i]:
			x = b[i] - a[i]
			inc = inc + [i + 1] * x
		elif a[i] > b[i]:
			y = a[i] - b[i]
			dec = dec + [i + 1] * y
	print(len(inc))
	for i in range(len(inc)):
		print(dec[i], inc[i])
