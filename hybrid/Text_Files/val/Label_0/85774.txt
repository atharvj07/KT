#include <bits/stdc++.h>
using namespace std;

int main() {
  cin.tie(nullptr);
  ios::sync_with_stdio(false);
  int n, k;
  cin >> n >> k;
  vector<long long> a(n);
  for (auto&& e : a) {
    cin >> e;
    --e;
  }
  a.push_back(0);
  for (int i = n - 1; i >= 0; --i) {
    a[i] += a[i + 1];
    a[i] %= k;
  }
  long long res = 0;
  map<int, int> mp;
  for (int r = 0; r <= n; ++r) {
    if (r - k >= 0) {
      --mp[a[r - k]];
    }
    res += mp[a[r]]++;
  }
  cout << res << '\n';
}
