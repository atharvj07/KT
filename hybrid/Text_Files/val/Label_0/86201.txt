#include <cstdio>
#include <cmath>

using namespace std;

const double eps = 1.0e-7;
const double pi = acos(-1);

bool equals(double a, double b) { return abs(a-b) < eps; }

int main() {
  const static double rr = 1.0/(2.0*(1.0-cos(pi*2.0/3.0)));
  double N;
  while(scanf("%lf",&N) != EOF && N) {
    double n;
    for(n = 1; n < N && !equals(0,fmod(3*n,N)); n+=1);
    n *= 3;
    printf("%.6f\n",
	   rr*pi - n*( (pi*2/n-sin(pi*2/n))*rr/2.0
		       + tan((pi*(n-2)/n-pi/3.0)/2.0)*2*rr*(1-cos(pi*2/n))/4.0 ));
  }
  return 0;
}
