var text = readline();
var arr = text.match(/<\/?[a-z]>/g).map(function(t){return t.replace(/<|>/g, '')});
var hash = {};
var currIndent = 0;
for (var i = 0; i < arr.length; i++) {
    var val = arr[i];
    if (val.search('/') != -1) {
        print(' '.repeat(hash[val.replace('/', '')].pop()) + '<' + val + '>');
        currIndent -= 2;
    } else {
        print(' '.repeat(currIndent) + '<' + val + '>');
        hash[val] = hash[val] || [];
        hash[val].push(currIndent);
        currIndent += 2;
    }
}
