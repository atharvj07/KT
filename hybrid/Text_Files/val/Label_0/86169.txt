#include <bits/stdc++.h>
using namespace std;
int n, m;
string mp[105];
int main() {
  cin >> n >> m;
  for (int i = 0; i < n; i++) cin >> mp[i];
  int p = 1, pos = 0;
  long long sum = 0;
  for (int i = 0; i < n - 1; i++) {
    int l, r;
    l = r = pos;
    int a, b;
    a = b = 0;
    while (1) {
      if (p) {
        sum += r - pos;
        pos = r;
        if (mp[i + 1][r] == '.') {
          sum++;
          break;
        }
        if (r < m - 1 && mp[i][r + 1] == '.')
          r++;
        else if (r < m - 1 && mp[i][r + 1] == '+')
          r++, p ^= 1, sum++;
        else
          p ^= 1, a = 1, sum++;
      } else {
        sum += pos - l;
        pos = l;
        if (mp[i + 1][l] == '.') {
          sum++;
          break;
        }
        if (l && mp[i][l - 1] == '.')
          l--;
        else if (l && mp[i][l - 1] == '+')
          l--, p ^= 1, sum++;
        else
          p ^= 1, b = 1, sum++;
      }
      if (a && b) {
        cout << "Never" << endl;
        return 0;
      }
    }
  }
  cout << sum << endl;
}
