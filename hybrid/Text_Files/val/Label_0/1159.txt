j = 0

def fun(x):
	global j
	j += 1
	return [int(x), int(j)]
n = int(input())
l = list(map(fun, input().split()))
c = 0
for i in range(1, n):
	c += l[i][0]
t = c + l[0][0]
i = 0
o = l[0][0]
s = l[0][0]
m = [1]
l.sort()
while i < n:
	if s >= t // 2 + 1:
		break
	elif o >= 2 * l[i][0]:
		if l[i][1] != 1:
			s += l[i][0]
			c -= l[i][0]
			m.append(l[i][1])
	else:
		break
	i += 1
if s <= c:
	print('0')
else:
	print(len(m))
	for i in m:
		print(i, end=' ')
