#include <cstdio>
#include <cstring>
#include <climits>
using namespace std;

char s[200002];
int dp[1 << 26];

int main(){
  scanf("%s", s + 1); int L = strlen(s + 1);

  int M = 1 << 26;

  for(int i = 0; i < M; i++) dp[i] = INT_MAX;

  dp[0] = 0; int v = 0;

  for(int l = 1; l <= L; l++){
    v ^= 1 << (s[l] - 'a');

    for(int i = 0; i < 26; i++){
      int w = v ^ 1 << i;

      if(dp[w] != INT_MAX && dp[w] + 1 < dp[v]) dp[v] = dp[w] + 1;
    }
  }

  dp[0] = 1; printf("%d\n", dp[v]);
  return 0;
}
