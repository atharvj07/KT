tes = int(input())
for j in range(0, tes):
	size = int(input())
	s = input()
	t = input()
	arr = []
	l = 0
	k = 0
	for i in range(0, size):
		if s[i] != t[i]:
			arr.append(i)
			l = l + 1
		if l > 2:
			print('NO')
			k = 1
			break
	if l == 2 and s[arr[0]] == s[arr[1]] and (t[arr[0]] == t[arr[1]]):
		print('YES')
	elif l < 2 or k == 0:
		print('NO')
