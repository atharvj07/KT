#include <bits/stdc++.h>
const long double PI = acos(-1);
const long double eps = 0.0000000001;
const long long INF = 0x3fffffffffffffff;
long long n, d1, d2, x[100005], y[100005], px[100005], py[100005], ans;
bool good = true;
long long dian(long long Ax, long long Ay, long long Bx, long long By) {
  return Ax * Bx + Ay * By;
}
long long cha(long long Ax, long long Ay, long long Bx, long long By) {
  return Ax * By - Bx * Ay;
}
long long dis(long long x, long long y) { return x * x + y * y; }
long long sqr(long long x) { return x * x; }
signed main() {
  std::ios::sync_with_stdio(false);
  std::cin >> n >> d1 >> d2;
  d1 *= d1;
  d2 *= d2;
  for (long long i = 1; i <= n; ++i) {
    static long long sx, sy, tx, ty;
    std::cin >> sx >> sy >> tx >> ty;
    px[i] = tx - sx;
    py[i] = ty - sy;
    if (i == 1) continue;
    x[i] = px[i] - px[i - 1];
    y[i] = py[i] - py[i - 1];
    if (dian(-px[i - 1], -py[i - 1], x[i], y[i]) >= 0 &&
        dian(-px[i - 1], -py[i - 1], x[i], y[i]) <= dis(x[i], y[i])) {
      if (dis(px[i - 1], py[i - 1]) >= d2) good = true;
      if (good && sqr(cha(-px[i - 1], -py[i - 1], x[i], y[i])) <=
                      dis(x[i], y[i]) * d1) {
        good = false;
        ++ans;
      }
      if (dis(px[i], py[i]) >= d2) good = true;
    } else {
      if (dis(px[i - 1], py[i - 1]) >= d2) good = true;
      if (good && dis(px[i - 1], py[i - 1]) <= d1) {
        good = false;
        ++ans;
      }
      if (dis(px[i], py[i]) >= d2) good = true;
      if (good && dis(px[i], py[i]) <= d1) {
        good = false;
        ++ans;
      }
    }
  }
  std::cout << ans << std::endl;
  return 0;
}
