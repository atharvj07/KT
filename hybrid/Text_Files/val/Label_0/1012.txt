import sys, math
input = sys.stdin.readline

def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	s = input()
	return s[:len(s) - 1]

def invr():
	return map(int, input().split())
t = 1
for tt in range(t):
	(n, h, l, r) = invr()
	a = inlt()
	q = [0] + [-1] * (h - 1)
	for i in range(len(a)):
		w = [-1] * h
		for qw in range(h):
			if q[qw] == -1:
				continue
			if l <= (qw + a[i]) % h <= r:
				w[(qw + a[i]) % h] = max(w[(qw + a[i]) % h], q[qw] + 1)
			else:
				w[(qw + a[i]) % h] = max(w[(qw + a[i]) % h], q[qw])
			if l <= (qw + a[i] - 1) % h <= r:
				w[(qw + a[i] - 1) % h] = max(w[(qw + a[i] - 1) % h], q[qw] + 1)
			else:
				w[(qw + a[i] - 1) % h] = max(w[(qw + a[i] - 1) % h], q[qw])
		q = w
	print(max(q))
