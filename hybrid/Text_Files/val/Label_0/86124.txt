#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i< (n); i++)
using ll = long long;
using namespace std;

int main(void){
	int D;
	cin >> D;
	int c[26];
	int s[D][26];
	rep(i,26) cin >> c[i];
	rep(i,D) rep(j,26) cin >> s[i][j];

	int t[D];
	rep(i,D){
		int max_s = s[i][0];
		int jmax = 0;
		rep(j,26){
			if (s[i][j]>=max_s){
				max_s = s[i][j];
				jmax =j;
			}
		}
		t[i] = 1;
	}
	rep(i,D) cout << t[i] << endl;
	


return 0;
}
