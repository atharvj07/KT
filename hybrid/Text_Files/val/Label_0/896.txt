(a, ta) = map(int, input().split())
(b, tb) = map(int, input().split())
(fh, fm) = map(int, input().split(':'))
start = fh * 60 + fm
stb = 5 * 60
(bl, al) = ([], [])
while stb < 24 * 60:
	bl.append((stb, stb + tb))
	stb += b

def inter(x, y):
	if x[0] >= y[1] or y[0] >= x[1]:
		return 0
	return 1
my = (start, start + ta)
ans = 0
for z in bl:
	ans += inter(my, z)
print(ans)
