import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collection;
import java.util.Deque;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

public class P651E {

    private static class Node implements Comparable<Node> {
        public final int i;
        public final int j;
        public final int a;
        public final List<Node> predecessors = new ArrayList<>(2);
        public final List<Node> peers = new ArrayList<>(4);
        public int c;

        public Node(int i, int j, int a) {
            this.i = i;
            this.j = j;
            this.a = a;
        }

        public Collection<Node> allPeers() {
            final Set<Node> allPeers = new HashSet<>();
            allPeers.add(this);
            Deque<Node> deque = new ArrayDeque<>();
            deque.addLast(this);
            while (!deque.isEmpty()) {
                final Node node = deque.removeLast();
                for (Node peer : node.peers) {
                    if (!allPeers.contains(peer)) {
                        allPeers.add(peer);
                        deque.add(peer);
                    }
                }
            }
            return allPeers;
        }

        @Override
        public int compareTo(Node other) {
            return this.a - other.a;
        }
    }

    private static int[][] compress(int n, int m, int[][] A) {
        final Node[][] nodes = new Node[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                nodes[i][j] = new Node(i, j, A[i][j]);
            }
        }

        final Node[] row = new Node[m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                row[j] = nodes[i][j];
            }
            Arrays.sort(row);
            Node prev = null;
            for (final Node node : row) {
                if (prev == null) {
                    // No op
                } else if (node.a == prev.a) {
                    node.peers.add(prev);
                    prev.peers.add(node);
                } else {
                    node.predecessors.add(prev);
                }
                prev = node;
            }
        }

        final Node[] col = new Node[n];
        for (int j = 0; j < m; j++) {
            for (int i = 0; i < n; i++) {
                col[i] = nodes[i][j];
            }
            Arrays.sort(col);
            Node prev = null;
            for (final Node node : col) {
                if (prev == null) {
                    // No op
                } else if (node.a == prev.a) {
                    node.peers.add(prev);
                    prev.peers.add(node);
                } else {
                    node.predecessors.add(prev);
                }
                prev = node;
            }
        }

        final Node[] arr = new Node[n*m];
        int k = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                arr[k++] = nodes[i][j];
            }
        }
        Arrays.sort(arr);
        for (Node node : arr) {
            if (node.c > 0) {
                continue;
            }
            if (node.peers.isEmpty()) {
                int c = 1;
                for (Node child : node.predecessors) {
                    c = Math.max(c, child.c + 1);
                }
                node.c = c;
            } else {
                final Collection<Node> peers = node.allPeers();
                int c = 1;
                for (Node peer : peers) {
                    for (Node child : peer.predecessors) {
                        c = Math.max(c, child.c + 1);
                    }
                }
                for (Node peer : peers) {
                    peer.c = c;
                }
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                A[i][j] = nodes[i][j].c;
            }
        }
        return A;
    }

    public static void main(String[] args) throws IOException {
        final BufferedReader in = new BufferedReader(
            new InputStreamReader(System.in));
        String[] s = in.readLine().split(" ");
        final int n = Integer.parseInt(s[0]);
        final int m = Integer.parseInt(s[1]);
        final int[][] A = new int[n][m];
        for (int i = 0; i < n; i++) {
            s = in.readLine().split(" ");
            for (int j = 0; j < m; j++) {
                A[i][j] = Integer.parseInt(s[j]);
            }
        }
        final int[][] C = compress(n, m, A);
        final StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            sb.append(C[i][0]);
            for (int j = 1; j < m; j++) {
                sb.append(' ').append(C[i][j]);
            }
            sb.append('\n');
        }
        System.out.print(sb);
    }

    private static final void log(Object o) {
        System.err.println(o);
    }
}
