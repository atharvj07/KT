import java.util.HashMap;
import java.util.LinkedList;
import java.util.Scanner;

/**
 * Created with IntelliJ IDEA.

 * Time     : 15:38
 * Project  : Suggested Friends
 */
public class Main {
    public static void main(String[] args) {
        HashMap<String, Integer> nameToID;
        LinkedList<Integer>[] friends;
        String[] iDtoName;
        boolean[][] isFriends;
        Scanner in;

        friends = new LinkedList[10005];
        nameToID = new HashMap<String, Integer>();
        iDtoName = new String[10005];
        isFriends = new boolean[10005][10005];
        in = new Scanner(System.in);

        int m = in.nextInt();

        for(int i = 0; i < 10005; i++) {
            friends[i] = new LinkedList<Integer>();
        }
        for(int i = 0; i < m; i++) {
            String s1 = in.next();
            String s2 = in.next();
            int u1, u2;
            if(!nameToID.containsKey(s1)) {
                u1 = nameToID.size();
                nameToID.put(s1, nameToID.size());
                iDtoName[u1] = s1;
            } else {
                u1 = nameToID.get(s1);
            }
            if(!nameToID.containsKey(s2)) {
                u2 = nameToID.size();
                nameToID.put(s2, nameToID.size());
                iDtoName[u2] = s2;
            } else {
                u2 = nameToID.get(s2);
            }
            friends[u1].add(u2);
            friends[u2].add(u1);
            isFriends[u1][u2] = true;
            isFriends[u2][u1] = true;
        }
        int n = nameToID.size();
        System.out.println(n);
        for(int i = 0; i < n; i++) {
            if(friends[i].size() > n - 2) {
                System.out.println(iDtoName[i] + " " + 0);
                continue;
            }
            int[] tmp = new int[10005];
            int max = 0, cnt = 0;
            for(int j : friends[i]) {
                for(int k : friends[j]) {
                    if(i == k || isFriends[i][k]) continue;
                    tmp[k]++;
                    if(tmp[k] > max) {
                        max = tmp[k];
                        cnt = 0;
                    }
                    if(tmp[k] == max && max > 0) {
                        cnt++;
                    }
                }
            }
            if(cnt == 0) cnt = n - 1 - friends[i].size();
            System.out.println(iDtoName[i] + " " + cnt);
        }
    }
}
