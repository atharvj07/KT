#include <bits/stdc++.h>
using namespace std;
long long fastpow(long long a, long long b,
                  long long m = (long long)(1e9 + 7)) {
  long long res = 1;
  a %= m;
  while (b > 0) {
    if (b & 1) res = (res * a) % m;
    a = (a * a) % m;
    b >>= 1;
  }
  return res;
}
long long n, k;
string dig;
int32_t main() {
  ios_base ::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  ;
  cin >> n >> k >> dig;
  string res(n, '0');
  cout << n << endl;
  for (long long i = 0; i < k; i++) {
    res[i] = dig[i];
    for (long long j = i + k; j < n; j += k) {
      res[j] = dig[i];
    }
  }
  if (res < dig) {
    for (long long i = k - 1; i >= 0; i--) {
      if (res[i] < '9') {
        for (long long j = i; j < n; j += k) {
          res[j]++;
        }
        for (long long j = i + 1; j < k; j++) {
          for (long long c = j; c < n; c += k) {
            res[c] = '0';
          }
        }
        break;
      }
    }
  }
  cout << res << endl;
  return 0;
}
