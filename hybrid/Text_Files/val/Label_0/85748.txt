#include<iostream>
#include<algorithm>

using namespace std;

int nt;
int t[30];
int m[101];

int gd(int d){
  for(int i=0;i+d<=100;i++){
    if(t[i%nt]>m[d+i])return d+i;
  }
  return 999;
}

int main(){
  while(cin>>nt,nt){
    for(int i=0;i<nt;i++){
      cin>>t[i];
    }
    fill(m,m+101,24);
    int n;
    cin>>n;
    while(n--){
      int d,mm;
      cin>>d>>mm;
      m[d]=min(m[d],mm);
    }
    int cd=1;
    int nc=0;
    for(;;){
      int dl=gd(cd);
      if(dl==999)break;
      int ml=0,x;
      for(int i=1;i+cd<=dl;i++){
	int id=gd(i+cd);
	if(id>ml){
	  ml=id;
	  x=i+cd;
	}
      }
      cd=x;
      nc++;
    }
    cout<<nc<<endl;
  }
  return 0;
}
