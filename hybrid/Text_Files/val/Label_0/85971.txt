#include<bits/stdc++.h>
using namespace std;
struct edge {
  int to, cost;
};
const int INF = 1 << 30;
typedef pair< int, int > Pi;
 
int N, M, C;
vector< edge > graph[100000];
Pi min_cost[100000];
 
void Dijkstra()
{
  for(int i = 0; i < N; i++) {
    min_cost[i] = make_pair(INF, i);
  }
  priority_queue< Pi, vector< Pi >, greater< Pi > > que;
  que.push(make_pair(0, 0));
  min_cost[0].first = 0;
  while(!que.empty()) {
    Pi p = que.top(); que.pop();
    if(min_cost[p.second].first < p.first) continue;
    for(int i = 0; i < graph[p.second].size(); i++) {
      const edge& e = graph[p.second][i];
      if(min_cost[e.to].first <= p.first + e.cost) continue;
      min_cost[e.to].first = p.first + e.cost;
      que.push(make_pair(min_cost[e.to].first, e.to));
    }
  }
}
 
 
int main()
{
  long long ALL = 0;
  scanf("%d %d %d", &N, &M, &C);
  for(int i = 0; i < M; i++) {
    int a, b, c;
    scanf("%d %d %d", &a, &b, &c);
    --a, --b;
    graph[a].push_back((edge){b, c});
    graph[b].push_back((edge){a, c});
    ALL += c;
  }
  Dijkstra();
  sort(min_cost, min_cost + N);
  vector< int > nums(N);
  for(int i = 0; i < N; i++) {
    nums[i] = min_cost[i].first;
  }
  nums.erase(unique(nums.begin(), nums.end()), nums.end());
  vector< bool > used(N, false);
  long long ret = 1LL << 60;
  for(int i = 0, ptr = 0; i < nums.size(); i++) {
    for( ; ptr < N && nums[i] == min_cost[ptr].first; ++ptr) {
      int curr = min_cost[ptr].second;
      used[curr] = true;
      for(int j = 0; j < graph[curr].size(); j++) {
        if(used[graph[curr][j].to]) ALL -= graph[curr][j].cost;
      }
    }
    ret = min(ret, ALL + (long long)C * nums[i]);
  }
  cout << ret << endl;
}
