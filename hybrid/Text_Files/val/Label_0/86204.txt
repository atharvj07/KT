#include <bits/stdc++.h>
using namespace std;
int main() {
  int a1, a2, a3, b1, b2, b3, n, flag = 1;
  cin >> a1 >> a2 >> a3;
  cin >> b1 >> b2 >> b3;
  cin >> n;
  int cup = a1 + a2 + a3;
  int medal = b1 + b2 + b3;
  if (n > 0)
    if (cup % 5 == 0) {
      n = n - (cup / 5);
      if (n < 0) flag = 0;
    } else {
      n = n - (cup / 5 + 1);
      if (n < 0) flag = 0;
    }
  if (n >= 0) {
    if (medal % 10 == 0) {
      n = n - (medal / 10);
      if (n < 0) flag = 0;
    } else {
      n = n - (medal / 10 + 1);
      if (n < 0) flag = 0;
    }
  }
  if (flag == 0)
    cout << "NO" << endl;
  else
    cout << "YES" << endl;
  return 0;
}
