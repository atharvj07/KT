// package question1;

import java.util.Scanner;

public class Solution{
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-->0){
            int a = sc.nextInt();
            if(a>3 && a%2==0){
                System.out.println(2);
            }
            else if(a>3 && a%2!=0){
                System.out.println(3);
            }
            else {
                 if(a==1){
                    System.out.println(0);
                }
                 else if(a==3){
                     System.out.println(2);
                 }
                else {
                    System.out.println(1);
                }
            }
        }
    }
}
