gcd = lambda a, b: gcd(b, a % b) if b else a

def main():
	n = int(input())
	if n == 1:
		print(0, 0)
		return
	x = 2 * n // 3
	if 2 * n % 2:
		x += 1
	s = 0
	for i in range(x):
		print(i, 0)
		s += 1
	for j in range(1, x, 2):
		print(j, 3)
		s += 1
	while n - s:
		s += 1
		i += 1
		print(i, 0)
main()
