from collections import deque
import heapq
import sys
input = sys.stdin.readline

def bfs1(s):
	q = deque()
	q.append(s)
	visit = [0] * (n + 1)
	visit[s] = 1
	while q:
		i = q.popleft()
		for j in G[i]:
			if not visit[j]:
				visit[j] = 1
				q.append(j)
	return visit

def bfs2(t):
	q = deque()
	q.append(t)
	visit0 = [0] * (n + 1)
	visit0[t] = 1
	c = 0
	while q:
		i = q.popleft()
		c += 1
		for j in G[i]:
			if not visit0[j] and (not visit[j]):
				visit0[j] = 1
				q.append(j)
	return c

def bfs3(t):
	q = deque()
	q.append(t)
	visit[t] = 1
	while q:
		i = q.popleft()
		for j in G[i]:
			if not visit[j]:
				visit[j] = 1
				q.append(j)
	return
(n, m, s) = map(int, input().split())
G = [[] for _ in range(n + 1)]
for _ in range(m):
	(u, v) = map(int, input().split())
	G[u].append(v)
visit = bfs1(s)
h = []
for i in range(1, n + 1):
	if not visit[i]:
		c = bfs2(i)
		heapq.heappush(h, (-c, i))
ans = 0
while h:
	(c, i) = heapq.heappop(h)
	if not visit[i]:
		ans += 1
		bfs3(i)
print(ans)
