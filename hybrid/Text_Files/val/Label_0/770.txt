n = int(input())
l = list(map(int, input().split()))
res = []
for i in range(n - 1):
	res.append(abs(l[i] - l[i + 1]))
temp = list(set(res))
if len(l) == 1:
	print(0)
elif len(l) == 2:
	if abs(l[0] - l[1]) % 2 == 0:
		print(abs(l[0] - l[1]) // 2)
	else:
		print(abs(l[0] - l[1]))
elif len(set(l)) == 1:
	print(0)
elif len(set(l)) == 2:
	temp = list(set(l))
	if abs(temp[0] - temp[1]) % 2 == 0:
		print(abs(temp[0] - temp[1]) // 2)
	else:
		print(abs(temp[0] - temp[1]))
elif len(set(l)) == 3:
	temp = list(set(l))
	temp.sort()
	if temp[0] - temp[1] == temp[1] - temp[2]:
		print(abs(temp[0] - temp[1]))
	else:
		print(-1)
else:
	print(-1)
