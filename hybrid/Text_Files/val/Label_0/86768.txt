class TaskC {
    long[] S;
    long to1[][];
    long toN[][];

    public void solve(int testNumber, InputReader in, PrintWriter out) {
        int M = in.nextInt();
        int N = in.nextInt();

        N = Math.min(N, 77);
        S = new long[N + 1];
        S[0] = 1;
        S[1] = 2;
        for(int i = 2; i <= N; i++)
            S[i] = S[i - 1] + S[i - 2];

        to1 = new long[N + 1][3];
        toN = new long[N + 1][3];

        for(int i = 0; i < M; i++) {
            long a = in.nextLong();
            long b = in.nextLong();

            if(a > b) {
                long aux = a; a = b; b = aux;
            }
            for(int j = 0; j <= N; j++) {
                Arrays.fill(to1[j], -1);
                Arrays.fill(toN[j], -1);
            }

            long dist = go(a, b, N);
            out.println(dist);
        }
    }

    private long go(long a, long b, int n) {
        long ret = Long.MAX_VALUE;

        if(n == 0) return 0;
        if(n == 1) {
            if(a == b)
                return 0;
            return 1;
        }

        if(a <= S[n - 1] && b <= S[n - 1]) {
            ret = Math.min(ret, go(a, b, n - 1));
            long amin = Math.min(first(0, a, n - 1), last(0, a, n - 1));
            long bmin = Math.min(first(1, b, n - 1), last(1, b, n - 1));
            ret = Math.min(ret, amin + bmin + 2);
        }
        else if(a > S[n - 1] && b > S[n - 1]) {
            ret = go(a - S[n - 1], b - S[n - 1], n - 2);
        }
        else {
            long amin = Math.min(first(0, a, n - 1), last(0, a, n - 1));
            ret = amin + 1 + first(1, b - S[n - 1], n - 2);
        }

        return ret;
    }

    private long last(int t, long x, int n) {
        if(n == 0) return 0;
        if(n == 1) {
            if(x == 2)
                return 0;
            return 1;
        }

        if(toN[n][t] != -1)
            return toN[n][t];

        long ret = Long.MAX_VALUE;

        if(x <= S[n - 1]) {
            long amin = Math.min(first(t, x, n - 1), last(t, x, n - 1));
            ret = amin + 1 + last(2, 1, n - 2);
        }
        else {
            ret = last(t, x - S[n - 1], n - 2);
        }

        toN[n][t] = ret;

        return ret;
    }


    private long first(int t, long x, int n) {
        if(n == 0) return 0;
        if(n == 1) {
            if(x == 1)
                return 0;
            return 1;
        }

        if(to1[n][t] != -1)
            return to1[n][t];

        long ret = Long.MAX_VALUE;
        if(x <= S[n - 1]) {
            ret = Math.min(ret, first(t, x, n - 1));
            long bmin = Math.min(first(t, x, n - 1), last(t, x, n - 1));
            ret = Math.min(ret, bmin + 2);
        }
        else {
            ret = 1 + first(t, x - S[n - 1], n - 2);
        }

        to1[n][t] = ret;

        return ret;
    }
}

class InputReader {
    private InputStream stream;
    private byte[] buf = new byte[1024];
    private int curChar;
    private int numChars;

    public InputReader(InputStream stream) {
        this.stream = stream;
    }

    public int read() {
        if (numChars == -1)
            throw new InputMismatchException();
        if (curChar >= numChars) {
            curChar = 0;
            try {
                numChars = stream.read(buf);
            } catch (IOException e) {
                throw new InputMismatchException();
            }
            if (numChars <= 0)
                return -1;
        }
        return buf[curChar++];
    }

    public int nextInt() {
        return Integer.parseInt(nextString());
    }

    public long nextLong() {
        return Long.parseLong(nextString());
    }

    public String nextString() {
        int c = read();
        while (isSpaceChar(c))
            c = read();
        StringBuffer res = new StringBuffer();
        do {
            res.appendCodePoint(c);
            c = read();
        } while (!isSpaceChar(c));
        return res.toString();
    }

    private boolean isSpaceChar(int c) {
        return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
    }

    }
