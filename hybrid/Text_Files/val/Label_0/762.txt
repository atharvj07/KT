from collections import defaultdict
t = int(input())
for _ in range(t):
	n = int(input())
	ans = 0
	arr = list(map(int, input().strip().split()))
	cnt = [[0 for i in range(201)] for j in range(n)]
	freq = defaultdict(int)
	ind = defaultdict(list)
	for j in range(n):
		freq[arr[j]] += 1
		for i in range(201):
			cnt[j][i] = (cnt[j - 1][i] if j > 0 else 0) + (i == arr[j])
		ans = max(ans, freq[arr[j]])
		ind[arr[j]].append(j)
	for a in range(201):
		for sz in range(1, n + 1):
			if 2 * sz <= len(ind[a]):
				r = ind[a][sz - 1]
				l = ind[a][len(ind[a]) - sz]
				bb = 0
				for b in range(201):
					if b != a:
						bb = max(bb, cnt[l][b] - cnt[r][b])
				ans = max(ans, bb + 2 * sz)
			else:
				break
	print(ans)
