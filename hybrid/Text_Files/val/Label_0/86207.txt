#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a = 0, b = 0, x, y;
  int i;
  for (i = 0; i < 3; i++) {
    cin >> x;
    a += x;
  }
  for (i = 0; i < 3; i++) {
    cin >> y;
    b += y;
  }
  cin >> n;
  int c = a / 5 + b / 10;
  if (a % 5) c++;
  if (b % 10) c++;
  if (c <= n)
    cout << "YES";
  else
    cout << "NO";
  return 0;
}
