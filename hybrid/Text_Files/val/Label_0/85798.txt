#include<bits/stdc++.h>
using namespace std;

#define int long long

signed main(){
  int N;
  vector<vector<int>> A;

  cin>>N;

  A.resize(N, vector<int>(N));

  if(N == 2) {
    cout<<-1<<endl;
    return 0;
  }

  for(int i = 0; i < 3; i++){
    A[i][0] = (i+1)%3;
    A[i][1] = (i+2)%3;
  }

  for(int i = 3; i < N; i++){
    //cout<<"i = "<<i<<endl;
    for(int j = 0; j < i; j++){
      A[j][i-1] = i;
      //cout<<"j = "<<j<<" i = "<<i<<endl;
    }
     for(int j = 0; j <= i; j++){
      A[i][j] = j;
    }
     swap(A[i-1][i-2], A[i-1][i-1]);
  }

  for(int i = 0; i < N; i++){
    for(int j = 0; j < N-1; j++){
      if(j) cout<<" ";
      cout<<A[i][j]+1;
    }
    cout<<endl;
  }

  return 0;
}
