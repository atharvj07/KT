#include <bits/stdc++.h>
using namespace std;
const int INF = 0x3f3f3f3f;
template <typename X>
inline bool minimize(X& p, X q) {
  if (p <= q) return 0;
  p = q;
  return 1;
}
template <typename X>
inline bool maximize(X& p, X q) {
  if (p >= q) return 0;
  p = q;
  return 1;
}
char Map[105][10005];
int main() {
  int N, M;
  scanf("%d%d", &N, &M);
  for (int i = 1; i <= N; ++i)
    scanf(" %s", Map[i] + 1), Map[i][0] = Map[i][M + 1] = '#';
  long long ans = 0;
  int cur = 1;
  const bool Right = true, Left = false;
  bool dir = Right;
  for (int i = 1; i < N; ++i) {
    if (Map[i + 1][cur] == '.') {
      ++ans;
      continue;
    }
    int left = cur, right = cur;
    bool find = false;
    while (!find) {
      if (Map[i][left - 1] == '#' && Map[i][right + 1] == '#') {
        cout << "Never" << endl;
        return 0;
      }
      ans += right - left + 1;
      if (dir == Right) {
        while (Map[i][right + 1] == '.') {
          ++ans;
          if (Map[i + 1][++right] == '.') {
            find = true;
            cur = right;
            break;
          }
        }
        if (Map[i][right + 1] == '+') Map[i][right + 1] = '.';
      } else {
        while (Map[i][left - 1] == '.') {
          ++ans;
          if (Map[i + 1][--left] == '.') {
            find = true;
            cur = left;
            break;
          }
        }
        if (Map[i][left - 1] == '+') Map[i][left - 1] = '.';
      }
      if (!find) dir = !dir;
    }
  }
  cout << ans << endl;
  return 0;
}
