def solve(a):
	d = {}
	for i in a:
		if i in d:
			d[i] += 1
		else:
			d[i] = 1
	return d

	n = int(input())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))

				i += 1
			while j >= 0 and a[j] == b[j]:

			elif a[i] > b[i] and a[j] < b[j]:
				a[i] -= 1

				ans.append([i + 1, j + 1])
			elif a[i] > b[i] and a[j] > b[j]:
				j -= 1
