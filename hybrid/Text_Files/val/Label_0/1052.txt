t = int(input())
for x in range(t):
	n = int(input())
	a = list(map(int, input().split(' ')))[:n]
	b = list(map(int, input().split(' ')))[:n]
	if n == 1:
		if a[0] != b[0]:
			print('NO')
		else:
			print('YES')
	else:
		(x, y) = (n + 1, n + 1)
		for i in range(n):
			if a[i] == 1:
				x = i
				break
		for i in range(n):
			if a[i] == -1:
				y = i
				break
		for i in range(0, n):
			bool = 0
			if a[i] == b[i]:
				bool = 1
			elif a[i] < b[i]:
				if i > x:
					bool = 1
			elif i > y:
				bool = 1
			if bool == 0:
				break
		if bool == 0:
			print('NO')
		elif bool == 1:
			print('YES')
