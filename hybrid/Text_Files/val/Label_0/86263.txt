#include <bits/stdc++.h>
using namespace std;
int N, M, K, deg[100001];
long long ans[100001], sum;
vector<int> in[100001];
int main() {
  cin >> N >> M;
  for (int i = 1; i <= M; i++) {
    int u, v;
    cin >> u >> v;
    deg[u]++, deg[v]++;
    if (u > v) swap(u, v);
    in[u].push_back(v);
  }
  for (int i = 1; i <= N; i++)
    sum += ans[i] = 1ll * in[i].size() * (deg[i] - in[i].size());
  cout << sum << endl;
  cin >> K;
  for (int i = 1; i <= K; i++) {
    int x;
    cin >> x;
    sum -= ans[x];
    ans[x] = 0;
    for (auto y : in[x])
      sum -= ans[y], in[y].push_back(x),
          sum += ans[y] = 1ll * in[y].size() * (deg[y] - in[y].size());
    in[x].clear();
    cout << sum << endl;
  }
  return 0;
}
