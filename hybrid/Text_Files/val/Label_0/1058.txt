	for i in range(n):

			ans = max(ans, l)
			if i + 2 < n or i - 2 >= 0:
				ans = max(ans, l + 1)
		elif len(A[i]) == 1:
			if i + 1 < n and i - 1 >= 0:
				ans = max(ans, len(A[i - 1]) + len(A[i + 1]))
			if (i + 3 < n or i - 3 >= 0) and (i + 1 < n and i - 1 >= 0):
				ans = max(ans, len(A[i - 1]) + len(A[i + 1]) + 1)
