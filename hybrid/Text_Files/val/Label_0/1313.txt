for _ in range(int(input())):
	x = int(input())
	y = list(input())
	diff = 0
	for i in range(x // 2):
		if y[i] != y[x - i - 1]:
			if y[i] == '0':
				y[i] = '1'
			else:
				y[x - i - 1] = '1'
			diff += 1
	n = y.count('0')
	if diff == 0:
		if n == 0:
			print('DRAW')
		elif n % 2 == 0 or n == 1:
			print('BOB')
		else:
			print('ALICE')
	elif diff == 1 and n == 1:
		print('DRAW')
	else:
		print('ALICE')
