for _ in range(int(input())):
	n = int(input())
	q = []
	for i in range(n):
		(a, b) = list(map(int, input().split()))
		q.append((a, i, b))
	if n == 1:
		print(1)
		continue
	q.sort()
	goFrom = 0
	goFromTime = q[0][0]
	now = goFromTime + abs(q[0][2])
	goTo = q[0][2]
	ans = 0
	if goTo > goFrom:
		willBe = goFrom + (q[1][0] - q[0][0]) * 1
		if willBe >= goTo:
			ans += 1
	else:
		willBe = goFrom + (q[1][0] - q[0][0]) * -1
		if willBe <= goTo:
			ans += 1
	for i in range(1, n):
		if now <= q[i][0]:
			if i == n - 1:
				ans += 1
				break
			goFrom = goTo
			goTo = q[i][2]
			if goTo > goFrom:
				willBe = min(goFrom + (q[i + 1][0] - q[i][0]) * 1, goTo)
				if willBe >= goTo:
					ans += 1
			else:
				willBe = max(goFrom + (q[i + 1][0] - q[i][0]) * -1, goTo)
				if willBe <= goTo:
					ans += 1
			goFromTime = q[i][0]
			now = goFromTime + abs(goFrom - q[i][2])
		else:
			if i == n - 1:
				if goTo > goFrom:
					pN = goFrom + (q[i][0] - goFromTime) * 1
					willBe = goTo
					if pN <= q[i][2] and willBe >= q[i][2]:
						ans += 1
				else:
					pN = goFrom + (q[i][0] - goFromTime) * -1
					willBe = goTo
					if pN >= q[i][2] and willBe <= q[i][2]:
						ans += 1
				break
			if goTo > goFrom:
				pN = goFrom + (q[i][0] - goFromTime) * 1
				willBe = min(goTo, goFrom + (q[i + 1][0] - goFromTime) * 1)
				if pN <= q[i][2] and willBe >= q[i][2]:
					ans += 1
			else:
				pN = goFrom + (q[i][0] - goFromTime) * -1
				willBe = max(goTo, goFrom + (q[i + 1][0] - goFromTime) * -1)
				if pN >= q[i][2] and willBe <= q[i][2]:
					ans += 1
	print(ans)
