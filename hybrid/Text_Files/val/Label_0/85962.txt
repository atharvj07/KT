#include <bits/stdc++.h>
using namespace std;
void Fast_Read_Out() {
  ios_base::sync_with_stdio(0);
  cin.tie(), cout.tie();
}
void Random() {
  unsigned int seed;
  asm("rdtsc" : "=A"(seed));
  srand(seed);
}
unsigned int Time() {
  unsigned int time = clock() / 1000.00;
  return time;
}
const int mod = (int)1e9 + 7;
const int N = int(4e5) + 123;
vector<int> g[N];
int timer, used[N];
int tin[N], tout[N], a[N], level[N];
int t[N * 4], add[4 * N], T[N * 4], Add[N * 4];
void push(int tl, int tr, int v) {
  if (tl != tr) {
    add[v + v] += add[v];
    add[v + v + 1] += add[v];
  }
  t[v] += 1LL * (tr - tl + 1) * add[v];
  add[v] = 0;
}
void Push(int tl, int tr, int v) {
  if (tl != tr) {
    Add[v + v] += Add[v];
    Add[v + v + 1] += Add[v];
  }
  T[v] += 1LL * (tr - tl + 1) * Add[v];
  Add[v] = 0;
}
void Update(int tl, int tr, int v, int l, int r, int val) {
  push(tl, tr, v);
  if (l > tr || tl > r) return;
  if (l <= tl && tr <= r) {
    add[v] += val;
    push(tl, tr, v);
    return;
  }
  int tm = (tl + tr) / 2;
  Update(tl, tm, v + v, l, r, val);
  Update(tm + 1, tr, v + v + 1, l, r, val);
  t[v] = t[v + v] + t[v + v + 1];
}
void Update_1(int tl, int tr, int v, int l, int r, int val) {
  Push(tl, tr, v);
  if (l > tr || tl > r) return;
  if (l <= tl && tr <= r) {
    Add[v] += val;
    Push(tl, tr, v);
    return;
  }
  int tm = (tl + tr) / 2;
  Update_1(tl, tm, v + v, l, r, val);
  Update_1(tm + 1, tr, v + v + 1, l, r, val);
  T[v] = T[v + v] + T[v + v + 1];
}
void dfs(int v, int lvl = 0) {
  used[v] = 1;
  level[v] = lvl;
  tin[v] = ++timer;
  for (auto to : g[v])
    if (!used[to]) dfs(to, lvl ^ 1);
  tout[v] = ++timer;
}
int Get(int tl, int tr, int v, int l, int r) {
  push(tl, tr, v);
  if (tl > r || tr < l) return 0;
  if (l <= tl && tr <= r) return t[v];
  int tm = (tl + tr) / 2;
  return Get(tl, tm, v + v, l, r) + Get(tm + 1, tr, v + v + 1, l, r);
}
int Get_1(int tl, int tr, int v, int l, int r) {
  Push(tl, tr, v);
  if (tl > r || tr < l) return 0;
  if (l <= tl && tr <= r) return T[v];
  int tm = (tl + tr) / 2;
  return Get_1(tl, tm, v + v, l, r) + Get_1(tm + 1, tr, v + v + 1, l, r);
}
int main() {
  Fast_Read_Out();
  Random();
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i < n; i++) {
    int x, y;
    cin >> x >> y;
    g[x].push_back(y);
    g[y].push_back(x);
  }
  dfs(1);
  for (int i = 1; i <= m; i++) {
    char c;
    cin >> c;
    if (c == '1') {
      int x, val;
      cin >> x >> val;
      if (level[x])
        Update(1, n + n, 1, tin[x], tout[x], val),
            Update_1(1, n + n, 1, tin[x], tout[x], -val);
      else
        Update(1, n + n, 1, tin[x], tout[x], -val),
            Update_1(1, n + n, 1, tin[x], tout[x], val);
    } else {
      int x;
      cin >> x;
      if (level[x])
        cout << Get(1, n + n, 1, tin[x], tin[x]) + a[x] << endl;
      else
        cout << Get_1(1, n + n, 1, tin[x], tin[x]) + a[x] << endl;
    }
  }
}
