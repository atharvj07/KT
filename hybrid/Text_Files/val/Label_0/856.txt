N = int(input())
Arr = [int(_) for _ in input().split()]
dic = {}
for i in range(N):
	dic[Arr[i]] = i
curr_i = -1
main_index = 0
for j in range(N, 0, -1):
	k = dic[j]
	if k > curr_i:
		curr_i = k
		print('\n' * (k - main_index), end='')
		print(j, end=' ')
		main_index = k
	elif k < curr_i:
		print(j, end=' ')
