#include<bits/stdc++.h>
using namespace std;
int cumsum[200005];
int a[200005];
int main()
{
	int t;
	scanf("%d",&t);
	while(t--)
	{
		int n;
		scanf("%d",&n);
		for(int i=0;i<n;i++)
		{
			scanf("%d",&a[i]);
		}
		for(int i=0;i<n;i++)
		{
			cumsum[i] = 0;
		}

		for(int i=0;i<n;i++)
		{
			if(a[i] == 0)
				continue;
			cumsum[i+1]--;
			cumsum[max(0,i-a[i] + 1)]++;
		}

		for(int i=0;i<n;i++)
		{
			if(i != 0)
				cumsum[i]+=cumsum[i-1];
			if(cumsum[i] == 0)
			{
				printf("0 ");
			}
			else
			{ 
				printf("1 ");
			}
		}
		printf("\n");
	}
	return 0;
}
