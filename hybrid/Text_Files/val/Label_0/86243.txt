#include <bits/stdc++.h>
using namespace std;
int t;
int cnt[(int)(1e6 + 10)];
int maxx = 0;
bool ok(int n, int m, pair<int, int> p) {
  int dcnt[(int)(1e6 + 10)];
  for (int i = 1; i <= t; ++i) dcnt[i] = cnt[i];
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) {
      int dist = (p.first - i) < 0 ? -(p.first - i) : (p.first - i);
      dist += (p.second - j) < 0 ? -(p.second - j) : (p.second - j);
      if (dist == 0) continue;
      if (--dcnt[dist] < 0) return false;
    }
  }
  for (int i = 1; i <= t; ++i)
    if (dcnt[i]) return false;
  return true;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin >> t;
  for (int i = 1; i <= t; ++i) {
    int x;
    cin >> x;
    if (!x) continue;
    ++cnt[x];
    if (x > maxx) maxx = x;
  }
  int f;
  for (int i = 1; i <= t; ++i)
    if (cnt[i] != 4 * i) {
      f = i;
      break;
    }
  int sqrtt = sqrt(t);
  for (int n = 1; n <= sqrtt; ++n) {
    if (t % n) continue;
    int m = t / n;
    vector<pair<int, int> > valid;
    for (int i = 1; i <= n; ++i) {
      for (int j = 1; j <= m; ++j) {
        int sum = (n - i) < 0 ? -(n - i) : (n - i);
        sum += (m - j) < 0 ? -(m - j) : (m - j);
        if (sum != maxx) continue;
        if (i + f - 1 > n) continue;
        if (i - (f - 1) < 1) continue;
        if (j + f - 1 > m) continue;
        if (j - (f - 1) < 1) continue;
        if (!((i + f - 1 == n and i - (f - 1) >= 1) or
              (j + f - 1 == m and j - (f - 1) >= 1) or
              (i + f - 1 <= n and i - (f - 1) == 1) or
              (j + f - 1 <= m and j - (f - 1) == 1)))
          continue;
        valid.push_back(make_pair(i, j));
      }
    }
    int up = valid.size() - 1;
    for (int i = 0; i <= up; ++i) {
      if (ok(n, m, valid[i])) {
        cout << n << " " << m << "\n";
        cout << valid[i].first << " " << valid[i].second << "\n";
        return 0;
      }
    }
  }
  cout << -1 << "\n";
  return 0;
}
