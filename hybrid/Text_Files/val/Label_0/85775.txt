#include <bits/stdc++.h>
#define int long long

using namespace std;

int n=0,k=0;
int a[200001];
int s[200001];
map<int, int> Map;

signed main()
{
	cin >> n >> k;
	for (int i=1; i<=n; i++)
	{
		cin >> a[i];
	}
	s[0] = 0;
	for (int i=1; i<=n; i++)
	{
		s[i] = s[i-1]+a[i];
	}
	Map[0]++;
	int res=0;
	for (int i=1; i<=n; i++)
	{
		if (i-k>=0)	
		{
			Map[(s[i-k]-(i-k))%k]--;		
		}	
		res+=Map[(s[i]-i)%k];
		Map[(s[i]-i)%k]++;	
	}		
	cout << res;
	return 0;
}
