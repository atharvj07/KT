import java.io.*;
import java.util.*;

public class D {

    public static StringTokenizer tokenizer = null;
    public static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

    public static String nextToken() throws IOException {
        if (tokenizer == null || !tokenizer.hasMoreTokens()) {
            tokenizer = new StringTokenizer(reader.readLine());
        }
        return tokenizer.nextToken();
    }

    public static int nextInt() throws IOException {
        return Integer.parseInt(nextToken());
    }

    public static long nextLong() throws IOException {
        return Long.parseLong(nextToken());
    }

    public static double nextDouble() throws IOException {
        return Double.parseDouble(nextToken());
    }

    public static class Pair {
        long quantity;
        char symbol;

        public Pair(long quantity, char symbol) {
            this.quantity = quantity;
            this.symbol = symbol;
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (o == null || getClass() != o.getClass()) return false;

            Pair pair = (Pair) o;

            if (quantity != pair.quantity) return false;
            return symbol == pair.symbol;

        }
    }

    public static int[] zValues(long[] array) {
        int n = array.length;
        int[] z = new int[n];
        for (int i = 1, l = 0, r = 0; i < n; i++) {
            if (i <= r) {
                z[i] = Math.min(r - i + 1, z[i - l]);
            }
            while (i + z[i] < n && array[z[i]] == array[i + z[i]]) {
                z[i]++;
            }
            if (i + z[i] - 1 > r) {
                l = i;
                r = i + z[i] - 1;
            }
        }
        return z;
    }

    public static List<Integer> find(long[] text, long[] pattern) {
        long[] array = new long[text.length + pattern.length + 1];
        for (int i = 0; i < pattern.length; i++) {
            array[i] = pattern[i];
        }
        array[pattern.length] = -1;
        for (int i = 0; i < text.length; i++) {
            array[i + pattern.length + 1] = text[i];
        }
        int[] z = zValues(array);
        List<Integer> result = new ArrayList<>();
        for (int i = 0; i < text.length; i++) {
            if (z[i + pattern.length + 1] == pattern.length) {
                result.add(i);
            }
        }
        return result;
    }

    public static void main(String[] args) throws IOException {
        int n = nextInt();
        int m = nextInt();

        ArrayList<Pair> t = new ArrayList<>(n);
        ArrayList<Pair> s = new ArrayList<>(m);
        for (int i = 0; i < n; i++) {
            String token = nextToken();
            char symbol = token.charAt(token.length() - 1);
            int quantity = Integer.parseInt(token.substring(0, token.length() - 2));
            if (t.size() > 0 && t.get(t.size() - 1).symbol == symbol) {
                t.get(t.size() - 1).quantity += quantity;
            } else {
                t.add(new Pair(quantity, symbol));
            }
        }
        for (int i = 0; i < m; i++) {
            String token = nextToken();
            char symbol = token.charAt(token.length() - 1);
            int quantity = Integer.parseInt(token.substring(0, token.length() - 2));
            if (s.size() > 0 && s.get(s.size() - 1).symbol == symbol) {
                s.get(s.size() - 1).quantity += quantity;
            } else {
                s.add(new Pair(quantity, symbol));
            }
        }

        if (s.size() == 1) {
            long result = 0;
            long quantity = s.get(0).quantity;
            char symbol = s.get(0).symbol;
            for (Pair pair: t) {
                if (pair.symbol == symbol && pair.quantity >= quantity) {
                    result += pair.quantity - quantity + 1;
                }
            }
            System.out.println(result);
            return;
        }

        long big = 1_000_000_000_000L;
        long[] text = new long[t.size()];
        for (int i = 0; i < t.size(); i++) {
            char c = t.get(i).symbol;
            int cNum = c - 'a';
            long quantity = t.get(i).quantity;
            text[i] = cNum * big + quantity;
        }
        long[] pattern = new long[s.size() - 2];
        for (int i = 0; i < s.size() - 2; i++) {
            char c = s.get(i + 1).symbol;
            int cNum = c - 'a';
            long quantity = s.get(i + 1).quantity;
            pattern[i] = cNum * big + quantity;
        }

        List<Integer> places = find(text, pattern);
        char cFirst = s.get(0).symbol, cLast = s.get(s.size() - 1).symbol;
        long qFirst = s.get(0).quantity, qLast = s.get(s.size() - 1).quantity;
        long result = 0;
        for (int index: places) {
            if (index == 0 || index == t.size() - pattern.length) {
                continue;
            }
            char cRealFirst = t.get(index - 1).symbol, cRealLast = t.get(index + pattern.length).symbol;
            long qRealFirst = t.get(index - 1).quantity, qRealLast = t.get(index + pattern.length).quantity;
            if (cFirst == cRealFirst && cLast == cRealLast && qFirst <= qRealFirst && qLast <= qRealLast) {
                result++;
            }
        }
        System.out.println(result);
    }

}
