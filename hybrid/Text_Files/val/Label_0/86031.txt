using ul = unsigned long long;

bool solve(int n, ul k, int a) {
	if(n == 0) return true;
	if(n-1 <= 60 && k >= (1uLL << (n-1))) return false;
	DEBUG("\n  solve(%d, %llx, %d)\n", n, k, a);
	int x = n;
	for(int i = n-2; i >= 0; i--){
		if(i >= 60){
			x = 1;
			break;
		}
		ul m = 1uLL << i;
		if(k < m) {
			x = (n-1)-i;
			break;
		}
		k -= m;
	}
	DEBUG("  x = %d\n", x);
	for(int i = x-1; i >= 0; i--) printf("%d ", a+i);
	return solve(n-x, k, a+x);
}
