#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
const long long mod = 1e9 + 7;
template <class T>
inline void read(T &res) {
  char c;
  T flag = 1;
  while ((c = getchar()) < '0' || c > '9')
    if (c == '-') flag = -1;
  res = c - '0';
  while ((c = getchar()) >= '0' && c <= '9') res = res * 10 + c - '0';
  res *= flag;
}
int n;
int e, b[N], t[N << 1], nxt[N << 1];
long long w[N], ans;
map<long long, int> mp[N];
void add(int u, int v) {
  t[++e] = v;
  nxt[e] = b[u];
  b[u] = e;
}
long long gcd(long long x, long long y) { return y == 0 ? x : gcd(y, x % y); }
void dfs(int u, int fa) {
  for (map<long long, int>::iterator it = mp[fa].begin(); it != mp[fa].end();
       ++it) {
    long long tmp = gcd(it->first, w[u]);
    mp[u][tmp] += it->second;
    ans = (ans + tmp % mod * it->second % mod) % mod;
  }
  mp[u][w[u]]++;
  ans = (ans + w[u]) % mod;
  for (int i = b[u]; i; i = nxt[i]) {
    int v = t[i];
    if (v == fa) continue;
    dfs(v, u);
  }
}
int main() {
  read(n);
  for (int i = 1; i <= n; i++) read(w[i]);
  for (int i = 1; i < n; i++) {
    int x, y;
    read(x), read(y);
    add(x, y);
    add(y, x);
  }
  dfs(1, 0);
  cout << ans << endl;
  return 0;
}
