from sys import stdin
a = lambda : stdin.readline().split()
for _ in range(int(stdin.readline())):
	n = int(stdin.readline())
	lst = [*map(int, a())]
	if n < 5:
		print(0, 0, 0)
		continue
	(g, s, b, item) = (0, 0, 0, n // 2)
	for i in range(n // 2 - 1, -1, -1):
		if lst[i] == lst[i + 1]:
			item -= 1
		else:
			break
	(cout, level) = (1, 1)
	for (i, x) in enumerate(lst[1:item]):
		if x == lst[i]:
			cout += 1
		elif level == 1:
			g += cout
			(cout, level) = (1, 2)
		elif level == 2:
			s += cout
			cout = 1
			if s > g:
				level = 3
		else:
			b += cout
			cout = 1
	if level == 3:
		b += cout
		if s > g and b > g:
			if g != 0 and s != 0 and (b != 0):
				print(g, s, b)
			else:
				print(0, 0, 0)
		else:
			print(0, 0, 0)
	else:
		print(0, 0, 0)
