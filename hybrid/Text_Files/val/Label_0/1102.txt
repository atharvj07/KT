def f(n, a):
	a = list(map(lambda s: s - 1, a))
	ar = [0] * (n + 1)
	l = 0
	ans = [1]
	hi = n - 1
	cnt = 0
	for i in a:
		if i == hi:
			ar[i] = 1
			hi -= 1
			while ar[hi] == 1:
				hi -= 1
				cnt -= 1
		else:
			ar[i] = 1
			cnt += 1
		ans.append(cnt + 1)
	return ans
a = int(input())
ls = list(map(int, input().strip().split()))
print(*f(a, ls))
