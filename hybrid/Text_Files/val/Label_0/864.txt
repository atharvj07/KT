n = int(input())
s = str(input())
result = ''
counter = 0
for i in range(s.__len__()):
	if s[i - 1] == '0' and s[i] == '0':
		if counter > 0:
			result += str(counter)
			counter = 0
		result += '0'
		continue
	if s[i] == '1':
		counter = counter + 1
	elif s[i] == '0':
		if counter > 0:
			result += str(counter)
		counter = 0
result += str(counter)
print(result)
