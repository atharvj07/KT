#include <iostream>
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll dp[200000],n,a[200000],b[200000],MOD=1000000007,ans;
ll f(ll x,ll y){
    if(y<=0)return 1;
    if(y%2==1)return f(x,y-1)*x%MOD;
    return f(x,y/2)*f(x,y/2)%MOD;
}
int main(void){
    cin>>n;
    for(int i=0;i<n;i++)cin>>dp[i];
    sort(dp,dp+n);
    for(ll i=0LL;i<n;i++){
        a[i]=f(2,i);
        b[n-i-1]=(f(2,i)+i*f(2,i-1)%MOD)%MOD;
    }
    for(int i=0;i<n;i++){
        ans+=dp[i]*a[i]%MOD*b[i]%MOD;
        ans%=MOD;
    }
    cout<<ans*f(2,n)%MOD<<endl;
}
