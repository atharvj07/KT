import java.util.*;
import java.lang.*;


public class Main {

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int l = scan.nextInt();
        int r = scan.nextInt();
        int[] s = new int[100001];
        for(int i=1;i<=n;i++){
            int t=scan.nextInt();
            s[i]=t;
        }
        boolean p = true;
        for(int i=1;i<=n;i++){
            int t=scan.nextInt();
            if(s[i]!=t && (i<l || i>r))
                p=false;
        }

        if(p)System.out.print("TRUTH");
        else System.out.print("LIE");
    }
}
