n = int(input())
arr = list(map(int, input().split()))
max1 = max(arr)
a = len(bin(max1)) - 2
lis = [0] * a
for i in range(a):
	for j in range(n):
		if arr[j] >= 2 ** i:
			x = bin(arr[j])
			x = x[2:]
			if x[-i - 1] == '1':
				lis[i] += 1
num = []
b = max(lis)
for i in range(b):
	s = 0
	for j in range(a):
		if lis[j] > 0:
			s += 2 ** j
			lis[j] -= 1
	num.append(s)
ans = 0
c = len(num)
for i in range(c):
	ans += num[i] ** 2
print(ans)
