const int MAXN = 2e5 + 5;

ll v[2][MAXN];

ll dp[2][MAXN];

int n;

void calc(int ind) {
	ll lim = 1ll << 34;
	dp[ind][0] = dp[ind][1] = 0ll;
	ll last = 0;
	REPN(i, 2, n) {
		dp[ind][i] = dp[ind][i - 1];
		int j;
		ll mul = 0ll;
		int new_last = last;
		for(j = i - 1; j > last && v[ind][j] < v[ind][j + 1]; j--) {
			for(; v[ind][j] < lim && v[ind][j] < v[ind][j + 1]; v[ind][j] *= 4, dp[ind][i] += 2);
			if (v[ind][j] >= lim) {
				new_last = max(new_last, j);
			}
			if (v[ind][j] < v[ind][j + 1]) {
				mul++;
			}
			dp[ind][i] += 2ll * mul;
		}
		if (last >= 1 && v[ind][last] < v[ind][last + 1]) {
			mul++;
		}
		dp[ind][i] += last * 2ll * mul;
		last = new_last;
	}
}

void solve(){
	scanf("%d", &n);
	REPN(i, 1, n) {
		scanf("%lld", &v[0][i]);
		v[1][n + 1 - i] = v[0][i];
	}
	REP(i, 0, 2) {
		calc(i);
	}
	ll ans = INF;
	REPN(i, 0, n) {
		ans = min(ans, i + dp[0][i] + dp[1][n - i]);
	}
	/*REP(j, 0, 2) {

}
