import java.io.*;
import java.util.*;

public class CF1267J extends PrintWriter {
	CF1267J() { super(System.out); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF1267J o = new CF1267J(); o.main(); o.flush();
	}

	void main() {
		int t = sc.nextInt();
		while (t-- > 0) {
			int n = sc.nextInt();
			int[] kk = new int[n];
			for (int i = 0; i < n; i++) {
				int c = sc.nextInt() - 1;
				kk[c]++;
			}
			int cnt = 0;
			for (int c = 0; c < n; c++)
				if (kk[c] > 0)
					kk[cnt++] = kk[c];
			kk = Arrays.copyOf(kk, cnt);
			int[] dd = new int[n + 3];
			int s_, s = -1, m;
			for (int h = 0; h < cnt; h++) {
				int k = kk[h];
				// a * s + b * (s - 1) == k
				// (a + b) * s == k + b
				// m * s == k + b, where 0 <= b <= m
				s_ = k + 2;
				for (m = 1; ; m++) {
					s = (k + m - 1) / m;
					if (k % m == 0)
						if (s_ > s + 1) {
							s_ = s + 1;
							dd[s_]++; dd[s_ + 1]--;
						}
					if (s_ > s) {
						s_ = s;
						dd[s_]++; dd[s_ + 1]--;
					}
					if (m >= s_) {
						dd[1]++; dd[s_]--;
						break;
					}
				}
			}
			for (s = 2; s <= n + 1; s++)
				dd[s] += dd[s - 1];
			// dd[1] == cnt
			s = n + 1;
			while (dd[s] != cnt)
				s--;
			m = 0;
			for (int h = 0; h < cnt; h++) {
				int k = kk[h];
				m += (k + s - 1) / s;
			}
			println(m);
		}
	}
}
