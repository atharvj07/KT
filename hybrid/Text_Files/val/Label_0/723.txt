def readNums(type=int):
	return list(map(type, input().split()))
(n, h, a, b, k) = readNums()
for _ in range(k):
	(ta, fa, tb, fb) = readNums()
	if ta == tb:
		print(abs(fa - fb))
	else:
		res = abs(tb - ta)
		if fa > b:
			print(fa - b + res + abs(fb - b))
		elif fa < a:
			print(a - fa + res + abs(fb - a))
		else:
			print(res + abs(fa - fb))
