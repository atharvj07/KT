for t in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	c = [b[i] - a[i] for i in range(n)]
	im1 = ip1 = n - 1
	if -1 in a:
		im1 = a.index(-1)
	if 1 in a:
		ip1 = a.index(1)
	m = min(im1, ip1)
	if min(c[:im1 + 1]) < 0 or max(c[:ip1 + 1]) > 0:
		print('NO')
	else:
		print('YES')
