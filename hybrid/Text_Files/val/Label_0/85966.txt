#include <bits/stdc++.h>
using namespace std;
long long a[110];
int main() {
  long long n, m, i, l = 1, r, mid, j;
  scanf("%lld%lld", &m, &n);
  r = m;
  for (i = 0; i < n; i++) {
    printf("1\n");
    fflush(stdout);
    scanf("%lld", &a[i]);
    if (a[i] == 0) break;
  }
  if (i == n) {
    j = 0;
    while (l <= r) {
      mid = (l + r) >> 1;
      printf("%lld\n", mid);
      fflush(stdout);
      scanf("%lld", &i);
      if (i == 0) break;
      i *= a[j];
      if (i == -1)
        r = mid - 1;
      else
        l = mid + 1;
      j = (j + 1) % n;
    }
  }
  return 0;
}
