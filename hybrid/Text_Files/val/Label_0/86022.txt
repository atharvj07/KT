#include <bits/stdc++.h>
using namespace std;
int main() {
  string str;
  cin >> str;
  int arr[str.length()], h = str.length() - 1, t = 0;
  for (int i = 0; i < str.length(); i++) {
    if (str[i] == 'l') {
      arr[h--] = i + 1;
    } else {
      arr[t++] = i + 1;
    }
  }
  for (int i = 0; i < str.length(); i++) cout << arr[i] << "\n";
  return 0;
}
