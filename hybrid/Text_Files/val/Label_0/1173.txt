(n, m) = [int(x) for x in input().split()]
q1 = [int(x) for x in input().split()]
q2 = [int(x) for x in input().split()]
a1 = a2 = 0
it1 = it2 = -1
ans = 0
while it1 < n - 1 or it2 < m - 1:
	if a1 <= a2:
		it1 += 1
		a1 += q1[it1]
	else:
		it2 += 1
		a2 += q2[it2]
	if a1 == a2:
		a1 = a2 = 0
		ans += 1
print(ans)
