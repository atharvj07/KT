#include <bits/stdc++.h>
int main() {
  long long p, q, a[100];
  int n, i;
  scanf("%lld%lld", &p, &q);
  scanf("%d", &n);
  for (i = 0; i < n; i++) {
    scanf("%lld", &a[i]);
  }
  for (i = 0; i < n; i++) {
    if (q == 0 || p / q < a[i]) {
      break;
    }
    p = p - a[i] * q;
    long long t;
    t = p;
    p = q;
    q = t;
  }
  if (i == n && q == 0)
    puts("YES");
  else
    puts("NO");
  return 0;
}
