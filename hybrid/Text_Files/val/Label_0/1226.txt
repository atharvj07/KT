from sys import stdin, stdout
import math
t = int(stdin.readline())
arr = list(map(int, stdin.readline().split()))
arr.sort()
k = 0
res = []
count = 0
if t % 2 != 0:
	k = (t + 1) // 2
	i = 0
	while i < k - 1:
		if arr[i] < arr[i + k - 1]:
			count += 1
		if i == 0:
			res.append(arr[i + k - 1])
		res.append(arr[i])
		res.append(arr[i + k])
		i += 1
else:
	k = t // 2
	i = 0
	while i < k - 1:
		if arr[i] < arr[i + k - 1] or (i == k - 2 and arr[i] < arr[i + k]):
			count += 1
		if arr[i] >= arr[i + k - 1] and (i == k - 2 and arr[i] < arr[i + k]):
			if i == 0:
				res.append(arr[i + k - 1])
			res.append(arr[i + k])
			res.append(arr[i])
			res.append(arr[i + k + 1])
		else:
			if i == 0:
				res.append(arr[i + k - 1])
			res.append(arr[i])
			res.append(arr[i + k])
			if i == k - 2:
				res.append(arr[i + k + 1])
		i += 1
print(count)
if count == 0:
	print(' '.join((str(i) for i in arr)))
else:
	print(' '.join((str(i) for i in res)))
