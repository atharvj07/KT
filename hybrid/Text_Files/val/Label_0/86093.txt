#include <bits/stdc++.h>
using namespace std;
long long int gcd(long long int a, long long int b) {
  if (b == 0) {
    return a;
  }
  return gcd(b, a % b);
}
vector<long long int> graph[100010];
long long int arr1[100010];
long long int ans = 0;
void DFS(long long int start, long long int parent,
         map<long long int, long long int> parmap) {
  long long int size, store, gcd1;
  map<long long int, long long int> childgcd;
  for (auto it : parmap) {
    gcd1 = gcd(arr1[start], it.first);
    childgcd[gcd1] += it.second;
  }
  childgcd[arr1[start]]++;
  for (auto it : childgcd) {
    store = it.second * it.first;
    if (store >= 1000000007) {
      store = store % 1000000007;
    }
    ans = (ans + store);
    if (ans >= 1000000007) {
      ans -= 1000000007;
    }
  }
  size = graph[start].size();
  for (long long int i = 0; i < size; i++) {
    store = graph[start][i];
    if (store == parent) {
      continue;
    }
    DFS(store, start, childgcd);
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long int vertice;
  cin >> vertice;
  for (long long int i = 1; i <= vertice; i++) {
    cin >> arr1[i];
  }
  long long int vert1, vert2;
  for (long long int i = 1; i < vertice; i++) {
    cin >> vert1 >> vert2;
    graph[vert1].push_back(vert2);
    graph[vert2].push_back(vert1);
  }
  map<long long int, long long int> waste;
  DFS(1, -1, waste);
  cout << ans;
  return 0;
}
