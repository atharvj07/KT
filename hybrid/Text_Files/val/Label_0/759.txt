from bisect import bisect
(n, m) = map(int, input().split())
a = [0] + [int(x) for x in input().split()]
b = [int(x) for x in input().split()]
for i in range(1, n + 1):
	a[i] += a[i - 1]
dorms = set(a)
for i in b:
	dorm = bisect(a, i)
	if a[dorm - 1] == i:
		dorm -= 1
	room = i - a[dorm - 1]
	print(dorm, room)
