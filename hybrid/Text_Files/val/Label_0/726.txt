from itertools import accumulate
for _ in range(int(input())):
	r = {'a': [], 'b': [], 'c': [], 'd': [], 'e': []}
	for _ in range(int(input())):
		s = input()
		for key in r:
			r[key].append(2 * s.count(key) - len(s))
	for key in r:
		if sum(r[key]) > 0:
			r[key] = len(r[key])
		else:
			r[key] = sum((x > 0 for x in accumulate(sorted(r[key], reverse=True))))
	print(max(r.values()))
