#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n, a, p, q, i, t, lc;
  cin >> p >> q >> n;
  for (i = 1; i <= n; i++) {
    cin >> a;
    if ((q == 0) || (a > p / q)) break;
    p = p - a * q;
    t = p;
    p = q;
    q = t;
  }
  if ((q != 0) | ((q == 0) && (i <= n)))
    cout << "NO";
  else
    cout << "YES";
  return 0;
}
