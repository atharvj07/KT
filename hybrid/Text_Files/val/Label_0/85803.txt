#include <bits/stdc++.h>
using namespace std;
int max(int a, int b, int c = INT_MIN, int d = INT_MIN) {
  int t = a;
  if (t < b)
    t = b;
  else if (t < c)
    t = c;
  else if (t < d)
    t = d;
  return t;
}
int min(int a, int b, int c = INT_MAX, int d = INT_MAX) {
  int t = a;
  if (t > b)
    t = b;
  else if (t > c)
    t = c;
  else if (t > d)
    t = d;
  return t;
}
void input(vector<int> &arr, int n = -1) {
  if (n = -1) n = arr.size();
  for (int c = 0; c < n; ++c) cin >> arr[c];
}
void output(vector<int> &arr, int n = -1) {
  if (n == -1) n = arr.size();
  for (int c = 0; c < n; ++c) cout << arr[c];
}
struct data {
  int a, b;
};
int comp(int g, data h) {
  if (g == h.b || g == h.a) return 1;
  return 0;
}
int main() {
  int n, m, common = -1;
  cin >> n >> m;
  std::vector<data> g(n), h(m);
  for (int c = 0; c < n; ++c) cin >> g[c].a >> g[c].b;
  for (int c = 0; c < m; ++c) cin >> h[c].a >> h[c].b;
  bool ans = true;
  for (int c = 0; c < n; ++c) {
    int resa = 0, resb = 0;
    for (int r = 0; r < m; ++r) {
      if (g[c].a == h[r].a && g[c].b == h[r].b ||
          g[c].b == h[r].a && g[c].a == h[r].b)
        continue;
      resa += comp(g[c].a, h[r]);
      resb += comp(g[c].b, h[r]);
    }
    if (resa > 0 && resb > 0) {
      cout << -1;
      return 0;
    }
    if (resa > 0) {
      if (common == -1)
        common = g[c].a;
      else if (common != g[c].a)
        ans = false;
    }
    if (resb > 0) {
      if (common == -1)
        common = g[c].b;
      else if (common != g[c].b)
        ans = false;
    }
  }
  for (int r = 0; r < m; ++r) {
    int resa = 0, resb = 0;
    for (int c = 0; c < n; ++c) {
      if (g[c].a == h[r].a && g[c].b == h[r].b ||
          g[c].b == h[r].a && g[c].a == h[r].b)
        continue;
      resa += comp(h[r].a, g[c]);
      resb += comp(h[r].b, g[c]);
    }
    if (resa > 0 && resb > 0) {
      cout << -1;
      return 0;
    }
  }
  if (ans)
    cout << common;
  else
    cout << 0;
  return 0;
}
