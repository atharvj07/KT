#include <bits/stdc++.h>
const int INF = 0x7fffffff;
using namespace std;
inline char nc() {
  static char buf[100000], *p1 = buf, *p2 = buf;
  if (p1 == p2) {
    p2 = (p1 = buf) + fread(buf, 1, 100000, stdin);
    if (p1 == p2) return EOF;
  }
  return *p1++;
}
inline int read() {
  int x = 0, f = 1;
  char CH = getchar();
  while (CH < '0' || CH > '9') {
    if (CH == '-') f = 0;
    CH = getchar();
  }
  while (CH >= '0' && CH <= '9') {
    x = (x << 3) + (x << 1) + (CH - 48);
    CH = getchar();
  }
  return f ? x : -x;
}
vector<int> v, vaux;
int solve(int l, int r, int calls) {
  if (calls == 0 || r - l <= 1) return calls;
  calls -= 2;
  int m = (l + r) / 2;
  for (int i = l; i < m; ++i) vaux[i] = v[m + i - l + (l + r) % 2];
  for (int i = r - 1; i >= m; i--) v[i] = v[l + i - m];
  for (int i = l; i < m; ++i) v[i] = vaux[i];
  calls = solve(l, m, calls);
  calls = solve(m, r, calls);
  return calls;
}
int main() {
  int n, k;
  cin >> n >> k;
  v.resize(n);
  vaux.resize(n);
  for (int i = 0; i < n; i++) v[i] = i + 1;
  if (k % 2 == 0 || solve(0, n, k - 1) != 0)
    cout << "-1\n";
  else
    for (int i = 0; i < n; i++) cout << v[i] << ' ';
  return 0;
}
