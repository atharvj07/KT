import java.util.Scanner;

    public class Main{
        public static void main(String [] args){
            Scanner sc = new Scanner(System.in);
            int n = sc.nextInt();
            int m = sc.nextInt();
            int cur = 0 , res = 1;
            for(int i = 0 ; i < n ; i++){
                int x = sc.nextInt();
                if(cur + x > m){
                    cur = x;
                    res++;
                }
                else{
                    cur += x;
                }
                
            }
            sc.close();
            System.out.println(res);
        }
    }
