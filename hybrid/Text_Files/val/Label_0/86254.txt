#include<bits/stdc++.h>

#define F first
#define S second
#define PB push_back
#define sz(s) int((s).size())
#define bit(n,k) (((n)>>(k))&1)

using namespace std;

typedef long long ll;
typedef pair<int,int> pii;

const int maxn = 2010;

int memo[4][6 * maxn];

int& f(int n, int k){
    return memo[n&3][k + 3 * maxn];
}

int main(){
    ios_base::sync_with_stdio(false); cin.tie(0); cout.tie();

    int n, m;
    cin >> n >> m;
    f(0, 0) = 1;
    for(int i = 1; i <= 3*n; i++){
	for(int j = -i; j <= i; j++){
	    f(i, j) = 0;
	    int num = 1;
	    for(int k = 1; k <= min(3, i); k++){
		f(i, j) = (1ll * f(i, j) + 1ll * num * f(i-k, j - (k == 1) + (k == 2))) % m;
		num = 1ll * num * (i-k) % m;
	    }
	}
    }
    int ans = 0;
    for(int i = 0; i <= 3*n; i++){
	ans = (ans + f(3*n, i)) % m;
    }
    return cout << ans << endl, 0;
}
