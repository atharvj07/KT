#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  string s1[n + 5], s2[m + 5];
  for (int i = 0; i < n; i++) cin >> s1[i];
  for (int i = 0; i < m; i++) cin >> s2[i];
  int x;
  cin >> x;
  for (int i = 0; i < x; i++) {
    int y;
    cin >> y;
    cout << s1[(y - 1) % n] + s2[(y - 1) % m] << endl;
  }
}
