import java.io.*;
import java.util.*;
public final class new_year_tree
{
    static BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
	static FastScanner sc=new FastScanner(br);
    static PrintWriter out=new PrintWriter(System.out);
	static Random rnd=new Random();
	static ArrayList<Integer>[] al;
	static int[] a,tin,tout,tour,lazy;
	static long[] tree;
	static int time=0;
	
	static void dfs(int u,int p)
	{
		tour[time]=u;tin[u]=time++;
		for(int x:al[u])
		{
			if(x!=p)
			{
				dfs(x,u);
			}
		}
		tout[u]=time-1;
	}
	
	static void build(int node,int s,int e)
	{
		if(s>e)
		{
			return;
		}
		if(s==e)
		{
			tree[node]=1L<<a[tour[s]];
		}
		else
		{
			int mid=(s+e)>>1;
			build(node<<1,s,mid);build(node<<1|1,mid+1,e);
			tree[node]=tree[node<<1]|tree[node<<1|1];
		}
	}
	
	static void update(int node,int s,int e,int l,int r,int val)
	{
		if(lazy[node]!=0)
		{
			tree[node]=1L<<lazy[node];
			if(s!=e)
			{
				lazy[node<<1]=lazy[node];lazy[node<<1|1]=lazy[node];
			}
			lazy[node]=0;
		}
		if(s>e || l>e || r<s)
		{
			return;
		}
		if(l<=s && r>=e)
		{
			tree[node]=(1L<<val);
			if(s!=e)
			{
				lazy[node<<1]=val;lazy[node<<1|1]=val;
			}
		}
		else
		{
			int mid=(s+e)>>1;
			update(node<<1,s,mid,l,r,val);update(node<<1|1,mid+1,e,l,r,val);
			tree[node]=tree[node<<1]|tree[node<<1|1];
		}
	}
	
	static long query(int node,int s,int e,int l,int r)
	{
		if(lazy[node]!=0)
		{
			tree[node]=1L<<lazy[node];
			if(s!=e)
			{
				lazy[node<<1]=lazy[node];lazy[node<<1|1]=lazy[node];
			}
			lazy[node]=0;
		}
		if(s>e || l>e || r<s)
		{
			return 0;
		}
		if(l<=s && r>=e)
		{
			return tree[node];
		}
		else
		{
			int mid=(s+e)>>1;
			long val1=query(node<<1,s,mid,l,r),val2=query(node<<1|1,mid+1,e,l,r);
			return (val1|val2);
		}
	}
	
	@SuppressWarnings("unchecked")
    public static void main(String args[]) throws Exception
    {
		int n=sc.nextInt(),m=sc.nextInt();a=new int[n];
		for(int i=0;i<n;i++)
		{
			a[i]=sc.nextInt();
		}
		al=new ArrayList[n];
		for(int i=0;i<n;i++)
		{
			al[i]=new ArrayList<Integer>();
		}
		for(int i=1;i<n;i++)
		{
			int u=sc.nextInt()-1,v=sc.nextInt()-1;
			al[u].add(v);al[v].add(u);
		}
		tin=new int[n];tout=new int[n];tour=new int[n];dfs(0,-1);tree=new long[4*n];lazy=new int[4*n];build(1,0,n-1);
		while(m>0)
		{
			int t=sc.nextInt();
			if(t==1)
			{
				int idx=sc.nextInt()-1,val=sc.nextInt();
				update(1,0,n-1,tin[idx],tout[idx],val);
			}
			else
			{
				int idx=sc.nextInt()-1;out.println(Long.bitCount(query(1,0,n-1,tin[idx],tout[idx])));
			}
			m--;
		}
		out.close();
    }
}
class FastScanner
{
    BufferedReader in;
    StringTokenizer st;

    public FastScanner(BufferedReader in) {
        this.in = in;
    }
	
    public String nextToken() throws Exception {
        while (st == null || !st.hasMoreTokens()) {
            st = new StringTokenizer(in.readLine());
        }
        return st.nextToken();
    }
	
	public String next() throws Exception {
		return nextToken().toString();
	}
	
    public int nextInt() throws Exception {
        return Integer.parseInt(nextToken());
    }

    public long nextLong() throws Exception {
        return Long.parseLong(nextToken());
    }

    public double nextDouble() throws Exception {
        return Double.parseDouble(nextToken());
    }
}
