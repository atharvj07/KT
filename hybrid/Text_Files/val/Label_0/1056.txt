import itertools as it
n = int(input())
A = [(trophy, len(list(gr))) for (trophy, gr) in it.groupby(input())]
if A[0][0] == 'S':
	A = A[1:]
if len(A) > 0 and A[-1][0] == 'S':
	A = A[:-1]
if len(A) == 0:
	print(0)
elif len(A) == 1:
	print(A[0][1])
elif len(A) == 3:
	if A[1][1] == 1:
		print(A[0][1] + A[2][1])
	else:
		print(max(A[0][1], A[2][1]) + 1)
else:
	best = max(map(lambda x: x[1], A[::2])) + 1
	for i in range(0, len(A) - 1, 2):
		if A[i + 1][1] == 1:
			best = max(best, A[i][1] + A[i + 2][1] + 1)
	print(best)
