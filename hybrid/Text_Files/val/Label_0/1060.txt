def solve():
	for t in range(int(input())):
		n = int(input())
		arr = list(map(int, input().split()))
		a = [[arr[i], i] for i in range(len(arr))]
		a.sort(key=lambda x: x[0], reverse=True)
		mid = (n + 2) // 2
		left = mid - 1
		right = mid + 1
		isleft = True
		ans = [None] * n
		res = 0
		for i in range(n):
			(cnt, idx) = (a[i][0], a[i][1])
			if isleft:
				ans[idx] = left
				cost = 2 * (mid - left)
				res += cnt * cost
				left -= 1
			else:
				ans[idx] = right
				cost = 2 * (right - mid)
				res += cnt * cost
				right += 1
			isleft = not isleft
		print(res)
		print(mid, end=' ')
		for ele in ans:
			print(ele, end=' ')
		print()
solve()
