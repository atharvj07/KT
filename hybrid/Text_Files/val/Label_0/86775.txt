import java.util.*;
public class modcrab
{
////heal=1
//strike=0
public static int decide(int a1,int a2,int c1,int h1,int h2)
{
if(h2-a1<=0)
return 0; //strike
if(h1-a2<=0)
return 1; //heal
return 0;//strike
}
public static void main(String[] args)
{
ArrayList<Integer> al=new<Integer> ArrayList();
Scanner in=new Scanner(System.in);
int a1;
int a2;
int c1;
int h1;
int h2;
int n=0; //num of phases required
h1=in.nextInt();
a1=in.nextInt();
c1=in.nextInt();
h2=in.nextInt();
a2=in.nextInt();
while(h1>0 && h2>0)
{
if(decide(a1,a2,c1,h1,h2)==0)
{//strike
al.add(0);
h2=h2-a1;
h1=h1-a2;
}
else
{//heal
al.add(1);
h1=h1+c1;
h1=h1-a2;
}
n++;
}
System.out.println(n);
for(int i=0;i<al.size();i++)
{if(al.get(i)==0)
System.out.println("STRIKE");
else
System.out.println("HEAL");
}
}
}
