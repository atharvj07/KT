import sys
r = sys.stdin.readline

def revAndnot(bit, idx):
	tmp = ''
	for i in range(idx):
		tmp += '1' if bit[i] == '0' else '0'
	return tmp[::-1] + bit[idx:]
for _ in range(int(r())):
	n = int(r())
	a = r().strip()
	b = r().strip()
	ans = []
	flip = False
	idx = 0
	for i in range(n - 1, -1, -1):
		if flip ^ (a[idx] == b[i]):
			ans.append(str(1))
		ans.append(str(i + 1))
		if flip:
			idx -= i
		else:
			idx += i
		flip ^= True
	print(len(ans), end=' ')
	print(' '.join(ans))
