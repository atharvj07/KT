#include<bits/stdc++.h>

using namespace std;

#define ZFor(i,n) for(int i=0;i<n;i++)
#define WFor(i,n) for(int i=1;i<=n;i++)
#define ZWFor(i,n) for(int i=0;i<=n;i++)
#define WZFor(i,n) for(int i=1;i<n;i++)
#define Max -2100000000
#define Min 2100000000
#define F first
#define S second
#define F1 first.first
#define F2 first.second
#define F3 second

typedef pair<int,int> P;
typedef pair<pair<int,int>,int> PP;


int main(void){
	
	int n,m,w[50],r[50],b[50],W=0,B=0,R=0,ans=Min;
	char hata[51][51];
	
	scanf("%d %d",&n,&m);
	ZFor(i,n) scanf("%s",hata[i]);
	
	ZFor(i,n){
		w[i]=0;
		r[i]=0;
		b[i]=0;
	}
	
	ZFor(i,n)ZFor(j,strlen(hata[i])){
		//printf("%c\n",hata[i][j]);
		if(hata[i][j]=='W') w[i]++;
		if(hata[i][j]=='R') r[i]++;
		if(hata[i][j]=='B') b[i]++;
	}
	//ZFor(i,n) printf("%d W=%d R=%d B=%d\n",i,w[i],r[i],b[i]);
	
	for(int i=0;i<n-2;i++){
		W+=r[i]+b[i];
		for(int j=i+1;j<n-1;j++){
			B+=w[j]+r[j];
			for(int k=j+1;k<n;k++) R+=w[k]+b[k];
			ans=min(ans,W+B+R);
			R=0;
		}
		B=0;
	}
	
	printf("%d\n",ans);
	
	return 0;
}
