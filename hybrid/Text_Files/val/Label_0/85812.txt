#include <bits/stdc++.h>
using namespace std;
long long n, m;
const long long N = 2005;
const long long mod = 1e9 + 7;
long long dp[N][N];
signed main() {
  cin >> n >> m;
  for (long long i = 1; i <= m; i++) dp[1][i] = 1;
  for (long long i = 2; i <= n; i++) {
    dp[i][1] = 1;
    long long s = 0;
    for (long long j = 2; j <= m; j++) {
      (s += dp[i - 1][j]) %= mod;
      (dp[i][j] += dp[i][j - 1] + s) %= mod;
    }
  }
  long long ans = 0;
  for (long long i = 1; i <= n; i++)
    for (long long j = 2; j <= m; j++) {
      (ans += (dp[i][j] - dp[i - 1][j] + mod) * dp[n + 1 - i][j] % mod *
              (m - j + 1)) %= mod;
    }
  cout << ans << endl;
  return 0;
}
