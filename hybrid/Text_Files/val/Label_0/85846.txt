#include <bits/stdc++.h>
using namespace std;
const int MAX_N = 10000;
const int MAX_M = 1000000;
char s[MAX_N + 4], t[MAX_M + 4];
int cs[26][MAX_N], ls[26];
int main() {
  scanf("%s%s", s, t);
  for (int i = 0; s[i]; i++) {
    int a = s[i] - 'a';
    cs[a][ls[a]++] = i;
  }
  int y = 1, x = 0;
  for (char *cpt = t; *cpt; cpt++) {
    int a = *cpt - 'a';
    if (ls[a] == 0) {
      puts("-1");
      return 0;
    }
    int i = lower_bound(cs[a], cs[a] + ls[a], x) - cs[a];
    if (i >= ls[a]) {
      y++;
      x = cs[a][0] + 1;
    } else
      x = cs[a][i] + 1;
  }
  printf("%d\n", y);
  return 0;
}
