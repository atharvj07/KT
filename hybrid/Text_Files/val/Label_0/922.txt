T = int(input(''))
for _ in range(T):
	n = int(input(''))
	p = list(map(int, input('').split(' ')))
	d = {}
	k = []
	t = 0
	for i in range(n):
		if p[i] not in d:
			k.append(p[i])
			d[p[i]] = 0
		d[p[i]] = d[p[i]] + 1
	l = len(k)
	s = 0
	b = 0
	if l < 3:
		print('0 0 0')
		continue
	g = d[k[0]]
	ind = -1
	for i in range(1, l, 1):
		if s <= g:
			s = s + d[k[i]]
		else:
			ind = i
			break
	if ind == -1:
		print('0 0 0')
		continue
	for i in range(ind, l, 1):
		if s + b + g + d[k[i]] > n // 2:
			break
		else:
			b = b + d[k[i]]
	if b <= g:
		print('0 0 0')
	else:
		print(str(g) + ' ' + str(s) + ' ' + str(b))
