#include <bits/stdc++.h>
using namespace std;
int gcd(int a, int b) { return b == 0 ? a : gcd(b, a % b); }
int main() {
  int n;
  cin >> n;
  int a[n], b[n];
  long long pr = 1;
  set<int> s;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    pr = pr * a[i] / gcd(pr, a[i]);
  }
  for (int i = 0; i < n; i++) {
    cin >> b[i];
  }
  int cnt1 = 0;
  for (int i = 0; i < min(pr, 100000000LL); i++) {
    for (int j = 0; j < n; j++) {
      if (i % a[j] == b[j]) {
        cnt1++;
        break;
      }
    }
  }
  cout << fixed << setprecision(6) << (cnt1 + .0) / pr << "\n";
}
