	static long addMod(long a,long b,long mod) {
		long c=a+b;
		if (c>=mod)
			c-=mod;
		return c;
	}

	static void process() throws Exception {


		out = new BufferedWriter(new OutputStreamWriter(System.out));
		reader = new InputReader(System.in);

		int n=reader.readInt();
		int m=reader.readInt();
		String[] map=new String[n];
		for (int i=0;i<n;i++)
			map[i]=reader.readString();

		int nmx=Math.max(n, m);

		int[][] delta= {{1,0},{0,1}};
		// d the previous direction
		// e the direction we went
		int[][] right=new int[n][m];
		int[][] down=new int[n][m];
		for (int u=n-1;u>=0;u--)
			for (int v=m-1;v>=0;v--) {
				int ad=0;
				if (map[u].charAt(v)=='R')
					ad=1;
				right[u][v]=ad;
				down[u][v]=ad;
				if (v+1<m)
					right[u][v]+=right[u][v+1];
				if (u+1<n)
					down[u][v]+=down[u+1][v];
			}

		long[][][] dp=new long[n][m][2];



		// let's look only at the moment where we change direction

		long[][][] diff=new long[n][m][2];
		long[][][] sum=new long[n][m][2];
		diff[0][0][0]=1;
		diff[0][0][1]=1;

		if (1<n)
			diff[1][0][0]=mod-1;
		if (1<m)
			diff[0][1][1]=mod-1;


		for (int i=0;i<n;i++) {
			for (int j=0;j<m;j++) {

				if (i>0)
					sum[i][j][0]=sum[i-1][j][0];
				if (j>0)
					sum[i][j][1]=sum[i][j-1][1];

				for (int d=0;d<2;d++) {
					//sum[i][j][d]=addMod(sum[i][j][d],diff[i][j][d],mod);
					sum[i][j][d]+=diff[i][j][d];
					if (sum[i][j][d]>=mod)
						sum[i][j][d]-=mod;

					dp[i][j][d]=sum[i][j][d];
				}

				int d;
				// first from above
				d=0;
				// we have m-1-j elements at the right
				// of which right[j+1] are blocked
				// so m-1-j-right[j+1] are free

				if (dp[i][j][d]>0) {
					if (j+1<m) {
						int blck=right[i][j+1];
						int all=m-1-j;
						int rem=all-blck;


						//log("i:"+i+" j:"+j+" rem:"+rem);
						//diff[i][j+1][1-d]=addMod(diff[i][j+1][1-d],dp[i][j][d],mod);
						diff[i][j+1][1-d]+=dp[i][j][d];
						if (diff[i][j+1][1-d]>=mod)
							diff[i][j+1][1-d]-=mod;

						if (j+1+rem<m) {
							//diff[i][j+1+rem][1-d]=addMod(diff[i][j+1+rem][1-d],mod-dp[i][j][d],mod);
							diff[i][j+1+rem][1-d]+=mod-dp[i][j][d];
							if (diff[i][j+1+rem][1-d]>=mod)
								diff[i][j+1+rem][1-d]-=mod;
						}
					}
				}
				d=1;
				// we have n-1-i elements below
				// of which down[i+1] are blocked
				// so n-1-i-down[i+1] are free

				if (dp[i][j][d]>0) {
					if (i+1<n) {
						int blck=down[i+1][j];
						int all=n-1-i;
						int rem=all-blck;

						//diff[i+1][j][1-d]=addMod(diff[i+1][j][1-d],dp[i][j][d],mod);
						diff[i+1][j][1-d]+=dp[i][j][d];
						if (diff[i+1][j][1-d]>=mod)
							diff[i+1][j][1-d]-=mod;
						if (i+1+rem<n) {
							//diff[i+1+rem][j][1-d]=addMod(diff[i+1+rem][j][1-d],mod-dp[i][j][d],mod);
							diff[i+1+rem][j][1-d]+=mod-dp[i][j][d];
							if (diff[i+1+rem][j][1-d]>=mod)
								diff[i+1+rem][j][1-d]-=mod;
						}
					}
				}
			}
		}

		long ans=addMod(dp[n-1][m-1][0],dp[n-1][m-1][1],mod);
		if (n==1 && m==1)
			ans=1;
		output(ans);


		try {
			out.close();
		} catch (Exception e) {
		}

	}

	static final class InputReader {
		private final InputStream stream;
		private final byte[] buf = new byte[1024];
		private int curChar;
		private int numChars;

		public InputReader(InputStream stream) {
			this.stream = stream;
		}

		private int read() throws IOException {
			if (curChar >= numChars) {
				curChar = 0;
				numChars = stream.read(buf);
				if (numChars <= 0) {
					return -1;
				}
			}
			return buf[curChar++];
		}

		public final String readString() throws IOException {
			int c = read();
			while (isSpaceChar(c)) {
				c = read();
			}
			StringBuilder res = new StringBuilder();
			do {
				res.append((char) c);
				c = read();
			} while (!isSpaceChar(c));
			return res.toString();
		}

		public final int readInt() throws IOException {
			int c = read();
			boolean neg = false;
			while (isSpaceChar(c)) {
				c = read();
			}
			char d = (char) c;
			// log("d:"+d);
			if (d == '-') {
				neg = true;
				c = read();
			}
			int res = 0;
			do {
				res *= 10;
				res += c - '0';
				c = read();
			} while (!isSpaceChar(c));
			// log("res:"+res);
			if (neg)
				return -res;
			return res;

		}

		public final long readLong() throws IOException {
			int c = read();
			boolean neg = false;
			while (isSpaceChar(c)) {
				c = read();
			}
			char d = (char) c;
			// log("d:"+d);
			if (d == '-') {
				neg = true;
				c = read();
			}
			long res = 0;
			do {
				res *= 10;
				res += c - '0';
				c = read();
			} while (!isSpaceChar(c));
			// log("res:"+res);
			if (neg)
				return -res;
			return res;

		}

		private boolean isSpaceChar(int c) {
			return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
		}
	}
