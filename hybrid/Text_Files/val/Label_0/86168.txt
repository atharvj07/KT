template <class T>

  if (b == 0) return a;
  return gcd(b, a % b);
}

  return a / gcd(a, b) * b;
}

vector<string> split(string str, char del) {
  vector<string> res;
  for (int i = 0, s = 0; i < SIZE(str); i++) {
    if (str[i] == del) {
      if (i - s != 0) res.push_back(str.substr(s, i - s));
      s = i + 1;
    } else if (i == SIZE(str) - 1) {
      res.push_back(str.substr(s));
    }
  }
  return res;
}
int H, W;
char field[101][10010];
int main() {
  cin >> H >> W;
  for (int i = 0; i < H; i++)
    for (int j = 0; j < W; j++) cin >> field[i][j];
  long long cost = 0;
  int ang = 0;
  int now = 0;
  for (int i = 0; i < H - 1; i++) {
    if (field[i + 1][now] == '.') {
      cost++;
      continue;
    }
    int leftPos = now;
    int rightPos = now;
    for (;; --leftPos)
      if (leftPos < 0 || field[i][leftPos] != '.') break;
    for (;; ++rightPos)
      if (rightPos >= W || field[i][rightPos] != '.') break;
    int leftHole = now;
    int rightHole = now;
    for (;; --leftHole)
      if (leftHole < 0 || field[i + 1][leftHole] == '.') break;
    for (;; ++rightHole)
      if (rightHole >= W || field[i + 1][rightHole] == '.') break;
    int cnt = 0;
    while (1) {
      if (ang == 0) {
        if (rightHole < W && rightHole < rightPos) {
          cost += rightHole - now + 1;
          now = rightHole;
          cnt = 0;
          break;
        } else if (rightPos < W && field[i][rightPos] == '+') {
          cnt = 0;
          cost += rightPos - now;
          field[i][rightPos] = '.';
          now = rightPos - 1;
          for (;; ++rightPos)
            if (rightPos >= W || field[i][rightPos] != '.') break;
        } else {
          cost += rightPos - now;
          now = rightPos - 1;
          cnt++;
        }
      } else {
        if (leftHole >= 0 && leftHole > leftPos) {
          cnt = 0;
          cost += -leftHole + now + 1;
          now = leftHole;
          break;
        } else if (leftPos >= 0 && field[i][leftPos] == '+') {
          cnt = 0;
          cost += -leftPos + now;
          field[i][leftPos] = '.';
          now = leftPos + 1;
          for (;; --leftPos)
            if (leftPos < 0 || field[i][leftPos] != '.') break;
        } else {
          cnt++;
          cost += now - leftPos;
          now = leftPos + 1;
        }
      }
      ang ^= 1;
      if (cnt == 2) {
        cout << "Never" << endl;
        return 0;
      }
    }
  }
  cout << cost << endl;
  return 0;
}
