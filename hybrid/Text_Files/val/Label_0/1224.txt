n = int(input())
arr = list(map(int, input().split()))
flag = 1
if n <= 2:
	print(0)
	print(*arr)
	flag = 0
arr.sort(reverse=True)
final = ['0']
pair = 0
if flag == 1:
	k = 0
	j = 1
	if n % 2 != 0:
		temp1 = arr[:n // 2 + 1]
		temp2 = arr[n // 2 + 1:]
	else:
		temp1 = arr[:n // 2]
		temp2 = arr[n // 2:]
		rem_val = temp2.pop(0)
	for i in range(0, len(temp2)):
		if temp1[k] == temp2[i] or temp1[j] == temp2[i]:
			pass
		else:
			pair += 1
		final.pop(-1)
		final.append(temp1[k])
		final.append(temp2[i])
		final.append(temp1[j])
		k += 1
		j += 1
	if n % 2 == 0:
		final.append(rem_val)
if flag == 1:
	print(pair)
	print(*final)
