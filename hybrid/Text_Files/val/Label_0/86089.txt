#include <bits/stdc++.h>
using namespace std;
int a[200005];
struct val {
  int l;
  int r;
  int i;
} q[200005];
int sq;
int mp[200005];
deque<int> s[200005];
deque<int> w[200005];
int ok[200005];
int ans[200005];
bool comp(val a, val b) {
  if (a.l / sq != b.l / sq)
    return a.l / sq < b.l / sq;
  else
    return a.r < b.r;
}
int main() {
  int n;
  scanf("%d", &n);
  sq = sqrt(n);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
  }
  int qu;
  scanf("%d", &qu);
  int l, r;
  for (int i = 1; i <= qu; i++) {
    scanf("%d %d", &l, &r);
    q[i].l = l;
    q[i].r = r;
    q[i].i = i;
  }
  sort(q + 1, q + qu + 1, comp);
  l = q[1].l;
  r = l - 1;
  int cnt = 0;
  int all = 0;
  for (int i = 1; i <= qu; i++) {
    while (r < q[i].r) {
      r++, mp[a[r]]++, cnt += mp[a[r]] == 1;
      int p = a[r];
      all -= ok[p] == 0 && !s[p].empty();
      if (s[p].size() > 0) {
        if (!w[p].empty()) ok[p] += (r - s[p].back()) != w[p].back();
        w[p].push_back(r - s[p].back());
      }
      s[p].push_back(r);
      all += ok[p] == 0 && !s[p].empty();
    }
    while (q[i].l < l) {
      l--, mp[a[l]]++, cnt += mp[a[l]] == 1;
      int p = a[l];
      all -= ok[p] == 0 && !s[p].empty();
      if (s[p].size() > 0) {
        if (!w[p].empty()) ok[p] += (s[p].front() - l) != w[p].front();
        w[p].push_front(s[p].front() - l);
      }
      s[p].push_front(l);
      all += ok[p] == 0 && !s[p].empty();
    }
    while (l < q[i].l) {
      mp[a[l]]--, cnt -= mp[a[l]] == 0;
      int p = a[l];
      all -= ok[p] == 0 && !s[p].empty();
      s[p].pop_front();
      if (s[p].size() > 0) {
        int tmp = s[p].front() - l;
        w[p].pop_front();
        if (!w[p].empty()) ok[p] -= tmp != w[p].front();
      }
      all += ok[p] == 0 && !s[p].empty();
      l++;
    }
    while (q[i].r < r) {
      mp[a[r]]--, cnt -= mp[a[r]] == 0;
      int p = a[r];
      all -= ok[p] == 0 && !s[p].empty();
      s[p].pop_back();
      if (s[p].size() > 0) {
        int tmp = r - s[p].back();
        w[p].pop_back();
        if (!w[p].empty()) ok[p] -= tmp != w[p].back();
      }
      all += ok[p] == 0 && !s[p].empty();
      r--;
    }
    ans[q[i].i] = cnt + (all == 0);
  }
  for (int i = 1; i <= qu; i++) cout << ans[i] << "\n";
  return 0;
}
