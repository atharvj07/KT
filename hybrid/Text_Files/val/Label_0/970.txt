import sys
from os import path

def open_inputs():
	if path.exists('input.txt'):
		sys.stdin = open('input.txt', 'r')
		sys.stdout = open('output.txt', 'w')

def get_int():
	return int(sys.stdin.readline().strip())

def get_ints():
	return list(map(int, sys.stdin.readline().strip().split()))

def get_string():
	return sys.stdin.readline().strip()

def printf(_s):
	sys.stdout.write(str(_s))

def main():
	open_inputs()
	cases = get_int()
	for case in range(cases):
		get_int()
		numbers = get_ints()
		numbers.sort()
		n = int(len(numbers) / 2)
		numbers_one = numbers[n:]
		numbers_two = numbers[:n]
		matches = 0
		while len(numbers_one) and len(numbers_two):
			one = numbers_one.pop()
			two = numbers_two.pop()
			if one == two:
				matches += 2
		matches += len(numbers_one) + len(numbers_two)
		printf(str(matches) + '\n')
main()
