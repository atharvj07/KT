#include<bits/stdc++.h>
using namespace std;
int n;
int a[205000],b[205000];
int ans[205000],len;
int main(){
  scanf("%d",&n);
  for(int i=1;i<=n;i++)scanf("%d",&a[i]);
  for(int i=n;i>=1;i--){
    int sum=0;
    for(int j=i*2;j<=n;j+=i)sum+=b[j];
    if(sum%2!=a[i]){
    	b[i]=1;
			ans[++len]=i;
    }
  }
  cout<<len<<endl;
  for(int i=1;i<=len;++i)cout<<ans[i]<<" ";
  return 0;
}
