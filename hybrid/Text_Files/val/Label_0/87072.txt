public class Main
{
	static class dian implements Comparable<dian>
	{
		int l;
		int r;
		int t;
		int id;//0是车,1是人
		public dian(int l,int r,int t,int id)
		{
			this.l=l;
			this.r=r;
			this.t=t;
			this.id=id;
		}
		@Override
		public int compareTo(dian o)
		{
			if (this.l!=o.l)return this.l-o.l;
			else 
			{
				return this.id-o.id;
			}
		}
	}
	static class node
	{
		int h;
		int id;
		public node()
		{
			h=-1;
		}
		public node(int h,int id)
		{
			this.h=h;
			this.id=id;
		}
	}
	static int time[];
	static node tree[];
	static void add(int l,int r,int k,int x,int value,int id)
	{
		if (l==r)
		{
			tree[k].h=value;
			tree[k].id=id;
			return;
		}
		int m=(l+r)>>1;
		if (x<=time[m])add(l,m,k<<1,x,value,id);
		else add(m+1,r,k<<1|1,x,value,id);
		
		if (tree[k<<1].h>tree[k<<1|1].h)
		{
			tree[k].h=tree[k<<1].h;
			tree[k].id=tree[k<<1].id;
		}
		else
		{
			tree[k].h=tree[k<<1|1].h;
			tree[k].id=tree[k<<1|1].id;
		}
	}
	static int find(int l,int r,int k,int a,int b,int value)
	{
		if (tree[k].h<value)return -1;
		if (l==r)return tree[k].id;
		int m=(l+r)>>1;
		
		int t=-1;
		if (a<=time[m])t=find(l,m,k<<1,a,b,value);
		if (t!=-1)return t;
		return find(m+1,r,k<<1|1,a,b,value);
	}
	
	public static void main(String[] args) throws IOException
	{
		Scanner sc=new Scanner(System.in);
		int n=ini();
		int m=ini();
		dian q[]=new dian [n+m+1];
		TreeSet<Integer> set=new TreeSet<>();
		for (int i=1;i<=n+m;i++)
		{
			int l=ini();
			int r=ini();
			int t=ini();
			q[i]=new dian(l,r,t,i);
			set.add(t);
		}
		Arrays.sort(q,1,n+m+1);
		time=new int [n+m+1];
		int o=0;
		for (Integer s:set)
		{
			time[++o]=s;
		}
		tree=new node [o<<2];
		int ans[]=new int [m];
		for (int i=1;i<o<<2;i++)
		{
			tree[i]=new node();
		}
		for (int i=1;i<=n+m;i++)
		{
			if (q[i].id<=n)//如果是车
			{
				add(1,o,1,q[i].t,q[i].r,q[i].id);
			}
			else//如果是人
			{
				ans[q[i].id-n-1]=find(1,o,1,q[i].t,o,q[i].r);
			}
		}
		for (int i=0;i<m;i++)
		{
			if (i==0)out.print(ans[i]);
			else out.print(" "+ans[i]);
		}
		
		out.flush();
	}

	static StreamTokenizer in=new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));
	static PrintWriter out=new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
	static int ini() throws IOException
	{
		in.nextToken();
		return (int)in.nval;
	}
	static String ins() throws IOException
	{
		in.nextToken();
		return in.sval;
	}
}
