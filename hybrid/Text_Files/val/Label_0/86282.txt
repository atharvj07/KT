#include <bits/stdc++.h>
const int N = 200005;
std::pair<int, int> operator-(std::pair<int, int> a, std::pair<int, int> b) {
  return std::pair<int, int>(a.first - b.first, a.second - b.second);
}
long long operator*(std::pair<int, int> a, std::pair<int, int> b) {
  return (long long)a.first * b.second - (long long)a.second * b.first;
}
bool is_odd(std::pair<int, int> a) { return (a.first | a.second) & 1; }
int n;
std::pair<int, int> p[N];
int stack[N], top;
bool banned[N];
bool cmp(int a, int b, int c) { return (p[c] - p[b]) * (p[b] - p[a]) >= 0; }
std::vector<int> convex_hull() {
  top = 1, stack[0] = 0;
  for (int i = 1; i <= n; ++i)
    if (!banned[i]) {
      while (top > 1 && cmp(stack[top - 2], stack[top - 1], i)) --top;
      stack[top++] = i;
    }
  return std::vector<int>(stack + 1, stack + top);
}
int main() {
  std::ios::sync_with_stdio(0), std::cin.tie(0);
  std::cin >> n;
  for (int i = 1; i <= n; ++i) std::cin >> p[i].first >> p[i].second;
  std::sort(p + 1, p + n + 1, [](std::pair<int, int> a, std::pair<int, int> b) {
    long long c = a * b;
    return c != 0 ? c > 0 : a < b;
  });
  std::vector<int> all = convex_hull();
  for (int i = 0; i < all.size(); i += 2) banned[all[i]] = 1;
  std::vector<int> a = convex_hull();
  for (int i = 0; i < all.size(); i += 2) banned[all[i]] = 0;
  if (n > 1) {
    for (int i = 1; i < all.size(); i += 2) banned[all[i]] = 1;
    std::vector<int> b = convex_hull();
    a.insert(a.end(), b.begin(), b.end());
  }
  for (int e : a)
    if (is_odd(p[e])) return std::cout << "Ani\n", 0;
  std::cout << "Borna\n";
  return 0;
}
