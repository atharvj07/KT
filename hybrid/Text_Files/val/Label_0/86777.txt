    static class TaskB {
        int n;
        String[] words;
        long[][] hashes;
        long base = 37;
        long mod = (int) 1e9 + 7;

        public void solve(int testNumber, InputReader in, OutputWriter out) {
            if (false) {
                test();
                return;
            }
            int t = in.readInt();
            while (t-- > 0) {
                n = in.readInt();
                words = IOUtils.readStringArray(in, n);
                long answer = solve();
                out.printLine(answer);
            }
        }

        void test() {
            int maxn = (int) 1e6;
            int wordCount = 100;
            n = wordCount;

            String s = "";
            int wordLen = maxn / wordCount;
            for (int i = 0; i < wordLen; i++) {
                s += 'a';
            }

            words = new String[wordCount];
            for (int i = 0; i < wordCount; i++) {
                words[i] = s;
            }
            System.out.println(solve());
        }

        long solve() {
            Arrays.sort(words, Comparator.comparingInt(String::length));
            hashes = new long[n][];
            for (int i = 0; i < n; i++) {
                hashes[i] = hash(i);
            }

            int maxLen = words[n - 1].length();
            int sumLen = Arrays.stream(words).mapToInt(String::length).sum();

            long[][] hashes = new long[maxLen + 1][];
            int[][] indexes = new int[sumLen + 1][];
            int hashCount = 0;
            for (int len = maxLen, first = n; len >= 1; len--) {
                while (first - 1 >= 0 && words[first - 1].length() >= len) {
                    first--;
                }
                long[] curHashes = new long[n - first];
                for (int i = first; i < n; i++) {
                    curHashes[i - first] = trueMod(getHash(i, 0, len - 1) * base);
                }
                Arrays.sort(curHashes);
                int size = 0;
                for (int i = 0; i < curHashes.length; i++) {
                    if (i == 0 || curHashes[i] != curHashes[i - 1]) {
                        curHashes[size++] = curHashes[i];
                    }
                }
                hashes[len] = Arrays.copyOfRange(curHashes, 0, size);
                indexes[len] = new int[size];
                for (int i = 0; i < size; i++) {
                    indexes[len][i] = hashCount++;
                }
            }

            boolean[] hasParent = new boolean[sumLen];
            int[] dp0 = new int[hashCount];
            int[] dp1 = new int[hashCount];
            for (int len = maxLen, first = n; len >= 1; len--) {
                while (first - 1 >= 0 && words[first - 1].length() >= len) {
                    first--;
                }
                long[] curHashes = hashes[len];
                int[] curIndexes = indexes[len];
                boolean[] used = new boolean[curIndexes.length];
                for (int i = first; i < n; i++) {
                    int at = binarySearch(curHashes, trueMod(getHash(i, 0, len - 1) * base));
                    if (!used[at]) {
                        used[at] = true;
                        int curIndex = curIndexes[at];
                        dp1[curIndex]++;
                        if (len > 1) {
                            long[] parHashes = hashes[len - 1];
                            long parHash = getHash(i, 1, len - 1);
                            int parAt = binarySearch(parHashes, parHash);
                            if (parAt >= 0) {
                                int[] parIndexes = indexes[len - 1];
                                int parIndex = parIndexes[parAt];
                                dp0[parIndex] += Math.max(dp0[curIndex], dp1[curIndex]);
                                dp1[parIndex] += dp0[curIndex];
                                hasParent[curIndex] = true;
                            }
                        }
                    }
                }
            }

            long answer = 0;
            for (int i = 0; i < hashCount; i++) {
                if (!hasParent[i]) {
                    answer += Math.max(dp0[i], dp1[i]);
                }
            }
            return answer;
        }

        int binarySearch(long[] a, long x) {
            int l = 0;
            int r = a.length;
            while (r - l > 1) {
                int m = l + r >> 1;
                if (a[m] <= x) {
                    l = m;
                } else {
                    r = m;
                }
            }
            return a[l] == x ? l : -1;
        }

        long[] hash(int wordIndex) {
            String s = words[wordIndex];
            long[] h = hashes[wordIndex] = new long[s.length()];
            h[0] = s.charAt(0) - 'a' + 1;
            long p = base;
            for (int i = 1; i < s.length(); i++, p = (p * base) % mod) {
                h[i] = trueMod(h[i - 1] + p * (s.charAt(i) - 'a' + 1));
            }
            return h;
        }

        long getHash(int wordIndex, int from, int to) {
            long h = hashes[wordIndex][to];
            if (from > 0) {
                h -= hashes[wordIndex][from - 1];
            }
            return trueMod(h);
        }

        long trueMod(long x) {
            return ((x % mod) + mod) % mod;
        }

    }

    static class OutputWriter {
        private final PrintWriter writer;

        public OutputWriter(OutputStream outputStream) {
            writer = new PrintWriter(new BufferedWriter(new OutputStreamWriter(outputStream)));
        }

        public OutputWriter(Writer writer) {
            this.writer = new PrintWriter(writer);
        }

        public void close() {
            writer.close();
        }

        public void printLine(long i) {
            writer.println(i);
        }

    }

    static class InputReader {
        private InputStream stream;
        private byte[] buf = new byte[1024];
        private int curChar;
        private int numChars;
        private InputReader.SpaceCharFilter filter;

        public InputReader(InputStream stream) {
            this.stream = stream;
        }

        public int read() {
            if (numChars == -1) {
                throw new InputMismatchException();
            }
            if (curChar >= numChars) {
                curChar = 0;
                try {
                    numChars = stream.read(buf);
                } catch (IOException e) {
                    throw new InputMismatchException();
                }
                if (numChars <= 0) {
                    return -1;
                }
            }
            return buf[curChar++];
        }

        public int readInt() {
            int c = read();
            while (isSpaceChar(c)) {
                c = read();
            }
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = read();
            }
            int res = 0;
            do {
                if (c < '0' || c > '9') {
                    throw new InputMismatchException();
                }
                res *= 10;
                res += c - '0';
                c = read();
            } while (!isSpaceChar(c));
            return res * sgn;
        }

        public String readString() {
            int c = read();
            while (isSpaceChar(c)) {
                c = read();
            }
            StringBuilder res = new StringBuilder();
            do {
                if (Character.isValidCodePoint(c)) {
                    res.appendCodePoint(c);
                }
                c = read();
            } while (!isSpaceChar(c));
            return res.toString();
        }

        public boolean isSpaceChar(int c) {
            if (filter != null) {
                return filter.isSpaceChar(c);
            }
            return isWhitespace(c);
        }

        public static boolean isWhitespace(int c) {
            return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
        }

        public interface SpaceCharFilter {
            public boolean isSpaceChar(int ch);

        }

    }

    static class IOUtils {
        public static String[] readStringArray(InputReader in, int size) {
            String[] array = new String[size];
            for (int i = 0; i < size; i++) {
                array[i] = in.readString();
            }
            return array;
        }

    }
