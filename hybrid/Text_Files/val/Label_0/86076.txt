#include <bits/stdc++.h>
using namespace std;
map<string, int> MP;
int go(int t, string s) {
  if (MP.find(s) != MP.end()) {
    return MP[s];
  }
  if (s.size() == 2) {
    return MP[s] = (s[0] - '0') * 2 + (s[1] - '0');
  }
  bool lose = true;
  bool vis[4] = {0};
  for (int i = (0); i < (int)(s.size()); i++) {
    string s2 = s;
    s2.erase(s2.begin() + i);
    int ret = go(1 - t, s2);
    vis[ret] = 1;
  }
  if (t == 0) {
    int i = 0;
    while (!vis[i]) ++i;
    return MP[s] = i;
  }
  if (t == 1) {
    int i = 3;
    while (!vis[i]) --i;
    return MP[s] = i;
  }
}
int stupid(string s) {
  MP.clear();
  int res = go(0, s);
  return res;
}
set<string> glo;
void exhaust(string &s, int idx) {
  if (idx == s.size()) {
    int ans = stupid(s);
    string bin = "";
    bin += '0' + (ans / 2);
    bin += '0' + (ans % 2);
    glo.insert(bin);
    return;
  }
  if (s[idx] != '?') return exhaust(s, idx + 1);
  s[idx] = '0';
  exhaust(s, idx + 1);
  s[idx] = '1';
  exhaust(s, idx + 1);
  s[idx] = '?';
}
set<string> doit(string s) {
  glo.clear();
  exhaust(s, 0);
  return glo;
}
