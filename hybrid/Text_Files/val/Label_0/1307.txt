def solve(n, d, h):
	if n < d + 1:
		return False
	if d > 2 * h:
		return False
	if d - h < 0:
		return False
	if d == 1 and n >= 3:
		return False
	cnt = 1
	h0 = d - h
	for i in range(h):
		print(cnt, cnt + 1)
		cnt += 1
	if h0:
		print(1, cnt + 1)
		cnt += 1
	for i in range(h0 - 1):
		print(cnt, cnt + 1)
		cnt += 1
	if h0:
		for i in range(n - d - 1):
			print(1, cnt + 1)
			cnt += 1
	else:
		for i in range(n - d - 1):
			print(2, cnt + 1)
			cnt += 1
	return True
(n, d, h) = [int(i) for i in input().split()]
if not solve(n, d, h):
	print(-1)
