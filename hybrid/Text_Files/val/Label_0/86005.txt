#include <bits/stdc++.h>
using namespace std;
const int N = 200002;
const int sizeTree = (1 << 20);
int tAdd[sizeTree], mn[sizeTree], mx[sizeTree];
long long tSum[sizeTree];
int n;
int a[N];
pair<int, int> b[N], c[N];
long long ans[N];
void push(int v, int tl, int tr) {
  if (tAdd[v] != 0) {
    int tm = (tl + tr) / 2;
    tAdd[2 * v] = mn[2 * v] = mx[2 * v] = tAdd[v];
    tAdd[2 * v + 1] = mn[2 * v + 1] = mx[2 * v + 1] = tAdd[v];
    tSum[2 * v] = 1ll * (tm - tl + 1) * tAdd[v];
    tSum[2 * v + 1] = 1ll * (tr - tm) * tAdd[v];
    tAdd[v] = 0;
  }
}
void build(int v, int tl, int tr) {
  if (tl == tr) {
    tAdd[v] = 0;
    tSum[v] = tl;
    mn[v] = mx[v] = tl;
    return;
  }
  int tm = (tl + tr) / 2;
  build(2 * v, tl, tm);
  build(2 * v + 1, tm + 1, tr);
  tSum[v] = tSum[2 * v] + tSum[2 * v + 1];
  mn[v] = min(mn[2 * v], mn[2 * v + 1]);
  mx[v] = max(mx[2 * v], mx[2 * v + 1]);
}
void assign(int v, int tl, int tr, int l, int r, int value) {
  if (l > r || mn[v] >= value) return;
  if (tl == l && tr == r && mx[v] <= value) {
    mx[v] = mn[v] = tAdd[v] = value;
    tSum[v] = 1ll * (tr - tl + 1) * value;
    return;
  }
  push(v, tl, tr);
  int tm = (tl + tr) / 2;
  assign(2 * v, tl, tm, l, min(r, tm), value);
  assign(2 * v + 1, tm + 1, tr, max(l, tm + 1), r, value);
  tSum[v] = tSum[2 * v] + tSum[2 * v + 1];
  mn[v] = min(mn[2 * v], mn[2 * v + 1]);
  mx[v] = max(mx[2 * v], mx[2 * v + 1]);
}
void update(int j, int pos) {
  if (b[j].first == -1)
    b[j].first = pos;
  else if (b[j].second == -1)
    b[j].second = pos;
  c[j].first = c[j].second;
  c[j].second = pos;
}
int main() {
  memset(b, -1, sizeof(b));
  memset(c, -1, sizeof(c));
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) scanf("%d", &a[i]);
  build(1, 0, n - 1);
  for (int i = 0; i < n; ++i) {
    for (int j = 1; j * j <= a[i]; ++j) {
      if (a[i] % j == 0) {
        update(j, i);
        if (a[i] / j != j) update(a[i] / j, i);
      }
    }
  }
  ans[0] = (n == 1 ? 1 : 3);
  for (int i = N - 1; i >= 1; --i) {
    ans[i] = 1ll * n * n - tSum[1];
    if (b[i].first != c[i].second) {
      assign(1, 0, n - 1, b[i].second + 1, n - 1, n);
      assign(1, 0, n - 1, b[i].first + 1, b[i].second, c[i].second);
      assign(1, 0, n - 1, 0, b[i].first, c[i].first);
    }
  }
  long long res = 0;
  for (int i = 1; i < N; ++i) {
    res += 1ll * (ans[i] - ans[i - 1]) * i;
  }
  cout << res << endl;
  return 0;
}
