#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
int main() {
  int t = 1;
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> t;
  int arr[t];
  int minn = INT_MAX;
  for (int i = 0; i < t; i++) {
    cin >> arr[i];
    if (arr[i] < minn) {
      minn = arr[i];
    }
  }
  int flag = 0;
  int ans = INT_MAX;
  int last;
  for (int i = 0; i < t; i++) {
    if (arr[i] == minn) {
      if (flag == 0) {
        flag = 1;
        last = i;
      } else {
        ans = min(i - last, ans);
        last = i;
      }
    }
  }
  cout << ans << endl;
}
