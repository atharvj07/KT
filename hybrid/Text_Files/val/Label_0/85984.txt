#include <cstdio>
#include <iostream>
#include <string> 
#include <cstring> 
using namespace std;
int main()
{
	int n;
	int head,dis,sum=0;
	char str[110],board[100];
	cin>>n;
	cin>>board;
	for(int i=0;i<=n-1;i++){
		cin>>str;
	  for(head=0;head<=strlen(str)-1;head++){
		 	for(dis=0;head+(strlen(board)-1)*dis<strlen(str);dis++){
		 	  	    for(int k=0;k<=strlen(board)-1;k++){
		 	  	    if(str[head+dis*k]!=board[k]) {goto p1;}
		 	  	 }
		 	  	 sum++;
		 	  	 goto p2;
		 	  	 p1:;
		 	  }	 	 
	  }
	  p2:;	
	}
	cout<<sum<<endl;
	return 0;
}
