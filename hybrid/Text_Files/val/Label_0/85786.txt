#include <bits/stdc++.h>
using namespace std;
const int MAX = 2e5;
int a[MAX];
int n, k;
int cnt = 1;
bool dfs(int l, int r) {
  if (cnt == k) {
    return true;
  }
  int mid = (l + r) / 2;
  if (cnt + 2 < k && (r - l) > 1) {
    cnt += 2;
    swap(a[mid - 1], a[mid]);
    return dfs(l, mid) || dfs(mid, r);
  } else if (cnt + 2 == k && (r - l) > 1) {
    cnt += 2;
    swap(a[mid - 1], a[mid]);
    return true;
  }
  return false;
}
int main(int argc, char const *argv[]) {
  while (scanf("%d%d", &n, &k) != EOF) {
    for (int i = 0; i < n; i++) {
      a[i] = i + 1;
    }
    if (!dfs(0, n)) {
      printf("-1\n");
      continue;
    }
    for (int i = 0; i < n - 1; i++) {
      printf("%d ", a[i]);
    }
    printf("%d\n", a[n - 1]);
  }
  return 0;
}
