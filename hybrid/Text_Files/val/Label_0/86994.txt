import java.util.*;

public class PermutationForgery {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int t = scan.nextInt();
        Stack<Integer> stack = new Stack<Integer>();
        
        while(t!=0) {
            int n = scan.nextInt();
            for(int i=0; i<n; i++) {
                stack.push(scan.nextInt());
            }
            for(int i=0;i<n;i++) {
                System.out.print(stack.pop() + " ");
            }
            System.out.print("\n");
            t--;
        }
        scan.close();
    }
}
