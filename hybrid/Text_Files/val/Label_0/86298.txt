int sg[31] = {0, 1, 2, 1, 4, 3, 2, 1, 5,  6, 2, 1, 8,  7,  5, 9,
              8, 7, 3, 4, 7, 4, 2, 1, 10, 9, 3, 6, 11, 12, 14};
bool use[100000];
int main() {
  int n, tot, ans = 0;
  cin >> n;
  tot = n;
  for (int i = 2; i * i <= n; i++) {
    if (!use[i]) {
      long long tmp = 1, cnt = 0;
      while (tmp * i <= n) {
        tmp *= i;
        cnt++;
        if (tmp < 100000) use[tmp] = 1;
      }
      tot -= cnt;
      ans ^= sg[cnt];
    }
  }
  ans ^= tot % 2;
  puts(ans ? "Vasya" : "Petya");
  return 0;
}
