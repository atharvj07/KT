#include <bits/stdc++.h>
using namespace std;
const int MAXN = 5e5 + 5;
const int MAXM = 4e7 + 5;
const int MOD = 998244353;
map<int, int> mp;
int a[MAXN];
long long powmod(long long x, long long y) {
  long long rst = 1;
  for (; y; y >>= 1) {
    if (y & 1) rst = rst * x % MOD;
    x = x * x % MOD;
  }
  return rst;
}
int main() {
  int n;
  while (~scanf("%d", &n)) {
    mp.clear();
    for (int i = 1; i <= n; i++) {
      scanf("%d", &a[i]);
      mp[a[i]] = i;
    }
    int maxx = 0, flag = 0, num = 0;
    for (int i = 1; i <= n; i++) {
      maxx = max(maxx, mp[a[i]]);
      if (maxx == i) {
        if (!flag) {
          flag = 1;
          continue;
        }
        num++;
      }
    }
    long long ans = powmod(2, num);
    printf("%lld\n", ans);
  }
}
