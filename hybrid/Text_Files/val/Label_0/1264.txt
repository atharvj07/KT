n = int(input())
a = list(map(int, input().split()))
s = a[-1]
ans = ['+']
for v in reversed(a[:-1]):
	if s > 0:
		s -= v
		ans.append('-')
	else:
		s += v
		ans.append('+')
if 0 <= s <= a[-1]:
	print(''.join(reversed(ans)))
else:
	s = -a[-1]
	ans = ['-']
	for v in reversed(a[:-1]):
		if s > 0:
			s -= v
			ans.append('-')
		else:
			s += v
