k = eval(input())
a = input().split()
for i in range(k):
	a[i] = int(a[i])
tot = 0
for i in range(k):
	tot += a[i]
if k == 2:
	if a[0] > a[1]:
		print('1\n1')
	else:
		print(0)
else:
	can = a[0]
	party = []
	party.append(1)
	for i in range(k):
		if a[0] >= a[i] * 2 and len(party) < k - 1:
			can += a[i]
			party.append(i + 1)
	if can * 2 > tot:
		print(len(party))
		for i in party:
			print(i, end=' ')
	else:
		print(0)
