(n, h, l, r) = (int(i) for i in input().split(' '))
li = [int(i) for i in input().split(' ')]
d = {}
if l <= li[0] and li[0] <= r:
	d[li[0]] = 1
else:
	d[li[0]] = 0
if l <= li[0] - 1 and li[0] - 1 <= r:
	d[li[0] - 1] = 1
else:
	d[li[0] - 1] = 0
li.pop(0)
for i in li:
	td = {}
	for (j, jv) in d.items():
		tmp = (j + i) % h
		if l <= tmp and tmp <= r:
			td[tmp] = max(jv + 1, td.get(tmp, 1))
		else:
			td[tmp] = max(jv, td.get(tmp, 0))
		tmp -= 1
		tmp %= h
		if l <= tmp and tmp <= r:
			td[tmp] = max(jv + 1, td.get(tmp, 1))
		else:
			td[tmp] = max(jv, td.get(tmp, 0))
	d = td
m = 0
for (k, v) in d.items():
	m = max(m, v)
print(m)
