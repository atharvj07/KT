#include <bits/stdc++.h>
using namespace std;
int main() {
  vector<long long int> p, q;
  long long int n, k;
  scanf("%lld", &n);
  for (long long int i = 0; i < n; i++) {
    long long int v;
    scanf("%lld", &v);
    long long int m = 0, a = 0, b = 0;
    for (long long int j = 1; j <= v * 2; j++) {
      scanf("%lld", &k);
      if (k % 2 == 0) {
        p.push_back(j);
        a++;
      } else {
        q.push_back(j);
        b++;
      }
    }
    for (long long int j = 0; j < b - 1; j += 2) {
      if (m == v - 1) {
        break;
      } else {
        cout << q[j] << " " << q[j + 1] << "\n";
        m++;
      }
    }
    for (long long int j = 0; j < a - 1; j += 2) {
      if (m == v - 1) {
        break;
      } else {
        cout << p[j] << " " << p[j + 1] << "\n";
        m++;
      }
    }
    p.clear();
    q.clear();
  }
}
