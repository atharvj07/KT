def run():
	(n, k) = map(int, input().split())
	arr = list(map(int, input().split()))
	arr.sort()
	total = 0
	i = 0
	while k > 0:
		if i < n:
			if i == 0 and arr[i] != 0:
				print(arr[i])
				total += arr[i]
				k -= 1
			elif arr[i] - total != 0:
				arr[i] -= total
				print(arr[i])
				total += arr[i]
				k -= 1
			i += 1
		elif i >= n and k > 0:
			print(0)
			k -= 1
run()
