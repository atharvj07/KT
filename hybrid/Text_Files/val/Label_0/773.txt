import sys
n = int(input())
arr = list(map(int, input().split()))
counter = {}
for num in arr:
	counter[num] = 1
keys = sorted(counter.keys())
if len(keys) > 3:
	print(-1)
elif len(keys) == 3:
	if keys[1] - keys[0] == keys[2] - keys[1]:
		print(keys[1] - keys[0])
	else:
		print(-1)
elif len(keys) == 2:
	if (keys[1] - keys[0]) % 2 == 0:
		print(int((keys[1] - keys[0]) // 2))
	else:
		print(keys[1] - keys[0])
else:
	print(0)
