def fill_segment(start_i, start_j, seg_len, direction, possible_map):
	if direction == 'right':
		for j in range(seg_len):
			possible_map[start_i][start_j + j]['left'] = j
			possible_map[start_i][start_j + j]['right'] = seg_len - j - 1
	elif direction == 'down':
		for i in range(seg_len):
			possible_map[start_i + i][start_j]['up'] = i
			possible_map[start_i + i][start_j]['down'] = seg_len - i - 1
	return possible_map

def get_possible_count(field, n):
	possible_map = list()
	for row in field:
		possible_map.append(list())
		for cell in row:
			possible_map[-1].append({'up': 0, 'down': 0, 'left': 0, 'right': 0})
	for i in range(len(field)):
		row = field[i]
		start = [i, -1]
		for j in range(len(row)):
			if row[j] == '#' and start[1] != -1:
				seg_len = j - start[1]
				possible_map = fill_segment(start[0], start[1], seg_len, 'right', possible_map)
				start = [i, -1]
			if row[j] == '.' and start[1] == -1:
				start = [i, j]
		if start[1] != -1:
			seg_len = len(row) - start[1]
			possible_map = fill_segment(start[0], start[1], seg_len, 'right', possible_map)
	for j in range(len(field[0])):
		start = [-1, j]
		for i in range(len(field)):
			if field[i][j] == '#' and start[0] != -1:
				seg_len = i - start[0]
				possible_map = fill_segment(start[0], start[1], seg_len, 'down', possible_map)
				start = [-1, j]
			if field[i][j] == '.' and start[0] == -1:
				start = [i, j]
		if start[0] != -1:
			seg_len = len(field) - start[0]
			possible_map = fill_segment(start[0], start[1], seg_len, 'down', possible_map)
	return possible_map

def count_ans(possible_map, k):
	ans = [1, 1]
	Max = 0
	for i in range(len(possible_map)):
		for j in range(len(possible_map[i])):
			cnt = possible_map[i][j]
			left = cnt['left']
			right = cnt['right']
			hor_variants = 0
			if left + right >= k - 1 and field[i][j] != '#':
				leftpos = k - min(right, k - 1)
				rightpos = min(left, k - 1) + 1
				hor_variants = rightpos - leftpos + 1
			up = cnt['up']
			down = cnt['down']
			ver_variants = 0
			if up + down >= k - 1 and field[i][j] != '#':
				uppos = k - min(down, k - 1)
				downpos = min(up, k - 1) + 1
				ver_variants = downpos - uppos + 1
			if hor_variants + ver_variants > Max:
				Max = hor_variants + ver_variants
				ans = [i + 1, j + 1]
	return ans
(n, k) = map(int, input().split())
field = list()
for _ in range(n):
	row = input()
	field.append(row)
possible_map = get_possible_count(field, n)
print(' '.join(map(str, count_ans(possible_map, k))))
