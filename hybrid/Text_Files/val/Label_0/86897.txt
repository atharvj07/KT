import java.util.Scanner;

public class catherinecards {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		
		int cards = sc.nextInt();
		sc.nextLine();
		String colors = sc.nextLine();
		
		boolean rb = false, gb = false, bb = false;
		
		int r = 0, g = 0, b= 0;

		for (int i = 0; i < cards; i++) {
			if (colors.charAt(i) == 'R') r++;
			if (colors.charAt(i) == 'G') g++;
			if (colors.charAt(i) == 'B') b++;
		}
		
//		System.out.println(r + " " + g + " " + b);
		
		if (r == 0 && b == 0 && g > 0) gb = true;
		if (r == 0 && b > 0 && g == 0) bb = true;
		if (r > 0 && b == 0 && g == 0) rb = true;
		
		if (r > 0 && g > 0 && b == 0) {
			bb = true;
			if (r > 1) gb = true;
			if (g > 1) rb = true;
		}
		
		if (r > 0 && g == 0 && b > 0) {
			gb = true;
			if (b > 1) rb = true;
			if (r > 1) bb = true;
		}
		
		if (r == 0 && g > 0 && b > 0) {
			rb = true;
			if (b > 1) gb = true;
			if (g > 1) bb = true;
		}
		
		if (r > 0 && b > 0 && g > 0) {
			rb = true;
			gb = true;
			bb= true;
		}
		
		String out = "";
		if (bb) out += "B";
		if (gb) out += "G";
		if (rb) out += "R";
		
		System.out.println(out);
		
	}

}
