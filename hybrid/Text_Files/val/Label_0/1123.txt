import sys
from bisect import bisect_right
input = sys.stdin.readline
sys.setrecursionlimit(10 ** 6)
t = int(input())

def dfs(arr, canset):
	if len(set(arr)) <= 1:
		return
	mid = (min(arr) + max(arr)) // 2
	midi = bisect_right(arr, mid)
	l_arr = arr[:midi]
	canset.add(sum(l_arr))
	r_arr = arr[midi:]
	canset.add(sum(r_arr))
	dfs(l_arr, canset)
	dfs(r_arr, canset)

def solve():
	(n, q) = map(int, input().split())
	a = list(map(int, input().split()))
	a.sort()
	s = sum(a)
	can_set = set()
	can_set.add(s)
	dfs(a, can_set)
	for _ in range(q):
		s = int(input())
		if s in can_set:
			print('Yes')
		else:
			print('No')
for _ in range(t):
	solve()
