long long powmod(long long a, long long b) {
  long long res = 1;
  a %= mod;
  for (; b; b >>= 1) {
    if (b & 1) res = res * a % mod;
    a = a * a % mod;
  }
  return res;
}

  }

}
long long gcd(long long a, long long b) { return b ? gcd(b, a % b) : a; }
void upd(pair<long long, int> &u, pair<long long, int> v) {
  if (u < v) u = v;
}
const int N = 301000;
long long ans;
struct tree {
  int n, maxp, u, v;
  int q[N], f[N], vis[N], sz[N], ms[N];
  vector<int> e[N];
  long long dp[N], sump, v1, v2, ans;
  pair<long long, int> u1, u2, pu1, pu2;
  pair<int, int> ansp;
  void add(int u, int v) { e[u].push_back(v), e[v].push_back(u); }
  void input() {
    for (int i = 1; i < (int)n; i++) scanf("%d%d", &u, &v), add(u, v);
  }
  void dfs1(int u, int p, int dep) {
    dp[1] += dep;
    sz[u] = 1;
    for (int j = 0; j < (int)((int)(e[u]).size()); j++) {
      int v = e[u][j];
      if (v == p) continue;
      dfs1(v, u, dep + 1);
      sz[u] += sz[v];
    }
  }
  void dfs2(int u, int p) {
    for (int j = 0; j < (int)((int)(e[u]).size()); j++) {
      int v = e[u][j];
      if (v == p) continue;
      dp[v] = dp[u] + n - sz[v] * 2;
      dfs2(v, u);
    }
  }
  void precalc() {
    dfs1(1, 0, 0);
    dfs2(1, 0);
    maxp = 1;
    sump = 0;
    for (int i = 1; i < (int)n + 1; i++) {
      if (dp[i] >= dp[maxp]) maxp = i;
      sump += dp[i];
    }
  }
  void clear() {
    for (int i = 1; i < (int)n + 1; i++) e[i].clear();
    dp[1] = n = 0;
  }
  int find(int u) {
    int t = 1;
    q[0] = u;
    f[u] = -1;
    for (int i = 0; i < (int)t; i++) {
      u = q[i];
      for (int j = 0; j < (int)e[u].size(); j++) {
        int v = e[u][j];
        if (!vis[v] && v != f[u]) f[q[t++] = v] = u;
      }
      ms[q[i]] = 0;
      sz[q[i]] = 1;
    }
    for (int i = t - 1; i >= 0; i--) {
      ms[q[i]] = max(ms[q[i]], t - sz[q[i]]);
      if (ms[q[i]] * 2 <= t) return q[i];
      sz[f[q[i]]] += sz[q[i]];
      ms[f[q[i]]] = max(ms[f[q[i]]], sz[q[i]]);
    }
    return 0;
  }
  void dfs3(int u, int f, int dep) {
    upd(u1, make_pair(dp[u] * v1 + dep * v1 * v2, u));
    upd(u2, make_pair(dp[u] * v2 + dep * v1 * v2, u));
    for (int i = 0; i < (int)((int)(e[u]).size()); i++) {
      int v = e[u][i];
      if (v == f || vis[v]) continue;
      dfs3(v, u, dep + 1);
    }
  }
  void dfs(int u) {
    u = find(u);
    vis[u] = 1;
    pu1 = make_pair(dp[u] * v1, u);
    pu2 = make_pair(dp[u] * v2, u);
    for (int i = 0; i < (int)((int)(e[u]).size()); i++) {
      int v = e[u][i];
      if (vis[v]) continue;
      u1 = make_pair(dp[u] * v1, u);
      u2 = make_pair(dp[u] * v2, u);
      dfs3(v, u, 1);
      if (pu1.first + u2.first > ans)
        ans = pu1.first + u2.first, ansp = make_pair(pu1.second, u2.second);
      if (pu2.first + u1.first > ans)
        ans = pu2.first + u1.first, ansp = make_pair(u1.second, pu2.second);
      upd(pu1, u1);
      upd(pu2, u2);
    }
    for (int i = 0; i < (int)((int)(e[u]).size()); i++) {
      int v = e[u][i];
      if (vis[v]) continue;
      dfs(v);
    }
  }
  pair<int, int> solve(int p1, int p2) {
    v1 = p1;
    v2 = p2;
    ans = dp[1] * (p1 + p2);
    ansp = make_pair(1, 1);
    dfs(1);
    return ansp;
  }
} t[3], T;
void calc(int p1, int p2, int p3) {
  pair<int, int> pr = t[p1].solve(t[p2].n, t[p3].n);
  T.clear();
  for (int i = 1; i < (int)t[p1].n + 1; i++)
    for (int j = 0; j < (int)((int)(t[p1].e[i]).size()); j++) {
      T.e[i + T.n].push_back(t[p1].e[i][j] + T.n);
    }
  T.n += t[p1].n;
  for (int i = 1; i < (int)t[p2].n + 1; i++)
    for (int j = 0; j < (int)((int)(t[p2].e[i]).size()); j++) {
      T.e[i + T.n].push_back(t[p2].e[i][j] + T.n);
    }
  T.n += t[p2].n;
  for (int i = 1; i < (int)t[p3].n + 1; i++)
    for (int j = 0; j < (int)((int)(t[p3].e[i]).size()); j++) {
      T.e[i + T.n].push_back(t[p3].e[i][j] + T.n);
    }
  T.n += t[p3].n;
  T.add(pr.first, t[p2].maxp + t[p1].n);
  T.add(pr.second, t[p3].maxp + t[p1].n + t[p2].n);
  T.precalc();
  ans = max(ans, T.sump);
}
int main() {
  for (int i = 0; i < (int)3; i++) scanf("%d", &t[i].n);
  for (int i = 0; i < (int)3; i++) t[i].input(), t[i].precalc();
  calc(0, 1, 2);
  calc(1, 0, 2);
  calc(2, 0, 1);
  printf("%I64d\n", ans / 2);
}
