import collections
t = int(input())
for i in range(t):
	n = int(input())
	nums = collections.Counter()
	nums2 = list(map(int, input().split()))
	for i in nums2:
		nums[i] += 1
	del nums2
	nums2 = list(nums.values())
	nums2.sort(reverse=True)
	if n % 2 == 1:
		if nums2[0] <= n // 2:
			print(1)
			continue
		elif nums2[0] > n // 2:
			print(nums2[0] - n + nums2[0])
			continue
	elif n % 2 == 0:
		if nums2[0] <= n // 2:
			print(0)
		else:
			print(nums2[0] - n + nums2[0])
			continue
