#include <bits/stdc++.h>
const long long MXN = 1e5 + 7;
using namespace std;
long long n, d, a[MXN], ans;
int main() {
  ios_base::sync_with_stdio(0);
  ;
  cin >> n >> d;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  int l = 1;
  for (int i = 1; i <= n; i++) {
    while (abs(a[i] - a[l]) > d) l++;
    long long cnt = i - l - 1;
    if (cnt > 0) ans += (cnt * (cnt + 1)) / 2;
  }
  cout << ans;
}
