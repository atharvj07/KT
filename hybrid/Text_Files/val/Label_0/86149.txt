#include <bits/stdc++.h>

int dir[5] = {-1, 0, 1, 0, -1};
long long int ipow(long long int x, long long int n) {
  long long int result = 1;
  while (n) {
    if (n & 1) result = result * x;
    n = n / 2;
    x = x * x;
  }
  return result;
}
void change(string &s) {
  int n = s.size();
  for (int i = 1; i < n; i += 2) {
    s[i] = '1';
  }
  for (int i = 0; i < n; i += 2) {
    s[i] = '0';
  }
}
int main() {
  long long int n, q;
  cin >> n >> q;
  string s(n, '&');
  while (q--) {
    long long int l, r;
    cin >> l >> r;
  }
  change(s);
  cout << s << "\n";
  return 0;
}
