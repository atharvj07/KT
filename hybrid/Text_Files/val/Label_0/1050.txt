def solve(size, a, b):
	plus = False
	minus = False
	for i in range(size):
		if a[i] > b[i] and (not minus):
			return 'NO'
		if a[i] < b[i] and (not plus):
			return 'NO'
		if a[i] == 1:
			plus = True
		if a[i] == -1:
			minus = True
	return 'YES'
for _ in range(int(input())):
	size = int(input())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	print(solve(size, a, b))
