#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a = 0, b = 0;
  int t;
  for (int i = 0; i < 3; ++i) {
    cin >> t;
    a += t;
  }
  a = (a + 4) / 5;
  for (int i = 0; i < 3; ++i) {
    cin >> t;
    b += t;
  }
  b = (b + 9) / 10;
  cin >> n;
  if (a + b > n)
    printf("NO");
  else
    printf("YES");
  return 0;
}
