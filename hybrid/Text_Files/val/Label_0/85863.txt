#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); ++i)
#define repit(it, li) for(auto it=li.begin(); it!=li.end(); it++)
using namespace std;
using ll = long long;
using P = pair<int,int>;

#define M 6400
bool dp[80][80][M];
int a[80][80];
int b[80][80];

int main(){
   int h, w;
   cin>>h>>w;
   rep(i, h) rep(j, w) cin>>a[i][j];
   rep(i, h) rep(j, w) cin>>b[i][j];

   rep(i, h){  
      rep(j, w){
         int l=abs(a[i][j]-b[i][j]);
         rep(k, M) dp[i][j][k]=0;
         if(i==0 && j==0){
            dp[i][j][l]=1;                                                                                       }   
         else{
            for(int k=0; k<M; k++) if(i!=0 && dp[i-1][j][k] || j!=0 && dp[i][j-1][k]){
               dp[i][j][max(k-l, -(k-l))]=1;
               if(k+l<M) dp[i][j][k+l]=1;
            }
         }
      }
   }

   int p=0;
   for(p=0; p<M; p++) if(dp[h-1][w-1][p]==1) break;

   cout<<p<<endl;
   return 0;
}
