
import java.util.Scanner;
public class StockArbitraging {

	public static void main(String[] args) {
		Scanner sc=new Scanner (System.in);
		int n=sc.nextInt();
		int m=sc.nextInt();
		int r=sc.nextInt();
		int minA=1001;
		int A[]=new int[n];
		int B[]=new int[m];
		for (int x=0;x<n;x++) 
		{
			A[x]=sc.nextInt();
		   if (minA>A[x])
			   minA=A[x];
		}
		int maxB=0;
		for (int x=0;x<m;x++) 
		{
			B[x]=sc.nextInt();
			if (maxB<B[x])
				maxB=B[x];
		}
		int p=0;
		if (minA>maxB)
			System.out.println(r);
		else {
			p=(r/minA)*maxB;
		System.out.println(p+(r%minA));
		}
			
			
		
	}

}
