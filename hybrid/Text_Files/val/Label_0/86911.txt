import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
	static int n, p, k;
	static int[] a, b;
	static int[][][][] dp;
	static final int inf = -1000000000;

	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st;
		st = new StringTokenizer(br.readLine());
		n = Integer.parseInt(st.nextToken());
		p = Integer.parseInt(st.nextToken());
		k = Integer.parseInt(st.nextToken());
		a = new int[n + 1];
		b = new int[n + 1];
		if (k == 0) {
			System.out.println(0);
			return;
		}
		if (p > 2 * ((n + k - 1) / k))
			p = 2 * ((n + k - 1) / k);
		st = new StringTokenizer(br.readLine());
		int s = Integer.parseInt(st.nextToken());
		while (s-- > 0) {
			a[Integer.parseInt(st.nextToken())] = 1;
		}
		st = new StringTokenizer(br.readLine());
		s = Integer.parseInt(st.nextToken());
		while (s-- > 0) {
			b[Integer.parseInt(st.nextToken())] = 1;
		}
		dp = new int[2][p + 1][k + 1][k + 1];
		for (int i = 0; i < 2; i++)
			for (int j = 0; j < p + 1; j++)
				for (int x = 0; x < k + 1; x++)
					for (int y = 0; y < k + 1; y++)
						dp[i][j][x][y] = inf;
		dp[0][0][0][0] = 0;
		for (int i = 1; i <= n; i++) {
			int crt = i & 1;
			int prev = crt ^ 1;
			// A only
			// - From nothing
			for (int j = 1; j <= p; j++)
				dp[crt][j][k - 1][0] = Math.max(dp[crt][j][k - 1][0], dp[prev][j - 1][0][0] + a[i]);
			// - From something
			for (int j = 1; j <= p; j++)
				for (int x = 0; x < k - 1; x++)
					dp[crt][j][x][0] = Math.max(dp[crt][j][x][0], dp[prev][j][x + 1][0] + a[i]);
			// B only
			// - From nothing
			for (int j = 1; j <= p; j++)
				dp[crt][j][0][k - 1] = Math.max(dp[crt][j][0][k - 1], dp[prev][j - 1][0][0] + b[i]);
			// - From something
			for (int j = 1; j <= p; j++)
				for (int x = 0; x < k - 1; x++)
					dp[crt][j][0][x] = Math.max(dp[crt][j][0][x], dp[prev][j][0][x + 1] + b[i]);
			// A and B
			// - From some A
			for (int j = 1; j <= p; j++)
				for (int x = 0; x < k - 1; x++)
					if (dp[prev][j - 1][x + 1][0] + (a[i] | b[i]) > dp[crt][j][x][k - 1])
						dp[crt][j][x][k - 1] = dp[prev][j - 1][x + 1][0] + (a[i] | b[i]);
			// - From some B
			for (int j = 1; j <= p; j++)
				for (int x = 0; x < k - 1; x++)
					if (dp[prev][j - 1][0][x + 1] + (a[i] | b[i]) > dp[crt][j][k - 1][x])
						dp[crt][j][k - 1][x] = dp[prev][j - 1][0][x + 1] + (a[i] | b[i]);
			// - From some A and B
			for (int j = 2; j <= p; j++)
				for (int x = 0; x < k - 1; x++)
					for (int y = 0; y < k - 1; y++)
						if (dp[prev][j][x + 1][y + 1] + (a[i] | b[i]) > dp[crt][j][x][y])
							dp[crt][j][x][y] = dp[prev][j][x + 1][y + 1] + (a[i] | b[i]);
			// None
			for (int j = 0; j <= p; j++)
				if (dp[prev][j][0][0] > dp[crt][j][0][0])
					dp[crt][j][0][0] = dp[prev][j][0][0];

			for (int j = 0; j <= p; j++)
				for (int x = 0; x < k; x++)
					for (int y = 0; y < k; y++)
						dp[prev][j][x][y] = inf;
		}
		int res = inf;
		for (int i = 0; i <= p; i++)
			for (int x = 0; x < k; x++)
				for (int y = 0; y < k; y++)
					if (dp[n & 1][i][x][y] > res)
						res = dp[n & 1][i][x][y];
		System.out.println(res);
	}
}
