import sys, math
input = sys.stdin.readline

def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	s = input()
	return list(s[:len(s) - 1])

def invr():
	return map(int, input().split())
t = 1
t = inp()
for tt in range(t):
	a = insr()
	qw = []
	q = 0
	for i in range(len(a)):
		if i < len(a) - 1:
			w = a[i + 1]
		else:
			w = 0
		if a[i] == '?':
			for j in ['a', 'b', 'c']:
				if j != w and j != q:
					qw.append(j)
					q = j
					break
		else:
			if a[i] == w:
				print(-1)
				break
			qw.append(a[i])
			q = a[i]
	else:
		print(''.join(qw))
