def Solve(x, L):
	if x == k[0]:
		return L == S
	if (x, tuple(L)) in Mem:
		return False
	if L == S:
		return False
	E = []
	for i in range(len(L)):
		E.append(L[Q[i] - 1])
	if Solve(x + 1, E):
		return True
	E = [0] * len(L)
	for i in range(len(L)):
		E[Q[i] - 1] = L[i]
	if Solve(x + 1, E):
		return True
	Mem[x, tuple(L)] = 1
	return False
Mem = {}
k = [0]
(n, k[0]) = map(int, input().split())
P = list(range(1, n + 1))
Q = list(map(int, input().split()))
S = list(map(int, input().split()))
if Solve(0, P):
	print('YES')
else:
	print('NO')
