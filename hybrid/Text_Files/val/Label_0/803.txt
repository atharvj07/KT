import math
t = int(input())
inputs = []
for _ in range(t):
	line = input().split(' ')
	(n, k) = (int(line[0]), int(line[1]))
	xs = [int(i) for i in input().split(' ')]
	inputs.append([n, k, xs])
for ele in inputs:
	(n, k, xs) = ele
	result = 0
	for i in range(k):
		if i == 0:
			result = max(result, xs[i] - 0)
		else:
			result = max(result, math.floor((xs[i] - xs[i - 1]) / 2) + 1)
		if i == k - 1:
			result = max(result, n + 1 - xs[i])
	print(result)
