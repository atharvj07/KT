#include<iostream>
#include<algorithm>
#include<string>
using namespace std;

string flag[51];
int n, m;
int res = 1 << 29;
int cnt(int from, int to, char color){
	int res = 0;
	
	for(int i = from;i < to;i++){
		for(int j = 0;j < m;j++){
			if(flag[i][j] == color)res++;
		}
	}
	
	return res;
}

int main(){								 
	cin >> n >> m;
	for(int i = 0;i < n;i++){
		cin >> flag[i];
	}

	for(int i = 1;i < n;i++){
		for(int j = i + 1;j < n;j++){
			int tmp = 0;
			tmp += cnt(0, i, 'W');
			tmp += cnt(i, j, 'B');
			tmp += cnt(j, n, 'R');
			res = min(res, n * m - tmp);
		}
	}
	cout << res << endl;
	return 0;
}
