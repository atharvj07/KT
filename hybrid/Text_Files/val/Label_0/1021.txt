p = input().rstrip().split(' ')
a = int(p[0])
b = int(p[1])
c = int(p[2])
discriminant = (b ** 2 - 4 * a * c) ** 0.5
realroot1 = (-b + discriminant) / (2 * a)
realroot2 = (-b - discriminant) / (2 * a)
if realroot1 > realroot2:
	print(realroot1)
	print(realroot2)
else:
	print(realroot2)
	print(realroot1)
