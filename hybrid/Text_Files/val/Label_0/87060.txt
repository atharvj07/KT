import java.awt.geom.*;
import java.io.*;
import java.math.*;
import java.util.*;
import java.util.regex.*;
import static java.lang.Math.*;
public class A {

    public A() throws Exception {
        int n = in.nextInt();
        String[] names = new String[n];
        for (int i=0; i<n; i++) names[i] = in.next();

        boolean valid = true;
        boolean[][] dep = new boolean[26][26];
        for (int i=1; i<n; i++) {
            if (names[i-1].startsWith(names[i]) && names[i-1].length() > names[i].length()) {
                buf.append("Impossible\n");
                return;
            }
            int l = min(names[i-1].length(), names[i].length());
            for (int j=0; j<l; j++) {
                int c1 = names[i-1].charAt(j)-'a';
                int c2 = names[i].charAt(j)-'a';
                if (c1!=c2) {
                    dep[c1][c2] = true;
                    break;
                }
            }
        }
        for (int k=0; k<26; k++) {
            for (int i=0; i<26; i++) {
                for (int j=0; j<26; j++) {
                    dep[i][j] |= dep[i][k]&&dep[k][j];
                }
            }
        }
        for (int i=0; i<26; i++) {
            if (dep[i][i]) {
                buf.append("Impossible\n");
                return;
            }
        }

        boolean[] visited = new boolean[26];
        for (int i=0; i<26; i++) {
            if (visited[i]) continue;
            dfs(dep, visited, i);
        }
        buf.reverse();
        buf.append('\n');
    }

    void dfs(boolean[][] dep, boolean[] visited, int cur) {
        visited[cur] = true;
        for (int i=0; i<26; i++) {
            if (!dep[cur][i]) continue;
            if (visited[i]) continue;
            dfs(dep, visited, i);
        }
        buf.append((char)(cur+'a'));
    }

    public static Scanner in = new Scanner(System.in);
    public static StringBuilder buf = new StringBuilder();
    public static void debug(Object... arr) { // {{{
        System.err.println(Arrays.deepToString(arr));
    } // }}}
    public static void main(String[] args) throws Exception { // {{{
        new A();
        System.out.print(buf);
    } // }}}
    public static class Scanner { // {{{
        BufferedReader br;
        String line;
        StringTokenizer st;
        public Scanner(InputStream in) {
            br = new BufferedReader(new InputStreamReader(in));
        }
        public boolean hasNext() throws IOException {
            while ((st==null||!st.hasMoreTokens())&&(line=br.readLine())!=null)
                st = new StringTokenizer(line);
            return st.hasMoreTokens();
        }
        public String next() throws IOException {
            if (hasNext()) return st.nextToken();
            throw new NoSuchElementException();
        }
        public int nextInt() throws IOException {
            return Integer.parseInt(next());
        }
        public long nextLong() throws IOException {
            return Long.parseLong(next());
        }
        public double nextDouble() throws IOException {
            return Double.parseDouble(next());
        }
    } // }}}
}
