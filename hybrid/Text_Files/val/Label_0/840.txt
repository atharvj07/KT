n = int(input())
a = list(map(int, input().split()))
d = dict()
poses = dict()
ans = [0 for i in range(n)]
for i in range(n):
	if a[i] not in d:
		d[a[i]] = 1
		poses[a[i]] = [i]
	else:
		d[a[i]] += 1
		poses[a[i]].append(i)
color = 1
for i in d.keys():
	while d[i]:
		if len(poses[i]) < n - i:
			print('Impossible')
			exit()
		d[i] -= n - i
		for j in range(n - i):
			ans[poses[i].pop()] = color
		color += 1
print('Possible')
print(*ans)
