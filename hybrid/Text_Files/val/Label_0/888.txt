def check(M):
	for i in range(n):
		for j in range(m):
			if i - 1 >= 0 and M[i - 1][j] + m == M[i][j]:
				return False
			if i + 1 < n and M[i + 1][j] == M[i][j] + m:
				return False
			if j - 1 >= 0 and M[i][j - 1] + 1 == M[i][j]:
				return False
			if j + 1 < m and M[i][j + 1] == M[i][j] + 1:
				return False
	return True

def transpose(M):
	n = len(M)
	m = len(M[0])
	R = [[0 for i in range(n)] for j in range(m)]
	for i in range(n):
		for j in range(m):
			R[j][i] = M[i][j]
	return R
