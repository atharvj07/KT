import java.io.*; import java.util.*;
public class CF1375E {
    public static class Pair implements Comparable<Pair>{
        int ind; long val;
        public Pair(int a, long b){
            this.ind=a; this.val=b;
        }
        public int compareTo(Pair other){
            if(this.val>other.val)return -1;
            if(this.val<other.val)return 1;
            if(this.ind>other.ind)return -1;
            return 1;
        }
    }
    public static void main(String[] args) throws IOException {
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        int N=Integer.parseInt(br.readLine());
        long[] a=new long[N]; 
        StringTokenizer st=new StringTokenizer(br.readLine());
        for (int i = 0; i < N; i++) {
            a[i]=(long) N*Integer.parseInt(st.nextToken())+i; 
        }
        int cnt=0; StringBuilder sb=new StringBuilder();
        
        for (int i = 0; i <N;i++) {
            ArrayList<Pair> arl=new ArrayList<>();
            
            for (int j = i+1; j < N; j++) {
                if(a[j]<a[i]){
                    arl.add(new Pair(j,a[j]));
                }
            }
            
            Collections.sort(arl);
            
            
            for (Pair p: arl) {
                
                cnt++; 
                a[i]+=a[p.ind];
                a[p.ind]=a[i]-a[p.ind];
                a[i]-=a[p.ind];
                sb.append(i+1).append(" ").append(1+p.ind).append("\n");
                
                
            }
            
        }
        System.out.println(cnt);
        System.out.println(sb.toString());
    }
    
}
