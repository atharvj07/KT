#include <bits/stdc++.h>
using namespace std;
long long a[100], p, q;
int i, j, k, m, n, x, y, z;
int main() {
  scanf("%I64d%I64d", &p, &q);
  scanf("%d", &n);
  for (i = 1; i <= n; i++) scanf("%I64d", &a[i]);
  int flag = 1;
  for (i = 1; i <= n; i++) {
    if (a[i] > 1 << 31 - 1 && q > 1 << 31 - 1) {
      flag = 0;
      break;
    }
    if (p - a[i] * q < 0) {
      flag = 0;
      break;
    }
    p -= q * a[i];
    swap(p, q);
  }
  if (flag == 0 || q != 0) {
    printf("NO\n");
    return 0;
  }
  printf("YES\n");
  return 0;
}
