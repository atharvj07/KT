#define pb push_back


const int M = ten(5) + 100;
int n;
int x[M], y[M];
pint a[M], b[M];
int a_idx[M], b_idx[M];
int gao(int* p, int* q)
{
    rep(i, 2) rep(j, 2)
        if (p[i] == q[j])
            return ten(9);

    int pa[2], pb[2], qa[2], qb[2];
    rep(i, 2)
    {
        pa[i] = a[a_idx[p[i]]].first;
        pb[i] = b[b_idx[p[i]]].first;
        qa[i] = a[a_idx[q[i]]].first;
        qb[i] = b[b_idx[q[i]]].first;
    }

    int res = -1;
    rep(v, n)
    {
        int va = a[a_idx[v]].first;
        int vb = b[b_idx[v]].first;

        int pa_dis = max(abs(pa[0] - va), abs(pa[1] - va));
        int pb_dis = max(abs(pb[0] - vb), abs(pb[1] - vb));
        int p_dis = max(pa_dis, pb_dis);

        int qa_dis = max(abs(qa[0] - va), abs(qa[1] - va));
        int qb_dis = max(abs(qb[0] - vb), abs(qb[1] - vb));
        int q_dis = max(qa_dis, qb_dis);

        chmax(res, min(p_dis, q_dis));
    }
    return res;
}
int main()
{
    scanf("%d", &n);
    rep(i, n)
        scanf("%d%d", x + i, y + i);

    rep(i, n)
    {
        a[i] = pint(x[i] + y[i], i);
        b[i] = pint(x[i] - y[i], i);
    }
    sort(a, a + n);
    sort(b, b + n);
    rep(i, n)
    {
        a_idx[a[i].second] = i;
        b_idx[b[i].second] = i;
    }

    int res = ten(9);
    int p[] = { a[0].second, b[0].second };
    int q[] = { a[n - 1].second, b[n - 1].second };
    chmin(res, gao(p, q));
    swap(p[1], q[1]);
    chmin(res, gao(p, q));
    cout << res << endl;
}
