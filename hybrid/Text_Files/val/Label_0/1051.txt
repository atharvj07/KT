t = int(input())
while t > 0:
	n = int(input())
	ai = input().split(' ')
	a = [int(i) for i in ai]
	bi = input().split(' ')
	b = [int(i) for i in bi]
	pos = [0] * n
	neg = [0] * n
	(pos[0], neg[0]) = (int(a[0] > 0), int(a[0] < 0))
	for i in range(1, n):
		pos[i] = pos[i - 1] + (a[i] > 0)
		neg[i] = neg[i - 1] + (a[i] < 0)
	cond = True
	for i in range(n - 1, 0, -1):
		if b[i] != a[i]:
			if b[i] > a[i] and pos[i - 1] <= 0 or (b[i] < a[i] and neg[i - 1] <= 0):
				cond = False
				break
	if a[0] != b[0]:
		cond = False
	if cond:
		print('YES')
	else:
		print('NO')
	t -= 1
