const int mod = 1e9 + 7;

#define MAX_N 3000003

inline int modPow(int x, int n) {
  if(n == 0) return 1;
  int res = modPow(x, n/2);
  (res *= res) %= mod;
  if(n&1) (res *= x) %= mod;
  return res;
}
inline int modInv(int a) {
  return modPow(a, mod-2);
}
inline int modComb(int n, int r) {
  static int fact[MAX_N], inv[MAX_N];
  if(fact[0] == 0) {
    fact[0] = inv[0] = 1;
    reps(i, 1, MAX_N) {
      fact[i] = fact[i-1]*i % mod;
      inv[i] = modInv(fact[i]);
    }
  }
  if(r < 0 || n < r) return 0;
  return fact[n]*inv[r]%mod*inv[n-r]%mod;
}
inline int Homo(int n, int r) {
  return modComb(n+r-1, r);
}
