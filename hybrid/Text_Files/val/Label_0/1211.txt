f = {'OR': lambda x, y: x | y, 'AND': lambda x, y: x & y, 'XOR': lambda x, y: x ^ y}
(n, m) = map(int, input().split())
(p, u, v) = ([], [], [])
l = {'?': n}
for i in range(n):
	(q, s) = input().split(' := ')
	if ' ' in s:
		(x, t, y) = s.split()
		p += [(l[x], f[t], l[y])]
		u += [i]
	else:
		p += [int(s, 2)]
		v += [i]
	l[q] = i
s = [0] * (n + 1)

def g(k, l):
	s[n] = k
	for i in v:
		s[i] = p[i] & l > 0
	for i in u:
		(x, f, y) = p[i]
		s[i] = f(s[x], s[y])
	return sum(s) - k
a = b = ''
for j in range(m):
	l = 1 << m - j - 1
	(x, y) = (g(0, l), g(1, l))
	a += '01'[y < x]
	b += '01'[y > x]
print(a, b)
