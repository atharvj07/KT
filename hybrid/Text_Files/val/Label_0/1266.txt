def invert(s):
	t = ''
	for i in s:
		if i == '+':
			t += '-'
		else:
			t += '+'
	return t
n = int(input())
if n == 1:
	print('+')
	exit()
a = list(map(int, input().split()))
cur = a[-1]
s = '+'
for i in range(n - 2, 0, -1):
	if cur > 0:
		cur -= a[i]
		s += '-'
	else:
		cur += a[i]
		s += '+'
if cur >= a[0]:
	s += '-'
elif abs(cur) <= a[0] and cur <= 0:
	s += '+'
elif 0 < cur < a[0]:
	s = invert(s)
	s += '+'
else:
	s = invert(s)
	s += '-'
print(s[::-1])
