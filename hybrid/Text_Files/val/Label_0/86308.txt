#include <bits/stdc++.h>
using namespace std;
long long v[100000];
long long gcd(long long a, long long b) {
  if (!b) return a;
  return gcd(b, a % b);
}
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> v[i];
  sort(v, v + n);
  long long tot = 0;
  for (int i = 0; i < n; i++) tot += 2 * v[i] * (i - (n - (i + 1))) + v[i];
  long long den = n;
  long long g = gcd(tot, n);
  cout << tot / g << ' ' << den / g << endl;
  return 0;
}
