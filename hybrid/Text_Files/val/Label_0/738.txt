import sys
input = sys.stdin.readline
from collections import deque

def ri():
	return tuple(map(int, input().split(' ')))
for _ in range(int(input())):
	n = int(input())
	comms = deque()
	for _ in range(n):
		comms.append(ri())
	comms.append((int(10000000000.0), 0))
	prev_ti = 0
	pos = 0
	prev_pos = 0
	ep = 0
	target = []
	ss = 0
	while comms:
		(ti, xi) = comms.popleft()
		steps = min(abs(ep - prev_pos), ti - prev_ti)
		if prev_pos <= ep:
			pos = prev_pos + steps
			if target and prev_pos <= target[0][1] <= pos:
				ss += 1
		else:
			pos = prev_pos - steps
			if target and prev_pos >= target[0][1] >= pos:
				ss += 1
		if ep == pos:
			ep = xi
		target = [(ti, xi)]
		prev_ti = ti
		prev_pos = pos
	print(ss)
