n = int(input())
for i in range(n):
	s = str(input())
	s = list(s)
	S = ''
	for j in range(len(s)):
		if j < len(s) - 1:
			if s[j] != '?' and s[j] == s[j + 1]:
				S = '-1'
				break
			if s[j] == '?':
				if s[j + 1] != 'a' and s[j - 1] != 'a':
					s[j] = 'a'
				elif s[j + 1] != 'b' and s[j - 1] != 'b':
					s[j] = 'b'
				elif s[j + 1] != 'c' and s[j - 1] != 'c':
					s[j] = 'c'
		elif s[j] == '?':
			if s[j - 1] != 'a' and s[j - 1] != 'b':
				s[j] = 'a'
			else:
				s[j] = 'c'
		S = S + s[j]
	print(S)
