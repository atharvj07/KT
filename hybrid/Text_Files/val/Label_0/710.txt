s = input()
fi = s.find('AB')
se = s.find('BA')
ab = s.count('AB')
ba = s.count('BA')
if fi == -1 or se == -1:
	print('NO')
elif (s.count('ABAB') == 1 or s.count('BABA') == 1) and s.count('A') == 2 and (s.count('B') == 2):
	print('NO')
elif ab == 1 and ba == 1:
	if max(fi, se) - min(fi, se) > 1:
		print('YES')
	else:
		print('NO')
else:
	print('YES')
