#include <iostream>
using namespace std;

int main() {
  int cnt = 1;
  while(1) {
    int n;
    cin >> n;
    if(n == 0) break;
    int jpeg[n][n];

    jpeg[0][0] = 1;
    int vx, vy;
    int x, y;
    int num = 2;
    x = 0, y = 0;
    vx = 1, vy = 0;
    bool flag = false;
    while(1) {
      if(x == n-1 && y == n-1) break;

      if(x == n-1 && y == 0 && vx == 1 && vy == -1) {
	vx = 0;
	vy = 1;
	flag = true;
      }
      if(x == 0 && y == n-1 && vx == -1 && vy == 1) {
	vx = 1;
	vy = 0;
	flag = true;
      }

      x += vx;
      y += vy;
      if(x < 0 || x >= n) {
	x -= vx;
	y -= vy;
	vx = 0;
	vy = 1;
	continue;
      } else if(y < 0 || y >= n) {
	x -= vx;
	y -= vy;
	vx = 1;
	vy = 0;
	continue;
      }

      if((!flag && vy == 0) || (flag && vx == 0)) {
	vx = -1;
	vy = 1;
      } else if((!flag && vx == 0) || (flag && vy == 0)) {
	vx = 1;
	vy = -1;
      }
      jpeg[y][x] = num++;
    }

    cout << "Case " << cnt++ << ':' << endl;
    for(int i = 0; i < n; i++) {
      for(int j = 0; j < n; j++) {
	printf("%3d", jpeg[i][j]);
      }
      printf("\n");
    }
  }
  return 0;
}
