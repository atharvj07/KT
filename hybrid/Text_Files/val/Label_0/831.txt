(n, k) = [int(x) for x in input().split()]
dic = {}
for i in range(1, 7):
	dic[i] = []
counter = 0
opp = 0
for i in range(k):
	(a, b) = [int(x) for x in input().split()]
	for (i, j) in ((a, b), (b, a)):
		if i not in dic:
			dic[i] = [j]
		else:
			dic[i].append(j)
arr = 7
if n == 7:
	for item in dic:
		for elem in dic:
			counter = len(dic[item])
			for i in dic[item]:
				if i not in dic[elem]:
					counter -= 1
			arr = min(arr, counter)
	print(max(k - arr, 0))
else:
	print(k)
