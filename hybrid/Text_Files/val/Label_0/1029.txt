n = map(int, input().split())
t = list(map(int, input().split()))
t_l = [1]
t_r = [1]
curr_wyn = 1
for x in range(1, len(t)):
	if t[x - 1] < t[x]:
		curr_wyn += 1
		t_l.append(curr_wyn)
	else:
		curr_wyn = 1
		t_l.append(curr_wyn)
curr_wyn = 1
for x in range(len(t) - 2, -1, -1):
	if t[x + 1] > t[x]:
		curr_wyn += 1
		t_r.append(curr_wyn)
	else:
		curr_wyn = 1
		t_r.append(curr_wyn)
t_r = t_r[::-1]
curr_wyn = 0
for x in range(len(t_l)):
	if max(t_l[x], t_r[x]) >= curr_wyn:
		curr_wyn = max(t_l[x], t_r[x])
	if x + 2 <= len(t_l) - 1:
		if t[x + 2] > t[x]:
			if t_l[x] + t_r[x + 2] >= curr_wyn:
				curr_wyn = t_l[x] + t_r[x + 2]
print(curr_wyn)
