#include <bits/stdc++.h>
#pragma GCC optimize("Ofast")
using namespace std;
int a[200500];
struct Segment {
  int len, x, m;
  Segment* l = nullptr;
  Segment* r = nullptr;
  Segment() {}
  Segment(int len, int x, Segment* l, Segment* r)
      : len(len), x(x), l(l), r(r) {}
  bool operator<(const Segment other) const { return x > other.x; }
};
struct cmp {
  bool operator()(const Segment* a, const Segment* b) const {
    if (a->len == b->len) {
      return a->m < b->m;
    }
    return a->len > b->len;
  }
};
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  int n;
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }
  set<Segment*, cmp> second;
  Segment* prev = nullptr;
  for (int l = 0, r = 0; l < n;) {
    while (r + 1 <= n && a[r + 1] == a[r]) r++;
    Segment* x = new Segment(r - l + 1, a[l], nullptr, nullptr);
    x->l = prev;
    x->m = l;
    if (prev) prev->r = x;
    prev = x;
    second.insert(x);
    l = r + 1;
    r = l;
  }
  int cnt = 0;
  while (!second.empty()) {
    cnt++;
    Segment* v = *second.begin();
    second.erase(v);
    if (v->l != nullptr && v->r != nullptr) {
      if (v->l->x == v->r->x) {
        second.erase(v->l);
        second.erase(v->r);
        Segment* u = v->l;
        u->r = v->r->r;
        if (u->r) u->r->l = u;
        u->len += v->r->len;
        second.insert(u);
      } else {
        v->l->r = v->r;
        v->r->l = v->l;
      }
    } else {
      if (v->l) {
        v->l->r = nullptr;
      } else if (v->r) {
        v->r->l = nullptr;
      }
    }
  }
  cout << cnt;
  return 0;
}
