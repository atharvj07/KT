n = int(input())
x = list(map(int, input().split()))
(s, dp, mod) = (0, 1, 998244353)
for i in range(n):
	if i > 0:
		s = s * 100 % mod
	s = (s + (100 - x[i]) * dp * (i + 1)) % mod
	if i == n - 1:
		s = (s + x[i] * dp * (i + 1)) % mod
	dp = dp * x[i] % mod
ans = s * pow(dp, mod - 2, mod) % mod
print(ans)
