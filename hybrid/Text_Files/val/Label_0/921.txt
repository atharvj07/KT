for _ in range(int(input())):
	a = int(input())
	li = list(map(int, input().split()))
	d = {}
	e = []
	for i in li:
		try:
			d[i] += 1
		except KeyError:
			d[i] = 1
			e.append(i)
	m = a // 2
	k = 0
	su = 0
	for i in e:
		if su + d[i] <= m:
			k += 1
			su += d[i]
		else:
			break
	if k < 3:
		print(0, 0, 0)
	else:
		g = d[e[0]]
		h = 0
		ind = 1
		while h <= g and ind < k:
			h += d[e[ind]]
			ind += 1
		u = 0
		for i in range(ind, k):
			u += d[e[i]]
		if g < h and g < u:
			print(g, h, u)
		else:
			print(0, 0, 0)
