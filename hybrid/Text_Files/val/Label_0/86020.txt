#include <bits/stdc++.h>
using namespace std;
struct elem {
  int x;
  elem* next = nullptr;
  elem* prev = nullptr;
};
elem* ins(elem& e, int& x) {
  elem* pne = new elem;
  elem& ne = *pne;
  ne.x = x;
  ne.next = &e;
  if (e.prev != nullptr) {
    e.prev->next = &ne;
    ne.prev = e.prev;
  }
  e.prev = &ne;
  return pne;
}
elem* insf(elem& e, int& x) {
  elem* pne = new elem;
  elem& ne = *pne;
  ne.x = x;
  ne.prev = &e;
  if (e.next != nullptr) {
    e.next->prev = &ne;
    ne.next = e.next;
  }
  e.next = &ne;
  return pne;
}
int main() {
  char c1, c2;
  elem e;
  e.x = 1;
  elem* pe = &e;
  elem* b = pe;
  int n = 1;
  scanf("%c%c", &c1, &c2);
  while (c2 - '\n') {
    ++n;
    if (c1 == 'l') {
      pe = ins(*pe, n);
      if (pe->prev == nullptr) b = pe;
    } else
      pe = insf(*pe, n);
    c1 = c2;
    scanf("%c", &c2);
  }
  for (int i = 0; i < n; i++) {
    printf("%d\n", b->x);
    b = b->next;
  }
  return 0;
}
