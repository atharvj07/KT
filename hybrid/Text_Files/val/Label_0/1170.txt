import sys
from collections import defaultdict as dc
from collections import Counter
from bisect import bisect_right, bisect_left
import math
(n, m) = map(int, input().split())
l = []
for _ in range(n):
	l.append(list(input()))
c = list(map(int, input().split()))
s = 0
for j in range(m):
	x = dc(int)
	for i in range(n):
		x[l[i][j]] += 1
	s += c[j] * max(x.values())
print(s)
