#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n;
  int m;
  cin >> m >> n;
  int p[n];
  for (int i = 0; i < n; i++) {
    int y = 1;
    cout << y << endl;
    fflush(stdout);
    int x;
    cin >> x;
    if (x == 0 || x == -2) {
      return 0;
    } else if (x == 1) {
      p[i] = 1;
    } else
      p[i] = 0;
  }
  int l = 1;
  int r = m;
  int i = 0;
  while (l <= r) {
    int mid = (l + r) / 2;
    cout << mid << endl;
    fflush(stdout);
    int x;
    cin >> x;
    if (x == 0 || x == -2)
      return 0;
    else if (x == 1) {
      if (p[i % n] == 1) {
        l = mid + 1;
      } else
        r = mid - 1;
    } else {
      if (p[i % n] == 1) {
        r = mid - 1;
      } else
        l = mid + 1;
    }
    i++;
  }
  return 0;
}
