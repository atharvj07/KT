import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Scanner;

public class Main {

    static int n,k;
    static int[][] dp;
    static final int INF = 1000000007;

    public static void main(String[] args) {

        Scanner scan = new Scanner(System.in);

        PrintWriter out = new PrintWriter(System.out);

        n = scan.nextInt();
        k = scan.nextInt();
        dp = new int[n+1][k+1];

//      ArrayList<ArrayList<Integer>> divisor = new ArrayList<ArrayList<Integer>>();
//      divisor.add(0,new ArrayList<Integer>());

        short[][] divisor = new short[n+1][];

        for (int i = 1; i <= n ; i++) {
//          ArrayList<Integer> list = new ArrayList<Integer>();
            short[] list = new short[n];
            int cnt = 0;
            for (short j = 1; j <= i ; j++) {
//              if(i%j==0) list.add(j);
                if(i%j==0){
                    list[cnt] = j;
                    cnt++;
                }
            }
//          divisor.add(i, list);
            divisor[i] = list;
        }

        for (int i = 1; i <= n ; i++) {
            dp[i][1] = 1;
        }


        for (int j = 2; j <= k ; j++) {
            for (int i = 1; i <= n ; i++) {
                int sum = 0;
//              ArrayList<Integer> list = divisor.get(i);
                short[] list = divisor[i];
                for (int q = 0; q < n; q++) {
                    if(list[q]==0) break;
                    sum = (sum + dp[list[q]][j-1])% INF;
                }
//              for (Integer p : list) {
//                  sum = (sum + dp[p][j-1]) % INF;
//              }
                dp[i][j] = sum;
            }
        }

        int ans = 0;
        for (int i = 1; i <= n ; i++) {
            ans = (ans + dp[i][k]) % INF;
        }

        out.println(ans);
        out.flush();
    }
}
