import java.util.Scanner;


public class NPrimeFactors {

    public static void main(String[] args){
        Scanner input = new Scanner(System.in);
        while(input.hasNext()){
            System.out.println(getBin(input.nextLong()));
        }
        input.close();
    }
    
    public static long getBin(long num){
        char[] chars = (num+"").toCharArray();
        String newNum = "";
        for(int a=0; a<chars.length; a++)
            if(chars[a] > '1'){
                for(; a<chars.length; a++)
                    newNum += "1";
                break;
            }else newNum += chars[a];
                
        return Long.parseLong(newNum, 2);
    }
    
}
