(n, h, a, b, k) = map(int, input().split())
for i in range(k):
	(ta, fa, tb, fb) = map(int, input().split())
	res = 0
	if ta == tb:
		print(abs(fa - fb))
	else:
		res = abs(ta - tb)
		if fa < a:
			res += a - fa
			fa = a
		elif fa > b:
			res += fa - b
			fa = b
		res += abs(fa - fb)
		print(res)
