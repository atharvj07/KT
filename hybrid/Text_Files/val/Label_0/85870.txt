#include <bits/stdc++.h>
const int Inf = 2 * 1000 * 1000 * 1000;
long long LINF = (long long)4e18;
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long H;
  cin >> H;
  long long n;
  cin >> n;
  vector<long long> d(n);
  for (int(i) = 0; (i) < n; (i)++) cin >> d[i];
  long long sum_d = 0;
  for (int(i) = 0; (i) < n; (i)++) sum_d += d[i];
  long long ans = 1e18;
  long long tmp = 0;
  for (int i = 0; i < n; ++i) {
    H += d[i];
    tmp++;
    if (H <= 0) {
      ans = tmp;
      break;
    }
    if (sum_d < 0) ans = min(ans, tmp + ((H - 1) / -sum_d + 1) * n);
    if (H <= 0) break;
  }
  if (ans > 3 * 1e17) {
    cout << -1;
    return 0;
  }
  cout << ans;
  return 0;
}
