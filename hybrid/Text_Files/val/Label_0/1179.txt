t = int(input())
for _ in range(t):
	(n, m) = map(int, input().split())
	l = input()
	if '1' not in l:
		print(l)
	else:
		pos = []
		l = list(l)
		for i in range(n):
			if l[i] == '1':
				pos.append(i)
		p = pos[0]
		if p != 0:
			for j in range(1, m + 1):
				if p - j >= 0:
					l[p - j] = '1'
				else:
					break
		for i in range(len(pos) - 1):
			p = pos[i]
			p_1 = pos[i + 1]
			d = p_1 - p - 1
			for j in range(1, m + 1):
				if p + j < p_1:
					l[p + j] = '1'
				else:
					break
			for j in range(1, m + 1):
				if p_1 - j > p:
					l[p_1 - j] = '1'
				else:
					break
			if d % 2 == 1:
				l[(p_1 + p) // 2] = '0'
		p = pos[-1]
		if p != n - 1:
			for j in range(1, m + 1):
				if p + j < n:
					l[p + j] = '1'
				else:
					break
		print(''.join(l))
