Read = lambda : map(int, input().split())

def solve():
	if 2 * h < d:
		return False
	L = []
	for i in range(1, h + 1):
		L.append((i, i + 1))
	remain = d - h
	now = h + 2
	if not remain and n > h + 1:
		if d == 1:
			return False
		while now <= n:
			L.append((2, now))
			now += 1
	else:
		while now <= n:
			L.append((1, now))
			for _ in range(remain - 1):
				if now == n:
					break
				L.append((now, now + 1))
				now += 1
			now += 1
	for (x, y) in L:
		print(x, y)
	return True
while True:
	try:
		(n, d, h) = Read()
	except:
		break
	if not solve():
		print(-1)
