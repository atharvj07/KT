#include <bits/stdc++.h>
using namespace std;
int n;
int l[1005], r[1005];
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) scanf("%d %d", &l[i], &r[i]);
    int curr = l[1];
    printf("%d ", curr);
    curr++;
    for (int i = 2; i <= n; i++) {
      if (curr > r[i])
        printf("0 ");
      else {
        curr = max(curr, l[i]);
        printf("%d ", curr);
        curr++;
      }
    }
    printf("\n");
  }
  return 0;
}
