#include <bits/stdc++.h>
using namespace std;
int main() {
  int m, n;
  cin >> m >> n;
  bool p[n];
  for (int i = 0; i < n; ++i) {
    cout << 1 << '\n';
    fflush(stdout);
    int x;
    cin >> x;
    if (x == 0) exit(0);
    if (x == -2) exit(0);
    if (x == 1)
      p[i] = 1;
    else
      p[i] = 0;
  }
  int pol = 0;
  long long l = 1, r = m;
  while (1) {
    long long mid = (l + r) / 2;
    cout << mid << '\n';
    fflush(stdout);
    int x;
    cin >> x;
    if (x == 0) exit(0);
    if (x == -2) exit(0);
    if (p[pol] == 0) x = -x;
    if (x == -1) {
      r = mid - 1;
    } else
      l = mid + 1;
    pol++;
    pol %= n;
  }
  return 0;
}
