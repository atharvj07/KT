for t in range(int(input())):
	n = int(input())
	s = input()
	if n < 3:
		print(1)
	else:
		a = []
		b = []
		c = 1
		j = 0
		for i in range(1, n):
			if s[i] == s[i - 1]:
				c += 1
			else:
				a.append(c)
				if c > 1:
					b.append(j)
				c = 1
				j += 1
		a.append(c)
		if c > 1:
			b.append(j)
		la = len(a)
		lb = len(b)
		if lb == 0:
			print((la + 1) // 2)
		else:
			ia = ib = r = 0
			while ia < la:
				r += 1
				if a[ia] == 1:
					a[b[ib]] -= 1
					if a[b[ib]] == 1:
						ib += 1
						if ib >= lb:
							r += (la - ia) // 2
							break
				else:
					ib += 1
					if ib >= lb:
						r += (la - ia) // 2
						break
				ia += 1
			print(r)
