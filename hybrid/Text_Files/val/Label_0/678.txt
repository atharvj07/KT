import math
t = int(input())
for test in range(t):
	(n, k) = map(int, input().split(' '))
	if n % 2 == 0:
		a = k % n
		if k % n == 0:
			a += n
		print(a)
	else:
		three = int(n / 2)
		a = k % n
		if k % n == 0:
			a += n
		z = math.ceil(k / three) - 1
		ans = a + z
		r = ans % n
		if r == 0:
			r += n
		print(r)
