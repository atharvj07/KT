(n, l) = map(int, input().split())
c = list(map(int, input().split()))
eff = [0]
best = 0
for i in range(1, n):
	if c[i] < 2 ** (i - best) * c[best]:
		eff.append(i)
		best = i
cost = 0
fcost = 10 ** 1000
for curr in range(len(eff) - 1, -1, -1):
	num = l // 2 ** eff[curr]
	l -= 2 ** eff[curr] * num
	cost += num * c[eff[curr]]
	fcost = min(fcost, cost + c[eff[curr]])
print(min(fcost, cost))
