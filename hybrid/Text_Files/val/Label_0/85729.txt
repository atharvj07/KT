#include <bits/stdc++.h>
using namespace std;
struct queueMax {
  stack<pair<long long, long long>> s1, s2;
  long long size() { return s1.size() + s2.size(); }
  bool isEmpty() { return (size() == 0); }
  void clear() {
    while (!s1.empty()) {
      s1.pop();
    }
    while (!s2.empty()) {
      s2.pop();
    }
  }
  long long getMax() {
    if (isEmpty()) {
      return -1e18;
    }
    if (!s1.empty() && !s2.empty()) {
      return max(s1.top().second, s2.top().second);
    }
    if (!s1.empty()) {
      return s1.top().second;
    }
    return s2.top().second;
  }
  void push(long long val) {
    if (s2.empty()) {
      s2.push({val, val});
    } else {
      s2.push({val, max(val, s2.top().second)});
    }
  }
  void pop() {
    if (s1.empty()) {
      while (!s2.empty()) {
        if (s1.empty()) {
          s1.push({s2.top().first, s2.top().first});
        } else {
          s1.push({s2.top().first, max(s2.top().first, s1.top().second)});
        }
        s2.pop();
      }
    }
    s1.pop();
  }
};
long long memo[5005][5005];
int32_t main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long n, K, x;
  cin >> n >> K >> x;
  vector<long long> arr(n + 1);
  for (long long i = 1; i <= n; ++i) {
    cin >> arr[i];
  }
  const long long inf = 1e17;
  for (long long i = 0; i <= n; ++i) {
    for (long long j = 0; j <= x; ++j) {
      memo[i][j] = -inf;
    }
  }
  memo[0][0] = 0;
  queueMax q;
  for (long long j = 1; j <= x; ++j) {
    q.clear();
    q.push(memo[0][j - 1]);
    for (long long i = 1; i <= n; ++i) {
      memo[i][j] = max(memo[i][j], q.getMax() + arr[i]);
      q.push(memo[i][j - 1]);
      if (i - K >= 0) q.pop();
    }
  }
  long long res = -inf;
  for (long long i = 1; i <= n; ++i) {
    if (n - i < K) {
      res = max(res, memo[i][x]);
    }
  }
  if (res < 0) {
    cout << "-1\n";
  } else
    cout << res << '\n';
  return 0;
}
