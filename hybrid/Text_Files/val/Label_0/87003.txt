    // for task C vk_cup qualification1, returns hamming distance between 2 char arrays
    private int hamming(char[] chars1, char[] chars2) {
        int hamDist = 0;
        for (int i = 0; i < 6; i++) {
            if (chars1[i] != chars2[i]) hamDist++;
        }
        return hamDist;
    }

    private int min(int a, int b, int c) {
        if (a < b)
            return a < c ? a : c;
        else
            return b < c ? b : c;
    }


    //BE CAREFUL
    private char nextChar() throws IOException {
        in.nextToken();
        return in.sval.charAt(0);
    }

    private char nextOrdinaryChar() throws IOException {
        in.nextToken();
        return (char) in.ttype;
    }

    private String nextString() throws IOException {
        in.nextToken();
        return in.sval;
    }


    int nextInt() throws IOException {
        in.nextToken();
        return (int) in.nval;
    }


    double nextDouble() throws IOException {
        in.nextToken();
        return in.nval;
    }

    class Node {
        Node next, prev;
        String s;

        public Node(Node next, Node prev, String s) {
            this.next = next;
            this.prev = prev;
            this.s = s;
        }
    }
