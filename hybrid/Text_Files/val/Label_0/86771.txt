import java.util.Scanner;

public class Main{

	public static void main(String[] args) {
		Scanner in=new Scanner(System.in);
		int n=in.nextInt();
		int v[]=new int[n*n];
		for (int i = 0; i < v.length; i++) {
			v[i]=i+1;
		}
		
		for (int i = 0; i < v.length/2; i++) {
			System.out.println(v[i]+" "+v[((n*n)-1)-i]);
		}

	}

}
