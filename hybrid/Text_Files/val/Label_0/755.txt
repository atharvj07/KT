(n, m) = map(int, input().split())
l1 = list(map(int, input().split()))
l2 = list(map(int, input().split()))
dist = [0 for i in range(n + 2)]
for i in range(len(l1)):
	dist[i] = dist[i - 1] + l1[i]
dist.insert(0, 0)
pointer = 0
i = 0
while i < m:
	if l2[i] <= dist[pointer + 1]:
		print(pointer + 1, l2[i] - dist[pointer])
		i += 1
	else:
		pointer += 1
