bool isPrime(int n)
{
  if (n < 2) return false;
  if (n < 4) return true;
  if (!(n & 1))   return false;
  if (n % 3 == 0) return false;

  int lim = sqrt(n);
  for (int i = 5; i <= lim; i += 6)
    if (n % i == 0 || n % (i + 2) == 0)
      return false;

  return true;
}

//-------------------------------PRIME FACTORS----------------------------------------//
map<int, int> primeFact(int n) {
  map<int, int> mp;
  while (n % 2 == 0) {
    mp[2]++;
    n = n / 2;
  }
  for (int i = 3; i <= sqrt(n); i += 2) {
    while (n % i == 0) {
      mp[i]++;
      n = n / i;
    }
  }
  if (n > 2) {
    mp[n]++;
  }
  return mp;
}

//----------------------------BINOMIAL COEFFICIENT------------------------------------//
int binomialCoeff(int n, int k)
{
  if (k > n)return 0;
  int C[n + 1][k + 1];
  int i, j;
  for (i = 0; i <= n; i++) {
    for (j = 0; j <= min(i, k); j++) {
      if (j == 0 || j == i)
        C[i][j] = 1;
      else
        C[i][j] = (C[i - 1][j - 1] % (mod) + C[i - 1][j] % (mod)) % (mod);
    }
  }

  return C[n][k];
}

int countDivisors(int n) {
  int cnt = 0;
  for (int i = 1; i <= sqrt(n); i++) {
    if (n % i == 0) {
      if (n / i == i)cnt++;
      else {
        cnt += 2;
      }
    }
  }
  return cnt;
}
int reverse_number(int n) {
  int rev = 0, r;
  while (n != 0) {
    r = n % 10;
    rev = rev * 10 + r;
    n /= 10;
  }
  return rev;
}
