for _ in range(int(input())):
	input()
	p = map(int, input().split())
	even = []
	odd = []
	for (i, x) in enumerate(p):
		i += 1
		if x % 2 == 0:
			even.append(i)
		else:
			odd.append(i)
	if len(even) != 0:
		print(len(even))
		print(' '.join([str(x) for x in even]))
	elif len(odd) >= 2:
		print(2)
		print(' '.join([str(x) for x in odd[:2]]))
	else:
		print(-1)
