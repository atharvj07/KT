#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a, mn = INT_MAX, ans = INT_MAX;
  cin >> n;
  map<int, int> mp;
  map<int, int> mp_ans;
  for (int i = 1; i <= n; i++) {
    cin >> a;
    if (mp[a] == 0)
      mp[a] = i;
    else {
      if (a <= mn) {
        mn = a;
        if (mp_ans[a] != 0) {
          ans = min(mp_ans[a], i - mp[a]);
          mp_ans[a] = ans;
        } else {
          mp_ans[a] = i - mp[a];
        }
      }
      mp[a] = i;
    }
  }
  cout << mp_ans[mn];
}
