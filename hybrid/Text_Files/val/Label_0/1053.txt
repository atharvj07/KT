import sys
from math import sqrt, floor
from collections import deque, Counter
inp = sys.stdin.readline
read = lambda : list(map(int, inp().split()))

def a():
	ans = ''
	for _ in range(int(inp())):
		(n, m) = read()
		tem = []
		pat = list('BW' * 100)
		for i in range(n):
			tem += pat[i % 2:m + i % 2] + ['\n']
		if tem.count('W') + 1 == tem.count('B'):
			ans += ''.join(tem)
		else:
			for i in range(-1, -len(tem), -1):
				if tem[i] == 'W':
					tem[i] = 'B'
					break
			ans += ''.join(tem)
	print(ans)

def b():
	ans = ''
	for _ in range(int(inp())):
		n = int(inp())
		a = read()
		b = read()
		nums = set()
		tem = 'YES'
		for i in range(n):
			if a[i] < b[i]:
				if not 1 in nums:
					tem = 'NO'
					break
				else:
					nums.add(a[i])
			elif a[i] > b[i]:
				if not -1 in nums:
					tem = 'NO'
					break
				else:
					nums.add(a[i])
			else:
				nums.add(a[i])
		ans += tem + '\n'
	print(ans)
b()
