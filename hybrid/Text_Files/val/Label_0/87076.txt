import java.io.*;
import java.util.*;

public class CF259B {
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		int[][] aa = new int[3][3];
		for (int i = 0; i < 3; i++) {
			StringTokenizer st = new StringTokenizer(br.readLine());
			for (int j = 0; j < 3; j++)
				aa[i][j] = Integer.parseInt(st.nextToken());
		}
		int xpy = aa[0][2] + aa[2][0];
		int xmy = aa[1][2] - aa[0][1];
		int x = (xpy + xmy) / 2;
		int y = (xpy - xmy) / 2;
		aa[0][0] = x;
		aa[2][2] = y;
		aa[1][1] = aa[0][0] + aa[0][1] + aa[0][2] - aa[1][0] - aa[1][2];
		for (int i = 0; i < 3; i++)
			System.out.println(aa[i][0] + " " + aa[i][1] + " " + aa[i][2]);
	}
}
