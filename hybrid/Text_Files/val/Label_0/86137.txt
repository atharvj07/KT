#include <bits/stdc++.h>
using namespace std;

int main(){
  int n,k;
  cin>>n>>k;
  vector<int> h(n);
  for(int i=0; i<n; i++)cin>>h[i];

  vector<int> dp(n, 1000000000);

  dp[0]=0;
  for(int i=1; i<n; ++i){
    int x=max(0, i-k);
    for(int j=x; j<i; ++j)dp[i]=min(dp[i], dp[j]+abs(h[i]-h[j]));
  }

  cout<<dp[n-1]<<endl;
  

}
