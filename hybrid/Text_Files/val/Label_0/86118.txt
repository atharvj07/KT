#include <bits/stdc++.h>

using namespace std;

const int maxn = 305,mo = int(1e9) + 7;

bitset<maxn> a[maxn];
int f[maxn][maxn],pw[maxn * maxn],n;

int pow(int a,int b)
{
	int tmp = 1;
	for(;b;b >>= 1,a = a * 1ll * a % mo)
		if (b & 1) tmp = tmp * 1ll * a % mo;
	return tmp;
}

int main()
{
	scanf("%d", &n);
	for(int i = 1;i <= n;i ++)
		for(int j = 1,v;j <= n;j ++)
			scanf("%d", &v),a[i][j] = v;
	int r = 0;
	for(int i = 1,j = 1;i <= n && j <= n;j ++)
	{
		for(int k = i;k <= n;k ++) if (a[k][j]) {swap(a[i],a[k]);break;}
		if (!a[i][j]) continue;
		++ r;
		for(int k = i + 1;k <= n;k ++) if (a[k][j]) a[k] ^= a[i];
		++ i;
	}
	pw[0] = 1;
	for(int i = 1;i <= n * n;i ++) pw[i] = pw[i - 1] * 2ll % mo;
	f[0][0] = 1;
	for(int i = 1;i <= n;i ++)
		for(int j = 0;j <= i;j ++)
		{
			f[i][j] = f[i - 1][j] * 1ll * pw[j] % mo; 
			if (j) f[i][j] = (f[i][j] + f[i - 1][j - 1] * 1ll * (pw[n] - pw[j - 1]) % mo) % mo;
		}
	int ans = 0;
	for(int i = r;i <= n;i ++)
		ans = (ans + f[n][i] * 1ll * pw[n * (n - i)] % mo * f[i][r]) % mo;
	printf("%d\n", (ans * 1ll * pow(f[n][r],mo - 2) % mo + mo) % mo);
	return 0;
}
