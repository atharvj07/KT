#include<bits/stdc++.h>
#include<algorithm>
#include<vector>
#include<queue>
using namespace std;

int main(){
	int n;
	scanf("%d",&n);
	string c;
	cin>>c;
	int res=0;
	for(int i=0;i<n;i++){
		int ok=0;
		string s;
		cin>>s;
		for(int j=0;j<s.length();j++){
			for(int k=1;k<s.length();k++){
				for(int l=0;l<c.length();l++){
					if(j+k*l>s.length())break;
					if(s[j+k*l]!=c[l])break;
					if(l==c.length()-1)ok=1;
				}
				if(ok)break;
			}
			if(ok)break;
		}
		if(ok)res++;
	}
	printf("%d\n",res);
}
