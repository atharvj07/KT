from collections import deque

def solve():
	t = int(input())
	for i in range(t):
		n = int(input())
		m = [int(s) for s in input().split()]
		d = deque([m[0]])
		for j in range(1, n):
			if m[j] < d[0]:
				d.appendleft(m[j])
			else:
				d.append(m[j])
		print(' '.join((str(s) for s in d)))
solve()
