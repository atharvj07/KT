#include<bits/stdc++.h>
using namespace std;
#define int long long


struct edge{int to,cost;};
typedef pair<int,int>P;
const int INF=LLONG_MAX/100;
#define fi first
#define se second


int N,M,C;
vector<edge>G[100000];
int dist[100000];
bool used[100000];
signed main(){
    cin.tie();
    ios_base::sync_with_stdio(false);
    int sum=0;
    cin>>N>>M>>C;
    for(int i=0;i<M;i++){
        int a,b,c;
        cin>>a>>b>>c;
        G[--a].push_back((edge){--b,c});
        G[b].push_back((edge){a,c});
        sum+=c;
    }


    fill_n(dist,N,INF);
    dist[0]=0;

    priority_queue<P,vector<P>,greater<P> >Q;
    Q.push(P(0,0));

    while(Q.size()){
        P p=Q.top();Q.pop();
        if(dist[p.se]<p.fi)continue;
        for(int i=0;i<G[p.se].size();i++){
            edge &e=G[p.se][i];
            if(dist[e.to]<=p.fi+e.cost)continue;
            dist[e.to]=p.fi+e.cost;
            Q.push(P(dist[e.to],e.to));
        }
    }

    vector<P>V;
    for(int i=0;i<N;i++)V.push_back(P(dist[i],i));
    sort(V.begin(),V.end());

    int mi=sum;

    //for(int i=0;i<N;i++)cout<<"dist["<<i<<"]="<<dist[i]<<endl;

    fill_n(used,N,false);
    int cur=0;
    while(cur<N){
        int to=cur;
        while(to<N&&V[to].fi==V[cur].fi){
            used[V[to].se]=true;
            for(int i=0;i<G[V[to].se].size();i++){
                edge &e=G[V[to].se][i];
                if(!used[e.to])continue;
                sum-=e.cost;
            }
            to++;
        }
        mi=min(mi,C*V[cur].fi+sum);
        cur=to;
        //cout<<V[cur].fi<<" "<<sum<<endl;
    }

    cout<<mi<<endl;
    return 0;
}
