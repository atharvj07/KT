#include <bits/stdc++.h>
using namespace std;
const int N = 100000;
int main() {
  int n, h, d;
  cin >> n >> h >> d;
  if (n == 2 && h == 1 && d == 1) {
    cout << 1 << ' ' << 2 << endl;
    return 0;
  }
  if (d * 2 < h || h == 1) {
    cout << -1;
    return 0;
  }
  if (d == 1 && h == 2) {
    for (int i = 2; i <= n; i++) printf("%d %d\n", 1, i);
    return 0;
  }
  for (int i = 1; i <= d; i++) {
    printf("%d %d\n", i, i + 1);
  }
  for (int i = 1, j = d + 1; i <= h - d; i++, j++) {
    if (i == 1)
      printf("%d %d\n", 1, j + 1);
    else
      printf("%d %d\n", j, j + 1);
  }
  for (int i = h + 2; i <= n; i++) printf("2 %d\n", i);
}
