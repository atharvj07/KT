INF = float('inf')

def inp():
	return int(sys.stdin.readline())

def inpl():
	return list(map(int, sys.stdin.readline().split()))

def inpl_1():
	return list(map(lambda x: int(x) - 1, sys.stdin.readline().split()))
M = 27
for _ in range(inp()):
	n = inp()
	A = inpl()
	aa = [[0] * (n + 1) for _ in range(M)]
	numind = [[] for _ in range(M)]
	for (i, x) in enumerate(A):
		numind[x].append(i)
		for j in range(M):
			aa[j][i + 1] = aa[j][i] + (x == j)
	res = 0
	for i in range(M):
		res = max(res, aa[i][-1])
	for a in range(1, M):
		for cnta in range(1, aa[a][-1] // 2 + 1):
			l = numind[a][cnta - 1]
			r = numind[a][aa[a][-1] - cnta]
			cnt = [0] * M
			for i in range(l, r + 1):
				if A[i] != a:
					cnt[A[i]] += 1
			ans = cnta * 2 + max(cnt)
			res = max(res, ans)
	print(res)
