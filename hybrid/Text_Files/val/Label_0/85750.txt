#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main(void) {
    int t;
    while (cin>>t && t) {
        vector<int> s(t);
        for (int i = 0; i < t; i++) cin >> s[i];
        int n;
        cin >> n;
        vector<int> d(101, 24);
        for (int i = 0; i < n; i++) {
            int a, b;
            cin >> a >> b;
            a--;
            d[a] = min(d[a], b);
        }
        
        vector<vector<int> > dp(101, vector<int>(t, 101));
        dp[0][0] = 0;
        for (int i = 0; i < 100; i++) {
            for (int j = 0; j < t; j++) {
                if (s[j] <= d[i]) dp[i+1][(j+1)%t] = min(dp[i+1][(j+1)%t], dp[i][j]);
                dp[i+1][1] = min(dp[i+1][1], dp[i][j]+1);
            }
        }
        cout << *min_element(dp[100].begin(), dp[100].end()) << endl;
    }
}
