import math
nbTest = int(input())

def findHighest2Pow(n):
	return int(math.log(n, 2))
for k in range(nbTest):
	n = int(input())
	T = list(map(int, input().split(' ')))
	sol = 0
	dic = {}
	for k in range(0, n):
		pow2 = findHighest2Pow(T[k])
		if pow2 in dic.keys():
			sol += dic[pow2]
			dic[pow2] += 1
		else:
			dic[pow2] = 1
	print(sol)
