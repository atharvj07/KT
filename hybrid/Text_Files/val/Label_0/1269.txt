(n, p) = map(int, input().split())
s = list(input())
j = n
i = 1
while i < n:
	if s[i] > s[0]:
		j = i
		break
	elif s[i] == s[0]:
		k = i
		l = 0
		while k < n and s[k] == s[l]:
			k += 1
			l += 1
		if k >= n or s[k] > s[l]:
			j = i
			break
		else:
			i = k
	else:
		i += 1
print((''.join(s[:j]) * p)[:p])
