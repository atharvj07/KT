#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, l, a;
  scanf("%lld %lld %lld", &n, &l, &a);
  if (n == 0)
    printf("%lld\n", l / a);
  else {
    long long count = 0;
    long long i = 0;
    while (n--) {
      long long x, y;
      scanf("%lld %lld", &x, &y);
      count += (x - i) / a;
      i = x + y;
    }
    count += (l - i) / a;
    printf("%lld\n", count);
  }
  return 0;
}
