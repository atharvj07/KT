(n, q) = input().split()
(n, q) = (int(n), int(q))
s = input()
abc = s.count('abc')
s = list(s)
lst = []
while q > 0:
	(i, c) = input().split()
	i = int(i)
	if s[i - 1] != c:
		if i >= 1 and i + 1 <= len(s) - 1:
			if s[i - 1] == 'a' and s[i] == 'b' and (s[i + 1] == 'c'):
				abc -= 1
			elif c == 'a' and s[i] == 'b' and (s[i + 1] == 'c'):
				abc += 1
		if i >= 2 and i <= len(s) - 1:
			if s[i - 1] == 'b' and s[i - 2] == 'a' and (s[i] == 'c'):
				abc -= 1
			elif c == 'b' and s[i - 2] == 'a' and (s[i] == 'c'):
				abc += 1
		if i >= 3:
			if s[i - 1] == 'c' and s[i - 2] == 'b' and (s[i - 3] == 'a'):
				abc -= 1
			elif c == 'c' and s[i - 2] == 'b' and (s[i - 3] == 'a'):
				abc += 1
	s[i - 1] = c
	lst.append(str(abc))
	q -= 1
print('\n'.join(lst))
