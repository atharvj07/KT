(a, b, c) = map(int, input().split())
d = b ** 2 - 4 * a * c
ans2 = []
if a == 0 and c == 0:
	print('%.6f' % 0)
elif a == 0:
	ans = -1 * c / b
	print('%.6f' % ans)
elif d == 0:
	ans = -1 * b / (2 * a)
	print('%.6f' % ans)
else:
	ans = (-b + d ** 0.5) / (2 * a)
	ans1 = (-b - d ** 0.5) / (2 * a)
	ans2.append(ans)
	ans2.append(ans1)
	ans2.sort()
	print('%.6f' % ans2[1])
	print('%.6f' % ans2[0])
