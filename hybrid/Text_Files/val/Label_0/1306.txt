N = list(map(int, input().split()))
if N[2] < N[1] / 2 or (N[1] == 1 and N[0] != 2):
	print(-1)
else:
	for i in range(N[2]):
		print(i + 1, i + 2)
	if N[2] != N[1]:
		print(1, N[2] + 2)
	for i in range(N[1] - N[2] - 1):
		print(i + N[2] + 2, i + N[2] + 3)
	if N[2] == N[1]:
		for i in range(N[0] - N[1] - 1):
			print(2, i + N[1] + 2)
	else:
		for i in range(N[0] - N[1] - 1):
			print(1, i + N[1] + 2)
