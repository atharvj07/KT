def noc(x, y, d):
	n = y - x + 1
	n += x % d
	if (y + 1) % d != 0:
		n += d - (y + 1) % d
	return int(n / d) - 1
for tc in range(int(input())):
	(x, y) = map(int, input().split())
	ans = 0
	i = 0
	while True:
		temp = noc(x, y, 10 ** i)
		ans += temp
		i += 1
		if temp == 0 or i > 100:
			break
	print(ans)
