#include<cstdio>
using namespace std;
#define N 2005
#define mod 1000000007
int dp[N][N],su[N][N],n,k;
int pw(int a,int p){if(p<0)return 1;int as=1;while(p){if(p&1)as=1ll*as*a%mod;a=1ll*a*a%mod;p>>=1;}return as;}
int main()
{
	scanf("%d%d",&n,&k);
	dp[n-1][0]=1;
	for(int i=n-1;i>=0;i--)
	for(int j=0;j<n;j++)
	{
		dp[i][j]=(dp[i][j]+su[i+1][j-1])%mod;
		if(i)if(n<=i+j&&j)dp[i-1][j]=(dp[i-1][j]+dp[i][j])%mod;
		su[i][j]=(su[i][j-1]+dp[i][j])%mod;
	}
	int as=0;
	for(int i=0;i<=n;i++)
	as=(as+1ll*dp[n-k][i]*pw(2,n-k-1))%mod;
	printf("%d\n",as);
}
