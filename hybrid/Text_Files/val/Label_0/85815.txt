#include <bits/stdc++.h>
using namespace std;
long long pd[2010][2010][2];
long long spd[2010][2010][2];
int main() {
  int n, m;
  for (long long r = 1; r >= 0; r--)
    for (long long hm = 1; hm <= 2000; hm++) {
      long long acc = 0;
      for (long long w = 2; w <= 2000; w++) {
        if (r)
          acc = (acc + spd[w][hm - 1][1] + pd[w][hm - 1][1]) % 1000000007LL;
        else
          acc = (acc + spd[w][hm - 1][1]) % 1000000007LL;
        pd[w][hm][r] = (1 + acc) % 1000000007LL;
        spd[w + 1][hm][r] = (spd[w][hm][r] + pd[w][hm][r]) % 1000000007LL;
      }
    }
  while (scanf("%d%d", &n, &m) == 2) {
    long long res = 0;
    for (long long w = 2; w <= m; w++)
      for (long long t = 1; t <= n; t++)
        res = (res + (((((pd[w][t][1] - pd[w][t - 1][1] + 1000000007LL) %
                         1000000007LL) *
                        pd[w][n - t + 1][1]) %
                       1000000007LL) *
                      (m - w + 1)) %
                         1000000007LL) %
              1000000007LL;
    printf("%d\n", (int)res);
  }
  return 0;
}
