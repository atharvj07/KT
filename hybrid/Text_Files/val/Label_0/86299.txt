#include <bits/stdc++.h>
using namespace std;
const int inf = ((1 << 30) - 1);
const long long linf = ((1ll << 62) - 1);
const int cmax = 1e+5;
const int cc = 30;
int n;
set<int> used;
map<int, int> cnt;
inline int way(int i) {
  long long x = i;
  int ans = 0;
  for (; x <= n; ans++, x *= i) used.insert(x);
  return ans;
}
int grandi[] = {0, 1, 2, 1, 4, 3, 2, 1, 5,  6, 2, 1, 8,  7,  5, 9,
                8, 7, 3, 4, 7, 4, 2, 1, 10, 9, 3, 6, 11, 12, 14};
map<int, int> st;
int mex(const vector<int> &stack) {
  static bool u[32];
  memset(u, 0, sizeof(u));
  for (int i = 0; i < ((int)(stack).size()); i++)
    if (stack[i] < 32) u[stack[i]] = true;
  for (int i = 0; i < 32; i++)
    if (!u[i]) return i;
  assert(false);
}
int make(int mask, int n) {
  if (st.find(mask) != st.end()) return st[mask];
  if (mask + 1 == 1 << n) return 0;
  vector<int> stack;
  for (int i = 0; i < n; i++)
    if (!(mask & (1 << i))) {
      int newMask = mask;
      for (int j = i + 1; j <= n; j += i + 1) newMask |= 1 << (j - 1);
      stack.push_back(make(newMask, n));
    }
  return st[mask] = mex(stack);
}
void buildGrandi() {
  for (int i = 1; i <= cc; i++) {
    st.clear(), grandi[i] = make(0, i);

  }
  cout << "|";
}
int main() {

  for (int i = 2; i * i <= n; i++)
    if (used.find(i) == used.end()) cnt[i] = way(i);
  ({});
  int ans = (n - ((int)(used).size())) & 1;
  for (__typeof((cnt).begin()) i = (cnt).begin(); i != (cnt).end(); ++i)
    ans ^= grandi[i->second];
  ({});

}
