(n, d, h) = map(int, input().strip().split())
if n == 2:
	if d == 1 and h == 1:
		print(1, 2)
	else:
		print(-1)
elif 2 * h < d or d == 1:
	print(-1)
else:
	for i in range(1, h + 1):
		print(i, i + 1)
	if d > h:
		print(1, h + 2)
		for i in range(h + 2, d + 1):
			print(i, i + 1)
	p = 2 if d == h else 1
	for i in range(d + 2, n + 1):
		print(p, i)
