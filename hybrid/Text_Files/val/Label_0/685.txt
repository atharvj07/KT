n = int(input())
target = 'abacaba'

def comp(s):
	cnt = 0
	for i in range(len(s) - 6):
		match = True
		for j in range(7):
			if s[i + j] != target[j]:
				match = False
				break
		if match:
			cnt += 1
	return cnt

def potential(s):
	potlist = []
	for i in range(len(s) - 6):
		pot = True
		for j in range(7):
			if s[i + j] != target[j] and s[i + j] != '?':
				pot = False
				break
		if pot:
			potlist.append(i)
	return potlist
while n > 0:
	t = int(input())
	tmp = input()
	cp = comp(tmp)
	st = list(tmp)
	isans = False
	for i in range(t):
		s = st[:]
		for j in range(t):
			if s[j] != '?':
				continue
			elif i <= j < i + 7:
				s[j] = target[j - i]
			else:
				s[j] = 'd'
		if comp(s) == 1:
			isans = True
			print('yes')
			print(''.join(s))
			break
	if not isans:
		print('no')
	n -= 1
