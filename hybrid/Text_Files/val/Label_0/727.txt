for _ in range(int(input())):
	n = int(input())
	a = []
	b = []
	c = []
	d = []
	e = []
	for i in range(n):
		arr = list(input())
		l = len(arr)
		cnt = [0, 0, 0, 0, 0]
		for j in arr:
			cnt[ord(j) - 97] += 1
		a.append(cnt[0] * 2 - l)
		b.append(cnt[1] * 2 - l)
		c.append(cnt[2] * 2 - l)
		d.append(cnt[3] * 2 - l)
		e.append(cnt[4] * 2 - l)
	a.sort(reverse=True)
	b.sort(reverse=True)
	c.sort(reverse=True)
	d.sort(reverse=True)
	e.sort(reverse=True)
	a_c = a[0]
	b_c = b[0]
	c_c = c[0]
	d_c = d[0]
	e_c = e[0]
	ans = 0
	if a_c > 0:
		for i in range(1, n):
			a_c += a[i]
			if a_c < 1:
				ans = max(i, ans)
				break
	if b_c > 0:
		for i in range(1, n):
			b_c += b[i]
			if b_c < 1:
				ans = max(i, ans)
				break
	if c_c > 0:
		for i in range(1, n):
			c_c += c[i]
			if c_c < 1:
				ans = max(i, ans)
				break
	if d_c > 0:
		for i in range(1, n):
			d_c += d[i]
			if d_c < 1:
				ans = max(i, ans)
				break
	if e_c > 0:
		for i in range(1, n):
			e_c += e[i]
			if e_c < 1:
				ans = max(i, ans)
				break
	if a_c >= 1 or b_c >= 1 or c_c >= 1 or (d_c >= 1) or (e_c >= 1):
		ans = n
	print(ans)
