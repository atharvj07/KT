for _ in range(int(input())):
	n = int(input())
	l1 = list(map(int, input().split()))
	l2 = list(map(int, input().split()))
	if sum(l1) != sum(l2):
		print(-1)
	else:
		l = []
		r = []
		for i in range(n):
			if l1[i] < l2[i]:
				l.extend([i + 1] * (l2[i] - l1[i]))
			elif l1[i] > l2[i]:
				r.extend([i + 1] * (l1[i] - l2[i]))
		if len(l) != 0:
			print(len(l))
			for i in range(len(l)):
				print(r[i], l[i])
		else:
			print(0)
