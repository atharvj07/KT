(n, m) = map(int, input().split())
mass = [[] for i in range(m)]
for i in range(n):
	x = [k for k in input()]
	for j in range(m):
		mass[j].append(x[j])
maxmass = []
for i in mass:
	maxans = max(i.count('A'), i.count('B'), i.count('C'), i.count('D'), i.count('E'))
	maxmass.append(maxans)
points = [int(i) for i in input().split()]
ans = 0
for i in range(len(maxmass)):
	ans += maxmass[i] * points[i]
print(ans)
