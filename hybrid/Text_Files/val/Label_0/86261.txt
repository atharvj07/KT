#include <algorithm>
#include <cstring>
#include <cstdio>
const int p[] = {2, 3, 5, 7, 11, 13, 17}, lim[] = {8, 5, 3, 2, 2, 2, 2}, len = 7; 
int pos[305], val[305], arr[15], sum[305], n, ans = -1e9; 
void dfs(int cur, int g = 1)
{
	if (cur >= len)
	{
		memset(sum, 0, sizeof(sum)); 
		for (int i = 0; i < n; i++)
		{
			int x = pos[i]; 
			bool f = true; 
			for (int j = 0; j < len; j++)
			{
				for (int k = 0; k < arr[j]; k++)
				{
					if (!(x % p[j]))
						x /= p[j]; 
				}
				if (!(x % p[j]))
					f = false; 
			}
			if (f)
				sum[x] += val[i];
		}
		int res = sum[1]; 
		for (int i = 2; i <= 300; i++)
			res += std::max(sum[i], 0); 
		ans = std::max(ans, res); 
		return;
	}
	for (arr[cur] = 0; arr[cur] <= lim[cur]; arr[cur]++)
	{
		dfs(cur + 1, g); 
		g *= arr[cur]; 
	}
}
int main()
{
	// freopen("CF17-TR-H.in", "r", stdin); 
	scanf("%d", &n); 
	for (int i = 0; i < n; i++)
		scanf("%d%d", pos + i, val + i); 
	dfs(0); 
	printf("%d\n", ans);
	return 0; 
}
