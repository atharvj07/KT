#include <bits/stdc++.h>
int main() {
  long long int n, l, a, prev = 0, count = 0;
  scanf("%lld %lld %lld", &n, &l, &a);
  for (int i = 0; i < n; i++) {
    long long int t, k;
    scanf("%lld %lld", &t, &k);
    if (t - prev >= a) count += (t - prev) / a;
    prev = t + k;
  }
  count += (l - prev) / a;
  printf("%lld", count);
}
