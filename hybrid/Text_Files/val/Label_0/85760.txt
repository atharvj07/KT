bool judge(int points_x[], int points_y[]) {
  int comm_point1_index = -1;
  int comm_point2_index = -1;
  for (int i = 0; i < 3; i++) {
    for (int j = i + 1; j < 3; j++) {
      for (int k = 2 * i; k <= 2 * i + 1; k++) {
        for (int l = 2 * j; l <= 2 * j + 1; l++) {
          if (points_x[k] == points_x[l] && points_y[k] == points_y[l]) {
            comm_point1_index = k;
            comm_point2_index = l;
            goto breakfor1;
          }
        }
      }
    }
  }
breakfor1:
  if (comm_point1_index < 0) {
    return false;
  }
  int index_a = comm_point1_index / 2;
  int index_b = comm_point2_index / 2;
  int index_c = 3 - index_a - index_b;
  int a_x = points_x[index_a * 2 + (comm_point1_index + 1) % 2] -
            points_x[comm_point1_index];
  int a_y = points_y[index_a * 2 + (comm_point1_index + 1) % 2] -
            points_y[comm_point1_index];
  int b_x = points_x[index_b * 2 + (comm_point2_index + 1) % 2] -
            points_x[comm_point2_index];
  int b_y = points_y[index_b * 2 + (comm_point2_index + 1) % 2] -
            points_y[comm_point2_index];
  if ((long long)a_x * b_x < -(long long)a_y * b_y ||
      (long long)a_x * b_y == (long long)a_y * b_x)
    return false;
  if (judge1(points_x, points_y, 2 * index_a, 2 * index_a + 1, 2 * index_c) &&
          judge1(points_x, points_y, 2 * index_b, 2 * index_b + 1,
                 2 * index_c + 1) ||
      judge1(points_x, points_y, 2 * index_a, 2 * index_a + 1,
             2 * index_c + 1) &&
          judge1(points_x, points_y, 2 * index_b, 2 * index_b + 1,
                 2 * index_c)) {
    return true;
  }
  return false;
}
