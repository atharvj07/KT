import math
import sys
from collections import defaultdict, Counter
from functools import lru_cache
t = int(input())
for _ in range(t):
	n = int(input())
	s1 = input()
	s2 = input()
	count = 0
	ind = []
	for i in range(n):
		if s1[i] != s2[i]:
			count += 1
			ind.append(i)
	if count == 2:
		if s1[ind[0]] == s1[ind[1]] and s2[ind[1]] == s2[ind[0]]:
			print('Yes')
		else:
			print('No')
	else:
		print('No')
