import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class E {

    private static Scanner in;

    private static int n, k, numFactors;
    private static int[][] a;
    private static int[] factors, factorization;
    private static long answer;

    private static void solve() {
        n = in.nextInt();
        k = in.nextInt();
        if (k == 1) {
            System.out.println((long) n * (n + 1) / 2);
            System.exit(0);
        }
        factor();
        a = new int[n][numFactors];
        for (int i = 0; i < n; i++) {
            int num = in.nextInt();
            for (int j = 0; j < numFactors; j++) {
                while (num % factors[j] == 0) {
                    a[i][j]++;
                    num /= factors[j];
                }
            }
        }
        int lo = 0;
        int hi = -1;
        int[] fac = new int[numFactors];
        while (true) {
            if (lo <= hi && works(fac)) {
                answer += (long) (n - hi);
                for (int i = 0; i < numFactors; i++)
                    fac[i] -= a[lo][i];
                lo++;
            } else {
                hi++;
                if (hi >= n)
                    break;
                for (int i = 0; i < numFactors; i++)
                    fac[i] += a[hi][i];
            }
        }
        System.out.println(answer);
    }

    private static boolean works(int[] fac) {
        for (int i = 0; i < numFactors; i++)
            if (fac[i] < factorization[i])
                return false;
        return true;
    }

    private static void factor() {
        Set<Integer> f = new HashSet<Integer>();
        int k2 = k;
        loop:
        while (true) {
            for (int i = 2; i <= Math.sqrt(k2); i++) {
                if (k2 % i == 0) {
                    f.add(i);
                    k2 /= i;
                    continue loop;
                }
            }
            break;
        }
        f.add(k2);
        k2 = k;
        numFactors = f.size();
        factors = new int[f.size()];
        factorization = new int[f.size()];
        int i = 0;
        for (int factor : f)
            factors[i++] = factor;
        for (i = 0; i < numFactors; i++) {
            while (k2 % factors[i] == 0) {
                factorization[i]++;
                k2 /= factors[i];
            }
        }
    }

    public static void main(String[] args) {
        in = new Scanner(System.in);
        solve();
        System.exit(0);
    }

    private static int[] nextArray(int length) {
        int[] a = new int[length];
        for (int i = 0; i < length; i++)
            a[i] = in.nextInt();
        return a;
    }
}
