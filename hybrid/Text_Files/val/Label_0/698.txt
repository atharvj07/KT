num = int(input())
(n, m) = (0, 0)
for i in range(5, int(num ** 0.5) + 5):
	if num % i == 0 and num // i >= 5:
		(n, m) = (i, num // i)
		break
if n == 0 and m == 0:
	print(-1)
else:
	s = ''
	arr = ['a', 'e', 'i', 'o', 'u']
	for i in range(n):
		ind = i
		ind %= 5
		for j in range(m):
			s += arr[ind]
			ind = (ind + 1) % 5
	print(s)
