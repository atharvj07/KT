#include <bits/stdc++.h>
using namespace std;
const int maxN = 1e5 + 13, sq = 400;
int n, a[maxN], ans[maxN], par[maxN], q, cur1, cur2, arr[maxN], l[maxN],
    r[maxN], dis[maxN];
map<int, int> mp, mark;
deque<int> deq[maxN];
bool cmp(int i, int j) {
  if (l[i] / sq != l[j] / sq) return l[i] < l[j];
  if (r[i] != r[j]) return r[i] < r[j];
  return i < j;
}
int find_par(int x) {
  if (x == par[x]) return x;
  return par[x] = find_par(par[x]);
}
bool check(int x) {
  if (deq[x].size() == 0) return 0;
  if (deq[x].size() <= 2) return 1;
  int s1 = deq[x].back(), e = deq[x].front();
  deq[x].pop_back();
  int s2 = deq[x].back();
  bool ret = 0;
  if (find_par(s2) == find_par(e)) ret = 1;
  deq[x].push_back(s1);
  return ret;
}
void add(int x, bool fl) {
  int val = a[x];
  if (mark[val]++ == 0) cur1++;
  if (check(val)) cur2--;
  if (fl)
    deq[val].push_front(x);
  else
    deq[val].push_back(x);
  if (check(val)) cur2++;
}
void remove(int x, bool fl) {
  int val = a[x];
  if (--mark[val] == 0) cur1--;
  if (check(val)) cur2--;
  if (fl)
    deq[val].pop_front();
  else
    deq[val].pop_back();
  if (check(val)) cur2++;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> n;
  iota(par, par + n, 0);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    if (mp.find(a[i]) != mp.end()) {
      int u = mp[a[i]];
      if (i - u == dis[u] || dis[u] == 0) par[i] = u;
      dis[i] = i - u;
    }
    mp[a[i]] = i;
  }
  cin >> q;
  for (int i = 0; i < q; i++) {
    cin >> l[i] >> r[i];
    l[i]--, r[i]--;
  }
  iota(arr, arr + q, 0);
  sort(arr, arr + q, cmp);
  int lf = 0, rg = 0;
  add(lf, 0);
  for (int i = 0; i < q; i++) {
    int x = arr[i];
    while (r[x] < rg) {
      remove(rg, 1);
      rg--;
    }
    while (r[x] > rg) {
      rg++;
      add(rg, 1);
    }
    while (l[x] > lf) {
      remove(lf, 0);
      lf++;
    }
    while (l[x] < lf) {
      lf--;
      add(lf, 0);
    }
    ans[x] = cur1 + (cur2 ? 0 : 1);
  }
  for (int i = 0; i < q; i++) cout << ans[i] << '\n';
}
