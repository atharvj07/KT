import sys

def solve():
	t = int(sys.stdin.readline())
	for ti in range(t):
		(x, y, p, q) = map(int, sys.stdin.readline().split())
		if p == 0:
			if x == 0:
				print(0)
			else:
				print(-1)
			continue
		if p == q:
			if x == y:
				print(0)
			else:
				print(-1)
			continue
		if p * y == x * q:
			print(0)
		elif p * y > x * q:
			z = (q * (y - x) + q - p - 1) // (q - p)
			ans = (z + q - 1) // q * q - y
			print(ans)
		else:
			z = (q * x + p - 1) // p
			ans = (z + q - 1) // q * q - y
			print(ans)
solve()
