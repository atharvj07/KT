input = stdin.readline
(power, mod) = ({}, 1000000007)

def getpower(k):
	if k == 0:
		return 1
	if k == 1:
		return 2
	if k in power:
		return power[k]
	power[k] = getpower(k // 2) * getpower(k - k // 2) % mod
	return power[k]
for _ in range(int(input())):
