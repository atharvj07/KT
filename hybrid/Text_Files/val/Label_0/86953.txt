public class c128C {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    private String getString() throws IOException {return br.readLine();}
    private String[] getTok() throws IOException {return br.readLine().split(" ");}
    private int getInt() throws IOException {return Integer.valueOf(br.readLine());}
    private int[] getInt(int n) throws IOException {
        int[] data= new int[n]; tok= br.readLine().split(" ");
        for (int i=0; i<n; i++) data[i]= Integer.valueOf(tok[i]);
        return data;
    }
    /*******************************************************************************/
    String[] tok;
    String s;

    int n;
    int d, a, b;
    Client[] memory;

    public void solve() throws IOException {
        int i=0, j=0, k;
        long x, y, mem;

        tok= getTok();
        n= Integer.valueOf(tok[0]);
        d= Integer.valueOf(tok[1]);
        tok= getTok();
        a= Integer.valueOf(tok[0]);
        b= Integer.valueOf(tok[1]);
        memory= new Client[n];
        for (i=0; i<n; i++) {
            tok= getTok();
            x= Long.valueOf(tok[0]);
            y= Long.valueOf(tok[1]);
            memory[i]= new Client(i, x*a + y*b);
        }
        Arrays.sort(memory);
        int noOfClients= 0;
        StringBuffer sb= new StringBuffer();
        for (i=0; i<n; i++) {
            if (d-memory[i].value>=0) {
                ++noOfClients;
                d-= memory[i].value;
                sb= sb.append(memory[i].id).append(" ");
            } else {
                break;
            }
        }
        System.out.printf("%d\n", noOfClients);
        if (sb.length()>0)
            System.out.printf("%s\n", sb.substring(0, sb.length()-1));
    }

    private class Client implements Comparable<Client> {
        int id;
        long value;

        public Client(int i, long v) { id= i+1; value= v;}

        public int compareTo(Client c) {
            return c.value- value>0?-1:1;
        }
    }

    public static void main(String[] args) throws IOException {
        new c128C().solve();
    }
}
