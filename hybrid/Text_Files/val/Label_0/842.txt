def customsort(t):
	return t[1]
n = int(input())
a = list(map(int, input().split()))
a_new = [(index, a[index]) for index in range(len(a))]
a_new.sort(key=lambda x: x[1])
b = []
cont = True
i = 0
k = 1
while i < n and cont:
	for j in range(n - a_new[i][1]):
		b.append(k)
	k += 1
	for j in range(len(a_new) - a_new[i][1]):
		if i + j >= n or a_new[i + j][1] != a_new[i][1]:
			print('Impossible')
			cont = False
			break
	i += n - a_new[i][1]
if cont:
	print('Possible')
	b_original = [0] * n
	for i in range(n):
		b_original[a_new[i][0]] = b[i]
	print(*b_original)
