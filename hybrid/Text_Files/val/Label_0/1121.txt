import sys
import bisect
input = sys.stdin.readline

def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	s = input()
	return list(s[:len(s) - 1])

def invr():
	return map(int, input().split())

def compbreaks(ass, breaks, pairs, offset):
	if ass[0] == ass[-1]:
		return
	m = (ass[0] + ass[-1]) // 2
	i = bisect.bisect(ass, m)
	bisect.insort(breaks, i + offset)
	pairs += [(offset, offset + i), (offset + i, offset + len(ass))]
	if i < len(ass):
		compbreaks(ass[:i], breaks, pairs, offset)
		compbreaks(ass[i:], breaks, pairs, offset + i)

def go(ass, pairs, prefixes, s):
	for (x, y) in pairs:
		a = prefixes[y] - prefixes[x]
		if a == s:
			return True
	return False

def solve(ass, qs, n, q):
	ass = sorted(ass)
	breaks = [0]
	pairs = [(0, n)]
	compbreaks(ass, breaks, pairs, 0)
	breaks.append(n)
	prefixes = [0]
	a = 0
	for i in range(n):
		a += ass[i]
		prefixes.append(a)
	goods = set()
	for (x, y) in pairs:
		a = prefixes[y] - prefixes[x]
		goods.add(a)
	for s in qs:
		if s in goods:
			print('Yes')
		else:
			print('No')
for _ in range(inp()):
	(n, q) = invr()
	ass = inlt()
	qs = [inp() for i in range(q)]
	solve(ass, qs, n, q)
