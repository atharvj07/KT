import sys
input = sys.stdin.readline

def inp():
	return int(input())

def st():
	return input().rstrip('\n')

def lis():
	return list(map(int, input().split()))

def ma():
	return map(int, input().split())
t = inp()
while t:
	t -= 1
	(n, m) = ma()
	p = 10 ** 9 + 7
	x = 0
	for i in range(m):
		(l, r, x1) = ma()
		x = x | x1
	x = x * 2 ** (n - 1)
	x %= p
	print(x)
