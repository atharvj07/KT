      public static void main(String hi[]) throws Exception
      {
         BufferedReader infile = new BufferedReader(new InputStreamReader(System.in));
         StringTokenizer st = new StringTokenizer(infile.readLine());
         int N = Integer.parseInt(st.nextToken());
         int Q = Integer.parseInt(st.nextToken());
         int[] arr = new int[N];
         st = new StringTokenizer(infile.readLine());
         for(int i=0; i < N; i++)
            arr[i] = Integer.parseInt(st.nextToken());
         int max = arr[0];
         for(int x: arr)
            max = Math.max(max, x);
         if(max == 0)
         {
            StringBuilder sb = new StringBuilder("YES\n");
            for(int i=0; i < N; i++)
               sb.append(Q+" ");
            System.out.println(sb);
         }
         else if(max > Q)
            System.out.println("NO");
         else
         {
            ArrayDeque<Integer> q = new ArrayDeque<Integer>();
            boolean did = false;
            for(int i=0; i < N; i++)
            {
               if(arr[i] == 0 && !did && max < Q)
               {
                  arr[i] = Q;
                  did = true;
               }
               if(arr[i] > 0)
                  q.add(i);
            }
            if(!did && max < Q)
            {
               System.out.println("NO");
               return;
            }
            while(q.size() > 0)
            {
               int curr = q.poll();
               if(curr > 0 && arr[curr-1] == 0)
               {
                  arr[curr-1] = arr[curr];
                  q.add(curr-1);
               }
               if(curr < N-1 && arr[curr+1] == 0)
               {
                  arr[curr+1] = arr[curr];
                  q.add(curr+1);
               }
            }
            if(!moves(arr, Q))
               System.out.println("NO");
            else
            {
               StringBuilder sb = new StringBuilder("YES\n");
               for(int i=0; i < N; i++)
                  sb.append(arr[i]+" ");
               System.out.println(sb);
            }
         }
      }
      public static boolean moves(int[] arr, int Q)
      {
         int N = arr.length;
         ArrayDeque<Integer>[] locs = new ArrayDeque[Q+1];
         for(int i=1; i <= Q; i++)
            locs[i] = new ArrayDeque<Integer>();
         for(int i=0; i < N; i++)
            locs[arr[i]].add(i);
         FenwickTree bit = new FenwickTree(N+3);
         for(int v=1; v <= Q; v++)
         {
            int min = -1;
            int max = -1;
            for(int dex: locs[v])
            {
               if(min == -1)
                  min = max = dex;
               max = dex;
            }
            if(min != max && bit.find(min+1, max+1) > 0)
               return false;
            for(int x: locs[v])
               bit.add(x+1, 1);
         }
         return true;
      }
   }
   class FenwickTree
   {
      //1 indexed
      public int[] tree;
      public int size;
      
      public FenwickTree(int size)
      {
         this.size = size;
         tree = new int[size+5];
      }
      public void add(int i, int v)
      {
         while(i <= size)
         {
            tree[i] += v;
            i += i&-i;
         }
      }
      public int find(int i)
      {
         int res = 0;
         while(i >= 1)
         {
            res += tree[i];
            i -= i&-i;
         }
         return res;
      }
      public int find(int l, int r)
      {
         return find(r)-find(l-1);
      }
   }
