import java.io.BufferedReader;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.OutputStream;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.StringTokenizer;

public class Main {
	public static void main(String[] args) {
		InputStream inputStream = System.in;
		OutputStream outputStream = System.out;
		Scanner in = new Main().new Scanner(inputStream);
		PrintWriter out = new PrintWriter(outputStream);
		Main solver = new Main();
		solver.solve(1, in, out);
		out.close();
	}

	public void solve(int testNumber, Scanner in, PrintWriter out) {
		int n = in.nextInt();
		int m = in.nextInt();
		int xi = in.nextInt();
		
		ArrayList<String> shifts = new ArrayList<String>();
		HashSet<Character> inKeeb = new HashSet<Character>();
		HashSet<Character> inRange = new HashSet<Character>();
		char[][] keeb = new char[n][m];
		
		for(int i = 0; i < n; i++) {
			char[] cs = in.readLine().toCharArray();
			for(int j = 0; j < m; j++) {
				char key = cs[j];
				if(key == 'S') {
					shifts.add(i + "," + j);
				} else {
					keeb[i][j] = key;
					inKeeb.add(key);
				}
			}
		}
		
		for(String shift : shifts) {
			String[] coords = shift.split(",");
			int x = Integer.parseInt(coords[0]);
			int y = Integer.parseInt(coords[1]);
			int range = xi;
			
			for(int cx = Math.max(0, x - range); cx <= Math.min(x + range, n-1); cx++) {
				for(int cy = Math.max(0, y - range); cy <= Math.min(y + range, m-1); cy++) {
					double dist = Math.sqrt((cx - x) * (cx - x) + (cy - y) * (cy - y));
					if(dist <= xi) {
						inRange.add(keeb[cx][cy]);
					}
				}
			}
		}
		
		int lift = 0;
		int count = in.nextInt();
		char[] queries = in.readLine().toCharArray();
		
		boolean impossible = false;
		for(int i = 0; i < count; i++) {
			char c = queries[i];
			
			if(Character.isUpperCase(c)) {
				c = (char) (c - 'A' + 'a');
				if(!inKeeb.contains(c) || shifts.size() == 0) {
					impossible = true;
					break;
				}
				if(!inRange.contains(c)) {
					lift++;
				}
			} else if(!inKeeb.contains(c)) {
				impossible = true;
				break;
			}
		}
		
		if(impossible) out.println("-1");
		else out.println(lift);
	}
	
	class Scanner {
		BufferedReader br;
		StringTokenizer in;

		public Scanner(InputStream inputStream) {
			br = new BufferedReader(new InputStreamReader(inputStream));
		}

		public String next() {
			return hasMoreTokens() ? in.nextToken() : null;
		}

		public String readLine() {
			try {
				return br.readLine();
			} catch (Exception e) {
				e.printStackTrace();
				return null;
			}
		}

		boolean hasMoreTokens() {
			while (in == null || !in.hasMoreTokens()) {
				String s = readLine();
				if (s == null)
					return false;
				in = new StringTokenizer(s);
			}
			return true;
		}

		public String nextString() {
			return next();
		}

		public int nextInt() {
			return Integer.parseInt(nextString());
		}

		public int[] nextIntArray(int n) {
			int[] a = new int[n];
			for (int i = 0; i < n; i++) {
				a[i] = nextInt();
			}
			return a;
		}
	}
}
