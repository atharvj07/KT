#include <bits/stdc++.h>
using namespace std;
const long double E = 2.71828182845904523536;
const long double pi = acos(-1);
const double eps = 1e-9;
const long long mod = 1e9 + 7;
const long long inf = 1e9;
const int N = 5050;
vector<pair<int, pair<int, int>>> ans;
vector<int> num;
long long mem[N][N];
int n, k, v;
int a[N];
long long dp(int i, int val) {
  if (i == n + 1) {
    if (val == v % k) return 0;
    return -inf;
  }
  long long &ret = mem[i][val];
  if (ret != -1) return ret;
  return ret = max(1 + dp(i + 1, (val + a[i]) % k), dp(i + 1, val));
}
void getpath(int i, int val) {
  if (i == n + 1) return;
  long long val1 = 1 + dp(i + 1, (val + a[i]) % k), val2 = dp(i + 1, val);
  if (val1 >= val2) {
    num.push_back(i);
    getpath(i + 1, (val + a[i]) % k);
  } else
    getpath(i + 1, val);
}
int main() {
  scanf("%d%d%d", &n, &k, &v);
  for (int i = 1; i <= n; i++) scanf("%d", &a[i]);
  if (v % k == 0) {
    int sum = a[1];
    for (int i = 2; i <= n; i++) {
      sum += a[i];
      ans.push_back({inf, {i, 1}});
    }
    if (sum / k < v / k) {
      printf("NO");
      return 0;
    }
    if (v) ans.push_back({v / k, {1, 2}});
    printf("YES\n");
    for (auto x : ans)
      printf("%d %d %d\n", x.first, x.second.first, x.second.second);
    return 0;
  }
  memset(mem, -1, sizeof mem);
  if (dp(1, 0) < 0) {
    printf("NO");
    return 0;
  }
  getpath(1, 0);
  long long res = a[num[0]];
  int id = num[0];
  for (int i = 1; i < num.size(); i++) {
    res += a[num[i]];
    a[num[i]] = 0;
    ans.push_back({inf, {num[i], id}});
  }
  if (res >= v) {
    int cnt = (res - v) / k;
    int j = id - 1;
    if (j <= 0) j += 2;
    if (cnt) ans.push_back({cnt, {id, j}});
  } else {
    int j = id - 1;
    if (j <= 0) j += 2;
    long long sum = a[j];
    for (int i = 1; i <= n; i++) {
      if (i == id || i == j || !a[i]) continue;
      sum += a[i];
      ans.push_back({inf, {i, j}});
    }
    int cnt = (v - res) / k;
    int cnt2 = sum / k;
    if (cnt > cnt2) {
      printf("NO");
      return 0;
    }
    ans.push_back({cnt, {j, id}});
  }
  printf("YES\n");
  for (auto x : ans)
    printf("%d %d %d\n", x.first, x.second.first, x.second.second);
  return 0;
}
