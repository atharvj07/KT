#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

typedef long long ll;

int main() {
    ll n;
    cin >> n;
    vector<ll> a(n);
    for(int i=0;i<n;++i) cin >> a[i];

    ll pos[n][16], neg[n][16];
    for(int x=0;x<=15;++x) {
        pos[n-1][x] = x;
        neg[0][x] = x;
    }

    for(int i=n-2;i>=0;--i) {
        ll x4 = 1;
        for(int x=0;x<=15;++x) {
            int j = 0;
            ll j4 = 1;
            while(a[i] * x4 > a[i+1] * j4) {
                j4 *= 4;
                ++j;
            }
            if(j <= 15) {
                pos[i][x] = pos[i+1][j] + x;
            } else {
                pos[i][x] = pos[i+1][15] + (n-i-1) * (j-15) + x;
            }
            x4 *= 4;
        }
    }

    for(int i=1;i<n;++i) {
        ll x4 = 1;
        for(int x=0;x<=15;++x) {
            int j = 0;
            ll j4 = 1;
            while(a[i-1] * j4 < a[i] * x4) {
                j4 *= 4;
                ++j;
            }
            if(j <= 15) {
                neg[i][x] = neg[i-1][j] + x;
            } else {
                neg[i][x] = neg[i-1][15] + i * (j-15) + x;
            }
            x4 *= 4;
        }
    }

    ll ans = pos[0][0] * 2;
    for(int i=1;i<n;++i) {
        ans = min(ans, i + neg[i-1][0] * 2 + pos[i][0] * 2);
    }
    cout << ans << "\n";
}
