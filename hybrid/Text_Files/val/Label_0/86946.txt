

import java.util.* ;
import java.io.*;



public class A 
{
    public static void main(String[] args) throws IOException 
    {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
        int n = Integer.parseInt(br.readLine());
        
        int space = n*2 ;
        
        for (int i = 0; i < n ; i++) {
            
            out.printf("%"+space+"s","");
            space-=2 ;
            for (int j = 0; j < i ; j++) {
                out.print(j+" ");
            }
            if(i== 0)
                out.print(i);
            else
                out.print(i+" ");
            for (int j = i-1 ; j >= 0 ; j--) {
                if(j==0)
                    out.print(j);
                else
                    out.print(j+" ");
            }
            out.println();
        }
        
        for (int i = 0; i <= n ; i++) {
            out.print(i+" ");
        }
        for (int i = n-1 ; i >= 0 ; i--) {
            if(i==0)
                out.print(i);
            else
                out.print(i+" ");
        }
        out.println();
        space = 2 ;
        for (int i = n-1 ; i >= 0 ; i--) {
            out.printf("%"+space+"s","");
            space += 2 ;
            for (int j = 0; j < i ; j++) {
                out.print(j+" ");
            }
            if(i==0)
                out.print(i);
            else
                out.print(i+" ");
            for (int j = i-1 ; j >= 0 ; j-- ) {
                if(j==0)
                    out.print(j);
                else
                    out.print(j+" ");
            }
            out.println();
        }
        
        br.close();
        out.close();
    }
        
}
