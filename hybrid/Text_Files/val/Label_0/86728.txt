import java.io.*;
import java.util.Locale;
import java.util.StringTokenizer;

/**
 * Created by DNs on 15.05.2016.
 */
public class Main implements Runnable {
  static BufferedReader in;
  static PrintWriter out;
  static StringTokenizer tok = new StringTokenizer("");

  public static void main(String[] args) {
    new Thread(new Main()).start();
  }

  @Override
  public void run() {
    Locale.setDefault(Locale.ENGLISH);
    try {
      init();
    } catch (FileNotFoundException e) {
      e.printStackTrace();
    }
    try {
      solve();
    } catch (IOException e) {
      e.printStackTrace();
    }
    out.close();
  }


  private String readString() {
    while (!tok.hasMoreTokens()) {
      try {
        tok = new StringTokenizer(in.readLine());
      } catch (IOException e) {
        e.printStackTrace();
      }
    }
    return tok.nextToken();
  }

  private Integer readInt() {
    return Integer.parseInt(readString());
  }

  private Long readLong() {
    return Long.parseLong(readString());
  }

  private Double readDouble() {
    return Double.parseDouble(readString());
  }

  private void init() throws FileNotFoundException {
    in = new BufferedReader(new InputStreamReader(System.in));
    out = new PrintWriter(System.out);
    /*in = new BufferedReader(new FileReader("stupid_rmq.in"));
    out = new PrintWriter("stupid_rmq.out");*/
  }

  private void solve() throws IOException {
    int n = readInt();
    int x = readInt();
    double[][] p = new double[1][2 * x + 2];
    for (int i = 0; i < x + 1; i++) {
      p[0][i] = readDouble();
    }
    x *= 2;
    double[][] matrix = new double[x + 2][x + 2];
    for (int i = 0; i < x + 2; i++) {
      for (int j = 0; j < x + 2; j++) {
        for (int k = 0; k < x + 2; k++) {
          if ((i ^ k) == j) {
            matrix[i][j] = p[0][k];
            break;
          }
        }
      }
    }
    double[][] kek = multiply(p, binPow(matrix, n - 1));
    out.print(1.0 - kek[0][0]);
  }

  private double[][] binPow(double[][] matrix, int power) {
    if (power == 0) {
      double[][] edin = new double[matrix.length][matrix.length];
      for (int i = 0; i < matrix.length; i++) {
        edin[i][i] = 1;
      }
      return edin;
    }
    if (power == 1) return matrix.clone();
    if (power % 2 == 0) {
      double[][] temp = binPow(matrix.clone(), power / 2);
      return multiply(temp.clone(), temp.clone());
    } else {
      return multiply(matrix.clone(), binPow(matrix.clone(), power - 1));
    }

  }

  private double[][] multiply(double[][] first, double[][] second) {
    int n = first.length;
    int k = first[0].length;
    int m = second[0].length;
    double[][] result = new double[n][m];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < k; j++) {
        for (int l = 0; l < m; l++) {
          result[i][l] += first[i][j] * second[j][l];
        }
      }
    }
    return result;
  }


}
