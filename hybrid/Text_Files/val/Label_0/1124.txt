t = int(input())
while t > 0:
	t -= 1
	(n, q) = input().split()
	(n, q) = (int(n), int(q))
	ar = [int(op) for op in input().split()]
	sr = []
	for i in range(q):
		sr.append(int(input()))
	dt = {}

	def fun(ar):
		if len(ar) == 0:
			return 0
		else:
			avg = int((max(ar) + min(ar)) / 2)
			la = []
			ra = []
			for i in range(len(ar)):
				if ar[i] > avg:
					ra.append(ar[i])
				else:
					la.append(ar[i])
			ans = 0
			if not (len(la) == 0 or len(ra) == 0):
				ans = fun(la) + fun(ra)
				dt[ans] = 1
			else:
				ans = sum(la) + sum(ra)
			dt[ans] = 1
			return ans
	dt[fun(ar)] = 1
	for i in range(q):
		if dt.get(sr[i], -1) == 1:
			print('Yes')
		else:
			print('No')
