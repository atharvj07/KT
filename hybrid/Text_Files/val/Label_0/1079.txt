import sys
ii = lambda : sys.stdin.readline().strip()
idata = lambda : [int(x) for x in ii().split()]

def solve():
	(n, m) = idata()
	posle_progulki_po_strokam = []
	matrix = []
	summ = 0
	flag = True
	for i in range(n):
		s = idata()
		matrix += [s[:]]
		d = 0
		q = 0
		for j in range(m):
			if s[j] and d:
				if s[j] - q <= d:
					flag = False
				for w in range(d):
					s[j - w - 1] = s[j - w] - 1
				d = 0
				q = s[j]
			elif s[j]:
				if q >= s[j]:
					flag = False
				q = s[j]
			else:
				d += 1
		posle_progulki_po_strokam += [s[:]]
	for i in range(m):
		s = []
		for k in range(n):
			s += [posle_progulki_po_strokam[k][i]]
		s1 = []
		for k in range(n):
			s1 += [matrix[k][i]]
		d = 0
		q = 0
		for j in range(n):
			if s1[j] and d:
				if s1[j] - q <= d:
					flag = False
				for w in range(d):
					s[j - w - 1] = min(s[j - w] - 1, s[j - w - 1])
				if j - d != 0 and s[j - d] <= s[j - d - 1]:
					flag = False
				d = 0
				q = s1[j]
			elif s1[j]:
				if q >= s1[j]:
					flag = False
				q = s1[j]
			else:
				d += 1
		summ += sum(s)
	if not flag:
		print(-1)
		return
	print(summ)
	return
solve()
