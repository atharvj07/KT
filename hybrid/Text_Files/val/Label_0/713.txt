l = input()
i = 0
f1 = False
f2 = False
f3 = False
f4 = False
while i < len(l) - 1:
	if (l[i] == 'A' and l[i + 1] == 'B') and (not f1):
		f1 = True
		i += 2
	elif (l[i] == 'B' and l[i + 1] == 'A') and (not f2):
		f2 = True
		i += 2
	else:
		i += 1
i = len(l) - 1
if not (f1 and f2):
	while i > 1:
		if (l[i] == 'B' and l[i - 1] == 'A') and (not f3):
			f3 = True
			i -= 2
		elif (l[i] == 'A' and l[i - 1] == 'B') and (not f4):
			f4 = True
			i -= 2
		else:
			i -= 1
if f1 and f2 or (f3 and f4):
	print('YES')
else:
	print('NO')
