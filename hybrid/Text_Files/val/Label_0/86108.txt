#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  int c = 0;
  int m;
  cin >> n >> m;
  while (n % m != 0) {
    n++;
    c++;
  }
  for (int i = 0; i < m; i++) {
    if (c) {
      cout << (n / m) - 1 << " ";
      c--;
      continue;
    }
    if (i - 1 == m) {
      cout << n / m;
      break;
    }
    cout << n / m << " ";
  }
  return 0;
}
