def between(begin, end, pos):
	return begin <= pos <= end or begin >= pos >= end
t = int(input())
for _ in range(t):
	n = int(input())
	(pre_t, pre_x, pre_pos, pos, dest, valid_num) = (0, 0, 0, 0, 0, 0)
	for i in range(n):
		(cur_t, cur_x) = map(int, input().split())
		pre_pos = pos
		if cur_t - pre_t >= abs(dest - pos):
			pos = dest
			dest = cur_x
		else:
			direction = 1 if dest > pos else -1
			pos += (cur_t - pre_t) * direction
		if i > 0 and between(pos, pre_pos, pre_x):
			valid_num += 1
		(pre_t, pre_x) = (cur_t, cur_x)
	if between(dest, pos, pre_x):
		valid_num += 1
	print(valid_num)
