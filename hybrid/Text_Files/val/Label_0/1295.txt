import sys
input = sys.stdin.readline
(n, q) = map(int, input().split())
s = input()[:-1]
t = s.count('abc')
w = list(s)
for _ in range(q):
	(i, x) = input()[:-1].split()
	i = int(i)
	if w[i - 1] == x:
		print(t)
	else:
		d = ''.join(w[max(0, i - 3):min(i + 2, n)])
		c1 = d.count('abc')
		w[i - 1] = x
		d = ''.join(w[max(0, i - 3):min(i + 2, n)])
		c2 = d.count('abc')
		t -= c1 - c2
		print(t)
