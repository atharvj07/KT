def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	s = input()
	return list(s[:len(s)])

def invr():
	return map(int, input().split())
n = inp()
z = []
for i in range(20):
	z.append(2 ** i - 1)
l = inlt()
l.sort()
b = [0] * n
for bit in range(20):
	cnt = 0
	for j in range(n):
		cnt += l[j] >> bit & 1
	for i in range(cnt):
		b[i] |= 1 << bit
res = 0
for x in b:
	res += x ** 2
print(res)
