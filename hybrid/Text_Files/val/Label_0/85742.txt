#include<bits/stdc++.h>
using namespace std;
#define M 200005
int a[M],b[M];
int main(){
	int n,ans=0;
	scanf("%d",&n);
	for(int i=1;i<=n;++i)scanf("%d",&a[i]);
	for(int i=n;i>=1;--i){
		int v=0;
		for(int j=i+i;j<=n;j+=i)v^=b[j];
		b[i]=v^a[i];
		if(b[i])++ans;
	}
	printf("%d\n",ans);
	for(int i=1;i<=n;++i)if(b[i])printf("%d ",i);
}
