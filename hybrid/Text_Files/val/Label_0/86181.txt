#include <bits/stdc++.h>
using namespace std;
struct node {
  int value;
  int r;
} a[200001];
int l[200001];
int n, r, c;
bool cmp(int i1, int i2) {
  if (a[i1].value == a[i2].value) return i1 > i2;
  return a[i1].value < a[i2].value;
}
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i].value;
    l[i] = i;
  }
  sort(l + 1, l + n + 1, cmp);
  for (int i = 1; i <= n; i++) {
    if (a[l[i]].value == a[l[i - 1]].value) {
      a[l[i]].r = a[l[i - 1]].r;
    } else {
      a[l[i]].r = l[i];
    }
  }
  for (int i = 1; i <= n; i++) {
    if (i > r) {
      c++;
    }
    r = max(r, a[i].r);
  }
  int res = 1;
  for (int i = 1; i <= c - 1; i++) {
    res = res * 2 % 998244353;
  }
  cout << res;
  return 0;
}
