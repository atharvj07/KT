#include <bits/stdc++.h>
using namespace std;
double n, r, v, x, dx, pi = 3.1415926525;
bool findn(int nn) { return x >= nn * 2 * pi * r; }
bool findt(double tt) { return dx > v * tt + 2 * r * sin(v * tt / 2 / r); }
int main() {
  cin >> n >> r >> v;
  for (int i = 0; i < n; i++) {
    int a, b;
    cin >> a >> b;
    x = b - a;
    int hi = 2000000000, lo = 0, med;
    while (hi > lo) {
      med = (hi + lo + 1) / 2;
      if (findn(med))
        lo = med;
      else
        hi = med - 1;
    }
    int nn = lo;
    dx = x - nn * 2 * pi * r;
    double hii = 2 * pi * r / v, loo = 0, meed;
    while (hii - loo > 0.00000001 * max(1.0, loo)) {
      meed = (hii + loo) / 2;
      if (findt(meed))
        loo = meed;
      else
        hii = meed;
    }
    printf("%.7f\n", loo + nn * 2 * pi * r / v);
  }
}
