t = int(input())

def aux(a):
	a = sorted(a, key=lambda x: x[0] - x[1], reverse=True)
	if a[0][0] <= a[0][1]:
		return 0
	res = 1
	cur_sum = a[0][0]
	rem_sum = a[0][1]
	for i in range(1, len(a)):
		if cur_sum + a[i][0] > rem_sum + a[i][1]:
			cur_sum += a[i][0]
			rem_sum += a[i][1]
			res += 1
	return res
for _ in range(t):
	n = int(input())
	words = []
	for i in range(n):
		words.append(input())
	hash_words = []
	for word in words:
		h_word = [0, 0, 0, 0, 0]
		for j in range(len(word)):
			h_word[ord(word[j]) - ord('a')] += 1
		hash_words.append(h_word)
	res = 0
	for i in range(5):
		a = []
		for j in range(n):
			a.append((hash_words[j][i], sum(hash_words[j]) - hash_words[j][i]))
		cur_res = aux(a)
		res = max(cur_res, res)
	print(res)
