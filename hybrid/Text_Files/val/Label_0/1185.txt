cin = input().split()
n = int(cin[0])
m = int(cin[1])
data = list(map(int, input().split()))
for x in range(1, max(len(data), len(bin(m)) - 2) + 10):
	if x < len(data):
		if data[x] > data[x - 1] * 2:
			data[x] = data[x - 1] * 2
	else:
		data.append(data[x - 1] * 2)
for x in reversed(range(0, len(data) - 1)):
	data[x] = min(data[x + 1], data[x])
m = bin(m)
m = list(reversed(m))
ans = 0
for x in range(max(len(data), len(m) - 2)):
	if x >= len(m) - 2:
		ans = min(ans, data[x])
	elif m[x] == '1':
		ans += int(data[x])
	else:
		ans = min(ans, data[x])
print(ans)
