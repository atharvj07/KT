t = int(input())
for mmm in range(t):
	(n, m) = map(int, input().split())
	time = 0
	wa = [i - 1 for i in map(int, input().split())]
	a = [0 for i in range(n)]
	for i in wa:
		a[i] = 1
	while a.count(1) != len(a):
		time += 1
		j = 0
		b = a.copy()
		while j < n:
			if b[j] == 1 and j != 0 and (j != n - 1):
				(a[j - 1], a[j + 1]) = (1, 1)
			elif b[j] == 1 and j == 0:
				a[j + 1] = 1
			elif b[j] == 1 and j == n - 1:
				a[j - 1] = 1
			j += 1
	print(time + 1)
