(n, k) = list(map(int, input().split()))
a = list(map(int, input().split()))
b = list(map(int, input().split()))
new = [a[0]]
for i in range(1, n):
	new.append(new[i - 1] + a[i])
(pointer, curr) = (0, 0)
for each in b:
	while each > new[pointer]:
		pointer += 1
	if pointer > 0:
		curr = new[pointer - 1]
	print(pointer + 1, each - curr, sep=' ')
