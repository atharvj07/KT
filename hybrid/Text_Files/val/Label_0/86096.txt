#include <bits/stdc++.h>
const int maxn = 1e6 + 7;
const int mod = 1e9 + 7;
const double eps = 1e-8;
using namespace std;
long long gcd(long long a, long long b) { return b == 0 ? a : gcd(b, a % b); }
long long a[maxn];
int n, t;
vector<int> g[maxn];
long long ans;
map<long long, int> p[maxn];
void dfs(int u, int fa) {
  p[u][a[u]]++;
  for (auto i : p[fa]) p[u][gcd(a[u], i.first)] += i.second;
  for (auto v : g[u]) {
    if (v == fa) continue;
    dfs(v, u);
  }
  for (auto i : p[u]) ans = (ans + i.first * i.second % mod) % mod;
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i < n + 1; i++) {
    scanf("%lld", &a[i]);
  }
  for (int i = 0; i < n - 1; i++) {
    int u, v;
    scanf("%d%d", &u, &v);
    g[u].push_back(v);
    g[v].push_back(u);
  }
  dfs(1, 0);
  printf("%lld\n", ans);
  return 0;
}
