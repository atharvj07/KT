import sys
import math
from functools import reduce
import bisect

def getN():
	return int(input())

def getNM():
	return map(int, input().split())

def getList():
	return list(map(int, input().split()))

def input():
	return sys.stdin.readline().rstrip()

def index(a, x):
	i = bisect.bisect_left(a, x)
	if i != len(a) and a[i] == x:
		return i
	return False
(n, m) = getNM()
adj = [set() for _ in range(n)]
for _ in range(m):
	(a, b) = getNM()
	a -= 1
	b -= 1
	adj[a].add(b)
	adj[b].add(a)
if n < 7 or m == 0:
	print(m)
	exit()
ans = 0
for i in range(7):
	for j in range(i, 7):
		ans = max(ans, m - len(adj[i] & adj[j]))
print(ans)
