import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;

public class ProblemC {
    
    
    public static void main(String[] args) throws IOException {
        BufferedReader s = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.valueOf(s.readLine());
        String[] names = new String[n];
        for (int i = 0 ; i < n ; i++) {
            names[i] = s.readLine();
        }
        
        int[][] dp = new int[26][26];
        int flen = names[0].length();
        int ffc = names[0].charAt(0) - 'a';
        int flc = names[0].charAt(flen-1) - 'a';
        dp[ffc][flc] = flen; 
        for (int i = 1 ; i < n ; i++) {
            int nlen = names[i].length();
            int from = names[i].charAt(0) - 'a';
            int to = names[i].charAt(nlen-1) - 'a';
            for (int f = 0 ; f < 26 ; f++) {
                if (dp[f][from] >= 1) {
                    dp[f][to] = Math.max(dp[f][to], dp[f][from] + nlen);
                }
            }
            dp[from][to] = Math.max(dp[from][to], nlen);
        }
        
        long max = 0;
        for (int i = 0 ; i < 26 ; i++) {
            max = Math.max(max, dp[i][i]);
        }
        System.out.println(max);
    }
}
