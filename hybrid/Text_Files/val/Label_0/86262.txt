#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
 
int Q, p, x, a[309], v[7] = { 2,3,5,7,11,13,17 }, w[7] = { 8,5,3,2,2,2,2 }, maxn;
vector<int>primes;
 
void dfs(vector<int>vec) {
	if (vec.size() == 7) {
		long long s = 1;
		for (int i = 0; i < vec.size(); i++) { for (int j = 0; j < vec[i]; j++) s *= 1LL * v[i]; }
		for (int i = 7; i < primes.size(); i++) {
			int score = a[primes[i]];
			for (int j = 2; j*primes[i] <= 300; j++) {
				if (s%j == 0) score += a[j*primes[i]];
			}
			if (score > 0) { vec.push_back(1); }
			else { vec.push_back(0); }
		}
		int sum = 0;
		for (int i = 2; i <= 300; i++) {
			int cx = i; bool OK = true;
			for (int j = 0; j < primes.size(); j++) {
				int t = 0;
				while (cx%primes[j] == 0) { cx /= primes[j]; t++; }
				if (vec[j] < t) OK = false;
				if (cx == 1) break;
			}
			if (OK == true) sum += a[i];
		}
		maxn = max(maxn, sum);
		return;
	}
	for (int i = 0; i <= w[vec.size()]; i++) {
		vector<int>vec2 = vec; vec2.push_back(i);
		dfs(vec2);
	}
}
 
int main() {
	cin >> Q;
	for (int i = 2; i <= 300; i++) {
		bool OK = false;
		for (int j = 2; j*j <= i; j++) { if (i%j == 0)OK = true; }
		if (OK == false) primes.push_back(i);
	}
	for (int i = 1; i <= Q; i++) {
		cin >> p >> x;
		a[p] = x;
	}
	dfs(vector<int>{});
	cout << maxn << endl;
	return 0;
}
