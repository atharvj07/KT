#include <iostream>
#include <vector>
#include <algorithm>
#define rep(i,n) for(int i=0; i<(int)(n); i++)

using namespace std;
typedef long long LL;
const LL mod=1e9+7;

LL modpow(LL x, LL n){
	LL r=1;
	while(n){
		if(n&1) r=r*x%mod;
		x=x*x%mod;
		n>>=1;
	}
	return r;
}

int main(){
  int N;
  cin >> N;
  vector<LL> C(N);
  rep(i,N) cin >> C[i];
  sort(C.begin(),C.end());
  vector<LL> P(N);
  rep(i,N){
    P[i]=modpow(2,2*N-2)*(N+1-i);
    P[i]%=mod;
  }
  LL ans=0;
  rep(i,N){
    ans+=C[i]*P[i];
    ans%=mod;
  }
  cout << ans << endl;

  return 0;
}
