x = list(map(int, input().split()))
field = []
for i in range(x[0]):
	q = input()
	field.append([i for i in q])
points = [[0] * x[0] for i in range(x[0])]
checkerfield = []
checkerfieldtrans = []
ship = ['.'] * x[1]
for i in range(x[0]):
	for j in range(x[0]):
		checkerfield.append(field[i][j])
		checkerfieldtrans.append(field[j][i])
maxi = 0
maxiind = [0, 0]
for r in range(x[0]):
	for p in range(x[0] - x[1] + 1):
		fp = p + x[1]
		if checkerfield[p + r * x[0]:fp + r * x[0]] == ship:
			for i in range(x[1]):
				points[r][i + p] = points[r][i + p] + 1
				if points[r][i + p] >= maxi:
					maxi = points[r][i + p]
					maxiind = [r, i + p]
pointstrans = [[points[j][i] for j in range(x[0])] for i in range(x[0])]
for r in range(x[0]):
	for p in range(x[0] - x[1] + 1):
		fp = p + x[1]
		if checkerfieldtrans[p + r * x[0]:fp + r * x[0]] == ship:
			for i in range(x[1]):
				pointstrans[r][i + p] = pointstrans[r][i + p] + 1
				if pointstrans[r][i + p] >= maxi:
					maxi = pointstrans[r][i + p]
					maxiind = [i + p, r]
print(maxiind[0] + 1, maxiind[1] + 1)
