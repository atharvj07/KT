num = input()
inGame = [1, 2]
check = 1
for _ in range(int(num)):
	x = int(input())
	nextNum = 6 - (inGame[0] + inGame[1])
	if x == int(inGame[0]):
		inGame[1] = nextNum
		continue
	elif x == inGame[1]:
		inGame[0] = nextNum
		continue
	else:
		check = 0
		break
if check == 1:
	print('YES')
else:
	print('NO')
