(n, p, q) = [int(j) for j in input().split()]
s = str(input())
(x, y) = (-1, -1)
found = False
for j in range(101):
	for k in range(101):
		res = j * p + k * q
		if res == n:
			(x, y) = (j, k)
			found = True
			break
		elif res > n:
			break
	if found == True or j * p > n:
		break
if found == False:
	print(-1)
else:
	(p1, p2) = (0, p)
	List = []
	for j in range(x):
		List.append(s[p1:p2])
		p1 += p
		p2 += p
	p2 = p1 + q
	for j in range(y):
		List.append(s[p1:p2])
		p1 += q
		p2 += q
	print(len(List))
	for j in range(len(List)):
		print(List[j])
