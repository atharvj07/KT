def process(e):
	answer = 0
	graphs = [{}, {}]

	def find_root(x, i):
		g = graphs[i]
		L = []
		while x != g[x]:
			L.append(x)
			x = g[x]
		for y in L:
			g[y] = x
		return x
	for (a, b, w) in e:
		g = graphs[w]
		if a not in g and b not in g:
			g[a] = min(a, b)
			g[b] = min(a, b)
		elif b not in g:
			a2 = find_root(a, w)
			g[b] = a2
		elif a not in g:
			b2 = find_root(b, w)
			g[a] = b2
		else:
			a2 = find_root(a, w)
			b2 = find_root(b, w)
			g[a2] = min(a2, b2)
			g[b2] = min(a2, b2)
	roots = [{}, {}]
	for i in range(2):
		for x in graphs[i]:
			x2 = find_root(x, i)
			if x2 not in roots[i]:
				roots[i][x2] = 0
			roots[i][x2] += 1
		for x in roots[i]:
			answer += roots[i][x] * (roots[i][x] - 1)
	for x in graphs[0]:
		if x in graphs[1]:
			x1 = find_root(x, 0)
			x2 = find_root(x, 1)
			answer += (roots[0][x1] - 1) * (roots[1][x2] - 1)
	return answer
n = int(input())
e = []
for i in range(n - 1):
	(a, b, w) = [int(x) for x in input().split()]
	e.append([a, b, w])
print(process(e))
