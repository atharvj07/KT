direction = input()
keyboard = ['qwertyuiop', 'asdfghjkl;', 'zxcvbnm,./']
mapping = {}
for i in range(len(keyboard)):
	for j in range(1, len(keyboard[i])):
		if direction == 'L':
			mapping[keyboard[i][j - 1]] = keyboard[i][j]
		else:
			mapping[keyboard[i][j]] = keyboard[i][j - 1]
print(''.join([mapping[c] for c in input()]))
