(n, p, q) = map(int, input().split())
(a, b, f) = (0, 0, 0)
st = input()
for v in range(0, n + 1):
	for u in range(0, n + 1):
		if v * p + u * q == n:
			a = v
			b = u
			f = 1
			break
if f == 0:
	print('-1')
	quit()
print(a + b)
x = 0
while x < n:
	if a > 0:
		print(st[x:x + p])
		x = x + p
		a = a - 1
	else:
		print(st[x:x + q])
		x = x + q
