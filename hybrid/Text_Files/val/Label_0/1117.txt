(a, b, c) = list(map(int, input().split()))
pos1 = 0
pos2 = 0
pos3 = 0
if a + b <= c:
	pos1 = 1
if a + c <= b:
	pos2 = 1
if b + c <= a:
	pos3 = 1
if pos1 == 0 and pos2 == 0 and (pos3 == 0):
	print(0)
else:
	if pos1 == 1:
		pos1 = c - (a + b) + 1
	else:
		pos1 = 10000000000
	if pos2 == 1:
		pos2 = b - (a + c) + 1
	else:
		pos2 = 10000000000
	if pos3 == 1:
		pos3 = a - (b + c) + 1
	else:
		pos3 = 10000000000
	print(min(pos1, min(pos2, pos3)))
