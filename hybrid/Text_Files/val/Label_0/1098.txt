import math
from collections import defaultdict
for i1 in range(int(input())):
	n = int(input())
	l = list(map(int, input().split()))
	if n == 1:
		print(0)
		continue
	dict = defaultdict(int)
	for i in l:
		dict[int(math.log(i, 2))] += 1
	ans = 0
	for i in dict.values():
		ans += i * (i - 1) // 2
	print(ans)
