t = int(input())
s = []
for kfnf in range(0, t):
	a = input()
	if 'aa' in a or 'bb' in a or 'cc' in a:
		s.append(-1)
	elif 'a' not in a and 'b' not in a and ('c' not in a):
		j = len(a)
		l = j % 3
		i = ''
		i += int(j // 3) * 'abc'
		if l <= 1:
			i += l * 'a'
		else:
			i += 'ab'
		s.append(i)
	else:
		u = list(a)
		b = len(u)
		if u[0] == '?' and u[1] == '?':
			u[0] = 'a'
		elif u[0] == '?' and u[1] != '?':
			if u[1] == 'a' or u[1] == 'c':
				u[0] = 'b'
			else:
				u[0] = 'a'
		if u[b - 1] == '?' and u[b - 2] == '?':
			u[b - 1] = 'a'
		elif u[b - 1] == '?' and u[b - 2] != '?':
			if u[b - 2] == 'a' or u[b - 2] == 'b':
				u[b - 1] = 'c'
			else:
				u[b - 1] = 'a'
		for i in range(1, b - 1):
			if u[i] == '?':
				if u[i + 1] == '?':
					if u[i - 1] == 'a':
						u[i] = 'b'
					elif u[i - 1] == 'b' or u[i - 1] == 'c':
						u[i] = 'a'
				else:
					p = [u[i - 1], u[i + 1]]
					p.sort()
					if p == ['a', 'b'] or p == ['b', 'b'] or p == ['a', 'a']:
						u[i] = 'c'
					elif p == ['a', 'c'] or p == ['a', 'a']:
						u[i] = 'b'
					else:
						u[i] = 'a'
		d = ''
		for i in u:
			d += i
		s.append(d)
for i in s:
	print(i)
