from math import inf

def is_increasing_matrix(matrix, r, c):
	for i in range(r):
		for j in range(c):
			top = -1 if i == 0 else matrix[i - 1][j]
			bottom = inf if i == r - 1 else matrix[i + 1][j]
			left = -1 if j == 0 else matrix[i][j - 1]
			right = inf if j == c - 1 else matrix[i][j + 1]
			if not (left < matrix[i][j] < right and top < matrix[i][j] < bottom):
				return False
	return True

def main():
	(r, c) = map(int, input().split())
	matrix = []
	for i in range(r):
		matrix.append(list(map(int, input().split())))
	changed = False
	for i in range(r - 2, 0, -1):
		for j in range(c - 2, 0, -1):
			upper_margin = matrix[i][j + 1] + matrix[i + 1][j] if 0 in [matrix[i][j + 1], matrix[i + 1][j]] else min(matrix[i][j + 1], matrix[i + 1][j])
			lower_margin = max(matrix[i][j - 1], matrix[i - 1][j])
			if lower_margin >= upper_margin - 1:
				print(-1)
				return
			if matrix[i][j] == 0:
				matrix[i][j] = upper_margin - 1
				changed = True
	if not changed:
		if not is_increasing_matrix(matrix, r, c):
			print(-1)
			return
	print(sum([sum(row) for row in matrix]))
main()
