#include <bits/stdc++.h>
using namespace std;
template <typename T>
bool MinPlace(T &a, const T &b) {
  if (a > b) {
    a = b;
    return true;
  }
  return false;
}
template <typename T>
bool MaxPlace(T &a, const T &b) {
  if (a < b) {
    a = b;
    return true;
  }
  return false;
}
vector<long long> adj[100005];
long long in[100005], out[100005];
void addedge(long long u, long long v) { adj[u].push_back(v); }
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  long long n, m;
  cin >> n >> m;
  for (long long i = 1; i <= m; i++) {
    long long u, v;
    cin >> u >> v;
    if (u > v) swap(u, v);
    in[v]++;
    out[u]++;
    addedge(u, v);
  }
  long long ans = 0;
  for (long long i = 1; i <= n; i++) {
    ans += in[i] * out[i];
  }
  cout << ans << "\n";
  long long q;
  cin >> q;
  while (q--) {
    long long v;
    cin >> v;
    ans -= in[v] * out[v];
    for (auto k : adj[v]) {
      ans -= out[k] * in[k];
      out[k]++;
      in[k]--;
      ans += out[k] * in[k];
      adj[k].push_back(v);
    }
    in[v] += adj[v].size();
    out[v] = 0;
    adj[v].clear();
    cout << ans << "\n";
  }
}
