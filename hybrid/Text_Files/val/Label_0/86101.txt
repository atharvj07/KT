#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2e5 + 50;
int n, m;
int k, a[MAXN];
int head[MAXN], tot;
struct Edge {
  int nex, to, dis;
} eg[MAXN];
void add(int a, int b, int c = 1) {
  eg[++tot] = {head[a], b, c};
  head[a] = tot;
}
int dis[MAXN], vis[MAXN];
priority_queue<pair<int, int>, vector<pair<int, int> >,
               greater<pair<int, int> > >
    pq;
void Dijkstra(int rt) {
  memset(dis, 0x3f, sizeof(dis));
  memset(vis, 0, sizeof(vis));
  while (!pq.empty()) pq.pop();
  dis[rt] = 0;
  pq.push({0, rt});
  while (!pq.empty()) {
    pair<int, int> t = pq.top();
    pq.pop();
    if (vis[t.second]) continue;
    vis[t.second] = 1;
    for (int i = head[t.second]; i; i = eg[i].nex) {
      int to = eg[i].to;
      if (dis[to] > t.first + eg[i].dis) {
        dis[to] = t.first + eg[i].dis;
        pq.push(make_pair(dis[to], to));
      }
    }
  }
}
int minn = 0, maxx = 0;
pair<int, int> rec[MAXN];
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1, a, b; i <= m; i++) {
    scanf("%d%d", &a, &b);
    add(b, a);
    rec[tot] = make_pair(a, b);
  }
  scanf("%d", &k);
  for (int i = 1; i <= k; i++) scanf("%d", &a[i]);
  Dijkstra(a[k]);
  memset(head, 0, sizeof(head));
  tot = 0;
  for (int i = 1; i <= m; i++) add(rec[i].first, rec[i].second);
  for (int i = 1; i < k; i++)
    if (dis[a[i]] != dis[a[i + 1]] + 1) minn++;
  for (int i = 1; i < k; i++) {
    if (dis[a[i]] != dis[a[i + 1]] + 1)
      maxx++;
    else
      for (int j = head[a[i]]; j; j = eg[j].nex) {
        int to = eg[j].to;
        if (to != a[i + 1] && dis[to] + 1 == dis[a[i]]) {
          maxx++;
          break;
        }
      }
  }
  printf("%d %d\n", minn, maxx);
  return 0;
}
