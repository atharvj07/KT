using namespace std;
#define rep(i,m,n) for(int (i)=(int)(m);i<(int)(n);i++)
#define REP(i,n) rep(i,0,n)
#define ll long long

ll dp[6060][2020];

int main() {
	ios::sync_with_stdio(false);
	cin.tie(0);


	ll n, m;
	cin >> n >> m;
	dp[0][0] = 1;
	n *= 3;
	REP(i, n) {
		REP(j, n / 3 + 1) {
			if (dp[i][j] == 0)continue;
			(dp[i + 1][j] += dp[i][j]) %= m;
			(dp[i + 2][j + 1] += dp[i][j] * (n - i - 1) % m) %= m;
			(dp[i + 3][j + 1] += dp[i][j] * (n - i - 1) * (n - i - 2) % m) %= m;
		}
	}
	ll ans = 0;
	REP(i, n / 3 + 1)(ans += dp[n][i]) %= m;
	cout << ans << "\n";

	return 0;
}
