import os, io
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
(n, m, k) = map(int, input().split())
kList = list(map(int, input().split()))
adjacencyList = []
maxDist = 0
for _ in range(n):
	adjacencyList.append([])
for _ in range(m):
	(p, q, r) = map(int, input().split())
	adjacencyList[p - 1].append((q - 1, r))
	adjacencyList[q - 1].append((p - 1, r))
	if r > maxDist:
		maxDist = r
l = 0
r = maxDist + 1
while l < r:
	m = (l + r) // 2
	isVisited = [False] * n
	dfsStack = []
	dfsStack.append(kList[0] - 1)
	isVisited[kList[0] - 1] = True
	while dfsStack:
		curVertex = dfsStack.pop()
		for (elem, dist) in adjacencyList[curVertex]:
			if not isVisited[elem] and dist <= m:
				isVisited[elem] = True
				dfsStack.append(elem)
	allVisited = True
	for elem in kList:
		if not isVisited[elem - 1]:
			allVisited = False
			break
	if allVisited:
		r = m
	else:
		l = m + 1
print(' '.join([str(l)] * k))
