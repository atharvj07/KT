#include <cstdio>
#include <algorithm>
#include <utility>
#include <vector>
#include <map>
using namespace std;

typedef pair<int,int> pii;

void dec(map<int,int> &mp, int x){
	if(!--mp[x]){ mp.erase(x); }
}

int dif(map<int,int> &mp){
	map<int,int>::iterator it = mp.end();
	--it;
	return it->first - mp.begin()->first;
}

bool check(const vector<pii> &v, int y0, int th){
	int n = v.size();
	map<int,int> mpx, mpy;
	for(int i = 0; i < n; ++i){
		++mpx[v[i].first];
		++mpy[v[i].second];
	}
	if(dif(mpx) <= th && dif(mpy) <= th){ return true; }

	int lt = 0;
	for(int i = 0; i < n; ++i){
		while(v[lt].first < v[i].first - th){
			if(v[lt].second <= y0 + th){
				++mpx[v[lt].first];
				++mpy[v[lt].second];
			}
			++lt;
		}
		if(v[i].second <= y0 + th){
			dec(mpx, v[i].first);
			dec(mpy, v[i].second);
			if(dif(mpx) <= th && dif(mpy) <= th){ return true; }
		}
	}
	return false;
}

int main(){
	int n;
	scanf("%d", &n);
	vector<pii> v(n);
	int y0 = 1 << 30;
	for(int i = 0; i < n; ++i){
		int x, y;
		scanf("%d%d", &x, &y);
		v[i] = pii(x + y, x - y);
		y0 = min(y0, v[i].second);
	}
	sort(v.begin(), v.end());
	
	int left = 1, right = 300000;
	while(left < right){
		int mid = (left + right) >> 1;
		if(check(v, y0, mid)){
			right = mid;
		}
		else{
			left = mid + 1;
		}
	}
	printf("%d\n", left);
}
