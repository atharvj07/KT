t = int(input())
while t > 0:
	t -= 1
	n = int(input())
	s1 = input()
	s2 = input()
	ls = []
	for i in range(n):
		if s1[i] != s2[i]:
			ls.append(i)
	if len(ls) == 0:
		print('YES')
	elif len(ls) != 2:
		print('NO')
	elif s1[ls[0]] == s1[ls[1]] and s2[ls[0]] == s2[ls[1]]:
		print('YES')
	else:
		print('NO')
