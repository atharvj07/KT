#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a[100000], f[2][100000], s = 0;
  bool b[1048576];
  cin >> n;
  memset(b, 0, sizeof(b));
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    f[0][i] = a[i];
    b[a[i]] = 1;
  }
  for (bool tag = 0; n > 1; tag = !tag) {
    int s = 0;
    for (int j = 0; j < n - 1; j++)
      if (f[tag][j] != f[tag][j + 1]) {
        f[!tag][s] = f[tag][j] | f[tag][j + 1];
        b[f[!tag][s]] = 1;
        s++;
      }
    n = s;
  }
  for (int i = 0; i < 1048576; i++)
    if (b[i]) s++;
  cout << s << endl;
  return 0;
}
