#define NMAX 105
const long double PI = acos(-1);

int n,ans[NMAX][NMAX],P[NMAX];

void circ(){
  int aux = P[n];
  for (int i=n;i>1;i--){
    P[i] = P[i-1];
  }
  P[1] = aux;
}

int main(){
  ios::sync_with_stdio(false);
  cin >> n;
  if (n<=2){
    cout << "-1\n";
    return 0;
  }
  for (int i=1;i<=n;i++){
    P[i] = i;
  }
  for (int i=1;i<=n;i++){
    if (n%2==1){
      int nr = 0;
      for (int j=i+1;j<=n;j++){
        nr++;
        // if (ans[j][nr] == i)
          // cout << "NUU\n";
        ans[i][nr] = j;
      }

      for (int j=1;j<i;j++){
        nr++;
        // if (ans[j][nr] == i)
          // cout << "NUU\n";
        ans[i][nr] = j;
      }
    }
    else{
      int nr = 0;
      if (P[1] == i && i != 1) circ();
      for (int j=1;j<=n;j++){
        if (P[j] == i) continue;
        nr++;
        // if (ans[P[j]][nr] == i)
          // cout << "NU\n";
        ans[i][nr] = P[j];
      }
      circ();
    }
  }

  for (int i=1;i<=n;i++){
    for (int j=1;j<n;j++){
      cout << ans[i][j] << ' ';
    }
    cout << '\n';
  }
  return 0;
}
