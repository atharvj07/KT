from sys import stdin
from math import sqrt
import time
inf = stdin
(n, h, l, r) = map(int, inf.readline().split(' '))
arr1 = list(map(int, inf.readline().split(' ')))
dp = [[-1 for i in range(h)] for j in range(n + 1)]
dp[0][0] = 0
for i in range(1, n + 1):
	for j in range(h):
		prev1 = (j - arr1[i - 1] + h) % h
		prev2 = (j - arr1[i - 1] + 1 + h) % h
		toadd = 0
		if l <= j and r >= j:
			toadd = 1
		if dp[i - 1][prev1] != -1:
			dp[i][j] = max(dp[i][j], dp[i - 1][prev1] + toadd)
		if dp[i - 1][prev2] != -1:
			dp[i][j] = max(dp[i][j], dp[i - 1][prev2] + toadd)
ans = 0
for i in range(n + 1):
	ans = max(dp[i])
print(ans)
