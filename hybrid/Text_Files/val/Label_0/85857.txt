#include <bits/stdc++.h>
using namespace std;
template <class T>
bool uin(T &a, T b) {
  return a > b ? (a = b, true) : false;
}
template <class T>
bool uax(T &a, T b) {
  return a < b ? (a = b, true) : false;
}
const int maxn = 2011, maxm = 2011;
bitset<maxm> b[maxn];
char buf[maxm];
int dist[maxn][maxn];
pair<int, int> me[maxn];
int used[maxn];
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.precision(10);
  cout << fixed;
  int T;
  scanf("%d", &T);
  while (T--) {
    int n, m;
    scanf("%d%d", &n, &m);
    for (int i = 0; i < (int)(n); ++i)
      for (int j = 0; j < (int)(maxm); ++j) b[i][j] = 0;
    vector<int> cnt(m);
    for (int i = 0; i < (int)(m); ++i) {
      scanf("%s", buf);
      for (int j = 0; j < (int)(n); ++j)
        if (buf[j] == '1') b[j].set(i), ++cnt[i];
    }
    for (int i = 0; i < (int)(n); ++i)
      for (int j = 0; j < (int)(n); ++j) {
        dist[i][j] = (b[i] & b[j]).count();
      }
    for (int i = 0; i < (int)(n); ++i) me[i] = {-1e9, -1}, used[i] = 0;
    int Z = 0;
    vector<pair<int, int> > e;
    for (int i = 0; i < (int)(n); ++i) {
      int v = -1;
      for (int j = 0; j < (int)(n); ++j) {
        if (used[j]) continue;
        if (v == -1 || me[j] > me[v]) v = j;
      }
      used[v] = 1;
      if (me[v].second != -1) e.push_back({v, me[v].second});
      for (int j = 0; j < (int)(n); ++j) uax(me[j], make_pair(dist[v][j], v));
    }
    bool ok = true;
    for (int j = 0; j < (int)(m); ++j) {
      int z = 0;
      for (auto w : e)
        if (b[w.first][j] && b[w.second][j]) ++z;
      ok &= z == cnt[j] - 1;
    }
    if (ok) {
      cout << "YES\n";
      for (auto w : e) cout << w.first + 1 << ' ' << w.second + 1 << '\n';
    } else
      cout << "NO\n";
  }
  return 0;
}
