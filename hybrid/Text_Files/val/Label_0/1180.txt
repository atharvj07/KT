for _ in range(int(input())):
	(n, m) = list(map(int, input().split()))
	s = input()
	prev = s
	for __ in range(m):
		curr = ''
		for i in range(n):
			if i == 0:
				if s[i] == '0' and s[i + 1] == '1':
					curr += '1'
				else:
					curr += s[i]
			elif i == n - 1:
				if s[i] == '0' and s[i - 1] == '1':
					curr += '1'
				else:
					curr += s[i]
			elif s[i] == '0' and int(s[i - 1]) + int(s[i + 1]) == 1:
				curr += '1'
			else:
				curr += s[i]
		if curr == s:
			break
		s = curr
	print(curr)
