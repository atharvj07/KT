def arrays(n, arr1, arr2):
	if arr1 == arr2:
		print('0')
		return
	count = 0
	results = []
	while arr1 != arr2:
		ri = -1
		rj = -1
		for i in range(n):
			a1 = arr1[i]
			a2 = arr2[i]
			if ri == -1 and a1 - 1 >= a2:
				ri = i
			if rj == -1 and a1 + 1 <= a2:
				rj = i
		if ri > -1 and rj > -1:
			arr1[ri] = arr1[ri] - 1
			arr1[rj] = arr1[rj] + 1
			count += 1
			results.append(f'{ri + 1} {rj + 1}')
		else:
			break
	if arr1 != arr2:
		print('-1')
	else:
		print(count)
		print('\n'.join(results))
t = int(input())
for i in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	arrays(n, a, b)
