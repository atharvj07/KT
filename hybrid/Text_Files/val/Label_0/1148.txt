import collections as cc
import itertools as it
import bisect as bi
import sys
sys = sys.stdin.readline
I = lambda : list(map(int, input().split()))

def find(x, i):
	while x != parent[i][x]:
		x = parent[i][x]
	return x

def union(x, y, i):
	a = find(x, i)
	b = find(y, i)
	parent[i][a] = parent[i][b] = min(a, b)
	if min(a, b) == a:
		sz[i][a] += sz[i][b]
	else:
		sz[i][b] += sz[i][a]
(n,) = I()
parent = [[i for i in range(n + 1)] for j in range(2)]
sz = [[1 for i in range(n + 1)] for j in range(2)]
for i in range(n - 1):
	t = I()
	union(*t)
ans = 0
for i in range(1, n + 1):
	if parent[0][i] == i:
		ans += sz[0][i] * (sz[0][i] - 1)
	if parent[1][i] == i:
		ans += sz[1][i] * (sz[1][i] - 1)
	ans += (sz[0][find(i, 0)] - 1) * (sz[1][find(i, 1)] - 1)
print(ans)
