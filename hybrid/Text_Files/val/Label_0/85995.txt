long long modulo(long long a, long long b, long long n) {
  long long x = 1, y = a;
  while (b > 0) {
    if (b % 2 == 1) {
      x = (x * y) % n;
    }
    y = (y * y) % n;
    b /= 2;
  }
  return x % n;
}

struct node {
  mutable long long v, i, n;
  node() {}
  node(long long x, long long y, long long z = 0) {
    v = x;
    i = y;
    n = z;
  }
  bool operator<(const node& y) const {
    if (v == y.v)
      if (i != y.i)
        return i < y.i;
      else
        return n < y.n;
    else
      return v < y.v;
  }
};
ostream& operator<<(ostream& o, node x) {
  o << '(' << x.i << ',' << x.v;
  if (x.n != 0) o << ',' << x.n;
  o << ")\t";
}
bool cmp(node x, node y) {
  if (x.i != y.i)
    return x.i < y.i;
  else if (x.v != y.v)
    return x.v > y.v;
  else
    return x.n < y.n;
}
