import getpass
import sys
import math
import functools

def ria():
	return [int(i) for i in input().split()]
files = True
if getpass.getuser() == 'frohenk' and files:
	sys.stdin = open('test.in')
n = ria()[0]
ar = []
for i in range(n):
	st = input()
	ar.append((st.count('s'), st.count('h'), st))
ar.sort(key=functools.cmp_to_key(lambda x, y: -x[0] * y[1] + x[1] * y[0]))
fst = ''
for (d, a, b) in ar:
	fst += b
suma = 0
kek = 0
for i in fst:
	if i == 's':
		kek += 1
	else:
		suma += kek
print(suma)
sys.stdout.close()
