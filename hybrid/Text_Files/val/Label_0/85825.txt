#include <bits/stdc++.h>
using namespace std;
long long arr[100009];
int main() {
  long long N, D;
  scanf(" %I64d %I64d", &N, &D);
  for (int i = 0; i < N; i++) scanf(" %I64d", &arr[i]);
  long long ans = 0;
  long long left = 0;
  long long add = 0;
  for (int i = 2; i < N; i++) {
    while (arr[left] + D < arr[i]) left++;
    add = (i - left);
    ans += (add * (add - 1)) / 2;
  }
  printf("%I64d\n", ans);
  return 0;
}
