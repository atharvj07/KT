#include <bits/stdc++.h>
using namespace std;
int const N = 500 + 10, mod = 1e9 + 7;
int n, a[N], dp[N][N];
int32_t main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int l = n - 1; l >= 0; l--) {
    dp[l][l] = 1;
    for (int r = l + 1; r < n; r++) {
      if (a[l] < a[l + 1])
        dp[l][r] = dp[l + 1][r];
      else
        dp[l][r] = 0;
      for (int k = l + 1; k <= r; k++) {
        if (k == r || a[l] < a[k + 1]) {
          (dp[l][r] += dp[l + 1][k] * (k == r ? 1 : dp[k + 1][r]) % mod) %= mod;
        }
      }
    }
  }
  if (n == 1)
    cout << "1\n";
  else
    cout << dp[1][n - 1] << '\n';
}
