#include <bits/stdc++.h>
using namespace std;
const int INF = 0x3f3f3f3f;
const int MAXN = (int)2e5 + 5;
long long n, p, q, x;
int main() {
  cin >> p >> q >> n;
  while (n) {
    cin >> x;
    if (!q || x > p / q) break;
    p -= x * q;
    swap(p, q);
    n--;
  }
  if (n || q)
    puts("NO");
  else
    puts("YES");
  return 0;
}
