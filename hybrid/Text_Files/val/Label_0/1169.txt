import sys
NM = list(map(int, input().split()))
n = NM[0]
m = NM[1]
choices = []
for _ in range(n):
	choices.append(list(input()))
Choices = [[None for _ in range(n)] for _ in range(m)]
for i in range(m):
	for j in range(n):
		Choices[i][j] = choices[j][i]
res = list(map(int, input().split()))
tot = 0
for i in range(m):
	tot += res[i] * max(Choices[i].count('A'), Choices[i].count('B'), Choices[i].count('C'), Choices[i].count('D'), Choices[i].count('E'))
print(tot)
