int n;
int a[N], nxt[N], pos[N];

void sol()
{
    scanf("%d", &n);
    for(int i = 1; i <= n; ++i) scanf("%d", &a[i]);
    for(int i = 0; i <= n; ++i) pos[i] = n+1;
    for(int i = n; i >= 0; --i) {
        nxt[i] = pos[a[i]];
        pos[a[i]] = i;
    }
    int t1 = 0, t2 = 0, ans = 0;
    for(int i = 1; i <= n; ++i) {
        if(a[t1] == a[i]) t1 = i;
        else if(a[t2] == a[i]) t2 = i;
        else if(nxt[t1]<nxt[t2]) t2=i,ans++;
        else t1=i,ans++;
    }
    printf("%d\n", ans);
}
