primera_linea = input()
mensaje_1 = input()
mensaje_2 = input()
(n, m) = primera_linea.split(' ')
largos_1 = mensaje_1.split(' ')
largos_2 = mensaje_2.split(' ')
resultado = suma_1 = suma_2 = i = j = 0
n = int(n)
m = int(m)
while True:
	if i >= n and j >= m:
		break
	if suma_1 <= suma_2:
		suma_1 += int(largos_1[i])
		i += 1
	else:
		suma_2 += int(largos_2[j])
		j += 1
	if suma_1 == suma_2:
		resultado += 1
print(resultado)
