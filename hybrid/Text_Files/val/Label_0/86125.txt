#include<bits/stdc++.h>
using namespace std;
int main(){
	int n; scanf("%d",&n);
	vector<int> c(26);
	for(int i=0;i<c.size();++i) scanf("%d",&c[i]);
	vector<vector<int> > s(n);
	for(int i=0;i<n;++i){
		s[i].resize(26);
		for(int j=0;j<26;++j)
			scanf("%d",&s[i][j]);
	}
	for(int i=0;i<n;++i){
		int k=0;
		for(int j=1;j<26;++j)
			if(s[i][j]>s[i][k]) 
				k=j;
		printf("%d\n",k+1);
	}
	return 0;
}
