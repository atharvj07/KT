#include <bits/stdc++.h>
using namespace std;
const int dbg = 0;
struct FatalMessage {
  FatalMessage(const char* file, int line, const char* cond) {
    std::cerr << file << " L" << line << " failed: " << cond << ". ";
  }
  ~FatalMessage() {
    std::cerr << std::endl;
    std::terminate();
  }
};
template <typename T>
const FatalMessage& operator<<(const FatalMessage& s, const T& t) {
  std::cerr << t;
  return s;
}
template <typename C>
void SortAndUniq(C& c) {
  sort(c.begin(), c.end());
  c.erase(unique(c.begin(), c.end()), c.end());
}
unsigned long long GetCount1(unsigned long long prefix_len,
                             unsigned long long prefix,
                             unsigned long long suffix_len,
                             unsigned long long N) {
  unsigned long long ans = 0;
  for (unsigned long long s = 0; s < suffix_len; ++s) {
    ans += 1ULL << s;
  }
  if (prefix < (N >> suffix_len)) {
    ans += 1ULL << suffix_len;
  } else if (prefix == (N >> suffix_len)) {
    ans += (N & (1ULL << suffix_len) - 1) + 1;
  }
  return ans;
}
unsigned long long GetCount(unsigned long long prefix_len,
                            unsigned long long prefix,
                            unsigned long long suffix_len,
                            unsigned long long N) {
  unsigned long long ans = GetCount1(prefix_len, prefix, suffix_len, N);
  if (prefix % 2 == 0) {
    ans += GetCount1(prefix_len, prefix + 1, suffix_len, N);
  }
  if (dbg) {
    cerr << N << ' ' << prefix << ' ' << ans << '\n';
  }
  return ans;
}
unsigned long long Len(unsigned long long x) {
  unsigned long long ans = 0;
  while (x >> ans) {
    ++ans;
  }
  return ans;
}
unsigned long long GetCount(unsigned long long prefix, unsigned long long N) {
  if (prefix > N) {
    return 0;
  }
  return GetCount(Len(prefix), prefix, Len(N) - Len(prefix), N);
}
unsigned long long GetCountBF(unsigned long long prefix, unsigned long long N) {
  unsigned long long ans = 0;
  for (unsigned long long x = prefix; x <= N; ++x) {
    unsigned long long y = x;
    while (y > prefix) {
      if (y % 2) {
        y -= 1;
      } else {
        y /= 2;
      }
    }
    if (y == prefix) {
      ans += 1;
    }
  }
  return ans;
}
void Verify(unsigned long long N, unsigned long long P) {
  unsigned long long a = GetCount(P, N);
  unsigned long long b = GetCountBF(P, N);
  if (a != b) {
    cerr << N << ' ' << P << ": " << a << " vs " << b << '\n';
    std::terminate();
  }
}
int main() {
  if (dbg >= 9) {
    for (unsigned long long N = 1; N <= 1024; ++N) {
      for (unsigned long long P = 1; P <= N + 1; ++P) {
        Verify(N, P);
      }
    }
  }
  if (0) {
    for (unsigned long long N = 1; N <= 1024; ++N) {
      Verify(N, N);
    }
  }
  cin.sync_with_stdio(false);
  cout.sync_with_stdio(false);
  unsigned long long K, N;
  cin >> N >> K;
  unsigned long long L = 0;
  for (; N >> L; ++L)
    ;
  unsigned long long ans = 1;
  for (unsigned long long oddeven = 1; oddeven <= 2; ++oddeven) {
    if (GetCount(oddeven, N) < K) {
      continue;
    }
    unsigned long long lo = 0, hi = N + 1;
    while (lo + 1 < hi) {
      unsigned long long mid = (lo + hi) / 2;
      if (GetCount(oddeven + mid * 2, N) < K) {
        hi = mid;
      } else {
        lo = mid;
      }
    }
    ans = max(ans, oddeven + lo * 2);
  }
  cout << ans << endl;
}
