#include <iostream>
#include <utility>
using namespace std;
#define MOD 1000000007


long long kaijo[5000001];
long long inv[5000001];



long long mul(long long x,long long y){
    return (x*y)%MOD;
}

long long bekijo(long long x,long long n){
    if(n==0)return 1;
    if(n%2==1)return mul(x,bekijo(x,n-1));
    return mul(bekijo(x,n/2),bekijo(x,n/2));
}

long long nCk(int n,int k){
    if(k==-1)return n==k?1:0;
    return mul(kaijo[n],mul(inv[k],inv[n-k]));
}

int main(void){
    kaijo[0]=1;
    for(long long i=1;i<=5000000;i++){
        kaijo[i]=mul(kaijo[i-1],i);
    }
    
    inv[5000000]=bekijo(kaijo[5000000],MOD-2);
    for(long long i=5000000-1;i>=0;i--){
        inv[i]=mul(inv[i+1],i+1);
    }
    
    while(1){
        long long int A,B,C,Sx,Sy;
        scanf("%lld%lld%lld%lld%lld",&A,&B,&C,&Sx,&Sy);
        if(A+B+C==0)break;
        
        /*引き分けた総得点をi点とするとチームXがA回の試合で合計Sx-i点勝ってYがB回の試合でSy-i点勝った*/
        /*Sx-i>=AかつSy-i>=Bのときのみ考える、つまりiの範囲は0からmin(Sx-A,Sy-B)*/
        /*勝ち点の振り分け方はSx-i点をA個の箱に全部最低1個ずつ振り分ける方法なので(Sx-i-1)C(A-1)*/
        /*同様に負け点の振り分けは(Sy-i-1)C(B-1)*/
        /*引き分け点の振り分けはA+B+C個の箱に(0個も許して)分ける方法なので(i+1)H(A+B+C-1)=(i+A+B+C-1)C(A+B+C-1)*/
        /*さらに勝った試合AをA+B+C個のどこに置き、BをB+Cのどこに置くかで(A+B+C)CA * (B+C)CB通り*/
        
        long long ans=0;
        for(int i=0;i<=min(Sx-A,Sy-B);i++){
            long long po=1;
            po=mul(po,nCk(Sx-i-1,A-1));
            po=mul(po,nCk(Sy-i-1,B-1));
            po=mul(po,nCk(i+A+B+C-1,A+B+C-1));
            po=mul(po,nCk(A+B+C,A));
            po=mul(po,nCk(B+C,B));
            ans+=po;
            ans%=MOD;
        }
        printf("%lld\n",ans);
        
    }
    
    
    return 0;
}
