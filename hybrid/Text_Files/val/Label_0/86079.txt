#include <bits/stdc++.h>
using namespace std;
int n, a[100009], ind, l, r, md = 1;
long long int sm[100009], gr;
map<long long int, int> mp, mps;
int main() {
  int i, j;
  scanf("%d", &n);
  for (i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
    sm[i] += sm[i - 1] + a[i];
    mp[sm[i]]++, mps[a[i]]++;
  }
  if (sm[n] % 2 != 0) return printf("NO"), 0;
  gr = sm[n] / 2;
  for (i = 1; i <= n; i++)
    if (sm[i] == gr) return printf("YES"), 0;
  for (i = 1; i < n - 1; i++) {
    long long int nd = gr - sm[i - 1];
    if (mp[sm[i] + nd] > 0 || mps[nd] > 0) return printf("YES"), 0;
    mp[sm[i]]--, mps[a[i]]--;
  }
  return printf("NO"), 0;
}
