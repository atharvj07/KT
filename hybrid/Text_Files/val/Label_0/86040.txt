// D - Xor Sum 4
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define rp(i,s,e) for(int i=(s);i<(e);++i)
#define MOD 1000000007
 
ll O[60];
 
int main(){
	int N; cin>>N;
	rp(i,0,N){ ll a; scanf("%lld",&a); for(int j=0; j<60 && a; ++j, a>>=1) O[j]+=a&1; }
	ll ans = 0;
	rp(i, 0, 60) ans = (ans + (O[i]*(N-O[i]) % MOD) * ((1LL<<i) % MOD)) % MOD;
	cout<< ans <<endl;
}
