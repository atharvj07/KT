import java.util.*;

public class problemANumbers{

	public static void main(String[] args){
		Scanner myS = new Scanner (System.in);

		
		long n = myS.nextLong();
		long m = myS.nextLong();
		
		/*
		long n = 1000000;
		long m = 1000000;
		*/
		long sum = 0;
		long nRem = n % 5;
		long nMult = (n - nRem)/5;
		long mRem = m % 5;
		long mMult = (m - mRem)/5;

		sum += nMult*mMult*5;
		sum += nMult*mRem;
		sum += mMult*nRem;

		long remSum = nRem + mRem;
		if(remSum >= 5){
			sum += (remSum-4);
		}
		System.out.println(sum);

	}
}
