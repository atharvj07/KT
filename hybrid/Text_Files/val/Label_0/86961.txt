import java.util.Scanner;

public class que4 {

	public static void main(String[] args) {
		Scanner scn = new Scanner(System.in);
		int test = scn.nextInt();
		for (int t = 0; t < test; t++) {
			int r = scn.nextInt();
			int c = scn.nextInt();
			char[][] arr = new char[r][c];
			for (int i = 0; i < r; i++) {
				String s = scn.next();
				for (int j = 0; j < c; j++) {
					arr[i][j] = s.charAt(j);

				}
			}
			int count1 = 0;
			int count2 = 0;
			int count3 = 0;
			int count4 = 0;
			boolean tl = false;
			boolean tr = false;
			boolean br = false;
			boolean bl = false;
			boolean ac=false;
			boolean ar=false;
			int count = 0;
			for (int i = 0; i < arr.length; i++) {
				int cl=0;
				for (int j = 0; j < arr[0].length; j++) {
					if (arr[i][j] == 'A') {
						count++;
						cl++;
					}
					
				}
				if(cl==arr[0].length){
					ac=true;
				}
			}
			for (int i = 0; i < arr[0].length; i++) {
				int cr=0;
				for (int j = 0; j < arr.length; j++) {
					if (arr[j][i] == 'A') {
					
						cr++;
					}
					
				}
				if(cr==arr.length){
					ar=true;
					break;
				}
			}
			for (int i = 0; i < arr[0].length; i++) {
				if (arr[0][i] == 'A') {
					if (i == 0) {
						tl = true;
					}
					if (i == arr[0].length - 1) {
						tr = true;
					}
					count1++;

				}
				if (arr[arr.length - 1][i] == 'A') {
					if (i == 0) {
						bl = true;
					}
					if (i == arr[0].length - 1) {
						br = true;
					}
					count2++;
				}
			}
			for (int i = 0; i < arr.length; i++) {
				if (arr[i][0] == 'A') {
					count3++;
				}
				if (arr[i][arr[0].length - 1] == 'A') {
					count4++;
				}
			}
			if (count == arr.length * arr[0].length) {
				System.out.println(0);
			} else if (count1 == arr[0].length || count2 == arr[0].length || count3 == arr.length
					|| count4 == arr.length) {
				System.out.println(1);
			} else if (tl || tr || br || bl||ar||ac) {
				System.out.println(2);
			} else if (count1 > 0 || count2 > 0 || count3 > 0 || count4 > 0) {
				System.out.println(3);
			} else if (count > 0) {
				System.out.println(4);
			} else {
				System.out.println("MORTAL");
			}
		}
	}

}
