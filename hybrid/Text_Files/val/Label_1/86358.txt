import java.util.Scanner;
 
public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
 
    int t = sc.nextInt();
    for (int i = 0; i < t; i++) {
      int a = sc.nextInt();
      int b = sc.nextInt();
      int c = sc.nextInt();
      int d = sc.nextInt();
 
      int x = a + (b - a) / 2;
      int y = b + (c - b) / 2;
      int z = c + (d - c) / 2;
 
      System.out.println(x + " " + y + " " + z);
    }
  }
}
