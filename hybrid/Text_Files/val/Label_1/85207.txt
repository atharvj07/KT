#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    ll gcd = a[0], ans = gcd;
    for (int i = 1; i < n; i++) {
      gcd = __gcd(gcd, 1ll * a[i] - a[i - 1]);
      if (gcd == 1) {
        cout << -1 << endl;
        goto yes;
      }
      ans = max(ans, 1ll * a[i] - a[i - 1]);
    }
    if (ans == 1)
      cout << 0 << endl;
    else
      cout << ans << " " << gcd << endl;
  yes:
    ;
  }
  return 0;
}
