import java.util.*;
import java.io.*;

public class Main {
    static int mod = (int)1e9 + 7;
    static int[][] dp;
    static boolean[][] vis;
    static int n, m;
    static ArrayList<ArrayList<Integer>> g;
    
    static int solve(int mask, int last) {
        if(vis[mask][last]) return dp[mask][last];
        vis[mask][last] = true;
        dp[mask][last] = 1;
        for(int i = 0; i < m; i++) {
            if((mask&(1<<i)) == 0) {
                int u = i - m*(i/m);
                int v = i/m;
                if(u == last) continue;
                if(u == 0) {
                    if(g.get(v).size() == 1) {
                        dp[mask][last] = (dp[mask][last] + solve(mask, v))%mod;
                    }
                }
                else {
                    int prev = g.get(u).get(g.get(u).size() - 1);
                    if(prev == mask) dp[mask][last] = (dp[mask][last] + solve(mask, v))%mod;
                    prev = g.get(u).get(g.get(u).size() - 2);
                    if(prev == mask) dp[mask][last] = (dp[mask][last] + solve(mask^(1<<i), v))%mod;
                }
            }
        }
        return dp[mask][last];
    }
    
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(in.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        
        g = new ArrayList<ArrayList<Integer>>();
        for(int i = 0; i <= n; i++) g.add(new ArrayList<Integer>());
        dp = new int[1<<m][n+1];
        vis = new boolean[1<<m][n+1];
        
        for(int i = 0; i < m; i++) {
            st = new StringTokenizer(in.readLine());
            int u = Integer.parseInt(st.nextToken()) - 1;
            int v = Integer.parseInt(st.nextToken()) - 1;
            g.get(u).add(v);
        }
        int ans = solve(0, 0);
        System.out.println(ans);
    }
}
