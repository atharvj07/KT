

#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
void setIO() { cin.tie(0)->sync_with_stdio(0); }
const int maxn = 3005;
const int maxm = 200005;
struct Edge {
    int to, w;
};
vector<Edge> G[maxn];
int dep[maxn], fa[maxn], mx[maxn], id[maxn], dfn[maxn], low[maxn], p[maxn], x[maxn];
int n, m, cnt, tim, ans;
bool vis[maxn];
void dfs(int u, int f, int dist) {
    fa[u] = f;
    dep[u] = dep[f] + 1;
    mx[u] = dist;
    low[u] = dfn[u] = ++tim;
    for (auto e : G[u]) {
        int v = e.to;
        if (v == f) continue;
        if (!dfn[v]) {
            dfs(v, u, e.w);
            low[u] = min(low[u], low[v]);
            if (low[v] > dfn[u]) G[v].push_back({u, e.w}), G[u].push_back({v, e.w});
        } else
            low[u] = min(low[u], dfn[v]);
    }
}
void find(int u, int dep) {
    dfn[u] = ++tim;
    id[tim] = u;
    p[dep] = u;
    vis[u] = true;
    for (auto e : G[u])
        if (!vis[e.to]) find(e.to, dep + 1);
}
int main() {
    setIO();
    cin >> n;
    for (int i = 1, a, b, c; i < n; i++) {
        cin >> a >> b >> c;
        G[a].push_back({b, c});
        G[b].push_back({a, c});
    }
    dfs(1, 1, 0);
    for (int i = 1; i <= n; i++) low[i] = fa[i];
    for (int i = 1; i <= n; i++)
        if (low[i] == i) find(i, 1);
    for (int i = 1; i <= n; i++) cin >> x[i];
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= x[i]; j++) {
            if (i == p[j]) {
                cnt++;
                break;
            }
        }
    for (int j = n; j >= 1; j--)
        if (cnt == n) break; else p[j] = p[j + 1];
    for (int i = 1; i <= n; i++) ans = max(ans, mx[p[i]]);
    cout << ans << endl;
    return 0;
}
