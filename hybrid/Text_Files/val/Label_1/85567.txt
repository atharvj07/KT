#include <iostream>
#include <cstring>

using namespace std;

int main() {
  int t, sx, sy, ex, ey;
  cin >> t >> sx >> sy >> ex >> ey;
  char dir[t + 1];
  cin >> dir;
  int x = sx, y = sy;
  int time = 0;
  for (int i = 0; i < t; i++) {
    if (x == ex && y == ey) {
      cout << time << endl;
      return 0;
    }
    if (dir[i] == 'E') x++;
    if (dir[i] == 'S') y--;
    if (dir[i] == 'W') x--;
    if (dir[i] == 'N') y++;
    time++;
  }
  cout << "-1" << endl;
  return 0;
}
