from collections import defaultdict

n = int(input())
years = defaultdict(int)
for _ in range(n):
    b, d = map(int, input().split())
    years[b] += 1
    years[d] -= 1
alive, maxAlive, yearWithMaxAlive = 0, 0, 0
for year, change in sorted(years.items()):
    alive += change
    if alive > maxAlive:
        maxAlive = alive
        yearWithMaxAlive = year
print(yearWithMaxAlive, maxAlive)
