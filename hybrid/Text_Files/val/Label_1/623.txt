import sys
n = int(input())
if n <= 2:
    print(n)
    if n == 1:
        print(".")
    else:
        print("C.")
        print(".C")
else:
    if n % 2 == 0:
        print((n // 2) * (n // 2))
        for i in range(n):
            for j in range(n):
                if (i % 2 == 0 and j % 2 == 0) or (i % 2 == 1 and j % 2 == 1):
                    print("C", end='')
                else:
                    print(".", end='')
            print()
    else:
        print((n // 2) * (n // 2 + 1))
        for i in range(n):
            for j in range(n):
                if (i % 2 == 0 and j % 2 == 0) or (i % 2 == 1 and j % 2 == 1):
                    if j == n - 1:
                        print("C")
                    else:
                        print("C", end='')
                else:
                    print(".", end='')
            print()
