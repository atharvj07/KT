#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int mod = 1e9 + 7;
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    ll cnt1,cnt2,x,y;
    cin >> cnt1 >> cnt2 >> x >> y;
    ll ans = 0;
    for(int i = 1; i <= (cnt1 + cnt2); i++){
        if(i % x == 0 || i % y == 0)
            continue;
        ans++;
    }
    cout << ans << "\n";
    return 0;
}
