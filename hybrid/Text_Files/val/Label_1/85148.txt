#include <bits/stdc++.h>
using namespace std;
const int maxn = 101;
int a[maxn], n;
vector<int> ans;
bool solve(int cur, int prv, vector<bool> &vis) {
  if (cur == 0) return false;
  if (vis[cur]) return true;
  vis[cur] = true;
  for (int i = 1; i <= n; i++) {
    if (i == prv) continue;
    if (a[cur] >= 1 && solve(i, cur, vis)) {
      a[cur]--;
      ans.push_back(cur), ans.push_back(i);
      return true;
    }
  }
  return false;
}
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  vector<bool> vis(n + 1);
  if (!solve(1, 1, vis))
    cout << -1 << endl;
  else {
    cout << ans.size() / 2 << endl;
    for (int i = 0; i < ans.size(); i += 2) cout << ans[i] << " " << ans[i + 1] << endl;
  }
}
