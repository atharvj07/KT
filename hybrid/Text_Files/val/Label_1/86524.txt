import java.util.Scanner;

public class Main {
    private static final int TIME_LIMIT = 10800;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[][] initialPositions = new int[m][2];
        int[][] designatedPositions = new int[m][2];
        for (int i = 0; i < m; i++) {
            initialPositions[i][0] = scanner.nextInt();
            initialPositions[i][1] = scanner.nextInt();
        }
        for (int i = 0; i < m; i++) {
            designatedPositions[i][0] = scanner.nextInt();
            designatedPositions[i][1] = scanner.nextInt();
        }
        int[][][] moves = new int[TIME_LIMIT][2][2];
        int time = 0;
        for (int i = 0; i < m; i++) {
            int[] initialPosition = initialPositions[i];
            int[] designatedPosition = designatedPositions[i];
            if (initialPosition[0] == designatedPosition[0] && initialPosition[1] == designatedPosition[1]) {
                continue;
            }
            int[] currentPosition = initialPosition;
            while (currentPosition[0] != designatedPosition[0] || currentPosition[1] != designatedPosition[1]) {
                if (time >= TIME_LIMIT) {
                    System.out.println(0);
                    return;
                }
                int nextPositionX;
                int nextPositionY;
                if (currentPosition[0] < designatedPosition[0]) {
                    nextPositionX = currentPosition[0] + 1;
                } else if (currentPosition[0] > designatedPosition[0]) {
                    nextPositionX = currentPosition[0] - 1;
                } else {
                    nextPositionX = currentPosition[0];
                }
                if (currentPosition[1] < designatedPosition[1]) {
                    nextPositionY = currentPosition[1] + 1;
                } else if (currentPosition[1] > designatedPosition[1]) {
                    nextPositionY = currentPosition[1] - 1;
                } else {
                    nextPositionY = currentPosition[1];
                }
                if (n >= nextPositionX && nextPositionX >= 1 && n >= nextPositionY && nextPositionY >= 1) {
                    moves[time][0][0] = currentPosition[0];
                    moves[time][0][1] = currentPosition[1];
                    moves[time][1][0] = nextPositionX;
                    moves[time][1][1] = nextPositionY;
                    time++;
                    currentPosition[0] = nextPositionX;
                    currentPosition[1] = nextPositionY;
                } else {
                    System.out.println(0);
                    return;
                }
            }
        }
        System.out.println(time);
        for (int i = 0; i < time; i++) {
            System.out.println(moves[i][0][0] + " " + moves[i][0][1] + " " + moves[i][1][0] + " " + moves[i][1][1]);
        }
    }
}
