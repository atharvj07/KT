import heapq

n = int(input())
potions = list(map(int, input().split()))

health = 0
min_heap = []
count = 0

for potion in potions:
    if health + potion >= 0:
        heapq.heappush(min_heap, potion)
        health += potion
        count += 1
    else:
        if min_heap and potion > min_heap[0]:
            health += potion - heapq.heappop(min_heap)
            heapq.heappush(min_heap, potion)
        else:
            continue

print(count)
