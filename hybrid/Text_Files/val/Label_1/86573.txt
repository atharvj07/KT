import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long x = scanner.nextLong();
        long y = scanner.nextLong();

        // Convert x and y to binary
        String xBinary = Long.toBinaryString(x); // Convert x to binary
        String yBinary = Long.toBinaryString(y); // Convert y to binary

        // Compare the binary representations
        int result = xBinary.compareTo(yBinary);

        // Print the comparison result
        if (result < 0) {
            System.out.println("<"); // x^y < y^x
        } else if (result > 0) {
            System.out.println(">"); // x^y > y^x
        } else {
            System.out.println("="); // x^y = y^x
        }
    }
}
