import java.util.Scanner;

public class Main {
    static int n, m, k;
    static int[][] pos, goal;
    static int[][] map;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        pos = new int[m + 1][2];
        goal = new int[m + 1][2];
        map = new int[n + 1][n + 1];
        for (int i = 1; i <= m; i++) {
            pos[i][0] = sc.nextInt();
            pos[i][1] = sc.nextInt();
            map[pos[i][0]][pos[i][1]] = i;
        }
        for (int i = 1; i <= m; i++) {
            goal[i][0] = sc.nextInt();
            goal[i][1] = sc.nextInt();
        }
        k = 0;
        for (int i = 1; i <= m; i++) {
            if (pos[i][0] != goal[i][0] || pos[i][1] != goal[i][1]) {
                bfs(i);
            }
        }
        System.out.println(k);
        for (int i = 1; i <= k; i++) {
            System.out.println(ans[i][0] + " " + ans[i][1] + " " + ans[i][2] + " " + ans[i][3]);
        }
    }

    static int[][] dir = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
    static int[][] ans;

    static void bfs(int idx) {
        int[][] q = new int[n * n + 1][2];
        int[][] dis = new int[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                dis[i][j] = -1;
            }
        }
        int fr = 1, re = 1;
        q[fr][0] = pos[idx][0];
        q[fr][1] = pos[idx][1];
        dis[pos[idx][0]][pos[idx][1]] = 0;
        ans = new int[3 * n * n][4];
        while (fr <= re) {
            int x = q[fr][0];
            int y = q[fr][1];
            fr++;
            if (x == goal[idx][0] && y == goal[idx][1]) {
                int nx = x, ny = y;
                int cnt = dis[x][y];
                while (cnt > 0) {
                    for (int i = 0; i < 4; i++) {
                        int tx = nx + dir[i][0];
                        int ty = ny + dir[i][1];
                        if (tx >= 1 && tx <= n && ty >= 1 && ty <= n && dis[tx][ty] == cnt - 1) {
                            ans[k][0] = nx;
                            ans[k][1] = ny;
                            ans[k][2] = tx;
                            ans[k][3] = ty;
                            k++;
                            nx = tx;
                            ny = ty;
                            break;
                        }
                    }
                    cnt--;
                }
                return;
            }
            for (int i = 0; i < 4; i++) {
                int tx = x + dir[i][0];
                int ty = y + dir[i][1];
                if (tx >= 1 && tx <= n && ty >= 1 && ty <= n && map[tx][ty] == 0 && dis[tx][ty] == -1) {
                    re++;
                    q[re][0] = tx;
                    q[re][1] = ty;
                    dis[tx][ty] = dis[x][y] + 1;
                }
            }
        }
    }
}
