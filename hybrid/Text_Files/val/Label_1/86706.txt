import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Main {

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        long y1 = Long.parseLong(line[1]);
        line = br.readLine().split(" ");
        List<Long> a = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            a.add(Long.parseLong(line[i]));
        }
        line = br.readLine().split(" ");
        int m = Integer.parseInt(line[0]);
        long y2 = Long.parseLong(line[1]);
        line = br.readLine().split(" ");
        List<Long> b = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            b.add(Long.parseLong(line[i]));
        }
        a.add(y1);
        b.add(y2);
        Collections.sort(a);
        Collections.sort(b);
        long best = 0;
        for (long i : a) {
            int idx = Collections.binarySearch(b, i);
            if (idx < 0) {
                idx = ~idx - 1;
            }
            if (idx >= 0) {
                best = Math.max(best, idx + b.size() - 1 - Collections.binarySearch(b, -i));
            }
        }
        System.out.println(best);
    }
}
