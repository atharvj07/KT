#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> h(n);
  for (int i = 0; i < n; ++i) cin >> h[i];
  int ans = 0;
  for (int i = 0; i < n - 1; ++i) {
    if (h[i + 1] - 1 > h[i]) ans += h[i + 1] - 1 - h[i];
  }
  cout << ans << endl;
  return 0;
}
