#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int MN = 2e5 + 5;
int n;
ll a[MN], k[MN], pw[MN];
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> k[i] >> a[i];
        pw[i] = 1LL << k[i];
    }
    ll ans = 0;
    for (int i = 1; i <= n; i++) ans = max(ans, (a[i] + pw[i] - 1) / pw[i]);
    cout << floor(log2(ans) + 1e-9) << '\n';
}
