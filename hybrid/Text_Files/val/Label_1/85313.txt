#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll n, m, k, ans = LLONG_MIN;
string s;

int main() {
  cin >> n >> m;
  ans = n * n - m * m;
  if (ans >= 0) {
    while (n--) { s += 'o'; }
    while (m--) { s += 'x'; }
  } else
    s = "-1";
  cout << ans << '\n' << s << '\n';
  return 0;
}
