def find_m_cute_sequence(a, b, m):
    if a == b:
        return [1, a]
    for k in range(2, 51):
        min_sum = a + 1
        max_sum = a + (2 ** (k - 2)) * m
        if min_sum <= b <= max_sum:
            sequence = [a]
            b -= a
            for i in range(k - 2, 0, -1):
                # Calculate the current term as the minimum between b - sum of next terms and 1 + m
                current = min((b - (1 << i) * m) // (1 << i), m) + 1
                sequence.append(sequence[-1] + current)
                b -= (1 << i) * (current - 1)
            sequence.append(sequence[-1] + b)
            return [k] + sequence
    return [-1]

q = int(input())
for _ in range(q):
    a, b, m = map(int, input().split())
    result = find_m_cute_sequence(a, b, m)
    print(*result)
