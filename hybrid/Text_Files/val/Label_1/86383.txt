import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class TimeTravel {

    static int n;
    static List<Integer>[] graph;
    static int[] time;
    static boolean[] visited;
    static int[] answerVertices;
    static int[] answerTimes;
    static int k;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        graph = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            graph[u].add(v);
            graph[v].add(u);
        }
        time = new int[n + 1];
        Arrays.fill(time, Integer.MAX_VALUE);
        visited = new boolean[n + 1];
        answerVertices = new int[2 * n];
        answerTimes = new int[2 * n];
        k = 0;
        dfs(1, 0);
        System.out.println(k);
        for (int i = 0; i < k; i++) {
            System.out.println(answerVertices[i] + " " + answerTimes[i]);
        }
    }

    private static void dfs(int vertex, int current) {
        visited[vertex] = true;
        time[vertex] = current;
        answerVertices[k] = vertex;
        answerTimes[k] = current;
        k++;
        for (int next : graph[vertex]) {
            if (!visited[next]) {
                dfs(next, current + 1);
                answerVertices[k] = vertex;
                answerTimes[k] = current;
                k++;
            } else if (time[next] < current - 1) {
                dfs(next, current - 1);
                answerVertices[k] = vertex;
                answerTimes[k] = current;
                k++;
            }
        }
    }
}
