#include <bits/stdc++.h>
using namespace std;

const int N = 117;
char t[N][N];
int d[N][N];
int n,m;

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> t[i] + 1;
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (t[i][j] == 'B') {
                d[i][j] = min(d[i - 1][j],min(d[i][j - 1],d[i - 1][j - 1])) + 1;
            }
        }
    }
    pair<int,int> ans = make_pair(0,0);
    int mx = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (d[i][j] > mx) {
                ans = make_pair(i,j);
                mx = d[i][j];
            }
        }
    }
    cout << ans.first << " " << ans.second << endl;
    return 0;
}
