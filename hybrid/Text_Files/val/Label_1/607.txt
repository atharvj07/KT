from collections import defaultdict

n = int(input())
s = input()

d = defaultdict(int)

for c in s:
    d[c] += 1

ans = [d['R'], d['G'], d['B']]
ans.sort()
if ans[0] > (ans[1] + ans[2] + 1) // 3:
    print(-1)
else:
    t = ['G'] * n
    for i in range(n):
        if s[i] == 'R':
            if ans[1] > (ans[2] + 1) // 3:
                t[i] = 'B'
                ans[1] -= 1
                ans[2] += 1
            else:
                t[i] = 'G'
                ans[1] += 1
        elif s[i] == 'G':
            if ans[0] > (ans[2] + 1) // 3:
                t[i] = 'R'
                ans[0] -= 1
                ans[2] += 1
            else:
                t[i] = 'G'
                ans[0] += 1
        elif s[i] == 'B':
            if ans[0] > (ans[1] + 1) // 3:
                t[i] = 'R'
                ans[0] -= 1
                ans[1] += 1
            else:
                t[i] = 'B'
                ans[0] += 1

    print(n - sum(c == s[i] for i, c in enumerate(t)))
    print(''.join(t))
