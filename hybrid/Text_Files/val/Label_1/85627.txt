#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    for (int i = 0; i < n; ++i) { std::cin >> a[i]; }
    int ans = 1;
    for (int i = 1; i < n; ++i) {
      if (a[i] != a[i - 1]) ++ans;
    }
    if (a[n - 1] != n) ++ans;
    std::cout << ans << "\n";
  }
  return 0;
}
