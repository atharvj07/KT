#include <cstdio>
#include <vector>
#include <queue>
using namespace std;

const int inf = 0x3f3f3f3f;

int main() {
  int n, m, d;
  scanf("%d%d%d", &n, &m, &d);
  vector<bool> dam(n + 1, false);
  vector<vector<int>> dis(n + 1, vector<int>(n + 1, inf));
  for (int i = 0; i < m; i++) {
    int t;
    scanf("%d", &t);
    dam[t] = true;
  }
  for (int i = 1; i < n; i++) {
    int u, v;
    scanf("%d%d", &u, &v);
    dis[u][v] = 1;
    dis[v][u] = 1;
  }
  for (int k = 1; k <= n; k++)
    for (int i = 1; i <= n; i++)
      for (int j = 1; j <= n; j++) dis[i][j] = min(dis[i][j], dis[i][k] + dis[k][j]);
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    int mx = 0;
    for (int j = 1; j <= n; j++) mx = max(mx, dis[i][j]);
    if (mx <= d && dam[i] == false) ans++;
  }
  printf("%d\n", ans);
  return 0;
}
