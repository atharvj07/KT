#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    long long k, b;
    cin >> k >> b;
    ans = max(ans, k + 1);
    a[k] += b;
  }
  while (a[ans] > 0) {
    a[ans] /= 4;
    ans++;
  }
  cout << ans - 1 << endl;
  return 0;
}
