import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

public class Solution {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        Fact[] facts = new Fact[N];
        for (int i = 0; i < N; i++) {
            facts[i] = new Fact(scanner.nextInt(), scanner.nextInt());
        }
        Arrays.sort(facts, Comparator.comparingInt(f -> f.a));

        int maxPlayers = 0;
        int currentRank = 0;
        for (Fact fact : facts) {
            if (fact.b == 0) {
                // Skip facts with B_i = 0, as they don't contribute to the player count.
                continue;
            }

            // Check if the fact is consistent with the current rank.
            if (fact.b <= currentRank) {
                // Inconsistent fact, cannot add more players.
                break;
            }

            // Update the maximum number of players and current rank.
            maxPlayers += fact.b - currentRank;
            currentRank = fact.b;
        }

        System.out.println(maxPlayers);
    }

    static class Fact {
        int a;
        int b;

        Fact(int a, int b) {
            this.a = a;
            this.b = b;
        }
    }
}
