#include <bits/stdc++.h>
using namespace std;
int main() {
    int q;
    cin >> q;
    deque<int> dq;
    set<int> s;
    while (q--) {
        char ch;
        int x;
        cin >> ch >> x;
        if (ch == 'L') {
            dq.push_front(x);
            s.insert(x);
        }
        if (ch == 'R') {
            dq.push_back(x);
            s.insert(x);
        }
        if (ch == '?') {
            int ans = min(abs((int)(distance(dq.begin(), find(dq.begin(), dq.end(), x)))), abs((int)(distance(dq.begin(), find(dq.begin(), dq.end(), x)) - dq.size())));
            cout << ans << '\n';
        }
    }
    return 0;
}
