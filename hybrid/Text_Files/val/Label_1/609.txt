n = int(input().strip())
s = input().strip()

# Possible patterns for a nice garland
patterns = ['RGB', 'RBG', 'GRB', 'GBR', 'BRG', 'BGR']

min_changes = n
best_pattern = ''

# Check each pattern and find the one with minimum changes needed
for pattern in patterns:
    changes = 0
    for i in range(n):
        if s[i] != pattern[i % 3]:
            changes += 1
    if changes < min_changes:
        min_changes = changes
        best_pattern = ''.join([pattern[i % 3] for i in range(n)])

print(min_changes)
print(best_pattern)
