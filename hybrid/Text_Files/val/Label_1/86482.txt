import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        int j = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 1) {
                ans += Math.abs(i - j);
                j++;
            }
        }
        System.out.println(ans);
    }
}
