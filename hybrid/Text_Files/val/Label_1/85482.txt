#include <iostream>
#include <vector>
#include <algorithm>
int main()
{
    int n;
    std::cin >> n;
    std::vector<int> height(n);
    for (int i = 0; i < n; i++)
        std::cin >> height[i];
    std::sort(height.begin(), height.end());
    int sum = 0, ans = 0;
    for (int i = 0; i < n; i++)
    {
        sum += height[i];
        ans = std::max(ans, sum - height[i + 1]);
    }
    std::cout << ans << std::endl;
}
