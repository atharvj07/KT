#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  string s, t;
  cin >> n >> s >> t;
  if (s[0] == s[1] || s[0] == t[0] || s[0] == t[1] || s[1] == t[0] || s[1] == t[1])
    cout << "NO" << endl;
  else {
    string res = "";
    for (int i = 0; i < n; i++) res += "abc";
    cout << "YES" << endl
         << res << endl;
  }

}
