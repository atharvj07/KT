for _ in range(int(input())):
    n, m = map(int, input().split())
    a = [int(x) for x in input().split()]
    p = set([int(x) - 1 for x in input().split()])
    bad = False
    for i in range(n - 1):
        if (a[i] > a[i + 1] and i not in p) or (a[i] == a[i + 1] and i in p):
            bad = True
            break
    print("NO" if bad else "YES")
