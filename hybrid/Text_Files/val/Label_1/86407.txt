import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int s = scanner.nextInt();
        int l = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; ++i) {
            a[i] = scanner.nextInt();
        }
        Arrays.sort(a);
        int i = 0, j = 0, ans = 0;
        while (i < n) {
            while (j < n && a[j] <= a[i] + s) ++j;
            if (j - i >= l) {
                ++ans;
                i = j;
            } else {
                ++i;
            }
        }
        System.out.println(ans > 0 ? ans : -1);
    }
}
