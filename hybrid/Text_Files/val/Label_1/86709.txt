import java.util.*;

public class Main {

	int n, m, y1, y2, ans;
	int[] a, b;

	void solve() {
		n = in.nextInt();
		y1 = in.nextInt();
		a = new int[n];
		for (int i = 0; i < n; i++) {
			a[i] = in.nextInt();
		}
		m = in.nextInt();
		y2 = in.nextInt();
		b = new int[m];
		for (int i = 0; i < m; i++) {
			b[i] = in.nextInt();
		}
		if (n == 0 || m == 0) {
			System.out.println(0);
			return;
		}
		for (int ai : a) {
			for (int bi : b) {
				if (ai == bi) {
					ans += 2;
					break;
				}
				if ((ai < bi && (y2 - y1) / 2 < y2 - bi) || (ai > bi && (y2 - y1) / 2 < y2 - y1 - ai)) {
					ans += 1;
				}
			}
		}
		System.out.println(ans);
	}

	Scanner in = new Scanner(System.in);

	public static void main(String[] args) {
		new Main().solve();
	}
}
