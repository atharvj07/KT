import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] a = new int[2], b = new int[2], c = new int[2];
        for (int i = 0; i < 2; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
            c[i] = sc.nextInt();
        }
        boolean ans = false;
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 4; j++) {
                int[] tmp = new int[]{a[0] - c[0], a[1] - c[1]};
                if (tmp[0] == b[0] && tmp[1] == b[1]) ans = true;
                for (int k = 0; k < i; k++) tmp = rotate(tmp);
                for (int k = 0; k < j; k++) tmp = add(tmp, c);
                if (tmp[0] == b[0] && tmp[1] == b[1]) ans = true;
            }
            a = rotate(a);
        }
        System.out.println(ans ? "YES" : "NO");
    }

    private static int[] rotate(int[] v) {
        return new int[]{-v[1], v[0]};
    }

    private static int[] add(int[] v1, int[] v2) {
        return new int[]{v1[0] + v2[0], v1[1] + v2[1]};
    }
}
