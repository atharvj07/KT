#include <iostream>
#include <string>

using namespace std;

int main() {
  string field[10];
  for (int i = 0; i < 10; i++) { cin >> field[i]; }
  for (int i = 0; i < 10; i++) {
    for (int j = 0; j < 10; j++) {
      if (field[i][j] == '.') {
        for (int di = -1; di <= 1; di++) {
          for (int dj = -1; dj <= 1; dj++) {
            if (di == 0 && dj == 0) continue;
            int cnt = 1;
            for (int k = 1; k < 5; k++) {
              int ni = i + k * di;
              int nj = j + k * dj;
              if (ni < 0 || ni >= 10 || nj < 0 || nj >= 10 ||
                  field[ni][nj] != 'X') {
                break;
              }
              cnt++;
            }
            if (cnt == 5) {
              cout << "YES" << endl;
              return 0;
            }
          }
        }
      }
    }
  }
  cout << "NO" << endl;
  return 0;
}
