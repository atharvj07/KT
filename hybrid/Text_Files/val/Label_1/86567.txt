import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int m = in.nextInt();
    int[] monsters = new int[n];
    int[] specialCells = new int[m];
    for (int i = 0; i < n; i++) {
      monsters[i] = in.nextInt();
    }
    for (int i = 0; i < m; i++) {
      specialCells[i] = in.nextInt();
    }
    Arrays.sort(monsters);
    Arrays.sort(specialCells);
    int ans = 0;
    int i = 0;
    int j = 0;
    while (i < n && j < m) {
      if (monsters[i] <= specialCells[j]) {
        ans++;
        i++;
        j++;
      } else {
        j++;
      }
    }
    System.out.println(ans);
  }
}
