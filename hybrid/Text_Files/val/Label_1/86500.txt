import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<Integer, Integer> bars = new HashMap<>();
        for (int i = 0; i < n; i++) {
            int l = sc.nextInt();
            bars.put(l, bars.getOrDefault(l, 0) + 1);
        }
        int maxHeight = 0, numTowers = 0;
        for (Map.Entry<Integer, Integer> entry : bars.entrySet()) {
            int height = entry.getValue();
            maxHeight = Math.max(maxHeight, height);
            numTowers += (height + maxHeight - 1) / maxHeight;
        }
        System.out.println(maxHeight + " " + numTowers);
    }
}
