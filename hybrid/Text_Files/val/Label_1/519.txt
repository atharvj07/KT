t = int(input())
for _ in range(t):
    a, b = map(int, input().split())
    diff = abs(a - b)
    if diff == 0:
        print(0)
    elif diff % 2 == 0:
        if a > b:
            print(1)
        else:
            print(2)
    else:
        if a > b:
            print(2)
        else:
            print(1)
