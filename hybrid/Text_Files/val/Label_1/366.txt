import sys
N = int(sys.stdin.readline())
B = list(sorted(map(int,sys.stdin.readline().split())))
odd = []
even = []
for x in B:
    if x%2:
        odd.append(x)
    else:
        even.append(x)
if len(odd)==len(even):
    print(0)
else:
    if len(odd)>len(even):
        odd.pop()
    else:
        even.pop()
    print(1)
    print(odd[0])
