import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int r = sc.nextInt();
            if (n == 0 && r == 0) {
                break;
            }
            int[] cards = new int[n + 1];
            for (int i = 1; i <= n; i++) {
                cards[i] = i;
            }
            for (int i = 0; i < r; i++) {
                int p = sc.nextInt();
                int c = sc.nextInt();
                int[] temp = new int[c];
                for (int j = 0; j < c; j++) {
                    temp[j] = cards[p + j];
                }
                for (int j = p + c; j < cards.length; j++) {
                    cards[j - c] = cards[j];
                }
                for (int j = 0; j < temp.length; j++) {
                    cards[cards.length - 1 - j] = temp[j];
                }
            }
            System.out.println(cards[cards.length - 1]);
        }
        sc.close();
    }
}
