#include <cstdio>
#include <cstdlib>
char a[12][12];
int b[12][12], c[12][12];
int main() {
  bool f = 0;
  for (int i = 1; i <= 10; i++) {
    scanf("%s", &a[i][1]);
    for (int j = 1; j <= 10; j++) {
      if (a[i][j] == 'X')
        b[i][j] = 1;
      else if (a[i][j] == 'O')
        c[i][j] = 1;
      else
        f = 1;
    }
  }
  if (!f) {
    printf("NO");
    return 0;
  }
  for (int i = 1; i <= 10; i++) {
    for (int j = 1; j <= 10; j++) {
      if (a[i][j] == '.')
        if (b[i][j - 4] + b[i][j - 3] + b[i][j - 2] + b[i][j - 1] + b[i][j] >=
                4 ||
            c[i][j - 4] + c[i][j - 3] + c[i][j - 2] + c[i][j - 1] + c[i][j] >=
                4 ||
            b[i - 4][j] + b[i - 3][j] + b[i - 2][j] + b[i - 1][j] + b[i][j] >=
                4 ||
            c[i - 4][j] + c[i - 3][j] + c[i - 2][j] + c[i - 1][j] + c[i][j] >=
                4 ||
            b[i - 4][j - 4] + b[i - 3][j - 3] + b[i - 2][j - 2] +
                    b[i - 1][j - 1] + b[i][j] >=
                4 ||
            c[i - 4][j - 4] + c[i - 3][j - 3] + c[i - 2][j - 2] +
                    c[i - 1][j - 1] + c[i][j] >=
                4 ||
            b[i - 4][j + 4] + b[i - 3][j + 3] + b[i - 2][j + 2] +
                    b[i - 1][j + 1] + b[i][j] >=
                4 ||
            c[i - 4][j + 4] + c[i - 3][j + 3] + c[i - 2][j + 2] +
                    c[i - 1][j + 1] + c[i][j] >=
                4) {
          printf("YES");
          return 0;
        }
    }
  }
  printf("NO");
  return 0;
}
