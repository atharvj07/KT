import sys

t = int(input())

for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    
    dp = [[0] * 2 for _ in range(n+1)]
    
    for i in range(n):
        for j in range(2):
            dp[i+1][j] = max(dp[i+1][j], dp[i][j])
            
            if (j == 0 and a[i] > 0) or (j == 1 and a[i] < 0):
                dp[i+1][j] = max(dp[i+1][j], dp[i][1-j]+a[i])
                
    print(max(dp[n]))
