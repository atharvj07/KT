import sys
input = sys.stdin.readline

n = int(input())
s = list(input().strip())
t = [''] * n
ans = 0

last = {}
for i in range(n):
    if s[i] not in last:
        last[s[i]] = -3
    if (i - last[s[i]]) % 3 == 0:
        t[i] = s[i]
        last[s[i]] = i
    else:
        ans += 1
        t[i] = 'R' if s[i] != 'R' else 'G' if s[i] != 'G' else 'B'
        last[t[i]] = i

print(ans)
print(''.join(t))
