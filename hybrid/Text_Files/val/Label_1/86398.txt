import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.Comparator;
import java.util.LinkedList;
import java.util.List;
import java.util.PriorityQueue;
import java.util.Queue;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] nk = br.readLine().split(" ");
        int n = Integer.parseInt(nk[0]);
        int k = Integer.parseInt(nk[1]);
        List<Book> books = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String[] tab = br.readLine().split(" ");
            books.add(new Book(Integer.parseInt(tab[0]), Integer.parseInt(tab[1]), Integer.parseInt(tab[2])));
        }
        long ans = solve(books, k);
        System.out.println(ans);
    }

    private static long solve(List<Book> books, int k) {
        Collections.sort(books, new Comparator<Book>() {
            @Override
            public int compare(Book o1, Book o2) {
                return o1.likes - o2.likes;
            }
        });
        int likesAlice = 0;
        int likesBob = 0;
        long ans = 0;
        int i = books.size() - 1;
        while (i >= 0 && (likesAlice < k || likesBob < k)) {
            Book book = books.get(i);
            if (book.likes > 0) {
                likesAlice++;
                ans += book.time;
            }
            if (book.likes == 2) {
                likesBob++;
                ans += book.time;
            }
            i--;
        }
        if (likesAlice >= k && likesBob >= k) {
            return ans;
        } else {
            return -1;
        }
    }

    private static class Book {
        int time;
        int likes;
        int likesAlice;

        public Book(int time, int likesAlice, int likesBob) {
            this.time = time;
            this.likesAlice = likesAlice;
            this.likes = likesAlice + likesBob;
        }
    }
}
