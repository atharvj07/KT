import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int t = sc.nextInt();
    while (t-- > 0) {
      long a = sc.nextLong();
      long b = sc.nextLong();
      long c = sc.nextLong();
      long d = sc.nextLong();
      if (b - a > d - c) {
        System.out.println((a + b) / 2 + " " + b + " " + (c + d) / 2);
      } else {
        System.out.println((a + b) / 2 + " " + (b + c) / 2 + " " + (c + d) / 2);
      }
    }
  }
}
