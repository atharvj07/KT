#include<bits/stdc++.h>
using namespace std;
typedef pair<int, int> par;
typedef pair<par, int> ppar;

struct cmp
{
  bool operator() (const ppar& a, const ppar& b)
  {
    return a.second < b.second;
  }
};

int main()
{
  int n;
  cin>>n;
  vector<int> adj[n+1];
  vector<int> w[n+1];
  vector<int> x(n+1);
  int a, b, c;
  for(int i=1;i<n;i++)
  {
    cin>>a>>b>>c;
    adj[a].push_back(b);
    w[a].push_back(c);
    adj[b].push_back(a);
    w[b].push_back(c);
  }

  for(int i=1;i<=n;i++)
  {
    cin>>x[i];
  }

  priority_queue<ppar, vector<ppar>, cmp> pq;
  int ans=0;

  for(int i=1;i<=n;i++)
  {
    pq.push(make_pair(make_pair(i,i), 0));
  }

  while(!pq.empty())
  {
    ppar p = pq.top(); pq.pop();
    int u=p.first.first, v=p.first.second, wei=p.second;

    if(x[u]<=0) continue;
    ans=max(wei, ans);
    x[u]--;

    for(int i=0;i<adj[u].size();i++)
    {
      int child=adj[u][i];
      if(child==v) continue;
      pq.push(make_pair(make_pair(child, u), max(w[u][i], wei)));
    }
  }

  cout<<ans<<endl;

  return 0;
}
