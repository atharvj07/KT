import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long x = input.nextLong();
        long y = input.nextLong();
        if (x == y) {
            System.out.println("=");
        } else if (x < y) {
            long temp = x;
            x = y;
            y = temp;
        }

        if (x % y == 0) {
            if (x / y > 1) {
                System.out.println("<");
            } else {
                System.out.println("=");
            }
        } else {
            while (x % y != 0) {
                long temp = x % y;
                x = y;
                y = temp;
            }
            if (x / y > 1) {
                System.out.println("<");
            } else {
                System.out.println(">");
            }
        }
    }
}
