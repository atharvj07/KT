#include <iostream>
#include <deque>
using namespace std;
int main() {
    int q;
    cin >> q;
    deque<int> a;
    while (q--) {
        char c;
        int x;
        cin >> c >> x;
        if (c == 'L') {
            a.push_front(x);
        } else if (c == 'R') {
            a.push_back(x);
        } else {
            int ans = min(abs(a.front() - x), abs(a.back() - x));
            cout << ans << '\n';
        }
    }
    return 0;
}
