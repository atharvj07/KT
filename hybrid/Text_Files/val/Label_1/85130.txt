#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll T[1001], x[1001], y[1001];
int main() {
  ll n;
  cin >> n;
  for (ll i = 1; i <= n; i++) cin >> x[i] >> y[i];
  for (ll i = 1; i <= n; i++) {
    ll ai, bi;
    cin >> ai >> bi;
    T[i] = abs(ai) + abs(bi);
  }
  ll t = *max_element(T + 1, T + n + 1);
  for (ll i = 1; i <= n; i++) {
    if (T[i] + T[i + 1] == t)
      cout << x[i] - 2 * (x[i] - x[i + 1]) << " " << y[i] - 2 * (y[i] - y[i + 1]);
  }
  return 0;
}
