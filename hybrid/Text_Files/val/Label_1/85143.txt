#include<bits/stdc++.h>
using namespace std;
const int mxN = 2e5+5;
int32_t main() {
    int n, m; cin >> n >> m;
    string s; cin >> s;
    unordered_map<char, int> freq;
    for( int i = 0; i < n; ++i ) ++freq[s[i]];
    for( int i = 0; i < m; ++i ) {
        int x, y, len;
        cin >> x >> y >> len; --x, --y;
        string a = s.substr(x, len), b = s.substr(y, len);
        unordered_map<char, int> freq1;
        for( int j = 0; j < len; ++j ) ++freq1[a[j]];
        bool no = 0;
        for( int j = 0; j < len; ++j ) {
            if( freq[a[j]] != freq1[b[j]] ) no = 1;
        }
        if( no ) cout << "NO" << endl;
        else cout << "YES" << endl; 
    }
    return 0;
}
