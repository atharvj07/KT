#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        int n;
        cin >> n;
        set <int> p;
        for (int i = 1; i <= n; i++)
        {
            int x;
            cin >> x;
            p.insert(x);
        }
        cout << p.size() << endl;
    }
}
