#include <iostream>

using namespace std;

bool isValid(int a, int b, int c) {
  return (a + b > c) && (a + c > b) && (b + c > a);
}

int main() {
  int x, y;
  cin >> x >> y;

  int ans = 0;
  while (x != y) {
    if (x > y) {
      int diff = x - y;
      if (diff % 3 == 0) x -= diff / 3;
      else x -= diff % 3;
    } else {
      int diff = y - x;
      if (diff % 2 == 0) x += diff / 2;
      else x += diff / 2 + 1;
    }
    ans++;
  }

  cout << ans << endl;

  return 0;
}
