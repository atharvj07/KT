#include <bits/stdc++.h>
#include <vector>
using namespace std;
int c=1e9,n;
vector<int>a,b,swapped;
int main() {
cin>>n; a=vector<int>(n+1); b=a; b[0]=0;
for (int i=1; i<=n; i++) cin>>a[i], b[a[i]]=i;
for (int i=1; i<=n; i++)
if (a[i]!=i&&b[i]!=i){
swapped.push_back(i); swapped.push_back(b[i]); c=min(c,min(i,b[i]));}
c=n/2-c, c=max(c,0);
cout<<(c?c-1:0)<<endl;
for (int i=0; i<c; i++) cout<<swapped[i]<<' '<<swapped[i+1]<<endl;
return 0; }
