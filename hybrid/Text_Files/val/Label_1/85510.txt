#include <algorithm>
#include <cmath>
#include <iostream>
#include <vector>
using namespace std;

int n;
vector<int> m, r;

int main() {
  cin >> n;
  m.resize(n);
  r.resize(n);
  for (int i = 0; i < n; i++) cin >> m[i];
  for (int i = 0; i < n; i++) cin >> r[i];
  double res = 0;
  for (int i = 0; i < n; i++) {
    if (r[i] == 0) res = 1;
    int t = 1;
    while (1) {
      if (t % m[i] == r[i]) { res += 1.0 / t; break; }
      t++;
    }
  }
  res /= n;
  cout.precision(6);
  cout << fixed << res << endl;
  return 0;
}
