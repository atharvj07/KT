import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            int[] arr = new int[4];
            for (int i = 0; i < 4; i++) {
                arr[i] = input.nextInt();
            }
            System.out.println((arr[0] + arr[1] + 1) / 2 + " " + (arr[1] + arr[2] + 1) / 2 + " " + (arr[2] + arr[3] + 1) / 2);
        }
    }
}
