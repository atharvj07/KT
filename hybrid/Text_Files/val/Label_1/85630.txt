#include <bits/stdc++.h>
using namespace std;
#define int long long
const int maxn = 3e5 + 5;
int n, m, x[maxn], p[maxn], ans;
bool ok;
int32_t main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> x[i];
    for (int i = 1; i <= m; i++) cin >> p[i];
    x[n + 1] = (1LL << 60);
    for (int i = 1; i <= m; i++) {
        int j = 1, last = 0;
        while (x[j] < p[i]) j++;
        if (j > 1) {
            if (x[j] % p[i] == 0) last = x[j] / p[i];
        }
        if (last > 0 && last <= x[j] - p[i]) {
            int y = x[j] - last * p[i];
            if (y <= 0) continue;
            bool ok = true;
            for (int j = 1; j <= n; j++) {
                int pos = lower_bound(x + 1, x + n + 2, y + (x[j] - y) / p[i] * p[i]) - x;
                if (pos > n || x[pos] != x[j]) {
                    ok = false;
                    break;
                }
            }
            if (ok) {
                ok = true;
                ans = y;
                cout << "YES" << endl;
                cout << ans << " " << i << endl;
                return 0;
            }
        }
    }
    if (!ok) cout << "NO" << endl;
}
