import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    String s = sc.nextLine();
    System.out.println(s.equals(new StringBuilder(s).reverse().toString().replace('b', 'd').replace('d', 'b').replace('p', 'q').replace('q', 'p').toString()) ? "Yes" : "No");
  }
}
