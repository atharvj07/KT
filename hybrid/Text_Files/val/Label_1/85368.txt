#include <bits/stdc++.h>
using namespace std;
typedef long long ll; constexpr int mod = 1e9 + 7; constexpr char nl = '\n';

int main() {
    int n, k; cin >> n >> k;
    string ans(2*n-1, 'z');
    vector<string> g(n);
    for (auto& s : g) cin >> s;
    auto change = [&](char& c, char d) {
        if (c > d && k) {
            c = d; k--;
            return true;
        }
        return false;
    };
    function<void(int,int,string)> dfs = [&](int i, int j, string cur) {
        if (i == n) { ans = min(ans, cur); return; }
        if (j == n) { dfs(i+1, 0, cur); return; }
        if (change(ans[i+j], g[i][j])) dfs(i, j+1, cur+g[i][j]);
        if (change(ans[i+j], g[i][j])) dfs(i+1, j+1, cur+g[i][j]);
        dfs(i, j+1, cur+g[i][j]);
    };
    dfs(0, 0, "");
    cout << ans << nl;
}
