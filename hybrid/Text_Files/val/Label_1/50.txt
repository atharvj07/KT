import sys
from collections import defaultdict

inf = sys.maxsize

def check(st):
	ans = defaultdict(int)
	for i in range(len(st)):
		ans[st[i]] ^= 1
	cnt = 0
	for i in ans:
		if ans[i] == 1: cnt += 1
	if cnt <= 1: return "Yes"
	else: return "No"

def main():
	t = int(input())
	for i in range(t):
		n = int(input())
		st = input()
		tt = input()
		x = check(st + tt)
		print(x)
		if x == "Yes":
			st += tt
			ans = []
			for i in range(n):
				if st[i] == st[n + i]: continue
				for j in range(n, n + i):
					if st[i] == st[j]:
						ans.append([i + 1, j - n + 1])
						st[i], st[j] = st[j], st[i]
						break
			print(len(ans))
			for cr in ans: print(*cr)
	return

main()
