#include <iostream>
using namespace std;

int main() {
  int w, h;
  cin >> w >> h;
  char a[h][w];
  for (int i = 0; i < h; i++) cin >> a[i];

  for (int i = 0; i < 2 * h; i++) {
    for (int j = w - 1; j >= 0; j--) {
      for (int k = 0; k < 2; k++) cout << a[i / 2][j];
    }
    cout << endl;
  }
}
