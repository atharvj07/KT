#include<bits/stdc++.h>
using namespace std;
int main(){
	int n,k,ans=0;
	cin>>n>>k;
	int a[n];
	for(int i=0;i<n;i++) cin>>a[i];
	for(int i=0;i<n-1;i++){
		if(a[i]+1-a[i+1]>k) ans+=a[i]+1-a[i+1]-k,a[i+1]=a[i]+1-k;
		if(a[i]+1-a[i+1]<k) ans+=k-(a[i]+1-a[i+1]),a[i]=k-a[i+1]+a[i+1];
	}
	cout<<ans<<endl;
	for(int i=0;i<n-1;i++){
		if(a[i]+1-a[i+1]>k) cout<<"+ "<<i+1<<" "<<a[i]+1-a[i+1]-k<<endl;
		if(a[i]+1-a[i+1]<k) cout<<"- "<<i+1<<" "<<k-(a[i]+1-a[i+1])<<endl;
	}
}
