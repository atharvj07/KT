import sys
sys.setrecursionlimit(1<<20)

def build(a, tree, node, start, end):
    if start == end:
        tree[node] = a[start]
        return tree[node]
    mid = (start + end) >> 1
    tree[node] = build(a, tree, 2 * node + 1, start, mid) ^ build(a, tree, 2 * node + 2, mid + 1, end)
    return tree[node]

def query(tree, node, start, end, l, r):
    if l <= start and end <= r:
        return tree[node]
    elif r < start or l > end:
        return 0
    mid = (start + end) >> 1
    return query(tree, 2 * node + 1, start, mid, l, r) ^ query(tree, 2 * node + 2, mid + 1, end, l, r)

n = int(input())
a = list(map(int, input().split()))
q = int(input())
tree = [0] * 4 * n
build(a, tree, 0, 0, n - 1)
for _ in range(q):
    l, r = map(int, input().split())
    l -= 1
    r -= 1
    print(query(tree, 0, 0, n - 1, l, r))
