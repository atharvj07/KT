#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &e : a) cin >> e;
  sort(a.begin(), a.end());
  int mx = 0;
  int s1 = 0, s2 = n - 1;
  while (s1 <= s2) {
    mx = max(mx, s2 - s1 + 1);
    while (s1 < s2 && a[s1] == a[s1 + 1]) s1++;
    while (s1 < s2 && a[s2] == a[s2 - 1]) s2--;
    s1++, s2--;
  }
  cout << mx << endl;
}
