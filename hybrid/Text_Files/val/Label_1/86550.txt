import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        if(m % 2 == 1 || n % 3 != 0) {
            System.out.println("NO");
            return;
        }
        List<Integer>[] adj = new List[n + 1];
        for(int i = 0; i <= n; ++i) {
            adj[i] = new ArrayList<>();
        }
        for(int i = 0; i < m; ++i) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a].add(b);
            adj[b].add(a);
        }
        int[] dep = new int[n + 1];
        int cur = 1;
        for(int i = 1; i <= n; ++i) {
            if(dep[i] == 0) {
                if(!dfs(i, cur, dep, adj)) {
                    System.out.println("NO");
                    return;
                }
                cur++;
            }
        }
        System.out.println("YES");
        for(int i = 1; i <= n; ++i) {
            System.out.print(dep[i] + " ");
        }
    }

    static boolean dfs(int u, int d, int[] dep, List<Integer>[] adj) {
        dep[u] = d;
        int cnt = 0;
        for(int v : adj[u]) {
            if(dep[v] == d) return false;
            if(dep[v] == 0) {
                cnt++;
                if(cnt > 1) return false;
                if(!dfs(v, d + 1, dep, adj)) return false;
            }
        }
        return true;
    }
}
