#include <cstdio>

int main() {
  int n;
  scanf("%d", &n);
  int a[1440];
  for (int i = 0; i < 1440; i++) { a[i] = 0; }
  for (int i = 0; i < n; i++) {
    int h, m;
    scanf("%d %d", &h, &m);
    a[h * 60 + m]++;
  }
  int ans = 0, cur = 0;
  for (int i = 0; i < 1440; i++) {
    cur -= a[i];
    if (cur < 0) {
      cur = 0;
      ans++;
    }
    cur += a[i + 1];
  }
  printf("%d\n", ans);
  return 0;
}
