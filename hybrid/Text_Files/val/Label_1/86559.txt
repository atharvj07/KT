import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[][] queries = new int[q][2];
        for (int i = 0; i < q; i++) {
            queries[i][0] = sc.nextInt() - 1;
            queries[i][1] = sc.nextInt() - 1;
        }
        int[] dp = new int[n];
        Arrays.fill(dp, -1);
        for (int i = 0; i < q; i++) {
            if (dp[queries[i][0]] == -1) {
                dp[queries[i][0]] = isReachable(a, queries[i][0], queries[i][1]) ? 1 : 0;
            }
            System.out.println(dp[queries[i][0]] == 1 ? "Shi" : "Fou");
        }
    }

    private static boolean isReachable(int[] a, int x, int y) {
        if (x == y) {
            return true;
        }
        if ((a[x] & a[x + 1]) == 0) {
            return false;
        }
        return isReachable(a, x + 1, y);
    }
}
