#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll inf = 1e18, mod = 1e9 + 7;
const int N = 1e5 + 10;
ll n, p, x, g;
map<ll, ll> A;
int main() {
  cin.tie(0)->sync_with_stdio(0);
  cin >> n;
  while (n--) {
    cin >> x;
    p = 0;
    for (ll i = 2; i * i <= x; ++i) {
      if (x % i == 0) {
        while (x % i == 0) x /= i, ++p;
        A[i] = max(A[i], p);
      }
    }
    if (x > 1) A[x] = max(A[x], p = 1);
  }
  g = 1;
  for (auto &it : A) {
    g *= (1ll << it.second);
  }
  cout << g << '\n';
  return 0;
}
