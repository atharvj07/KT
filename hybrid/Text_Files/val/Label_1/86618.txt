import java.util.Scanner;
class Main {
    static int mod = 1000000007;
    static long[][] comb = new long[1002][1002];
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) a[i] = sc.nextInt();
        long ans = 1;
        for (int i = 0; i < k; i++) {
            ans = ans * comb(n, a[i]) % mod;
        }
        for (int i = 1; i <= n; i++) {
            long p = 1;
            for (int j = 0; j < k; j++) {
                p = p * (i - a[j]) % mod;
            }
            ans = ans * p % mod;
        }
        ans = (ans + mod) % mod;
        System.out.println(ans);
    }
    static long comb(int n, int r) {
        if (n < r) return 0;
        if (comb[n][r] > 0) return comb[n][r];
        if (r == 0 || r == n) return comb[n][r] = 1;
        return comb[n][r] = (comb(n - 1, r - 1) + comb(n - 1, r)) % mod;
    }
}
