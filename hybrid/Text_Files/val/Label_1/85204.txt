#include <bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int m = -1;
    bool f = 1;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        if (__gcd(a[i], a[j]) != __gcd(a[0], a[1])) {
          f = 0;
          break;
        }
        m = __gcd(a[i], a[j]);
      }
      if (f == 0) break;
    }
    int g = __gcd(a[0], m);
    if (g > 1) {
      cout << 0 << endl;
      continue;
    }
    if (m == -1 || f == 0) {
      cout << -1 << endl;
    } else {
      int c = (a[1] - a[0] + m) % m;
      cout << m << " " << c << endl;
    }
  }
  return 0;
}
