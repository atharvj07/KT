#include <bits/stdc++.h>
using namespace std;
#define nl '\n'
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;

void solve() {
    int n; cin >> n;
    string l, r; cin >> l >> r;
    vector<int> l_q, r_q;
    vector<pii> ans;
    for (int i = 0; i < n; i++) {
        if (l[i] == '?') l_q.push_back(i);
        if (r[i] == '?') r_q.push_back(i);
    }
    for (int i = 0; i < l_q.size(); i++) {
        for (int j = 0; j < r_q.size(); j++) {
            ans.push_back({l_q[i], r_q[j]});
        }
    }
    for (int i = 0; i < n; i++) {
        if (l[i] != '?' && r[i] != '?') {
            if (l[i] == r[i]) {
                ans.push_back({i, i});
            }
        }
    }
    cout << ans.size() << nl;
    for (pii p : ans) {
        cout << p.first + 1 << ' ' << p.second + 1 << nl;
    }
}

int main() {
    solve();
}
