import java.util.ArrayList;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class NetworkPacketRouting {

    private static int[][] connections;
    private static int[] shortestPath;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        connections = new int[n + 1][n + 1];

        for (int i = 1; i <= n; i++) {
            int ri = scanner.nextInt();
            int ki = scanner.nextInt();
            for (int j = 0; j < ki; j++) {
                int tj = scanner.nextInt();
                connections[ri][tj] = 1;
            }
        }

        int p = scanner.nextInt();
        for (int i = 0; i < p; i++) {
            int si = scanner.nextInt();
            int di = scanner.nextInt();
            int vi = scanner.nextInt();

            shortestPath = new int[n + 1];
            for (int j = 1; j <= n; j++) {
                shortestPath[j] = -1;
            }

            if (bfs(si, di, vi)) {
                System.out.println(shortestPath[di]);
            } else {
                System.out.println("NA");
            }
        }
    }

    private static boolean bfs(int start, int destination, int ttl) {
        Queue<Integer> queue = new LinkedList<>();
        queue.offer(start);
        shortestPath[start] = 0;

        while (!queue.isEmpty()) {
            int current = queue.poll();
            if (current == destination) {
                return true;
            }

            for (int next = 1; next <= connections.length - 1; next++) {
                if (connections[current][next] == 1 && shortestPath[next] == -1) {
                    shortestPath[next] = shortestPath[current] + 1;
                    if (shortestPath[next] < ttl) {
                        queue.offer(next);
                    }
                }
            }
        }

        return false;
    }
}
