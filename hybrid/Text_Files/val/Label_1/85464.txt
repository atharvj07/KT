#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x, lst = 0;
        cin >> n;
        set<int> s;
        for (int i = 1, a; i <= n; i++) {
            cin >> a;
            if (a == x) lst = i;
            s.insert(a);
        }
        int sz = s.size();
        if (sz > 2)
            cout << "NO" << endl;
        else {
            if (sz == 1)
                cout << "YES" << endl;
            else if (s.count(lst) == 0)
                cout << "YES" << endl;
            else if (lst - 1 < n - lst)
                cout << "YES" << endl;
            else
                cout << "NO" << endl;
        }
    }
}
