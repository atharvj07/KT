#include <iostream>
#include <vector>

const int N = 10;
char a[N][N];

bool checkRow(int i, int j) {
  int cnt = 1;
  for (int k = j + 1; k < N && a[i][k] == 'X'; ++k) {
    ++cnt;
  }
  for (int k = j - 1; k >= 0 && a[i][k] == 'X'; --k) {
    ++cnt;
  }
  return cnt >= 5;
}

bool checkCol(int i, int j) {
  int cnt = 1;
  for (int k = i + 1; k < N && a[k][j] == 'X'; ++k) {
    ++cnt;
  }
  for (int k = i - 1; k >= 0 && a[k][j] == 'X'; --k) {
    ++cnt;
  }
  return cnt >= 5;
}

bool checkDiag1(int i, int j) {
  int cnt = 1;
  int x = i, y = j;
  while (x >= 0 && y >= 0 && a[x][y] == 'X') {
    cnt++, x--, y--;
  }
  x = i, y = j;
  while (x < N && y < N && a[x][y] == 'X') {
    cnt++, x++, y++;
  }
  return cnt >= 5;
}

bool checkDiag2(int i, int j) {
  int cnt = 1;
  int x = i, y = j;
  while (x >= 0 && y < N && a[x][y] == 'X') {
    cnt++, x--, y++;
  }
  x = i, y = j;
  while (x < N && y >= 0 && a[x][y] == 'X') {
    cnt++, x++, y--;
  }
  return cnt >= 5;
}

int main() {
  for (int i = 0; i < N; ++i) {
    for (int j = 0; j < N; ++j) {
      std::cin >> a[i][j];
    }
  }

  bool win = false;
  for (int i = 0; i < N && !win; ++i) {
    for (int j = 0; j < N && !win; ++j) {
      if (a[i][j] == '.') {
        a[i][j] = 'X';
        win |= checkRow(i, j) || checkCol(i, j) || checkDiag1(i, j) ||
               checkDiag2(i, j);
        a[i][j] = '.';
      }
    }
  }

  std::cout << (win ? "YES" : "NO") << '\n';
}
