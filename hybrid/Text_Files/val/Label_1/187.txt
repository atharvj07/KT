import sys
input = sys.stdin.readline

for _ in range(int(input())):
    (n, c) = map(int, input().split())
    w = list(map(int, input().split()))
    w.sort(reverse=True)
    ans = 0
    i = 0
    while i < n and ans + w[i] <= c:
        ans += w[i]
        i += 1
    if ans >= c / 2:
        print(i)
        for j in range(i):
            print(j + 1, end=' ')
        print()
    else:
        print(-1)
