#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.begin(), a.end());
  int ans = 0;
  int rem = m * k;
  for (int i = 0; i < n; i++) {
    if (a[i] > rem) break;
    rem -= a[i];
    ans++;
  }
  cout << ans << endl;
  return 0;
}
