import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        char[] s = sc.next().toCharArray();
        int m = sc.nextInt();
        for (int i = 0; i < m; i++) {
            int k = sc.nextInt();
            int d = sc.nextInt();
            for (int j = 0; j + k - 1 < s.length; j++) {
                for (int p = 0, q = j + d - 1; q < j + k; p += d, q += d) {
                    char tmp = s[j + p];
                    s[j + p] = s[q];
                    s[q] = tmp;
                }
            }
            System.out.println(new String(s));
        }
    }
}
