#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = 0;
    for (int i = 1; i < n; i++) {
        if (a[i] + 1 - a[i - 1] != k) {
            if (a[i] + 1 - k > a[i - 1]) {
                cout << '-' << ' ' << i << ' ' << (a[i] + 1 - k - a[i - 1]) << '\n';
                a[i] = a[i - 1] + k - 1;
                ans++;
            } else {
                cout << '+' << ' ' << i << ' ' << (k - (a[i - 1] - a[i])) << '\n';
                a[i] = a[i - 1] + k;
                ans++;
            }
        }
    }
    cout << ans << '\n';
    return 0;
}
