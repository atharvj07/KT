import java.util.*;
import java.io.*;

public class Main {

  public static void main(String[] args) throws IOException {
    BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st1 = new StringTokenizer(in.readLine());
    int n = Integer.parseInt(st1.nextToken());
    long y1 = Long.parseLong(st1.nextToken());
    StringTokenizer st2 = new StringTokenizer(in.readLine());
    long[] a = new long[n];
    for (int i = 0; i < n; i++) {
      a[i] = Long.parseLong(st2.nextToken());
    }
    StringTokenizer st3 = new StringTokenizer(in.readLine());
    int m = Integer.parseInt(st3.nextToken());
    long y2 = Long.parseLong(st3.nextToken());
    StringTokenizer st4 = new StringTokenizer(in.readLine());
    long[] b = new long[m];
    for (int i = 0; i < m; i++) {
      b[i] = Long.parseLong(st4.nextToken());
    }
    long ans = 0;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        long da = a[i] - b[j];
        if (da >= 0 && (da % (y2 - y1)) == 0) {
          ans = Math.max(ans, da / (y2 - y1) + 2);
        }
      }
    }
    System.out.println(ans);
  }
}
