import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        long y1 = scanner.nextLong();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int m = scanner.nextInt();
        long y2 = scanner.nextLong();
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = scanner.nextInt();
        }

        long x1 = a[0];
        long x2 = b[0];
        long dx = x2 - x1;
        long dy = y2 - y1;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            long x = a[i];
            if (dx * (x - x1) % dy == 0) {
                ans++;
            }
        }
        for (int i = 0; i < m; i++) {
            long x = b[i];
            if (dx * (x - x1) % dy == 0) {
                ans++;
            }
        }

        System.out.println(ans);
    }

}
