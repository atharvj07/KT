import sys

n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

if (a == [0 for i in range(n - 1)]) or (b == [0 for i in range(n - 1)]):
    print("YES")
    print("0" * n)
else:
    t = [0 for i in range(n)]
    for i in range(n - 1):
        if a[i] == 3 or b[i] == 3:
            print("NO")
            sys.exit()
        if a[i] == 0 and b[i] == 0:
            t[i] = 0
        elif a[i] == 0 and b[i] == 1:
            t[i] = 1
            t[i + 1] = 1
        elif a[i] == 0 and b[i] == 2:
            if i == 0:
                t[i] = 1
                t[i + 1] = 2
            else:
                print("NO")
                sys.exit()
        elif a[i] == 1 and b[i] == 0:
            t[i] = 1
        elif a[i] == 1 and b[i] == 1:
            t[i] = 1
        elif a[i] == 1 and b[i] == 2:
            t[i] = 3
        elif a[i] == 2 and b[i] == 0:
            if i == 0:
                t[i] = 2
                t[i + 1] = 2
            else:
                print("NO")
                sys.exit()
        elif a[i] == 2 and b[i] == 1:
            t[i] = 3
        elif a[i] == 2 and b[i] == 2:
            t[i] = 2
    print("YES")
    for i in t:
        print(i, end=" ")
