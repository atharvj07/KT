#include <iostream>
int main() {
  int n;
  std::cin >> n;
  int a[n], mi = 123456789, ma = 0, mini, maxi;
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
    if (a[i] == 1) {
      mi = i;
    }
    if (a[i] == n) {
      maxi = i;
      ma = i;
    }
  }
  if (maxi == 0) {
    std::cout << mi;
    return 0;
  }
  if (mi > maxi) {
    std::cout << n - 1 - mi;
  } else {
    std::cout << ma;
  }
}
