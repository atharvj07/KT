n = int(input())
phone_book = {}

for _ in range(n):
    data = input().split()
    name = data[0]
    numbers = set(data[2:])
    if name in phone_book:
        phone_book[name].update(numbers)
    else:
        phone_book[name] = numbers

for name in phone_book:
    numbers = list(phone_book[name])
    filtered_numbers = []
    for num in numbers:
        if not any(num != other and other.endswith(num) for other in numbers):
            filtered_numbers.append(num)
    phone_book[name] = filtered_numbers

print(len(phone_book))
for name, numbers in phone_book.items():
    print(name, len(numbers), ' '.join(numbers))
