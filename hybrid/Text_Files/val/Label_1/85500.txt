#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll n,a[100001],b[100001];
int main(){
    cin>>n;
    for(int i=1;i<=n;++i){
        cin>>a[i];
        b[a[i]]++;
    }
    for(int i=1;i<=100000;++i){
        if(b[i]%2==1){
            cout<<"Conan\n";
            return 0;
        }
    }
    cout<<"Agasa\n";
    return 0;
}
