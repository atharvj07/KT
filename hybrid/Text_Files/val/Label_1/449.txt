def check_sequence(n, a, b):
    for i in range(n - 1):
        if b[i] > a[i]:
            return "NO", []
        if (a[i] & b[i]) != b[i]:
            return "NO", []
    
    t = [0] * n
    for i in range(4):
        if (i | (i+1)%4) == a[0] and (i & (i+1)%4) == b[0]:
            t[0] = i
            break
    else:
        return "NO", []
    
    for i in range(1, n):
        for j in range(4):
            if (t[i-1] | j) == a[i-1] and (t[i-1] & j) == b[i-1]:
                t[i] = j
                break
        else:
            return "NO", []
    
    return "YES", t

n = int(input().strip())
a = list(map(int, input().strip().split()))
b = list(map(int, input().strip().split()))

result, sequence = check_sequence(n, a, b)
print(result)
if result == "YES":
    print(*sequence)
