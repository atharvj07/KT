def f(b):
    if len(b) == 1:
        return b[0]
    return f([b[i] ^ b[i+1] for i in range(len(b)-1)])

def solve(a):
    max_val = 0
    for i in range(len(a)):
        for j in range(i, len(a)):
            max_val = max(max_val, f(a[i:j+1]))
    return max_val

n = int(input().strip())
a = list(map(int, input().strip().split()))
q = int(input().strip())

for _ in range(q):
    l, r = map(int, input().strip().split())
    print(solve(a[l-1:r]))
