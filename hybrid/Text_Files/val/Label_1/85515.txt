#include <vector>
#include <string>
#include <iostream>
using namespace std;
int main() {
    int t, n, a, b;
    cin >> t;
    while (t--) {
        cin >> n >> a >> b;
        string s;
        if (a - b <= 2) {
            for (int i = 0; i * a < n; ++i) {
                for (char c = 'a' + i * b % 26; c <= 'z' && c - 'a' < b; ++c) {
                    s += c;
                }
            }
        } else {
            for (int i = 0; i * a < n; ++i) {
                for (char c = 'a' + i % 2; c <= 'z' && c - 'a' < b; ++c) {
                    s += c;
                }
            }
        }
        s.resize(n);
        cout << s << '\n';
    }
}
