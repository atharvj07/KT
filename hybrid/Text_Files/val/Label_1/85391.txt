#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod=1e9+7;
vector< vector<ll> > g;
vector<bool> vis;
ll n;
void dfs(ll node,ll pre,ll mood,ll prv,ll par,vector<ll> &path){
  if(vis[node]){
    path.push_back(node);
    return;
  }
  vis[node]=1;
  path.push_back(node);
  for(auto x:g[node]){
    if(x==pre){
      continue;
    }
    if(x==par){
      dfs(x,node,mood,prv,par,path);
      return;
    }
    dfs(x,node,mood^1,prv,par,path);
  }
  if(prv){
    path.pop_back();
  }
}
int main()
{
    cin>>n;
    g.resize(n+1);
    vis.resize(n+1,false);
    for(ll i=0;i<n-1;i++){
      ll x,y;
      cin>>x>>y;
      g[x].push_back(y);
      g[y].push_back(x);
    }
    ll ans=0;
    for(ll i=1;i<=n;i++){
      if(vis[i]){
        continue;
      }
      vector<ll> path;
      dfs(i,-1,0,0,0,path);
      ll sum=0;
      unordered_map<ll,ll> frq;
      for(auto x:path){
        frq[x]++;
      }
      for(auto x:path){
        if(frq[x]&1){
          sum++;
        }
      }
      ll way=(1ll<<sum);
      way%=mod;
      ans+=way;
      ans%=mod;
    }
    cout<<ans<<endl;
}
