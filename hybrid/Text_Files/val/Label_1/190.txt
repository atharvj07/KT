t = int(input().strip())
for _ in range(t):
    n, W = map(int, input().split())
    weights = list(map(int, input().split()))
    items = sorted(((w, i) for i, w in enumerate(weights, start=1)), reverse=True)

    total_weight = 0
    selected_items = []
    for weight, idx in items:
        if total_weight + weight <= W:
            total_weight += weight
            selected_items.append(idx)
        if total_weight >= -(-W // 2):
            break

    if total_weight >= -(-W // 2):
        print(len(selected_items))
        print(' '.join(map(str, selected_items)))
    else:
        print(-1)
