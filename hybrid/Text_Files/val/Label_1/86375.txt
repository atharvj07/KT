import java.util.Scanner;

public class TreePermutation {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt();
        }

        if (!isValidPermutation(n, p)) {
            System.out.println("NO");
            return;
        }

        System.out.println("YES");
        for (int i = 1; i < n; i++) {
            System.out.println(p[i] + " " + p[i - 1]);
        }
    }

    private static boolean isValidPermutation(int n, int[] p) {
        boolean[] visited = new boolean[n + 1];
        for (int i = 0; i < n; i++) {
            if (visited[p[i]]) {
                return false;
            }
            visited[p[i]] = true;
        }
        return true;
    }
}
