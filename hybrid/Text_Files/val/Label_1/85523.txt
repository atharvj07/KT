#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  int cur = 0, ans = 0;
  while (cur < n) {
    while (cur < n && a[cur] != a[0]) cur++;
    if(cur == n) break;
    ans += cur + 1;
    cur++;
  }
  cout << ans << endl;
}
