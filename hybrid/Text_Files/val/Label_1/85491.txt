#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
  int n;
  cin >> n;
  vector<ll> arr(n);
  for (int i = 0; i < n; i++) cin >> arr[i];

  vector<ll> dp(n, 0);
  dp[1] = arr[0] + max(arr[n - 1], arr[1]);
  dp[2] = arr[0] + arr[1] + max(arr[n - 2], arr[2]);

  for (int i = 3; i < n; i++) {
    dp[i] = max({
      dp[i - 2] + arr[i - 1] + max(arr[i - 3], arr[i]),
      dp[i - 1] + arr[i] + max(arr[i - 2], arr[0]),
    });
  }

  cout << dp[n - 1] << endl;
  return 0;
}
