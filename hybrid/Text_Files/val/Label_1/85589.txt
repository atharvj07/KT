#include<bits/stdc++.h>
using namespace std;
int main(){
  long long n;
  cin>>n;
  long long l=0,r=1e9,mid;
  while(l<=r){
      mid=(l+r)/2;
      long long t=mid,sm=0;
      while(t){
         sm+=t%10;
         t/=10;
      }
      if(mid*mid+mid*sm-n==0){
         break;
      }
      else if(mid*mid+mid*sm-n<0){
         l=mid+1;
      }
      else{
         r=mid-1;
      }
  }
  if(l>r){
     cout<<-1;
  }
  else{
     cout<<mid;
  }
  return 0;
}
