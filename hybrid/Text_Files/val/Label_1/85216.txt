#include <iostream>
#include <cstring>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    string s, t;
    cin >> s >> t;
    int a = 0, b = 0;
    for (int i = 0; i < n; i++)
        if (s[i] == 'a')
            a = i + 1;
        else
            break;
    for (int i = 0; i < n; i++)
        if (t[i] == 'b')
            b = i + 1;
        else
            break;
    if (a == 0 || b == 0)
        cout << 2 * (k + 1) - 1;
    else
        cout << 2 * (k + 1);
}
