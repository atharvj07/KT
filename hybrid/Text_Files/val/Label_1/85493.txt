#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, ans = 0;
  cin >> n;
  vector<long long> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i]; }
  for (int i = n - 1; i > 0; i -= 2) {
    long long sum = v[i] + v[i - 1];
    v[i] = sum;
    v.erase(v.begin() + i - 1);
  }
  cout << v[0] << endl;
  return 0;
}
