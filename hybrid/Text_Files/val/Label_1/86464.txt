import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]), p = Integer.parseInt(line[1]);

        // Read and store the coders' accusations
        int[][] accusations = new int[n][2];
        for (int i = 0; i < n; i++) {
            line = reader.readLine().split(" ");
            accusations[i][0] = Integer.parseInt(line[0]);
            accusations[i][1] = Integer.parseInt(line[1]);
        }

        // Sort the coders' accusations by the number of people they accused
        Arrays.sort(accusations, (a, b) -> a[0] + a[1] - (b[0] + b[1]));

        // Count the number of ways to choose two suspects
        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int agreed = 0;
                for (int k = 0; k < n; k++) {
                    if (accusations[k][0] == accusations[i][0] || accusations[k][0] == accusations[j][0] ||
                            accusations[k][1] == accusations[i][0] || accusations[k][1] == accusations[j][0]) {
                        agreed++;
                    }
                }
                if (agreed >= p) {
                    count++;
                }
            }
        }

        // Output the number of possible two-suspect sets
        System.out.println(count);
    }
}
