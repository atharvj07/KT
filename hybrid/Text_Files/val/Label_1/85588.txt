#include <iostream>
using namespace std;
int main() {
  long long n;
  cin >> n;
  long long l = 1, r = n;
  while (l <= r) {
    long long mid = (l + r) >> 1;
    long long sum = 0, x = mid;
    while (x > 0) {
      sum += x % 10;
      x /= 10;
    }
    if (mid * mid + mid * sum - n <= 0)
      r = mid - 1;
    else
      l = mid + 1;
  }
  if (r * r + r * (r + 1) - n == 0)
    cout << r;
  else
    cout << -1;
}
