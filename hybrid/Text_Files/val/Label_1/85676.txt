#include <iostream>
using namespace std;

int main() {
  string b[2], e[2];
  bool flag = 1;
  for (int i = 0; i < 2; i++) cin >> b[i];
  for (int i = 0; i < 2; i++) cin >> e[i];
  for (int i = 0; i < 2; i++) {
    for (int j = 0; j < 2; j++) {
      if (b[i][j] != e[i][j] && b[i][j] != 'X' && e[i][j] != 'X')
        flag = 0;
    }
  }
  cout << (flag ? "YES\n" : "NO\n");
}
