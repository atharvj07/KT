#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int mn = n, cnt = 0;
  for (int i = 0; i < n - 1; i++) {
    if (a[i] > a[i + 1]) {
      mn = min(mn, a[i]);
      cnt++;
    }
  }
  cout << mn << ' ' << cnt << '\n';
  return 0;
}
