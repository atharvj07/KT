def min_coins_to_one(n, k, A, B):
    if k == 1:
        return (n - 1) * A
    coins = 0
    while n > 1:
        if n % k == 0:
            n //= k
            coins += min(B, (n * k - n) * A)
        else:
            diff = n - n // k * k
            n -= diff
            coins += diff * A
            if n == 0:
                n += diff
                coins -= diff * A
                break
    return coins

n = int(input().strip())
k = int(input().strip())
A = int(input().strip())
B = int(input().strip())

print(min_coins_to_one(n, k, A, B))
