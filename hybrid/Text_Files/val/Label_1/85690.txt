#include <limits.h>
#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, ans = INT_MIN;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) { cin >> v[i]; }
    for (int i = 0; i < n; i++)
        for (int j = i + 1; j < n; j++) {
            if (v[i] == 1 && v[j] == n) ans = max(ans, j - i);
            if (v[i] == n && v[j] == 1) ans = max(ans, j - i);
        }
    cout << ans;
    return 0;
}
