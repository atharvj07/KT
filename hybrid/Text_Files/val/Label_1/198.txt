import sys
input = sys.stdin.readline

n,m = map(int,input().split())
k = list(map(int,input().split()))
sale,need = [],0
for i in k: need += i
for _ in range(m):
    d,t = map(int,input().split())
    sale.append((d,t))
sale.sort()
ans = 0
for d,t in sale:
    if d > ans:
        ans += need
    need -= k[t-1]
print(ans+1)
