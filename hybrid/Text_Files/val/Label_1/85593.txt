#include <iostream>
using namespace std;
bool solve() {
  long long a, b, c, d, x, y, x1, y1, x2, y2;
  cin >> a >> b >> c >> d;
  cin >> x >> y >> x1 >> y1 >> x2 >> y2;
  if (x > x2 || x < x1 || y > y2 || y < y1) return 0;
  if (a + d + y1 - y > c) return 0;
  if (a + d + y2 - y > c) return 0;
  if (c + b + x2 - x > a) return 0;
  if (c + b + x1 - x > a) return 0;
  return 1;
}
int main() {
  long long t;
  cin >> t;
  while (t--) cout << (solve() ? "YES" : "NO") << "\n";
  return 0;
}
