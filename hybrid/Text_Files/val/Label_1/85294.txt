#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<vector<char>> map1(n, vector<char>(n)), map2(n, vector<char>(n));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      cin >> map1[i][j];
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      cin >> map2[i][j];
    }
  }

  vector<vector<char>> rotated_map2 = map2;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      rotated_map2[i][j] = map2[j][n - i - 1];
    }
  }

  cout << (map1 == map2 || map1 == rotated_map2 ? "Yes" : "No") << endl;
  return 0;
}
