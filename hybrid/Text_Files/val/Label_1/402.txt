n=input()
if '0' in n:
    print(-1)
else:
    k=len(n)
    f=[(1<<i)-1 for i in range(k+1)]
    for i in range(2,k+1):
        last=n[-i]
        cur=0
        for j in range(i-1,-1,-1):
            if n[j]>last:
                cur|=f[j+1]
            elif n[j]==last:
                cur|=f[j+1]-1
        if (1<<i-1)&cur==0:
            print(-1)
            exit()
        f[i]=cur
    if f[k]&1:
        ans=f[k].bit_length()//2-1
    else:
        ans=f[k].bit_length()//2-2
    print(ans)
