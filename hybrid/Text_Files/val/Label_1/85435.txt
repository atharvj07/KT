#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll bm[21];
vector<ll>v;
int main() {
    int n; cin >> n;
    for(int i = 0; i < n; i++) {
        ll x; cin >> x;
        v.push_back(x);
    }
    bm[20] = 1;
    for(int i = 19; i >= 0; i--) {
        bm[i] = bm[i + 1] * 2;
    }
    ll ans = 0;
    for(auto u: v) {
        for(int i = 0; i < 20; i++) {
            if((bm[i] & u)) {
                for(auto v: v) {
                    if((bm[i] & v) == 0) {
                        ans += bm[i] * bm[i];
                    }
                }
            }
        }
    }
    cout << ans << endl;
    return 0;
}
