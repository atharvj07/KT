import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] fish = new int[n + 1];
        int max = 0;
        for (int i = 0; i < q; i++) {
            int a = sc.nextInt();
            int v = sc.nextInt();
            fish[a] += v;
            if (fish[a] > max) {
                max = fish[a];
            }
            List<Integer> list = new ArrayList<>();
            for (int j = 1; j <= n; j++) {
                if (fish[j] == max) {
                    list.add(j);
                }
            }
            Collections.sort(list);
            System.out.println(list.get(0) + " " + max);
        }
    }
}
