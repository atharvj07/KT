#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int a[4], b[4];
  for (int i = 0; i < 4; i++) {
    cin >> a[i] >> b[i];
  }
  sort(a, a + 4);
  sort(b, b + 4);
  if (a[1] > b[2] && b[1] > a[2]) {
    cout << "Draw";
  } else if (a[1] > b[2]) {
    cout << "Team 1";
  } else {
    cout << "Team 2";
  }
  return 0;
}
