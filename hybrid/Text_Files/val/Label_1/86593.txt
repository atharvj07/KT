import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int values[] = new int[n + 1];
        int[][] arr = new int[m + 1][2];
        for (int i = 1; i <= m; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            int val = sc.nextInt();
            arr[i][0] = x;
            arr[i][1] = y;
            values[x] = Math.max(values[x], val);
            values[y] = Math.max(values[y], val);
        }
        long ans = 0;
        for (int i = 1; i <= n; i++) {
            ans += values[i];
        }
        System.out.println(ans);
    }
}
