n=int(input())
a=input()
b=input()
i=j=k=l=0
p=[i,j]
q=[k,l]
for x in a:
    if x=='N':
        j+=1
    elif x=='E':
        i+=1
    elif x=='S':
        j-=1
    elif x=='W':
        i-=1
    p=[i,j]
    q=p.copy()
    for y in b:
        if y=='N':
            l+=1
        elif y=='E':
            k+=1
        elif y=='S':
            l-=1
        elif y=='W':
            k-=1
        if q==[k,l]:
            break
    if q!=[k,l]:
        print('NO')
        break
else:
    print('YES')
