T = int(input().strip())
for _ in range(T):
    x, y = map(int, input().split())
    if x >= y:
        print("YES")
    else:
        possible = False
        while x >= 2:
            if x >= y:
                possible = True
                break
            if x % 2 == 0:
                x = x * 3 // 2
            else:
                break
        if possible:
            print("YES")
        else:
            print("NO")
