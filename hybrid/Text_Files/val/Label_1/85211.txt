#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mxN = 1e5+5;
int n, x, k;
ll pfx[mxN], a[mxN];
int main() {
    cin >> n >> x >> k;
    for(int i = 0; i < n; i++) { cin >> a[i]; pfx[i+1] = pfx[i] + (a[i]%x==0); }
    ll ans = 0;
    for(int i = 0; i < n; i++) {
        ans += pfx[i]+1;
        if(a[i]%x==0) { ans += max(0ll, pfx[n]-pfx[i]-1); }
    }
    cout << ans << "\n";
}
