import sys

n, m = map(int, sys.stdin.readline().split())
k = list(map(int, sys.stdin.readline().split()))
days = [[] for _ in range(n+1)]
for i in range(m):
    d, t = map(int, sys.stdin.readline().split())
    days[t].append(d)
ans = 0
need = sum(k)
bought = 0
while bought < need:
    ans += 1
    for i in range(1, n+1):
        if k[i-1] and (not days[i] or days[i][0] <= ans):
            if not days[i] or ans == days[i][0]:
                days[i].pop(0)
            k[i-1] -= 1
            bought += 1
print(ans)
