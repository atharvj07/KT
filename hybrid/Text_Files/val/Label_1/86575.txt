import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long x = in.nextLong(), y = in.nextLong();
        int cmp = Long.compare(dig(x, y), dig(y, x));
        System.out.println(cmp == 0 ? "=" : cmp > 0 ? ">" : "<");
    }

    public static long dig(long x, long y) {
        return (long)Math.pow(10, (int)Math.log10(x) * y);
    }
}
