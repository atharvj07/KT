import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] adj[] = new int[n + 1][n + 1];
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a][b] = adj[b][a] = 1;
        }
        if (m % 3 != 0) {
            System.out.println("NO");
            return;
        }
        for (int i = 1; i <= n; i++) {
            int cnt = 0;
            for (int j = 1; j <= n; j++) {
                cnt += adj[i][j];
            }
            if (cnt % 3 != 0) {
                System.out.println("NO");
                return;
            }
        }
        System.out.println("YES");
        int dep = 1;
        for (int i = 1; i <= n; i++) {
            int cnt = 0;
            for (int j = 1; j <= n; j++) {
                cnt += adj[i][j];
            }
            if (cnt == 3) {
                for (int j = 1; j <= n; j++) {
                    if (adj[i][j] == 1) {
                        System.out.print(dep++ + " ");
                        adj[i][j] = adj[j][i] = 0;
                    }
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            int cnt = 0;
            for (int j = 1; j <= n; j++) {
                cnt += adj[i][j];
            }
            if (cnt == 1) {
                for (int j = 1; j <= n; j++) {
                    if (adj[i][j] == 1) {
                        System.out.print(dep++ + " ");
                        adj[i][j] = adj[j][i] = 0;
                    }
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            int cnt = 0;
            for (int j = 1; j <= n; j++) {
                cnt += adj[i][j];
            }
            if (cnt == 1) {
                for (int j = 1; j <= n; j++) {
                    if (adj[i][j] == 1) {
                        System.out.print(dep++ + " ");
                        adj[i][j] = adj[j][i] = 0;
                    }
                }
            }
        }
    }
}
