#include<bits/stdc++.h>
using namespace std;
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  if (n * m % k) {
    cout << "NO" << endl;
    return 0;
  }
  int h = (n * m) / k;
  if (h % 2 == 1) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  cout << 0 << " " << 0 << endl;
  cout << n << " " << h / 2 << endl;
  cout << n / 2 << " " << h << endl;
  return 0;
}
