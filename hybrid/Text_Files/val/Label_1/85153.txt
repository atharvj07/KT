#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    if (!(a[n - 1] == n && a[0] == 1)) {
        if (a[n - 1] > n || a[0] < 1) { cout << -1 << endl; }
        if (a[n - 1] > a[0] + k - 1) { cout << -1 << endl; }
        long long ans = n - a[0] - 1;
        cout << ans << endl;
        for (long long i = 1; i <= ans; i++) { cout << "+ " << i << " " << k << endl; }
    } else { cout << "0\n"; }
    return 0;
}
