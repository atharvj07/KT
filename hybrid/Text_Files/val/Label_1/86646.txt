import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }

        int m = input.nextInt();

        ArrayList<Pair> queries = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int k = input.nextInt();
            int pos = input.nextInt();
            queries.add(new Pair(k, pos));
        }

        int[] dp = new int[n + 1];
        int[] prev = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            dp[i] = Integer.MIN_VALUE;
        }

        for (int i = n - 1; i >= 0; i--) {
            for (int j = 1; j <= n; j++) {
                if (dp[j] < dp[j - 1]) {
                    dp[j] = dp[j - 1];
                    prev[j] = prev[j - 1];
                }

                if (dp[j] < dp[j - 1] + a[i]) {
                    dp[j] = dp[j - 1] + a[i];
                    prev[j] = i;
                }
            }
        }

        for (Pair query : queries) {
            int k = query.first;
            int pos = query.second;

            int cur = prev[k];
            for (int i = k - 1; i >= 0; i--) {
                if (cur != prev[i]) {
                    break;
                }

                cur = prev[cur];
                pos--;
            }

            System.out.println(a[cur]);
        }
    }

    public static class Pair {
        public int first;
        public int second;

        public Pair(int first, int second) {
            this.first = first;
            this.second = second;
        }
    }
}
