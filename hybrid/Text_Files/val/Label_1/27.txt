import sys

n = int(sys.stdin.readline())
blocks = []
for _ in range(n):
    blocks.append(list(map(int, sys.stdin.readline().split())))

blocks.sort(key=lambda x: x[1], reverse=True)
L = [blocks[0]]
R = [blocks[0][::-1]]
for block in blocks[1:]:
    if block[0] == L[-1][2]:
        L.append(block)
    elif block[-1] == R[-1][0]:
        R.append(block[::-1])
    elif block[0] == R[-1][2]:
        R.append(block)
    elif block[-1] == L[-1][0]:
        L.append(block[::-1])

if len(L) > len(R):
    print(sum(block[1] for block in L))
else:
    print(sum(block[1] for block in R))
