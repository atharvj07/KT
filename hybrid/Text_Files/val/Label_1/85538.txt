#include <iostream>
#include <vector>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        vector<int> b(n, -1);
        for (int i = 0; i < n; i++) {
            int pos = (i + a[i % n] % n + n) % n;
            if (b[pos] != -1) {
                cout << "NO" << endl;
                goto next;
            }
            b[pos] = i;
        }
        cout << "YES" << endl;
    next:
        continue;
    }
}
