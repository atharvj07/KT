#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define pb push_back
#define mp make_pair

const int N = 2e5;
vector<pair<ll, ll>> adj[N];
ll tot, par[N], sz[N], ans, a[N];
vector<pair<ll, ll>> edge;

ll find(ll x) {
    return par[x] == x ? x : par[x] = find(par[x]);
}

void merge(ll x, ll y, ll z) {
    ll p = find(x), q = find(y);
    if (p == q)
        return;
    if (sz[p] < sz[q])
        swap(p, q);
    par[q] = p;
    sz[p] += sz[q];
    tot -= z;
    ans += z;
}

int main() {
    ll t, n, p, i, j, x, y, w, k, l, r;
    cin >> t;
    while (t--) {
        cin >> n >> p;
        for (i = 0; i < n; i++) {
            adj[i].clear();
            par[i] = i;
            sz[i] = 1;
        }
        for (i = 0; i < n; i++) { cin >> a[i]; }
        tot = 0;
        for (i = 0; i < n - 1; i++) {
            vector<ll> temp;
            temp.pb(a[i]);
            ll g = 0;
            for (j = i + 1; j < n; j++) {
                temp.pb(a[j]);
                g = __gcd(g, a[j]);
                w = *min_element(temp.begin(), temp.end());
                adj[i].pb(mp(j + 1, w));
                tot += w;
                if (g == w)
                    edge.pb(mp(i + 1, j + 1));
            }
        }
        for (i = 0; i < n; i++) { adj[i].pb(mp(i + 1, p)); }

        for (auto x : edge)
            merge(x.first - 1, x.second - 1, x.second);

        cout << tot << endl;
    }
    return 0;
}
