import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int n = scanner.nextInt();
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            if (n == 0 && a == 0 && b == 0) {
                break;
            }
            if (a > b) {
                a ^= b;
                b ^= a;
                a ^= b;
            }
            int max = (n / a) * a;
            int count = 0;
            for (int i = 1; i <= n; i++) {
                if (i > max) {
                    max += a;
                }
                if (i % a != 0 && i % b != 0) {
                    count++;
                }
            }
            System.out.println(count);
        }
        scanner.close();
    }
}
