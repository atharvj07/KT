n=int(input())
a=[int(x) for x in input().split()]
sereja=0
dima=0
t=True
while n>0:
    if t:
        if a[0]>a[n-1]:
            sereja+=a[0]
            a.pop(0)
        else:
            sereja+=a[n-1]
            a.pop(n-1)
        n-=1
    else:
        if a[0]>a[n-1]:
            dima+=a[0]
            a.pop(0)
        else:
            dima+=a[n-1]
            a.pop(n-1)
        n-=1
    t= not t
print(sereja,dima)
