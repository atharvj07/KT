#include<bits/stdc++.h>
using namespace std;
const int N=100009,M=100000007;
vector<int> v[N];
int n,ans[N];
int main(){
    cin>>n;
    for(int i=1;i<=n;i++){
        int x;cin>>x;v[x].push_back(i);
    }
    bool f=0;
    for(int i=1;i<=n;i++){
        if(v[i].size()!=(n-1-v[i].size())){
            f=1;
            break;
        }
        else if(v[i].size()==0){
            for(int j=1;j<=n;j++){
                if(j!=i){
                    ans[j]=i;
                }
            }
            break;
        }
        else if(v[i].size()==n-1){
            for(int j=1;j<=n;j++){
                if(j!=i){
                    ans[j]=j;
                }
            }
            break;
        }
        else{
            ans[v[i][0]]=i;
            for(int j=1;j<v[i].size();j++){
                ans[v[i][j]]=v[i][0];
            }
            int d=0;
            for(int j=1;j<=n;j++){
                if(i==j) continue;
                if(d){
                    if(ans[j]!=d){
                        f=1;
                        break;
                    }
                }
                else if(!v[i].size()){
                    d=j;
                }
            }
            if(f) break;
        }
    }
    if(f){
        cout<<"Impossible";
    }
    else{
        cout<<"Possible\n";
        for(int i=1;i<=n;i++){
            cout<<ans[i]<<" ";
        }
    }
    return 0;
}
