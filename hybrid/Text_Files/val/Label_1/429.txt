n = int(input())
s = input()

def recolor_garland(s):
    recolors = 0
    t = list(s)
    for i in range(1, n):
        if t[i] == t[i-1]:
            recolors += 1
            for color in "RGB":
                if (i == n-1 or color != t[i+1]) and color != t[i-1]:
                    t[i] = color
                    break
    return f"{recolors}\n{''.join(t)}"

print(recolor_garland(s))
