#include <iostream>
#include <vector>

int main() {
    int t, n, x, s = 0;
    std::vector<int> v;
    std::cin >> t;
    while (t--) {
        std::cin >> n;
        while (n--) {
            std::cin >> x;
            v.push_back(x);
            s += (x % 2);
        }
        (s == 0 || s == n) ? std::cout << "Yes\n" : std::cout << "No\n";
        s = 0;
        v.clear();
    }
}
