import java.util.*;
import java.io.*;

public class Olympiad {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] skills = new int[n];
        String[] line = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            skills[i] = Integer.parseInt(line[i]);
        }

        int[] programmers = new int[n];
        int[] mathematicians = new int[n];
        int[] pe = new int[n];
        int pCount = 0, mCount = 0, peCount = 0;

        for (int i = 0; i < n; i++) {
            if (skills[i] == 1) {
                programmers[pCount++] = i;
            } else if (skills[i] == 2) {
                mathematicians[mCount++] = i;
            } else {
                pe[peCount++] = i;
            }
        }

        int w = Math.min(pCount, Math.min(mCount, peCount));
        System.out.println(w);

        for (int i = 0; i < w; i++) {
            System.out.println((programmers[i] + 1) + " " + (mathematicians[i] + 1) + " " + (pe[i] + 1));
        }
    }
}
