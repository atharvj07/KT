#include <iostream>
#include <iomanip>
#include <vector>
#include <climits>
using namespace std;
int main() {
    int w, h;
    cin >> w >> h;
    vector<string> image(h);
    for (int i = 0; i < h; i++) cin >> image[i];
    for (int j = 0; j < 2 * h; j++) {
        for (int i = w - 1; i >= 0; i--) {
            for (int k = 0; k < 2; k++) cout << image[j % h][i];
        }
        cout << endl;
    }
}
