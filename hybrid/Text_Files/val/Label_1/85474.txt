#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    if (i == 1 || i == 2) {
      cout << 'O';
    } else {
      int a = 1, b = 1, c;
      for (int j = 3; j <= i; j++) {
        c = a + b;
        a = b;
        b = c;
      }
      if (c == i) {
        cout << 'O';
      } else {
        cout << 'o';
      }
    }
  }
  cout << endl;
  return 0;
}
