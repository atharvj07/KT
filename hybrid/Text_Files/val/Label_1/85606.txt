#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  vector<bool> used(n + 1, false);
  vector<int> ans(m, 0);

  for (int i = 0; i < m; i++) {
    int difficulty;
    cin >> difficulty;

    if (!used[difficulty]) {
      used[difficulty] = true;
      bool can_hold = true;
      for (int j = 1; j <= n; j++) {
        if (!used[j]) {
          can_hold = false;
          break;
        }
      }
      if (can_hold) {
        ans[i] = 1;
        for (int j = 1; j <= n; j++) {
          used[j] = false;
        }
      }
    }
  }

  for (int i : ans) {
    cout << i;
  }
  cout << endl;

  return 0;
}
