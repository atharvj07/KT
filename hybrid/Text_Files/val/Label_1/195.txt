import sys
input = sys.stdin.readline

n, m = map(int, input().split())
k = list(map(int, input().split()))
d = []
t = []
for i in range(m):
    a, b = map(int, input().split())
    d.append(a)
    t.append(b)

ans = 0
for i in range(m):
    if k[t[i]-1] > 0:
        k[t[i]-1] -= 1
        ans = d[i]
print(ans)
