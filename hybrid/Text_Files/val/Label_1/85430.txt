#include <bits/stdc++.h>
#define int int64_t
using namespace std;
int solve() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto &x : a) cin >> x;
    int gcd = a[0];
    for (auto &x : a) gcd = __gcd(gcd, x);
    int cnt = 0;
    for (int i = 1; i * i <= gcd; i++) {
        if (gcd % i == 0) {
            cnt++;
            if (i != gcd / i) cnt++;
        }
    }
    return cnt;
}
int32_t main() { cout << solve() << endl; }
