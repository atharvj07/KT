#include <iostream>
#include <vector>
using namespace std;

const int MAXN = 1e5 + 10;
int N, sz[MAXN], par[MAXN], ans[MAXN];
bool used[MAXN];
vector<int> G[MAXN];

void dfs1(int u, int p) {
    sz[u] = 1;
    par[u] = p;
    for (int v : G[u])
        if (v != p)
            dfs1(v, u), sz[u] += sz[v];
}

void dfs2(int u, int p, int total) {
    ans[u] = max(total - sz[u], sz[u]);
    for (int v : G[u])
        if (v != p)
            dfs2(v, u, total);
}

int find_centroid(int u) {
    dfs1(u, u);
    dfs2(u, u, sz[u]);
    int centroid = u, min_sz = N;
    for (int v : G[u])
        if (v != par[u])
            min_sz = min(min_sz, ans[v]);
    for (int v : G[u])
        if (v != par[u] && ans[v] == min_sz)
            centroid = v;
    return centroid;
}

int main() {
    int T;
    cin >> T;
    while (T--) {
        cin >> N;
        for (int u = 1; u <= N; u++)
            G[u].clear(), used[u] = 0;
        for (int i = 1; i < N; i++) {
            int u, v;
            cin >> u >> v;
            G[u].push_back(v);
            G[v].push_back(u);
        }
        int centroid = find_centroid(1);
        for (int v : G[centroid])
            if (!used[v]) {
                cout << centroid << ' ' << v << '\n';
                used[centroid] = used[v] = 1;
                cout << centroid << ' ' << v << '\n';
                break;
            }
    }
    return 0;
}
