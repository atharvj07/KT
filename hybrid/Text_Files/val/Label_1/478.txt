import sys

input = sys.stdin.readline

n, m = map(int, input().split())
a = list(map(int, input().split()))
p = [0] * (n + 1)
q = [0] * (n + 1) * 2

s = sum(a)

for _ in range(m):
    t, *data = map(int, input().split())
    if t == 1:
        x, v = data
        p[x - 1] += v - a[x - 1]
        a[x - 1] = v
        s += v - a[x - 1]
    elif t == 2:
        l, r = data
        l -= 1
        print((s + p[l] + (q[2 * l] - q[2 * r])) % 1000000000)
    else:
        l, r, d = data
        l -= 1
        p[l] += d
        p[r] -= d
        q[2 * l] += (r - l) * d
        q[2 * r] -= (r - l) * d
        s += (r - l) * d
