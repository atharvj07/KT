#include <bits/stdc++.h>
using namespace std;

int main() {
    cin.tie(nullptr)->sync_with_stdio(false);
    int q;
    cin >> q;
    map<int, int> mp;
    map<int, int> mp2;
    while (q--) {
        char c;
        cin >> c;
        if (c == 'L' || c == 'R') {
            int x;
            cin >> x;
            if (c == 'L') {
                auto it = mp.find(x);
                if (it == mp.end()) mp[x] = 0;
                mp[x]++;
            } else {
                auto it = mp2.find(x);
                if (it == mp2.end()) mp2[x] = 0;
                mp2[x]++;
            }
        } else {
            int x;
            cin >> x;
            auto it = mp.find(x);
            auto it2 = mp2.find(x);
            if (it == mp.end() && it2 == mp2.end()) {
                cout << -1 << "\n";
                continue;
            }
            int ans;
            if (it != mp.end()) ans = min(mp[x], it2 == mp2.end() ? mp[x] : mp2[x] + 1);
            else ans = it2->second;
            cout << ans << "\n";
        }
    }
    return 0;
}
