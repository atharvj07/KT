import java.io.*;
import java.util.*;

public class Main {
    static final int inf = (int) 1e9;

    static int[] getOptimalSequence(int[] a, int k) {
        int n = a.length;
        int[][] dp = new int[n + 1][k + 1];
        int[][] p = new int[n + 1][k + 1];
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                dp[i][j] = -inf;
                p[i][j] = -1;
            }
        }

        dp[0][0] = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= k; j++) {
                dp[i][j] = Math.max(dp[i - 1][j], dp[i - 1][j - 1] + a[i - 1]);
                if (dp[i][j] != dp[i - 1][j]) {
                    p[i][j] = 0;
                } else {
                    p[i][j] = 1;
                }
            }
        }

        int[] res = new int[k];
        int i = n;
        int j = k;
        while (i > 0 && j > 0) {
            if (p[i][j] == 1) {
                i--;
            } else {
                j--;
                res[j] = a[i - 1];
                i--;
            }
        }

        return res;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        String[] s = in.readLine().split(" ");
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(s[i]);
        }

        int m = Integer.parseInt(in.readLine());
        for (int i = 0; i < m; i++) {
            s = in.readLine().split(" ");
            int k = Integer.parseInt(s[0]);
            int pos = Integer.parseInt(s[1]) - 1;

            int[] res = getOptimalSequence(a, k);
            System.out.println(res[pos]);
        }
    }
}
