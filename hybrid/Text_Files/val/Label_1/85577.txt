#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end());
    int ans = 1, cnt = 1;
    for (int x : a) {
      if (cnt >= x) ans++;
      else break;
      cnt++;
    }
    cout << ans << endl;
  }
  return 0;
}
