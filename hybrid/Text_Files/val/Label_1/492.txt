t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    wrong = 0
    for i in range(n):
        if a[i] != i + 1:
            wrong += 1
    if wrong == 0:
        print(0)
    else:
        left = 0
        while a[left] == left + 1:
            left += 1
        right = n - 1
        while a[right] == right + 1:
            right -= 1
        if any(a[i] == i + 1 for i in range(left, right + 1)):
            print(2)
        else:
            print(1)
