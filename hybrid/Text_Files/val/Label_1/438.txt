n=int(input())
a=[int(input()) for i in range(n)]
sa=[abs(i) for i in a]
sa.sort()
a1=[]
a2=[]
s=0
for i in a:
    s+=i
    if i>0:
        a1.append(abs(i))
    else:
        a2.append(abs(i))
if s>0:
    print('first')
elif s<0:
    print('second')
else:
    if a1==sa:
        print('first')
    elif a2==sa:
        print('second')
    else:
        for i in range(len(sa)):
            if a1[i]!=sa[i]:
                if a1[i]>sa[i]:
                    print('first')
                else:
                    print('second')
                break
