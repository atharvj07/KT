#include<bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long int n, x, y, d;
        cin >> n >> x >> y >> d;
        if (x == y)
            cout << 0 << '\n';
        else if ((y - x) % d == 0)
            cout << (y - x) / d << '\n';
        else
            cout << -1 << '\n';
    }
    return 0;
}
