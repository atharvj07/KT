import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt();
        int[] present = new int[n + 1];
        int count = 0;
        for (int i = 1; i <= n; i++) {
            if (present[i] == 0) {
                count++;
            }
        }
        for (int i = 0; i < m; i++) {
            String msg = input.next();
            int id = input.nextInt();
            if (msg.equals("-")) {
                present[id]--;
            } else {
                present[id]++;
            }
            if (present[id] == 0) {
                count--;
            }
        }
        System.out.println(count);
        for (int i = 1; i <= n; i++) {
            if (present[i] > 0) {
                System.out.print(i + " ");
            }
        }
    }
}
