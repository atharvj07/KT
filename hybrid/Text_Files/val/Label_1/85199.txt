#include<bits/stdc++.h>
using namespace std;

int main(){
	int n, m;
	cin >> n >> m;
	pair<int, int> center;
	for(int i = 1; i <= n; i++){
		for(int j = 1; j <= m; j++){
			char c; cin >> c;
			if(c == 'B'){
				if(center.first == 0){
					center = make_pair(i, j);
				}
				else{
					center = { (center.first + i) / 2, (center.second + j) / 2 };
				}
			}
		}
	}
	cout << center.first << " " << center.second << "\n";
	return 0;
}
