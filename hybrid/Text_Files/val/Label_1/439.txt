n = int(input())
a = [int(input()) for _ in range(n)]
pts = [0, 0]
seq = []
last = 0
for x in a:
  if x > 0:
    pts[0] += x
    if x > last:
      seq.append(pts[0])
      last = x
  else:
    pts[1] -= x
    if -x > last:
      seq.append(-pts[1])
      last = -x
if pts[0] == pts[1]:
  if seq == []:
    print("second")
  elif seq[0] > 0:
    print("first")
  else:
    print("second")
elif pts[0] > pts[1]:
  print("first")
else:
  print("second")
