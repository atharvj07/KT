import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = i + 1;
        }
        int[] dif = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            dif[i] = Math.abs(p[i] - p[i + 1]);
        }
        Arrays.sort(dif);
        int[] ans = new int[n];
        ans[0] = p[0];
        int idx = 0;
        int cnt = 0;
        for (int i = 1; i < n; i++) {
            if (idx < n - 1 && cnt == k) {
                idx++;
            }
            ans[i] = p[idx + dif[idx]];
            cnt++;
        }
        for (int i : ans) {
            System.out.print(i + " ");
        }
        System.out.println();
    }
}
