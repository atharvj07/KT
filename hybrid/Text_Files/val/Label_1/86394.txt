import java.util.*;

class Main {
    static int n, m, q, p[], ra[], t, c = 0;
    static boolean[] f;
    static ArrayList<Integer>[] adj;
    static int[] ai, bi, ci;

    static int find(int u) {
        int r = u;
        while (r != p[r])
            r = p[r];
        int i = u, j;
        while (i != r) {
            j = p[i];
            p[i] = r;
            i = j;
        }
        return r;
    }

    static boolean unite(int u, int v) {
        int a = find(u), b = find(v);
        if (a == b)
            return false;
        if (ra[a] > ra[b]) {
            p[b] = a;
        } else {
            p[a] = b;
            if (ra[a] == ra[b])
                ra[b]++;
        }
        return true;
    }

    static void add(int u) {
        f[u] = true;
        for (int v : adj[u]) {
            if (f[v]) {
                t = u;
                c = v;
                return;
            }
            add(v);
        }
    }

    static boolean check() {
        for (int i = 0; i < q; i++) {
            int u = ai[i], v = bi[i], w = ci[i];
            for (int j = 0; j < q; j++) {
                if (i != j && u == ai[j] && v == bi[j])
                    w = Math.min(w, ci[j]);
                if (i != j && v == ai[j] && u == bi[j])
                    w = Math.min(w, ci[j]);
            }
            if (find(u) != find(w) || find(v) != find(w))
                return false;
        }
        return true;
    }

    static void dfs1(int u, int p) {
        f[u] = true;
        for (int v : adj[u]) {
            if (v != p) {
                dfs1(v, u);
            }
        }
    }

    static void dfs(int u, int p) {
        f[u] = true;
        for (int v : adj[u]) {
            if (v != p && !f[v]) {
                f[v] = true;
                dfs(v, u);
                unite(u, v);
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        q = sc.nextInt();
        p = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            p[i] = i;
        }
        ra = new int[n + 1];
        t = 0;
        c = 0;
        adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a].add(b);
            adj[b].add(a);
        }
        ai = new int[q];
        bi = new int[q];
        ci = new int[q];
        for (int i = 0; i < q; i++) {
            ai[i] = sc.nextInt();
            bi[i] = sc.nextInt();
            ci[i] = sc.nextInt();
        }
        if (check()) {
            f = new boolean[n + 1];
            Arrays.fill(f, false);
            dfs1(1, 1);
            f = new boolean[n + 1];
            Arrays.fill(f, false);
            dfs(1, 1);
            for (int i = 1; i <= n; i++) {
                if (t == i || c == i) {
                    t = 0;
                    c = 0;
                    continue;
                }
                if (find(i) == find(t) || find(i) == find(c)) {
                    t = 0;
                    c = 0;
                }
            }
            if (t == 0 && c == 0) {
                System.out.println((long) Math.pow(2, n - 1) % 1000000007);
            } else {
                System.out.println(0);
            }
        } else {
            System.out.println(0);
        }
    }
}
