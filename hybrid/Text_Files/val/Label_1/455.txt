import sys
def get(): return list(map(int, sys.stdin.readline().split()))

n, m = get()
adj = [[] for i in range(n+1)]
for _ in range(m):
    a, b = get()
    adj[a].append(b)
    adj[b].append(a)

vis = [False] * (n+1)
group = [[]]

def dfs(x):
    if vis[x]:
        return
    vis[x] = True
    group[-1].append(x)
    for y in adj[x]:
        dfs(y)

for i in range(1, n+1):
    if not vis[i]:
        group.append([])
        dfs(i)

tot = 0
for i in range(1, len(group)):
    print(len(group[i]))
    for j in group[i]:
        tot += 1
        print(tot, tot)
