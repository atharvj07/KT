#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, d, b;
    cin >> n >> d >> b;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int ans = 0;
    int s1 = 0, s2 = 0;
    for (int i = 0; i < n; i++) {
        if (i <= d) {
            s1 += a[i];
        }
        if (i >= n - 1 - d) {
            s2 += a[i];
        }
    }
    s1 += (n - d - 1) * b;
    s2 += (n - d - 1) * b;
    if (s1 % b != 0) {
        ans++;
    }
    if (s2 % b != 0) {
        ans++;
    }
    cout << ans << endl;
    return 0;
}
