#include <bits/stdc++.h>
using namespace std;
int main() {
    int x, y;
    cin >> x >> y;
    if(x % 2 == 0 && y % 2 != 0) return cout << -1, 0;
    if(x % 3 == 0 && y % 3 != 0) return cout << -1, 0;
    int op = x / 3 - y / 3;
    x %= 3, y %= 3;
    if(x == 2 && y == 2) op += 3;
    if(x == 1 && y == 1) op += 6;
    if(x == 0 && y == 0) op += 0;
    cout << op << endl;
    return 0;
}
