#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  if (a[0] == 0) { cout << -1 << '\n'; return 0; }
  int cnt = 0;
  vector<pair<int, int>> ans;
  for (int i = 1; i < n; i++) {
    if (a[i] > 0) { cnt = i; }
  }
  if (cnt == 0) { cout << -1 << '\n'; return 0; }
  int num = min(a[0], n - 1);
  for (int i = 2; i <= cnt && num > 0; i++) {
    if (a[i] > 0) { ans.push_back({1, i}); num--; }
  }
  for (int i = cnt + 1; i < n && num > 0; i++) {
    if (a[i] > 0) { ans.push_back({1, i}); num--; }
  }
  for (int i = 2; i <= cnt && num > 0; i++) {
    if (a[i] > 0) { a[i]--; ans.push_back({i, cnt}); }
  }
  for (int i = cnt + 1; i < n && num > 0; i++) {
    if (a[i] > 0) { a[i]--; ans.push_back({i, cnt}); }
  }
  for (int i = 2; i <= cnt && a[1] > 0 && a[cnt] > 0; i++) {
    if (a[i] > 0) { a[1]--; a[cnt]--; ans.push_back({1, cnt}); }
  }
  for (int i = cnt + 1; i < n && a[1] > 0 && a[cnt] > 0; i++) {
    if (a[i] > 0) { a[1]--; a[cnt]--; ans.push_back({1, cnt}); }
  }
  for (int i = 1; i < cnt;) {
    if (a[i] > 1 && a[cnt] > 0) {
      a[i] -= 2;
      a[cnt] -= 1;
      ans.push_back({i, cnt});
    } else if (a[i] == 1 && a[cnt + 1] > 0) {
      a[i] -= 1;
      a[cnt + 1] -= 1;
      ans.push_back({i, cnt + 1});
      i++;
    } else {
      i++;
    }
  }
  for (int i = cnt + 2; i < n;) {
    if (a[i] > 1 && a[cnt] > 0) {
      a[i] -= 2;
      a[cnt] -= 1;
      ans.push_back({i, cnt});
    } else if (a[i] == 1 && a[cnt + 1] > 0) {
      a[i] -= 1;
      a[cnt + 1] -= 1;
      ans.push_back({i, cnt + 1});
      i++;
    } else {
      i++;
    }
  }
  cout << ans.size() << '\n';
  for (int i = 0; i < ans.size(); i++) {
    cout << ans[i].first << ' ' << ans[i].second << '\n';
  }
  return 0;
}
