import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int a = sc.nextInt();
            int b = sc.nextInt();
            if (n == 0 && a == 0 && b == 0) break;
            int ans = 0;
            for (int i = 0; i <= n; i++) {
                if ((i % a != 0) && (i % b != 0)) ans++;
            }
            System.out.println(ans);
        }
        sc.close();
    }
}
