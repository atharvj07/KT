#include <bits/stdc++.h>

using namespace std;

int a[200'001];
vector<int> adj[200'001];

int dfs(int root, int par) {
    int min_v = a[root];
    for (int v : adj[root]) {
        if (v == par) continue;
        min_v = min(min_v, dfs(v, root));
    }
    return min_v;
}

int main() {
    int n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    int mx = 1e9;
    for (int i = 1; i <= n; i++) {
        mx = min(mx, dfs(i, -1));
    }
    cout << mx << endl;
    return 0;
}
