import java.util.Scanner;

public class Main {
    static int N, M;
    static int[][] G;
    static boolean[] vis;
    static int mod = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        M = sc.nextInt();
        G = new int[N][N];
        for (int i = 0; i < M; i++) {
            int s = sc.nextInt() - 1;
            int t = sc.nextInt() - 1;
            G[s][t] = G[t][s] = 1;
        }
        vis = new boolean[N];
        int ans = dfs(0, 0);
        System.out.println(ans);
    }

    static int dfs(int cur, int prev) {
        vis[cur] = true;
        int ans = 0;
        for (int i = 0; i < N; i++) {
            if (G[cur][i] == 1 && !vis[i]) {
                if (i == prev) {
                    ans = (ans + dfs(i, prev)) % mod;
                } else {
                    ans = (ans + mod - dfs(i, cur)) % mod;
                }
            }
        }
        return ans + 1;
    }
}
