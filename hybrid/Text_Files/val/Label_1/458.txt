def solve_painter_problem():
    n, m = map(int, input().split())
    harmonize = [[] for _ in range(n + 1)]
    for _ in range(m):
        a, b = map(int, input().split())
        harmonize[a].append(b)
        harmonize[b].append(a)
    
    positions = [[] for _ in range(n + 1)]
    col = 1
    for color in range(1, n + 1):
        row = color
        positions[color].append((row, col))
        for adj in harmonize[color]:
            if len(positions[adj]) == 0:
                col += 1
                positions[adj].append((row, col))
                positions[color].append((row, col))
        
    for color in range(1, n + 1):
        print(len(positions[color]))
        for pos in positions[color]:
            print(*pos)

solve_painter_problem()
