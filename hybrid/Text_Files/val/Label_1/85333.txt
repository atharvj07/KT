#include <bits/stdc++.h>
using namespace std;
typedef long long int ll;
const ll N = 1e5 + 1, mod = 1e9 + 7, inf = 1e9 + 70;

void solve() {
    int n, x;
    cin >> n >> x;
    vector<ll> a(n);
    for (auto &it : a) { cin >> it; }
    ll ans1 = 0, ans2 = 0;
    for (auto &it : a) {
        ans1 += ceil((double)it / x);
        ll rem = it % x;
        ans2 += ceil((double)it / x) - (rem == 0);
    }
    cout << ans1 << ' ' << ans2 << '\n';
}
int main() {
    int t;
    cin >> t;
    while (t--) { solve(); }
    return 0;
}
