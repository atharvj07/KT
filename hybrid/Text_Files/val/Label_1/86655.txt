import java.util.*;
import java.io.*;

class Main {
    static long[] a;
    static long[] b;
    static long[] c;
    static long[] d;
    static long INF = Long.MAX_VALUE / 2;

    static void merge(int l, int m, int r) {
        int n1 = m - l + 1;
        int n2 = r - m;
        for (int i = 0; i < n1; ++i) {
            c[i] = a[l + i];
            d[i] = b[l + i];
        }
        for (int i = 0; i < n2; ++i) {
            c[n1 + i] = a[m + i + 1];
            d[n1 + i] = b[m + i + 1];
        }
        Long[] ca = new Long[n1 + n2];
        Long[] cb = new Long[n1 + n2];
        for (int i = 0; i < n1 + n2; ++i) {
            ca[i] = c[i];
            cb[i] = d[i];
        }
        Arrays.sort(ca);
        Arrays.sort(cb);
        for (int i = 0; i < n1 + n2; ++i) {
            a[l + i] = ca[i];
            b[l + i] = cb[i];
        }
    }

    static void mergeSort(int l, int r) {
        if (l < r) {
            int m = (l + r) / 2;
            mergeSort(l, m);
            mergeSort(m + 1, r);
            merge(l, m, r);
        }
    }

    static long solve(int n) {
        mergeSort(0, n - 1);
        long ans = 0;
        for (int i = 0; i < n; ++i) {
            if (b[i] > a[i]) {
                ans += (b[i] - a[i]) * i;
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        a = new long[n];
        b = new long[n];
        c = new long[n];
        d = new long[n];
        for (int i = 0; i < n; ++i) {
            a[i] = sc.nextLong();
        }
        for (int i = 0; i < n; ++i) {
            b[i] = sc.nextLong();
        }
        System.out.println(solve(n));
    }
}
