#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
map<ll, ll> compress;
vector<ll> pos[100005];
int main() {
  ll n, x;
  cin >> n;
  for (ll i = 1, cnt = 0; i <= n; ++i) {
    cin >> x;
    if (!compress.count(x)) compress[x] = ++cnt;
    pos[compress[x]].push_back(i);
  }
  ll mx = 0, ans;
  bool flag = 0;
  for (auto& vec : pos) {
    if (vec.size() & 1) {
      flag = 1;
      break;
    }
    if (vec.size() > mx) mx = vec.size(), ans = vec[0];
  }
  if (flag) cout << "Impossible\n";
  else {
    cout << "Possible\n";
    for (ll i = 1; i <= n; ++i) cout << ans << ' ';
    cout << '\n';
  }
  return 0;
}
