import sys

n = int(input())
a = [int(x) for x in input().split()]
b = [int(x) for x in input().split()]

if n == 2:
    if a[0] == 0:
        print("YES")
        print("0 0")
    else:
        print("NO")
else:
    t = [0] * n
    t[0] = a[0] | b[0]
    for i in range(1, n - 1):
        t[i] = a[i] | b[i]
        if t[i] < t[i-1]:
            print("NO")
            sys.exit()
    t[n-1] = a[n-2] | b[n-2]
    print("YES")
    print(' '.join([str(x) for x in t]))
