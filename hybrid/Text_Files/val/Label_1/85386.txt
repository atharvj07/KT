#include <cstdio>
#include <vector>

inline int rd() {
  int x;
  scanf("%d", &x);
  return x;
}

int main() {
  std::vector<std::vector<char>> f(10, std::vector<char>(10));
  int cx = 0, co = 0;
  for (int i = 0; i < 10; ++i) {
    scanf(" %s", &f[i][0]);
    for (int j = 0; j < 10; ++j) {
      (f[i][j] == 'X') ? ++cx : (f[i][j] == 'O') ? ++co : 0;
    }
  }
  if (cx != co) {
    printf("NO\n");
    return 0;
  }
  int dx[] = {-1, 0, 1, 1, 1, 0, -1, -1};
  int dy[] = {-1, -1, -1, 0, 1, 1, 1, 0};
  int ans = 0;
  for (int i = 0; i < 10; ++i) {
    for (int j = 0; j < 10; ++j) {
      if (f[i][j] != '.') {
        continue;
      }
      for (int d = 0; d < 8; ++d) {
        int cnt = 1;
        for (int k = 0; k < 4; ++k) {
          int x = i + k * dx[d], y = j + k * dy[d];
          if (0 <= x && x < 10 && 0 <= y && y < 10 && f[x][y] == 'X') {
            ++cnt;
          }
        }
        if (cnt > 4) {
          ++ans;
        }
      }
    }
  }
  printf("%s\n", ans > 0 ? "YES" : "NO");
  return 0;
}
