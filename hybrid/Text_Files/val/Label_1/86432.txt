import java.util.Scanner;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] p = new int[n];
        int[] s = new int[n];
        int[] b = new int[n];
        for(int i = 0; i < n; i++) {
            p[i] = in.nextInt();
        }
        for(int i = 0; i < n; i++) {
            s[i] = in.nextInt();
        }
        for(int i = 0; i < n; i++) {
            b[i] = in.nextInt();
        }
        int[] inc = new int[m];
        int[] pref = new int[m];
        for(int i = 0; i < m; i++) {
            inc[i] = in.nextInt();
        }
        for(int i = 0; i < m; i++) {
            pref[i] = in.nextInt();
        }
        for(int i = 0; i < m; i++) {
            int cnt = 0;
            for(int j = 0; j < n; j++) {
                if(p[j] <= inc[i] && inc[i] <= s[j] && Math.abs(b[j] - pref[i]) <= inc[i] - p[j]) {
                    cnt++;
                }
            }
            System.out.print(cnt + " ");
        }
    }
}
