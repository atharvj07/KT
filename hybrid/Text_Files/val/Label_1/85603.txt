#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<bool> used(n + 1, false);
  string ans;
  for (int i = 0; i < m; i++) {
    int diff;
    cin >> diff;
    if (!used[diff]) {
      used[diff] = true;
      bool round = true;
      for (int j = 1; j <= n; j++) {
        if (!used[j]) round = false;
      }
      if (round) ans += '1';
      else ans += '0';
    } else
      ans += '0';
  }
  cout << ans << endl;
  return 0;
}
