#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define mod 1000000007

const int N = 1e6 + 5;

char s[N];
ll pow2[N], inv[N];

ll bigmod(ll a, ll p) {
    if (p == 0) return 1;
    if (p == 1) return a;
    if (p % 2 == 0) {
        ll tmp = bigmod(a, p / 2);
        return ((tmp % mod) * (tmp % mod)) % mod;
    } else {
        return ((bigmod(a, p - 1) % mod) * (a % mod)) % mod;
    }
}

void calc_hash() {
    pow2[0] = 1;
    for (int i = 1; i < N; i++)
        pow2[i] = (pow2[i - 1] * 2) % mod;
    inv[N - 1] = bigmod(pow2[N - 1], mod - 2);
    for (int i = N - 2; i >= 0; i--)
        inv[i] = (inv[i + 1] * 2) % mod;
}

ll get_hash(int l, int r) {
    return ((s[l] - s[r + 1] + mod) % mod * inv[l] % mod) % mod;
}

int main() {
    calc_hash();

    int n, m;
    cin >> n >> m;
    string p;
    cin >> p;
    int l = p.size();
    vector<int> v;
    for (int i = 0; i < m; i++) {
        int x;
        cin >> x;
        v.push_back(x);
    }

    s[n + 1] = '$';
    for (int i = n; i >= 1; i--) {
        s[i] = s[i + 1] + p[i - 1] - 'a' + 1;
        s[i] %= 26;
        s[i] = (s[i] + 26) % 26 + 'a';
    }

    ll cur, ans = 0;
    for (int i = 0; i + l - 1 <= v[0] - 1; i++) {
        cur = get_hash(i, i + l - 1);
        for (int j = 1; j < m; j++) {
            if (cur != get_hash(v[j] - 1, v[j] - 1 + l - 1))
                break;
            else {
                if (j == m - 1) {
                    ans++;
                    ans %= mod;
                }
            }
        }
    }

    cout << ans << endl;
    return 0;
}
