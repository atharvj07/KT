import java.util.*;
public class Main{
    public static void main(String[] args){
        Scanner in = new Scanner(System.in);
        final int[][] x = new int[3][2];
        for(int i=0; i<3; i++){
            for(int j=0; j<2; j++){
                x[i][j] = in.nextInt();
            }
        }
        for(int i=0; i<4; i++){
            if(x[0][0]+x[2][0]==x[1][0] && x[0][1]+x[2][1]==x[1][1]){
                System.out.println("YES");
                return;
            }
            int a = x[0][0];
            x[0][0] = -x[0][1];
            x[0][1] = a;
        }
        System.out.println("NO");
    }
}
