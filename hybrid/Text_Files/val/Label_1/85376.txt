#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, r;
    cin >> n >> r;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    sort(v.begin(), v.end());
    bool can = true;
    for (int i = 0; i < n; i++) {
        if (v[i].first > r) {
            can = false;
        }
        r += v[i].second;
    }
    if (r < 0) can = false;
    cout << (can ? "YES" : "NO") << '\n';
}
