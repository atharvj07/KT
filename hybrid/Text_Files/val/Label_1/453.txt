import sys
input = sys.stdin.readline
n, m = map(int, input().split())
adj = [[] for i in range(n+1)]
for i in range(m):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)
painted = [0] * (n+1)
vis = [0] * (n+1)
ans = []

def dfs(node, color, ans_):
    vis[node] = 1
    ans_.append((node, color))
    painted[color] += 1
    for u in adj[node]:
        if vis[u] == 0:
            dfs(u, color, ans_)

for i in range(1, n+1):
    if painted[i] == 0:
        ans_ = []
        dfs(i, i, ans_)
        ans.append((len(ans_), ans_))

print(len(ans))
for x in ans:
    print(x[0])
    for y in x[1]:
        print(*y)
