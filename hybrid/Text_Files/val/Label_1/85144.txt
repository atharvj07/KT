#include<iostream>
#include <vector>
#include <queue>
using namespace std;
int main(){
int n,a[1000];
cin>>n;
for(int i=1;i<=n;i++) cin>>a[i];
queue<pair<int,int> > q;
bool vis[n+5]={0};
q.push(make_pair(1,0));
vis[1]=1;
int ans=0;
while(!q.empty()){
int s=q.front().first;
int cnt=q.front().second;
q.pop();
ans+=cnt;
if(cnt==0){
for(int i=2;i<=n;i++){
if(vis[i]==0){
vis[i]=1;
q.push(make_pair(i,a[i]));
q.push(make_pair(i,a[i]-1));
}
}
}
else{
for(int i=2;i<=n;i++){
if(a[s]==0||a[i]==0)continue;
if(vis[i]==0){
a[s]--;a[i]--;
vis[i]=1;
q.push(make_pair(i,a[i]));
q.push(make_pair(i,a[i]-1));
}
}
}
}
if(ans%2==0)cout<<ans/2<<endl;
else cout<<-1<<endl;
return 0;
}
