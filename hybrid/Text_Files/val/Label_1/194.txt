def min_days_to_start_playing(n, m, k, special_offers):
    total_microtransactions = sum(k)
    max_days = total_microtransactions * 2  # Worst case, buying all without any offer
    special_offers.sort()

    def can_buy_in_days(days):
        burles = days  # Earned burles equals the number of days
        purchases = [0] * n  # Track purchases for each type
        for day, type_ in special_offers:
            if day > days: break  # Ignore offers beyond the available days
            # Calculate maximum purchases for this offer within the available burles
            while burles > 0 and purchases[type_ - 1] < k[type_ - 1]:
                purchases[type_ - 1] += 1
                burles -= 1
        # Calculate remaining purchases without offers within the remaining burles
        for i in range(n):
            burles -= 2 * (k[i] - purchases[i])
        return burles >= 0

    # Binary search to find the minimum days
    low, high = 1, max_days
    while low < high:
        mid = (low + high) // 2
        if can_buy_in_days(mid):
            high = mid
        else:
            low = mid + 1
    return low

n, m = map(int, input().split())
k = list(map(int, input().split()))
special_offers = [tuple(map(int, input().split())) for _ in range(m)]
print(min_days_to_start_playing(n, m, k, special_offers))
