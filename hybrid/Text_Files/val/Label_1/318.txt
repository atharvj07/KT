import sys

input = sys.stdin.readline

n, m = map(int, input().split())
k = list(map(int, input().split()))
tot = sum(k)
need = [tot] * n
sale = [False] * n
ans = 0

for _ in range(m):
    d, t = map(int, input().split())
    t -= 1
    ans = max(ans, d)
    if not sale[t]:
        sale[t] = True
        need[t] -= k[t]
    tot -= k[t]

while tot > 0:
    ans += 1
    for i in range(n):
        if need[i] > 0 and not sale[i]:
            tot -= 1
            need[i] -= 1

print(ans)
