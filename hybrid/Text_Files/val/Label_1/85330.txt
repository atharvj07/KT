#include <bits/stdc++.h>
using namespace std;
#define int long long
int n, a[110];
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 1; i <= n; i++) { a[i] = i; }
        for (int i = n - 1; i >= 2; i--) {
            a[i - 1] += a[i];
        }
        for (int i = 1; i <= n; i++) {
            cout << a[i] << " ";
        }
        cout << endl;
    }
}
