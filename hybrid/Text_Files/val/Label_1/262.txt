import sys
input = sys.stdin.readline

n = int(input())
a = [list(map(int, input().split())) for i in range(n)]
m = 10**9 + 7
f = [[[0 for i in range(2)] for j in range(1<<n)] for k in range(n+1)]
f[0][1][1] = 1
for mask in range(1, 1<<n):
    lst = -1
    for i in range(n):
        if mask & (1<<i):
            lst = i
    for i in range(n):
        if i == lst: continue
        if a[lst][i] == 0: continue
        for subset in range((1<<n)-1):
            if subset & (1<<lst) or subset & (1<<i): continue
            if f[i][mask | (1<<i)][1]:
                f[lst][mask][0] = (f[lst][mask][0] + f[i][mask | (1<<i)][1]) % m
                f[lst][mask][1] = (f[lst][mask][1] + f[i][mask | (1<<i)][0]) % m
print(f[0][(1<<n)-1][0])
