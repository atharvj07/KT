from collections import defaultdict, deque

n, m, s = map(int, input().split())
g = defaultdict(list)
ed = []
for i in range(m):
    a, b, c = map(int, input().split())
    if a == 1:
        g[b].append(c)
    else:
        g[b].append(c)
        g[c].append(b)
        ed.append((-1, b, c))

d = [0] * (n + 1)

def bfs(st, sign):
    q = deque()
    q.append((st, 0))
    ans = set()
    ans.add(st)
    d[st] = 0
    while q:
        v, dd = q.popleft()
        for u in g[v]:
            if u not in ans:
                ans.add(u)
                q.append((u, dd + sign))
                d[u] = dd + sign
    return ans

ans1 = bfs(s, 1)
ans2 = bfs(s, -1)

print(len(ans1))
for x in ed:
    if d[x[1]] > d[x[2]]:
        print('+', end='')
    else:
        print('-', end='')
print()
print(len(ans2))
for x in ed:
    if d[x[1]] < d[x[2]]:
        print('+', end='')
    else:
        print('-', end='')
