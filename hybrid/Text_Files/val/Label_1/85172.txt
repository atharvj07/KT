#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        string s;
        cin>>s;
        bool f=1;
        for(int i=0;i<n/2;i++){
            if(s[i]!=s[n-i-1]){
                if(s[i]==(s[n-i-1]-1) || s[i]==(s[n-i-1]+1));
                else {f=0; break;}
            }
        }
        if(f){cout<<"YES"<<endl;}
        else{cout<<"NO"<<endl;}
    }
    return 0;
}
