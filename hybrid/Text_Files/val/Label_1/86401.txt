import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int[] t = new int[n], a = new int[n], b = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextInt();
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        int[] dp = new int[1 << n];
        Arrays.fill(dp, Integer.MAX_VALUE);
        dp[0] = 0;
        for (int mask = 0; mask < 1 << n; mask++) {
            int acnt = Integer.bitCount(mask & ((1 << n) - 1));
            int bcnt = Integer.bitCount((mask >> n) & ((1 << n) - 1));
            if (acnt >= k && bcnt >= k) {
                for (int i = 0; i < n; i++) {
                    if (a[i] == 1 && (mask & (1 << i)) == 0) {
                        dp[mask | (1 << i)] = Math.min(dp[mask | (1 << i)], dp[mask] + t[i]);
                    } else if (b[i] == 1 && ((mask >> n) & (1 << i)) == 0) {
                        dp[mask | (1 << i) | (1 << (i + n))] = Math.min(dp[mask | (1 << i) | (1 << (i + n))], dp[mask] + t[i]);
                    }
                }
            }
        }
        int ans = dp[(1 << n) - 1];
        System.out.println(ans == Integer.MAX_VALUE ? -1 : ans);
    }
}
