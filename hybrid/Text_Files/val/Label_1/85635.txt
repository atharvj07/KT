#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> p[i];
  }
  vector<int> in(n + 1), out(n + 1);
  for (int i = 1; i <= n; i++) {
    in[p[i]]++;
    out[i]++;
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    if (in[i] == 0) {
      ans += out[i] * (out[i] - 1) / 2;
    } else if (out[i] == 0) {
      ans += in[i] * (in[i] - 1) / 2;
    } else {
      ans += in[i] * out[i];
    }
  }
  int x, y;
  for (int i = 1; i <= n; i++) {
    if (in[i] == 0) {
      x = i;
      int j = p[i];
      while (j != i) {
        out[j]--;
        j = p[j];
      }
      break;
    }
  }
  for (int i = 1; i <= n; i++) {
    if (out[i] == 0) {
      y = i;
      int j = p[i];
      while (j != i) {
        in[j]--;
        j = p[j];
      }
      break;
    }
  }
  ans -= in[x] * out[x];
  ans += in[y] * out[y];
  cout << ans << endl;
  return 0;
}
