import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    List<Integer> heights = new ArrayList<>();
    int[] bars = new int[1001];
    for (int i = 1; i <= n; i++) bars[scanner.nextInt()]++;
    for (int i = 1; i <= 1000; i++) {
      if (bars[i] > 0) {
        int height = bars[i];
        while (heights.size() < height) heights.add(0);
        heights.set(height - 1, heights.get(height - 1) + 1);
        bars[i] -= heights.get(height - 1);
      }
    }
    int maxHeight = heights.size();
    int numTowers = 0;
    for (int height : heights) numTowers += height;
    System.out.println(maxHeight + " " + numTowers);
  }
}
