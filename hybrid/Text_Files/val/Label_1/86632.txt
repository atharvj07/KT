import java.util.*;
public class Main {
    static boolean isNatural(String s) {
        int n = s.length();
        int[] l = new int[n + 1], r = new int[n + 1];
        for (int i = 0; i < n; ++i) {
            if (s.charAt(i) == 'L' || s.charAt(i) == 'R') {
                if (l[i] == 0) { l[i] = i + 1; }
                r[i] = i + 1;
            }
        }
        for (int i = n - 1; i >= 0; --i) {
            if (s.charAt(i) == 'L' || s.charAt(i) == 'R') {
                if (r[i] == 0) { r[i] = i + 1; }
                l[i] = i + 1;
            }
        }
        for (int i = 0; i < n; ++i) {
            if (s.charAt(i) == 'U' || s.charAt(i) == 'D') {
                if (l[i] == 0 || r[i] == 0 || l[i] > r[i]) { return false; }
            }
        }
        return true;
    }
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        while (n-- > 0) {
            String s = in.next();
            System.out.println(isNatural(s) ? "Yes" : "No");
        }
    }
}
