MOD = 998244353

def power(x, y, p):
    res = 1
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def solve(n, k):
    if k == 1:
        return 0
    if k == 2:
        return (4 * power(2, n - 1, MOD) - 2) % MOD
    ans = 0
    f = [0] * (n + 1)
    f[0] = 1
    for i in range(1, n + 1):
        f[i] = f[i - 1] * i % MOD
    inv = [0] * (n + 1)
    inv[n] = power(f[n], MOD - 2, MOD)
    for i in range(n, 0, -1):
        inv[i - 1] = inv[i] * i % MOD
    for i in range(1, n + 1):
        for j in range(i, n + 1):
            tmp = f[n] * inv[i] % MOD * inv[j] % MOD * inv[n - i] % MOD * inv[n - j] % MOD
            tmp = tmp * power(2, n * (n - i) + n * (n - j) - (n - i) * (n - j), MOD) % MOD
            if (i + j - k) * 2 >= n:
                ans = (ans + tmp) % MOD
            if i != j and (i * 2 >= k or j * 2 >= k):
                ans = (ans + tmp) % MOD
    return ans

n, k = map(int, input().split())
print(solve(n, k))
