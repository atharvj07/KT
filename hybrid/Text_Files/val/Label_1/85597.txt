#include <bits/stdc++.h>
using namespace std;

int a[5], b[5];

int main() {
  for (int i = 1; i <= 4; i++) { cin >> a[i] >> b[i]; }
  if (a[1] > b[3] && b[1] > a[2] && a[2] < b[4]) {
    cout << "Team 1" << endl;
  } else if (a[3] > b[1] && b[3] > a[4] && a[4] < b[2]) {
    cout << "Team 2" << endl;
  } else {
    cout << "Draw" << endl;
  }
}
