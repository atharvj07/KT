import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] p = new int[n];
        int[] s = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
            s[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        int[] inc = new int[m];
        int[] pref = new int[m];
        for (int i = 0; i < m; i++) {
            inc[i] = sc.nextInt();
            pref[i] = sc.nextInt();
        }
        int[] ans = new int[m];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (p[j] <= inc[i] && inc[i] <= s[j] && Math.abs(b[j] - pref[i]) <= inc[i] - p[j]) {
                    ans[i]++;
                }
            }
        }
        for (int i = 0; i < m; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
