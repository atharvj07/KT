n, t, c = map(int, input().split())
prisoners = list(map(int, input().split()))

valid_segments = 0
bad_prisoners = 0

for i in range(n):
    if prisoners[i] > t:
        bad_prisoners += 1
    if i >= c and prisoners[i - c] > t:
        bad_prisoners -= 1
    if bad_prisoners == 0 and i >= c - 1:
        valid_segments += 1

print(valid_segments)
