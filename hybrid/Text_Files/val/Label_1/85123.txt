#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back
#define mp make_pair
const int mod=1e9+7,N=1e5+7;
const double PI=acos(-1.0);
bool vis[N];
vector<int>edj[N],adj[N],path;
pair<int,int>ppr[N];
int dist[N],a[N],dp[N][20];
int32_t main() {
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);
    int n,m,k,i,j,x;
    cin>>n>>m>>k;
    for(i=1;i<=n;i++)cin>>a[i];
    for(i=1;i<=n;i++){
        while(!path.empty())path.pop_back();
        for(j=i;j<=n;j++){
            path.pb(j);
            x=(j-i+1+m-1)/m;
            dp[j][0]=a[j]-k*x;
            for(int l=1;(1<<l)<=j-i+1;l++){
                dp[j][l]=max(dp[j][l-1],dp[j-(1<<(l-1))][l-1]);
            }
        }
        ppr[i]=mp(dp[n][(int)log2(j-i+1)],j);
    }
    sort(ppr+1,ppr+n+1);
    int ans=0;
    for(i=1;i<=n;i++){
        ans=max(ans,ppr[i].first);
        x=(ppr[i].second-i+1+m-1)/m;
        if(i+x-1<=n)ans=max(ans,ppr[i].first+dp[i+x-1][(int)log2(x)]);
    }
    cout<<ans<<"\n";
    return 0;
}
