import java.util.Scanner;

public class BallOrder {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] throwsAllowed = new int[n];
        for (int i = 0; i < n; i++) {
            throwsAllowed[i] = scanner.nextInt();
        }

        long[][] dp = new long[n + 1][n + 1];
        dp[0][0] = 1;

        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= n; j++) {
                // Calculate the number of ways to place the i-th ball with j throws remaining
                long ways = 0;
                if (j > 0) {
                    ways += dp[i - 1][j - 1] * (i - 1); // Place the ball in a new position with a remaining throw
                }
                if (j >= throwsAllowed[i - 1]) {
                    ways += dp[i - 1][j - throwsAllowed[i - 1]] * (n - i + 1); // Place the ball in an existing position with all throws used
                }
                dp[i][j] = ways % MOD;
            }
        }

        System.out.println(dp[n][n]);
    }
}
