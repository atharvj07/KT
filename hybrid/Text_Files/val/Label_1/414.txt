import sys
readline = sys.stdin.readline

def f(b):
    if len(b) == 1:
        return b[0]
    return f([b[i - 1] ^ b[i] for i in range(1, len(b))])

n = int(readline())
a = list(map(int, readline().split()))
m = int(readline())
for l, r in [map(int, readline().split()) for i in range(m)]:
    print(f(a[l - 1:r]))
