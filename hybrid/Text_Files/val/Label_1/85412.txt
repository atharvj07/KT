#include <bits/stdc++.h>
using namespace std;
struct events
{
    string event;
    int num;
    bool operator<(const events &a) const
    {
        return event != a.event ? event < a.event : num < a.num;
    }
};
int main()
{
    int n;
    cin >> n;
    vector<events> v1;
    vector<events> v2;
    for (int i = 0; i < n; i++)
    {
        string e;
        cin >> e;
        int x;
        cin >> x;
        events v;
        v.event = e;
        v.num = x;
        v1.push_back(v);
    }
    v2.push_back(v1[0]);
    for (int i = 1; i < n; i++)
    {
        if (v2.back().event == "insert")
        {
            v2.push_back(v1[i]);
        }
        else
        {
            if (v2.back().num > v1[i].num)
                v2.push_back(v1[i]);
        }
    }
    cout << v2.size() << "\n";
    for (int i = 0; i < v2.size(); i++)
    {
        cout << v2[i].event << " " << v2[i].num << "\n";
    }
    return 0;
}
