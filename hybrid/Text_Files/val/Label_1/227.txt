n,m=map(int,input().split());l=[]
for i in range(m):l.append(list(map(int,input().split())));
a=[[0 for i in range(n)]for j in range(n)]
for i in l:a[i[0]-1][i[1]-1]=1
for i in range(n):
 a[i][0]=1;a[i][n-1]=1;a[0][i]=1;a[n-1][i]=1
ans=0
for i in range(n):
 for j in range(n):
  if(a[i][j]==0):
   a[i][j]=2
   k=j+1
   while(k<n and a[i][k]==0):
    a[i][k]=2
    k+=1
   k=j-1
   while(k>=0 and a[i][k]==0):
    a[i][k]=2
    k-=1
   k=i+1
   while(k<n and a[k][j]==0):
    a[k][j]=2
    k+=1
   k=i-1
   while(k>=0 and a[k][j]==0):
    a[k][j]=2
    k-=1
   ans+=1
