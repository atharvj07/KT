#include <cstdio>
#include <vector>
#include <algorithm>
int main() {
    int n;
    scanf("%d", &n);
    std::vector<long long> a(n);
    for (int i = 0; i < n; ++i) {
        scanf("%lld", &a[i]);
    }
    std::sort(a.begin(), a.end());
    printf("%lld\n", a[std::max(0, n - 2)] + 1);
    return 0;
}
