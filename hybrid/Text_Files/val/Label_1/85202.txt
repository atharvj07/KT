#include <iostream>
#include <cstdio>
#include <cstring>
using namespace std;

int n, m, a[120][120], t[120][120];

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        string s;
        cin >> s;
        for (int j = 1; j <= m; j++) {
            a[i][j] = (s[j - 1] == 'B');
            t[i][j] = t[i - 1][j] + t[i][j - 1] - t[i - 1][j - 1] + a[i][j];
        }
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (i == j || i == n - j + 1 || i == 1 || j == 1 || i == n || j == m ||
                t[i + ((n - i - 1) >> 1)][j + ((m - j - 1) >> 1)] - t[i - ((n - i - 1) >> 1) - 1][j + ((m - j - 1) >> 1)] - t[i + ((n - i - 1) >> 1)][j - ((m - j - 1) >> 1) - 1] + t[i - ((n - i - 1) >> 1) - 1][j - ((m - j - 1) >> 1) - 1]) {
                cout << i << ' ' << j << endl;
                return 0;
            }
        }
    }
    return 0;
}
