import sys
import bisect

input = sys.stdin.readline
n = int(input())
b = sorted([int(x) for x in input().split()])
k = 0
ans = []
for i in range(n):
    if i+1 in b or i+2 in b:
        k += 1
        ans.append(b[i])
    elif i+3 in b and b[i] != b[i+1]:
        k += 1
        ans.append(b[i])
print(k)
print(' '.join(map(str, ans)))
