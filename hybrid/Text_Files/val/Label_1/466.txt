n = int(input())
d = {}
for i in range(n):
    s = input().split()
    name = s[0]
    k = int(s[1])
    phones = list()
    for j in range(2, 2 + k):
        phones.append(s[j])
    phones.sort()
    d[name] = d.get(name, set()) | set(phones)
print(len(d))
for name in sorted(d):
    phones = list(d[name])
    phones.sort()
    print(name, len(phones), *phones)
