import sys

n = int(input())
x1, x2 = map(int, input().split())
lines = []
for _ in range(n):
    k, b = map(int, input().split())
    lines.append((k, b))

for i in range(n - 1):
    k1, b1 = lines[i]
    for j in range(i + 1, n):
        k2, b2 = lines[j]
        if k1 != k2:
            x = (b2 - b1) / (k1 - k2)
            if x1 < x < x2:
                print("YES")
                sys.exit()
print("NO")
