import sys
input = sys.stdin.readline
n, k = map(int, input().split())
f = list(map(int, input().split()))
w = list(map(int, input().split()))
s = [0] * n
m = [w[i] for i in range(n)]
Q = [i for i in range(n) if f[i] == i]
for _ in range(k):
    Q2 = []
    for i in Q:
        if s[i] + w[f[i]] > s[f[i]]:
            s[f[i]] = s[i] + w[f[i]]
            m[f[i]] = min(m[f[i]], m[i] + w[f[i]])
        elif s[i] + w[f[i]] == s[f[i]]:
            m[f[i]] = min(m[f[i]], m[i] + w[f[i]])
        Q2.append(f[i])
    Q = Q2
for i in range(n):
    print(s[i], m[i])
