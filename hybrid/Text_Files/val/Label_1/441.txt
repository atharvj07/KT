import heapq
n,k=map(int,input().split())
a=list(map(int,input().split()))
g=[[]for i in range(n+1)]
for i in range(n-1):
    x,y=map(int,input().split())
    g[x].append(y)
    g[y].append(x)
vis=[0]*(n+1)
ans=[]
def dfs(x):
    vis[x]=1
    heapq.heappush(ans,a[x-1])
    if len(ans)>k:
        heapq.heappop(ans)
    for y in g[x]:
        if not vis[y]:
            dfs(y)
dfs(1)
print(ans[0])
