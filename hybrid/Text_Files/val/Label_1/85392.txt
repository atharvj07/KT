#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 5008;
int a[N];
ll dp[N][N], ans[100008];
int main() {
  int n, q;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  cin >> q;
  for (int l = n - 1; l >= 0; l--) {
    for (int r = l; r < n; r++) {
      dp[l][r] = (r == l ? a[l] : dp[l][r - 1] ^ a[r]);
    }
  }
  for (int i = 0; i < q; i++) {
    int x, y;
    cin >> x >> y;
    x--, y--;
    ans[i] = dp[x][y];
  }
  for (int i = 0; i < q; i++) cout << ans[i] << "\n";
}
