#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> a(n, vector<int>(m));
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        cin >> a[i][j];
      }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        ans += min({abs(a[i][j] - a[i][m - 1 - j]), abs(a[i][j] - a[n - 1 - i][j]),
                   abs(a[i][j] - a[n - 1 - i][m - 1 - j])});
      }
    }
    cout << ans << endl;
  }
  return 0;
}
