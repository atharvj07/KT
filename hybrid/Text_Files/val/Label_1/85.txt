def can_make_zero(n, a):
    left, right = 0, n - 1
    while left <= right:
        if a[left] < a[right]:
            a[right] -= a[left]
            left += 1
        elif a[left] > a[right]:
            a[left] -= a[right]
            right -= 1
        else:
            left += 1
            right -= 1
    return all(x == 0 for x in a)

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    print("YES" if can_make_zero(n, a) else "NO")
