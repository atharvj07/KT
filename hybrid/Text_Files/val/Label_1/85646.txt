#include <vector>
#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> c(n + 1);
  for (int i = 1; i <= n; ++i) {
    cin >> c[i];
  }
  string ans(n, 'a');
  vector<bool> used(26, false);
  for (int i = n; i >= 1; --i) {
    int l = c[i];
    for (int j = 0; j < 26; ++j) {
      if (!used[j] && l > 0) {
        used[j] = true;
        l--;
        for (int k = 0; k < i; ++k) {
          ans[k] = char(j + 'a');
        }
      }
    }
  }
  cout << ans << endl;
  return 0;
}
