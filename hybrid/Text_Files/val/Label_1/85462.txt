#include<iostream>
#include<vector>
using namespace std;
int main(){
    int n;
    cin>>n;
    if(n%2==1)cout<<"NO\nNO";
    else{
        cout<<"YES\n";
        vector<int> ans1,ans2;
        for(int i=1;i<=n;i++){
            if(i%2==1)ans1.push_back(i+1),ans2.push_back(i);
            else ans1.push_back(i-1),ans2.push_back(i);
        }
        for(auto p:ans1)cout<<p<<" ";
        cout<<"\n";
        for(auto p:ans2)cout<<p<<" ";
        cout<<"\n";
    }
}
