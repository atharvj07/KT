n=int(input())
l=list(map(int,input().split()))
x=[[0]*21 for _ in range(n+1)]
for i in range(n-1,-1,-1):
  for j in range(20):
    x[i][j]=min(x[i+1][j],x[i+1][(j+l[i])%20],x[i+1][(j+l[i]+1)%20])
    a=[0]*20
    a[j]=1
    for k in range(i+1,n):
      for w in range(20):
        a[w]=min(a[w],a[(w+l[k])%20],a[(w+l[k]+1)%20])
    x[i][j]=min(x[i][j],sum(a)%(10))
print(*x[0])
