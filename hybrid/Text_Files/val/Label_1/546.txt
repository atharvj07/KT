import sys
input = sys.stdin.readline
n,k = map(int,input().split())
p = list(map(int,input().split()))
c = list(map(int,input().split()))
p.sort()
c_ = [[] for i in range(n)]
for i in range(n):
    pos = p.index(p[i])
    c_[pos].append(c[i])
for i in range(1,n):
    c_[i] += c_[i-1]
ans = []
for i in range(n):
    if i>0 and i+k<n:
        ans.append(c_[i+k-1]-c_[i-1])
    elif i==0 and k>0:
        ans.append(c_[k-1])
    else:
        ans.append(c[i])
print(*ans)
