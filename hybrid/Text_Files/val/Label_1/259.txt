mod = 1000000007
n = int(input())
adj = [[] for i in range(n)]
for i in range(n):
    l = list(map(int, input().split()))
    for j in range(n):
        if i != j and l[j]:
            adj[i].append(j)
dp = [[[[0] * 2 for l in range(2)] for j in range(1 << n)] for i in range(n)]
dp[0][1 << 0][0][0] = 1
dp[0][1 << 0][0][1] = 1
dp[0][1 << 0][1][0] = 0
dp[0][1 << 0][1][1] = 0
for i in range(n):
    for mask in range(1 << n):
        for on in range(2):
            for in_cycle in range(2):
                for j in adj[i]:
                    if not (mask & (1 << j)):
                        dp[j][mask | (1 << j)][(on | (i == j))][(in_cycle | (on and (i == j)))] += dp[i][mask][on][in_cycle]
                        dp[j][mask | (1 << j)][(on | (i == j))][(in_cycle | (on and (i == j)))] %= mod
print(dp[n - 1][(1 << n) - 1][1][1])
