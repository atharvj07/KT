#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  if (n & 1) {
    cout << "NO\nNO\n";
  } else {
    cout << "YES\n";
    for (int i = 1; i < n; i += 2)
      cout << i + 1 << " ";
    for (int i = 2; i <= n; i += 2)
      cout << i - 1 << " ";
    cout << "\nYES\n";
    for (int i = 1; i < n; i += 2)
      cout << i << " ";
    for (int i = 2; i <= n; i += 2)
      cout << i + 1 << " ";
    cout << "\n";
  }
}
