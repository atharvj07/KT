#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int maxn = 4005;
int n, p[maxn], v[maxn], d[maxn];

int main() {
  cin >> n;
  for (int i = 1; i <= n; ++i) cin >> v[i] >> d[i] >> p[i];
  vector<int> ans;
  queue<int> q;
  for (int i = 1; i <= n; ++i) q.push(i);
  while (!q.empty()) {
    bool f = true;
    int t = q.front();
    for (int i = 1; i <= n; ++i) {
      if (i != t) { p[i] -= v[t]; }
    }
    while (!q.empty() && p[t] >= 0) {
      q.pop();
      f = false;
      ans.push_back(t);
      for (int i = 1; i <= n; ++i) {
        if (i != t) { p[i] -= d[t]; }
      }
      if (!q.empty()) {
        t = q.front();
        for (int i = 1; i <= n; ++i) {
          if (i != t) { p[i] -= v[t]; }
        }
      } else
        break;
    }
    if (f) {
      ans.push_back(t);
      q.pop();
    }
  }
  cout << ans.size() << endl;
  for (auto x : ans) cout << x << " ";
  cout << endl;
  return 0;
}
