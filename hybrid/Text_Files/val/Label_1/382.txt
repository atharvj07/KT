n=int(input())
a=list(map(int,input().split()))
q,ans=[a.count(i) for i in range(1,n+1)].count(0),[0]*n
for i in a:ans[i-1]+=1
for i,x in enumerate(ans):
    if x==0:
        for j in range(n):
            if q==0:break
            if ans[j]==2:
                ans[i]=1
                ans[j]-=1
                a[i]=j+1
                q-=1
print(q)
print(' '.join(map(str,a)))
