#include<bits/stdc++.h>
using namespace std;
char a[15][15],b[15][15];
int n,f;
bool check(int x)
{
    if(x==0){for(int i=0;i<n;i++)for(int j=0;j<n;j++)if(a[i][j]!=b[i][j])return false;}
    if(x==1){for(int i=0;i<n;i++)for(int j=0;j<n;j++)if(a[n-i-1][j]!=b[i][j])return false;}
    if(x==2){for(int i=0;i<n;i++)for(int j=0;j<n;j++)if(a[i][n-j-1]!=b[i][j])return false;}
    if(x==3){for(int i=0;i<n;i++)for(int j=0;j<n;j++)if(a[n-i-1][n-j-1]!=b[i][j])return false;}
    if(x==4){for(int i=0;i<n;i++)for(int j=0;j<n;j++)if(a[j][i]!=b[i][j])return false;}
    if(x==5){for(int i=0;i<n;i++)for(int j=0;j<n;j++)if(a[i][n-j-1]!=b[i][j])return false;}
    if(x==6){for(int i=0;i<n;i++)for(int j=0;j<n;j++)if(a[n-j-1][i]!=b[i][j])return false;}
    if(x==7){for(int i=0;i<n;i++)for(int j=0;j<n;j++)if(a[n-j-1][n-i-1]!=b[i][j])return false;}
    return true;
}
int main()
{
    cin>>n;
    for(int i=0;i<n;i++)for(int j=0;j<n;j++)cin>>a[i][j];
    for(int i=0;i<n;i++)for(int j=0;j<n;j++)cin>>b[i][j];
    for(int i=0;i<8;i++)
    {
        if(check(i))
        {
            cout<<"Yes";
            return 0;
        }
    }
    cout<<"No";
    return 0;
}
