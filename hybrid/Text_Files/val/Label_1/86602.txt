import java.util.Scanner;

public class TeamScientificDecathlonOlympias {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] skills = new int[n];

        for (int i = 0; i < n; i++) {
            skills[i] = scanner.nextInt();
        }

        int[] count = new int[4];
        for (int skill : skills) {
            count[skill]++;
        }

        int w = Math.min(count[1], Math.min(count[2], count[3]));

        System.out.println(w);

        for (int i = 0, j = 1, k = 1, l = 1; w > 0; w--) {
            while (skills[i] != 1 || count[1] == 0) {
                i++;
            }
            while (skills[j] != 2 || count[2] == 0) {
                j++;
            }
            while (skills[k] != 3 || count[3] == 0) {
                k++;
            }
            System.out.println((i + 1) + " " + (j + 1) + " " + (k + 1));
            count[1]--;
            count[2]--;
            count[3]--;
            i++;
            j++;
            k++;
            l++;
        }
    }
}
