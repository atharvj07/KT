import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt();
        int b = in.nextInt();
        if (a > b) {
            System.out.println("Masha");
        } else if (a < b) {
            System.out.println("Dasha");
        } else {
            System.out.println("Equal");
        }
    }
}
