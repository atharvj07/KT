import sys
n, k = map(int, input().split())
p = list(map(int, input().split()))
c = list(map(int, input().split()))
lst = []
for i in range(n):
    lst.append((p[i], c[i]))
lst.sort()
dp = [-1] * n
dp[n-1] = lst[n-1][1]
for i in range(n-2,-1,-1):
    mx = -1
    for j in range(i+1,min(i+k+1,n)):
        if lst[i][0] > lst[j][0]:
            mx = max(mx,dp[j])
    dp[i] = max(lst[i][1],mx+lst[i][1])
print(*dp)
