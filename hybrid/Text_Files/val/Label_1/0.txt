def longest_almost_increasing_subsequence(a):
    n = len(a)
    dp = [1] * n
    for i in range(1, n):
        for j in range(i):
            if min(a[j], a[j + 1]) <= min(a[i - 1], a[i]) and dp[j] + 1 > dp[i]:
                dp[i] = dp[j] + 1
    return max(dp)

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    print(longest_almost_increasing_subsequence(a))
