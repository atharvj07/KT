import java.io.*;
import java.math.*;
import java.util.*;

class Main {
  static int mod = 1000000007;
  static long[][] dp;
  static int[][] g;
  static int n, m;
  static boolean[] vis;

  public static void dfs(int x) {
    vis[x] = true;
    for (int y : g[x]) {
      if (!vis[y]) {
        dfs(y);
      }
    }
  }

  public static long solve(int mask, int u) {
    if (dp[mask][u] != -1) {
      return dp[mask][u];
    }
    long ans = 0;
    if (mask == (1 << m) - 1) {
      return ans = 1;
    }
    for (int i = 0; i < m; i++) {
      if ((mask & (1 << i)) == 0) {
        int v = g[u][i];
        if (vis[v]) {
          ans = (ans + solve(mask | (1 << i), v)) % mod;
        }
      }
    }
    return dp[mask][u] = ans;
  }

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
    StringTokenizer st = new StringTokenizer(br.readLine());
    n = Integer.parseInt(st.nextToken());
    m = Integer.parseInt(st.nextToken());
    dp = new long[1 << m][n + 1];
    g = new int[n + 1][m];
    vis = new boolean[n + 1];
    for (int i = 0; i < m; i++) {
      st = new StringTokenizer(br.readLine());
      int x = Integer.parseInt(st.nextToken());
      int y = Integer.parseInt(st.nextToken());
      g[x][i] = y;
    }
    dfs(1);
    long ans = 0;
    for (int i = 0; i < m; i++) {
      ans = (ans + solve(1 << i, g[1][i])) % mod;
    }
    bw.write(ans + "\n");
    bw.flush();
  }
}
