#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::vector<int> v(2 * n);
    for (int& x : v) std::cin >> x;
    int sum = 0;
    for (int x : v) sum += (x & 1);
    if (sum & 1)
      std::cout << "Yes\n";
    else
      std::cout << "No\n";
  }
}
