import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);

    int n = input.nextInt();
    int m = input.nextInt();

    int[] p = new int[n];
    for (int i = 0; i < n; i++) p[i] = input.nextInt();

    int[] s = new int[n];
    for (int i = 0; i < n; i++) s[i] = input.nextInt();

    int[] b = new int[n];
    for (int i = 0; i < n; i++) b[i] = input.nextInt();

    int[] inc = new int[m];
    for (int i = 0; i < m; i++) inc[i] = input.nextInt();

    int[] pref = new int[m];
    for (int i = 0; i < m; i++) pref[i] = input.nextInt();

    for (int i = 0; i < m; i++) {
      int count = 0;
      for (int j = 0; j < n; j++) {
        if (p[j] <= inc[i] && inc[i] <= s[j] && Math.abs(b[j] - pref[i]) <= inc[i] - p[j]) count++;
      }
      System.out.print(count + " ");
    }

    input.close();
  }
}
