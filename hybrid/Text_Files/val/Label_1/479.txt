import sys

n, m = map(int, input().split())
a = list(map(int, input().split()))
for i in range(m):
    t, *q = map(int, input().split())
    if t == 1:
        a[q[0] - 1] = q[1]
    elif t == 2:
        l, r = q
        print(
            ((1 + (r - l + 1) * (r - l)) * ((a[l - 1] + a[r - 1]) // 2)) % 1000000000)
    else:
        l, r, d = q
        for j in range(l - 1, r):
            a[j] += d
