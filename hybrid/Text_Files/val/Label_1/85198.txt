#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m, k, a;
  cin >> n >> m >> k;
  vector<int> v(n + 1), c(m);
  for (int i = 1; i <= n; i++) { cin >> a; v[a]++; }
  for (int i = 0; i < m; i++) { cin >> a; c[a]++; }
  int ans = 0;
  for (int i = 0; i < m; i++) { ans += c[i] - (v[i] > 0); }
  cout << ans << endl;
  return 0;
}
