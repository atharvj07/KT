#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const int MOD = 1e9 + 7;

vector<ll> pow10;

void init() {
  pow10.push_back(1);
  for (int i = 1; i <= 1e6; i++) pow10.push_back(pow10.back() * 10 % MOD);
}

ll fastPow(ll x, ll p) {
  if (p == 0) return 1;
  ll ans = fastPow(x, p / 2);
  ans = ans * ans % MOD;
  if (p % 2) ans = ans * x % MOD;
  return ans;
}

int main() {
  init();
  int n, m;
  cin >> n >> m;
  string p;
  cin >> p;
  vector<int> y(m);
  for (int i = 0; i < m; i++) cin >> y[i], y[i]--;
  int sz = p.size();
  ll ans = 0;
  for (int i = 0; i < m; i++) {
    int j = i + 1;
    while (j < m && y[j] <= y[i] + sz - 1) j++;
    j--;
    ans = (ans + pow10[y[j] - (i == 0 ? 0 : y[i - 1]) - sz + 1]) % MOD;
  }
  cout << ans << endl;
  return 0;
}
