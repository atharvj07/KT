import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int numFriends = Integer.parseInt(br.readLine());
        StringTokenizer st = new StringTokenizer(br.readLine());
        int[] friendCounts = new int[numFriends];
        for (int i = 0; i < numFriends; i++) {
            friendCounts[i] = Integer.parseInt(st.nextToken());
        }

        int sum = 0;
        for (int friendCount : friendCounts) {
            sum += friendCount;
        }

        int numWays = 0;
        for (int i = 1; i <= 5; i++) {
            for (int j = 1; j <= 5; j++) {
                int total = i + sum + j;
                if (total % numFriends + 1 == 1) {
                    numWays++;
                }
            }
        }

        System.out.println(numWays);
    }
}
