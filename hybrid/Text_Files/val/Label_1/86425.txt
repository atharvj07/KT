import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.List;

public class Main {
    private final static int INF = 1000000000;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int[] n_m = Arrays.stream(br.readLine().split(" ")).mapToInt(x -> Integer.parseInt(x)).toArray();
        int n = n_m[0], m = n_m[1];

        int[] p = Arrays.stream(br.readLine().split(" ")).mapToInt(x -> Integer.parseInt(x)).toArray();
        int[] s = Arrays.stream(br.readLine().split(" ")).mapToInt(x -> Integer.parseInt(x)).toArray();
        int[] b = Arrays.stream(br.readLine().split(" ")).mapToInt(x -> Integer.parseInt(x)).toArray();
        int[] inc = Arrays.stream(br.readLine().split(" ")).mapToInt(x -> Integer.parseInt(x)).toArray();
        int[] pref = Arrays.stream(br.readLine().split(" ")).mapToInt(x -> Integer.parseInt(x)).toArray();

        int[] count = new int[m];
        for (int j = 0; j < m; ++j) {
            for (int i = 0; i < n; ++i) {
                if (p[i] <= inc[j] && s[i] >= inc[j] && Math.abs(b[i] - pref[j]) <= inc[j] - p[i]) {
                    ++count[j];
                }
            }
        }
        StringBuilder sb = new StringBuilder();
        for (int x : count) {
            sb.append(x).append(' ');
        }
        System.out.println(sb.toString());
    }
}
