#include <iostream>
using namespace std;

int main() {
  long long n, m, k;
  cin >> n >> m >> k;
  if (n * m % k != 0) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
    long long a = n * m / k - 1;
    cout << (a * k / m + 1) << " 0" << endl;
    cout << "0 " << a % k << endl;
    cout << n << " " << k - a % k - 1 << endl;
  }
  return 0;
}
