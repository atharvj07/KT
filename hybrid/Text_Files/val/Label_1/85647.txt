#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll c[2002];
string s;
#define vi(x) vector<x>
#define pb push_back
const ll mod = 1000000007;
inline ll add(ll x, ll y) { x += y; if(x >= mod) x -= mod; return x;}
inline ll mul(ll x, ll y) { ll z = 1LL * x * y; return z % mod;}
inline ll power(ll x, ll y) { if(y == 0) return 1; ll p = power(x, y / 2); p = mul(p, p); if(y % 2) return mul(p, x); return p;}
#define endl "\n"
#define all(s) s.begin(), s.end()
int main() {
	ll n;
	cin >> n;
	for(ll i = 1; i <= n; i++) cin >> c[i];
	if(c[n] > 1) {
		cout << "Impossible" << endl;
		return 0;
	}
	if(c[n]) {
		for(ll i = 0; i < n; i++) {
			s += 'a' + i;
		}
		cout << s << endl;
		return 0;
	}
	for(ll i = 0; i < n; i++) {
		if(c[i]) {
			char ch = 'a';
			if(i > 0) ch = s[i - 1] + 1;
			while(c[i]--) s += ch;
		}
	}
	cout << s << endl;
	return 0;
}
