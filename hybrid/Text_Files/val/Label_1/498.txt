import sys

def fast_pow(base, e, mod):
    ans = 1
    while e > 0:
        if e % 2 == 1:
            ans = ans * base % mod
        base = base * base % mod
        e //= 2
    return ans

m = 1000000007

def solve(a, b, m):
    if a > b:
        return -1
    if a == b:
        return 1, [a]
    ans = []
    L = len(str(b))
    ans.append(a)
    for i in range(1, L):
        cur = a * fast_pow(10, i, m) % m
        diff = (b - cur) % m
        ans.append(diff)
        a = (a + diff) % m
    return len(ans), ans

q = int(sys.stdin.readline().strip())
for _ in range(q):
    a, b, m = map(int, sys.stdin.readline().strip().split())
    ans, seq = solve(a, b, m)
    if ans == -1:
        print(-1)
    else:
        print(ans)
        print(*seq)
