n, m = map(int, input().split())
a = list(map(int, input().split()))
b = []
for i in range(m):
    x, y = map(int, input().split())
    b.append((x, y))
b.sort()
ans = 0
for i in b:
    x, y = i
    if x > ans:
        ans = x
    if a[y - 1] > 0:
        a[y - 1] -= 1
    else:
        ans += 1
print(ans)
