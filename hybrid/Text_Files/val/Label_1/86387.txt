import java.util.Scanner;

public class MartianLuckyNumbers {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        int b = scanner.nextInt();
        int n = scanner.nextInt();
        scanner.nextLine(); 
        String s = scanner.nextLine();

        int[] prefixSums = new int[n + 1];
        int[] powersOfK = new int[n + 1];

        powersOfK[0] = 1;
        for (int i = 1; i <= n; i++) {
            powersOfK[i] = (powersOfK[i - 1] * k) % k;
        }

        for (int i = 0; i < n; i++) {
            int digit = s.charAt(i) - '0';
            prefixSums[i + 1] = (prefixSums[i] + digit * powersOfK[i]) % k;
        }

        int luckySubstrings = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j <= n; j++) {
                int sum = (prefixSums[j] - prefixSums[i] + k) % k;
                if (sum == b) {
                    luckySubstrings++;
                }
            }
        }

        System.out.println(luckySubstrings);
    }
}
