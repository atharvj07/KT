n,m,x,y,z,p=list(map(int,input().split()))
d=[[(i,j) for i in range(1,n+1)]for j in range(1,m+1)]
for i in range(0,x%4):d=list(zip(*d[::-1]))
for i in range(0,y%2):d=[i[::-1]for i in d]
for i in range(0,z%4):d=list(zip(*d[::-1]))
for i in range(p):
    a,b=list(map(int,input().split()))
    print(*d[a-1][b-1])
