import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);

		int n = in.nextInt();
		int q = in.nextInt();

		int mx = 0;
		int maxNum = 0;
		TreeMap<Integer, Integer> map = new TreeMap<>();

		while (q-- > 0) {
			int a = in.nextInt();
			int v = in.nextInt();

			map.put(a, map.getOrDefault(a, 0) + v);

			if (map.get(a) > mx) {
				mx = map.get(a);
				maxNum = a;
			} else if (map.get(a) == mx) {
				maxNum = Math.min(maxNum, a);
			}

			System.out.println(maxNum + " " + mx);
		}
	}
}
