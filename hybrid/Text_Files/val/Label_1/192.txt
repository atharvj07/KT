import sys
for _ in range(int(input())):
    n, W = map(int, input().split())
    a = [int(_) for _ in input().split()]
    a.sort()
    m, rem = divmod(W, 2), 0
    for i in range(n):
        if rem < m:
            rem += a[i]
            m -= 1
        else:
            sys.stdout.write(f'{i+1}\n')
            for j in range(n):
                if a[j] in a[:i+1]:
                    sys.stdout.write(f"{j+1} ")
            break
    else:
        sys.stdout.write('-1\n')
