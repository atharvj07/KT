import math

n, m = map(int, input().split())
k = list(map(int, input().split()))
s = []
for _ in range(m):
    s.append(list(map(int, input().split())))

ans = math.ceil(max(k) / 2) * 2
for i in range(m):
    if k[s[i][1] - 1] > 0:
        ans = min(ans, s[i][0])
        k[s[i][1] - 1] -= 1
print(ans)
