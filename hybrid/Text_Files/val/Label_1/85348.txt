#include <iostream>
using namespace std;
long long abs(long long x) {
	return x > 0 ? x : -x;
}
int main() {
	int t;
	cin >> t;
	while (t--) {
		long long n, x, y, d;
		cin >> n >> x >> y >> d;
		if (x == y) {
			cout << 0 << endl;
		} else if ((y - x) % d == 0 && y >= x) {
			cout << (y - x) / d << endl;
		} else if ((x - y) % d == 0 && x > y) {
			cout << (x - y) / d << endl;
		} else if ((n - x + y) % d == 0 && y < x) {
			cout << (n - x + y) / d << endl;
		} else if ((n - y + x) % d == 0 && x < y) {
			cout << (n - y + x) / d << endl;
		} else {
			cout << -1 << endl;
		}
	}
	return 0;
}
