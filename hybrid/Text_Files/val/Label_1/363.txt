def min_erase_for_bipartite(n, b):
    def highest_power_of_2(x):
        return x & (~(x - 1))

    b_powers = [highest_power_of_2(x) for x in b]
    count = {}
    for bp in b_powers:
        if bp in count:
            count[bp] += 1
        else:
            count[bp] = 1
    max_count = max(count.values())
    erase_count = n - max_count
    max_power = max(count, key=count.get)
    erase_elements = [b[i] for i in range(n) if b_powers[i] != max_power]
    return erase_count, erase_elements

n = int(input().strip())
b = list(map(int, input().strip().split()))

erase_count, erase_elements = min_erase_for_bipartite(n, b)
print(erase_count)
print(*erase_elements)
