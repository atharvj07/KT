#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> boxes;
  for (int i = 0; i < n; i++) {
    int k, a;
    cin >> k >> a;
    boxes.push_back({k, a});
  }
  sort(boxes.begin(), boxes.end());
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (boxes[i].first > ans) {
      ans = boxes[i].first;
    }
    boxes[i].second--;
    if (boxes[i].second == 0) {
      i--;
      ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
