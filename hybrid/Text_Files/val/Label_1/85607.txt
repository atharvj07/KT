#include <bits/stdc++.h>
#define int long long
using namespace std;
const int mxN = 2e5 + 5;
const int mod = 1e9 + 7;
int n, t, c;
int a[mxN];
int32_t main() {
    cin >> n >> t >> c;
    for (int i = 0; i < n; i++) cin >> a[i];
    int l = 0, cnt = 0, ans = 0;
    for (int r = 0; r < n; r++) {
        if (!cnt) l = r;
        cnt += a[r] > t;
        if (cnt == c) {
            ans = (ans + (l + 1)) % mod;
            cnt -= a[l] > t;
            l++;
        }
    }
    cout << ans;
}
