#include <bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin >> n;
    string s;
    cin >> s;
    int cnt = 0, k = 0;
    vector<int> v;
    for (char c : s)
    {
        if (c == 'B')
        {
            cnt++;
            if (cnt > 0 && (cnt > k || k == 0))
            {
                k = cnt;
                v.clear();
                v.push_back(cnt);
            }
        }
        else
            cnt = 0;
    }
    cout << v.size() << endl;
    for (auto x : v)
        cout << x << " ";
}
