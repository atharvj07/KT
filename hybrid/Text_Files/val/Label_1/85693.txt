#include<bits/stdc++.h>
using namespace std;
#define ll long long

map<ll,ll>pass;

void upd(ll l,ll r,ll v)
{
	while(l<=r){
		pass[l]+=v;
		l=(l+1)/2;
	}
}

long long passQuery(ll l,ll r){
	ll s=0;
	while(l<=r){
		if(l%2==1)s+=pass[l++];
		if(r%2==0)s+=pass[r--];
		l/=2;r/=2;
	}
	return s;
}

int main()
{
	long long q,a,b,c;
	cin>>q;
	while(q--){
		cin>>a;
		if(a==1){
			cin>>b>>c>>a;
			upd(b,c,a);
		}
		else{
			cin>>b>>c;
			cout<<passQuery(b,c)<<endl;
		}
	}
	return 0;
}
