n = int(input())
t = list(map(int, input().split()))
T = int(input())

t.sort()
max_visits = 1

for i in range(n):
    current_visits = 1
    for j in range(i + 1, n):
        if t[j] - t[i] <= T:
            current_visits += 1
        else:
            break
    max_visits = max(max_visits, current_visits)

print(max_visits)
