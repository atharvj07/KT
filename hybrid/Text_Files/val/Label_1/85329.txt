#include <bits/stdc++.h>
using namespace std;
#define LL long long
const int MAXN = 1000006;
bool God[MAXN + 1];
int main(){
    int T, n;
    cin >> T;
    God[1] = God[2] = 0, God[3] = 1;
    for (int i = 1; i <= MAXN; i++){
        if (i % 2 == 1){
            God[i + 1] = !God[i - 1];
        }else{
            God[i + 1] = !God[i - 2];
        }
    }
    while (T--){
        cin >> n;
        int p = 1;
        while (God[p] == 0){
            p++;
        }
        vector<int> ans;
        while (n){
            if (God[n]){
                if (ans.size() > 0 && p > 2){
                    ans.push_back(2);
                    n -= 2;
                }else{
                    ans.push_back(p);
                    n -= p;
                }
            }else{
                ans.push_back(1);
                n--;
            }
        }
        reverse(ans.begin(), ans.end());
        for (int i = 0; i < ans.size(); i++){
            cout << ans[i] << ' ';
        }
        cout << '\n';
    }
}
