import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        Map<Character, Integer>[] maps = new Map[m];
        for (int i = 0; i < m; i++) {
            maps[i] = new HashMap<>();
        }
        for (int i = 0; i < n; i++) {
            String s = scanner.next();
            for (int j = 0; j < m; j++) {
                maps[j].merge(s.charAt(j), 1, Integer::sum);
            }
        }
        long answer = 0;
        for (int i = 0; i < m; i++) {
            for (Character character : maps[i].keySet()) {
                if (maps[i].get(character) > 1) {
                    answer += maps[i].get(character) - 1;
                }
            }
        }
        System.out.println(answer);
    }
}
