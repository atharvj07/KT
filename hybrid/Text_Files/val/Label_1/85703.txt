#include <iostream>
#include <vector>

using namespace std;

int main() {
  int s, b;
  cin >> s >> b;
  vector<int> a(s), d(b), g(b);
  for (int i = 0; i < s; i++) cin >> a[i];
  for (int i = 0; i < b; i++) cin >> d[i] >> g[i];
  vector<int> ans(s);
  for (int i = 0; i < s; i++) {
    for (int j = 0; j < b; j++) {
      if (a[i] >= d[j]) ans[i] += g[j];
    }
  }
  for (int i = 0; i < s; i++) cout << ans[i] << " ";
  return 0;
}
