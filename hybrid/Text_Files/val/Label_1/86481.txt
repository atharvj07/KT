import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = scan.nextInt();

        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 1) {
                ans += i;
                a[i] = 0;
            }
        }
        for (int i = n - 1; i >= 0; i--) {
            if (a[i] == 1) {
                ans += n - 1 - i;
                a[i] = 0;
            }
        }
        System.out.println(ans);
    }
}
