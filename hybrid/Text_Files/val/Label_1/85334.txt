#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    vector<int> a(n);
    for (int &i : a) cin >> i;
    int cur = 0, mx = 0;
    for (int i : a) {
      cur += i / x + (i % x != 0);
      mx += (i + x - 1) / x;
    }
    cout << cur << ' ' << mx << '\n';
  }
}
