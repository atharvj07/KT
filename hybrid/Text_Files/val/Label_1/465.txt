from collections import defaultdict

n = int(input())
name_nums = defaultdict(lambda: [])
for _ in range(n):
    name, k, *nums = input().split()
    name_nums[name] += [num for num in nums if not any(num in n for n in name_nums[name] if len(n) > len(num))]

print(len(name_nums))
for name, nums in sorted(name_nums.items()):
    print(name, len(nums), *nums)
