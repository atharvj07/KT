#include <bits/stdc++.h>
#define ll long long
using namespace std;
ll a[(int)2e5 + 5];
ll f[(int)2e5 + 5];
ll g[(int)2e5 + 5];
int main() {
  ll n;
  cin >> n;
  for (int i = 1; i <= n; ++i) { cin >> a[i]; }
  ll ans = 0;
  if (n <= 3) {
    for (int i = 1; i <= n; ++i) { ans += a[i]; }
  } else {
    for (int i = 1; i <= n; ++i) {
      f[(i + 1) % n + 1] += a[i];
      g[i] += a[(i + 1) % n + 1];
    }
    for (int i = 1; i <= n; ++i) {
      f[i] = max(f[i], f[i - 1]);
      g[i] = max(g[i], g[i - 1]);
    }
    for (int i = 1; i <= n; ++i) {
      ans = max({ans, f[(i + 1) % n + 1] + g[i - 1],
                f[i - 1] + g[(i + 1) % n + 1]});
    }
  }
  cout << ans << endl;
}
