import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
q = int(input())
ms = [[] for _ in range(n+1)]
for _ in range(q):
    s, t, u = map(int, input().split())
    for i in ms[s]:
        if i[0] == t:
            ms[s].remove(i)
            break
    if u != 0:
        ms[s].append([t, u])
    ms[s].sort()
    ans = 0
    res = [0] * n
    for k in ms[s]:
        t, u = k
        res[s-1] += 1
        if u != 0:
            res[u-1] += 1
            ans += 1
        if res[s-1] >= a[s-1]:
            break
    for i, v in enumerate(res):
        ans += max(0, a[i] - v)
    print(ans)
