import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] lengths = new int[n];
        for (int i = 0; i < n; i++) {
            lengths[i] = input.nextInt();
        }
        Arrays.sort(lengths);
        int tallest = 0;
        int numTowers = 0;
        for (int i = 0; i < n; i++) {
            if (tallest < lengths[i]) {
                tallest = lengths[i];
                numTowers++;
            }
        }
        System.out.println(tallest + " " + numTowers);
    }
}
