n, m = map(int, input().split())
banned = {tuple(map(int, input().split())) for _ in range(m)}

# For n even, no two chips can swap places, so we focus on not hitting banned cells.
# For n odd, center row/column can't be used due to potential swap, reducing usable border by n-1
usable_border = 4 * (n - 1) if n % 2 == 0 else 4 * (n - 1) - (n - 1)

# Check if any of the banned cells is on the border but not a corner
for x, y in banned:
    if x == 1 or x == n or y == 1 or y == n:
        if not (x == 1 and (y == 1 or y == n)) and not (x == n and (y == 1 or y == n)):
            usable_border -= 1

# Output the maximum number of points Gerald can earn
print(max(0, min(usable_border, n - 1)))
