#include <iostream>
#include <unordered_map>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  unordered_map<int, pair<int, int>> X;
  unordered_map<int, pair<int, int>> Y;
  vector<pair<int, int>> clues(n);
  for (int i = 0; i < n; i++) {
    int a, b;
    cin >> a >> b;
    clues[i] = {a, b};
  }
  for (int i = 0; i < n; i++) {
    int x, y;
    cin >> x >> y;
    X[x] = clues[i];
    Y[y] = clues[i];
  }
  auto itx = X.begin();
  auto ity = Y.begin();
  cout << itx->first + itx->second.first << ' ' << ity->first + ity->second.second
       << endl;
  return 0;
}
