#include<bits/stdc++.h>
using namespace std;
typedef long long int ll;
int main(){
  int t;
  cin>>t;
  while(t--){
    ll n,a[100000];
    cin>>n;
    for(ll i=0;i<n;++i){
      cin>>a[i];
    }
    ll x[n]={0};
    ll f=0;
    for(ll i=0;i<n;++i){
      ll y=(i+a[i%n]+n)%n;
      if(x[y]==1)f=1;
      x[y]=1;
    }
    if(f)cout<<"NO\n";
    else cout<<"YES\n";
  }
  return 0;
}
