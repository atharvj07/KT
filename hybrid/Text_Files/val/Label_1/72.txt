t = int(input().strip())
for _ in range(t):
    a, b = map(int, input().strip().split())
    s = list(input().strip())
    n = len(s)
    possible = True

    for i in range(n):
        if s[i] == '?':
            if s[n-1-i] != '?':
                s[i] = s[n-1-i]
                if s[i] == '0':
                    a -= 1
                else:
                    b -= 1
        else:
            if s[n-1-i] == '?':
                s[n-1-i] = s[i]
                if s[i] == '0':
                    a -= 1
                else:
                    b -= 1
            elif s[n-1-i] != s[i]:
                possible = False
                break
            else:
                if s[i] == '0':
                    a -= 1
                else:
                    b -= 1

    for i in range(n//2):
        if s[i] == '?':
            if a > 1:
                s[i] = s[n-1-i] = '0'
                a -= 2
            elif b > 1:
                s[i] = s[n-1-i] = '1'
                b -= 2
            else:
                possible = False
                break

    if n % 2 == 1 and s[n//2] == '?':
        if a > 0:
            s[n//2] = '0'
            a -= 1
        elif b > 0:
            s[n//2] = '1'
            b -= 1
        else:
            possible = False
    
    if a != 0 or b != 0 or not possible:
        print(-1)
    else:
        print("".join(s))
