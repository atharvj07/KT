#include <iostream>
#include <string>

using namespace std;

int main() {
  int m, s;
  cin >> m >> s;

  int ms = s / m;
  int md = s % m;

  string minNum, maxNum;
  if (md == 0) {
    for (int i = 0; i < m; i++) minNum += '0' + ms;
  } else {
    for (int i = 0; i < md; i++) minNum += '0' + ms + 1;
    for (int i = 0; i < m - md; i++) minNum += '0' + ms;
  }

  for (int i = 0; i < m; i++) maxNum += '0' + ms;
  for (int i = 0; i < md; i++) maxNum[i] = '9';

  if (minNum <= maxNum) cout << minNum << ' ' << maxNum << '\n';
  else cout << "-1 -1\n";

  return 0;
}
