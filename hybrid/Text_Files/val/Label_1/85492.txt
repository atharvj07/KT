#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
	int n; cin >> n;
	ll mx = LLONG_MIN;
	vector<ll> a(n);
	for (auto &x : a) cin >> x;
	for (int i = 0; i < n; i++) {
		ll sm = a[i] + a[(i + 1) % n];
		ll x = max(sm, max(a[(i - 1 + n) % n], a[(i + 1) % n]));
		mx = max(mx, x);
	}
	cout << mx << endl;
	return 0;
}
