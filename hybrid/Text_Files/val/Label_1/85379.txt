#include <bits/stdc++.h>

using namespace std;

const int maxN = 5005;

int n, k;
string s, t;

int main() {
    cin >> n >> k >> s;
    while (k >= (int) s.length()) {
        t += s;
        k -= (int) s.length();
    }
    for (int i = 0; i < k; i++) t += s[i];
    cout << t << '\n';
    return 0;
}
