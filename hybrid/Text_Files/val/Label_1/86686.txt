import java.util.*;

 class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 1; i < n; i++) {
            if (a[i] < a[i - 1]) {
                ans += a[i - 1] / a[i];
                if (a[i - 1] % a[i] != 0) {
                    ans++;
                }
                a[i - 1] = -2;
            }
        }
        if (a[n - 1] < 0) {
            System.out.println(-1);
        } else {
            System.out.println(ans);
        }
    }
}
