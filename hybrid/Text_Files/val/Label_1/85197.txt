#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, k;
    cin >> n >> m >> k;
    vector<int> c(n + 1);
    for (int i = 1; i <= n; i++) cin >> c[i];
    vector<pair<int, int>> need(m);
    for (int i = 0; i < m; i++) {
        int l, r;
        cin >> l >> r;
        need[i] = {l, r};
    }
    vector<int> cnt(k + 1);
    for (int i = 0; i < m; i++) {
        if (c[need[i].first] != c[need[i].second]) {
            cnt[c[need[i].first]]++;
            cnt[c[need[i].second]]++;
        }
    }
    int ans = 0;
    for (int i = 1; i <= k; i++) {
        if (cnt[i] > 0) ans += cnt[i] / 2;
    }
    cout << ans << endl;
}
