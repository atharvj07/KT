n, k = map(int, input().strip().split())
f=[int(x) for x in input().strip().split()]
w=[int(x) for x in input().strip().split()]
s=[0]*n
m=[0]*n
for i in range(n):
    if k==1:
        s[i]=w[i]
        m[i]=w[i]
    else:
        s[i]+=s[f[i]]+w[i]
        m[i]=min(m[f[i]],w[i])
for i in range(n):
    print(s[i],m[i])
