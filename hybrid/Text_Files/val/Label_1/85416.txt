#include <bits/stdc++.h>

using namespace std;

struct node {
    int val, prio;
    node() {}
    node(int v, int pr) : val(v), prio(pr) {}
    bool operator<(node x) const { return prio > x.prio; }
};

const int maxn = 100005;
priority_queue<node> pq;
int n;

int main() {
    cin >> n;
    while (n--) {
        string s;
        cin >> s;
        if (s == "insert") {
            int x;
            cin >> x;
            pq.push(node(x, x));
        } else if (s == "getMin") {
            int x;
            cin >> x;
            while (!pq.empty() && x != pq.top().val) {
                node temp = pq.top();
                pq.pop();
                pq.push(node(temp.val + temp.val, temp.prio));
            }
        } else {
            while (!pq.empty() && pq.top().val == 0) pq.pop();
        }
    }
    cout << pq.size() << '\n';
    while (!pq.empty()) {
        cout << "insert " << pq.top().val << '\n';
        pq.pop();
    }
    return 0;
}
