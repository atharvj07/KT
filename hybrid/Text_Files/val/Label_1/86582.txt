import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        while (true) {
            int n = in.nextInt();
            int a = in.nextInt();
            int b = in.nextInt();
            if (n == 0 && a == 0 && b == 0) {
                break;
            }
            int count = 0;
            for (int i = 0; i <= n; i++) {
                if ((i % a != 0) && (i % b != 0)) {
                    count++;
                }
            }
            System.out.println(count);
        }
    }
}
