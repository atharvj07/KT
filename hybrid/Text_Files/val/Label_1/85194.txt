#include <bits/stdc++.h>
using namespace std;
int n,m,k,c[200005],l[200005],r[200005];
int main(){
    ios_base::sync_with_stdio(false); cin.tie(NULL);
    cin>>n>>m>>k;
    for(int i=1;i<=n;i++) cin>>c[i];
    for(int i=1;i<=m;i++) cin>>l[i]>>r[i];
    for(int i=1;i<=m;i++) if(c[l[i]]!=c[r[i]]){ cout<<0<<endl; return 0; }
    map<int,bool> mp;
    for(int i=1;i<=m;i++) mp[c[l[i]]]=1;
    int cnt=0;
    for(int i=1;i<=m;i++) if(!mp[c[r[i]]]) cnt++;
    cout<<cnt<<endl;
}
