#include <bits/stdc++.h>
#define ll long long
#define endl "\n"
#define vi vector<ll>
#define qi queue<ll>
#define pb push_back
#define mp make_pair
#define D1(a) cout << #a " = " << a << endl;
#define D2(a, b) cout << #a " = " << a << ", " << #b " = " << b << endl;
using namespace std;

int main()
{
    int n;
    cin >> n;
    string l, r;
    cin >> l >> r;
    vi left, right;
    for (int i = 0; i < n; i++)
    {
        if (l[i] != r[i] && l[i] != '?' && r[i] != '?')
            continue;
        else
        {
            left.pb(i + 1);
            right.pb(i + 1);
        }
    }
    int cnt = min(n / 2, (int)left.size());
    cout << cnt << endl;
    for (int i = 0; i < cnt; i++)
    {
        cout << left[i] << " " << right[i] << endl;
    }
    return 0;
}
