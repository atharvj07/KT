import sys
input = sys.stdin.readline

n,m=map(int,input().split())
d,q,boss=[0]*n,[0]*n,[-1]*n
for i in range(m):
    t,*a=map(int,input().split())
    if t==1:
        boss[a[0]-1]=a[1]-1
    elif t==2:
        for i in range(a[0]-1,n,boss[i]):
            d[i]=1
    else:
        if not q[a[0]-1]:
            for i in range(a[0]-1,n,boss[i]):
                q[i]=d[i]
        print(('YES','NO')[q[a[0]-1]==a[1]])
