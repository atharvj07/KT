#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x, y, d;
        cin >> n >> x >> y >> d;
        if (x <= y) {
            if ((y - x) % d == 0)
                cout << (y - x) / d << endl;
            else
                cout << -1 << endl;
        } else {
            if ((n - x + y) % d == 0)
                cout << (n - x + y) / d << endl;
            else
                cout << -1 << endl;
        }
    }
}
