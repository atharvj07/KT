n = int(input())
a = list(map(int, input().split()))
mx = max(a)
for i in range(0, n, 3):
    if mx == a[i]:
        print("chest")
        break
for i in range(1, n, 3):
    if mx == a[i]:
        print("biceps")
        break
for i in range(2, n, 3):
    if mx == a[i]:
        print("back")
        break
