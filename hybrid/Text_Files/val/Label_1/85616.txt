#include <bits/stdc++.h>
using namespace std;
int t, n;
vector<int> v;
int main() { cin >> t; while (t--) { cin >> n; v.resize(2 * n); for (int &i : v) cin >> i; sort(v.begin(), v.end()); int ok = 1; for (int i = 1; i < 2 * n; i += 2) if (!(v[i] - v[i - 1] & 1)) ok = 0; cout << (ok ? "Yes" : "No") << endl; v.clear(); } }
