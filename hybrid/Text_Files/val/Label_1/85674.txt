#include <bits/stdc++.h>
using namespace std;
char m[3][3], a[3][3];
int main() {
  for (int i = 0; i < 2; i++)
    for (int j = 0; j < 2; j++) cin >> m[i][j];
  for (int i = 0; i < 2; i++)
    for (int j = 0; j < 2; j++) cin >> a[i][j];
  return cout << ((a[0][0] == m[0][0] && a[0][1] == m[0][1] && a[1][0] == m[1][0] &&
                   a[1][1] == m[1][1])
                    ? "YES"
                    : "NO"),
         0;
}
