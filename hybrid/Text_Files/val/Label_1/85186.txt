#include <bits/stdc++.h>

using namespace std;

#define ll long long

int main()
{
    ll n, k, m;
    cin >> n >> k >> m;
    if (n == 1)
    {
        cout << (k) % m << endl;
        return 0;
    }
    ll dp[n + 2][k + 2][2][2];
    memset(dp, 0, sizeof dp);
    dp[1][0][1][0] = 1;
    for (int i = 1; i <= n; i++)
    {
        for (int mod = 0; mod <= k; mod++)
        {
            for (int lead = 0; lead <= 1; lead++)
            {
                for (int prv = 0; prv <= 1; prv++)
                {
                    for (int dig = 0; dig <= 9; dig++)
                    {
                        ll curmod = (dig + mod) % k;
                        ll curlead = lead, curprv = prv;
                        if (lead == 0 && dig == 0) curlead = 1;
                        if (prv == 1 && dig != 0) curprv = 0;
                        dp[i + 1][curmod][curlead][curprv] += dp[i][mod][lead][prv];
                        dp[i + 1][curmod][curlead][curprv] %= m;
                    }
                }
            }
        }
    }
    cout << dp[n][0][0][1] << endl;

    return 0;
}
