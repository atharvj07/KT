t = int(input().strip())
for _ in range(t):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    p = set(map(int, input().split()))
    for i in range(n):
        for j in range(n-1):
            if a[j] > a[j + 1] and j + 1 in p:
                a[j], a[j + 1] = a[j + 1], a[j]
    if a == sorted(a):
        print("YES")
    else:
        print("NO")
