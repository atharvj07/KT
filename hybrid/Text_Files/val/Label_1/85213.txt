#include <iostream>
#include <vector>
#include <map>
using namespace std;

int main() {
  int n, x, k;
  cin >> n >> x >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  map<long long, int> last;
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    ans += i - last[a[i]];
    last[a[i] + x] = i;
  }
  cout << ans << endl;
  return 0;
}
