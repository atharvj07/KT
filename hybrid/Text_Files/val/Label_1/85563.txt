#include <iostream>
#include <vector>

using namespace std;

int main() {
    long long n;
    cin >> n;

    vector<int> denominations = {1, 5, 10, 20, 100};
    long long sum = 0;
    long long count = 0;

    for (int i = denominations.size() - 1; i >= 0; i--) {
        while (n >= denominations[i]) {
            n -= denominations[i];
            count++;
        }
    }

    cout << count << endl;

    return 0;
}
