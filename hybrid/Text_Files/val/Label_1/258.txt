n = int(input())
a = [list(map(int, input().split())) for i in range(n)]
m = 10 ** 9 + 7
f = [[[[-1 for _ in range(5)] for _ in range(1 << n)] for _ in range(n)] for _ in range(n)]

def dp(x, y, s, t):
    if x == y and s == (1 << n) - 1 and t == 0:
        return 1
    if f[x][y][s][t] != -1:
        return f[x][y][s][t]
    f[x][y][s][t] = 0
    if a[x][y] and ((s >> y) & 1) == ((s >> x) & 1):
        f[x][y][s][t] += dp(x, y, s, (t + 1) % 4)
    for z in range(n):
        if z == x or z == y or ((s >> z) & 1) != t % 2:
            continue
        if a[y][z]:
            f[x][y][s][t] += dp(y, z, s | (1 << z), t)
    f[x][y][s][t] %= m
    return f[x][y][s][t]

print(dp(0, 0, 1, 0))
