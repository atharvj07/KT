#include <bits/stdc++.h>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        vector<int> freq(n + 1);
        for (int i = 0; i < n; i++) {
            int x; cin >> x;
            freq[x]++;
        }
        long long ans = 0;
        for (int i = 1; i <= n; i++) {
            ans += (long long)freq[i] * (freq[i] - 1) * (freq[i] - 2) / 6;
        }
        cout << ans << '\n';
    }
    return 0;
}
