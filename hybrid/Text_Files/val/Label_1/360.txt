shift = input().strip()
message = input().strip()

keyboard = 'qwertyuiopasdfghjkl;zxcvbnm,./'
shifted_message = ''

if shift == 'R':
    for char in message:
        shifted_message += keyboard[keyboard.index(char) - 1]
else:
    for char in message:
        shifted_message += keyboard[keyboard.index(char) + 1]

print(shifted_message)
