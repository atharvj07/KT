import java.util.*;

public class GoldWar {

    static class Edge {
        int u, v;

        public Edge(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    static class Spaceship {
        int x, a, f;

        public Spaceship(int x, int a, int f) {
            this.x = x;
            this.a = a;
            this.f = f;
        }
    }

    static class Base {
        int x, d;

        public Base(int x, int d) {
            this.x = x;
            this.d = d;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<Edge> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt() - 1;
            int v = scanner.nextInt() - 1;
            edges.add(new Edge(u, v));
        }

        int s = scanner.nextInt();
        int b = scanner.nextInt();
        int k = scanner.nextInt();
        int h = scanner.nextInt();

        Spaceship[] spaceships = new Spaceship[s];
        for (int i = 0; i < s; i++) {
            int x = scanner.nextInt() - 1;
            int a = scanner.nextInt();
            int f = scanner.nextInt();
            spaceships[i] = new Spaceship(x, a, f);
        }

        Base[] bases = new Base[b];
        for (int i = 0; i < b; i++) {
            int x = scanner.nextInt() - 1;
            int d = scanner.nextInt();
            bases[i] = new Base(x, d);
        }

        int[][] distances = floydWarshall(n, edges);

        int minCost = Integer.MAX_VALUE;
        for (int dummyBases = 0; dummyBases <= b; dummyBases++) {
            int cost = dummyBases * h;
            int attackedBases = 0;

            for (Spaceship spaceship : spaceships) {
                int maxDistance = spaceship.f;
                for (Base base : bases) {
                    if (spaceship.a >= base.d && distances[spaceship.x][base.x] <= maxDistance) {
                        attackedBases++;
                        break;
                    }
                }
            }

            cost += (b - attackedBases) * k;
            minCost = Math.min(minCost, cost);
        }

        System.out.println(minCost);
    }

    static int[][] floydWarshall(int n, List<Edge> edges) {
        int[][] distances = new int[n][n];
        for (int i = 0; i < n; i++) {
            Arrays.fill(distances[i], Integer.MAX_VALUE);
        }

        for (Edge edge : edges) {
            distances[edge.u][edge.v] = 1;
            distances[edge.v][edge.u] = 1;
        }

        for (int k = 0; k < n; k++) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    distances[i][j] = Math.min(distances[i][j], distances[i][k] + distances[k][j]);
                }
            }
        }

        return distances;
    }
}
