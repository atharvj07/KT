import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            long a = in.nextLong(), b = in.nextLong(), c = in.nextLong(), d = in.nextLong();
            long x = a, y = b, z = c;
            if (b - a < d - c) {
                x = b;
            }
            if (c - b < d - x) {
                y = c;
            }
            System.out.println(x + " " + y + " " + z);
        }
    }
}
