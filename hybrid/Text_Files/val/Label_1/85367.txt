#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 2005, oo = 1e9;
int n, k;
string ans;
char a[maxn][maxn];
vector<pair<int, int>> pos;
void dfs(int x, int y, int kk) {
    if (x == n) {
        for (int j = y; j <= n; j++) ans += a[x][j];
        return;
    }
    if (y == n) {
        for (int i = x; i <= n; i++) ans += a[i][y];
        return;
    }
    if (a[x][y] != ans.back()) {
        if (kk) {
            pos.push_back({x, y});
            --kk;
            ans += a[x][y];
            dfs(x + 1, y, kk);
            dfs(x, y + 1, kk);
            pos.pop_back();
        } else
            dfs(x, y + 1, kk);
    } else {
        ans += a[x][y];
        dfs(x + 1, y, kk);
        dfs(x, y + 1, kk);
    }
}
int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++) cin >> a[i][j];
    ans.resize(2 * n - 1);
    dfs(1, 1, k);
    for (auto it : pos) a[it.first][it.second] = ans[it.first + it.second - 2];
    for (int i = 1; i <= n; i++, cout << '\n')
        for (int j = 1; j <= n; j++) cout << a[i][j];
}
