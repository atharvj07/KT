#include <iostream>
#include <vector>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> divs[MAXN];
pair<int, int> dp[MAXN][10]; // dp[i][j] = (cnt, last_occur)

int main() {
    int n, x, k;
    cin >> n >> x >> k;
    int a[n + 1];
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        for (int j = 1; j * j <= x; j++) {
            if (x % j == 0) {
                divs[a[i]].push_back(j);
                if (x / j != j) divs[a[i]].push_back(x / j);
            }
        }
    }
    for (int i = n; i >= 1; i--) {
        for (int div : divs[a[i]]) {
            int cnt = 0, last = -1;
            for (int j = i + 1; j <= n; j++) {
                if (a[j] >= div) {
                    if (last == j - 1) cnt++;
                    else cnt = 1;
                    last = j;
                }
                dp[i][div].first = max(dp[i][div].first, cnt + dp[j][div].first);
                dp[i][div].second = max(dp[i][div].second, dp[j][div].second);
            }
            if (last == i) {
                if (cnt + dp[i][div].first < k) continue;
                dp[i][div].second = i;
            }
            dp[i][div].second = max(dp[i][div].second, dp[i][div].first - 1 + dp[i][div].first);
        }
    }
    int ans = 0;
    for (int i = 1; i <= n; i++)
        for (int div : divs[a[i]]) {
            if (dp[i][div].first < k) continue;
            int j = dp[i][div].second;
            if (j < 0) j = i;
            if (j >= i + k) ans++;
        }
    cout << ans << endl;
    return 0;
}
