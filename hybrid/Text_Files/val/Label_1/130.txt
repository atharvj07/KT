MOD = 998244353

def count_trees(n):
    if n == 1:
        return 1
    h = 0
    while (1 << h) - 1 < n:
        h += 1
    full = (1 << (h - 1)) - 1
    last = n - 1 - full
    if last >= (1 << (h - 2)):
        return 0
    if h % 2 == 0:
        return 1 if last == (1 << (h - 2)) - 1 else 0
    else:
        return 2 if last == 0 else 0

n = int(input().strip())
print(count_trees(n) % MOD)
