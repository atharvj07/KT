import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] r = new int[4];
        for (int i = 1; i < 4; i++) {
            r[i] = sc.nextInt();
        }
        int d = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        long ans = 0;
        int level = 0;
        int loaded = 0;
        for (int i = 0; i < n; i++) {
            if (i > 0) {
                ans += d;
            }
            ans += Math.max(0, a[i] - (loaded == 2 ? 1 : 0));
            ans += r[2] - loaded;
            loaded = 2;
            if (a[i] > 0) {
                ans += r[1] - 1;
                loaded = 1;
            }
        }
        System.out.println(ans);
    }
}
