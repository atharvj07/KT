#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 5;
int n, d, b;
int a[N];
int l[N], r[N];

int main() {
  cin >> n >> d >> b;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 1; i <= n; i++) {
    l[i] = max(i - d, 1);
    r[i] = min(i + d, n);
  }
  int st = 1, en = n;
  int cnt1 = 0, cnt2 = 0;
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    if (st > en) break;
    if (a[st] + cnt1 != b) ans = 1;
    if (a[en] + cnt2 != b) ans = 1;
    if (a[st] + cnt1 + a[en] + cnt2 == b) {
      cnt1 -= b - a[st];
      a[st] = b;
      cnt2 -= b - a[en];
      a[en] = b;
      st++;
      en--;
    } else if (a[st] + cnt1 > a[en] + cnt2) {
      cnt1 -= b - a[st];
      a[st] = b;
      st++;
    } else {
      cnt2 -= b - a[en];
      a[en] = b;
      en--;
    }
    for (int j = st; j <= en; j++) {
      cnt1 += max(0, a[j] + cnt1 - b);
      cnt2 += max(0, a[j] + cnt2 - b);
      a[j] = min(a[j] + cnt1, b);
      a[j] = min(a[j] + cnt2, b);
    }
  }
  cout << ans << endl;
}
