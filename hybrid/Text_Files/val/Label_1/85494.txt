#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  while (n > 1) {
    vector<long long> b;
    for (int i = 1; i < n; i++) { b.emplace_back(a[i - 1] + a[i]); }
    n = b.size();
    a = b;
  }
  cout << a[0] << endl;
  return 0;
}
