#include <bits/stdc++.h>
 
#define ll long long
 
using namespace std;
 
const ll MAXN = 1e5 + 10;
 
ll n, sz[MAXN], par[MAXN], ans1, ans2, ans3, ans4;
 
vector<ll> adj[MAXN];
 
vector<pair<ll, ll>> ans;
 
void dfs_sz(ll node, ll p) {
    sz[node] = 1;
    par[node] = p;
    for (ll next : adj[node]) {
        if (next != p) {
            dfs_sz(next, node);
            sz[node] += sz[next];
        }
    }
}
 
ll get_centroid(ll node, ll p, ll tot) {
    for (ll next : adj[node]) {
        if (next != p) {
            if (sz[next] > tot / 2) return get_centroid(next, node, tot);
        }
    }
    return node;
}
 
void find_centroid() {
    dfs_sz(1, 1);
    ll centroid = get_centroid(1, 1, n);
    ans1 = centroid;
    for (ll next : adj[centroid]) {
        if (sz[next] > n / 2) {
            ans2 = next;
            ans3 = centroid;
            ans4 = next;
        }
    }
}
 
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    ll t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (ll i = 1; i <= n; i++) {
            adj[i].clear();
        }
        for (ll i = 1; i < n; i++) {
            ll a, b;
            cin >> a >> b;
            adj[a].push_back(b);
            adj[b].push_back(a);
        }
        find_centroid();
        cout << ans1 << " " << ans2 << endl;
        cout << ans3 << " " << ans4 << endl;
    }
    return 0;
}
