#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int s, b;
  cin >> s >> b;

  vector<int> a(s);
  for (int i = 0; i < s; i++) {
    cin >> a[i];
  }

  vector<pair<int, int>> b_sorted(b);
  for (int i = 0; i < b; i++) {
    cin >> b_sorted[i].first >> b_sorted[i].second;
  }
  sort(b_sorted.begin(), b_sorted.end());

  vector<int> ans(s);
  for (int i = 0; i < s; i++) {
    int cur_gold = 0;
    for (int j = 0; j < b; j++) {
      if (a[i] >= b_sorted[j].first) {
        cur_gold += b_sorted[j].second;
      }
    }
    ans[i] = cur_gold;
  }

  for (int i = 0; i < s; i++) {
    cout << ans[i] << endl;
  }

  return 0;
}
