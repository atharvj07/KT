#include <bits/stdc++.h>
using namespace std;

#define rep(i, a, b) for (int i = (a); i < (b); ++i)

using ll = long long;

const int N = 1e5 + 10;

ll a[N];
int n, d, b;
int cnt[N];

int main() {
  cin >> n >> d >> b;
  rep(i, 0, n) cin >> a[i];
  rep(i, 0, n) cnt[a[i]]++;

  int cur = b;
  int ans = 0;
  rep(i, 0, N) {
    if (i != b) {
      int need = max(0, i - b);
      ll have = cnt[i];
      if (have < need) {
        ans++;
        cnt[cur] -= need - have;
      } else if (have > need) {
        ans++;
        cnt[cur] += have - need;
      }
      if (cnt[cur] < 0) break;
    }
    cur++;
    if (cur > b) break;
  }

  cout << ans << endl;
  return 0;
}
