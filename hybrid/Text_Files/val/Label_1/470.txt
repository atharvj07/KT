n = int(input())
a = list(map(int, input().split()))

suffix_sum = [0] * n
suffix_sum[n - 1] = a[n - 1]
for i in range(n - 2, -1, -1):
    suffix_sum[i] = suffix_sum[i + 1] + a[i]

res = 0
cur = 0
for i in range(n):
    cur += a[i]
    if cur < 0:
        cur = 0
        res = i
    res = max(res, i + 1 if suffix_sum[i] >= 0 else i)

print(res)
