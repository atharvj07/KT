#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n,f[1005],i,j=0;
    cin>>n;
    f[0]=1;f[1]=1;
    for(i=2; i<n; i++) f[i]=f[i-1]+f[i-2];
    for(i=0;i<n;i++){
        if(j<=n){
            if(f[j]==i+1){
                cout<<"O";
                 j++;
            }
            else cout<<"o";
        }
    }
    cout<<"\n";
    return 0;
}
