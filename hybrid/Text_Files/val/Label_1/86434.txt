import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
  public static void main(String[] args) throws Exception {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] nm = br.readLine().split(" "), nms = br.readLine().split(" "), nss = br.readLine().split(" "), nbs = br.readLine().split(" "), incs = br.readLine().split(" "), prefs = br.readLine().split(" ");
    for (int i = 0, j = 0, l = Integer.parseInt(nm[1]), p, s, b, inc, pref, ok; i < l; i++) {
      ok = 0;
      p = Integer.parseInt(nms[i]);
      s = Integer.parseInt(nss[i]);
      b = Integer.parseInt(nbs[i]);
      inc = Integer.parseInt(incs[i]);
      pref = Integer.parseInt(prefs[i]);
      if (p <= inc && inc <= s && Math.abs(b - pref) <= (inc - p)) ok = 1;
      System.out.print(ok + " ");
    }
  }
}
