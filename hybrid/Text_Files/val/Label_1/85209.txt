#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int n, x, k;
    cin >> n >> x >> k;
    vector<int> v(n);
    for (int& x : v) cin >> x;
    vector<int> pre(10000001), suf(10000001);
    for (int i = 0; i < n; i++) pre[v[i]]++;
    for (int i = 10000000; i >= 1; i--) {
        if (i % x == 0) suf[i] += suf[i + x];
        suf[i] += pre[i];
    }
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        if (k > 0) ans += suf[v[i] + 1];
        else if (v[i] % x == 0) ans += pre[v[i]] - 1;
    }
    cout << ans << '\n';
}
