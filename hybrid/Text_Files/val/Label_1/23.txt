n = int(input())
blocks = [list(map(int, input().split())) for i in range(n)]
M = 401
dp = [[[0 for j in range(M)] for k in range(6)] for z in range(n+1)]
for j in range(1, n+1):
    color1, value, color2 = blocks[j-1]
    for k in range(6):
        for prev in range(M):
            dp[j][k][prev] = max(dp[j][k][prev], dp[j-1][k][prev])
            dp[j][color1][prev] = max(dp[j][color1][prev], dp[j-1][k][prev]+value)
            dp[j][color2][color1] = max(dp[j][color2][color1], dp[j-1][k][prev]+value)
print(dp[n][0][0])
