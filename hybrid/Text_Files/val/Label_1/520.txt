t = int(input())
for _ in range(t):
    n, x, y = map(int, input().split())
    diff = y - x
    for i in range(1, diff):
        if diff % i == 0 and diff // i + 1 <= n:
            step = i
            break
    max_elements = diff // step + 1
    start = y - step * (n - 1)
    if start < 1:
        start += (step * ((n - max_elements) // 2 + (n - max_elements) % 2))
    for i in range(n):
        print(start + step * i, end=' ')
    print()
