#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
  ll n, m, k, ans = 0;
  cin >> n >> m >> k;
  multiset<ll> a;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    a.insert(x);
  }
  while (!a.empty() && ans < n) {
    ll sum = 0;
    multiset<ll>::reverse_iterator it = a.rbegin();
    while (it != a.rend() && sum + *it <= k) {
      sum += *it;
      ans++;
      it++;
    }
    a.erase(a.begin(), it.base());
  }
  cout << ans << endl;
}
