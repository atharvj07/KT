#include <bits/stdc++.h>
typedef long long ll;
#define REP(i,n) for(int i=0;i<(int)n;i++)
using namespace std;

ll dfs(vector<vector<ll>>& adj,ll node, ll par,const vector<ll>& weight){
    ll mn=weight[node];
    for(auto &e:adj[node]){
        if(e!=par)mn=min(mn, dfs(adj,e,node,weight));
    }
    return mn;
}

int main(){
    ll n,k;
    cin >> n >> k;
    vector<ll> weight(n+1);
    REP(i,n)cin >> weight[i+1];

    vector<vector<ll>> adj(n+1);
    REP(i,n-1){
        ll a,b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }

    ll ans=LLONG_MAX;
    REP(rt,n){
        ll tmp=dfs(adj,rt+1,-1,weight);
        ans=min(ans,tmp);
    }

    cout << ans;
    return 0;
}
