#include <iostream>
#include <vector>
using namespace std;

int n;
vector<vector<char>> a, b;

bool rot90() {
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            if (a[i][j] != b[n - j - 1][i])
                return false;
    return true;
}

bool rot180() {
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            if (a[i][j] != b[n - i - 1][n - j - 1])
                return false;
    return true;
}

bool rot270() {
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            if (a[i][j] != b[j][n - i - 1])
                return false;
    return true;
}

bool flipv() {
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            if (a[i][j] != b[i][n - j - 1])
                return false;
    return true;
}

bool fliph() {
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            if (a[i][j] != b[n - i - 1][j])
                return false;
    return true;
}

int main() {
    cin >> n;
    a.resize(n, vector<char>(n));
    b.resize(n, vector<char>(n));
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            cin >> a[i][j];
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            cin >> b[i][j];
    if (a == b || rot90() || rot180() || rot270() || flipv() || fliph())
        cout << "Yes" << endl;
    else
        cout << "No" << endl;
    return 0;
}
