#include <bits/stdc++.h>
using namespace std;
#define sz(v) ((long long)(v).size())

int n, m, d, u, v, p, mp[100005];
vector<int> adj[100005], dis[100005];

void dfs(int u, int pre) {
    for (auto v : adj[u]) {
        if (v == pre)
            continue;
        int len = dis[u][v];
        dis[v][u] = dis[u][v];
        dis[v].push_back(len + dis[u][pre]);
        dfs(v, u);
    }
}

int main() {
    freopen("Input.txt", "r", stdin);
    cin >> n >> m >> d;
    for (int i = 0; i < m; i++) { cin >> p; mp[p] = 1; }
    while (cin >> u >> v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
        dis[u].push_back(1);
        dis[v].push_back(1);
    }
    for (int j = 1; j <= n; j++) {
        dfs(j, j);
    }
    int ans = 0;
    for (int j = 1; j <= n; j++) {
        int ok = 0;
        for (int p : dis[j]) {
            ok |= mp[j] & (p <= d);
        }
        ans += ok;
    }
    cout << ans << endl;
    return 0;
}
