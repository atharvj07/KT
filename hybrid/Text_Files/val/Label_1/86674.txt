import java.io.*;
import java.util.Arrays;

class Main {
    public static void main(String[] args) throws IOException {
        Reader reader = new InputStreamReader(System.in);
        BufferedReader in = new BufferedReader(reader);
        int n = Integer.parseInt(in.readLine());
        char[] s = in.readLine().toCharArray();
        int[] stack = new int[n];
        int stackPtr = 0;
        int[] left = new int[n];
        int[] right = new int[n];
        for (int i = 0; i < n; ++i) {
            if (s[i] == '(') stack[stackPtr++] = i;
            else {
                if (stackPtr > 0) stackPtr--;
                left[i] = stackPtr > 0 ? stack[stackPtr - 1] : -1;
            }
        }
        Arrays.fill(stack, 0);
        stackPtr = 0;
        for (int i = n - 1; i >= 0; --i) {
            if (s[i] == ')') stack[stackPtr++] = i;
            else {
                if (stackPtr > 0) stackPtr--;
                right[i] = stackPtr > 0 ? stack[stackPtr - 1] : -1;
            }
        }
        int maxBeauty = 0;
        int l = 0, r = 0;
        for (int i = 0; i < n; ++i) {
            for (int j = i + 1; j < n; ++j) {
                if ((i == j) ||
                        (s[i] == '(' && s[j] == ')' && (i == left[j] || j == right[i])) ||
                        (s[i] == ')' && s[j] == '(' && (i == right[j] || j == left[i]))) {
                    int beauty = 0;
                    if (i != 0) {
                        int beautyFromStart = 0;
                        if (s[i - 1] == '(') {
                            beautyFromStart += 2;
                            if (left[i - 1] >= 0 && s[left[i - 1]] == '(') {
                                beautyFromStart += 2;
                            }
                        } else if (s[i - 1] == ')' && left[i - 1] >= 0) {
                            beautyFromStart += 2;
                            if (s[left[i - 1]] == '(') {
                                beautyFromStart += 2;
                            }
                        }
                        int beautyFromEnd = 0;
                        if (s[j - 1] == '(') {
                            beautyFromEnd += 2;
                            if (left[j - 1] >= 0 && s[left[j - 1]] == '(') {
                                beautyFromEnd += 2;
                            }
                        } else if (s[j - 1] == ')' && left[j - 1] >= 0) {
                            beautyFromEnd += 2;
                            if (s[left[j - 1]] == '(') {
                                beautyFromEnd += 2;
                            }
                        }
                        int best = Math.max(beautyFromStart, beautyFromEnd);
                        if (best > 0) beauty += best;
                    }
                    if (j != n - 1) {
                        int beautyFromStart = 0;
                        if (s[j + 1] == '(') {
                            beautyFromStart += 2;
                            if (left[j + 1] >= 0 && s[left[j + 1]] == '(') {
                                beautyFromStart += 2;
                            }
                        } else if (s[j + 1] == ')' && left[j + 1] >= 0) {
                            beautyFromStart += 2;
                            if (s[left[j + 1]] == '(') {
                                beautyFromStart += 2;
                            }
                        }
                        int beautyFromEnd = 0;
                        if (s[i + 1] == '(') {
                            beautyFromEnd += 2;
                            if (left[i + 1] >= 0 && s[left[i + 1]] == '(') {
                                beautyFromEnd += 2;
                            }
                        } else if (s[i + 1] == ')' && left[i + 1] >= 0) {
                            beautyFromEnd += 2;
                            if (s[left[i + 1]] == '(') {
                                beautyFromEnd += 2;
                            }
                        }
                        int best = Math.max(beautyFromStart, beautyFromEnd);
                        if (best > 0) beauty += best;
                    }
                    if (beauty > maxBeauty) {
                        maxBeauty = beauty;
                        l = i;
                        r = j;
                    }
                }
            }
        }
        System.out.println(maxBeauty);
        System.out.println((l + 1) + " " + (r + 1));
    }
}
