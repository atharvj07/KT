#include <bits/stdc++.h>
using namespace std;
int solve()
{
    int t, sx, sy, ex, ey;
    cin >> t >> sx >> sy >> ex >> ey;
    string s;
    cin >> s;
    int dx[] = {1, 0, -1, 0};
    int dy[] = {0, -1, 0, 1};
    for (int i = 0, x = sx, y = sy, j = 0;; i++)
    {
        if (x == ex and y == ey)
            return i;
        if (j == 4)
            return -1;
        if (s[i % t] == 'E')
            x++;
        else if (s[i % t] == 'S')
            y--;
        else if (s[i % t] == 'W')
            x--;
        else if (s[i % t] == 'N')
            y++;
        if (x < sx or x > ex or y < sy or y > ey)
            j++;
    }
}
int main()
{
    cout << solve() << endl;
    return 0;
}
