import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] arr = new int[n];

        for (int i = 0; i < n; ++i) {
            arr[i] = scanner.nextInt();
        }

        long[][] dp = new long[n][6];

        dp[n - 1][arr[n - 1]] = 1;

        for (int i = n - 2; i >= 0; --i) {
            for (int j = 1; j <= 5; ++j) {
                dp[i][j] += dp[i + 1][j];
                dp[i][j] += dp[i + 1][(j + arr[i]) % 6];
            }
        }

        long ans = dp[0][1] + dp[0][2] + dp[0][3] + dp[0][4] + dp[0][5];

        System.out.println(ans);
    }
}
