#include <bits/stdc++.h>
using namespace std;
#define int long long
int n,m,k,q;
int a[200005],b[200005];
int vis[200005];
set<int>st;
vector<int>g[200005];
int f(int x)
{
    if(x==n) return 0;
    if(vis[x]==1) return 1e18;
    vis[x]=1;
    if(st.find(x)!=st.end()) return 1;
    int ans=1e18;
    if(g[x].size()>0)
    {
        for(auto it:g[x])
        ans=min(ans,1+f(it));
    }
    return ans;
}
main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    cin>>n>>m>>k>>q;
    for(int i=1;i<=k;i++)
    {
        int x,y;
        cin>>x>>y;
        a[x]=y;
    }
    for(int i=1;i<=q;i++)
    {
        int x;
        cin>>x;
        b[x]=1;
    }
    for(int i=1;i<n;i++)
    {
        for(int j=1;j<=m;j++)
        {
            if(b[j])
            {
                if(a[i]==j)
                st.insert(i);
                if(a[i+1]==j)
                g[i].push_back(i+1);
            }
        }
    }
    cout<<f(1)<<"\n";
}
