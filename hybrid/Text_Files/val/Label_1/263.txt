from sys import stdin, stdout
mod = int(1e9) + 7
adj = [[] for i in range(501)]
for i in range(1,int(stdin.readline())):
    row = list(map(int, stdin.readline().split()))
    for j in range(i):
        adj[i].append(row[j])
        adj[j].append(row[i])
dp = [[0,0,0] for i in range(501)]
def rec(u,p,mask):
    if dp[u][mask] != 0: return dp[u][mask]
    last = -1 if mask == 0 else p
    ans = 0
    for v in range(len(adj[u])):
        if v == last or adj[u][v] == 0 or (mask & (1 << v)): continue
        if mask == 0: ans += rec(v,u,1<<v)
        else: ans += rec(v,u,mask|(1<<v))
    dp[u][mask] = ans
    return ans%mod
print(rec(0,-1,0))
