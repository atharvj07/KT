#include <bits/stdc++.h>

int main() {
    int n;
    std::cin >> n;
    std::string s, t;
    std::cin >> s >> t;
    if (s[0] == s[1] || t[0] == t[1] || s == t) {
        std::cout << "NO" << std::endl;
    } else {
        std::string res = "abc";
        for (int i = 0; i < n - 1; i++) {
            res = "a" + res + "b" + res + "c" + res;
        }
        std::cout << "YES" << std::endl;
        std::cout << res << std::endl;
    }
    return 0;
}
