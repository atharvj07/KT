from collections import defaultdict, deque
import sys

input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
a.sort()
g = defaultdict(list)
vis = defaultdict(int)
for z in range(1, n + 1):
    for j in a:
        if z + j <= n + 1: g[z].append(z + j)
        if z - j >= 1: g[z].append(z - j)

b = []
for z in range(1, n + 1):
    if vis[z] == 0:
        Q = deque()
        Q.append((z, 1))
        b.append((1, []))
        while len(Q) != 0:
            v, col = Q.popleft()
            if vis[v] == col: continue
            vis[v] = col
            b[-1][1].append(v)
            for u in g[v]:
                if vis[u] == 0: Q.append((u, 3 - col))

low = len(b)
ans = -1
for k in b:
    if len(k[1]) <= low:
        low = len(k[1])
        ans = k

print(low)
if ans != -1:
    for z in ans[1]:
        print(a[z - 1], end=' ')
