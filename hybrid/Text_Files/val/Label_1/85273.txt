#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  long long h, m;
  cin >> n;
  vector<pair<pair<long long, long long>, long long>> v;
  for (long long i = 0; i < n; i++) {
    cin >> h >> m;
    v.push_back({{h, m}, i});
  }
  sort(v.begin(), v.end());
  long long ans = 0, cnt = 0;
  for (auto i : v) {
    if (i.first.first > cnt) {
      ans++;
      cnt = i.first.first;
    }
    cnt++;
  }
  cout << ans << endl;
  return 0;
}
