from collections import defaultdict, deque

n, m, s = map(int, input().split())
graph = defaultdict(list)
for _ in range(m):
    t, u, v = map(int, input().split())
    graph[u].append((t, v))
    graph[v].append((t, u))

def bfs(graph, s, reverse=False):
    visited = set()
    queue = deque([(s, None)])
    reachable = set()
    while queue:
        u, prev = queue.popleft()
        if u in visited:
            continue
        visited.add(u)
        reachable.add(u)
        for t, v in graph[u]:
            if reverse and t == 2 and v != prev:
                queue.append((v, u))
            elif not reverse and t != 2 or t == 1 and v != prev:
                queue.append((v, u))
    return reachable

reachable_max = bfs(graph, s)
reachable_min = bfs(graph, s, True)

print(len(reachable_max))
for u in graph:
    for t, v in graph[u]:
        if t == 2 and v in reachable_max:
            print('+', end='')
        if t == 2 and v in reachable_min:
            print('-', end='')

print(len(reachable_min))
for u in graph:
    for t, v in graph[u]:
        if t == 2 and v in reachable_max:
            print('-', end='')
        if t == 2 and v in reachable_min:
            print('+', end='')
