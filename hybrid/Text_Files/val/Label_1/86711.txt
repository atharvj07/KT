import java.util.*;
import java.io.*;

class Main {
    static int solve(int[] a, int[] b, int n, int m, int y1, int y2) {
        int yDiff = y2 - y1;
        int[] aModDiff = new int[n];
        int maxCount = 0;

        for (int i = 0; i < n; i++) {
            aModDiff[i] = a[i] % yDiff;
        }

        Arrays.sort(b);

        for (int j = 0; j < m; j++) {
            int l = b[j] - yDiff;
            int r = b[j] + yDiff;
            int low = Arrays.binarySearch(aModDiff, l);
            if (low < 0) {
                low = -low - 1;
            }
            int high = Arrays.binarySearch(aModDiff, r);
            if (high < 0) {
                high = -high - 1;
            }
            maxCount = Math.max(maxCount, high - low);
        }

        return maxCount + 2;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int y1 = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int m = scanner.nextInt();
        int y2 = scanner.nextInt();
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = scanner.nextInt();
        }
        System.out.println(solve(a, b, n, m, y1, y2));
    }
}
