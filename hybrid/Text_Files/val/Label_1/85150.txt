#include<bits/stdc++.h>
using namespace std;
int main(){
    ios_base::sync_with_stdio(false); cin.tie(nullptr); cout.tie(nullptr);
    int n,m,k,q; cin>>n>>m>>k>>q;
    vector<pair<int,int>>tr(k);
    for(auto& t:tr) cin>>t.first>>t.second;
    vector<int>sp(m),ans(k,1e9);
    for(int i=0; i<q; i++) cin>>sp[i];
    for(int i=0; i<k; i++){
        int r=tr[i].first,c=tr[i].second;
        if(c==1){
            ans[i]=r-1;
            continue;
        }
        int lc=upper_bound(sp.begin(),sp.end(),c-1)-sp.begin();
        int rc=lower_bound(sp.begin(),sp.end(),c+1)-sp.begin();
        ans[i]=min(ans[i],r-1+min(abs(sp[lc]-c), abs(sp[rc]-c)));
    }
    int res=*min_element(ans.begin(),ans.end());
    if(res==1e9) cout<<-1<<'\n';
    else cout<<res<<'\n';
    return 0;
}
