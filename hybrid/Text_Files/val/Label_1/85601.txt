#include <iostream>
using namespace std;
int main()
{
    int a[4], b[4], c = -1;
    for( int i = 0 ; i < 4 ; i++ )
        cin >> a[i] >> b[i];
    for( int i = 0 ; i < 2 ; i++ )
    {
        if( a[2 * i] > b[2 * i - 1] && b[2 * i] > a[2 * i - 1] )
            c = i + 1;
        else if( a[2 * i - 1] > b[2 * i] && b[2 * i - 1] > a[2 * i] )
            c = i + 3;
    }
    if( c == -1 )
        cout << "Draw" << endl;
    else
        cout << "Team " << c << endl;
    return 0;
}
