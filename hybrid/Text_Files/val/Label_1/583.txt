import sys

n, m = [int(i) for i in input().split()]
sub = [i for i in range(n + 1)]

def find_boss(x):
    while x != sub[x]:
        x = sub[x]
    return x

def union_boss(a, b):
    root_a = find_boss(a)
    root_b = find_boss(b)
    if root_a != root_b:
        sub[root_a] = root_b

doc_sign = []

for i in range(m):
    t, *vals = [int(j) for j in input().split()]
    if t == 1:
        union_boss(*vals)
    elif t == 2:
        doc_sign.append(vals[0])
    elif t == 3:
        root = find_boss(vals[0])
        print("YES" if root in doc_sign else "NO")
