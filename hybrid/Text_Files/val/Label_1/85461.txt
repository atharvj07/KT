#include<bits/stdc++.h>
using namespace std;
#define ll long long
bool check(vector<int>a){
    for (int i=1;i<a.size();i++){
        if ((a[i]&a[i-1])) return 1;
    }
    return 0;
}
int main()
{
    ll n;cin>>n;
    vector<int>p,q;
    if (n==2) {cout<<"NO\nNO";return 0;}
    if (n==3) {cout<<"NO\nNO";return 0;}
    if (n==4) {cout<<"YES\n4 2 1 3\nYES\n2 4 1 3"; return 0;}
    for (int i=1;i<=n;i++){
        p.push_back(i);
    }
    for (int i=1;i<=n;i++){
        q.push_back(i);
    }
    if (n%2==0){
        int temp=p[n/2];
        p[n/2]=p[n/2-1];
        p[n/2-1]=temp;
    }
    swap(q[n-1],q[n-2]);
    if (check(p)) cout<<"NO\n";
    else {
        cout<<"YES\n";
        for (int i=0;i<n;i++) cout<<p[i]<<" ";
        cout<<"\n";
    }
    if (check(q)) cout<<"NO\n";
    else {
        cout<<"YES\n";
        for (int i=0;i<n;i++) cout<<q[i]<<" ";
        cout<<"\n";
    }
    return 0;
}
