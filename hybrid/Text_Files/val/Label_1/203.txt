import sys
input = sys.stdin.readline

n, m, k = map(int, input().split())
c = list(map(int, input().split()))
l, r = [0] * m, [0] * m
for i in range(m):
    l[i], r[i] = map(int, input().split())
d = {}
for i in range(m):
    if (l[i], r[i]) not in d:
        d[(l[i], r[i])] = 1
    else: d[(l[i], r[i])] += 1
ans = n - len(d)
print(ans)
