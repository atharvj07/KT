import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int w = sc.nextInt();
        int n = sc.nextInt();
        // coordinates of black squares
        int[][] black = new int[n][2];
        for (int i = 0; i < n; i++) {
            black[i][0] = sc.nextInt();
            black[i][1] = sc.nextInt();
        }

        // distance between white squares
        long dist = 0;
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                boolean isBlack = false;
                for (int[] b : black) {
                    if (b[0] == i && b[1] == j) {
                        isBlack = true;
                    }
                }
                if (!isBlack) {
                    int minDist = Integer.MAX_VALUE;
                    for (int[] b : black) {
                        int d = Math.abs(i - b[0]) + Math.abs(j - b[1]);
                        minDist = Math.min(minDist, d);
                    }
                    dist += minDist;
                }
            }
        }

        // choose two white squares
        long res = (dist % 1000000007) * (h * w - n) % 1000000007 * ((h * w - n - 1) % 1000000007) % 1000000007;
        System.out.println(res);
    }
}
