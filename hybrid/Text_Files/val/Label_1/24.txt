import sys
input = sys.stdin.readline

n = int(input())
d = []
for i in range(n):
    c1, v, c2 = map(int, input().split())
    d.append([c1, v, c2])
d.sort(key = lambda x: x[1])
dp = {}
dp[d[0][2]] = d[0][1]
for i in range(1, n):
    for j in list(dp):
        if j == d[i][0]:
            continue
        dp[d[i][2]] = max(dp.get(d[i][2], 0), dp[j] + d[i][1])
print(max(dp.values()))
