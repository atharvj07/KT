#include <bits/stdc++.h>
using namespace std;
#define MOD 1000000007ll

typedef long long ll;
typedef pair<int,int> ii;
#define PB push_back
#define MP make_pair
#define F first
#define S second
const int MN = 100001;
int N, a[MN];
int main() {
    cin >> N;
    for (int i = 0; i < N; i++) cin >> a[i];
    ll ans = N, cnt = 0;
    for (int i = 0; i < N; i++)
        if (a[i] == i) cnt++;
    for (int i = 0; i < N; i++)
        if (a[a[i]-1] == i) cnt += 2;
    ans += cnt;
    ans %= MOD;
    cout << ans << '\n';
    return 0;
}
