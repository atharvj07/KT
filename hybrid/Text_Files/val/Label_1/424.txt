t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    # Initial sum of elements at even positions.
    original_sum = sum(a[i] for i in range(0, n, 2))
    # The idea is to find the maximum subarray sum, considering the difference
    # between swapping and not swapping for each pair.
    max_diff = 0
    current_diff = 0
    for i in range(1, n, 2):
        # Calculate the difference if we swap the current element with the previous one.
        diff = a[i] - (a[i - 1] if i - 1 >= 0 else 0)
        current_diff = max(diff, current_diff + diff)
        max_diff = max(max_diff, current_diff)
    # Reset current_diff for even index swaps (considering odd indexes as starting points).
    current_diff = 0
    for i in range(2, n, 2):
        diff = a[i - 1] - (a[i] if i < n else 0)
        current_diff = max(diff, current_diff + diff)
        max_diff = max(max_diff, current_diff)
    print(original_sum + max_diff)
