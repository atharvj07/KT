#include <bits/stdc++.h>
using namespace std;
#define nl '\n'
#define INF 1e9
#define ll long long

struct DSU
{
    int n;
    vector<int> par, size;
    DSU(int _n) : n(_n)
    {
        par.resize(n + 1);
        size.resize(n + 1);
        iota(par.begin(), par.end(), 0);
        fill(size.begin(), size.end(), 1);
    }
    int getPar(int x) { return par[x] == x ? x : par[x] = getPar(par[x]); }
    void unite(int x, int y)
    {
        x = getPar(x), y = getPar(y);
        if (x == y) return;
        if (size[x] > size[y])
            swap(x, y);
        par[x] = y;
        size[y] += size[x];
    }
};

int main()
{
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    int n, m;
    cin >> n >> m;
    DSU dsu(n);
    map<pair<int, int>, int> mp;
    while (m--)
    {
        int u, v, c;
        cin >> u >> v >> c;
        if (c == 'B')
            mp[{u, v}] = mp[{v, u}] = 1;
        else
            mp[{u, v}] = mp[{v, u}] = 2;
        dsu.unite(u, v);
    }
    map<int, vector<int>> lst;
    for (auto &e : mp)
    {
        int u = dsu.getPar(e.first.first), v = dsu.getPar(e.first.second);
        lst[u].push_back(e.first.first);
        lst[v].push_back(e.first.first);
        lst[u].push_back(e.first.second);
        lst[v].push_back(e.first.second);
        lst[u].push_back(e.second);
        lst[v].push_back(e.second);
    }
    for (auto &e : lst)
    {
        int x = e.first;
        vector<int> l = e.second;
        sort(l.begin(), l.end());
        l.resize(unique(l.begin(), l.end()) - l.begin());
        int sz = l.size();
        if (sz & 1)
        {
            cout << -1 << nl;
            return 0;
        }
        for (int i = 0; i < sz; i += 2)
        {
            if (l[i] != l[i + 1])
            {
                cout << -1 << nl;
                return 0;
            }
        }
    }
    set<int> st;
    for (int i = 1; i <= n; i++)
        st.insert(dsu.getPar(i));
    vector<int> ans;
    for (auto &e : st)
        ans.push_back(e);
    int k = ans.size();
    cout << k << nl;
    for (int x : ans)
        cout << x << ' ';
    cout << nl;
    return 0;
}
