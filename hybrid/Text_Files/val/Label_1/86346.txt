import java.util.*;

public class NetworkDesign {

    static int n, p;
    static List<List<Integer>> connections;
    static List<Packet> packets;

    static class Packet {
        int source, destination, ttl;

        public Packet(int source, int destination, int ttl) {
            this.source = source;
            this.destination = destination;
            this.ttl = ttl;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        n = scanner.nextInt();
        connections = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int router = scanner.nextInt();
            int connectionsCount = scanner.nextInt();
            List<Integer> routers = new ArrayList<>();
            for (int j = 0; j < connectionsCount; j++) {
                routers.add(scanner.nextInt());
            }
            connections.add(routers);
        }

        p = scanner.nextInt();
        packets = new ArrayList<>();
        for (int i = 0; i < p; i++) {
            packets.add(new Packet(scanner.nextInt(), scanner.nextInt(), scanner.nextInt()));
        }

        for (Packet packet : packets) {
            int result = findMinimumRouters(packet.source, packet.destination, packet.ttl);
            System.out.println(result == Integer.MAX_VALUE ? "NA" : result);
        }
    }

    static int findMinimumRouters(int source, int destination, int ttl) {
        if (source == destination) {
            return 0;
        }

        Queue<int[]> queue = new LinkedList<>();
        queue.offer(new int[]{source, 1});
        boolean[] visited = new boolean[n + 1];
        visited[source] = true;

        while (!queue.isEmpty()) {
            int[] current = queue.poll();
            int router = current[0];
            int hops = current[1];

            if (router == destination) {
                return hops;
            }

            for (int nextRouter : connections.get(router - 1)) {
                if (!visited[nextRouter] && ttl > hops) {
                    visited[nextRouter] = true;
                    queue.offer(new int[]{nextRouter, hops + 1});
                }
            }
        }

        return Integer.MAX_VALUE;
    }
}
