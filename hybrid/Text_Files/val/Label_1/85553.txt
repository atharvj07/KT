#include<bits/stdc++.h>
using namespace std;
const int maxN = 3e5 + 10;

vector<pair<int, int>> edge[maxN];
int dp[maxN];
int ans;

void dfs(int node, int w) {
    dp[node] = w;
    for (auto &it : edge[node]) {
        if (dp[it.first] < it.second) {
            dfs(it.first, it.second);
        }
    }
}
signed main() {
    int n, m; cin >> n >> m;
    while (m--) {
        int x, y, z; cin >> x >> y >> z;
        edge[x].push_back({y, z});
    }
    for (int i = 1; i <= n; i++) {
        dfs(i, 0);
        ans = max(ans, dp[i]);
    }
    cout << ans << endl;
    return 0;
}
