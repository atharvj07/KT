#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0;
  int sum = 0;
  for (int i = 0; i < n; i++) {
    if (sum + a[i] <= k) {
      sum += a[i];
      ans++;
    } else {
      m--;
      sum = a[i];
      if (m < 0) break;
      ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
