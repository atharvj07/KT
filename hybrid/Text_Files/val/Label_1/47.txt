for _ in range(int(input())):
    n = int(input())
    sa = input()
    tb = input()
    if sa == tb: print('Yes')
    else:
        t = list(tb)
        i = 0
        for c in sa:
            if i < n and t.count(c) > 0:
                t.remove(c)
                i += 1
        if len(t) > 0: print('No')
        else:
            print('Yes')
            ans = []
            ta = list(sa)
            for i in range(n):
                if ta[i] != tb[i]:
                    for j in range(i+1,n):
                        if ta[j] == tb[j] and tb[i] == ta[i]:
                            ans.append([i+1,j+1])
                            ta[i],ta[j] = ta[j],ta[i]
                            break
            print(len(ans))
            for p in ans: print(*p)
