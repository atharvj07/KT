#include<bits/stdc++.h>
using namespace std;

int main(){
	int n;
	cin>>n;
	map<string,vector<string> >mp;
	vector<string> hostname,hostpath;
	int x=0;
	int y=0;
	for(int i=1;i<=n;i++){
		string s;
		cin>>s;
		int j=7;
		string host="",path="";
		while(j<s.length()){
			if(s[j]=='/'){
				path=s.substr(j);
				break;
			}
			else host+=s[j];
			j++;
		}
		hostname.push_back(host);
		hostpath.push_back(path);
	}
	for(int i=0;i<n;i++){
		for(int j=i+1;j<n;j++){
			if(hostname[i]==hostname[j]){
				if(find(mp[hostname[i]].begin(),mp[hostname[i]].end(),hostpath[j])==mp[hostname[i]].end()) mp[hostname[i]].push_back(hostpath[j]);
				if(find(mp[hostname[i]].begin(),mp[hostname[i]].end(),hostpath[i])==mp[hostname[i]].end()) mp[hostname[i]].push_back(hostpath[i]);
			}
		}
	}
	cout<<mp.size()<<endl;
	for(auto i:mp)
	{
		cout<<i.first<<" ";
		for(auto j:i.second) cout<<j<<" ";
		cout<<endl;
	}
	return 0;
}
