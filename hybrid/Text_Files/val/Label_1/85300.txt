#include <bits/stdc++.h>
#define int long long
using namespace std;
vector<int> g[100001];
int sub_size[100001];
bool vis[100001];
int centroid_size, centroid_node;
int n;

void dfs1(int node, int parent) {
  sub_size[node] = 1;
  for (auto x : g[node]) {
    if (x == parent || vis[x]) continue;
    dfs1(x, node);
    sub_size[node] += sub_size[x];
  }
}

void dfs2(int node, int parent, int sz) {
  int mx_sz = sz - sub_size[node];
  for (auto x : g[node]) {
    if (x == parent || vis[x]) continue;
    mx_sz = max(mx_sz, sub_size[x]);
  }
  if (mx_sz <= n / 2) {
    centroid_size = mx_sz, centroid_node = node;
  }
  for (auto x : g[node]) {
    if (x == parent || vis[x]) continue;
    dfs2(x, node, sz);
  }
}

int find_centroid() {
  dfs1(1, 1);
  n = sub_size[1];
  dfs2(1, 1, n);
  return centroid_node;
}

int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    int u, v, x, y;
    cin >> n;
    for (int i = 0; i <= n; i++) {
      g[i].clear();
      sub_size[i] = 0;
      vis[i] = false;
    }
    for (int i = 1; i < n; i++) {
      cin >> u >> v;
      g[u].push_back(v);
      g[v].push_back(u);
    }
    int centroid = find_centroid();
    for (int i = 0; i < g[centroid].size(); i++) {
      vis[g[centroid][i]] = true;
      x = centroid, y = g[centroid][i];
      break;
    }
    centroid = find_centroid();
    for (int i = 0; i < g[centroid].size(); i++) {
      if (!vis[g[centroid][i]]) {
        vis[g[centroid][i]] = true;
        break;
      }
    }
    cout << x << ' ' << y << endl;
    cout << centroid << ' ' << g[centroid][0] << endl;
  }
  return 0;
}
