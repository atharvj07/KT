import java.util.*;
import java.io.*;

public class Main {

    static int n, m, q, ans;
    static int[] p;
    static int[][] lca;
    static boolean[][] dp;
    static boolean[] vis;

    static int find(int x) {
        return p[x] == x ? x : (p[x] = find(p[x]));
    }

    static void dfs(int cur, int pa) {
        vis[cur] = true;
        for (int i = 0; i < n; i++) {
            if (i == cur || vis[i] || lca[cur][i] != -1 || lca[i][cur] != -1) continue;
            if (find(cur) != find(i)) {
                p[find(i)] = find(cur);
                lca[cur][i] = lca[i][cur] = pa;
                dfs(i, pa);
            }
            else {
                dp[cur][i] = dp[i][cur] = true;
                dfs(i, cur);
            }
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(in.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        q = Integer.parseInt(st.nextToken());
        p = new int[n];
        lca = new int[n][n];
        dp = new boolean[n][n];
        vis = new boolean[n];
        for (int i = 0; i < n; i++) p[i] = i;
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(in.readLine());
            int u = Integer.parseInt(st.nextToken()) - 1;
            int v = Integer.parseInt(st.nextToken()) - 1;
            lca[u][v] = lca[v][u] = -1;
        }
        for (int i = 0; i < q; i++) {
            st = new StringTokenizer(in.readLine());
            int u = Integer.parseInt(st.nextToken()) - 1;
            int v = Integer.parseInt(st.nextToken()) - 1;
            int c = Integer.parseInt(st.nextToken()) - 1;
            lca[u][v] = lca[v][u] = c;
        }
        dfs(0, 0);
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) continue;
                if (dp[i][j]) {
                    ans = -1;
                    break;
                }
            }
        }
        System.out.println(ans == -1 ? 0 : (1 << m));
    }
}
