#include <bits/stdc++.h>

using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n, m;
  cin >> n >> m;
  string s, t;
  cin >> s >> t;
  vector<int> pos[26];
  for (int i = 0; i < n; i++) pos[s[i] - 'a'].push_back(i);
  int ans = 0;
  for (int i = 0; i < m; i++) {
    int c = t[i] - 'a';
    if (pos[c].empty()) {
      cout << -1 << '\n';
      return 0;
    }
    int j = 0;
    while (pos[c][j] <= i) j++;
    if (j < pos[c].size()) ans = max(ans, pos[c][j] - i);
  }
  cout << ans << '\n';
}
