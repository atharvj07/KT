import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        // Read the number of windows and service times
        String[] windowCounts = reader.readLine().split(" ");
        int k1 = Integer.parseInt(windowCounts[0]);
        int k2 = Integer.parseInt(windowCounts[1]);
        int k3 = Integer.parseInt(windowCounts[2]);

        String[] serviceTimes = reader.readLine().split(" ");
        int t1 = Integer.parseInt(serviceTimes[0]);
        int t2 = Integer.parseInt(serviceTimes[1]);
        int t3 = Integer.parseInt(serviceTimes[2]);

        // Read the number of people and their arrival times
        int n = Integer.parseInt(reader.readLine());
        String[] arrivalTimes = reader.readLine().split(" ");
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = Integer.parseInt(arrivalTimes[i]);
        }

        // Sort the arrival times in ascending order
        Arrays.sort(c);

        // Initialize the service times for each person
        int[] s1 = new int[n];
        int[] s2 = new int[n];
        int[] s3 = new int[n];

        // Calculate the service times for each person
        s1[0] = t1;
        s2[0] = t2 + t1;
        s3[0] = t3 + s2[0];
        for (int i = 1; i < n; i++) {
            s1[i] = Math.max(c[i], s1[i - 1]) + t1;
            s2[i] = Math.max(c[i], s2[i - 1]) + t2;
            s3[i] = Math.max(c[i], s3[i - 1]) + t3;
        }

        // Find the maximum service time
        int maxServiceTime = s3[n - 1];

        // Print the maximum service time
        System.out.println(maxServiceTime);
    }
}
