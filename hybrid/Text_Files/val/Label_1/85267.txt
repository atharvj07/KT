#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == '>' && i + 1 < n) {
        if (s[i + 1] == '<') {
          i++;
          ans++;
        }
      } else if (s[i] == '<' && i > 0) {
        if (s[i - 1] == '>') {
          i--;
          ans++;
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
