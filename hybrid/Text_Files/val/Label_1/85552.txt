#include <cstdio>
#include <algorithm>
#include <vector>
using namespace std;

#define maxn 300005
struct edge{
	int to;
	int cost;
	edge(){}
	edge(int to, int cost):to(to),cost(cost){}
	bool operator < (edge rhs){ return rhs.to == this->to ? rhs.cost < this->cost : rhs.to > this->to; }
};

vector<edge> G[maxn];
int dp[2][maxn], ans = 0;
int n, m;

int main(){
	scanf("%d %d", &n, &m);
	for(int i=0;i<m;i++){
		int u, v, w;
		scanf("%d %d %d", &u, &v, &w);
		G[u].push_back(edge(v, w));
	}
	
	for(int i=0;i<=n;i++) dp[1][i] = dp[0][i] = -1e9;
	
	dp[0][1] = 0;
	int t=1, nt=0;
	for(int i=1;i<=n;i++){
		for(int j=0;j<G[i].size();j++){
			int cost = G[i][j].cost;
			for(int k=1;k<=n;k++){
				if(dp[nt][k] < dp[t][k]) ans = max(ans, dp[t][k]);
				dp[nt][k] = max(dp[nt][k], max(dp[t][k], dp[t][i]) + cost);
			}
			nt^=1; t^=1;
			dp[nt][i]=-1e9;
		}
	}
	
	for(int i=1;i<=n;i++) if(dp[nt][i] < dp[t][i]) ans = max(ans, dp[t][i]);
	
	printf("%d", ans);
	
	return 0;
}
