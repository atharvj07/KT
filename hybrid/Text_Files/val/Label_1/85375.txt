#include<bits/stdc++.h>
using namespace std;

int main(){
    int n,r;
    cin>>n>>r;
    vector<pair<int,int>> v;
    for(int i=0;i<n;++i){
        int p,q;
        cin>>p>>q;
        v.push_back({p,q});
    }

    for(int i=0;i<n;++i){
        if(v[i].first<=r) r+=v[i].second;
        else{
            cout<<"NO"<<endl;
            return 0;
        }
    }
    cout<<"YES"<<endl;
}
