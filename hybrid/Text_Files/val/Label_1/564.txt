import sys
input = sys.stdin.readline

n, m, k = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
ans = 0
for i in range(n):
    for j in range(m):
        if a[i][j] == 0:
            continue
        if i + 1 < n and j + 1 < m:
            if a[i + 1][j] == 1 and a[i][j + 1] == 1 and a[i + 1][j + 1] == 1:
                a[i][j] = 2
                a[i + 1][j] = 2
                a[i][j + 1] = 2
                a[i + 1][j + 1] = 2
                ans += 1
        if i + 2 < n and i % 2 == 0 and j + 1 < m and a[i + 1][j] == 1 and a[i + 2][j] == 1 and a[i + 1][j + 1] == 1:
            a[i][j] = 2
            a[i + 1][j] = 2
            a[i + 2][j] = 2
            a[i + 1][j + 1] = 2
            ans += 1
        if j + 2 < m and j % 2 == 0 and i + 1 < n and a[i][j + 1] == 1 and a[i][j + 2] == 1 and a[i + 1][j + 1] == 1:
            a[i][j] = 2
            a[i][j + 1] = 2
            a[i][j + 2] = 2
            a[i + 1][j + 1] = 2
            ans += 1
flag = False
for i in range(n):
    for j in range(m):
        if a[i][j] == 1:
            flag = True
if ans <= k and flag == False:
    print(ans)
else:
    print(-1)
