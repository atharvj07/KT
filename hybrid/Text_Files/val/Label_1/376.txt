t = int(input())
for _ in range(t):
    x, y = map(int, input().split())
    if y==0:
        print(x*2-1)
    elif y==1:
        print(x*2)
    elif x==0:
        print(y*2-1)
    elif x==1:
        print(y*2)
    else:
        if x==1 or y==1:
            a = x+y-1
            if a%2==1:
                a+=1
            b = a+2
            print(b)
        else:
            a = x+y
            if a%2==0:
                a+=1
            b = a+4
            print(b)
