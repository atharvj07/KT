#include<bits/stdc++.h>
using namespace std;
const int maxN = 4001;
long long v[maxN], d[maxN], p[maxN], idx[maxN];
int n, cnt, cr[maxN];
void go(int u) {
	if(cr[u]) return;
	cr[u] = 1;
	long long now = p[u];
	for(int i = 1; i < n; ++i) {
		long long id = idx[i];
		now -= i < u ? v[id] : (i == u ? 0 : v[id] - i + u);
		if(p[id] >= now) now = p[id];
		else {
			go(id);
			now = p[u] - d[id] - i + u;
			cnt--;
		}
	}
	cnt++;
}
int main() {
	cin >> n;
	for(int i = 0; i < n; ++i) {
		cin >> v[i] >> d[i] >> p[i];
		idx[v[i]] = i;
	}
	for(int i = 0; i < n; ++i) {
		if(!cr[i]) {
			cnt = 0;
			go(i);
		}
	}
	cout << cnt << endl;
	for(int i = 0; i < n; ++i) {
		if(cr[i]) cout << i + 1 << " ";
	}
	return 0;
}
