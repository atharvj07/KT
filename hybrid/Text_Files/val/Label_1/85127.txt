#include <bits/stdc++.h>
using namespace std ;
int main(){
    int n;cin>>n;int a[n];for(int i=0;i<n;i++)cin>>a[i];sort(a,a+n);int ans=1,prv=a[0];
    for(int i=1;i<n;i++){
        if(a[i]-prv<=5){
            ans++;
        }
        else{
            prv=a[i];
            ans=1;
        }
    }
    cout<<ans<<'\n';
}
