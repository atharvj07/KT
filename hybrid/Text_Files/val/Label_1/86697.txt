import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n + 1];
        int cnt = 0;
        for (int i = 0; i < m; i++) {
            int op = sc.next().charAt(0);
            int x = sc.nextInt();
            if (op == '+') a[x] = 1;
            else a[x] = 0;
            boolean ok = false;
            for (int j = 1; j <= n; j++) {
                if (a[j] > 0) {
                    ok = true;
                    break;
                }
            }
            if (ok) cnt++;
        }
        System.out.println(cnt);
        for (int i = 1; i <= n; i++) {
            if (a[i] > 0) System.out.print(i + " ");
        }
        System.out.println();
    }
}
