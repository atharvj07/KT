import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while (t-- > 0) {
            long a = sc.nextLong();
            long b = sc.nextLong();
            long c = sc.nextLong();
            long d = sc.nextLong();

            long x = (a + b) / 2;
            long y = (b + c) / 2;
            long z = (c + d) / 2;

            System.out.println(x + " " + y + " " + z);
        }
    }
}
