import java.util.Scanner;

public class Olympiad {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] students = new int[n];
        for (int i = 0; i < n; i++) {
            students[i] = scanner.nextInt();
        }

        int[] programmers = new int[n];
        int[] mathematicians = new int[n];
        int[] athletes = new int[n];
        int pCount = 0, mCount = 0, aCount = 0;
        for (int i = 0; i < n; i++) {
            if (students[i] == 1) {
                programmers[pCount++] = i + 1;
            } else if (students[i] == 2) {
                mathematicians[mCount++] = i + 1;
            } else {
                athletes[aCount++] = i + 1;
            }
        }

        int teams = Math.min(pCount, Math.min(mCount, aCount));
        System.out.println(teams);
        for (int i = 0; i < teams; i++) {
            System.out.println(programmers[i] + " " + mathematicians[i] + " " + athletes[i]);
        }
    }
}
