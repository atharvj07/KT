#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int l = 0, r = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == '>')
        l++;
      if (s[i] == '<')
        r++;
    }
    cout << min(l, r) << endl;
  }
  return 0;
}
