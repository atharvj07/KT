import sys

def solve():
    k = int(input())
    for _ in range(k):
        n = int(input())
        s = input()
        t = input()
        if s == t:
            print("Yes")
            print(0)
        else:
            diffs = []
            for i in range(n):
                if s[i] != t[i]:
                    diffs.append(i)
            if len(diffs) % 2 == 0:
                print("Yes")
                print(len(diffs) // 2)
                for i in range(0, len(diffs), 2):
                    print(diffs[i] + 1, diffs[i + 1] + 1)
            else:
                print("No")

solve()
