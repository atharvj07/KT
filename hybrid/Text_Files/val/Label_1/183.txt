n = int(input().strip())
prices = list(map(int, input().strip().split()))
q = int(input().strip())
budgets = [int(input().strip()) for _ in range(q)]

prices.sort()

for budget in budgets:
    answer = 0
    low, high = 0, n-1
    while low <= high:
        mid = (low + high) // 2
        if prices[mid] <= budget:
            answer = mid + 1
            low = mid + 1
        else:
            high = mid - 1
    print(answer)
