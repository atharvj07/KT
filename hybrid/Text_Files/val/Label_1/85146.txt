#include<iostream>
#include<queue>
#include<vector>
using namespace std;

int n;
vector<int> a;
queue<int> q;
vector<pair<int,int>> v;

int main() {
    cin >> n;
    a.resize(n + 1);
    bool f = 0;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        if (!a[i]) f = 1;
    }
    if (f) { cout << -1 << endl; return 0; }
    q.push(1);
    while (!q.empty()) {
        int t = q.front(); q.pop();
        for (int i = 1; i <= n; i++) {
            if (i == t) continue;
            if (a[i] > 0) {
                v.push_back({t, i});
                a[i]--;
                a[t]--;
                if (a[t] == 0) q.push(t);
                if (a[i] == 0) q.push(i);
            }
        }
    }
    cout << v.size() << endl;
    for (auto x : v) {
        cout << x.first << " " << x.second << endl;
    }
    return 0;
}
