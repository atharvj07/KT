#include <iostream>
using namespace std;
#define int long long
const int mod = static_cast<int>(1e9) + 7;
int fastpow(int a, int p) {
    if (p == 0) { return 1; }
    int x = fastpow(a, p / 2);
    x = x * x % mod;
    if (p & 1) { x = x * a % mod; }
    return x;
}
int32_t main() {
    int n, q;
    cin >> n >> q;
    int ans[q + 1]{}, pref[n + 5]{};
    for (int i = n; i >= 1; i--) { pref[i] = (pref[i + 1] + fastpow(2, i - 1)) % mod; }
    for (int i = 1; i <= q; i++) {
        int x, y;
        char c;
        cin >> x >> y >> c;
        if (c == 'U') { ans[i] = pref[y + 1]; } else { ans[i] = pref[n - y + 1]; }
    }
    for (int i = 1; i <= q; i++) { cout << ans[i] << '\n'; }
}
