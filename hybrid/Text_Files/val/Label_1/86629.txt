import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        boolean isMirror = true;
        for (int i = 0; i < s.length() / 2; i++) {
            if (s.charAt(i) == 'b' && s.charAt(s.length() - i - 1) == 'd' ||
                s.charAt(i) == 'd' && s.charAt(s.length() - i - 1) == 'b' ||
                s.charAt(i) == 'p' && s.charAt(s.length() - i - 1) == 'q' ||
                s.charAt(i) == 'q' && s.charAt(s.length() - i - 1) == 'p') {
                continue;
            } else {
                isMirror = false;
                break;
            }
        }
        System.out.println(isMirror ? "Yes" : "No");
    }
}
