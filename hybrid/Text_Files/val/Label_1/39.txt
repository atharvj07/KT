import sys

n, m = map(int, input().split())
a = list(map(int, input().split()))
l = [[]]
r = [[]]

for i in range(n):
    if a[i] not in l[-1]:
        l.append([i])
        r.append([i])
    l[-1].append(i)

for i in range(m):
    x, y = map(int, input().split())
    x -= 1
    y -= 1
    if a[x] == a[y]:
        print(min(abs(x - i) for i in l[a[x]] if x <= i <= y) if l[a[x]][-1] >= x else sys.maxsize)
    else:
        print(-1)
