#include <iostream>
using namespace std;
typedef long long ll;
ll gcd(ll a, ll b) {
    if (a%b==0) return b;
    return gcd(b,a%b);
}
int main() {
    ll a,b; cin >> a >> b;
    ll g = gcd(a,b);
    a/=g, b/=g;
    ll ans=0;
    while (b!=1) {
        if (a%b==0) {
            a/=b;
            ans++;
        } else {
            ll d=b, c=b/a;
            b=a%b;
            a=d;
            ans+=c;
        }
    }
    cout << ans << endl;
    return 0;
}
