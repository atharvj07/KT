#include <bits/stdc++.h>
using namespace std;
const int N = 10;
typedef pair<int, int> pii;

inline bool valid(int x, int y) {
    return 0 <= x && x < N && 0 <= y && y < N;
}

const int dx[] = {-1, 0, 1, 0};
const int dy[] = {0, 1, 0, -1};

int check(char g[][N], int x, int y) {
    if (g[x][y] != '.') return 0;
    for (int i = 0; i < 4; i++) {
        int cnt = 0;
        for (int j = 0;; j++) {
            int nx = x + j * dx[i], ny = y + j * dy[i];
            if (!valid(nx, ny) || g[nx][ny] != 'X') break;
            cnt++;
            if (cnt >= 5) return 1;
        }
    }
    return 0;
}

int main() {
    char g[N][N];
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            cin >> g[i][j];
        }
    }
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            if (check(g, i, j)) {
                cout << "YES\n";
                return 0;
            }
        }
    }
    cout << "NO\n";
}
