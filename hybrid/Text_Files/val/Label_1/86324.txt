import java.util.Scanner;
import java.util.ArrayList;
import java.util.HashMap;

public class Main {
    static boolean result = false;
    static boolean[] visited;
    static int[] prev;
    static ArrayList<Integer>[] adj;
    static HashMap<Integer, Integer> entrance;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        adj = new ArrayList[n + 1];
        visited = new boolean[n + 1];
        prev = new int[n + 1];
        entrance = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 1; i <= m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a].add(b);
            adj[b].add(a);
            if (a == 1) {
                entrance.put(b, i);
            } else if (b == 1) {
                entrance.put(a, i);
            }
        }
        dfs(1, 0);
        if (result) {
            System.out.println("Yes");
            for (int i = 2; i <= n; i++) {
                System.out.println(prev[i]);
            }
        } else {
            System.out.println("No");
        }
    }

    static void dfs(int v, int p) {
        if (visited[v]) {
            result = true;
            return;
        }
        visited[v] = true;
        for (int u : adj[v]) {
            if (u != p) {
                prev[u] = v;
                dfs(u, v);
                if (result) {
                    return;
                }
            }
        }
        if (entrance.containsKey(v)) {
            result = true;
            return;
        }
    }
}
