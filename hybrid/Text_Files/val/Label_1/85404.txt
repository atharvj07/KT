#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int MAXN = 2e5+10;
const ll MOD = 1e9+7;
ll a[MAXN], b[MAXN];
int main(){
    ll n,m,k;
    cin >> n >> m >> k;
    ll ans = 0;
    for(int i = 1; i <= n; i ++){
        cin >> a[i] >> b[i];
        if(a[i]>b[i]) swap(a[i],b[i]);
        ans += a[i];
    }
    sort(b+1,b+n+1,greater<ll>());
    for(int i = 1; i <= m && i <= n; i++) ans += b[i], k--;
    for(int i = m+1; i <= n && k>=2; i+=2) ans+=b[i], k-=2;
    cout << ans << "\n";
}
