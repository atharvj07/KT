#include<bits/stdc++.h>
using namespace std;
int main()
{
	long long n, a, b, h, d, ans = 0;
	cin >> n >> a >> b;
	vector<pair<long long, long long>> v;
	for (int i = 0; i < n; i++)
	{
		cin >> h >> d;
		v.push_back({ h, d });
		ans += d;
	}
	sort(v.begin(), v.end(), greater<pair<long long, long long>>()); // sorting in decreasing order
	for (int i = 0; i < n; i++)
	{
		if (b > 0 and v[i].second < v[i].first)
		{
			ans -= v[i].second;
			ans += v[i].first;
			b--;
		}
		else if (a > 0)
		{
			ans += v[i].first;
			a--;
		}
		else
			break;
	}
	cout << ans << endl;
	return 0;
}
