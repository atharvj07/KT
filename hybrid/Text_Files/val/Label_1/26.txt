def max_seq_value(blocks):
    # Initialize adjacency matrix for the graph
    adj_matrix = [[0]*5 for _ in range(5)]
    for block in blocks:
        color1, value, color2 = block
        adj_matrix[color1][color2] += value
        adj_matrix[color2][color1] += value

    # Try all combinations of removing one edge
    max_value = 0
    for i in range(1, 5):
        for j in range(i+1, 5):
            edge_removed_value = adj_matrix[i][j]
            adj_matrix[i][j] = adj_matrix[j][i] = 0
            max_value = max(max_value, find_max_cycle(adj_matrix) + edge_removed_value)
            adj_matrix[i][j] = adj_matrix[j][i] = edge_removed_value

    return max_value

def find_max_cycle(adj_matrix):
    max_path_value = 0
    for i in range(1, 5):
        visited = [False]*5
        max_path_value = max(max_path_value, dfs(i, adj_matrix, visited, 0))
    return max_path_value

def dfs(node, adj_matrix, visited, path_value):
    max_value = path_value
    visited[node] = True
    for neighbor in range(1, 5):
        if not visited[neighbor] and adj_matrix[node][neighbor] > 0:
            max_value = max(max_value, dfs(neighbor, adj_matrix, visited, path_value + adj_matrix[node][neighbor]))
    visited[node] = False
    return max_value

n = int(input().strip())
blocks = [list(map(int, input().strip().split())) for _ in range(n)]
print(max_seq_value(blocks))
