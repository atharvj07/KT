#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int N = 3e5+10;
ll a[N], ans[N];
vector<ll> adj[N], order;
bool vis[N], rev[N];

void dfs1(int u) {
    vis[u] = 1;
    for (int v : adj[u]) {
        if (!vis[v]) dfs1(v);
    }
    order.push_back(u);
}

void dfs2(int u) {
    rev[u] = 1;
    for (int v : adj[u]) {
        if (!rev[v]) dfs2(v);
    }
    ans[u] = a[u];
}

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        adj[i].clear();
    }
    ll p[m];
    for (int i = 0; i < m; i++) {
        cin >> p[i];
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            ll t = a[i] / p[j];
            ll rem = a[i] % p[j];
            if (rem != 0) t++;
            int L = i, R = n - 1;
            while (L <= R) {
                int mid = (L + R) / 2;
                if (t + p[j] * mid >= a[mid]) {
                    L = mid + 1;
                } else {
                    R = mid - 1;
                }
            }
            ll ind = L;
            if (ind != n && a[ind] == t * p[j]) {
                adj[i].push_back(ind);
            }
        }
    }

    for (int i = 0; i < n; i++) {
        if (!vis[i]) dfs1(i);
    }

    reverse(order.begin(), order.end());

    for (int u : order) {
        if (!rev[u]) dfs2(u);
    }

    int fl = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (ans[i] == a[i]) {
                fl = 1;
                cout << "YES" << endl;
                cout << a[i] << " " << j + 1 << endl;
                break;
            }
        }
        if (fl) break;
    }

    if (!fl) cout << "NO" << endl;
}
