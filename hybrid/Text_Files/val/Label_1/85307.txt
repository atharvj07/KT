#include <iostream>
#include <string>
#include <vector>
#include <unordered_map>
#include <set>

using namespace std;

int main() {
  int n;
  cin >> n;

  unordered_map<string, set<string>> hostname_to_path;
  unordered_map<string, bool> hostname_processed;

  for (int i = 0; i < n; i++) {
    string address;
    cin >> address;

    size_t pos = address.find('/');
    string hostname = address.substr(7, pos - 7);
    string path = address.substr(pos + 1);

    hostname_to_path[hostname].insert(path);
    hostname_processed[hostname] = false;
  }

  vector<set<string>> groups;

  for (auto &entry : hostname_to_path) {
    if (hostname_processed[entry.first]) continue;

    set<string> group = {entry.first};
    hostname_processed[entry.first] = true;

    for (auto &hostname : hostname_to_path) {
      if (hostname_to_path[entry.first] == hostname_to_path[hostname.first]) {
        hostname_processed[hostname.first] = true;
        group.insert(hostname.first);
      }
    }

    if (group.size() > 1) {
      groups.push_back(group);
    }
  }

  cout << groups.size() << endl;
  for (auto &group : groups) {
    for (auto &hostname : group) {
      cout << hostname << " ";
    }
    cout << endl;
  }

  return 0;
}
