#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define rep(i, n) for (int i = 0; i < (n); ++i)

int main() {
  ll n;
  cin >> n;

  vector<ll> ans(n - 1);
  for (ll k = 2; k <= n; ++k) {
    if (k % 2 == 0)
      ans[k - 2] = k / 2;
    else
      ans[k - 2] = k;
  }
  for (ll i : ans) cout << i << " ";
  cout << endl;
  return 0;
}
