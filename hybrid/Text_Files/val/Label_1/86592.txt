import java.util.*;
import java.io.*;
public class Main {
    static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    static int T, a;
    public static void main(String[] args) throws IOException {
        T = Integer.parseInt(br.readLine(), 10);
        while(T-->0) {
            a = Integer.parseInt(br.readLine(), 10);
            if(a <= 178) {
                System.out.println(180 / a);
            } else {
                System.out.println(-1);
            }
        }
    }
}
