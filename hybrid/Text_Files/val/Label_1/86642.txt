import java.util.*;

class Main {
    static int n, k;
    static long[][] dp;
    static int[] s, t, c;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        dp = new long[n + 1][k + 1];
        s = new int[n];
        t = new int[n];
        c = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
            t[i] = sc.nextInt();
            c[i] = sc.nextInt();
        }
        long ans = rec(0, 0);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            sb.append((dp[i + 1][k] == dp[i][k]) ? 0 : 1).append(' ');
        }
        System.out.println(sb.toString());
    }

    static long rec(int pos, int cnt) {
        if (pos == n || cnt == k) {
            return 0;
        }
        if (dp[pos][cnt] != 0) {
            return dp[pos][cnt];
        }
        long ans = rec(pos + 1, cnt);
        if (pos + t[pos] - 1 < n && cnt + 1 <= k && dp[pos + t[pos]][cnt + 1] < dp[pos][cnt] + c[pos]) {
            dp[pos + t[pos]][cnt + 1] = dp[pos][cnt] + c[pos];
            ans = Math.max(ans, rec(pos + t[pos], cnt + 1));
        }
        return dp[pos][cnt] = ans;
    }
}
