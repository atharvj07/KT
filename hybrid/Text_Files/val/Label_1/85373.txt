#include <bits/stdc++.h>
using namespace std;

const int N = 500005;
int n;
int m[N];
vector<pair<int, int>> adj[N];
int dp[N];
int64_t ans = 0;

void dfs(int u, int p) {
    dp[u] = m[u];
    for (auto x : adj[u]) {
        int v = x.first, w = x.second;
        if (v == p) continue;
        dfs(v, u);
        dp[u] = max(dp[u], min(m[u], dp[v] + w));
    }
    ans += dp[u];
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> m[i];
    for (int i = 2; i <= n; i++) {
        int p, w;
        cin >> p >> w;
        adj[i].push_back({p, w});
        adj[p].push_back({i, w});
    }
    dfs(1, 1);
    cout << ans << endl;
}
