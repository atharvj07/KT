n, m = map(int, input().split())
boss = [0]*n
docs = []

for _ in range(m):
    event = list(map(int, input().split()))
    if event[0] == 1:
        x, y = event[1]-1, event[2]-1
        boss[x] = y+1
    elif event[0] == 2:
        docs.append(event[1]-1)
    else:
        x, i = event[1]-1, event[2]-1
        signer = docs[i]
        found = "NO"
        while signer != 0:
            if signer == x:
                found = "YES"
                break
            signer = boss[signer-1]
        if signer == x:
            found = "YES"
        print(found)
