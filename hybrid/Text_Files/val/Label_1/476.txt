n, m = map(int, input().split())
a = list(map(int, input().split()))

def fib(n):
    fibs = [1, 1]
    for i in range(2, n+1):
        fibs.append(fibs[-1] + fibs[-2])
    return fibs

for _ in range(m):
    op = list(map(int, input().split()))
    if op[0] == 1:
        a[op[1]-1] = op[2]
    elif op[0] == 2:
        l, r = op[1], op[2]
        fibs = fib(r-l)
        print(sum(fibs[i] * a[l-1+i] for i in range(r-l+1)) % 1000000000)
    elif op[0] == 3:
        l, r, d = op[1], op[2], op[3]
        for i in range(l-1, r):
            a[i] += d
