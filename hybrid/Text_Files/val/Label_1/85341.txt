#include <bits/stdc++.h>
using namespace std;
#define Long long long
#define endl "\n"

const int N = 100;

Long t, n, m, a[N][N], dp1[N], dp2[N], ans;
vector<int> v;
map<int, int> mp;

int main()
{
	cin >> t;
	while (t--)
	{
		ans = 0;
		cin >> n >> m;
		for (int i = 0; i < n; i++)
			for (int j = 0; j < m; j++) cin >> a[i][j];
		for (int j = 0; j < m; j++)
		{
			v.clear();
			for (int i = 0; i < n; i++) v.push_back(a[i][j]);
			int cnt = 0;
			for (int i = 0; i < n; i++) cnt += v[i] != v[n - i - 1];
			ans += cnt / 2;
		}
		for (int i = 0; i < n; i++)
		{
			v.clear();
			for (int j = 0; j < m; j++) v.push_back(a[i][j]);
			int cnt = 0;
			for (int j = 0; j < m; j++) cnt += v[j] != v[m - j - 1];
			ans += cnt / 2;
		}
		cout << ans << endl;
	}
	return 0;
}
