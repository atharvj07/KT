#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int ans = 0;
    for (int i = 1; i < n; i++) {
        int diff = a[i] + 1 - a[i - 1];
        if (diff != k) {
            ans++;
            if (diff > k) {
                cout << "- " << i << " " << diff - k << endl;
            } else {
                cout << "+ " << i << " " << k - diff << endl;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
