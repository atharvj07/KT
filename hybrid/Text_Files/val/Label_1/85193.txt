#include <bits/stdc++.h>

using namespace std;

#define ll long long

const int MAXN = 3e5 + 10;

int n;
int a[MAXN];
int b[MAXN];

vector<pair<int, int>> ans;

int main() {
    
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    for (int i = 1; i <= n; i++) {
        b[a[i]] = i;
    }

    vector<pair<int, int>> v;

    for (int i = 1; i <= n; i++) {
        if (a[i] != i) {
            v.push_back({a[i], i});
        }
    }

    for (int i = 0; i < (int)v.size(); i++) {
        int p1 = v[i].first;
        int p2 = v[i].second;

        if (2 * abs(p1 - p2) >= n) {
            ans.push_back({p1, p2});
            swap(a[p1], a[p2]);
            swap(b[p1], b[p2]);
        }
    }

    int sz = ans.size();

    cout << sz << '\n';

    for (int i = 0; i < sz; i++) {
        cout << ans[i].first << ' ' << ans[i].second << '\n';
    }

    return 0;
}
