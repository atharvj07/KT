from collections import deque

n, m, s = map(int, input().split())

g = [[] for i in range(n+1)]
edge = []
for i in range(m):
    t, u, v = map(int, input().split())
    g[u].append((v, t))
    g[v].append((u, t))
    if t == 2:
        edge.append((u, v))

q = deque([s])
vis = [False for i in range(n+1)]
vis[s] = True
ans1 = []
ans2 = []
cnt = 0

while q:
    u = q.popleft()
    ans1.append(u)
    ans2.append(u)
    cnt += 1
    for v, t in g[u]:
        if not vis[v]:
            if t == 1:
                q.append(v)
                vis[v] = True
            elif u < v:
                ans1[-1] = v
                q.append(v)
                vis[v] = True
            else:
                ans2[-1] = v
                q.append(v)
                vis[v] = True

print(cnt)
for u, v in edge:
    if ans1[0] == u and ans1[-1] == v:
        print('+', end='')
    elif ans1[0] == v and ans1[-1] == u:
        print('-', end='')
print()

print(cnt)
for u, v in edge:
    if ans2[0] == u and ans2[-1] == v:
        print('+', end='')
    elif ans2[0] == v and ans2[-1] == u:
        print('-', end='')
