#include <bits/stdc++.h>
using namespace std;
string s;
const int maxn = (int)2e3 + 6;
long long c[maxn];
int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> c[i]; }
    if (c[n]) {
        for (int i = 0; i < c[n]; i++) { s.push_back('a' + n - 1); }
    }
    long long sum = c[n];
    for (int j = n - 1; j; j--) {
        sum += c[j];
        long long rem = sum % j;
        int rep = sum / j;
        for (int i = 0; i < j; i++) { s.push_back('a' + i); }
        for (int i = 1; i < rem; i++) { s.push_back(s[(i - 1) * rep]); }
    }
    cout << s << endl;
    return 0;
}
