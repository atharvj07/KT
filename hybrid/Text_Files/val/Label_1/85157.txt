#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, k, p = 0;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n - 1; i++) {
        if (a[i] + 1 - a[i + 1] < k) { p += k - (a[i] + 1 - a[i + 1]); a[i + 1] += k - (a[i] + 1 - a[i + 1]); cout << "+ " << i + 2 << " " << k - (a[i] + 1 - a[i + 1]) << endl; }
        if (a[i] + 1 - a[i + 1] > k && a[i + 1] > 1) { p += a[i] + 1 - a[i + 1] - k; a[i + 1] -= a[i] + 1 - a[i + 1] - k; cout << "- " << i + 2 << " " << a[i] + 1 - a[i + 1] - k << endl; }
    }
    cout << p << endl;
    return 0;
}
