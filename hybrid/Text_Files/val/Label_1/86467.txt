import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] p = new int[n];
        if (k == 1) {
            for (int i = 0; i < n; i++) p[i] = i + 1;
        } else {
            p[0] = k + 1;
            for (int i = 1; i < k; i++) p[i] = i;
            for (int i = k; i < n; i++) p[i] = i + 1;
            int[] t = new int[k];
            for (int i = 1; i < k; i++) t[i - 1] = p[i] - p[i - 1];
            Arrays.sort(t);
            int m = p[k - 1] + t[0];
            for (int i = k; i < n; i++) p[i] = m;
        }
        for (int i = 0; i < n; i++) System.out.print(p[i] + " ");
    }
}
