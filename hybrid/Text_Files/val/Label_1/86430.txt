import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] p = new int[n];
        int[] s = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            s[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }
        int[] inc = new int[m];
        int[] pref = new int[m];
        for (int i = 0; i < m; i++) {
            inc[i] = scanner.nextInt();
        }
        for (int i = 0; i < m; i++) {
            pref[i] = scanner.nextInt();
        }
        for (int i = 0; i < m; i++) {
            int cnt = 0;
            for (int j = 0; j < n; j++) {
                if (p[j] <= inc[i] && inc[i] <= s[j] && Math.abs(b[j] - pref[i]) <= inc[i] - p[j]) {
                    cnt++;
                }
            }
            System.out.print(cnt + " ");
        }
        System.out.println();
    }
}
