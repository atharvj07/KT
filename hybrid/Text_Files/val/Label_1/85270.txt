#include <cstdio>
#include <vector>
#include <algorithm>

int main() {
  int n;
  scanf("%d", &n);
  std::vector<int> v(n * 2);
  for (int i = 0; i < n; i++) {
    int h, m;
    scanf("%d %d", &h, &m);
    v[i] = h * 60 + m;
    v[i + n] = h * 60 + m + 1;
  }
  std::sort(v.begin(), v.end());
  int ans = 0, cnt = 0;
  for (int i = 0; i < v.size(); i++) {
    if (cnt == 0) ans++;
    cnt += (v[i] == v[i - 1] ? 0 : 1);
    cnt--;
  }
  printf("%d\n", ans);
  return 0;
}
