#include <iostream>
const int MOD = 1e9 + 7;

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    int a[n];
    for (int i = 0; i < n; i++) { std::cin >> a[i]; }
    bool ok = true, can = true;
    for (int j = 2; j < n; j++) {
      if (a[(j + 1) % n] - a[j] != a[j] - a[(j - 1 + n) % n]) {
        ok = false;
      }
      if (a[(j + 1) % n] - a[j] < 0) {
        can = false;
      }
    }
    if (!ok || !can) {
      std::cout << "-1\n";
      continue;
    }
    if (a[1] == a[0]) {
      std::cout << "0\n";
      continue;
    }
    int m = MOD;
    for (int i = 0; i < n; i++) {
      int dif = a[(i + 1) % n] - a[i];
      if (m > dif) m = dif;
    }
    std::cout << m << " " << a[1] - a[0] % m << '\n';
  }
  return 0;
}
