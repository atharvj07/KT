#include <bits/stdc++.h>
using namespace std;
const int MAXN = 5005, MAXQ = 1e5 + 5;

int a[MAXN], f[MAXN];

int main() {
#ifdef ONLINE_JUDGE
  ios_base::sync_with_stdio(false);
  cin.tie(0);
#endif
  int n, q;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = n; i >= 1; i--) f[i] = a[i] ^ f[i + 1];
  cin >> q;
  while (q--) {
    int l, r;
    cin >> l >> r;
    cout << f[l] << '\n';
  }
  return 0;
}
