import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] cnt = new int[n + 1];
        int max = 0;
        List<Integer> ans = new ArrayList<>();
        for (int i = 0; i < q; i++) {
            int a = sc.nextInt();
            int v = sc.nextInt();
            cnt[a] += v;
            if (cnt[a] > max) {
                max = cnt[a];
                ans = new ArrayList<>();
                ans.add(a);
            } else if (cnt[a] == max) {
                ans.add(a);
            }
            int ansMin = Integer.MAX_VALUE;
            for (int x : ans) {
                ansMin = Math.min(ansMin, x);
            }
            System.out.println(ansMin + " " + max);
        }
    }
}
