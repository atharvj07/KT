import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        char[] s = sc.next().toCharArray();
        int[] ans = new int[q];
        int[][] pos = new int[n + 2][2];
        for (int i = 0; i < n; i++) {
            if (s[i] == '<') {
                pos[i][1] = i;
            } else {
                pos[i][0] = i;
            }
        }
        for (int i = 0; i < q; i++) {
            int l = sc.nextInt() - 1;
            int r = sc.nextInt() - 1;
            int anss = 0;
            for (int j = l; j <= r; j++) {
                int[] p = pos[j];
                if (p[0] <= l || p[1] >= r) {
                    anss++;
                }
            }
            ans[i] = anss;
            for (int j = l; j <= r; j++) {
                pos[j][0] = r - (pos[j][0] - l);
                pos[j][1] = r - (pos[j][1] - l);
            }
        }
        for (int i = 0; i < q; i++) {
            System.out.println(ans[i]);
        }
    }
}
