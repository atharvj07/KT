#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<string> a(n, string(m, '0'));
  for (int i = 0; i < n; i++) cin >> a[i];
  bool ok = true;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (((i + j) % 2) == 0) {
        int top = i - 1, bot = i + 1, left = j - 1, right = j + 1;
        int sum = 0;
        sum += ((top >= 0 && a[top][j] == '1') ? 1 : 0);
        sum += ((bot < n && a[bot][j] == '1') ? 1 : 0);
        sum += ((left >= 0 && a[i][left] == '1') ? 1 : 0);
        sum += ((right < m && a[i][right] == '1') ? 1 : 0);
        if (sum % 2 == 0) { ok = false; }
      }
    }
  }
  if (!ok) { cout << "-1" << endl; }
  else {
    cout << 0 << endl;
  }
  return 0;
}
