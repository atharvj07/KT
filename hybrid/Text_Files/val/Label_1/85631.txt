#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<long long> xs(n);
  for (auto &x : xs) cin >> x;
  sort(xs.begin(), xs.end());
  vector<long long> ps(m);
  for (auto &p : ps) cin >> p;
  for (int i = 0; i < m; i++) {
    bool ok = true;
    for (int j = 0; j < n; j++)
      if ((xs[j] - 1 + ps[i] - 1) / ps[i] * ps[i] < xs[j]) { ok = false; break; }
    if (!ok) continue;
    cout << "YES\n" << 1 + (xs[0] - 1) % ps[i] << ' ' << i + 1 << '\n';
    return 0;
  }
  cout << "NO\n";
  return 0;
}
