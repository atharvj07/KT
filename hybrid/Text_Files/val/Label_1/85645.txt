#include <bits/stdc++.h>
using namespace std;
int a[2002], dp[2002][2002];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      dp[i][j] = -1;
    }
  }
  stack<int> s;
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    if (dp[i][1] == -1) {
      s.push(i);
      for (int j = 1; j <= s.size() && s.top() > 0;) {
        int c = s.top();
        s.pop();
        dp[c][1] = 1;
        dp[c][s.size() + 1] = 1;
        s.push(c - 1);
        if (s.top() < 1) s.pop();
        ans++;
      }
      while (!s.empty()) s.pop();
      s.push(i);
    }
    if (a[i] > 0) s.push(i);
    while (a[i] == 0) {
      if (!s.empty()) {
        s.pop();
        ans--;
      } else
        break;
    }
    int sz = s.size();
    for (int j = 1; j <= s.size() && s.top() > 0; j++) {
      int c = s.top();
      s.pop();
      dp[c][j] = 1;
      dp[c][s.size() + 1] = 1;
      s.push(c - 1);
      if (s.top() < 1) s.pop();
    }
    while (!s.empty()) s.pop();
    s.push(i);
  }
  // for(int i=1;i<=n;i++) {for(int j=1;j<=n;j++) cout<<dp[i][j]<<" ";cout<<'\n';}
  vector<char> v;
  int maxm = -1;
  for (int i = 1; i <= n; i++) {
    if (a[i] > 0) {
      maxm = i;
      break;
    }
  }
  for (int i = maxm; i <= n; i++) {
    int cnt = 0;
    for (int j = 1; j <= n - i + 1; j++) { cnt += dp[i + j - 1][j]; }
    if (cnt == a[i]) {
      for (int j = maxm; j < i; j++) v.push_back('a' + j - 1);
      char ch = 'a' + i - 1;
      for (int j = 0; j < a[i]; j++) v.push_back(ch);
      maxm = i + 1;
    }
  }
  for (int i = 0; i < v.size(); i++) cout << v[i];
}
