#include <bits/stdc++.h>
using namespace std;
#define nl '\n'
#define INF 1e9

int main() {
    int n, m; cin >> n >> m;
    string s, t; cin >> s >> t;
    vector<int> dp(m, INF);
    for (int i = 0; i < n; i++) {
        if (s[i] == t[0])
            dp[0] = 0;
        if (s[i] == t[0])
            for (int j = 1; j < m; j++)
                dp[j] = min(dp[j], dp[j - 1] + 1);
        if (s[i] == t[m - 1])
            for (int j = m - 2; j >= 0; j--)
                dp[j] = min(dp[j], dp[j + 1] + 1);
    }
    int ans = 0;
    for (int d : dp)
        ans = max(ans, m - 1 - d);
    cout << ans << nl;
}
