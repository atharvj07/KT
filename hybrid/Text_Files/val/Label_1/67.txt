import sys

def prime_factors(n: int):
    """Return all prime factors of n."""
    factors = []
    while n % 2 == 0:
        factors.append(2)
        n //= 2
    i = 3
    while i * i <= n:
        while n % i == 0:
            factors.append(i)
            n //= i
        i += 2
    if n > 2:
        factors.append(n)
    return factors

def max_prime_sum(n: int):
    """Return the maximum number of primes that sum to n."""
    factors = prime_factors(n)
    num_primes = len(factors)
    return num_primes

def prime_sum_representation(n: int):
    """Return a prime sum representation of n."""
    factors = prime_factors(n)
    return factors

n = int(input())
num_primes = max_prime_sum(n)
print(num_primes)
print(*prime_sum_representation(n))
