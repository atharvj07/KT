import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), ans = 0;
        char[][] arr = new char[n][m];
        int[][] cost = new int[n][m];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.next().toCharArray();
            for (int j = 0; j < m; j++) {
                cost[i][j] = sc.nextInt();
            }
        }
        for (int j = 0; j < m; j++) {
            int[] count = new int[26];
            for (int i = 0; i < n; i++) {
                count[arr[i][j] - 'a']++;
            }
            int min = 1000000000;
            for (int i = 0; i < 26; i++) {
                if (count[i] == 1) {
                    min = Math.min(min, cost[0][j]);
                }
            }
            for (int i = 0; i < n; i++) {
                if (arr[i][j] != arr[0][j]) {
                    ans += Math.min(min, cost[i][j]);
                }
            }
        }
        System.out.println(ans);
    }
}
