import java.util.Scanner;

public class CircleArrangement {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }

        if (!isArrangementPossible(arr)) {
            System.out.println("NO");
            return;
        }

        System.out.println("YES");
        int[] result = findArrangement(arr);
        for (int num : result) {
            System.out.print(num + " ");
        }
    }

    private static boolean isArrangementPossible(int[] arr) {
        for (int i = 0; i < arr.length; i++) {
            int prev = arr[(i - 1 + arr.length) % arr.length];
            int next = arr[(i + 1) % arr.length];
            if (arr[i] >= prev + next) {
                return false;
            }
        }
        return true;
    }

    private static int[] findArrangement(int[] arr) {
        int[] result = new int[arr.length];
        int minIndex = 0;
        for (int i = 1; i < arr.length; i++) {
            if (arr[i] < arr[minIndex]) {
                minIndex = i;
            }
        }

        int j = 0;
        for (int i = minIndex; i < arr.length; i++) {
            result[j++] = arr[i];
        }
        for (int i = 0; i < minIndex; i++) {
            result[j++] = arr[i];
        }

        return result;
    }
}
