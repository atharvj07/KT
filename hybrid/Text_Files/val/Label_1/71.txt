import sys

for _ in range(int(input())):
    n = int(input())
    a = [int(x) for x in input().split()]
    a.sort()
    now = a[0]
    ans = -1
    for i in range(1, n):
        if now == a[i]:
            continue
        if ans == -1:
            ans = a[i] ^ a[i - 1]
        elif ans != a[i] ^ a[i - 1]:
            ans = -1
            break
        now = a[i]
    print(ans)
