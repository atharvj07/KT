#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto &it : a) cin >> it;
    nth_element(a.begin(), a.begin(), a.end());
    int ans = 1;
    for (auto it : a) {
      if (it > ans) break;
      ans++;
    }
    cout << ans << endl;
  }
}
