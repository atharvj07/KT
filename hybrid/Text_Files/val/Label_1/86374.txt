import java.util.Scanner;

public class TreeInvariant {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] permutation = new int[n];
        for (int i = 0; i < n; i++) {
            permutation[i] = scanner.nextInt();
        }

        boolean[][] adjacencyMatrix = new boolean[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (permutation[i] == permutation[j]) {
                    adjacencyMatrix[i][j] = adjacencyMatrix[j][i] = true;
                }
            }
        }

        if (!isTree(adjacencyMatrix)) {
            System.out.println("NO");
            return;
        }

        System.out.println("YES");
        for (int i = 0; i < n - 1; i++) {
            for (int j = i + 1; j < n; j++) {
                if (adjacencyMatrix[i][j]) {
                    System.out.println((i + 1) + " " + (j + 1));
                    break;
                }
            }
        }
    }

    private static boolean isTree(boolean[][] adjacencyMatrix) {
        int n = adjacencyMatrix.length;
        boolean[] visited = new boolean[n];
        int count = 0;

        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                count++;
                dfs(adjacencyMatrix, visited, i);
            }
        }

        return count == 1;
    }

    private static void dfs(boolean[][] adjacencyMatrix, boolean[] visited, int i) {
        visited[i] = true;
        for (int j = 0; j < adjacencyMatrix.length; j++) {
            if (adjacencyMatrix[i][j] && !visited[j]) {
                dfs(adjacencyMatrix, visited, j);
            }
        }
    }
}
