shift = input()
msg = input()
ans = ""
for c in msg:
    if shift == "R":
        if c == 'z':
            ans += 'y'
        elif c == ';':
            ans += 'l'
        elif c == ',':
            ans += 'm'
        else:
            ans += chr(ord(c) - 1)
    else:
        if c == 'a':
            ans += 'z'
        elif c == 'q':
            ans += ';'
        elif c == 'p':
            ans += ','
        else:
            ans += chr(ord(c) + 1)
print(ans)
