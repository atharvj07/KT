#include <bits/stdc++.h>
using namespace std;

const int MAXN = 5e5 + 10;

int n, m[MAXN], ans[MAXN], best[MAXN], max_ans;
bool vis[MAXN];

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);

    cin >> n;
    for (int i = 1; i <= n; i++) cin >> m[i];
    for (int i = 1; i <= n; i++) {
        if (i > 1) vis[max(1, i - m[i])] = true;
        if (i < n) vis[max(1, i + m[i])] = true;
    }
    for (int i = 1; i <= n; i++) {
        if (vis[i]) best[i] = i - 1;
        else best[i] = i;
    }
    for (int i = n; i >= 1; i--) {
        if (i > 1) max_ans = max(best[max(1, i - m[i])], max_ans);
        ans[i] = min(m[i], max_ans + 1);
    }
    for (int i = 1; i <= n; i++) cout << ans[i] << " ";
    cout << '\n';
    return 0;
}
