#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;

    vector<vector<pair<int, int>>> adj(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back(make_pair(v, w));
    }

    int dp[n + 1][100001];
    for (int i = 0; i <= n; i++) {
        for (int j = 0; j <= 100000; j++) {
            dp[i][j] = -1;
        }
    }

    dp[1][0] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= 100000; j++) {
            if (dp[i][j] != -1) {
                for (auto edge : adj[i]) {
                    int v = edge.first;
                    int w = edge.second;
                    if (w > j) {
                        dp[v][w] = max(dp[v][w], dp[i][j] + 1);
                    }
                }
            }
        }
    }

    int ans = 0;
    for (int i = 0; i <= 100000; i++) {
        ans = max(ans, dp[n][i]);
    }

    cout << ans << endl;

    return 0;
}
