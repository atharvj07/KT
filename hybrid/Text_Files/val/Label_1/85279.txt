#include <bits/stdc++.h>
using namespace std;

struct node {
    int v, d, p;
};

const int maxn = 4001;
vector<int> ans;
node a[maxn];
int n, l, r;

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i].v >> a[i].d >> a[i].p;
    }
    l = 1, r = 2;
    while (l <= n) {
        if (a[l].p < 0) {
            for (int i = r; i <= n; i++) {
                a[i].p = a[i].p - a[l].d;
            }
            l++;
            continue;
        }
        a[l].p = a[l].p - a[r].v;
        r++;
        if (r > n or a[r].p < 0) {
            ans.push_back(l);
            l = r;
            r = l + 1;
        }
    }
    cout << ans.size() << endl;
    for (int i = 0; i < ans.size(); i++) {
        cout << ans[i] << ' ';
    }
    cout << endl;
    return 0;
}
