#include <bits/stdc++.h>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        vector<vector<int>> a(n, vector<int>(m));
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                cin >> a[i][j];
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] != a[i][m - j - 1]) {
                    ans += min(a[i][j], a[i][m - j - 1]);
                    a[i][j] = a[i][m - j - 1];
                }
                if (a[i][j] != a[n - i - 1][j]) {
                    ans += min(a[i][j], a[n - i - 1][j]);
                    a[i][j] = a[n - i - 1][j];
                }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
