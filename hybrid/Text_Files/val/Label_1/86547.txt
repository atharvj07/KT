import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.LinkedList;
import java.util.Queue;

public class Main {
    static char[][] grid;
    static boolean[][] visited;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());

        while (t-- > 0) {
            String[] dimensions = br.readLine().split(" ");
            int n = Integer.parseInt(dimensions[0]);
            int m = Integer.parseInt(dimensions[1]);

            grid = new char[n][m];
            visited = new boolean[n][m];

            for (int i = 0; i < n; i++) {
                String row = br.readLine();
                for (int j = 0; j < m; j++) {
                    grid[i][j] = row.charAt(j);
                }
            }

            boolean valid = true;

            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (grid[i][j] == '.' && !validMove(i, j)) {
                        valid = false;
                        break;
                    }
                }
            }

            if (valid) {
                System.out.println("YES");
                for (int i = 0; i < n; i++) {
                    for (int j = 0; j < m; j++) {
                        System.out.print(grid[i][j]);
                    }
                    System.out.println();
                }
            } else {
                System.out.println("NO");
            }
        }
    }

    private static boolean validMove(int x, int y) {
        Queue<int[]> queue = new LinkedList<>();
        queue.add(new int[] {x, y});
        visited[x][y] = true;

        int countR = 0;
        int countW = 0;

        while (!queue.isEmpty()) {
            int[] current = queue.poll();
            int i = current[0];
            int j = current[1];

            if (grid[i][j] == 'R') {
                countR++;
            } else if (grid[i][j] == 'W') {
                countW++;
            }

            if (countR > 0 && countW > 0) {
                return false;
            }

            if (i > 0 && !visited[i - 1][j] && grid[i - 1][j] != '#') {
                queue.add(new int[] {i - 1, j});
                visited[i - 1][j] = true;
            }

            if (j > 0 && !visited[i][j - 1] && grid[i][j - 1] != '#') {
                queue.add(new int[] {i, j - 1});
                visited[i][j - 1] = true;
            }

            if (i < grid.length - 1 && !visited[i + 1][j] && grid[i + 1][j] != '#') {
                queue.add(new int[] {i + 1, j});
                visited[i + 1][j] = true;
            }

            if (j < grid[0].length - 1 && !visited[i][j + 1] && grid[i][j + 1] != '#') {
                queue.add(new int[] {i, j + 1});
                visited[i][j + 1] = true;
            }
        }

        if (countR == 0) {
            grid[x][y] = 'W';
        } else {
            grid[x][y] = 'R';
        }

        return true;
    }
}
