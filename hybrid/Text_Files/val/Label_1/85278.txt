#include <bits/stdc++.h>
using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int n, s = 0;
    cin >> n;
    vector<int> a(n);
    for (auto &x : a) cin >> x;
    sort(a.begin(), a.end());
    for (int i = 0; i < n - 1; ++i) {
      s += (a[i + 1] - a[i] <= 1);
    }
    cout << s + 1 << '\n';
  }
}
