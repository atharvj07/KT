import sys

def gcd(a, b):
    if (a == 0):
        return b
    return gcd(b % a, a)

for t in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    x = a[0]
    for i in range(1, n):
        x = gcd(x, a[i])
    if x == 1:
        print(-1)
    else:
        print(x)
