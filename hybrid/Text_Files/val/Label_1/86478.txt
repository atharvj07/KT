import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n + 1];
        int cnt = 0;
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
            cnt += a[i];
        }
        int res = 0;
        for (int i = 1; i <= n && cnt > 0; i++) {
            if (a[i] == 1) {
                res += i - 1;
                cnt--;
            }
        }
        System.out.println(res);
    }
}
