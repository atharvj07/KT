#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
    ll n, m, k, ans = LLONG_MAX;
    cin >> n >> m >> k;
    ll a[n + 3], b[m + 3];
    for (int i = 0; i < n; ++i) cin >> a[i + 1];
    for (int i = 0; i < m; ++i) cin >> b[i + 1];
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            ll cur = 1, cnt = 0, l = i, r = j;
            while (cur <= k) {
                if (a[l] != b[r]) ++cnt;
                if (cnt == k) { ans = min(ans, cur); break; }
                ++l, ++r;
                if (l > n) l = 1;
                if (r > m) r = 1;
                ++cur;
            }
        }
    }
    if (ans == LLONG_MAX) cout << "-1" << '\n';
    else cout << ans << '\n';
}
