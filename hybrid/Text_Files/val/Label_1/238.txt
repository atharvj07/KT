for _ in range(int(input())):
    n, q = map(int, input().split())
    s = input()
    for _ in range(q):
        l, r = map(int, input().split())
        t = 0
        for i in range(l - 1, r - 1):
            if s[i] != s[i + 1]:
                t += 1
            if t >= 1:
                print('YES')
                break
        else:
            print('NO')
