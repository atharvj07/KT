#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll n, t, c;
ll arr[200005];
ll pre[200005];

int main() {
    cin >> n >> t >> c;
    for (ll i = 1; i <= n; i++) {
        cin >> arr[i];
        pre[i] = pre[i - 1] + (arr[i] <= t ? 1 : 0);
    }
    ll ans = 0;
    for (ll i = 1; i <= n; i++) {
        if (i + c - 1 > n) break;
        if (pre[i + c - 1] - pre[i - 1] == c) ans++;
    }
    cout << ans << "\n";
    return 0;
}
