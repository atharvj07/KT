n, m = map(int, input().split())
coords = list(map(int, input().split()))
taxi = list(map(int, input().split()))

# Separate riders and taxi drivers
riders, drivers = [], []
for i in range(n + m):
    if taxi[i] == 1:
        drivers.append(coords[i])
    else:
        riders.append(coords[i])

# For each rider, find the closest taxi driver
driver_counts = [0] * len(drivers)
for rider in riders:
    # Binary search to find the closest taxi driver
    left, right = 0, len(drivers) - 1
    while left < right:
        mid = (left + right) // 2
        if drivers[mid] < rider:
            left = mid + 1
        else:
            right = mid
    # Check if the previous taxi driver is closer
    if left > 0 and abs(drivers[left] - rider) >= abs(drivers[left - 1] - rider):
        left -= 1
    driver_counts[left] += 1

print(' '.join(map(str, driver_counts)))
