import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] monsters = new int[n];
        for (int i = 0; i < n; i++) {
            monsters[i] = in.nextInt();
        }
        int[] stars = new int[m];
        for (int i = 0; i < m; i++) {
            stars[i] = in.nextInt();
        }
        Arrays.sort(monsters);
        Arrays.sort(stars);
        int ans = 0;
        int i = 0, j = 0;
        while (i < n && j < m) {
            if (monsters[i] <= stars[j]) {
                ans++;
                i++;
                j++;
            } else {
                j++;
            }
        }
        System.out.println(ans);
    }
}
