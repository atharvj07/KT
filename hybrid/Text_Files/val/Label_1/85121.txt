#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  long long pref[n + 1];
  pref[0] = 0;
  for (int i = 0; i < n; i++) pref[i + 1] = pref[i] + a[i];
  long long ans = 0;
  for (int l = 0; l < n; l++) {
    for (int r = l; r < n; r++) {
      ans = max(ans, pref[r + 1] - pref[l] - k * ((r - l + 1 + m - 1) / m));
    }
  }
  cout << ans << endl;
  return 0;
}
