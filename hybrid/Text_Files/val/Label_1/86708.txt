import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int y1 = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int m = scanner.nextInt();
        int y2 = scanner.nextInt();
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = scanner.nextInt();
        }

        Arrays.sort(a);
        Arrays.sort(b);

        int maxSensors = 0;
        for (int i = 0; i < n; i++) {
            int left = 0;
            int right = m - 1;
            int j = -1;
            while (left <= right) {
                int mid = (left + right) / 2;
                if (a[i] * (y2 - y1) + b[mid] * (y2 - y1) <= (a[i] - b[mid]) * (y2 - y1)) {
                    j = mid;
                    left = mid + 1;
                } else {
                    right = mid - 1;
                }
            }
            if (j != -1) {
                maxSensors = Math.max(maxSensors, j + 1);
            }
        }

        System.out.println(maxSensors);
    }
}
