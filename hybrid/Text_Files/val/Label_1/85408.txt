#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 2e5 + 5;
int w[maxn], lcp[maxn];
int32_t main() {
  ios_base::sync_with_stdio(false); cin.tie(nullptr);
  int n, m; cin >> n >> m;
  string s, t; cin >> s >> t;
  for (int i = max(m - 1, 0); i < n; i++) {
    w[i] = (s[i] != t[i % m]);
  }
  for (int i = 1, j = 0; i < n; i++) {
    while (j && w[i] != w[j]) j = lcp[j - 1];
    lcp[i] = (w[i] == w[j]) ? ++j : 0;
  }
  cout << *max_element(lcp, lcp + n) << '\n';
}
