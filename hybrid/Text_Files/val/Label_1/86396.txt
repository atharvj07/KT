import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int q = sc.nextInt();
        int[] p = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            p[i] = i;
        }
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int pa = find(p, a);
            int pb = find(p, b);
            if (pa != pb) {
                p[pa] = pb;
            }
        }
        int ans = 0;
        boolean flag = true;
        for (int i = 0; i < q; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            int pa = find(p, a);
            int pb = find(p, b);
            int pc = find(p, c);
            if (pa == pb && pb == pc) {
                ans++;
            } else {
                flag = false;
            }
        }
        if (flag) {
            ans *= Math.pow(2, n - 1 - m);
            System.out.println(ans);
        } else {
            System.out.println(0);
        }
    }

    public static int find(int[] p, int x) {
        if (p[x] == x) {
            return x;
        } else {
            return p[x] = find(p, p[x]);
        }
    }
}
