#include <bits/stdc++.h>

using namespace std;

const int MAXN = 100005;
int N, M;
vector<int> adj[MAXN];
int a[MAXN];
bool mark[MAXN];

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> N >> M;
  for (int i = 0; i < M; i++) {
    int x, y;
    cin >> x >> y;
    adj[x].push_back(y);
  }
  for (int i = 0; i < N; i++) { cin >> a[i]; }
  queue<int> q;
  q.push(1);
  while (!q.empty()) {
    int u = q.front();
    q.pop();
    mark[u] = true;
    for (auto v : adj[u]) {
      if (!mark[v]) { q.push(v); }
    }
  }
  int cnt = 0;
  for (int i = 0; i < N; i++) {
    if (mark[a[i]]) { cnt++; }
  }
  cout << cnt << "\n";
  for (int i = 0; i < N; i++) {
    if (mark[a[i]]) { cout << a[i] << "\n"; }
  }
  return 0;
}
