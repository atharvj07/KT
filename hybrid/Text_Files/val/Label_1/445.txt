from collections import defaultdict
from heapq import heappop, heappush
n, k = map(int, input().split())
a = list(map(int, input().split()))
adj = defaultdict(list)
for i in range(n - 1):
    u, v = map(int, input().split())
    adj[u].append(v)
    adj[v].append(u)
vis = [False] * (n + 1)
ans = []
def dfs(u, p):
    heappush(ans, a[u - 1])
    vis[u] = True
    for v in adj[u]:
        if not vis[v] and v != p:
            dfs(v, u)
    if len(ans) > k:
        heappop(ans)
dfs(1, 1)
print(ans[0])
