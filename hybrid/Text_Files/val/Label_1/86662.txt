import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i] * (2 * i + 1);
        }
        int min = sum;
        for (int i = 1; i < n; i++) {
            sum -= a[i - 1] * (2 * i - 1);
            sum += a[i] * (2 * i + 1);
            min = Math.min(min, sum);
        }
        System.out.println(min);
    }
}
