#include <bits/stdc++.h>
using namespace std;

int main() {
    int T;
    cin >> T;

    while (T--) {
        int n, x;
        cin >> n >> x;
        vector<int> a(n);
        for (int &i : a) cin >> i;
        int ans[2] = {0, 0};
        for (int &i : a) { ans[1] += (i + x - 1) / x; }
        sort(a.rbegin(), a.rend());
        for (int &i : a) { ans[0] += (i + x - 1) / x; }

        for (int x : ans) cout << x << ' ';
        cout << '\n';
    }
}
