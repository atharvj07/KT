import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int p = in.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            int x = in.nextInt() - 1;
            int y = in.nextInt() - 1;
            arr[i] = (1 << x) | (1 << y);
        }
        int[] sum = new int[1 << n];
        for (int i = 0; i < (1 << n); i++) {
            for (int j = 0; j < n; j++) {
                if ((i & (1 << j)) != 0) {
                    sum[i] += arr[j];
                }
            }
        }
        long ans = 0;
        for (int i = 0; i < (1 << n); i++) {
            for (int j = i + 1; j < (1 << n); j++) {
                if (Integer.bitCount(sum[i] & sum[j]) >= p) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
