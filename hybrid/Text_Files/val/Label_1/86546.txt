import java.util.*;
import java.io.*;

class Main {
    static char[][][] neighbors = {
        {{'W', '.'}, {'.', 'R'}},
        {{'.', 'W'}, {'R', '.'}}
    };

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while(t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            char[][] grid = new char[n][m];
            boolean ok = true;
            for(int i = 0; i < n; i++) {
                grid[i] = sc.next().toCharArray();
            }
            for(int i = 0; i < n; i++) {
                for(int j = 0; j < m; j++) {
                    if(grid[i][j] == '.') {
                        boolean found = false;
                        for(char[][] nb: neighbors) {
                            boolean valid = true;
                            for(int x = 0; x < 2 && valid; x++) {
                                int nx = i + x;
                                int ny = j + 1 - x;
                                if(nx >= 0 && nx < n && ny >= 0 && ny < m) {
                                    valid &= grid[nx][ny] == nb[x][0];
                                }
                            }
                            if(valid) {
                                found = true;
                                grid[i][j] = nb[0][1];
                                break;
                            }
                            valid = true;
                            for(int x = 0; x < 2 && valid; x++) {
                                int nx = i + x;
                                int ny = j + x;
                                if(nx >= 0 && nx < n && ny >= 0 && ny < m) {
                                    valid &= grid[nx][ny] == nb[x][0];
                                }
                            }
                            if(valid) {
                                found = true;
                                grid[i][j] = nb[0][1];
                                break;
                            }
                        }
                        if(!found) {
                            ok = false;
                            break;
                        }
                    }
                }
            }
            if(!ok) {
                System.out.println("NO");
            } else {
                System.out.println("YES");
                for(int i = 0; i < n; i++) {
                    for(int j = 0; j < m; j++) {
                        System.out.print(grid[i][j]);
                    }
                    System.out.println();
                }
            }
        }
    }
}
