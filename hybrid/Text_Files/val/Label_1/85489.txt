#include<bits/stdc++.h>
using namespace std;

const int maxN = 3005;
int n, dp[maxN], sub[maxN], ans, edge[maxN][maxN], deg[maxN], x[maxN];
vector<int> g[maxN];

void dfs(int u, int p = 0) {
    sub[u] = 1;
    for (auto v : g[u]) {
        if (v == p) continue;
        dfs(v, u);
        sub[u] += sub[v];
    }
}
void subEdge(int u, int p = 0) {
    dp[u] = edge[u][p];
    for (auto v : g[u]) {
        if (v == p) continue;
        dp[u] = max(dp[u], edge[v][u] + dp[v]);
    }
    ans = max(ans, dp[u]);
    for (auto v : g[u]) {
        if (v == p) continue;
        int old = dp[v];
        dp[v] = max(dp[v], edge[u][v] + dp[u]);
        if (sub[u] * 2 >= n) {
            subEdge(v, u);
        }
        else {
            if (dp[u] == edge[v][u] + old) {
                dp[v] = old;
                subEdge(v, u);
            }
            else
                subEdge(v, u);
        }
        dp[v] = old;
    }
}
int main() {
    int a, b, c;
    cin >> n;
    for (int i = 0; i < n - 1; i++) {
        cin >> a >> b >> c;
        g[a].push_back(b);
        g[b].push_back(a);
        edge[a][b] = edge[b][a] = c;
        deg[a]++; deg[b]++;
    }
    for (int i = 1; i <= n; i++) {
        if (deg[i] == 1) {
            dfs(i);
            subEdge(i);
            break;
        }
    }
    cout << ans << '\n';
    return 0;
}
