#include <bits/stdc++.h>
using namespace std;

const int inf = 1e9;
map<pair<long long, long long>, long long> dist;
int a[100005];
long long b[100005], c[100005];
int main() {
  int q;
  cin >> q;
  long long ans = 0;
  for (int i = 0; i < q; i++) {
    int type;
    cin >> type;
    if (type == 1) {
      long long x, y, z;
      cin >> x >> y >> z;
      int d = 0;
      if (x > y) {
        d = (x - y - 1) / 2;
        x = y + 2 * d;
      } else if (x < y) {
        d = (y - x - 1) / 2;
        x = y - 2 * d;
      }
      long long dw = z * d;
      if (dist.count({x, y}) == 0) {
        dist[{x, y}] = dw;
        ans += dw;
        if (x != y) dist[{y, x}] = dw;
      } else
        ans += (dw - dist[{x, y}]);
      dist[{x, y}] = dw;
      a[i] = 1;
      b[i] = x;
      c[i] = y;
    } else {
      long long x, y;
      cin >> x >> y;
      if (x > y)
        cout << (ans - dist[{y + 2 * ((x - y - 1) / 2), x}] + a[i] * (b[i] < c[i]));
      else if (x < y)
        cout << (ans - dist[{x + 2 * ((y - x - 1) / 2), x}] + a[i] * (c[i] < b[i]));
      else
        cout << ans;
      cout << "\n";
    }
  }
  return 0;
}
