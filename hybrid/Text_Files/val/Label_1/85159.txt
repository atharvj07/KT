#include <bits/stdc++.h>
#define ll long long
#define asssert(cond) {if(!(cond)){cout<<"Assertion failed on line "<<__LINE__<<", exiting"<<endl; exit(0);}}
using namespace std;

const int MAXN = 505;
ll grid[MAXN][MAXN];

ll R, C;
bool found = false;
ll sum;
set<ll> seen;

bool isValid(ll x) {
    return x >= 1 && x <= (R*C);
}

void dfs(ll x, ll y, ll curSum) {
    if(x == R) {
        x = 0, y++;
    }
    if(x == R && y == C) {
        if(curSum == sum) {
            found = true;
            return;
        } else {
            return;
        }
    }
    if(grid[x][y] > 0) {
        dfs(x+1, y, curSum + grid[x][y]);
    } else {
        for(ll i = 1; i <= R*C; i++) {
            if(!seen.count(i)) {
                seen.insert(i);
                dfs(x+1, y, curSum + i);
                seen.erase(i);
            }
        }
    }
}

int main() {
    cin >> R >> C;
    sum = (R*C * (R*C + 1)) / 2;
    for(ll i = 0; i < R; i++) {
        for(ll j = 0; j < C; j++) {
            cin >> grid[i][j];
        }
    }

    for(ll i = 0; i < R; i++) {
        ll rowSum = 0;
        for(ll j = 0; j < C; j++) rowSum += grid[i][j];
        if(i == 0) sum = rowSum;
        asssert(sum == rowSum);
    }

    for(ll i = 0; i < C; i++) {
        ll colSum = 0;
        for(ll j = 0; j < R; j++) colSum += grid[j][i];
        asssert(sum == colSum);
    }

    ll diag1Sum = 0;
    for(ll i = 0; i < R; i++) {
        diag1Sum += grid[i][i];
    }
    asssert(sum == diag1Sum);

    ll diag2Sum = 0;
    for(ll i = 0; i < R; i++) {
        diag2Sum += grid[i][R-i-1];
    }
    asssert(sum == diag2Sum);

    dfs(0, 0, 0);

    if(found) {
        for(ll i = 0; i < R; i++) {
            for(ll j = 0; j < C; j++) {
                if(grid[i][j] == 0) {
                    cout << sum - (ll)accumulate(grid[i], grid[i] + C, 0LL) << endl;
                    return 0;
                }
            }
        }
    } else {
        cout << -1 << endl;
    }

    return 0;
}
