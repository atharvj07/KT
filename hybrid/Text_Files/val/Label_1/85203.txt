#include <bits/stdc++.h>
using namespace std;
int main(){
    int n, m, x, y, mx=0;
    cin>>n>>m;
    char a[n+1][m+2];
    for(int i=1; i<=n; i++){
        for(int j=1; j<=m; j++){
            cin>>a[i][j];
            if(a[i][j]=='B'){
                x=i, y=j;
                mx=1;
            }
        }
    }
    for(int i=x-1; i<=x+1; i++){
        for(int j=y-1; j<=y+1; j++){
            if(abs(x-i)+abs(y-j)%2==1 && i>=1 && i<=n && j>=1 && j<=m && a[i][j]=='B')
                mx=0;
        }
    }
    if(mx) cout<<x<<' '<<y<<endl;
    else cout<<-1<<endl;
    return 0;
}
