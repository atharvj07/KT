import java.util.Scanner;

public class IgorKsFolders {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int a = scanner.nextInt();
        int b = scanner.nextInt();

        int rows = (int) Math.ceil((double) n / m);
        int startRow = (int) Math.ceil((double) a / m);
        int endRow = (int) Math.ceil((double) b / m);

        int selections = 0;
        if (startRow == endRow) {
            selections = 1;
        } else {
            selections = 2; // First and last row selections
            if (startRow + 1 < endRow) {
                selections++; // Selection for middle rows
            }
        }

        System.out.println(selections);
    }
}
