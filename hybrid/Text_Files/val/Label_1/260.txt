MOD = 10**9 + 7

def countWays(n, can_connect):
    dp = [[0] * n for _ in range(n)]

    for length in range(2, n+1):
        for left in range(n-length+1):
            right = left + length - 1
            if length == 2:
                dp[left][right] = 1
            else:
                for mid in range(left+1, right+1):
                    if can_connect[left][mid]:
                        dp[left][right] += dp[left+1][mid-1] * dp[mid][right]
                        dp[left][right] %= MOD

    return dp[0][n-1]

n = int(input().strip())
can_connect = [list(map(int, input().strip().split())) for _ in range(n)]

print(countWays(n, can_connect))
