#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> cards(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> cards[i];
    }
    int last_removed = -1;
    for (int i = 0; i < n; ++i) {
        if (cards[i] > last_removed) {
            last_removed = cards[i];
        }
    }
    std::cout << (last_removed == cards[n - 1] ? "Conan" : "Agasa") << "\n";
    return 0;
}
