#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
int main() {
  int s, b;
  cin >> s >> b;
  vector<long long> a(s, 0);
  for (int i = 0; i < s; i++) cin >> a[i];
  vector<pair<long long, long long> > bpv(b);
  for (int i = 0; i < b; i++) cin >> bpv[i].first >> bpv[i].second;
  for (int i = 0; i < s; i++) {
    long long max_gold = 0;
    for (int j = 0; j < b; j++)
      if (bpv[j].first <= a[i]) max_gold += bpv[j].second;
    cout << max_gold << " ";
  }
}
