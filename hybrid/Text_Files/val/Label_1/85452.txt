#include <bits/stdc++.h>
using namespace std;
#define pb emplace_back
int main(){
    int n;
    cin>>n;
    string s;
    cin>>s;
    string mn=s;
    for(int rot=0;rot<n;++rot){
        string ns=s;
        for(int i=0;i<n;++i){
           int ni=(i+rot<n)?i+rot:i+rot-n;
           ns[ni]=(s[i]=='9'?'0':(s[i]-'0'+1+'0'));
        }
        if(ns<mn) mn=ns;
        s=ns;
    }
    cout<<mn<<endl;
}
