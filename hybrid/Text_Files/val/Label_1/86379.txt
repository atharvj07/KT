import java.util.Scanner;
import java.util.ArrayList;
import java.util.List;

public class TreeInvariant {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt() - 1;
        }

        // Check if a tree exists for the given permutation
        if (!hasTree(n, p)) {
            System.out.println("NO");
            return;
        }

        // Build the tree
        List<List<Integer>> adjList = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            adjList.add(new ArrayList<>());
        }
        for (int i = 0; i < n - 1; i++) {
            adjList.get(p[i]).add(i + 1);
            adjList.get(i + 1).add(p[i]);
        }

        // Print the tree
        System.out.println("YES");
        for (int i = 0; i < n - 1; i++) {
            System.out.println((adjList.get(i).get(0) + 1) + " " + (adjList.get(i).get(1) + 1));
        }
    }

    // Check if a tree exists for the given permutation
    private static boolean hasTree(int n, int[] p) {
        // Create a union-find data structure
        UnionFind uf = new UnionFind(n);

        // Iterate over the permutation and check if edges are valid
        for (int i = 0; i < n - 1; i++) {
            if (uf.find(i) == uf.find(p[i])) {
                return false;
            }
            uf.union(i, p[i]);
        }

        return true;
    }

    // Union-find data structure
    private static class UnionFind {
        int[] parent;
        int[] rank;

        public UnionFind(int n) {
            parent = new int[n];
            rank = new int[n];
            for (int i = 0; i < n; i++) {
                parent[i] = i;
                rank[i] = 0;
            }
        }

        public int find(int x) {
            if (parent[x] != x) {
                parent[x] = find(parent[x]);
            }
            return parent[x];
        }

        public void union(int x, int y) {
            int px = find(x);
            int py = find(y);
            if (px != py) {
                if (rank[px] > rank[py]) {
                    parent[py] = px;
                } else if (rank[px] < rank[py]) {
                    parent[px] = py;
                } else {
                    parent[py] = px;
                    rank[px]++;
                }
            }
        }
    }
}
