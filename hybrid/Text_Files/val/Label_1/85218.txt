#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s, t;
  cin >> s >> t;
  int c = n * (n + 1) / 2;
  for (int i = 0; i < n; ++i) {
    if (s[i] == 'a') {
      if (t[i] == 'b')
        c += i;
    } else {
      if (t[i] == 'a')
        c += n - 1 - i;
    }
  }
  cout << min(c, k) << endl;
  return 0;
}
