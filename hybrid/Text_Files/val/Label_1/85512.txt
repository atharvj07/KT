#include <bits/stdc++.h>

using namespace std;

const int NMAX = 16;
const double EPS = 1e-4;

int M[NMAX + 1], R[NMAX + 1];
int K;
double ans;

int gcd(int x, int y) {
    while (y) {
        x %= y;
        swap(x, y);
    }
    return x;
}

int main() {
    cin >> K;
    for (int i = 0; i < K; i++) {
        cin >> M[i];
    }
    for (int i = 0; i < K; i++) {
        cin >> R[i];
    }

    int lcm = M[0];
    for (int i = 1; i < K; i++) {
        lcm = lcm / gcd(lcm, M[i]) * M[i];
    }

    for (int d = 0; d < lcm; d++) {
        bool flag = true;
        for (int i = 0; i < K; i++) {
            if (d % M[i] != R[i]) {
                flag = false;
                break;
            }
        }
        if (flag) {
            ans += 1.0 * K / lcm;
        }
    }

    cout << fixed << setprecision(6) << ans << endl;
    return 0;
}
