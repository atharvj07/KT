#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);

  int n;
  cin >> n;

  vector<int> a(n);
  for (int& ai : a) { cin >> ai; }

  vector<int> ones(20);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 20; j++) {
      if ((a[i] >> j) & 1) ones[j]++;
    }
  }

  int64_t ans = 0;
  for (int j = 0; j < 20; j++) { ans += 1LL * ones[j] * (ones[j] - 1) * (1 << j); }

  for (int i = 0; i < n; i++) {
    ans += 1LL * a[i] * a[i];
    for (int j = 0; j < 20; j++) {
      if ((a[i] >> j) & 1) { ans += 1LL * ones[j] * (1 << j); }
    }
  }

  cout << ans << endl;
  return 0;
}
