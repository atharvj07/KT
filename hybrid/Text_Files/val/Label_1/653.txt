n = int(input())
t = [int(x) for x in input().split()]
T = int(input())
ans = 0
for i in range(n):
    left = i - 1
    right = i + 1
    while left >= 0 and right < n:
        if t[i] - t[left] <= T and t[right] - t[i] <= T:
            ans = max(ans, right - left + 1)
        left -= 1
        right += 1
print(ans)
