#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  string s, t;
  cin >> n >> s >> t;
  bool ok = true;
  if (n == 1) {
    if (s[0] == 'a' || s[1] == 'a' || t[0] == 'a' || t[1] == 'a') ok = false;
  }
  if (n == 2) {
    if (s[0] == 'b' || s[1] == 'b' || t[0] == 'b' || t[1] == 'b') {
      if (s[0] == 'c' || s[1] == 'c' || t[0] == 'c' || t[1] == 'c') {
        if (s[0] == 'a' && s[1] == 'a') ok = false;
        if (t[0] == 'a' && t[1] == 'a') ok = false;
      }
    }
  }
  if (!ok) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
    for (int i = 1; i <= n; i++) {
      if (s[0] == 'a' && t[0] == 'a') cout << 'b' << 'c' << 'a';
      else if (s[0] == 'a' && t[0] == 'b') cout << 'c' << 'a' << 'b';
      else if (s[0] == 'a' && t[0] == 'c') cout << 'b' << 'a' << 'c';
      else if (s[0] == 'b' && t[0] == 'a') cout << 'a' << 'c' << 'b';
      else if (s[0] == 'b' && t[0] == 'b') cout << 'c' << 'b' << 'a';
      else if (s[0] == 'b' && t[0] == 'c') cout << 'a' << 'b' << 'c';
      else if (s[0] == 'c' && t[0] == 'a') cout << 'b' << 'c' << 'a';
      else if (s[0] == 'c' && t[0] == 'b') cout << 'c' << 'a' << 'b';
      else if (s[0] == 'c' && t[0] == 'c') cout << 'a' << 'b' << 'c';
    }
    cout << endl;
  }
  return 0;
}
