#include <bits/stdc++.h>
#define int long long
using namespace std;
int a, b;
string ans;
int scr(string s){
    int cur = 0, ans = 0;
    for(int i = 0, j = 0; i < s.size(); i = j){
        while(j < s.size() && s[j] == s[i]) j++;
        int len = j - i;
        if(s[i] == 'o') ans += len * len;
        else ans -= len * len;
    }
    return ans;
}
signed main(){
    cin >> a >> b;
    int mxv = -1e18;
    for(int i = 0; i <= a; i++){
        for(int j = 0; j <= b; j++){
            if(i > a || j > b || i + j == 0) continue;
            string s = "";
            for(int k = 1; k <= i; k++) s += 'o';
            for(int k = 1; k <= j; k++) s += 'x';
            int v = scr(s);
            if(v > mxv){
                mxv = v;
                ans = s;
            }
        }
    }
    cout << mxv << endl << ans << endl;
}
