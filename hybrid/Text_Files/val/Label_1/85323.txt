#include <iostream>
#include <string>
#include <set>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    string s;
    cin >> n >> s;
    set<char> st;
    bool ok = true;
    for (int i = 0; i < n; i++) {
      if (st.count(s[i])) ok = false;
      st.insert(s[i]);
    }
    cout << (ok ? "YES" : "NO") << endl;
  }
  return 0;
}
