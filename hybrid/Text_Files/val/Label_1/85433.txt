#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int compute_gcd(int a, int b) {
  if (b == 0)
    return a;
  return compute_gcd(b, a % b);
}

int main() {
  int n;
  cin >> n;
  vector<int> num(n);
  for (int i = 0; i < n; i++) {
    cin >> num[i];
  }
  int gcd = num[0];
  for (int i = 1; i < n; i++) {
    gcd = compute_gcd(gcd, num[i]);
  }
  int ans = 0;
  for (int i = 1; i * i <= gcd; i++) {
    if (gcd % i == 0) {
      ans += 2;
      if (i * i == gcd)
        ans--;
    }
  }
  cout << ans << '\n';
  return 0;
}
