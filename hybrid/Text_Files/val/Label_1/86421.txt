import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int n = s.length(), m = sc.nextInt();
        StringBuilder ans = new StringBuilder();
        for (int i = 0; i < m; i++) {
            int k = sc.nextInt(), d = sc.nextInt();
            for (int j = 0; j < n - k + 1; j++) {
                boolean[] flag = new boolean[d];
                StringBuilder res = new StringBuilder();
                for (int l = j; l < j + k; l++) {
                    int idx = l % d;
                    if (!flag[idx]) {
                        res.append(s.charAt(l));
                        flag[idx] = true;
                    }
                }
                ans.append(res);
            }
            s = ans.toString();
            ans.setLength(0);
            System.out.println(s);
        }
    }
}
