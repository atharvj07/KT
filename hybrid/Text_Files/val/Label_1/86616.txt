import java.util.*;
import java.math.BigInteger;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = in.nextInt();
        }
        BigInteger ans = happy(n, k, a);
        System.out.println(ans);
    }
    static BigInteger happy(int n, int k, int[] a) {
        BigInteger[] fact = new BigInteger[n + 1];
        fact[0] = fact[1] = BigInteger.ONE;
        for (int i = 2; i <= n; i++) {
            fact[i] = fact[i - 1].multiply(BigInteger.valueOf(i));
        }
        BigInteger numerator = BigInteger.ONE;
        BigInteger denominator = BigInteger.ONE;
        for (int i = 0; i < k; i++) {
            numerator = numerator.multiply(fact[n - a[i]]);
            denominator = denominator.multiply(fact[a[i]]);
        }
        return numerator.multiply(denominator).mod(BigInteger.valueOf((long)Math.pow(10, 9) + 7));
    }
}
