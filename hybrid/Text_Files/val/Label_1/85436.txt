#include <iostream>
#include <vector>

typedef long long ll;
const ll mod = 1e9 + 7;

ll pow2(ll x) { return 1LL << x; }

ll msum(std::vector<ll> &a) {
  ll ans = 0;
  for (ll x : a) { ans = (ans + x) % mod; }
  return ans;
}

int main() {
  ll n;
  std::cin >> n;
  std::vector<ll> a(n);
  for (ll &x : a) { std::cin >> x; }
  std::vector<ll> c(21, 0);
  for (ll x : a) {
    for (ll j = 0; j < 21; ++j) {
      if (pow2(j) & x) { c[j]++; }
    }
  }
  ll ans = 0;
  for (ll j = 0; j < 21; ++j) {
    ans = (ans + pow2(2 * j) % mod * c[j] % mod * (c[j] - 1) % mod) % mod;
  }
  std::cout << ans << std::endl;
  return 0;
}
