#include <bits/stdc++.h>
using namespace std;
#define MP make_pair

int main() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  int b[n];
  for (int i = 0; i < n; i++) {
    if (a[i] == n - 1)
      b[i] = n;
    else {
      b[i] = (a[i] + 1);
      a[i + b[i] % n] = n - 1;
    }
  }
  bool ok = true;
  for (int i = 0; i < n; i++) ok &= (a[i] == n - 1);
  if (!ok) {
    cout << "Impossible\n";
  } else {
    cout << "Possible\n";
    for (int i = 0; i < n; i++) cout << b[i] << " ";
    cout << '\n';
  }
}
