import sys
input = sys.stdin.readline
n,m = map(int,input().split())
k = list(map(int,input().split()))
ans = 0
sum_cost = 0
for i in range(n):
    sum_cost+= k[i]*2
d = []
t = []
for i in range(m):
    x,y = map(int,input().split())
    d.append(x)
    t.append(y)
d.sort()
t.sort()
ptr = j = 0
while ptr < n and j<m:
    if t[j] == ptr+1:
        sum_cost-= k[ptr]*(d[j]-ans)
        j+=1
    else:
       sum_cost-= k[ptr]
    ptr+=1
    ans = d[j-1]
if sum_cost<=0:
    print(ans)
else:
    res = (sum_cost+1)//2
    print(ans+res)
