from collections import defaultdict

n, m = map(int, input().split())
adj = defaultdict(list)
boss = [-1] * n
last = [0] * n
curDoc = 1

for _ in range(m):
    t, *q = map(int, input().split())
    if t == 1:
        x, y = q
        adj[x].append(y)
        boss[y] = x
    elif t == 2:
        x, = q
        last[x] = curDoc
        curDoc += 1
    else:
        x, doc = q
        y = x
        while y != -1 and last[y] < doc:
            y = boss[y]
        if y == -1:
            print('NO')
        else:
            print('YES')
