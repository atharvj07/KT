import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] bars = new int[n];
        for (int i = 0; i < n; i++) {
            bars[i] = scanner.nextInt();
        }
        Arrays.sort(bars);
        int maxTower = bars[n - 1];
        int towerCount = 0;
        for (int i = 0; i < n; i++) {
            if (bars[i] == maxTower) {
                towerCount++;
            }
        }
        System.out.println(maxTower + " " + towerCount);
    }
}
