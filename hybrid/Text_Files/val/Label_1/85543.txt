#include <bits/stdc++.h>
using namespace std;
#define int long long int
#define pb push_back
#define endl '\n'
#define mod 1000000007
#define N 1000001
int parent[N], a[N];
bool vis[N], val[N];
vector<int> g[N];
priority_queue<int> q;
int findParent(int node)
{
    if (parent[node] == node)
        return node;
    return parent[node] = findParent(parent[node]);
}
void merge(int a, int b)
{
    int p1 = findParent(a);
    int p2 = findParent(b);
    if (p1 != p2)
        parent[p2] = p1;
}
void dfs(int node)
{
    vis[node] = 1;
    int p = findParent(node);
    if (val[p] == 0)
    {
        val[p] = 1;
        q.push(p);
    }
    for (auto child : g[node])
    {
        if (!vis[child])
            dfs(child);
    }
}
int32_t main()
{
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; ++i)
        parent[i] = i;
    for (int i = 0; i < m; ++i)
    {
        int p, q;
        cin >> p >> q;
        g[p].pb(q);
        merge(p, q);
    }
    for (int i = 1; i <= n; ++i)
    {
        cin >> a[i];
        if (a[i] != i)
            g[a[i]].pb(i);
    }
    for (int i = 1; i <= n; ++i)
    {
        if (!vis[i])
            dfs(i);
    }
    if (q.size() != n)
    {
        cout << -1 << endl;
        return 0;
    }
    for (int i = 1; i <= n; ++i)
    {
        cout << q.top() << endl;
        q.pop();
    }
    return 0;
}
