
#include <bits/stdc++.h>
using namespace std;
#define FOR(i,n) for(int i=0; i<n; i++)

int n, m, a, b, w, last, ans;
map<int, vector<pair<int,int>>>adj;

void dfs(int u){
    FOR(i,(int)adj[u].size()){
        if(adj[u][i].second>last){
            last=adj[u][i].second;
            ans++;
            dfs(adj[u][i].first);
        }
    }
}

int main(){
    cin>>n>>m;
    FOR(i,m){
        cin>>a>>b>>w;
        adj[a].push_back({b,w});
    }
    last=-1; ans=0;
    dfs(1);
    cout<<ans<<endl;
}
