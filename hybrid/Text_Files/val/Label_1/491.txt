import math
for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    b = [i for i in range(1, n + 1)]
    ans = 0
    for i in range(n):
        if a[i] != b[i]:
            d1 = abs(a[i] - b[i])
            d2 = n - abs(a[i] - b[i])
            ans += math.ceil((min(d1, d2) - 1) / 2)
    print(ans)
