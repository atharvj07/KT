#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int s, b;
    cin >> s >> b;

    vector<int> a(s);
    for (int i = 0; i < s; i++) {
        cin >> a[i];
    }

    vector<pair<int, int>> d(b);
    for (int i = 0; i < b; i++) {
        cin >> d[i].first >> d[i].second;
    }

    sort(d.begin(), d.end());

    vector<int> ans(s);
    for (int i = 0; i < s; i++) {
        int cur = 0;
        for (int j = 0; j < b; j++) {
            if (d[j].first <= a[i]) {
                cur += d[j].second;
            }
        }
        ans[i] = cur;
    }

    for (int i = 0; i < s; i++) {
        cout << ans[i] << " ";
    }
    cout << endl;

    return 0;
}
