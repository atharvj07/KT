#include <cstring>
#include <algorithm>
#include <cstdio>
const int N = 1e5+5;
int n, a[N], b[N], anss, ans;
int main() {
    scanf("%d", &n);
    for(int i = 1;i <= n;i++) scanf("%d", &a[i]);
    std::sort(a+1, a+1+n);
    b[a[1]] = 0;
    for(int i = 2;i <= n;i++) {
        if(a[i] == a[i-1]) b[a[i]] = b[a[i-1]];
        else b[a[i]] = b[a[i-1]]^1;
    }
    for(int i = 1;i <= n;i++)
    if(b[i]) anss++; else ans++;
    ans = std::max(ans, anss);
    for(int i = 2;i <= n;i++) {
        if(b[a[i]] != b[a[i-1]]) {
            ans = std::max(ans, anss+1);
            ans = std::max(ans, ans+1);
            if(b[a[i]]) anss++; else ans++;
        }
    }
    printf("%d\n", ans);
    return 0;
}
