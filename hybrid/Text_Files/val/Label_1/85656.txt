#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
using ll = long long int;
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    ll a, b, c, d;
    cin >> a >> b >> c >> d;
    if (c == d) {
        cout << min(a + b, (c - 1)) << endl;
    } else {
        ll Min = min(a, b);
        ll Max = max(a, b);
        bool flag = true;
        for (ll i = 2; i * i <= c; i++) {
            if (c % i == 0) {
                if (d % i != 0) {
                    flag = false;
                }
            }
        }
        if (flag == true) {
            cout << min(a + b, c - 1) << endl;
            return 0;
        }
        flag = true;
        for (ll i = 2; i * i <= d; i++) {
            if (d % i == 0) {
                if (c % i != 0) {
                    flag = false;
                }
            }
        }
        if (flag == true) {
            cout << min(a + b, d - 1) << endl;
            return 0;
        }
        cout << min(a + b, c + d - 1) << endl;
    }
    return 0;
}
