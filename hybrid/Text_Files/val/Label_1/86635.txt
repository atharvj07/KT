import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int numDataSets = scanner.nextInt();
        scanner.nextLine();
        for (int i = 0; i < numDataSets; i++) {
            String score = scanner.nextLine();
            System.out.println(isNatural(score) ? "Yes" : "No");
        }
    }

    private static boolean isNatural(String score) {
        int lastLeft = -1;
        int lastRight = -1;
        for (char arrow : score.toCharArray()) {
            switch (arrow) {
                case 'U':
                case 'D':
                    if (lastLeft == lastRight) {
                        return false;
                    }
                    break;
                case 'L':
                    if (lastRight == lastLeft + 1) {
                        return false;
                    }
                    lastLeft = lastRight;
                    lastRight = -1;
                    break;
                case 'R':
                    if (lastLeft == lastRight - 1) {
                        return false;
                    }
                    lastRight = lastLeft;
                    lastLeft = -1;
                    break;
            }
        }
        return true;
    }
}
