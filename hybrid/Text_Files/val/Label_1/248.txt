n, m = map(int, input().split())
ar = [list(input()) for i in range(n)]
l = {}
cnt = 0
r = 0
for i in range(n):
    for j in range(m):
        if ar[i][j] == '.':
            l[(i, j)] = cnt
            cnt += 1
        else:
            cnt += 1
c = [[] for i in range(cnt)]
for i in l:
    r = l[i]
    x = i[0]
    y = i[1]
    if x + 1 < n and ar[x + 1][y] == '.':
        c[r].append(l[(x + 1, y)])
    if x - 1 >= 0 and ar[x - 1][y] == '.':
        c[r].append(l[(x - 1, y)])
    if y + 1 < m and ar[x][y + 1] == '.':
        c[r].append(l[(x, y + 1)])
    if y - 1 >= 0 and ar[x][y - 1] == '.':
        c[r].append(l[(x, y - 1)])
vis = [False for i in range(cnt)]
color = [0 for i in range(cnt)]

def dfs(i):
    if vis[i]:
        return
    vis[i] = True
    for j in c[i]:
        if color[j] == 0 or color[j] == color[i]:
            color[j] = color[i] ^ 3
        dfs(j)
dfs(0)
a = []
for i in l:
    if ar[i[0]][i[1]] == '.':
        if color[l[i]] == 1:
            a.append('W')
        else:
            a.append('B')
    else:
        a.append('-')

for i in range(0, n):
    for j in range(0, m):
        print(a[i * m + j], end='')
    print()
