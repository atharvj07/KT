import sys
input = sys.stdin.readline
for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    p = list(map(int, input().split()))
    l = [[] for _ in range(n + 1)]
    for i in p:
        l[i].append(l[i + 1])
        l[i + 1] = [i] + l[i + 1]
    ans = []
    for pt in l[1]:
        t = []
        for i in pt:
            if not t or t[-1] <= a[i]:
                t.append(a[i])
            else:
                ans = 'NO'
                break
        if ans == 'NO':
            break
    if ans != 'NO':
        ans = 'YES'
    print(ans)
