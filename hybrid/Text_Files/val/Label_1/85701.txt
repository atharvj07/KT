#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int MAXL = 1e5 + 1;
ll a[MAXL];
ll b, d[MAXL], g[MAXL];
ll pre[MAXL];

int main() {
    ll s;
    cin >> s >> b;
    for (ll i = 1; i <= s; i++)
        cin >> a[i];
    for (ll i = 1; i <= b; i++)
        cin >> d[i] >> g[i];
    sort(d + 1, d + b + 1, greater<ll>());
    sort(g + 1, g + b + 1, greater<ll>());
    for (ll i = 1; i <= b; i++) {
        pre[i] = pre[i - 1] + g[i];
    }
    for (ll i = 1; i <= s; i++) {
        ll j = upper_bound(d + 1, d + b + 1, a[i], greater<ll>()) - d - 1;
        if (j <= b)
            cout << pre[j] << '\n';
        else
            cout << 0 << '\n';
    }
    return 0;
}
