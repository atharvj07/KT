#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, a, b;
    cin >> n >> a >> b;
    vector<pair<long long, long long>> v(n);
    for (int i = 0; i < n; ++i) {
        long long x, y;
        cin >> x >> y;
        v[i] = {x, y};
    }
    sort(v.begin(), v.end(), [](const auto& l, const auto& r) { return l.second > r.second; });
    long long sum = 0;
    for (int i = 0; i < min(a, (int)v.size()); ++i) {
        sum += v[i].first + v[i].second;
    }
    sort(v.begin(), v.end());
    for (int i = 0; i < min(b, (int)v.size()); ++i) {
        sum += v[i].second;
    }
    cout << sum << endl;
    return 0;
}
