#include <bits/stdc++.h>
#define N 101
using namespace std;
int dp[N][N], a[N][N], n, m;
int func() {
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (a[i][j] == a[i][m - j + 1] && a[i][j] == a[n - i + 1][j] &&
          a[i][j] == a[n - i + 1][m - j + 1])
        dp[i][j] = dp[i - 1][j - 1] + 0;
      else
        dp[i][j] = dp[i - 1][j - 1] + 1;
    }
  }
  return dp[n][m];
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= m; j++) { cin >> a[i][j]; }
    }
    cout << func() << endl;
  }
  return 0;
}
