#include <bits/stdc++.h>
using namespace std;

int main() {
  int a[4], b[4];
  for (int i = 0; i < 4; i++) { cin >> a[i] >> b[i]; }
  if (b[0] > a[3] && a[0] > b[3])
    cout << "Team 1" << endl;
  else if (b[2] > a[1] && a[2] > b[1])
    cout << "Team 2" << endl;
  else
    cout << "Draw" << endl;
}
