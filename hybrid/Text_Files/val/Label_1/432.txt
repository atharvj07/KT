n=int(input())
s=input()
l=['R','G','B']
c=0;t=''
for i in range(n):
    if i==0:
        t+=s[i]
    elif i>0 and s[i]!=s[i-1]:
        t+=s[i]
    elif i>0 and s[i]==s[i-1]:
        for j in l:
            if j!=s[i] and (i==n-1 or j!=s[i+1]):
                t+=j
                c+=1
                break;
print(c);print(t)
