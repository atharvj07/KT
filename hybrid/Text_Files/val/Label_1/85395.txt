#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int n, q;
const int mod=1e9+7;
ll a[5000];
int main(){
    cin>>n;
    for(int i=0;i<n;i++)cin>>a[i];
    cin>>q;
    for(int i=1;i<n;i++)a[i]^=a[i-1];
    while(q--){
        int l, r;
        cin>>l>>r;l--,r--;
        cout<<(a[r]^((l?a[l-1]:0)))<<"\n";
    }
}
