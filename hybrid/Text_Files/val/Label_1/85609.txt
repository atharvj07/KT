#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main()
{
    ll n,t,c;
    cin>>n>>t>>c;
    ll p[n+1];
    for(int i=1;i<=n;i++)
        cin>>p[i];
    ll suff[n+1];
    suff[n]=0;
    for(int i=n-1;i>=1;i--)
        suff[i]=suff[i+1]+p[i];
    ll ans=0;
    for(int i=1;i<=n-c+1;i++)
    {
        if(i-1<=0||p[i-1]>t)
        {
            if(suff[i+c-1]-p[i+c-1]<=t*c)
                ans++;
        }
    }
    cout<<ans<<endl;
}
