import java.util.Scanner;

public class BallOrder {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] throwsAllowed = new int[n];
        for (int i = 0; i < n; i++) {
            throwsAllowed[i] = scanner.nextInt();
        }
        scanner.close();

        // Calculate the factorial of n.
        long factorialN = 1;
        for (int i = 2; i <= n; i++) {
            factorialN = (factorialN * i) % MOD;
        }

        // Calculate the product of factorials of throws allowed.
        long productFactorials = 1;
        for (int throwsAllowedValue : throwsAllowed) {
            productFactorials = (productFactorials * factorial(throwsAllowedValue)) % MOD;
        }

        // Calculate the result.
        long result = (factorialN * productFactorials) % MOD;

        System.out.println(result);
    }

    private static long factorial(int n) {
        if (n == 0) {
            return 1;
        } else {
            return n * factorial(n - 1);
        }
    }
}
