import java.util.Scanner;

public class FolderDeletion {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int a = scanner.nextInt();
        int b = scanner.nextInt();

        int minSelections = 0;
        int currentRow = 1;
        int currentColumn = 1;

        for (int i = a; i <= b; i++) {
            if (currentColumn > m) {
                minSelections++;
                currentRow++;
                currentColumn = 1;
            }

            if (i >= currentRow * m - m + 1 && i <= currentRow * m) {
                currentColumn++;
            }
        }

        System.out.println(minSelections);
    }
}
