import java.util.*;
import java.io.*;

class Main {
  static class Pair {
    int l, r;

    public Pair(int l, int r) {
      this.l = l;
      this.r = r;
    }
  }

  public static void main(String[] args) throws IOException {
    BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer tokenizer = new StringTokenizer(in.readLine());
    int n = Integer.parseInt(tokenizer.nextToken());
    int q = Integer.parseInt(tokenizer.nextToken());
    String s = in.readLine();

    boolean[] conveyors = new boolean[n + 1];
    for (int i = 0; i < n; i++) {
      conveyors[i + 1] = s.charAt(i) == '>';
    }

    List<Pair> queries = new ArrayList<>();
    for (int i = 0; i < q; i++) {
      tokenizer = new StringTokenizer(in.readLine());
      int l = Integer.parseInt(tokenizer.nextToken());
      int r = Integer.parseInt(tokenizer.nextToken());
      queries.add(new Pair(l, r));
    }

    int[] maxBalls = new int[q];
    int[] next = new int[n + 1];
    int[] prev = new int[n + 1];
    for (int i = 1; i <= n; i++) {
      if (conveyors[i]) {
        next[i] = i + 1;
      } else {
        prev[i] = i - 1;
      }
    }

    for (int i = q - 1; i >= 0; i--) {
      Pair query = queries.get(i);
      int l = query.l, r = query.r;

      for (int j = l; j <= r; j++) {
        conveyors[j] = !conveyors[j];
      }

      for (int j = l; j <= r; j++) {
        if (conveyors[j]) {
          next[j] = j + 1;
          prev[j] = prev[j + 1];
          if (prev[j + 1] != 0) {
            next[prev[j + 1]] = j;
          }
        } else {
          prev[j] = j - 1;
          next[j] = next[j - 1];
          if (next[j - 1] != 0) {
            prev[next[j - 1]] = j;
          }
        }
      }

      int maxBall = 0;
      for (int j = 1; j <= n; j++) {
        if (next[j] == 0 && prev[j] == 0) {
          maxBall++;
        }
      }

      maxBalls[i] = maxBall;
    }

    for (int i : maxBalls) {
      System.out.println(i);
    }
  }
}
