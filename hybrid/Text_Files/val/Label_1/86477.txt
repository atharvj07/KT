import java.util.*;
import java.math.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int t = scanner.nextInt();
        double[] x = new double[n];
        double[] v = new double[n];
        double[] m = new double[n];
        int[] next = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = scanner.nextDouble();
            v[i] = scanner.nextDouble();
            m[i] = scanner.nextDouble();
            next[i] = -1;
        }
        for (int i = 0; i < t; i++) {
            for (int j = 0; j < n; j++) {
                x[j] += v[j];
                if (x[j] < -100 || x[j] > 100) {
                    v[j] = -v[j];
                }
                if (next[j] == -1) {
                    for (int k = j + 1; k < n; k++) {
                        if (next[k] == -1 && x[j] == x[k]) {
                            double v1 = v[j] * m[j];
                            double v2 = v[k] * m[k];
                            double v3 = (v1 + v2) / (m[j] + m[k]);
                            v[j] = v3 / m[j];
                            v[k] = v3 / m[k];
                            next[j] = k;
                            next[k] = j;
                            break;
                        }
                    }
                }
            }
        }
        for (double d : x) {
            System.out.println(String.format("%.6f", d));
        }
    }
}
