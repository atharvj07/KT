#include <bits/stdc++.h>
using namespace std;

const int maxn = 1e5 + 10;
int n, m;
int p[maxn];

void solve() {
  cin >> n;
  for (int i = 1; i <= n; ++i) cin >> p[i];
  int cur = n;
  int ans = (n * (n + 1)) / 2;
  for (int i = 1; i <= n; ++i) {
    if (p[i] > i) {
      ans -= cur--;
    }
  }
  cout << ans << endl;
}

int main() {
  solve();
  return 0;
}
