import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {

    private static class City {
        private boolean assigned = false;
        private int bridgeCount;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] input = reader.readLine().split(" ");
        int cities = Integer.parseInt(input[0]);
        int bridges = Integer.parseInt(input[1]);
        City[] citiesArr = new City[cities + 1];
        Arrays.setAll(citiesArr, i -> new City());
        Map<Integer, List<Integer>> bridgesMap = new HashMap<>();

        while (bridges-- > 0) {
            input = reader.readLine().split(" ");
            int a = Integer.parseInt(input[0]);
            int b = Integer.parseInt(input[1]);
            citiesArr[a].bridgeCount++;
            citiesArr[b].bridgeCount++;
            bridgesMap.computeIfAbsent(a, k -> new ArrayList<>()).add(b);
            bridgesMap.computeIfAbsent(b, k -> new ArrayList<>()).add(a);
        }

        if (cities % 3 != 0) {
            System.out.println("NO");
            return;
        }

        List<Integer> oddCities = new ArrayList<>();
        List<Integer> doubleOddCities = new ArrayList<>();
        for (int i = 1; i <= cities; i++) {
            if (citiesArr[i].bridgeCount % 2 == 1) {
                oddCities.add(i);
            }
            if (citiesArr[i].bridgeCount == 2) {
                doubleOddCities.add(i);
            }
        }

        if (oddCities.size() % 3 != 0 || doubleOddCities.size() % 2 != 0) {
            System.out.println("NO");
            return;
        }

        StringBuilder output = new StringBuilder("YES\n");
        int deputy = 1;
        while (!oddCities.isEmpty()) {
            for (int i = 1; i <= cities; i++) {
                if (!citiesArr[i].assigned && bridgesMap.get(i).size() == 1) {
                    output.append(deputy).append(" ");
                    citiesArr[i].assigned = true;
                    citiesArr[bridgesMap.get(i).get(0)].assigned = true;
                    deputy++;
                    break;
                }
            }
            oddCities.removeIf(city -> citiesArr[city].assigned);
        }

        while (!doubleOddCities.isEmpty()) {
            for (Integer city : doubleOddCities) {
                if (!citiesArr[city].assigned) {
                    output.append(deputy).append(" ");
                    citiesArr[city].assigned = true;
                    for (Integer neighbor : bridgesMap.get(city)) {
                        if (!citiesArr[neighbor].assigned) {
                            citiesArr[neighbor].assigned = true;
                            break;
                        }
                    }
                    deputy++;
                    break;
                }
            }
            doubleOddCities.removeIf(city -> citiesArr[city].assigned);
        }

        for (int i = 1; i <= cities; i++) {
            if (!citiesArr[i].assigned) {
                output.append(deputy).append(" ");
            }
        }

        System.out.println(output);
    }
}
