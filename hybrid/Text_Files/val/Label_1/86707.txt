import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), y1 = in.nextInt(), a[] = new int[n];
        for(int i = 0; i < n; ++i) a[i] = in.nextInt();
        int m = in.nextInt(), y2 = in.nextInt(), b[] = new int[m];
        for(int i = 0; i < m; ++i) b[i] = in.nextInt();
        long ans = 0;
        for(int i = 0, j = 0; i < n && j < m; ++i) {
            while(j < m && ((long)b[j] - a[i]) * (y2 - y1) < ((long)a[i] - b[j]) * (y2 - y1)) ++j;
            if(j < m) {
                if(((long)a[i] - b[j]) * (y2 - y1) == ((long)b[j] - a[i]) * (y2 - y1)) ans += 2;
                else ans += 1;
            }
        }
        System.out.println(ans);
    }
}
