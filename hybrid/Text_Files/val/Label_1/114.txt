def can_obtain(s, t):
    i, j = len(s) - 1, len(t) - 1
    while i >= 0 and j >= 0:
        if s[i] == t[j]:
            i -= 1
            j -= 1
        else:
            i -= 2  # simulate backspace by moving back an extra step
    return j == -1  # all characters in t matched

q = int(input())
for _ in range(q):
    s = input()
    t = input()
    print("YES" if can_obtain(s, t) else "NO")
