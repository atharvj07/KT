#include<bits/stdc++.h>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int a, b, c, d, x, y, x1, y1, x2, y2;
        cin >> a >> b >> c >> d >> x >> y >> x1 >> y1 >> x2 >> y2;
        if ((x1 <= x - a && x2 >= x + b) || (y1 <= y - c && y2 >= y + d)) cout << "YES" << endl;
        else cout << "NO" << endl;
    }
    return 0;
}
