#include <bits/stdc++.h>
using namespace std;
const int M = 30000;
int32_t main() {
  int n, r;
  cin >> n >> r;
  vector<pair<int, int>> v(n);
  for (auto &e : v) cin >> e.first >> e.second;
  int dp[M + 1];
  memset(dp, -1, sizeof(dp));
  dp[r] = 0;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = M; j >= 0; j--) {
      if (dp[j] == i) {
        if (j >= v[i].first) dp[j + v[i].second] = i + 1;
        if (j - v[i].first >= 0) dp[j - v[i].first] = i + 1;
      }
    }
    if (dp[0] == n) ans = 1;
  }
  cout << (ans == 1 ? "YES" : "NO");
  return 0;
}
