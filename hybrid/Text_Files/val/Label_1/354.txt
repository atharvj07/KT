import math

a_00, a_01, a_10, a_11 = map(int, input().split())

if a_00 == 0 and a_11 == 0 and (a_01 > 0 or a_10 > 0):
    print("Impossible")
elif a_01 == 0 and a_10 == 0 and (a_00 > 0 and a_11 > 0):
    print("Impossible")
else:
    count_0 = int((-1 + math.sqrt(1 + 4 * 2 * a_00)) / 2) if a_00 > 0 else 0
    count_1 = int((-1 + math.sqrt(1 + 4 * 2 * a_11)) / 2) if a_11 > 0 else 0

    if a_00 == 0 and a_11 == 0:
        if a_01 == 1 and a_10 == 0:
            print("01")
        elif a_01 == 0 and a_10 == 1:
            print("10")
        else:
            print("Impossible")
    elif count_0 * (count_0 - 1) // 2 == a_00 and count_1 * (count_1 - 1) // 2 == a_11 and count_0 + count_1 > 0:
        if a_01 + a_10 == count_0 * count_1:
            result = ""
            while count_0 > 0 or count_1 > 0:
                if a_10 >= count_1:
                    result += "0"
                    a_10 -= count_1
                    count_0 -= 1
                else:
                    result += "1"
                    count_1 -= 1
            print(result)
        else:
            print("Impossible")
    else:
        print("Impossible")
