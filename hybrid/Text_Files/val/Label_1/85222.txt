#include <iostream>
#include <vector>
#include <queue>

using namespace std;

const int INF = 1e9;
vector< vector< pair<int,int> > > adj;
vector<int> color;

bool bfs(int n, int st, vector<bool>& visited, vector<int>& dist){
    if(color[st]==-1) return true;

    queue<int> q; q.push(st);
    visited[st] = true;
    int cur_color = color[st];
    
    while(!q.empty()){
        int u=q.front(); q.pop();
        
        for(auto e : adj[u]){
            int v=e.first, w=e.second;
            if(!visited[v]){
                q.push(v);
                visited[v]=true;
                
                if(color[v]==-1)  return true;
                if(color[v]==cur_color)  dist[v] = dist[u]+1;
                else  dist[v] = dist[u];
            }
        }

    }
    return false;
}

int main(){
    int n, m; cin >> n >> m;
    adj.resize(n+1); color.resize(n+1,-1);   // -1 indicates uncolored

    while(m--){
        int u, v; char c;
        cin >> u >> v >> c;
        
        adj[u].push_back({v, c=='R'});
        adj[v].push_back({u, c=='R'});
        
        if(c=='R')  color[u]=color[v]=1;
        else  color[u]=color[v]=0;
    }

    if(color[1]==-1){
        cout << -1 << endl;
        return 0;
    }

    vector<bool> visited(n+1, false);
    vector<int> dist(n+1, INF);

    if(!bfs(1,1,visited,dist)){
        cout << -1 << endl;
        return 0;
    }
    
    int mx_idx=-1, mx_dist=-1;
    for(int i=1; i<=n; i++){
        if(!visited[i]) continue;
        
        if(dist[i]>mx_dist){
            mx_idx=i;
            mx_dist=dist[i];
        }
    }

    int ans = mx_dist;
    vector<int> path;
    while(mx_idx!=1){
        path.push_back(mx_idx);
        mx_idx=1;
        for(auto e : adj[mx_idx]){
            int v=e.first, w=e.second;
            if(dist[v]==dist[mx_idx]-1){
                mx_idx=v;
            }
        }
    }
    
    cout << ans << endl;
    while(!path.empty()){
        cout << path.back() << " ";
        path.pop_back();
    }
    
    return 0;
}
