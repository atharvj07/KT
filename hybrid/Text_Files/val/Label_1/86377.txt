import java.util.Scanner;

public class TreeInvariant {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt() - 1;
        }

        boolean[][] connected = new boolean[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                connected[i][j] = (i != j && p[i] == p[j]);
            }
        }

        boolean possible = true;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    if (connected[i][j] && connected[j][k] && !connected[i][k]) {
                        possible = false;
                        break;
                    }
                }
                if (!possible) {
                    break;
                }
            }
            if (!possible) {
                break;
            }
        }

        if (possible) {
            System.out.println("YES");
            for (int i = 1; i < n; i++) {
                System.out.println((i + 1) + " " + (p[i] + 1));
            }
        } else {
            System.out.println("NO");
        }
    }
}
