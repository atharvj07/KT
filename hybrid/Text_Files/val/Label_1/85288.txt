#include <vector>
#include <iostream>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  vector<string> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }

  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      int sum = 0;
      for (int k = i; k < n; k++) {
        for (int l = j; l < m; l++) {
          sum += a[k][l] - '0';
          if ((k - i + 1) % 2 == 0 && (l - j + 1) % 2 == 0 && sum % 2 == 0) {
            ans++;
          }
        }
      }
    }
  }

  cout << ans << endl;
}
