#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int i = 0; i < n; i++) {
    cin >> p[i];
  }
  int m = 0;
  for (int i = 0; i < n; i++) {
    int j = (i + n / 2) % n;
    if (p[i] > p[j]) {
      m++;
      swap(p[i], p[j]);
    }
  }
  cout << m << "\n";
  for (int i = 0; i < n; i++) {
    int j = (i + n / 2) % n;
    if (p[i] > p[j]) {
      cout << i + 1 << " " << j + 1 << "\n";
      swap(p[i], p[j]);
    }
  }
  return 0;
}
