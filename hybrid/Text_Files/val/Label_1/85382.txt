#include <algorithm>
#include <iostream>
#include <map>
#include <vector>
using namespace std;

const int MAXN = 150000;
vector<pair<int, int>> ans;
char l[MAXN + 2], r[MAXN + 2];
int n;

int main() {
  cin >> n;
  map<char, vector<int>> mp;
  for (int i = 1; i <= n; ++i) {
    cin >> l[i];
    mp[l[i]].push_back(i);
  }
  for (int i = 1; i <= n; ++i) {
    cin >> r[i];
    if (l[i] != r[i] && r[i] != '?') {
      if (mp[r[i]].empty()) {
        cout << 0 << endl;
        return 0;
      }
      mp[r[i]].pop_back();
      ans.push_back({i, i});
    } else if (l[i] != '?' || r[i] != '?') {
      if (!mp[l[i]].empty() || !mp[r[i]].empty()) {
        int x = mp[l[i]].empty() ? mp[r[i]].back() : mp[l[i]].back();
        ans.push_back({i, x});
        mp[l[i]].pop_back(), mp[r[i]].pop_back();
      }
    } else {
      if (!mp['?'].empty()) {
        int x = mp['?'].back();
        ans.push_back({i, x});
        mp['?'].pop_back();
      }
    }
  }
  for (auto p : mp) {
    for (int x : p.second) {
      ans.push_back({x, x});
    }
  }
  cout << ans.size() << endl;
  for (auto p : ans) {
    cout << p.first << " " << p.second << endl;
  }
  return 0;
}
