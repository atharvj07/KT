#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int i = 0; i < n; i++) cin >> p[i];
  int ans = n * (n - 1) / 2 - 1, cnt = 0;
  for (int i = 1; i < n; i++) {
    if (p[i] < p[i - 1]) ans -= i;
  }
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (p[i] > p[j]) cnt++;
    }
  }
  cout << ans << ' ' << cnt << '\n';
}
