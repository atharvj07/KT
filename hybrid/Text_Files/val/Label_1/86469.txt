import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        if (k == 1) {
            for (int i = 1; i <= n; i++) {
                System.out.print(i + " ");
            }
        } else if (k == n - 1) {
            for (int i = 1; i <= n; i++) {
                System.out.print((i + 1) % n + 1 + " ");
            }
        } else {
            int[] perm = new int[n];
            perm[0] = 1;
            int d = n / (k - 1);
            int r = n % (k - 1);
            for (int i = 1; i <= k - 1; i++) {
                for (int j = 1; j <= d; j++) {
                    perm[i * d - j] = (i + j) % n + 1;
                }
                if (r > 0) {
                    perm[i * d] = (i + d) % n + 1;
                    r--;
                }
            }
            for (int i = 0; i < n; i++) {
                System.out.print(perm[i] + " ");
            }
        }
    }
}
