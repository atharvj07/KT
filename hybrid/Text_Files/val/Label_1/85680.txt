#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ll t, n;
    cin >> t;
    while (t--) {
        cin >> n;
        ll ans = (n + 1949) / 2050;
        if (ans * 2050 == n) {
            cout << ans << '\n';
        } else {
            cout << -1 << '\n';
        }
    }
}
