import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        int[] users = new int[n + 1];
        int leaderCount = 0;
        Set<Integer> leaders = new HashSet<>();
        for (int i = 0; i < m; i++) {
            String message = input.next();
            int user = input.nextInt();
            if (message.equals("+")) {
                users[user]++;
                if (users[user] == 1) {
                    if (leaderCount == 0) {
                        for (int j = 1; j <= n; j++) {
                            if (users[j] > 0) {
                                leaders.add(j);
                            }
                        }
                    }
                    leaderCount++;
                }
            } else {
                users[user]--;
                if (users[user] == 0) {
                    leaderCount--;
                    if (leaderCount == 0) {
                        leaders.clear();
                    }
                }
            }
        }
        System.out.println(leaders.size());
        for (int leader : leaders) {
            System.out.print(leader + " ");
        }
        System.out.println();
    }
}
