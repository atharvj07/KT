#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int INF = 1e9;
const int MAXN = 3000;

vector<int> adj[MAXN + 1];
vector<int> w[MAXN + 1];
vector<int> x;
int d[MAXN + 1][MAXN + 1];

int dfs(int node, int par, int dist) {
    d[node][node] = 0;
    for (int i = 0; i < adj[node].size(); i++) {
        int u = adj[node][i];
        if (u == par) continue;
        d[node][u] = d[u][node] = max(dist, w[node][i]);
        dfs(u, node, max(dist, w[node][i]));
    }
}

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n - 1; i++) {
        int u, v, wt;
        cin >> u >> v >> wt;
        adj[u].push_back(v);
        adj[v].push_back(u);
        w[u].push_back(wt);
        w[v].push_back(wt);
    }
    for (int i = 0; i < n; i++) {
        int xi;
        cin >> xi;
        x.push_back(xi);
    }
    dfs(1, 1, 0);
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        int best = INF;
        for (int j = 1; j <= n; j++) {
            if (x[j - 1] == 0) continue;
            best = min(best, d[i][j]);
            x[j - 1]--;
        }
        ans = max(ans, best);
    }
    cout << ans << endl;
    return 0;
}
