n = int(input())
mod = 998244353

# Since each permutation of numbers 1 to n in lexicographic order contributes exactly one subarray of length n with the sum of n*(n+1)/2,
# and since there are (n-1)! permutations of the remaining n-1 numbers, the answer is n!.
# However, this counts each permutation separately, so we have to adjust for the fact that the sequence wraps around at the end,
# contributing additional valid sequences.

# To calculate factorial of n modulo mod
def factorial(n, mod):
    f = 1
    for i in range(1, n + 1):
        f = (f * i) % mod
    return f

# Main calculation
ans = factorial(n, mod) - (n - 1)
ans %= mod

print(ans)
