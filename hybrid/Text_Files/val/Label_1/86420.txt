import java.util.*;
import java.io.*;

class Main {
    static void swap(char[] arr, int i, int j) {
        char temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }

    static char[] dSort(int[] charCount, char[] arr, int start, int d) {
        int[] count = new int[d];
        for (int i = start; i < start + d; ++i)
            if (i < charCount.length)
                ++count[i % d];

        count[d - 1]--;
        for (int i = d - 2; i >= 0; --i)
            count[i] = count[i + 1] - 1;

        int j = 0;
        char[] newArr = new char[d];
        for (int i = 0; i < d; ++i) {
            while (count[i] >= 0) {
                newArr[count[i]--] = arr[j++];
            }
        }
        for (int i = start; i < start + d; ++i) {
            arr[i] = newArr[i - start];
        }
        return arr;
    }

    static void shuffle(char[] arr, int k, int d) {
        int n = arr.length;
        int[] charCount = new int[n];
        for (int i = 0; i < n; ++i) {
            charCount[i] = i % d;
        }

        for (int i = 0; i <= n - k; ++i) {
            arr = dSort(charCount, arr, i, d);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        char[] arr = br.readLine().toCharArray();
        int m = Integer.parseInt(br.readLine());
        while (m-- > 0) {
            String[] input = br.readLine().split(" ");
            int k = Integer.parseInt(input[0]);
            int d = Integer.parseInt(input[1]);
            shuffle(arr, k, d);
            System.out.println(new String(arr));
        }
    }
}
