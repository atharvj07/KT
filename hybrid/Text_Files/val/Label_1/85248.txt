#include <bits/stdc++.h>

using namespace std;
const int MOD = 1e9 + 7;

int main() {
    int n;
    cin >> n;
    long long sum = 0, ans = n;
    for (int i = 0; i < n; i++) {
        int h;
        cin >> h;
        ans = ans * i % MOD - sum % MOD;
        sum = (sum + h) % MOD;
        ans = (ans + MOD) % MOD;
    }
    cout << ans << endl;
    return 0;
}
