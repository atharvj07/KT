#include<bits/stdc++.h>
using namespace std;
const int mod=1e9+7;
int main()
{
    int n; cin>>n;
    vector<int>v(n+1,0);
    for(int i=1;i<=n;i++) cin>>v[i];
    int pre[n+1],suf[n+1];
    pre[0]=0;
    for(int i=1,j=0;i<=n;i++)
    {
       if(v[i]==v[i-1]) pre[i]=pre[i-1];
       else pre[i]=i;
    }
    suf[n+1]=0;
    for(int i=n,j=0;i>0;i--)
   {
       if(v[i]==v[i+1]) suf[i]=suf[i+1];
       else suf[i]=i;
   }

    int ans=0;
    for(int i=1;i<=n;i++)
    {
        ans=(ans+suf[i]-pre[i]-1+mod)%mod;
    }
    cout<<ans%mod<<endl;
}
