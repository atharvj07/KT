#include <bits/stdc++.h>

using namespace std;

int n, a[3005];
int cnt[3005];

int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 1; i <= n; i++)
      cnt[a[i]] = 0;
    int ans = 0;
    for (int i = 1; i <= n; i++) {
      cin >> a[i];
      for (int j = 1; j < i; j++) {
        if (a[i] == a[j] && j != i - 1)
          ans += cnt[a[j]];
        cnt[a[i]]++;
      }
    }
    cout << ans << "\n";
  }
}
