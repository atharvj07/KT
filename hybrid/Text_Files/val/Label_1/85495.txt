#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll maxn = 2e5 + 5;
ll n, m, k, p1, p2, sz, ans, a[maxn];
int main() {
  cin >> n >> m >> k;
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }
  for (int i = 0; i < n; ++i) {
    if (sz + a[i] <= k)
      sz += a[i], ++ans;
    else {
      ++p1;
      sz = a[i];
      --i;
    }
    if (p1 > m) break;
  }
  ans -= p1;
  while (p2 < n and k >= a[p2]) k -= a[p2++], ++ans;
  cout << ans << endl;
}
