import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextLong();
        }
        long ans = 0;
        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += Math.max(0, b[i] - a[i]);
            ans += Math.abs(sum - b[i]);
        }
        System.out.println(ans);
    }
}
