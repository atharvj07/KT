n, m = map(int, input().split())
a = list(map(int, input().split()))
q = [[int(i) for i in input().split()] for _ in range(m)]
ans = [-1]*m
b = {}
for i in range(n):
    if a[i] in b: b[a[i]].append(i)
    else: b[a[i]] = [i]
for i in range(m):
    if not a[q[i][0]-1] in b or not a[q[i][1]-1] in b: continue
    c = b[a[q[i][0]-1]]
    if c[0] > q[i][1]-1 or c[len(c)-1] < q[i][0]-1: continue
    d = b[a[q[i][1]-1]]
    if d[0] > q[i][1]-1 or d[len(d)-1] < q[i][0]-1: continue
    ans[i] = min(ans[i], abs(c[0]-d[0]), abs(c[0]-d[len(d)-1]), abs(c[len(c)-1]-d[0]), abs(c[len(c)-1]-d[len(d)-1]))
print(*ans)
