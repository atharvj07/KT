n = int(input())
s = input()

min_s = s
for i in range(n):
    # Add 1 to all digits
    s = ''.join(str((int(c) + 1) % 10) for c in s)
    
    # Try all rotations
    for j in range(n):
        rotated = s[j:] + s[:j]
        if rotated.lstrip('0') < min_s.lstrip('0'):
            min_s = rotated

print(min_s.lstrip('0') or '0')
