#include<bits/stdc++.h>
using namespace std;
int main()
{
	int n,m,k;
	cin>>n>>m>>k;
	int a[n+1];
	for(int i=1;i<=n;i++)
	{
		cin>>a[i];
	}
	int l,r;
	int b[k+1];
	for(int j=1;j<=m;j++)
	{
		cin>>l>>r;
		b[a[l]]++;
		b[a[r]]++;
	}
	int sum=0;
	for(int j=1;j<=k;j++)
	{
		if(b[j]%2==1)
		sum++;
	}
	cout<<sum/2;
	return 0;
}
