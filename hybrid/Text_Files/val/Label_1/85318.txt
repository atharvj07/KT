#include<bits/stdc++.h>

using namespace std;

typedef long long ll;

int main() {
    ll n, m, k;
    cin >> n >> m >> k;
    map<ll, ll> mp1, mp2;
    for(ll i=0; i<n; i++) {
        ll a;
        cin >> a;
        mp1[a]++;
    }

    for(ll i=0; i<m; i++) {
        ll b;
        cin >> b;
        mp2[b]++;
    }

    bool ok;

    for(auto x: mp1) {
        ll cur = x.first;
        ll c1 = x.second;
        ok = true;
        for(auto q: mp2) {
            ll cur2 = q.first;
            ll c2 = q.second;
            if(cur == cur2 && c1 == c2) ok = false;
        }

        if(ok) {
            for(ll i=1;; i++) {
                ll cur2 = (i-1)%m + 1;
                if(cur2 == cur) continue;
                k--;
                if(k == 0) {
                    cout << i << endl;
                    return 0;
                }
            }
        }
    }

    for(auto x: mp2) {
        ll cur = x.first;
        ll c1 = x.second;
        ok = true;
        for(auto q: mp1) {
            ll cur2 = q.first;
            ll c2 = q.second;
            if(cur == cur2 && c1 == c2) ok = false;
        }

        if(ok) {
            for(ll i=1;; i++) {
                ll cur2 = (i-1)%n + 1;
                if(cur2 == cur) continue;
                k--;
                if(k == 0) {
                    cout << i << endl;
                    return 0;
                }
            }
        }
    }

    return 0;
}
