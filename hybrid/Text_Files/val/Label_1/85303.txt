#include <bits/stdc++.h>

using namespace std;

const int N = 1e5 + 5;

int sz[N], sub[N], par[N];
vector<int> g[N];

void dfs1(int p, int v = 1) {
  sz[v] = 1;
  par[v] = p;
  for (int u : g[v]) {
    if (u != p) {
      dfs1(v, u);
      sz[v] += sz[u];
    }
  }
}

int dfs2(int p, int v = 1) {
  for (int u : g[v]) {
    if (u != p) {
      if (sz[u] > sz[v] / 2) {
        return dfs2(v, u);
      }
    }
  }
  return v;
}

int find_centroid(int root) {
  dfs1(0, root);
  int c = dfs2(0, root);
  return c;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    for (int i = 1; i < n; i++) {
      int u, v;
      cin >> u >> v;
      g[u].push_back(v);
      g[v].push_back(u);
    }
    int c = find_centroid(1);
    for (int u : g[c]) {
      cout << c << ' ' << u << '\n';
      cout << c << ' ' << u << '\n';
      break;
    }
    for (int i = 1; i <= n; i++) {
      g[i].clear();
      par[i] = 0;
      sz[i] = 0;
      sub[i] = 0;
    }
  }
  return 0;
}
