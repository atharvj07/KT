#include <bits/stdc++.h>
using namespace std;
#define mod 1000000007

long long fastpow(long long b, long long p) {
    if (!p) return 1;
    if (p == 1) return b % mod;
    if (p % 2) return (fastpow(b, p - 1) % mod * b % mod) % mod;
    long long z = fastpow(b, p / 2);
    return (z * z) % mod;
}

int main() {
    int n, k, m;
    cin >> n >> k >> m;
    if (k > 1)
        cout << (fastpow(2, n) - 1 + fastpow(k, n)) % m << endl;
    else
        cout << fastpow(2, n) % m << endl;
}
