n,m=map(int,input().split())
a=list(set(map(int,input().split())))
a.sort()
l=len(a)
mn=m
k=0
c=0
for i in a:
    if i<=mn:
        mn-=i
        k+=1
    else:
        break
l1=[]
mn=m
for j in reversed(a):
    if j<=mn:
        mn-=j
        l1.append(j)
        c+=1
    else:
        break

if c<k:
    k=c
    l1.reverse()
print(k)
print(*l1)
