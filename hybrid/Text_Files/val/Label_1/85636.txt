#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

const int MAXN = 100001;

int main() {
  int n;
  cin >> n;
  vector<int> p(n + 1);

  for (int i = 1; i <= n; i++) {
    cin >> p[i];
  }

  vector<int> seen(n + 1, -1);

  int ans = 0;
  for (int i = 1; i <= n; i++) {
    if (seen[i] != -1) {
      ans += seen[i];
      continue;
    }

    int len = 0;
    int cur = i;
    while (seen[cur] == -1) {
      len++;
      seen[cur] = len;
      cur = p[cur];
    }
    ans += seen[cur] + len - 1;
  }

  cout << ans << endl;

  return 0;
}
