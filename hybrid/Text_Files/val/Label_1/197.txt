import heapq
n,m=map(int,input().split())
a=list(map(int,input().split()))
b=[]
for i in range(m):
    x,y=map(int,input().split())
    b.append((x,y))
b.sort()
ans=0
need=sum(a)
money=0
h=[]
for j in b:
    x,y=j
    if money>=need:
        ans=x
        break
    money+=1
    heapq.heappush(h,(a[y-1],y))
    a[y-1]-=1
    if a[y-1]==0:
        a.pop(y-1)
        need-=1
while money<need:
    money+=1
    x=heapq.heappop(h)
    y=x[1]
    if x[0]==0:
        continue
    a.insert(y-1,x[0]-1)
    need-=1
    heapq.heappush(h,(a[y-1],y))
ans+=money-1
print(ans)
