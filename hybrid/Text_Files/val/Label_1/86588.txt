import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int t = scan.nextInt();
        while (t-- > 0) {
            int a = scan.nextInt();
            if (180 % a == 0)
                System.out.println(180 / a);
            else if (360 % a == 0)
                System.out.println(360 / a);
            else
                System.out.println(-1);
        }
    }
}
