#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    int ans = 0;
    int mn = 1e9;
    for (int i = 0; i < n; i++) {
        mn = min(mn, a[i]);
        ans += a[i] == mn;
    }

    cout << ans << '\n';

    return 0;
}
