for _ in range(int(input())):
    n = int(input())
    s, t = input(), input()
    if sorted(s) != sorted(t):
        print('No')
    else:
        ans = []
        for i in range(n):
            if s[i] != t[i]:
                for j in range(i + 1, n):
                    if s[j] == t[i] and t[j] == s[i]:
                        ans.append([i + 1, j + 1])
                        s, t = s[:i] + s[j] + s[i + 1:j] + s[i], t[:i] + t[j] + t[i + 1:j] + t[i]
                        break
                else:
                    for j in range(i + 1, n):
                        if s[j] == t[i]:
                            ans.extend([[i + 1, j + 1], [j + 1, i + 1]])
                            s, t = s[:i] + s[j] + s[i + 1:j] + s[i], t[:i] + t[j] + t[i + 1:j] + t[i]
                            break
        print('Yes')
        print(len(ans))
        for i, j in ans:
            print(i, j)
