import java.util.Scanner;

public class FolderDeletion {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int a = scanner.nextInt();
        int b = scanner.nextInt();

        int selections = 0;
        int start = a;
        int end = b;

        while (start <= end) {
            // Calculate the number of folders that can fit in a row
            int foldersInRow = Math.min(m, end - start + 1);

            // Calculate the number of full rows needed to select the remaining folders
            int fullRows = (foldersInRow + m - 1) / m;

            // Update the start and end indices to reflect the selected folders
            start += foldersInRow;
            end -= (foldersInRow - 1);

            // Increment the number of selections
            selections += fullRows;
        }

        System.out.println(selections);
    }
}
