#include<bits/stdc++.h>
#define vi vector<int>
using namespace std;

int main(){
  int n,a;
  cin>>n;
  vi m(n),f(n);
  for(a=0;a<n;a++)cin>>m[a];
  f[0]=m[0];
  for(a=1;a<n;a++){
    f[a]=min(m[a],f[a-1]+1);
  }
  for(a=n-2;a>=0;a--){
    f[a]=min(f[a],f[a+1]+1);
  }
  for(a=0;a<n;a++)cout<<f[a]<<" ";
  cout<<endl;
}
