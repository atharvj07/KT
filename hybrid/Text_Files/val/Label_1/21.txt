class TreeNode:
    def __init__(self, value=0, left=None, right=None):
        self.value = value
        self.left = left
        self.right = right

def build_tree(nodes):
    if not nodes:
        return None
    tree_nodes = {i: TreeNode(value=v) for i, (v, _, _) in enumerate(nodes, start=1)}
    for i, (_, l, r) in enumerate(nodes, start=1):
        if l != -1:
            tree_nodes[i].left = tree_nodes[l]
        if r != -1:
            tree_nodes[i].right = tree_nodes[r]
    return tree_nodes[1]

def find(t, x):
    if t == None:
        return False
    if t.value == x:
        return True
    if x < t.value:
        return find(t.left, x)
    else:
        return find(t.right, x)

def calculate_failures(root, values):
    count = 0
    for val in values:
        if not find(root, val):
            count += 1
    return count

n = int(input().strip())
nodes = []
values = set()
for _ in range(n):
    v, l, r = map(int, input().strip().split())
    nodes.append((v, l, r))
    values.add(v)

root = build_tree(nodes)
print(calculate_failures(root, values))
