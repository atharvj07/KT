#include <iostream>
 
using namespace std;
 
int main() {
  int a[4], b[4];
  for (int i = 0; i < 4; ++i)
    cin >> a[i] >> b[i];
  if (max(a[0], a[1]) > max(b[2], b[3]) && max(b[0], b[1]) > max(a[2], a[3]))
    cout << "Team 1" << endl;
  else if (max(a[2], a[3]) > max(b[0], b[1]) && max(b[2], b[3]) > max(a[0], a[1]))
    cout << "Team 2" << endl;
  else
    cout << "Draw" << endl;
  return 0;
}
