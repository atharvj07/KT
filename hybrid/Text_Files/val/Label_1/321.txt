import sys

input = sys.stdin.readline

n = int(input())
a = [0] * n
d = {}
for i in range(1,n):
    x,y=map(int,input().split())
    if x not in d: d[x]=[y]
    else: d[x].append(y)
    if y not in d: d[y]=[x]
    else: d[y].append(x)
l=[0]
def dfs(x,p):
    a[x-1]=1
    for y in d[x]:
        if y==p: continue
        else:
            dfs(y,x)
            a[x-1]^=a[y-1]
    l.append(a[x-1])

dfs(1,-1)
c=0
for i in range(n):
    if a[i]==0: c+=1
print(1,c+1)
