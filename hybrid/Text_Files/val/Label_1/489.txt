import sys
input = sys.stdin.readline
from collections import defaultdict

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    b = sorted(a)
    exist = defaultdict(int)
    for i in b: exist[i] = 1
    ans = 0
    for i in range(n):
        if a[i] != b[i]:
            j = i
            while j < n and a[j] != b[i]: j += 1
            k = j
            while k >= 0 and a[k] != b[j]: k -= 1
            if k < 0: ans += 1
            else: ans += 2
    print(ans)
