#include <cstdio>
#include <cmath>

int main() {
  long long int a, b;
  scanf("%lld %lld", &a, &b);
  long long int tmp;
  while (a != b) {
    if (a < b) {
      tmp = a;
      a = b;
      b = tmp;
    }
    if (a % b == 0) {
      a = a / b;
    } else {
      a = a - b;
    }
  }
  printf("%lld", (b < 19227314019894611328LL) ? b : -1);
  return 0;
}
