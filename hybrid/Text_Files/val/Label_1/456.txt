from collections import defaultdict
n, m = map(int, input().split())
d = defaultdict()
for i in range (m):
    a, b = map(int, input().split())
    if a not in d: d[a] = set()
    if b not in d: d[b] = set()
    d[a].add(b); d[b].add(a)

def dfs(x, y):
    v[x] = y
    for u in d[x]:
        if v[u] == 0:
            dfs(u, y)

c, v = 1, [0] * 101
for i in d:
    if v[i] == 0: dfs(i, c); c += 1

print(c - 1)
for i in range (1, n + 1):
    print(len(d[i]))
    for x in d[i]:
        print(' '.join(map(str, [x, i])))
