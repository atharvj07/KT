import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int d1 = scanner.nextInt();
        int d2 = scanner.nextInt();
        int d3 = scanner.nextInt();
        // Output the minimum distance that Patrick will have to walk in order to visit both shops and return to his house.
        System.out.println(Math.min(d1 + d2 + d3, d1 + d3 + d2));
    }
}
