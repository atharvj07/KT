import java.util.*;

public class NetworkSimulation {

    static class Router {
        int id;
        List<Integer> connections;

        public Router(int id, List<Integer> connections) {
            this.id = id;
            this.connections = connections;
        }
    }

    static class Packet {
        int source;
        int destination;
        int ttl;

        public Packet(int source, int destination, int ttl) {
            this.source = source;
            this.destination = destination;
            this.ttl = ttl;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        List<Router> routers = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int id = scanner.nextInt();
            int k = scanner.nextInt();
            List<Integer> connections = new ArrayList<>();
            for (int j = 0; j < k; j++) {
                connections.add(scanner.nextInt());
            }
            routers.add(new Router(id, connections));
        }

        int p = scanner.nextInt();
        List<Packet> packets = new ArrayList<>();
        for (int i = 0; i < p; i++) {
            packets.add(new Packet(scanner.nextInt(), scanner.nextInt(), scanner.nextInt()));
        }

        for (Packet packet : packets) {
            int minRouters = findMinRouters(routers, packet);
            System.out.println(minRouters == Integer.MAX_VALUE ? "NA" : minRouters);
        }
    }

    static int findMinRouters(List<Router> routers, Packet packet) {
        int[] distances = new int[routers.size()];
        Arrays.fill(distances, Integer.MAX_VALUE);
        distances[packet.source - 1] = 0;

        Queue<Integer> queue = new LinkedList<>();
        queue.offer(packet.source - 1);

        while (!queue.isEmpty()) {
            int currentRouter = queue.poll();
            int currentDistance = distances[currentRouter];

            if (currentDistance >= packet.ttl) {
                continue;
            }

            for (int nextRouter : routers.get(currentRouter).connections) {
                int nextDistance = currentDistance + 1;
                if (nextDistance < distances[nextRouter - 1]) {
                    distances[nextRouter - 1] = nextDistance;
                    queue.offer(nextRouter - 1);
                }
            }
        }

        int minRouters = distances[packet.destination - 1];
        return minRouters == Integer.MAX_VALUE ? Integer.MAX_VALUE : minRouters;
    }
}
