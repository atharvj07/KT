#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int cnt = 0;
    vector<int> v;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'B') {
            cnt++;
        } else {
            if (cnt) {
                v.push_back(cnt);
                cnt = 0;
            }
        }
    }
    if (cnt) {
        v.push_back(cnt);
    }
    cout << v.size() << "\n";
    for (int i : v) {
        cout << i << " ";
    }
    cout << "\n";
}
