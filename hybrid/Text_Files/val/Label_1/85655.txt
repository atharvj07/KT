#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
bool mark[30005];
int main(){
    ll a,b,c,d,ans,i,j,v;
    cin>>a>>b>>c>>d;
    v=max(c,d);
    for(i=2;i<=v;++i)mark[i]=1;
    for(i=2;i*i<=v;++i){
        if(mark[i]==1){
            for(j=i*i;j<=v;j+=i)mark[j]=0;
        }
    }
    for(j=2;j<=v;++j){
        if(mark[j]==1){
            if(j%c==0)a--;
            if(j%d==0)b--;
            if(a<=0 && b<=0)break;
        }
    }
    cout<<j<<endl;
}
