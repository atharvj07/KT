#include<bits/stdc++.h>
using namespace std;
int n, r;
struct Project{
    int cost, change;
    Project(int _cost, int _change):cost(_cost),change(_change){}
};
vector<Project> projects;
bool isPossible(){
    sort(projects.begin(), projects.end(), [](Project&a, Project&b){return a.cost>b.cost;});
    for(auto& p:projects){
        if(r<p.cost) return false;
        r+=p.change;
        if(r<0) return false;
    }
    return true;
}
int main(){
    cin>>n>>r;
    int cost, change;
    for(int i=0;i<n;++i){
        cin>>cost>>change;
        projects.push_back(Project(cost, change));
    }
    isPossible()?cout<<"YES":cout<<"NO";
    return 0;
}
