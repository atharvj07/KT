import java.util.*;

public class TreeInvariant {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt() - 1;
        }

        // Check if a valid tree exists.
        if (!isValidTree(n, p)) {
            System.out.println("NO");
            return;
        }

        // Build the tree.
        List<int[]> edges = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (p[i] == p[j]) {
                    edges.add(new int[]{i + 1, j + 1});
                }
            }
        }

        // Print the tree.
        System.out.println("YES");
        for (int[] edge : edges) {
            System.out.println(edge[0] + " " + edge[1]);
        }
    }

    // Checks if a valid tree exists for the given permutation.
    private static boolean isValidTree(int n, int[] p) {
        // A permutation is valid if and only if it contains a cycle of length n.
        boolean[] visited = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                if (!hasCycle(i, p, visited)) {
                    return false;
                }
            }
        }
        return true;
    }

    // Checks if there is a cycle starting from the given vertex.
    private static boolean hasCycle(int vertex, int[] p, boolean[] visited) {
        visited[vertex] = true;
        int nextVertex = p[vertex];
        if (visited[nextVertex]) {
            return true;
        }
        return hasCycle(nextVertex, p, visited);
    }
}
