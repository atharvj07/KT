#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int ans = 0, l = 0, r = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] != 'F') {
            r++;
        }
        if (s[i] == 'F' || i == n - 1) {
            if (l == r)
                ans += l;
            l = r = 0;
        } else {
            l++;
        }
    }
    cout << ans << endl;
    return 0;
}
