#include<bits/stdc++.h>
using namespace std;
#define int long long 
const int mod=1e9+7;
const int mxN=1e5+2; 
int n,b,d,a[mxN];
int32_t main()
{
    cin>>n>>d>>b;
    for(int i=1;i<=n;i++) cin>>a[i];
    int l=0,r=n-1,ans=n;
    while(l<r){
        if(a[l]+a[r]<=b){ l++; ans--; }
        else if(a[l]+a[r]==b+1) l++,r--,ans--;
        else{ l++,r--,ans--; }
    }
    cout<<ans<<endl;
    return 0;
}
