#include <bits/stdc++.h>
#define MOD 1000000007
using namespace std;
long long dp[1000001];
int main() {
    int n;
    cin >> n;
    vector<int> h(n);
    for (int i = 0; i < n; i++) { cin >> h[i]; }
    dp[0] = 1;
    dp[1] = 0;
    for (int i = 2; i <= n; i++) {
        dp[i] = ((dp[i - 2] * (h[i - 2] + h[i - 1]) % MOD) + dp[i - 1]) % MOD;
    }
    cout << dp[n] << endl;
    return 0;
}
