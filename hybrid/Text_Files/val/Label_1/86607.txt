import java.util.Scanner;

public class Decathlon {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] students = new int[n];

        for (int i = 0; i < n; i++) {
            students[i] = scanner.nextInt();
        }

        int programmers = 0, mathematicians = 0, sportsmen = 0;
        for (int skill : students) {
            if (skill == 1) {
                programmers++;
            } else if (skill == 2) {
                mathematicians++;
            } else {
                sportsmen++;
            }
        }

        int teams = Math.min(programmers, Math.min(mathematicians, sportsmen));
        System.out.println(teams);

        int programmerIndex = 0, mathematicianIndex = 0, sportsmanIndex = 0;
        for (int i = 0; i < teams; i++) {
            while (students[programmerIndex] != 1) {
                programmerIndex++;
            }
            while (students[mathematicianIndex] != 2) {
                mathematicianIndex++;
            }
            while (students[sportsmanIndex] != 3) {
                sportsmanIndex++;
            }
            System.out.println((programmerIndex + 1) + " " + (mathematicianIndex + 1) + " " + (sportsmanIndex + 1));
            programmerIndex++;
            mathematicianIndex++;
            sportsmanIndex++;
        }
    }
}
