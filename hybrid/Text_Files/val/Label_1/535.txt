import sys

def solve(n, a, b):
    if a == b:
        return "Yes"
    for i in range(1, n // 2 + 1):
        a = a[i:] + a[:i]
        if a == b:
            return "Yes"
    return "No"

t = int(input())
for i in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    print(solve(n, a, b))
