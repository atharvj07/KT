#include <iostream>
#include <vector>
#include <iomanip>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> m(n), r(n);
    for (int i = 0; i < n; i++) cin >> m[i];
    for (int i = 0; i < n; i++) cin >> r[i];

    double ans = 0;
    for (int i = 0; i < n; i++) ans += (double)r[i] / m[i];
    cout << fixed << setprecision(6) << ans << endl;
    return 0;
}
