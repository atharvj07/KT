#include <bits/stdc++.h>
using namespace std;
int n,m,k,a[200005],b[200005];
int main(){
    cin>>n>>m>>k;
    for(int i=1;i<=n;i++)cin>>a[i],b[i]=a[i];
    sort(a+1,a+1+n);
    int l=1,r=m,ans=0;
    while(l<=r){
        int md=(l+r)/2;
        bool chk=0;
        int sum=0;
        for(int i=1;i<=n;i++){
            if(sum+a[i]<=k*md){
                sum+=a[i];
            }
            else{
                sum=a[i];
                md--;
                if(md<0)break;
            }
        }
        if(md>=0)chk=1;
        if(chk)ans=md,l=md+1;
        else r=md-1;
    }
    int sum=0;
    for(int i=1;i<=n;i++){
        if(sum+b[i]<=k*ans)sum+=b[i];
        else break;
        ans--;
        if(ans==0)break;
    }
    cout<<ans+1<<endl;
}
