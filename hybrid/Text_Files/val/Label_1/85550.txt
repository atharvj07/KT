#include <algorithm>
#include <vector>
#include <iostream>

using namespace std;

long double sum(const vector<long double>& v, long long int start, long long int end) {
    long double sum = 0;
    for(long long int i=start; i<end; i++) {
        sum += v[i];
    }
    return sum;
}

long double solve(vector<long double>& v, long long int n, long long int n1, long long int n2) {
    if(n1+n2<n) {
        v.resize(n1+n2);
    }
    sort(v.begin(), v.end());

    long double ans = sum(v, 0, n2)/n2;
    ans += sum(v, n2, n2+n1)/n1;
    return ans;
}

int main() {
    long long int n, n1, n2;
    cin >> n >> n1 >> n2;

    vector<long double> v(n);
    for(long long int i=0; i<n; i++) {
        cin >> v[i];
    }

    cout << solve(v, n, n1, n2) << endl;
    return 0;
}
