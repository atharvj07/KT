import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        while (true) {
            int n = in.nextInt();
            int r = in.nextInt();
            if (n == 0 && r == 0) break;
            int[] deck = new int[n + 1];
            for (int i = 1; i <= n; i++) {
                deck[i] = i;
            }
            for (int i = 0; i < r; i++) {
                int p = in.nextInt();
                int c = in.nextInt();
                int[] temp = new int[n + 1];
                for (int j = 1; j <= p; j++) {
                    temp[j] = deck[j];
                }
                for (int j = p + c; j <= n; j++) {
                    deck[j - c] = deck[j];
                }
                for (int j = 1; j <= c; j++) {
                    deck[n - c + j] = temp[j];
                }
            }
            System.out.println(deck[n]);
        }
    }
}
