#include <iostream>

using namespace std;

int main() {
    int hh, mm;
    cin >> hh >> mm;
    long long H, D, C, N;
    cin >> H >> D >> C >> N;

    long double ans = D * max(0, 20 * 60 - hh * 60 - mm) / N;
    if (H <= ans) {
        cout << "0.0000";
    } else {
        ans = H / N;
        if (hh >= 20) {
            ans *= 0.8;
        }
        cout << ans * C;
    }
    return 0;
}
