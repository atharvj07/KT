#include <bits/stdc++.h>

int main() {
  int n;
  std::string s;
  std::cin >> n >> s;
  int cnt = 0;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'B') {
      cnt++;
    } else {
      if (cnt > 0) {
        ans++;
      }
      cnt = 0;
    }
  }
  if (cnt > 0) {
    ans++;
  }
  std::cout << ans << "\n";
  cnt = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'B') {
      cnt++;
    } else {
      if (cnt > 0) {
        std::cout << cnt << " ";
      }
      cnt = 0;
    }
  }
  if (cnt > 0) {
    std::cout << cnt << "\n";
  }
}
