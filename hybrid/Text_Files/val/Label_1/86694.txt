import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] users = new int[n + 1];
        int leaderCount = 0;
        List<Integer> leaders = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            String message = sc.next();
            if (message.charAt(0) == '+') {
                int id = sc.nextInt();
                if (users[id]++ == 0) {
                    leaderCount++;
                    if (leaderCount == 1) {
                        leaders.add(id);
                    }
                }
            } else {
                int id = sc.nextInt();
                if (--users[id] == 0) {
                    leaderCount--;
                    if (leaderCount == 0) {
                        leaders.clear();
                    } else {
                        leaders.remove(new Integer(id));
                    }
                }
            }
            if (leaderCount > 0) {
                System.out.println(leaderCount);
                for (int id : leaders) {
                    System.out.print(id + " ");
                }
                System.out.println();
            } else {
                System.out.println(0);
            }
        }
    }
}
