import java.util.*;
import java.io.*;

class Main {
    static int[] dishPrice;
    static int[] dishStandard;
    static int[] dishBeauty;
    static int[] personIncome;
    static int[] personBeauty;

    static int count(int income, int beauty) {
        int cnt = 0;
        for (int i = 0; i < dishPrice.length; i++) {
            if (dishPrice[i] <= income && income <= dishStandard[i] && Math.abs(dishBeauty[i] - beauty) <= income - dishPrice[i]) {
                cnt++;
            }
        }
        return cnt;
    }

    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        dishPrice = new int[n];
        dishStandard = new int[n];
        dishBeauty = new int[n];
        personIncome = new int[m];
        personBeauty = new int[m];
        for (int i = 0; i < n; i++) {
            dishPrice[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            dishStandard[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            dishBeauty[i] = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            personIncome[i] = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            personBeauty[i] = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            System.out.print(count(personIncome[i], personBeauty[i]) + " ");
        }
        System.out.println();
    }
}
