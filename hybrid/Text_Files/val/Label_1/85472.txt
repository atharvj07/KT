#include <iostream>
#include <vector>

using namespace std;

const int MAX_N = 1001;
bool isFibonacci[MAX_N];

void preprocess() {
    isFibonacci[1] = isFibonacci[2] = true;
    for (int i = 3; i < MAX_N; i++) {
        isFibonacci[i] = isFibonacci[i - 1] || isFibonacci[i - 2];
    }
}

int main() {
    preprocess();

    int n;
    cin >> n;

    for (int i = 1; i <= n; i++) {
        if (isFibonacci[i]) {
            cout << 'O';
        } else {
            cout << 'o';
        }
    }

    cout << endl;

    return 0;
}
