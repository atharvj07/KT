import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int t = sc.nextInt();
    Ball[] balls = new Ball[n];
    for (int i = 0; i < n; i++) {
      balls[i] = new Ball(sc.nextInt(), sc.nextInt(), sc.nextInt());
    }
    for (int i = 0; i < t; i++) {
      for (int j = 0; j < n; j++) {
        for (int k = j + 1; k < n; k++) {
          if (balls[j].willCollide(balls[k])) {
            balls[j].collide(balls[k]);
          }
        }
        balls[j].update();
      }
    }
    for (int i = 0; i < n; i++) {
      System.out.printf("%.6f\n", balls[i].x);
    }
  }

  static class Ball {
    double x;
    double v;
    double m;

    public Ball(int x, int v, int m) {
      this.x = x;
      this.v = v;
      this.m = m;
    }

    public boolean willCollide(Ball other) {
      return x < other.x && v > other.v;
    }

    public void collide(Ball other) {
      double tmp = v;
      v = (v * m - other.v * other.m) / (m - other.m);
      other.v = (other.v * other.m - tmp * m) / (other.m - m);
    }

    public void update() {
      x += v;
    }
  }
}
