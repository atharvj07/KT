A, B, n = map(int, input().split())
for _ in range(n):
    l, t, m = map(int, input().split())
    def is_valid(x):
        return A + (x - 1) * B <= t and (x - l + 1) * (2 * A + (l + x - 2) * B) <= 2 * m * t
    def bin_search():
        left, right = l, 10**6
        ans = -1
        while left <= right:
            mid = (left + right) // 2
            if is_valid(mid):
                ans = mid
                left = mid + 1
            else:
                right = mid - 1
        return ans
    print(bin_search())
