#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const long long inf = 1e18;

int main() {
    int n, m, x;
    cin >> n >> m;
    vector<int> a(n), b(m);
    for (int &x : a) {
        cin >> x;
    }
    for (int &y : b) {
        cin >> y;
    }
    cin >> x;
    vector<vector<long long>> c(n, vector<long long>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            c[i][j] = 1ll * a[i] * b[j];
        }
    }
    vector<vector<long long>> pref(n + 1, vector<long long>(m + 1));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            pref[i + 1][j + 1] = pref[i][j + 1] + pref[i + 1][j] - pref[i][j] + c[i][j];
        }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i; j < n; j++) {
            for (int k = 0; k < m; k++) {
                int l = k;
                while (l < m && pref[i + 1][l + 1] - pref[i][l + 1] - pref[i + 1][k] + pref[i][k] <= x) {
                    l++;
                }
                ans = max(ans, (j - i + 1) * (l - k));
            }
        }
    }
    cout << ans << '\n';
    return 0;
}
