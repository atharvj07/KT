import sys

for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    p = list(map(int, input().split()))
    b = [0] * n
    for i in range(m):
        b[p[i]] = b[p[i] - 1] ^ a[p[i]] ^ a[p[i] - 1]
    for i in range(n - 1):
        if b[i] > b[i + 1]:
            print("NO")
            sys.exit()
    print("YES")
