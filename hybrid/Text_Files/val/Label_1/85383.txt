#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5+5;
int n, ans, l[maxn], r[maxn], L[maxn], R[maxn];
char a[maxn], b[maxn];
string s(char *x) { return string(x); }
int main() {
    cin >> n;
    cin >> a >> b;
    for (int i = 1; i <= n; i++) {
        l[i] = (s(a) == "?" || s(a) == s(b)) ? 1 : 0;
        r[i] = (s(b) == "?" || s(a) == s(b)) ? 1 : 0;
    }
    for (int i = 1; i <= n; i++) l[i] += l[i-1], r[i] += r[i-1];
    for (int i = n; i >= 1; i--) L[i] = L[i+1] + l[i-1], R[i] = R[i+1] + r[i-1];
    for (int i = 1; i <= n; i++) {
        int mx1 = 0, mx2 = 0;
        mx1 = L[i], mx2 = R[i+1];
        ans = max(ans, mx1+mx2);
    }
    cout << ans << endl;
    for (int i = 1; i <= n; i++) {
        int mx1 = 0, mx2 = 0, idx = 0;
        mx1 = L[i], mx2 = R[i+1];
        if (ans == mx1 + mx2) idx = i;
        while(idx && mx1) {
            if (l[idx-1] > 0) mx1--, ans--, cout << idx << " " << idx << endl, idx = 0;
            else idx--;
        }
        idx = i;
        while(idx <= n && mx2) {
            if (r[idx] > 0) mx2--, ans--, cout << idx << " " << idx << endl, idx = n+1;
            else idx++;
        }
    }
}
