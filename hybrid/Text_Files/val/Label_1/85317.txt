#include <iostream>
#include <vector>

int main() {
  int n, m, k;
  std::cin >> n >> m >> k;
  std::vector<int> a(n), b(m);
  for (int i = 0; i < n; i++) std::cin >> a[i];
  for (int i = 0; i < m; i++) std::cin >> b[i];
  int ans = 0;
  for (int i = 0; i < k; i++) {
    if (a[(i % n)] != b[(i % m)]) ans++;
  }
  std::cout << ans;
  return 0;
}
