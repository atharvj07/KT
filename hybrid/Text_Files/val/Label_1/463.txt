import queue
def update(d,a,b,k):
    if (a,b)in d:del d[(a,b)]
    if k==0:return
    d[(a,b)]=k
def solve():
    n=int(input())
    a=list(map(int,input().split()))
    q=int(input())
    d=dict()
    for i in range(1,q+1):
        s,t,u=map(int,input().split())
        update(d,s,t,u)
        cur=queue.Queue()
        cur.put((1,0,0))
        ans=max(a)
        while not cur.empty():
            x,y,vis=cur.get()
            if x>ans:break
            if y>=a[x-1]:
                ans=min(ans,x+vis)
            for s,t,u in d.values():
                if s!=x:continue
                if t>y:
                    if not (x,y,vis+1) in d:
                        cur.put((x,y,vis+1))
                else:
                    if not (u,y+1,vis+1) in d:
                        cur.put((u,y+1,vis+1))
        print(ans)
solve()
