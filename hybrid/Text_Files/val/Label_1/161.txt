n, w = map(int, input().split())
changes = list(map(int, input().split()))

min_passengers = 0
max_passengers = w
current_passengers = 0

for change in changes:
    current_passengers += change
    min_passengers = max(min_passengers, -current_passengers)
    max_passengers = min(max_passengers, w - current_passengers)

if max_passengers >= min_passengers:
    print(max_passengers - min_passengers + 1)
else:
    print(0)
