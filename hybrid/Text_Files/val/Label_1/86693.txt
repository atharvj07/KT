import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();

        int[] degrees = new int[n + 1];
        for (int i = 0; i < m; i++) {
            char action = in.next().charAt(0);
            int id = in.nextInt();
            if (action == '+') {
                degrees[id]++;
            } else {
                degrees[id]--;
            }
        }

        int leaderCount = 0;
        List<Integer> leaders = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            if (degrees[i] > 0) {
                leaderCount++;
                leaders.add(i);
            }
        }

        System.out.println(leaderCount);
        for (int leader : leaders) {
            System.out.print(leader + " ");
        }
        System.out.println();
    }
}
