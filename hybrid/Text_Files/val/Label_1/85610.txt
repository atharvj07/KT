#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5+5;
long long n, t, c;
int a[maxn];
long long dp[maxn][4];
int main() {
  cin >> n >> t >> c;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  dp[n+1][3] = 1;
  for (int i = n; i >= 1; i--) {
    for (int j = 0; j < 3; j++) {
      if (j != 2) {
        dp[i][j] = dp[i+1][j];
        if (a[i] <= t) dp[i][j] += dp[i+1][j+1];
      } else {
        dp[i][j] = dp[i+1][j];
        if (a[i] <= t && i+c-1 <= n && a[i+c-1] <= t) { dp[i][j] += dp[i+c][0]; }
      }
    }
  }
  cout<<dp[1][0]<<"\n";
}
