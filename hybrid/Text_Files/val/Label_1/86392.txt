import java.util.Scanner;

public class MartianLuckyNumbers {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int k = scanner.nextInt();
        int b = scanner.nextInt();
        int n = scanner.nextInt();

        int[] digits = new int[n];
        for (int i = 0; i < n; i++) {
            digits[i] = scanner.nextInt();
        }

        int luckySubstrings = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                int digitalRoot = getDigitalRoot(digits, i, j, k);
                if (digitalRoot == b) {
                    luckySubstrings++;
                }
            }
        }

        System.out.println(luckySubstrings);
    }

    private static int getDigitalRoot(int[] digits, int start, int end, int k) {
        int sum = 0;
        for (int i = start; i <= end; i++) {
            sum += digits[i];
        }

        while (sum >= k) {
            int newSum = 0;
            while (sum > 0) {
                newSum += sum % k;
                sum /= k;
            }
            sum = newSum;
        }

        return sum;
    }
}
