#include <iostream>
#include <vector>
#include <numeric>
#include <unordered_map>
using namespace std;

int main() {
    int t; cin >> t;
    vector<pair<int, vector<int>>> arrays(t);
    for (int i = 0; i < t; i++) {
        int n; cin >> n;
        vector<int> a(n);
        for (int j = 0; j < n; j++) cin >> a[j];
        arrays[i] = {n, a};
    }
    for (auto& [n, a] : arrays) {
        unordered_map<int, int> freq;
        for (int i = 1; i < n; i++) freq[a[i]-a[i-1]]++;
        vector<int> diffs;
        for (auto& [diff, count] : freq) {
            if (count > n - 1) diffs.push_back(diff);
        }
        if ((int) diffs.size() <= 1) {
            if (diffs.empty()) cout << 0 << endl;
            else {
                int m = a[n-1] - a[0] + diffs[0];
                cout << m << " " << diffs[0] << endl;
            }
        }
        else cout << -1 << endl;
    }
}
