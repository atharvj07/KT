s = [list(input()) for i in range(int(input().split()[0]))]
for i in range(len(s)):
    for j in range(len(s[0])):
        if s[i][j] == '.':
            if (i % 2 == 0 and j % 2 == 0) or (i % 2 == 1 and j % 2 == 1):
                s[i][j] = 'W'
            else:
                s[i][j] = 'B'
for i in s:
    print(''.join(i))
