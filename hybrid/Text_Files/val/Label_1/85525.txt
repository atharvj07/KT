#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

template <class T> void rotate(vector<vector<T>>& m) {
  int n = m.size();
  m.resize(m[0].size());
  for (int i = 0; i < m.size(); ++i) {
    m[i].resize(n);
    for (int j = 0; j < n; ++j) { m[i][j] = std::move(m[n - j - 1][i]); }
  }
}

template <class T> void mirror(vector<vector<T>>& m) {
  for (auto& row : m) {
    reverse(row.begin(), row.end());
  }
}

template <class T> void zoom(vector<vector<T>>& m) {
  int n = m.size();
  int w = m[0].size();
  m.resize(n * 2);
  for (int i = 0; i < n; ++i) {
    m[i * 2].resize(w * 2);
    m[i * 2 + 1].resize(w * 2);
    for (int j = 0; j < w; ++j) {
      m[i * 2][j * 2] = std::move(m[i][j]);
      m[i * 2][j * 2 + 1] = std::move(m[i][j]);
      m[i * 2 + 1][j * 2] = std::move(m[i][j]);
      m[i * 2 + 1][j * 2 + 1] = std::move(m[i][j]);
    }
  }
}

int main() {
  int w, h;
  cin >> w >> h;
  vector<vector<char>> m(h, vector<char>(w));
  for (auto& row : m) {
    for (auto& col : row) { cin >> col; }
  }
  rotate(m);
  mirror(m);
  zoom(m);
  for (auto& row : m) {
    for (auto& col : row) { cout << col; }
    cout << "\n";
  }
  return 0;
}
