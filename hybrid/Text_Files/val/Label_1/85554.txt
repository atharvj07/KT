#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 3e5 + 5;
const int MAXM = 3e5 + 5;

struct Edge
{
    int u, v, w;
};

vector<Edge> edges[MAXN];
int dist[MAXN];
bool vis[MAXN];

int main()
{
    int n, m;
    cin >> n >> m;

    for (int i = 0; i < m; i++)
    {
        int u, v, w;
        cin >> u >> v >> w;
        edges[u].push_back({u, v, w});
    }

    int ans = 0;

    for (int i = 1; i <= n; i++)
    {
        if (!vis[i])
        {
            int cnt = 0;
            vis[i] = true;
            for (Edge e : edges[i])
            {
                if (e.w > dist[e.u])
                {
                    cnt++;
                    dist[e.v] = e.w;
                    vis[e.v] = true;
                }
            }
            ans = max(ans, cnt);
        }
    }

    cout << ans << endl;

    return 0;
}
