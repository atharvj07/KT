import java.util.Scanner;

public class TeamScientificDecathlonOlympias {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] skills = new int[n];

        for (int i = 0; i < n; i++) {
            skills[i] = scanner.nextInt();
        }

        int[] programmers = new int[n];
        int[] mathematicians = new int[n];
        int[] athletes = new int[n];

        int programmerCount = 0;
        int mathematicianCount = 0;
        int athleteCount = 0;

        for (int i = 0; i < n; i++) {
            switch (skills[i]) {
                case 1:
                    programmers[programmerCount++] = i + 1;
                    break;
                case 2:
                    mathematicians[mathematicianCount++] = i + 1;
                    break;
                case 3:
                    athletes[athleteCount++] = i + 1;
                    break;
                default:
                    break;
            }
        }

        int w = Math.min(programmerCount, Math.min(mathematicianCount, athleteCount));

        System.out.println(w);

        for (int i = 0; i < w; i++) {
            System.out.println(programmers[i] + " " + mathematicians[i] + " " + athletes[i]);
        }
    }
}
