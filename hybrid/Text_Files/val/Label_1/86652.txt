import java.util.*;
class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long a[] = new long[n + 1];
    long b[] = new long[n + 1];
    for (int i = 1; i <= n; i++) a[i] = sc.nextLong();
    for (int i = 1; i <= n; i++) b[i] = sc.nextLong();
    sc.close();
    long min = Long.MAX_VALUE;
    long dp[][] = new long[n + 2][n + 1];
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= n; j++) {
        dp[i][j] = dp[i - 1][j];
        if (b[j] - a[j] >= 0) dp[i][j] += b[j] - a[j];
        if (a[j] - b[j] > 0) {
          min = Math.min(min, dp[i][j] + a[j]);
          dp[i][j] = Math.min(dp[i][j], dp[i - 1][j - 1]);
        }
      }
    }
    System.out.println(min == Long.MAX_VALUE ? 0 : min);
  }
}
