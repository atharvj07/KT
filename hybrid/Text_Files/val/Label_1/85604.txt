#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(m);
    for (int i = 0; i < m; i++) cin >> a[i];
    vector<int> b(n + 1, 0);
    for (int i = 0; i < n + 1; i++) b[i] = i;
    string ans = "";
    for (int i = 0; i < m; i++) {
        int j = a[i];
        if (j != b[j]) ans += '0';
        else {
            ans += '1';
            while (j < n) {
                b[j] = b[j + 1];
                j++;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
