import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] size = new int[2];
        for (int i = 0; i < size.length; ++i) {
            size[i] = sc.nextInt();
        }
        sc.close();

        System.out.println(Math.max(Math.max(size[0], size[1]) / 3,
                Math.min(size[0], size[1]) / 2 + 1));
    }
}
