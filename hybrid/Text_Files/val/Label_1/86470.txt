import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[][] arr = new int[n][2];
        for (int i = 0; i < n; i++) {
            arr[i][0] = i + 1;
            arr[i][1] = Math.abs(i - (n - 1));
        }
        Arrays.sort(arr, (a, b) -> b[1] - a[1]);
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            ans[i] = arr[i][0];
        }
        System.out.println(Arrays.toString(ans).replace("[", "").replace("]", "").replace(",", ""));
    }
}
