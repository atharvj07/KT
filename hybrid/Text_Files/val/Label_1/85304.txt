#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5 + 5;

int n;
string s[MAXN];
map<string, int> id;
int fa[MAXN], sz[MAXN];

int find(int x) {
    if (x != fa[x]) fa[x] = find(fa[x]);
    return fa[x];
}
void merge(int x, int y) {
    x = find(x); y = find(y);
    if (x == y) return;
    if (sz[x] > sz[y]) swap(x, y);
    fa[x] = y;
    sz[y] += sz[x];
}

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> s[i];
        int pos = s[i].find('/');
        if (pos == -1) id[s[i]] = i;
        else id[s[i].substr(0, pos)] = i;
    }
    for (int i = 0; i < n; i++) {
        fa[i] = i; sz[i] = 1;
        for (int j = 0; j < i; j++) {
            if (s[i].find(s[j]) == 0 || s[j].find(s[i]) == 0) { merge(i, j); }
        }
    }
    map<int, vector<string> > ans;
    for (int i = 0; i < n; i++) ans[find(i)].push_back(s[i]);
    cout << ans.size() - 1 << endl;
    for (auto &p : ans) {
        if (p.second.size() == 1) continue;
        for (auto &x : p.second) { cout << x << ' '; } cout << endl;
    }
    return 0;
}
