#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<int> x(n), p(m);
  for (int i = 0; i < n; ++i) std::cin >> x[i];
  for (int j = 0; j < m; ++j) std::cin >> p[j];
  std::sort(p.begin(), p.end());
  for (int y = 0;; ++y) {
    int i = 0;
    while (i != m && p[i] <= x[0] - y) ++i;
    if (i != m) {
      bool all = true;
      for (int j = 0; j != n; ++j) {
        int z = std::max(x[j], y);
        if ((z - y) % p[i] != 0) { all = false; break; }
      }
      if (all) {
        std::cout << "YES" << std::endl;
        std::cout << y << ' ' << i + 1 << std::endl;
        return 0;
      }
    }
  }
  std::cout << "NO" << std::endl;
  return 0;
}
