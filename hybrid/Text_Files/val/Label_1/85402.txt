#include <iostream>
#include <vector>
#include <algorithm>

typedef long long ll;

using namespace std;

int main() {
    int n, a, b;
    ll hp, dmg, ans = 0;
    vector<pair<ll, ll>> v;

    cin >> n >> a >> b;
    for (int i = 0; i < n; i++) {
        cin >> hp >> dmg;
        v.push_back({hp, dmg});
    }

    sort(v.begin(), v.end());

    for (int i = 0; i < n; i++) {
        if (a > 0 && v[i].first < v[i].second) {
            ans += v[i].second;
            v[i].first = v[i].second;
            a--;
        } else {
            ans += v[i].first;
        }
    }

    sort(v.begin(), v.end(), [](const pair<ll, ll>& a, const pair<ll, ll>& b) { return a.second > b.second; });

    for (int i = 0; i < n && b > 0; i++) {
        ans += v[i].second;
        b--;
    }

    cout << ans << endl;

    return 0;
}
