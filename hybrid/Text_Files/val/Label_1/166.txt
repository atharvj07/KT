import sys

n, m, x, y, z, p = map(int, sys.stdin.readline().split())
candies = [list(map(int, sys.stdin.readline().split())) for _ in range(p)]

for candy in candies:
    x, y = candy
    x = (x - 1 + (z - x) % 4 + y - 1 + (1 - y) % 4) % n + 1
    y = (y - 1 + (x - 1) % 4 + 1 - x + (y - 1) % 4) % m + 1
    print(x, y)
