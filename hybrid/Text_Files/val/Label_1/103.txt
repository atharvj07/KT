n = int(input())
a = list(map(int, input().split()))

neg, pos = 0, 0
negative_count = 0
total_subsegments = n * (n + 1) // 2

for i in range(n):
    if a[i] < 0:
        negative_count += 1
    if negative_count % 2 == 0:
        pos += 1
    else:
        neg += 1

neg += negative_count * (negative_count - 1) // 2
pos = total_subsegments - neg

print(neg, pos)
