#include<bits/stdc++.h>
#include<cstdio>
using namespace std;
const int MAXN = 505;
int n, sum, a[MAXN][MAXN];
int R, C;
bool ok(){
  for(int i = 1; i <= n; i++){
    int sum_row = 0, sum_col = 0;
    for(int j = 1; j <= n; j++){
      sum_row += a[i][j];
      sum_col += a[j][i];
    }
    if(sum_row != sum || sum_col != sum)return 0;
  }
  int sum_dig1 = 0, sum_dig2 = 0;
  for(int i = 1; i <= n; i++){
    sum_dig1 += a[i][i];
    sum_dig2 += a[i][n - i + 1];
  }
  if(sum_dig1 != sum || sum_dig2 != sum)return 0;
  return 1;
}
void solve(){
  int add = sum - a[R][C];
  if(add <= 0 || add > n * n)
    cout << -1 << '\n';
  else 
    cout << add << '\n';
}
int main(){
  cin >> n;
  sum = (n * (n * n + 1)) / 2;
  for(int i = 1; i <= n; i++){
    for(int j = 1; j <= n; j++){
      cin >> a[i][j];
      if(a[i][j] == 0){
        R = i, C = j;
      }
    }
  }
  if(ok()){
    solve();
    return 0;
  }
  for(int i = 1; i <= n; i++){
    for(int j = 1; j <= n; j++){
      if(a[i][j] != a[R][C] && a[i][j] != 0){
        swap(a[i][j], a[R][C]);
        if(ok()){
          solve();
          return 0;
        }
        swap(a[i][j], a[R][C]);
      }
    }
  }
  cout << -1 << '\n';
  return 0;
}
