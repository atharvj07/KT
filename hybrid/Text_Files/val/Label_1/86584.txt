import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int n = scanner.nextInt();
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            if (n == 0 && a == 0 && b == 0) {
                break;
            }
            int count = 0;
            for (int i = 1; i <= n; i++) {
                if (!((i % a == 0) || (i % b == 0))) {
                    count++;
                }
            }
            System.out.println(count);
        }
    }
}
