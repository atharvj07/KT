#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll N = 3e5 + 9;
ll n, m, x[N], p[N];
bool ok[N];
int main() {
    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> x[i];
    for (int i = 0; i < m; i++) cin >> p[i];
    ll ans1 = 0, ans2 = 0;
    for (int j = 0; j < m; j++) {
        for (int i = 0; i < n; i++) {
            ok[i] = 0;
        }
        ll mn = 1e18;
        for (int i = 0; i < n; i++) {
            ll tmp = x[i] % p[j];
            tmp = p[j] - tmp;
            if (tmp <= mn) {
                mn = tmp;
                ans1 = tmp;
                ans2 = j + 1;
            }
            ll l = lower_bound(x + i, x + n, x[i] - tmp) - x;
            if (l == i) continue;
            if (x[l] == x[i] - tmp) ok[l] = 1;
        }
        int ch = 1;
        for (int i = 0; i < n && ch; i++) {
            if (!ok[i]) ch = 0;
        }
        if (ch) {
            cout << "YES" << endl;
            cout << ans1 << " " << ans2 << endl;
            return 0;
        }
    }
    cout << "NO" << endl;
    return 0;
}
