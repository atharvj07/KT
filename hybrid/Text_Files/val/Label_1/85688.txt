#include<bits/stdc++.h>

using namespace std;


int main() {
    int n;
    cin >> n;
    vector<int> v(n + 2), pos(n + 2);
    for (int i = 1; i <= n; ++i) {
        cin >> v[i];
        pos[v[i]] = i;
    }
    if (v[1] < v[n]) {
        cout << pos[n] - 1;
    } else {
        cout << pos[1] - 1;
    }

    return 0;
}
