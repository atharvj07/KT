import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static long minCost(String[] s, long[][] a, int n, int m) {
        long ans = 0;
        for (int i = 0; i < n; i++) {
            boolean[] vis = new boolean[26];
            long minCost = 0;
            for (int j = 0; j < m; j++) {
                if (vis[s[i].charAt(j) - 'a']) {
                    minCost = -1;
                    break;
                }
                vis[s[i].charAt(j) - 'a'] = true;
                minCost = Math.min(minCost, a[i][j]);
            }
            if (minCost != -1) ans += minCost;
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        String[] s = new String[n];
        long[][] a = new long[n][m];
        for (int i = 0; i < n; i++) {
            s[i] = sc.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = sc.nextLong();
            }
        }
        System.out.println(minCost(s, a, n, m));
    }
}
