import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    static int n, m;
    static int[] a, b, w;
    static List<Integer>[] g;
    static int[] match;
    static boolean[] used;

    public static void main(String[] args) {
        try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
            String[] line = br.readLine().split(" ");
            n = Integer.parseInt(line[0]);
            m = Integer.parseInt(line[1]);
            a = new int[m];
            b = new int[m];
            w = new int[m];
            g = new List[n + 1];
            match = new int[n + 1];
            used = new boolean[n + 1];
            Arrays.fill(match, -1);
            for (int i = 0; i <= n; i++) {
                g[i] = new ArrayList<>();
            }
            for (int i = 0; i < m; i++) {
                line = br.readLine().split(" ");
                a[i] = Integer.parseInt(line[0]);
                b[i] = Integer.parseInt(line[1]);
                w[i] = Integer.parseInt(line[2]);
                g[a[i]].add(i);
                g[b[i]].add(i);
            }
            int sum = 0;
            for (int i = 1; i <= n; i++) {
                Arrays.fill(used, false);
                sum += dfs(i);
            }
            System.out.println(sum);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public static int dfs(int v) {
        if (used[v]) return 0;
        used[v] = true;
        for (int i : g[v]) {
            if (match[a[i]] == -1 || dfs(match[a[i]]) != 0) {
                match[a[i]] = v;
                match[v] = a[i];
                return w[i];
            }
            if (match[b[i]] == -1 || dfs(match[b[i]]) != 0) {
                match[b[i]] = v;
                match[v] = b[i];
                return w[i];
            }
        }
        return 0;
    }
}
