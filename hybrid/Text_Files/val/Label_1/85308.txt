#include <bits/stdc++.h>
using namespace std;

map<string, set<string> > domains;
map<string, string> redir;
const int N=100001;
int main() {
	int n; cin>>n;
	for(int i=0; i<n; i++) {
		string page; cin>>page;
		int p=page.find("/", 7);
		string host=""; string path="";
		if(p>=0) {
			host=page.substr(7, p-7);
			path=page.substr(p+1);
		} else {
			host=page.substr(7);
		}
		int pp=host.find(".");
		string root=host.substr(pp+1);
		bool first=0;
		while(pp>=0) {
			string sub=host.substr(pp+1);
			if(sub.size()<root.size()) {
				root=sub;
				first=1;
			}
			if(first) path=redir[sub]+path;
			else if(domains.find(sub)!=domains.end()) domains[sub].insert(path);
			else {
				set<string> tmp;
				tmp.insert(path);
				domains[sub]=tmp;
			}
			pp=host.find(".", pp+1);
		}
		redir[host]=path+"/";
	}
	map<string, set<string> >::iterator it;
	int ans=0;
	for(it=domains.begin(); it!=domains.end(); it++) {
		if(it->second.size()>1) ans++;
	}
	cout<<ans<<"\n";
	for(it=domains.begin(); it!=domains.end(); it++) {
		if(it->second.size()>1) {
			set<string> paths=it->second;
			for(set<string>::iterator jt=paths.begin(); jt!=paths.end(); jt++) {
				cout<<it->first<<"/"<<*jt<<" ";
			}
			cout<<"\n";
		}
	}
}
