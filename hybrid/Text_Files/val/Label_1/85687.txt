#include <bits/stdc++.h>
#define int long long
using namespace std;

const int INF = 1e18;

const int N = 2e5 + 1;
vector<pair<int, int>> edges[N];

int n, m;
int a[N], dist[N];

void dijkstra(int src) {
  for (int i = 1; i <= n; i++) dist[i] = INF;
  dist[src] = 0;

  set<pair<int, int>> s;
  s.insert({0, src});

  while (!s.empty()) {
    int u = s.begin()->second;
    s.erase(s.begin());

    for (auto &e : edges[u]) {
      int v = e.first;
      int w = e.second;
      if (dist[v] > dist[u] + w) {
        s.erase({dist[v], v});
        dist[v] = dist[u] + w;
        s.insert({dist[v], v});
      }
    }
  }
}

void solve() {
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    int u, v, w;
    cin >> u >> v >> w;
    edges[u].push_back({v, w});
    edges[v].push_back({u, w});
  }
  for (int i = 1; i <= n; i++) cin >> a[i];

  for (int i = 1; i <= n; i++) {
    dijkstra(i);
    int ans = INF;
    for (int j = 1; j <= n; j++) {
      ans = min(ans, dist[j] * 2 + a[j]);
    }
    cout << ans << " ";
  }
}

int32_t main() {
  solve();
  return 0;
}
