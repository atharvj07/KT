import sys
rl = sys.stdin.readline

def get_ints(): return map(int, rl().strip().split())

_2pow30 = 1 << 30  # 2^30
for _ in range(int(rl())):
    n, w = get_ints()
    if w < _2pow30: print(-1); continue
    ok = [-1] * n
    wsum = 0
    lst = []
    for i in range(n):
        wi = int(rl())
        if ok[i] != -1: continue
        if wsum + wi >= _2pow30:
            ok[i] = 1; lst.append(i)
            wsum += wi
            continue
        for j in range(i + 1, n):
            if ok[j] != -1: continue
            if wsum + wi + int(rl()) >= _2pow30:
                ok[i] = ok[j] = 1; lst.extend([i, j])
                wsum += wi + int(rl())
                break
    if len(lst) > 0 and ok[0] != -1:
        print(len(lst)); print(*lst)
    else: print(-1)
