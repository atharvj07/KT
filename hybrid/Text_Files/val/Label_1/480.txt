import math

def fib(x):
    if x == 0 or x == 1:
        return 1
    return fib(x - 1) + fib(x - 2)

mod = (int) (1e9)

n, m = map(int, input().split())
a = list(map(int, input().split()))
for _ in range(m):
    t, *params = map(int, input().split())
    if t == 1:
        i, v = params
        a[i - 1] = v
    elif t == 2:
        l, r = params
        ans = 0
        for i in range(l - 1, r):
            ans += a[i] * fib(i - l + 2)
            ans %= mod
        print(ans)
    else:
        l, r, d = params
        for i in range(l - 1, r):
            a[i] += d
