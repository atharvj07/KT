#include <bits/stdc++.h>
using namespace std;
constexpr int maxn = 2e5 + 10;
struct ele {
    long long hp, dmg;
    bool operator<(const ele& a) const { return dmg > a.dmg; }
};
long long a[maxn], b[maxn];
int main() {
    int n, xa, xb;
    cin >> n >> xa >> xb;
    for (int i = 0; i < n; i++) cin >> a[i] >> b[i];
    sort(a, a + n);
    sort(b, b + n);
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        if (i < xa)
            ans += a[i] + a[i];
        else if (i - xa < xb)
            ans += b[i];
        else
            ans += b[i] + b[i];
    }
    cout << ans << endl;
    return 0;
}
