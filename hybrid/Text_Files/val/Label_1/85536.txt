#include<bits/stdc++.h>
using namespace std;
#define ll long long int
void solve(){
	ll n;
	cin>>n;
	vector<ll>a(n);
	for(ll i=0;i<n;i++){
		cin>>a[i];
	}
	ll c=0;
	for(ll i=0;i<n;i++){
		c+=a[i];
		if(c>=n){
			c=c%n;
		}
		if(c<0){
			c=(c%n+n)%n;
		}
	}
	if(c==0){
		cout<<"YES"<<endl;
	}
	else{
		cout<<"NO"<<endl;
	}
}
int main(){
	ll t;
	cin>>t;
	while(t--){
		solve();
	}
	return 0;
}
