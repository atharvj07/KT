#include <iostream>
#include <cstring>
#include <vector>

using namespace std;

const int MAXN = 50005, MAXM = 11;

string s;
int n, p[MAXN];
bool f[MAXN][27];
vector<pair<string, pair<int, int>>> v;

int main() {
  cin >> s;
  cin >> n;
  for (int i = 0; i < n; ++i) {
    string a;
    int l, r;
    cin >> a >> l >> r;
    v.push_back(make_pair(a, make_pair(l, r)));
  }
  p[s.size()] = -1;
  for (int i = s.size() - 1; i >= 0; --i) {
    p[i] = i + 1;
    for (int j = i + 1; j <= s.size(); ++j)
      if (s[i] == s[j] && p[j] < p[i]) p[i] = p[j];
  }
  for (int i = 0; i < s.size(); ++i) {
    for (int j = i; j < s.size(); ++j) {
      string now = s.substr(i, j - i + 1);
      int cnt = 0;
      for (int k = p[i]; k >= 0; k = p[k]) {
        string pref = s.substr(k, j - i + 1);
        if (now == pref) cnt++;
      }
      bool ok = 1;
      for (int k = 0; k < v.size(); ++k) {
        string a = v[k].first;
        int l = v[k].second.first, r = v[k].second.second;
        if (now == a) {
          ok &= cnt >= l && cnt <= r;
        }
      }
      if (ok) f[i][j - i + 1] = 1;
    }
  }
  int ans = 0;
  for (int i = 0; i < s.size(); ++i)
    for (int j = 0; j < 27; ++j)
      if (f[i][j]) ans++;
  cout << ans << endl;
  return 0;
}
