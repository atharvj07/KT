#include<bits/stdc++.h>
#define ll long long
using namespace std;
const ll INF = 1e18;
const int N = 2e5 + 5;

int n, m;
vector<pair<ll, ll>>adj[N];
ll a[N], dis[N];

void dijkstra(int src) {
    for (int i = 1; i <= n; i++)
        dis[i] = INF;
    priority_queue<pair<ll, int>>pq;
    pq.push({0, src});
    dis[src] = 0;
    while (!pq.empty()) {
        int u = pq.top().second;
        ll d = -pq.top().first;
        pq.pop();
        if (d > dis[u]) continue;
        for (auto e: adj[u]) {
            int v = e.first;
            ll w = e.second;
            if (dis[u] + w < dis[v]) {
                dis[v] = dis[u] + w;
                pq.push({-dis[v], v});
            }
        }
    }
}

int main() {
    int u, v, w;
    cin >> n >> m;
    for (int i = 1; i <= m; i++) {
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    for (int i = 1; i <= n; i++)
        cin >> a[i];
    for (int i = 1; i <= n; i++) {
        dijkstra(i);
        ll mn = INF;
        for (int j = 1; j <= n; j++) {
            mn = min(mn, 2 * dis[j] + a[j]);
        }
        cout << mn << " ";
    }
    cout << "\n";
    return 0;
}
