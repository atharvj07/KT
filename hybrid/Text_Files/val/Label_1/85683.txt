#include <bits/stdc++.h>
using namespace std;

const int inf = 1e9;
const int mx = 2e5 + 7;

int32_t main() {
    int n, m; cin >> n >> m;
    vector<vector<pair<int, int>>> g(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v, w; cin >> u >> v >> w;
        g[u].push_back({v, w});
        g[v].push_back({u, w});
    }
    vector<int> cost(n + 1);
    for (int i = 1; i <= n; i++) { cin >> cost[i]; }

    vector<int> dist(n + 1, inf);

    function<void(int, int)> dijkstra = [&](int src, int parent) {
        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
        pq.push({0, src});
        dist[src] = 0;

        while (!pq.empty()) {
            int dis = pq.top().first;
            int node = pq.top().second;
            pq.pop();

            if (dis > dist[node]) continue;

            for (auto &child : g[node]) {
                int child_node = child.first;
                int child_weight = child.second;
                if (child_node == parent) continue;
                if (dist[node] + child_weight < dist[child_node]) {
                    dist[child_node] = dist[node] + child_weight;
                    pq.push({dist[child_node], child_node});
                }
            }
        }
    };

    for (int i = 1; i <= n; i++) { dijkstra(i, -1); }

    for (int i = 1; i <= n; i++) {
        int ans = inf;
        for (int j = 1; j <= n; j++) {
            ans = min(ans, 2 * dist[j] + cost[j]);
        }
        cout << ans << " ";
    }
    cout << endl;
    return 0;
}
