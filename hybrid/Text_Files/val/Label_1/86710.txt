import java.util.*;
import java.io.*;
class Main {
    static int maxSensors(int[] a, int[] b, int y1, int y2) {
        int n = a.length, m = b.length;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                long x1 = a[i], y = y1, x2 = b[j], y3 = y2;
                if (x1 < x2) {
                    y = y2 - (y2 - y1) * (x2 - x1) / (y2 - y1);
                } else {
                    y = y1 + (y2 - y1) * (x1 - x2) / (y2 - y1);
                }
                if (y >= Math.min(y1, y2) && y <= Math.max(y1, y2))
                    ans = Integer.max(ans, i + 1 + j + 1);
            }
        }
        return ans;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.valueOf(line[0]), y1 = Integer.valueOf(line[1]);
        line = br.readLine().split(" ");
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = Integer.valueOf(line[i]);
        }
        line = br.readLine().split(" ");
        int m = Integer.valueOf(line[0]), y2 = Integer.valueOf(line[1]);
        line = br.readLine().split(" ");
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = Integer.valueOf(line[i]);
        }
        System.out.println(maxSensors(a, b, y1, y2));
    }
}
