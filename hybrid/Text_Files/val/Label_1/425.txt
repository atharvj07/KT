n=int(input())
s=input()
t=s[0]
need_change=False
for i in range(1,n):
    if s[i]!=s[i-1]:
        t+=s[i]
        need_change=False
    else:
        if not need_change:
            need_change=True
            t+=s[i-1] if s[i]==t[-1] else s[i]
        else:
            t+=s[i-1] if s[i]==s[i-1] else s[i]
print(n-len(t))
print(t)
