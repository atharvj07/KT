#include <bits/stdc++.h>

using namespace std;
#define endl "\n"

unordered_map<int, int> getFreq(string a) {
    unordered_map<int, int> freq;
    for (char c : a) { freq[c]++; }
    return freq;
}

int main() {
    int n, m;
    cin >> n >> m;
    string s;
    cin >> s;
    while (m--) {
        int x, y, len;
        cin >> x >> y >> len;
        string s1 = s.substr(x - 1, len);
        string s2 = s.substr(y - 1, len);
        if (getFreq(s1) == getFreq(s2))
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
}
