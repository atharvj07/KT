#include <bits/stdc++.h>
using namespace std;

const int N = 2005;
int n, m, x, p[N][N];
long long a[N], b[N], c[N][N];

int main() {
  ios::sync_with_stdio(0), cin.tie(0);
  cin >> n >> m;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 1; i <= m; i++) { cin >> b[i]; }
  cin >> x;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      cin >> c[i][j];
      p[i][j] = p[i - 1][j] + p[i][j - 1] - p[i - 1][j - 1] + c[i][j];
    }
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      for (int k = i; k <= n; k++) {
        for (int l = j; l <= m; l++) {
          if (p[k][l] - p[k][j - 1] - p[i - 1][l] + p[i - 1][j - 1] <= x) {
            ans = max(ans,
                     (k - i + 1) * (l - j + 1));  // Area of submatrix
          }
        }
      }
    }
  }
  cout << ans << endl;
  return 0;
}
