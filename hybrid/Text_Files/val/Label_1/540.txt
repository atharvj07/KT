n, t = map(int,input().split())
s = input()
p = s.find('.')
ans = s
while t:
    if p >= 0 and p < len(s) - 1:
        if s[p + 1] >= '5':
            s = s[:p] + str(int(s[:p]) + 1) + '0'
            if p > 0 and s[p-1] == '9':
                p -= 1
        else:
            s = s[:p+1]
        ans = max(ans, s)
    t -= 1
print(ans.rstrip('0'))
