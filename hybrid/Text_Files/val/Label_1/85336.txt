#include <cstdio>
#include <algorithm>
#define ll long long
using namespace std;
const int N = 100005, mod = 1000000007, oo = (int)1e9 + 7;
ll sum[N], seg[4 * N];
int n, x;
ll *a;
void build(int node, int st, int en)
{
    if (st == en)
    {
        seg[node] = *a % x == 0 ? *a / x : *a / x + 1;
        return;
    }
    int mid = (st + en) / 2;
    build(2 * node, st, mid);
    build(2 * node + 1, mid + 1, en);
    seg[node] = seg[2 * node] + seg[2 * node + 1];
}
ll get(int node, int st, int en, int l, int r)
{
    if (st > r || en < l)
        return 0;
    if (l <= st && en <= r)
        return seg[node];
    int mid = (st + en) / 2;
    return get(2 * node, st, mid, l, r) + get(2 * node + 1, mid + 1, en, l, r);
}
int main()
{
    int t;
    scanf("%d", &t);
    while (t--)
    {
        scanf("%d%d", &n, &x);
        a = new ll[n + 2];
        sum[0] = 0;
        for (int i = 1; i <= n; i++)
        {
            scanf("%lld", a + i);
            sum[i] = sum[i - 1] + a[i];
        }
        build(1, 1, n);
        ll ans1 = get(1, 1, n, 1, n);
        int idx1 = 0, idx2 = 0;
        ll mx = 0, mn = 1e18;
        for (int i = 1; i <= n; i++)
        {
            ll beauty = get(1, 1, n, i + 1, n);
            beauty += sum[i - 1] % x == 0 ? sum[i - 1] / x : sum[i - 1] / x + 1;
            if (beauty > mx)
            {
                mx = beauty;
                idx1 = i;
                idx2 = i;
            }
            if (beauty < mn)
            {
                mn = beauty;
                idx2 = i;
            }
        }
        if (idx1 == idx2)
        {
            printf("%lld %lld\n", ans1, ans1);
            continue;
        }
        ll sum1 = sum[idx1 - 1], sum2 = sum[idx2 - 1];
        ll beauty = get(1, 1, n, idx2 + 1, n);
        beauty += sum1 % x == 0 ? sum1 / x : sum1 / x + 1;
        beauty += sum2 % x == 0 ? sum2 / x : sum2 / x + 1;
        printf("%lld %lld\n", ans1, beauty);
    }
    return 0;
}
