a = input()
b = input()
sa, sb = [int(i) for i in a], [int(i) for i in b]
cnt = [0] * 200005
sum = [0] * 200005
for i in range(len(b)):
    cnt[sa[i % len(a)]] += 1
    sum[i] = cnt[sa[0]]
    for j in range(len(a)):
        cnt[sa[j]] -= sb[i - j]
        sum[i] += cnt[sa[j]] * (j != len(a) - 1)
print(sum[-1])
