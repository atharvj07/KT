#include <bits/stdc++.h>
using namespace std;
int a[1000001], b[1000001];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            int x;
            cin >> x;
            a[i] ^= x;
            b[j] ^= x;
        }
    }
    int sum1 = 0, sum2 = 0;
    for (int i = 1; i <= n; ++i) sum1 += (a[i] % 4 != 0 ? 2 : 0);
    for (int j = 1; j <= m; ++j) sum2 += (b[j] % 4 != 0 ? 2 : 0);
    int ans = sum1 / 2 + sum2 % 2;
    if (ans == n + m)
        cout << "-1";
    else
        cout << ans << "\n";
    return 0;
}
