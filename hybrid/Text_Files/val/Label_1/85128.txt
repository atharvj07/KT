#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

struct info {
	int a, l, r;
	info() {}
	info(int a, int l, int r) : a(a), l(l), r(r) {}
	bool operator<(const info &b) const {
		return a < b.a;
	}
};

int main() {
	int n; scanf("%d", &n);
	vector<info> v(n);
	for (int i = 0; i < n; ++i) {
		int a; scanf("%d", &a);
		v[i] = info(a, a - 5, a + 5);
	}
	sort(v.begin(), v.end());
	int ans = 1;
	vector<info> stk;
	stk.push_back(v[0]);
	for (int i = 1; i < n; ++i) {
		if (v[i].l > stk.back().r) {
			stk.push_back(v[i]);
			++ans;
		}
		else stk.back().r = max(stk.back().r, v[i].r);
	}
	printf("%d\n", ans);
	return 0;
}
