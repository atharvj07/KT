import java.util.Scanner;

public class IgorKSecretArchive {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int a = scanner.nextInt();
        int b = scanner.nextInt();

        int rows = (int) Math.ceil((double) n / m);
        int startRow = (int) Math.ceil((double) a / m);
        int endRow = (int) Math.ceil((double) b / m);

        int selections = 0;

        // Select all folders in the rows containing the target folders
        for (int i = startRow; i <= endRow; i++) {
            selections++;
        }

        // If the target folders are not in the same row, select the remaining folders individually
        if (startRow != endRow) {
            int startCol = (a - 1) % m + 1;
            int endCol = (b - 1) % m + 1;

            selections += Math.max(startCol, m - endCol + 1);
        }

        System.out.println(selections);
    }
}
