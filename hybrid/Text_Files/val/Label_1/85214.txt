#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n, p, x, ans = 0, gcd = 0, mn = 1e9;
    cin >> n >> p;
    vector<int> a(n + 1), g[n + 1];
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i <= n; i++) {
        x = a[i];
        mn = min(mn, a[i]);
        for (int j = i - 1; j >= 1; j--) {
            x = __gcd(x, a[j]);
            if (x <= mn) {
                g[i].push_back(j);
                g[j].push_back(i);
                ans += x;
            }
            mn = min(mn, a[j]);
        }
        gcd = __gcd(gcd, a[i]);
    }
    if (gcd > 1) ans += gcd;
    if (n > 1) ans += p;
    mn = 1e9;
    for (int i = 1; i <= n; i++) {
        mn = min(mn, a[i]);
    }
    if (mn <= p) ans = mn;
    cout << ans << "\n";
}
int main() {
    int t;
    cin >> t;
    while (t--) solve();
    return 0;
}
