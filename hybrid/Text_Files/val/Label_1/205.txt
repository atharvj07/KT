n, m, k = map(int, input().split())
socks = list(map(int, input().split()))
l, r = [], []
for i in range(m):
    x, y = map(int, input().split())
    l.append(x)
    r.append(y)
d = {}
for i in range(n):
    if (l[i], r[i]) not in d:
        d[(l[i], r[i])] = [socks[l[i] - 1], socks[r[i] - 1]]
    else:
        d[(l[i], r[i])].append(socks[l[i] - 1])
        d[(l[i], r[i])].append(socks[r[i] - 1])
ans = 0
for key, value in d.items():
    value.sort()
    if len(set(value)) > 1:
        ans += len(set(value)) - 1
print(ans)
