#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> p(n + 1);
    for (int i = 1; i <= n; ++i) {
        cin >> p[i];
    }
    vector<int> cycle;
    vector<bool> visited(n + 1, false);
    for (int i = 1; i <= n; ++i) {
        if (!visited[i]) {
            int j = i;
            while (!visited[j]) {
                visited[j] = true;
                cycle.push_back(j);
                j = p[j];
            }
            if (cycle.size() > 2) {
                int m = cycle.size();
                cout << m * (m - 1) << endl;
                return 0;
            }
            cycle.clear();
        }
    }
    cout << n << endl;
    return 0;
}
