import sys

input = sys.stdin.readline

n, m = map(int, input().split())
a = list(map(int, input().split()))
freq = {}
dist = [n] * (n + 1)
for i in range(n - 1, -1, -1):
    if a[i] in freq:
        dist[i] = min(dist[i], i - freq[a[i]])
    freq[a[i]] = i
ans = []
for _ in range(m):
    l, r = map(int, input().split())
    ans.append(dist[l - 1] if dist[l - 1] <= r - l else -1)
print(*ans, sep="\n")
