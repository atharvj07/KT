import sys

t = int(input())

def get_cute(a, b, m):
    if (b - a + 1) % m != 1:
        return -1

    return (b - a + 1) // m + 1

for _ in range(t):
    a, b, m = map(int, input().split())
    result = get_cute(a, b, m)
    print(result if result == -1 else f'{result} ', *range(a, b + 1, result))
