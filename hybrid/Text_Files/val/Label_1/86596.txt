import java.util.*;
import java.lang.*;

class Main
{
    static class Princess {
        int p1;
        int p2;
        int w;
        public Princess(int p1, int p2, int w) {
            this.p1 = p1;
            this.p2 = p2;
            this.w = w;
        }
    }
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        List<Princess> princesses = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int p1 = sc.nextInt() - 1;
            int p2 = sc.nextInt() - 1;
            int w = sc.nextInt();
            Princess princess = new Princess(p1, p2, w);
            princesses.add(princess);
        }
        
        int[] dp = new int[n];
        for (Princess p : princesses) {
            dp[p.p1] = Math.max(dp[p.p1], p.w);
            dp[p.p2] = Math.max(dp[p.p2], p.w);
        }
        
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += dp[i];
        }

        System.out.println(sum);
    }
}
