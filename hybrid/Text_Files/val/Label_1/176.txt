from collections import deque, defaultdict

def bfs_maximize(graph, s):
    visited = set()
    queue = deque([s])
    visited.add(s)
    while queue:
        vertex = queue.popleft()
        for neighbor in graph[vertex]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)
    return visited

def bfs_minimize(graph, s, rev_graph):
    visited = set()
    queue = deque([s])
    visited.add(s)
    while queue:
        vertex = queue.popleft()
        for neighbor in graph[vertex]:
            if neighbor not in visited and vertex not in rev_graph[neighbor]:
                visited.add(neighbor)
                queue.append(neighbor)
    return visited

n, m, s = map(int, input().split())
directed_graph = defaultdict(list)
reverse_graph = defaultdict(list)
undirected_edges = []

for _ in range(m):
    t, u, v = map(int, input().split())
    if t == 1:
        directed_graph[u].append(v)
    else:
        undirected_edges.append((u, v))
        directed_graph[u].append(v)
        directed_graph[v].append(u)
        reverse_graph[v].append(u)

# Maximize
for u, v in undirected_edges:
    directed_graph[u].append(v)
reachable_max = bfs_maximize(directed_graph, s)
print(len(reachable_max))
print('+' * len(undirected_edges))

# Minimize
for u, v in undirected_edges:
    directed_graph[u].remove(v)
    directed_graph[v].remove(u)
    directed_graph[v].append(u)  # reverse the edge for minimize
reachable_min = bfs_minimize(directed_graph, s, reverse_graph)
print(len(reachable_min))
print('-' * len(undirected_edges))
