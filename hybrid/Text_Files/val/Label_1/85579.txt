#include<bits/stdc++.h>
using namespace std;

#define ll long long
#define INF 2e9
int main() {
    ll t; cin >> t;
    while (t--) {
        ll n; cin >> n;
        vector<ll> a(n);
        for (auto &x: a) cin >> x;
        sort(a.begin(), a.end());
        ll cnt = 0;
        for (auto &x: a) {
            if (x <= cnt + 1) ++cnt;
            else break;
        }
        cout << cnt + 1 << endl;
    }
    return 0;
}
