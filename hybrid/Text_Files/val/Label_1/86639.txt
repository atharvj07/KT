import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int r1 = sc.nextInt();
        int r2 = sc.nextInt();
        int r3 = sc.nextInt();
        int d = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }

        long[][] dp = new long[n][2];
        dp[0][0] = arr[0] * r1 + r3;
        dp[0][1] = arr[0] * r1 + r3 + d;

        for (int i = 1; i < n; i++) {
            dp[i][0] = Math.min(dp[i - 1][0] + arr[i] * r1, dp[i - 1][1] + r3);
            dp[i][1] = Math.min(dp[i][0] + r2, dp[i - 1][1] + r3 + d);
        }

        System.out.println(dp[n - 1][0]);
    }
}
