n, m = map(int, input().split())
s = ['.' for i in range(m)]
for i in range(n):
    t = input()
    for j in range(m):
        if s[j] == '.' and t[j] == '.':
            if i > 0 and s[j - 1] == 'B':
                s[j] = 'W'
            elif j > 0 and s[j - 1] == 'W':
                s[j] = 'B'
            else:
                s[j] = 'B'
print('\n'.join([''.join(s)] * n))
