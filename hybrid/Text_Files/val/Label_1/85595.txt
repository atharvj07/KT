#include <bits/stdc++.h>
using namespace std;

int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        int l, r, d, u, x, y, x1, y1, x2, y2;
        cin >> l >> r >> d >> u >> x >> y >> x1 >> y1 >> x2 >> y2;
        bool ans = false;
        if (x + l >= x1 || x + r <= x2) ans = true;
        if (y + u >= y1 || y + d <= y2) ans = true;
        cout << (ans ? "YES" : "NO") << endl;
    }
    return 0;
}
