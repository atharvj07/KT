#include <bits/stdc++.h>
using namespace std;

#define pb push_back

struct edge {
    int u, v;
    char c;
    edge(int u, int v, char c): u(u), v(v), c(c) {}
};

const int MAXN = 1e5 + 5;
vector<edge> g[MAXN];
int color[MAXN], comp[MAXN], m;
vector<int> ans;

void dfs(int u, int c) {
    comp[u] = c;
    color[c]++;
    for (edge e : g[u]) {
        if (comp[e.v] == -1)
            dfs(e.v, c ^ (e.c == 'R'));
    }
}

bool check() {
    for (int i = 1; i <= m; i++)
        if (color[i] % 2 != 0) {
            return false;
        }
    return true;
}

int main() {
    int n, u, v;
    char c;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        cin >> u >> v >> c;
        g[u].pb(edge(u, v, c));
        g[v].pb(edge(v, u, c));
    }
    memset(comp, -1, sizeof comp);
    memset(color, 0, sizeof color);
    int cpt = 0;
    for (int i = 1; i <= n; i++) {
        if (comp[i] == -1) {
            dfs(i, cpt++);
        }
    }
    if (!check()) {
        cout << "-1" << endl;
        return 0;
    }
    for (int i = 1; i <= n; i++) {
        for (edge e : g[i]) {
            if (comp[i] != comp[e.v]) {
                ans.pb(i);
                break;
            }
        }
    }

    cout << ans.size() << endl;
    for (int i : ans) {
        cout << i << " ";
    }

    return 0;
}
