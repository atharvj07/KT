#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] != 'F') {
      ans++;
      int j = i + 1;
      while (j < n && s[j] != 'F') {
        j++;
      }
      i = j - 1;
    }
  }
  cout << ans << endl;
  return 0;
}
