import java.util.*;
import java.io.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int ang = in.nextInt();
            int n = 180 / gcd(ang, 180);
            if (n % 2 == 1 && n * ang != 360) {
                n /= 2;
            }
            System.out.println(n);
        }
    }

    public static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
