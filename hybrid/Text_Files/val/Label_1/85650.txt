#include <iostream>

using namespace std;

long long mn[101], mx[101];

int main() {
    int m, s;
    cin >> m >> s;
    if (s > 9 * m || (s == 0 && m > 1)) {
        cout << "-1 -1";
        return 0;
    }
    for (int i = m - 1, t = s; i >= 0; i--) {
        mn[i] = t / 9;
        t %= 9;
        t += mn[i] == 0 && i > 0;
    }
    for (int i = m - 1, t = s; i >= 0; i--) {
        mx[i] = t % 10;
        t /= 10;
    }
    for (int i = 0; i < m; i++) {
        cout << mn[i];
    }
    cout << " ";
    for (int i = 0; i < m; i++) {
        cout << mx[i];
    }
    cout << endl;
    return 0;
}
