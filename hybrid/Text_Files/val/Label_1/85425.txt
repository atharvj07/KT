#include <iostream>

long long GCD(long long a, long long b) {
    if (b == 0) return a;
    return GCD(b, a % b);
}

long long f(long long x) {
    if (x % 2 == 0) return x;
    else return x - 1;
}

int main() {
    long long n, k;
    std::cin >> n >> k;
    if (k >= n) {
        if (n % 2 == 0) std::cout << n - 1;
        else std::cout << n;
    } else if (k == 1) {
        std::cout << 0;
    } else {
        long long g = GCD(n, k);
        if (g == 1) {
            if (k % 2 == 1) std::cout << n;
            else std::cout << n - 1;
        } else {
            n = n / g;
            k = k / g;
            long long ans = f(n);
            if (ans == n) {
                ans--;
                if (ans == 0) ans = 1;
            }
            std::cout << ans * g;
        }
    }
    return 0;
}
