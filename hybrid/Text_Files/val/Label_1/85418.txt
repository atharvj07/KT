#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int cnt = 0;
  vector<int> ans;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'B') {
      cnt++;
    } else {
      if (cnt) ans.push_back(cnt);
      cnt = 0;
    }
  }
  if (cnt) ans.push_back(cnt);
  cout << ans.size() << endl;
  for (int i = 0; i < ans.size(); i++) cout << ans[i] << " ";
  cout << endl;
  return 0;
}
