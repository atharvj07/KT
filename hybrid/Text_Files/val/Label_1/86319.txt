import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int[] k = new int[3];
        int[] t = new int[3];
        int n;
        for (int i = 0; i < 3; i++) {
            k[i] = input.nextInt();
        }
        for (int i = 0; i < 3; i++) {
            t[i] = input.nextInt();
        }
        n = input.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = input.nextInt();
        }
        TreeSet<Integer> s1 = new TreeSet<Integer>();
        TreeSet<Integer> s2 = new TreeSet<Integer>();
        TreeSet<Integer> s3 = new TreeSet<Integer>();
        int[] cnt = new int[3];
        int[] cur = new int[3];
        Arrays.fill(cur, 1);
        for (int i = 0; i < n; i++) {
            int min = Integer.MAX_VALUE;
            int id = 0;
            for (int j = 0; j < 3; j++) {
                if (cur[j] <= k[j] && t[j] <= min) {
                    min = t[j];
                    id = j;
                }
            }
            cnt[id]++;
            if (id == 0) {
                s1.add(cnt[0]);
            } else if (id == 1) {
                s2.add(cnt[1]);
            } else {
                s3.add(cnt[2]);
            }
            cur[id] += t[id];
        }
        int ans = 0;
        while (!s1.isEmpty() || !s2.isEmpty() || !s3.isEmpty()) {
            ans++;
            if (!s1.isEmpty()) {
                int temp = s1.pollFirst();
                cur[0]--;
                if (cur[0] == 0) {
                    s2.add(cnt[1]);
                }
            }
            if (!s2.isEmpty()) {
                int temp = s2.pollFirst();
                cur[1]--;
                if (cur[1] == 0) {
                    s3.add(cnt[2]);
                }
            }
            if (!s3.isEmpty()) {
                s3.pollFirst();
                cur[2]--;
                if (cur[2] == 0) {
                    cur[0] = 1;
                }
            }
        }
        System.out.println(ans);
    }
}
