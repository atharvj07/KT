n = int(input())
h = [int(x) for x in input().split()]
b = [int(x) for x in input().split()]
d = dict()
for i in range(n): d[h[i]] = b[i]
h.sort()
dp = [0] * n
dp[0] = d[h[0]]
for i in range(1, n):
    dp[i] = d[h[i]]
    for j in range(i):
        if h[j] < h[i]: dp[i] = max(dp[i], dp[j] + d[h[i]])
print(max(dp))
