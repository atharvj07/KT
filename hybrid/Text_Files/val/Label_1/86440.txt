import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    // Read vector A
    int x1 = sc.nextInt();
    int y1 = sc.nextInt();
    // Read vector B
    int x2 = sc.nextInt();
    int y2 = sc.nextInt();
    // Read vector C
    int x3 = sc.nextInt();
    int y3 = sc.nextInt();

    // Check if vector A can be transformed into vector B using the given operations
    if (x1 + y3 == x2 && y1 - x3 == y2) {
      System.out.println("YES");
    } else if (x1 - y3 == x2 && y1 + x3 == y2) {
      System.out.println("YES");
    } else {
      System.out.println("NO");
    }
  }
}
