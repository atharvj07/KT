#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) { cin >> a[i]; }
        unordered_map<int, vector<int>> mp;
        for (int i = 0; i < n; ++i) { mp[a[i]].push_back(i); }
        long long ans = 0;
        for (auto &p : mp) {
            vector<int> v = p.second;
            int m = v.size();
            for (int i = 0; i < m; ++i) {
                for (int j = i + 1; j < m; ++j) {
                    ans += (upper_bound(v.begin(), v.end(), j) - v.begin()) * (lower_bound(v.begin(), v.end(), i) - v.begin());
                }
            }
        }
        cout << ans << endl;
    }
}
