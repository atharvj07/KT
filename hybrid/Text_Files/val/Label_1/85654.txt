#include <bits/stdc++.h>

using namespace std;

int main() {
  int c1, c2, x, y;
  cin >> c1 >> c2 >> x >> y;
  if (x > y)
    swap(x, y);
  unordered_map<int, bool> mp;
  for (int i = 2; i * i <= y; ++i) {
    if (!mp[i]) {
      for (int j = i + i; j <= y; j += i)
        mp[j] = true;
    }
  }
  int ans = 0;
  for (int i = 2; ; ++i) {
    if (!mp[i] && !mp[i / x] && !mp[i / y] && ans < c1 + c2)
      ans++;
    else if (ans >= c1 + c2)
      break;
  }
  cout << ans << endl;
  return 0;
}
