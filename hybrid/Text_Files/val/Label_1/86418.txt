import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int m = sc.nextInt();
        char[] arr = s.toCharArray();
        while (m-- > 0) {
            int k = sc.nextInt();
            int d = sc.nextInt();
            int[] cnt = new int[d];
            int[][] pos = new int[d][k];
            for (int i = 0; i < k; i++) {
                pos[i % d][cnt[i % d]++] = i;
            }
            int[] newPos = new int[k];
            for (int i = 0; i < d; i++) {
                for (int j = 0; j < cnt[i]; j++) {
                    newPos[pos[i][j]] = i * k / d + j;
                }
            }
            char[] newArr = new char[k];
            for (int i = 0; i < k; i++) {
                newArr[i] = arr[newPos[i]];
            }
            System.arraycopy(newArr, 0, arr, 0, k);
            System.out.println(new String(arr));
        }
    }
}
