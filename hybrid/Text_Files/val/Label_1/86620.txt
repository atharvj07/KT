import java.util.*;
import java.io.*;

public class Main {
    static int MOD = 1000000007;
    static int N, K;
    static int[] A;
    static long res = 0;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        K = sc.nextInt();
        A = new int[K];
        for (int i = 0; i < K; i++) A[i] = sc.nextInt();
        dfs(0, 1, 1);
        System.out.println(res);
    }
    static void dfs(int pos, long prod, long fact) {
        if (pos == K) {
            res = (res + prod * fact) % MOD;
            return;
        }
        long f = fact;
        for (int i = 1; i <= A[pos] && i <= N - A[pos] + 1; i++) {
            f = (f * i) % MOD;
            dfs(pos + 1, prod * fastpow(N, i) % MOD, fact);
            fact = fact * fastpow(i, MOD - 2) % MOD;
        }
    }
    static long fastpow(long b, long e) {
        long res = 1;
        while (e > 0) {
            if ((e & 1) > 0) res = (res * b) % MOD;
            b = (b * b) % MOD;
            e >>= 1;
        }
        return res;
    }
}
