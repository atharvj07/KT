from collections import defaultdict
from math import gcd

for _ in range(int(input())):
  n, c = input().split()
  n = int(n)
  a = input()
  d = defaultdict(list)
  for i in range(n):
    d[a[i]].append(i+1)
  x = []
  if len(d) == 1:
    print(0)
  else:
    first = next(iter(d))
    for k in d:
      if k != first:
        g = gcd(d[first][0], d[k][0])
        x.append(g)
    print(len(x))
    print(*x)
