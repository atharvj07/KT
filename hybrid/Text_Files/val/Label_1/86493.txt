import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        Arrays.sort(arr);
        int height = 0, count = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] > height) {
                height++;
                count = 0;
            }
            count++;
        }
        System.out.println(height + " " + count);
    }
}
