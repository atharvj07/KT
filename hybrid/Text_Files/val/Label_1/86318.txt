import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] windows = new int[3];
        for (int i = 0; i < 3; i++) {
            windows[i] = sc.nextInt();
        }
        int[] times = new int[3];
        for (int i = 0; i < 3; i++) {
            times[i] = sc.nextInt();
        }
        int n = sc.nextInt();
        int[] arrivals = new int[n];
        for (int i = 0; i < n; i++) {
            arrivals[i] = sc.nextInt();
        }

        int[] currentWindow = new int[3];
        int[] nextFreeTime = new int[3];
        int maxTime = 0;
        for (int i = 0; i < n; i++) {
            int minFreeTime = Integer.MAX_VALUE;
            int minWindow = -1;
            for (int j = 0; j < 3; j++) {
                if (nextFreeTime[j] < minFreeTime) {
                    minFreeTime = nextFreeTime[j];
                    minWindow = j;
                }
            }
            maxTime = Math.max(maxTime, arrivals[i] - minFreeTime);
            nextFreeTime[minWindow] = arrivals[i] + times[minWindow];
            currentWindow[minWindow]++;
            if (currentWindow[minWindow] == windows[minWindow]) {
                currentWindow[minWindow] = 0;
            }
        }

        System.out.println(maxTime);
    }
}
