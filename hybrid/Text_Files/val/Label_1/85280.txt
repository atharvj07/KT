#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
#define vi(x) vector<x>
#define pb push_back
#define all(x) x.begin(), x.end()
#define rall(x) x.rbegin(), x.rend()

const int maxn = 4007;

int v[maxn], d[maxn], p[maxn];
int n;

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> v[i] >> d[i] >> p[i];
  vector<int> res;
  for (int i = 1, j = 1, cnt = n; i <= n; i++, j++) {
    p[j] -= v[i];
    if (p[j] < 0) {
      cout << cnt - 1 << '\n';
      for (int e : res) cout << e << ' ';
      cout << '\n';
      return 0;
    }
    while (j + 1 <= n && p[j + 1] >= 0) {
      p[j + 1] -= v[i];
      j++;
    }
    j--;
    res.pb(i);
    cnt--;
  }
  cout << n << '\n';
  for (int e : res) cout << e << ' ';
  cout << '\n';
  return 0;
}
