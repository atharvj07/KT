import sys

input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
q = int(input())
mp = {}

for _ in range(q):
    s, t, u = map(int, input().split())
    if (s, t) in mp:
        del mp[(s, t)]
    if u != 0:
        mp[(s, t)] = u
    cnt = [0] * n
    ans = 0
    for t in mp:
        if cnt[t[0]] >= t[1]:
            cnt[t[1]] += 1
        else:
            ans += 1
            cnt[t[0]] += 1
            cnt[t[1]] += 1
    for i in range(n):
        ans += max(0, a[i] - (cnt[i] + (i in mp and cnt[i] >= t[1])))
    print(ans)
