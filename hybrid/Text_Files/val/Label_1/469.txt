n = int(input())
d = {}
for i in range(n):
    a = input().split()
    l = []
    for j in range(int(a[1])):
        b = a[2+j]
        if not b in d: d[b] = [a[0]]
        elif b not in d[a[0]]: d[b].append(a[0])
        l.append(b)
    for j in range(len(l)):
        for k in range(len(l)):
            if j != k and l[j].endswith(l[k]):
                if a[0] in d[l[k]]: d[l[k]].remove(a[0])
m = 0
for i in d:
    if len(d[i]) > 0:
        m += 1
        print(i, len(d[i]), *d[i])
print(m)
