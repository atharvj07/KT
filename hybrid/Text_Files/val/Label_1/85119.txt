#include<bits/stdc++.h>
using namespace std;
const int N=3e5;
int a[N],q[N],l=1,r;
long long pre[N],suff[N],dp[N],ans;
int main(){
    int n,m,k; cin>>n>>m>>k;
    for(int i=1;i<=n;i++) cin>>a[i],pre[i]=pre[i-1]+a[i],suff[n-i+1]=suff[n-i]+a[n-i+1];
    for(int i=1;i<=n;i++){
        while(l<=r&&q[l]<=i-m) l++;
        if(l<=r) dp[i]=max(dp[i],dp[q[l]]+pre[i]-pre[q[l]]);
        while(l<=r&&suff[r+1]>=suff[i+1]) r--;
        q[++r]=i;
    }
    for(int i=1;i<=n;i++) ans=max(ans,dp[i]+suff[i+1]-k*((i+m-1)/m));
    cout<<ans<<endl;
}
