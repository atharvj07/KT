#include <bits/stdc++.h>
#define ll long long
using namespace std;
const ll mod = 1000000007;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  ll n;
  cin >> n;
  vector<ll> a(n);
  for (ll i = 0; i < n; i++) cin >> a[i];
  ll sum = 0;
  vector<ll> pref(n);
  pref[0] = a[0];
  for (ll i = 1; i < n; i++) pref[i] = pref[i - 1] + a[i];
  map<ll, ll> cnt;
  ll ans = 0;
  for (ll i = 0; i < n; i++) {
    ans += (sum - (pref[i] - a[i]) + mod) % mod;
    sum += a[i];
    ans %= mod;
    cnt[a[i]]++;
    if (cnt[a[i]] == 2) ans -= 2, ans %= mod;
  }
  cout << ans << endl;
  return 0;
}
