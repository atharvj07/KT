n = int(input())
s = input()
L = ['R', 'G', 'B']
cnt = 0
need = s[0]
ans = []
for i in range(n):
    if s[i] != need:
        ans.append(s[i])
        if cnt < 2:
            cnt += 1
        else:
            need = L[(L.index(need) + 1) % 3]
            cnt = 0
    ans.append(s[i])
print(len(ans) - n)
print(''.join(ans))
