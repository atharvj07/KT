import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        List<Book> books = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int t = in.nextInt();
            int a = in.nextInt();
            int b = in.nextInt();
            books.add(new Book(t, a, b, i));
        }
        Collections.sort(books);
        int ans = -1;
        int cntA = 0;
        int cntB = 0;
        int sum = 0;
        for (Book book : books) {
            sum += book.t;
            if (book.a == 1) {
                cntA++;
            }
            if (book.b == 1) {
                cntB++;
            }
            if (cntA >= k && cntB >= k) {
                ans = sum;
                break;
            }
        }
        System.out.println(ans);
    }

    static class Book implements Comparable<Book> {
        int t;
        int a;
        int b;
        int id;

        public Book(int t, int a, int b, int id) {
            this.t = t;
            this.a = a;
            this.b = b;
            this.id = id;
        }

        @Override
        public int compareTo(Book o) {
            return this.id - o.id;
        }
    }
}
