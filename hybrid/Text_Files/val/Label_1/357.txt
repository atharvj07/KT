x = input()

a = [list(k) for k in ['qwertyuiop', 'asdfghjkl;', 'zxcvbnm,./']]

c = [list(k) for k in ['erttyuiqw', 'rdfghjasd', 'zcxvbnm,k']]

    d = b
else:
    d = c
