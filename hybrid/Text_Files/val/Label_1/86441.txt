import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] a = new int[2];
        int[] b = new int[2];
        int[] c = new int[2];
        for (int i = 0; i < 2; i++) {
            a[i] = scanner.nextInt();
            b[i] = scanner.nextInt();
            c[i] = scanner.nextInt();
        }
        if (a[0] + c[0] == b[0] && a[1] + c[1] == b[1] || a[0] - c[1] == b[0] && a[1] + c[0] == b[1] ||
                a[0] - c[0] == b[0] && a[1] - c[1] == b[1] || a[0] + c[1] == b[0] && a[1] - c[0] == b[1]) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
