#include <bits/stdc++.h>
using namespace std;
#define nl '\n'
typedef long long ll;
typedef long double ld;

const ll MOD = 1e9 + 7;

int main() {
    ll n;
    cin >> n;
    vector<ll> a(n);
    for (ll i = 0; i < n; i++)
        cin >> a[i];
    sort(a.begin(), a.end());
    ll l = 0, r = 0, ans = 0;
    while (r < n) {
        if (a[r] - a[l] <= 5) {
            ans = max(ans, r - l + 1);
            r++;
        } else {
            l++;
        }
    }
    cout << ans << nl;
    return 0;
}
