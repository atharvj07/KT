n, t = map(int, input().split())
g = input()
s = [0] * (n + 2)
for i in range(n - 1, -1, -1):
    s[i] = s[i + 1] + (ord(g[i]) - ord('0')) * (10 ** (n - i - 1))
cur = 0
while t and cur < n - 1:
    cur += 1
    t -= 1
    nxt = (s[cur + 1] + 5) * (10 ** (n - cur - 1))
    if nxt > s[cur]:
        g = g[:cur] + str((ord(g[cur]) - ord('0')) + 1) + '0' * (n - cur)
        cur = -1
print(g.rstrip('0'))
