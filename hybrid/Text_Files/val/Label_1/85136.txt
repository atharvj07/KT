#include <bits/stdc++.h>
#define MXSZ 2000
using namespace std;
typedef long long ll;

int n, m, a[MXSZ + 5], b[MXSZ + 5], c[MXSZ + 5], ans, x;
ll ms[MXSZ + 5][MXSZ + 5];

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);
    cin >> n >> m;
    for (int i = 1; i <= n; ++i) cin >> a[i];
    for (int j = 1; j <= m; ++j) cin >> b[j];
    cin >> x;
    for (int i = 1; i <= n; ++i)
        for (int j = 1; j <= m; ++j) c[i * j] += a[i] * b[j];
    sort(c + 1, c + n * m + 1);
    for (int i = 1; i <= n; ++i)
        for (int j = 1; j <= m; ++j) ms[i][j] = ms[i - 1][j] + ms[i][j - 1] - ms[i - 1][j - 1] + c[(i - 1) * m + j];
    for (int l1 = 1; l1 <= n; ++l1)
        for (int l2 = l1; l2 <= n; ++l2)
            for (int r1 = 1; r1 <= m; ++r1)
                for (int r2 = r1; r2 <= m; ++r2) {
                    ll sum = ms[l2][r2] - ms[l2][r1 - 1] - ms[l1 - 1][r2] + ms[l1 - 1][r1 - 1];
                    if (sum <= x) ans = max(ans, (l2 - l1 + 1) * (r2 - r1 + 1));
                }
    cout << ans << '\n';
    return 0;
}
