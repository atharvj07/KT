#include <iostream>

using namespace std;

int main() {
  long long n;
  cin >> n;
  long long l = 0, r = 3e9, ans = -1;
  while (l <= r) {
    long long mid = (l + r) / 2;
    long long sum = 0, x = mid;
    while (x) {
      sum += x % 10;
      x /= 10;
    }
    if (mid * mid + sum * mid - n == 0) {
      ans = mid;
      r = mid - 1;
    } else if (mid * mid + sum * mid - n < 0) {
      l = mid + 1;
    } else {
      r = mid - 1;
    }
  }
  cout << ans << endl;
}
