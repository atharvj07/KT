#include <bits/stdc++.h>

#define ll long long

using namespace std;

const int MOD = 1e9 + 7;

int main() {
    ll n, k, m;
    cin >> n >> k >> m;
    ll ans = 0;
    for (ll i = 1; i <= k; i++) {
        ll left = 0, right = 0, sum = 1;
        for (ll j = 1; j < n; j++) {
            (left *= 10) %= m;
            (right *= 10) %= m;
            (left += i) %= m;
            (right += i * (k / i)) % m;
            (sum *= 10) %= m;
        }
        (left *= i) %= m;
        (right *= i * (k / i)) % m;
        ans += (right - left + m) * sum % m;
        ans %= m;
    }
    cout << ans << endl;
    return 0;
}
