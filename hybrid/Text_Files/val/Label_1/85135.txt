#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < (n); i++)
#define all(x) (x).begin(), (x).end()
using namespace std;
using ll = long long;
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n), b(m);
  rep(i, n) cin >> a[i];
  rep(i, m) cin >> b[i];
  ll x;
  cin >> x;
  vector<vector<ll>> dp(n + 1, vector<ll>(m + 1, LLONG_MAX));
  dp[0][0] = 0;
  rep(i, n) rep(j, m) {
    dp[i + 1][j] = min(dp[i + 1][j], dp[i][j] + a[i] * b[j]);
    dp[i][j + 1] = min(dp[i][j + 1], dp[i][j] + a[i] * b[j]);
  }
  ll ans = 0;
  rep(i, n + 1) rep(j, m + 1) {
    if (dp[i][j] <= x) ans = max(ans, (ll)i * j);
  }
  cout << ans << endl;
}
