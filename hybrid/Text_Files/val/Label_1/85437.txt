#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (int)(n); i++)
int main() {
  long long n, t;
  cin >> n;
  priority_queue<long long, vector<long long>, greater<long long>> pq;
  rep(i, n) {
    cin >> t;
    pq.push(t);
  }
  while (pq.size() > 1) {
    long long a = pq.top();
    pq.pop();
    long long b = pq.top();
    pq.pop();
    long long c = a & b;
    long long d = a | b;
    pq.push(c * c);
    pq.push(d * d);
  }
  cout << pq.top() << endl;
}
