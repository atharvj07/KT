import java.util.*;
import java.io.*;

public class Main {

    static int[] a;
    static int[] lastSetBit;

    public static void solve() throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] _in = in.readLine().split(" ");
        int n = Integer.parseInt(_in[0]), q = Integer.parseInt(_in[1]);
        a = new int[n];
        lastSetBit = new int[n];

        int[] freq = new int[300000 + 1];

        _in = in.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(_in[i]);
            freq[a[i]]++;
            if (a[i] > 0) {
                lastSetBit[i] = getLastSetBit(a[i]);
            }
        }

        for (int x : freq) {
            if (x > 1) {
                System.out.println("Shi");
                return;
            }
        }

        for (int i = 0; i < q; i++) {
            _in = in.readLine().split(" ");
            int x = Integer.parseInt(_in[0]), y = Integer.parseInt(_in[1]);
            if (check(x, y)) {
                System.out.println("Shi");
            } else {
                System.out.println("Fou");
            }
        }
    }

    static boolean check(int x, int y) {
        y--;
        return (a[x - 1] | a[y]) > 0 && checkSetBit(x, y);
    }

    static boolean checkSetBit(int x, int y) {
        for (int i = x; i < y; i++) {
            if (lastSetBit[i] == lastSetBit[x] && ((a[i] | a[x]) == a[x])) {
                return true;
            }
        }
        return false;
    }

    static int getLastSetBit(int x) {
        int i = 0;
        while ((x & (1 << i)) == 0) {
            i++;
        }
        return i;
    }

    public static void main(String[] args) throws IOException {
        solve();
    }
}
