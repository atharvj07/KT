for _ in range(int(input())):
    a, b = map(int, input().split())
    x = 1 + a + b
    ans = 0
    for i in range(2, int(b ** 0.5) + 1):
        if b % i == 0:
            if x % (i + b // i) == 0:
                ans += 1
            if x % (b // i + i) == 0:
                ans += 1
    if x % b == 0:
        ans += 1
    print(ans)
