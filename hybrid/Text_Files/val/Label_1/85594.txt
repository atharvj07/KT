#include <iostream>
using namespace std;
typedef long long int ll;
int main() {
  ll t;
  cin >> t;
  while (t--) {
    ll a, b, c, d, x, y, x1, y1, x2, y2;
    cin >> a >> b >> c >> d >> x >> y >> x1 >> y1 >> x2 >> y2;
    bool flag = false;
    if (y1 <= y - c && y2 >= y + d && x1 <= x - a && x2 >= x + b)
      flag = true;
    cout << (flag ? "YES" : "NO") << endl;
  }
  return 0;
}
