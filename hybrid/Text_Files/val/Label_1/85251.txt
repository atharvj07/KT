#include<bits/stdc++.h>
using namespace std;
#define ios ios_base::sync_with_stdio(false);cin.tie(NULL);
using ll=long long;
int gcd(int a, int b) { return b == 0 ? a : gcd(b, a % b); }
int main() { 
	int n; cin >> n;
	int o = n / 2;
	if(o == 1) {
		cout << 0 << '\n';
	}
	else {
		cout << o - (n % 2 == 0) << '\n';
		int k = 0;
		for(int i = 1; i <= n; i++) {
			for(int j = i + 1; j <= n; j++) {
				if(gcd(i, j) > 1) {
					cout << i << ' ' << j << '\n';
					k++;
					if(k == o - (n % 2 == 0)) break;
				}
			}
		}
	}
}
