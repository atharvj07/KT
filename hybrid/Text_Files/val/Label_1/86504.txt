import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextInt();
        }
        Arrays.sort(arr);
        int max = 0;
        int cnt = 0;
        int last = -1;
        for (int i = 0; i < n; i++) {
            if (arr[i] == last) {
                cnt++;
            } else {
                last = arr[i];
                cnt = 1;
            }
            max = Math.max(max, cnt);
        }
        System.out.println(arr[n - max] + " " + max);
    }
}
