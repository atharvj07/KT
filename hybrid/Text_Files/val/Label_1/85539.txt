#include <bits/stdc++.h>
using namespace std;
int main(){
    ios::sync_with_stdio(0);cin.tie(0);
    long long t, n, a[200001];
    cin>>t;
    while(t--){
        cin>>n;
        for(long long i=0;i<n;++i) cin>>a[i];
        sort(a, a+n);
        long long l=a[0], r=a[0];
        for(long long i=1;i<n;++i){
            l=min(l, a[i]);
            r=max(r, a[i]);
        }
        for(long long i=0;i<n;++i){
            if(l+i!=a[i] or r-i!=a[i]) {cout<<"NO\n"; goto nxt;}
        }
        cout<<"YES\n";nxt:;
    }
}
