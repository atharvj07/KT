#include <iostream>
#include <queue>
#include <set>
#include <vector>

using namespace std;

const int MAXN = 1e5 + 10;

struct node {
  int val, freq;
  bool operator<(const node& a) const { return val > a.val; }
};

int n;
vector<pair<int, int>> events;

int main() {
  cin >> n;
  priority_queue<int> pq;
  set<int> s;
  for (int i = 0; i < n; i++) {
    string type;
    cin >> type;
    if (type == "insert") {
      int x;
      cin >> x;
      events.push_back({1, x});
      pq.push(x);
      s.insert(x);
    } else if (type == "getMin") {
      int x;
      cin >> x;
      if (pq.empty())
        events.push_back({-1, 0});
      else if (pq.top() != x) {
        events.push_back({-1, x});
        pq.pop();
        while (!pq.empty() && s.count(pq.top()) == 0) pq.pop();
      }
    } else if (type == "removeMin") {
      if (pq.empty())
        events.push_back({-1, 0});
      else {
        int x = pq.top();
        pq.pop();
        s.erase(x);
      }
    }
  }
  int ans = n;
  for (auto& event : events) {
    if (event.first == -1) ans++;
    else {
      if (event.first == 1) s.insert(event.second);
      else
        while (!pq.empty() && s.count(pq.top()) == 0) {
          ans++;
          pq.pop();
        }
    }
  }
  cout << ans << endl;
  for (auto& event : events) {
    if (event.first == -1)
      cout << "removeMin" << endl;
    else if (event.first == 1)
      cout << "insert " << event.second << endl;
    else
      cout << "getMin " << pq.top() << endl;
  }
  return 0;
}
