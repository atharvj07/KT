n = int(input())
a = list(map(int, input().split()))
q = int(input())

# Initialize milestones and resource production
milestones = [{} for _ in range(n)]
production = [0] * n

for _ in range(q):
    s, t, u = map(int, input().split())
    s -= 1
    if u > 0:
        u -= 1
    # Remove existing milestone if it exists
    if t in milestones[s]:
        prev_u = milestones[s][t]
        if prev_u != -1:  # Check if it's not a removed milestone
            production[prev_u] -= 1
        del milestones[s][t]
    # Add new milestone if u is not 0
    if u != -1:
        milestones[s][t] = u
        production[u] += 1
    else:  # Mark the milestone for removal
        milestones[s][t] = -1

    # Calculate minimum turns
    turns = 0
    for i in range(n):
        # Calculate how many of resource i needs to be produced
        needed = a[i] - production[i]
        if needed < 0:
            needed = 0
        turns += needed
    print(turns)
