import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        boolean[][] adj = new boolean[N][N];
        boolean[][] visited = new boolean[N][1 << N];
        int[][] dp = new int[N][1 << N];
        for (int i = 0; i < M; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            adj[a][b] = true;
        }
        int mod = 1000000007;
        for (int mask = 1; mask < (1 << N); mask++) {
            for (int i = 0; i < N; i++) {
                if ((mask & (1 << i)) == 0) {
                    for (int j = 0; j < N; j++) {
                        if (adj[i][j] && (mask & (1 << j)) != 0) {
                            visited[i][mask] = true;
                            if (dp[j][mask] == 0) {
                                dp[j][mask] = dfs(j, mask, adj, visited, dp, mod);
                            }
                            dp[i][mask] += dp[j][mask];
                            dp[i][mask] %= mod;
                        }
                    }
                }
            }
        }
        System.out.println(dp[0][(1 << N) - 1]);
    }

    public static int dfs(int i, int mask, boolean[][] adj, boolean[][] visited, int[][] dp, int mod) {
        if (visited[i][mask]) {
            return 0;
        }
        visited[i][mask] = true;
        int ret = 1;
        for (int j = 0; j < adj.length; j++) {
            if (adj[i][j] && (mask & (1 << j)) != 0) {
                if (dp[j][mask] == 0) {
                    dp[j][mask] = dfs(j, mask, adj, visited, dp, mod);
                }
                ret = (ret + dp[j][mask]) % mod;
            }
        }
        return ret;
    }
}
