#include <iostream>
#include <vector>

int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<int> a(m);
  for (int &x : a) std::cin >> x;

  std::vector<int> cnt(n + 1);
  for (int x : a) ++cnt[x];

  for (int i = 0; i < m; ++i) {
    --cnt[a[i]];
    int ok = 1;
    for (int j = 1; j <= n; ++j) if (cnt[j]) ok = 0;
    std::cout << ok;
  }

  return 0;
}
