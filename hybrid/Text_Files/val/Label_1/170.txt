n, m, x, y, z, p = map(int, input().split())
for _ in range(p):
    a, b = map(int, input().split())
    x = x % 4
    y = y % 2
    z = z % 4
    if z == 1:
        a, b = b, n - a + 1
    elif z == 2:
        a, b = n - a + 1, m - b + 1
    elif z == 3:
        a, b = m - b + 1, a
    if x % 2 != 0:
        a, b = b, a
    if y != 0:
        b = m - b + 1
    print(a, b)
