#include <iostream>

using namespace std;

typedef long long ll;

const int N = 1e5 + 10;

int n, k;
ll a[N], x;

int main() {
    cin >> n >> x >> k;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        a[i] = (a[i] + x - 1) / x;
    }

    ll ans = 0;
    for (int i = 1; i <= n; i++) {
        int cnt = 0;
        for (int j = i + 1; j <= n; j++) {
            if (a[i] < a[j] || (a[i] == a[j] && k == 0))
                cnt++;
            if (a[i] <= a[j] && a[i] + k >= a[j])
                ans += cnt;
        }
    }
    cout << ans << endl;
    return 0;
}
