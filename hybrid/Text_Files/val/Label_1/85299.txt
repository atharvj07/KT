#include <iostream>
#include <vector>
#include <queue>

using namespace std;

typedef long long ll;
typedef vector<ll> vll;
typedef pair<ll, ll> pll;

const ll mod = 1e9 + 7;

ll n;
vll adj[100001];
vll sz(100001, 0);

void dfs1(ll u, ll p) {
  sz[u] = 1;
  for (auto v : adj[u]) {
    if (v == p) continue;
    dfs1(v, u);
    sz[u] += sz[v];
  }
}

ll dfs2(ll u, ll p) {
  for (auto v : adj[u]) {
    if (v == p) continue;
    if (sz[v] > n / 2) return dfs2(v, u);
  }
  return u;
}

int main() {
  ll t;
  cin >> t;
  while (t--) {
    ll x, y;
    cin >> n;
    for (ll i = 0; i <= n; i++) {
      adj[i].clear();
      sz[i] = 0;
    }
    for (ll i = 1; i < n; i++) {
      cin >> x >> y;
      adj[x].push_back(y);
      adj[y].push_back(x);
    }
    dfs1(1, 1);
    ll centroid = dfs2(1, 1);
    for (auto v : adj[centroid]) {
      cout << centroid << " " << v << "\n";
      cout << centroid << " " << v << "\n";
      break;
    }
  }
  return 0;
}
