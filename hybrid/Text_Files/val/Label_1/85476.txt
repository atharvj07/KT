#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, d, b;
    cin >> n >> d >> b;
    vector<int> a(n + 2);
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }
    int ans = 0;
    for (int i = d + 1; i <= n - d; ++i) {
        if (a[i] > b) {
            if (a[i - d] + a[i + d] >= b) {
                ++ans;
                a[i - d] += a[i] - b;
                a[i + d] += a[i] - b;
            } else {
                ans += 2;
            }
            a[i] = b;
        }
    }
    cout << ans << endl;
    return 0;
}
