#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, x[1001], y[1001], a[1001], b[1001];
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> x[i] >> y[i];
  }
  for (int i = 0; i < n; ++i) {
    cin >> a[i] >> b[i];
  }
  long long sumx = 0, sumy = 0;
  for (int i = 0; i < n; ++i) {
    sumx += (x[i] + a[i]);
    sumy += (y[i] + b[i]);
  }
  cout << sumx / n << ' ' << sumy / n << endl;
}
