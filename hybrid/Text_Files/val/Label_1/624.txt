def can_be_erased(s):
    for i in range(len(s)):
        if s[i] == 'A':
            for j in range(len(s)):
                if s[j] == 'B':
                    return True
            break
    for i in range(len(s)):
        if s[i] == 'B':
            for j in range(len(s)):
                if s[j] == 'C':
                    return True
            break
    return False

for _ in range(int(input())):
    print('YES' if can_be_erased(input()) else 'NO')
