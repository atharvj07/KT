import java.util.*;
import java.io.*;

public class Main {
    static long a, b, min;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        a = sc.nextLong();
        b = sc.nextLong();
        min = gcd(a, b);
        if (a < b) {
            a /= min;
            b /= min;
            if (a < 2) {
                System.out.println("Masha");
            } else {
                System.out.println("Dasha");
            }
        } else if (a > b) {
            a /= min;
            b /= min;
            if (b < 2) {
                System.out.println("Dasha");
            } else {
                System.out.println("Masha");
            }
        } else {
            System.out.println("Equal");
        }
    }

    static long gcd(long a, long b) {
        while (b > 0) {
            long temp = a;
            a = b;
            b = temp % b;
        }
        return a;
    }
}
