t = int(input())
for _ in range(t):
    n = int(input())
    c = list(map(int, input().split()))
    min_odd = c[0]
    min_even = c[1]
    total_cost = c[0] * n + c[1] * n
    min_cost = total_cost
    odd_count = 1
    even_count = 1
    for i in range(2, n):
        if i % 2 == 0:
            odd_count += 1
            min_odd = min(min_odd, c[i])
        else:
            even_count += 1
            min_even = min(min_even, c[i])
        total_cost += c[i]
        current_cost = total_cost + (n - odd_count) * min_odd + (n - even_count) * min_even
        min_cost = min(min_cost, current_cost)
    print(min_cost)
