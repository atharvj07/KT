import java.util.*;
class Main{
    static int n, m;
    static boolean[] used;
    static boolean[] cut;
    static int[] tin;
    static int[] fup;
    static int timer = 0;
    static List<Integer>[] g;
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        g = new List[n];
        for(int i = 0; i < n; i++) g[i] = new ArrayList<>();
        for(int i = 0; i < m; i++){
            int u = sc.nextInt();
            int v = sc.nextInt();
            g[u].add(v);
            g[v].add(u);
        }
        used = new boolean[n];
        cut = new boolean[n];
        tin = new int[n];
        fup = new int[n];
        dfs(0, -1);
        for(int v = 0; v < n; v++){
            if(cut[v]) System.out.println(v);
        }
    }
    static void dfs(int v, int p){
        used[v] = true;
        tin[v] = fup[v] = timer++;
        int children = 0;
        for(int u : g[v]){
            if(!used[u]){
                dfs(u, v);
                fup[v] = Math.min(fup[v], fup[u]);
                if(fup[u] >= tin[v] && p != -1) cut[v] = true;
                children++;
            } else if(u != p){
                fup[v] = Math.min(fup[v], tin[u]);
            }
        }
        if(p == -1 && children > 1) cut[v] = true;
    }
}
