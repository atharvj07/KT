#include <bits/stdc++.h>
#define MX 100001
using namespace std;
vector<int>g[MX], l;
int par[MX], dep[MX], sub[MX], ipar[MX];
bool vis[MX];

int dfs1(int u, int p)
{
    vis[u] = true;
    par[u] = p;
    for(int v: g[u]) {
        if(vis[v]) continue;
        dep[v] = dep[u] + 1;
        int ans = dfs1(v, u);
        sub[u] += ans;
    }
    return sub[u] + 1;
}

void dfs2(int u, int p, int anc)
{
    ipar[u] = anc;
    for(int v: g[u]) {
        if(v == p) continue;
        dfs2(v, u, anc);
    }
}

int main()
{
    int n, m;
    cin >> n >> m;
    while(m--) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
    }

    dfs1(1, 0);
    dfs2(1, 0, 1);

    int ans = 0;
    for(int i = 1; i <= n; i++) {
        int x = i;
        while(x != 1) {
            if(!vis[x]) break;
            x = ipar[x];
        }
        if(x == 1) continue;
        vis[x] = false;
        ans++;
        l.push_back(x);
    }

    if(ans == 0) cout << -1 << "\n";
    else {
        cout << ans << "\n";
        for(int x: l) cout << x << "\n";
    }
    return 0;
}
