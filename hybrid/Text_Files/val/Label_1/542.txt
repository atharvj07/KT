import bisect
n, k = map(int, input().split())
b = [int(x) for x in input().split()]
c = [int(x) for x in input().split()]
x = sorted([a-b[i] for a, i in zip(c, range(n))])
ans = [0]*n
for j in range(k, len(x)): del x[j-k]
for i in range(n): t = bisect.bisect_left(x, -b[i]); ans[i] = c[i]+0 if t == 0 else x[t-1]
print(*ans)
