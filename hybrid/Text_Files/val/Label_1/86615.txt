import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            char[][] grid = new char[n][m];
            for (int i = 0; i < n; i++) {
                String s = sc.next();
                for (int j = 0; j < m; j++) {
                    grid[i][j] = s.charAt(j);
                }
            }
            boolean[][] visited = new boolean[n][m];
            int k = 0;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (!visited[i][j] && grid[i][j] != '.') {
                        k++;
                        dfs(grid, visited, i, j, k);
                    }
                }
            }
            if (k <= 26) {
                System.out.println("YES");
                System.out.println(k);
                for (int i = 0; i < n; i++) {
                    for (int j = 0; j < m; j++) {
                        if (grid[i][j] != '.') {
                            System.out.println((i + 1) + " " + (j + 1) + " " + (i + 1) + " " + (j + 1));
                        }
                    }
                }
            } else {
                System.out.println("NO");
            }
        }
    }

    public static void dfs(char[][] grid, boolean[][] visited, int i, int j, int k) {
        if (i < 0 || i >= grid.length || j < 0 || j >= grid[0].length || visited[i][j] || grid[i][j] == '.') {
            return;
        }
        visited[i][j] = true;
        grid[i][j] = (char) ('a' + k - 1);
        dfs(grid, visited, i + 1, j, k);
        dfs(grid, visited, i - 1, j, k);
        dfs(grid, visited, i, j + 1, k);
        dfs(grid, visited, i, j - 1, k);
    }
}
