n=int(input())
s=input()
if n<=2:
  print(0,s)
  exit()
ans=[]
for i in range(n):
  if i==0:
    c=s[i]
    ans.append(c)
  elif s[i]!=ans[-1]:
    c=s[i]
    ans.append(c)
  else:
    j=1
    while s[i]==s[(i-j)%n]:
      j+=1
    if (i-j)%n==0:
      j+=1
    c=s[(i-j)%n]
    ans.append(c)
print(n-len(ans))
print(''.join(ans))
