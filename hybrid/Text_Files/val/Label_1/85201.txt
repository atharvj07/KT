#include<iostream>
#include<vector>
using namespace std;
int n,m,fi,fj,ans_i,ans_j;
char a[116][116];

void dfs(int i,int j){
    if(a[i][j]=='W' || (i<1 || i>n || j<1 || j>m)) return;
    a[i][j]='W';
    dfs(i+1,j);
    dfs(i-1,j);
    dfs(i,j+1);
    dfs(i,j-1);
    if(fi==0){
        fi=i;
        fj=j;
    }else{
        ans_i=(i+fi)/2;
        ans_j=(j+fj)/2;
    }
}

int main(){
    cin>>n>>m;
    for(int i=1;i<=n;i++){
        cin>>a[i]+1;
    }
    for(int i=1;i<=n;i++){
        for(int j=1;j<=m;j++){
            if(a[i][j]=='B'){
                fi=0;
                dfs(i,j);
                cout<<ans_i<<" "<<ans_j<<endl;
            }
        }
    }
    return 0;
}
