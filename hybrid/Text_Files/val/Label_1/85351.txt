#include <iostream>

using namespace std;

int64_t solve() {
    int64_t n, x, y, d;
    cin >> n >> x >> y >> d;
    if (x > y) swap(x, y);
    if ((y - x) % d == 0) return (y - x) / d;
    if ((n - y + x) % d == 0) return (n - y + x) / d;
    return -1;
}

int main() {
    int t;
    cin >> t;
    while (t--) cout << solve() << '\n';
    return 0;
}
