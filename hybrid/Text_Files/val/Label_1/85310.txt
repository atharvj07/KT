#include <bits/stdc++.h>
using namespace std;

pair<int, string> solve(int a, int b) {
  // dp[i][j] = max score for using the first i 'o's and j 'x's
  vector<vector<int>> dp(a + 1, vector<int>(b + 1, -1e9));
  dp[0][0] = 0;
  for (int i = 0; i <= a; i++) {
    for (int j = 0; j <= b; j++) {
      // add an 'o'
      if (i > 0) dp[i][j] = max(dp[i][j], dp[i - 1][j] + (i * i));
      // add an 'x'
      if (j > 0) dp[i][j] = max(dp[i][j], dp[i][j - 1] - (j * j));
    }
  }
  int i = a, j = b;
  string s;
  while (i > 0 || j > 0) {
    if (i > 0 && dp[i][j] == dp[i - 1][j] + (i * i)) {
      s += 'o';
      i--;
    } else {
      s += 'x';
      j--;
    }
  }
  reverse(s.begin(), s.end());
  return {dp[a][b], s};
}

int main() {
  int a, b;
  cin >> a >> b;
  auto [score, s] = solve(a, b);
  cout << score << endl;
  cout << s << endl;
  return 0;
}
