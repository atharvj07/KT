import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {

    private static List<Integer> articulationPoints;
    private static int root, n, m, timer;
    private static boolean[] visited;
    private static int[] tin, up;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        root = 0;

        Graph graph = new Graph(n);
        visited = new boolean[n];
        tin = new int[n];
        up = new int[n];
        articulationPoints = new ArrayList<>();

        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int from = Integer.parseInt(st.nextToken());
            int to = Integer.parseInt(st.nextToken());
            graph.addEdge(from, to);
            graph.addEdge(to, from);
        }

        dfs(graph, root, -1);
        Collections.sort(articulationPoints);
        for (int articulationPoint : articulationPoints) {
            System.out.println(articulationPoint);
        }
    }

    private static void dfs(Graph graph, int current, int parent) {
        visited[current] = true;
        tin[current] = up[current] = timer++;
        int childrenCount = 0;
        for (int neighbor : graph.getNeighbors(current)) {
            if (!visited[neighbor]) {
                dfs(graph, neighbor, current);
                up[current] = Math.min(up[current], up[neighbor]);
                if (current != root && up[neighbor] >= tin[current]) {
                    articulationPoints.add(current);
                }
                childrenCount++;
            } else if (neighbor != parent) {
                up[current] = Math.min(up[current], tin[neighbor]);
            }
        }

        if (current == root && childrenCount > 1) {
            articulationPoints.add(current);
        }
    }

    private static class Graph {
        private final Map<Integer, List<Integer>> graph;
        private final int n;

        public Graph(int n) {
            this.n = n;
            graph = new HashMap<>();
            for (int i = 0; i < n; i++) {
                graph.put(i, new ArrayList<>());
            }
        }

        public void addEdge(int from, int to) {
            graph.get(from).add(to);
        }

        public Iterable<Integer> getNeighbors(int vertex) {
            return graph.get(vertex);
        }

        public int size() {
            return n;
        }
    }
}
