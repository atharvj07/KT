#include <iostream>
#include <string>

int main() {
  int T;
  std::cin >> T;
  for (int t = 0; t < T; ++t) {
    int n;
    std::string s;
    std::cin >> n >> s;
    bool pal = true;
    for (int i = 0; i < n / 2; ++i) {
      if (s[i] != s[n - 1 - i] && s[i] + 1 != s[n - 1 - i] &&
          s[i] - 1 != s[n - 1 - i]) {
        pal = false;
        break;
      }
    }
    std::cout << (pal ? "YES" : "NO") << std::endl;
  }
  return 0;
}
