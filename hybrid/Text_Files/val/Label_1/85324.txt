#include <vector>
#include <string>
#include <map>
#include <iostream>
#include <algorithm>
#include <unordered_set>

using namespace std;
int main() {
  string a;
  cin >> a;
  int n;
  cin >> n;
  vector<pair<string, pair<int, int>>> r(n);
  for (auto& v : r) {
    string b;
    cin >> b >> v.second.first >> v.second.second;
    v.first = b;
  }
  map<string, vector<pair<int, int>>> m;
  for (auto& v : r) {
    string b = v.first;
    int len = v.first.size();
    for (int i = 0; i < a.size(); ++i) {
      if (i + len > a.size()) {
        continue;
      }
      auto sub = a.substr(i, len);
      if (sub == b) {
        m[sub].emplace_back(i, i + len - 1);
      }
    }
  }
  unordered_set<string> s;
  for (auto& v : m) {
    string sub = v.first;
    bool f = true;
    for (auto& u : r) {
      int oc = 0;
      for (auto& w : v.second) {
        if (w.first >= u.second.first && w.second <= u.second.second) {
          ++oc;
        }
      }
      if (oc == 0 || oc > u.second.second) {
        f = false;
        break;
      }
    }
    if (f) {
      s.insert(sub);
    }
  }
  cout << s.size() << endl;
  return 0;
}
