#include<map>
#include<cstdio>
int main() {
	int n, ans[500001] = {};
	std::map<int, int> umap;
	std::scanf("%d", &n);
	for (int i = 2; i <= n; ++i) {
		for (std::map<int, int>::iterator it = umap.begin(); it != umap.end(); ++it) {
			int x = it->first;
			ans[i] = std::max(ans[i], std::max(x, i / x));
		}
		umap[i] = i;
	}
	for (int i = 2; i <= n; ++i) {
		if (!ans[i]) ans[i] = 1;
		std::printf("%d ", ans[i]);
	}
	return 0;
}
