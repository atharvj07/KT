#include <bits/stdc++.h>

using namespace std;

const int MN = 1e5 + 5;
vector<int> adj[MN];
int up[MN], a[MN], ans[MN], deg[MN], cnt;

int dfs1(int u, int p) {
    up[u] = p;
    for (int v : adj[u]) {
        if (v == p)
            continue;
        up[u] = min(up[u], dfs1(v, u));
    }
    return up[u];
}

int dfs2(int u, int p = 0) {
    if (up[u] < p || (p && up[u] == p))
        return cnt++, -1;
    ans[cnt++] = u;
    for (int v : adj[u])
        if (v != p)
            dfs2(v, u);
}

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int x, y;
        cin >> x >> y;
        adj[x].push_back(y);
        deg[y]++;
    }
    for (int i = 0; i < n; i++)
        if (deg[i + 1] == 0)
            dfs1(i + 1, i + 1);
    for (int i = 0; i < n; i++)
        cin >> a[i + 1];
    reverse(a + 1, a + 1 + n);
    int ok = 1;
    for (int i = 1; i <= n; i++) {
        if (up[a[i]] < up[i])
            ok = 0;
    }
    if (!ok) {
        cout << -1 << endl;
        return 0;
    }
    dfs2(1);
    cout << cnt << endl;
    for (int i = 0; i < cnt; i++)
        cout << ans[i] << endl;
}
