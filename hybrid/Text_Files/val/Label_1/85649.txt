#include <iostream>
#include <cstdio>
using namespace std;
int n, sum, list[100];
bool check(int tmp) {
    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum += tmp % 10;
        tmp /= 10;
    }
    return sum == ::sum;
}
int main() {
    cin >> n >> sum;
    for (int i = 0; i < n; i++) {
        list[i] = 9;
    }
    if (sum < n || sum > (n * 9)) {
        cout << -1 << ' ' << -1 << endl;
        return 0;
    }
    if (n == 1) {
        cout << sum << ' ' << sum << endl;
        return 0;
    }
    list[n - 1]--;
    while (!check(--list[n - 1])) {
        list[n - 1] = 9;
        int index = n - 1;
        while (--index >= 0 && list[index] == 0) {
            list[index] = 9;
            list[index + 1]--;
        }
    }
    for (int i = n - 1; i >= 0; i--) {
        cout << list[i];
    }
    cout << ' ';
    for (int i = 0; i < n; i++) {
        list[i] = 0;
    }
    int index = 0;
    while (sum - list[index] >= n - index - 1) {
        list[index] = sum - (n - index - 1);
        sum -= sum - (n - index - 1);
        index++;
    }
    list[index] = sum;
    for (int i = 0; i < n; i++) {
        cout << list[i];
    }
    return 0;
}
