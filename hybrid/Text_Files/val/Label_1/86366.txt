import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long a = sc.nextLong();
            long b = sc.nextLong();
            long c = sc.nextLong();
            long d = sc.nextLong();

            if (b == c) {
                System.out.println(a + " " + b + " " + d);
            } else {
                System.out.println(a + " " + b + " " + c);
            }
        }
        sc.close();
    }
}
