import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

public class Main {
	static class Record implements Comparable<Record> {
		int index, num;
		Record(int i, int n) { index=i; num=n; }
		@Override
		public int compareTo(Record r) { return r.num-num; }
	}
	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int n=sc.nextInt(), q=sc.nextInt(), ans_index=1, ans_num=0;
		Record[] records=new Record[n]; for(int i=0; i<n; i++) records[i]=new Record(i+1, 0);
		for(int j=0; j<q; j++) {
			int index=sc.nextInt()-1, num=sc.nextInt();
			records[index].num+=num;
			if(ans_num<records[index].num) ans_num=records[index].num;
			else if(ans_num==records[index].num) ans_index=Math.min(ans_index, records[index].index);
		}
		Arrays.sort(records, Comparator.comparingInt(r->r.num));
		System.out.println(ans_index+" "+records[n-1].num);
	}
}
