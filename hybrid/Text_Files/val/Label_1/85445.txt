#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i], a[i]--;

  vector<int> p(n + 2);
  for (int i = 0; i < n; i++) p[a[i]]++;
  for (int i = 1; i <= n; i++) p[i] += p[i - 1];

  int ans = n;
  for (int i = 0; i < n; i++) {
    ans = max(ans, p[a[i] - 1] + n - p[a[i]]);
  }
  cout << ans << endl;
}
