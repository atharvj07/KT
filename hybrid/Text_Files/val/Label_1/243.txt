n, m = map(int, input().split())
x, t = [int(x) for x in input().split()], [int(x) for x in input().split()]
idx = [i for i in range(n+m) if not t[i]]
res = []
for d in x:
    if not idx: break
    l, r = 0, len(idx) - 1
    while l <= r:
        mid = (l + r) // 2
        if x[idx[mid]] <= d: l = mid + 1
        else: r = mid - 1
    res.append(l)
print(*res)
