#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int MAXN = 1000 + 5;
ll grid[MAXN][MAXN];

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> grid[i][j];
        }
    }

    ll sum_row[MAXN] = {0}, sum_col[MAXN] = {0}, sum_diag1 = 0, sum_diag2 = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            sum_row[i] += grid[i][j];
            sum_col[j] += grid[i][j];
            if (i == j) sum_diag1 += grid[i][j];
            if (i + j == n - 1) sum_diag2 += grid[i][j];
        }
    }

    ll sum = 0;
    for (int i = 0; i < n; i++) {
        sum += grid[i][i];
    }
    if (sum_diag1 != sum || sum_diag2 != sum) {
        cout << -1 << endl;
        return 0;
    }

    bool flag = false;
    for (int i = 0; i < n; i++) {
        if (sum_row[i] != sum || sum_col[i] != sum) {
            if (flag) {
                cout << -1 << endl;
                return 0;
            }
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 0) {
                    grid[i][j] = sum - sum_row[i] + grid[i][j];
                    flag = true;
                }
            }
        }
    }

    for (int i = 0; i < n; i++) {
        if (sum_row[i] != sum || sum_col[i] != sum) {
            cout << -1 << endl;
            return 0;
        }
    }

    cout << grid[0][0] << endl;

    return 0;
}
