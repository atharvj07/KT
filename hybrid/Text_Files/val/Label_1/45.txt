q = int(input())
for _ in range(q):
    n = int(input())
    sticks = list(map(int, input().split()))
    sticks.sort()
    area = sticks[0] * sticks[-1]
    possible = True
    for i in range(n):
        if sticks[2*i] != sticks[2*i+1] or sticks[-(2*i+1)] != sticks[-(2*i+2)] or sticks[2*i] * sticks[-(2*i+1)] != area:
            possible = False
            break
    if possible:
        print("YES")
    else:
        print("NO")
