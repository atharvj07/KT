import sys

input = sys.stdin.readline

n = int(input())
a = [list(map(int, input().split())) for _ in range(n)]
a.sort(key=lambda x: x[1], reverse=True)
b, c = {}, {}
ans = 0
for x in a:
    if x[0] not in b: b[x[0]] = x
    if x[2] not in c: c[x[2]] = x
    if x[0] in c and x[2] in b and c[x[0]] == b[x[2]]:
        ans += x[1]
        del b[x[0]], c[x[2]]
print(ans)
