#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 18;
int n;
double m[MAXN], r[MAXN];

int gcd(int a, int b) {
    if (b == 0) return a;
    return gcd(b, a % b);
}

int lcm(int a, int b) {
    return a / gcd(a, b) * b;
}

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> m[i];
    for (int i = 0; i < n; i++) cin >> r[i];
    double ans = 0;
    int lcm_all = 1;
    for (int i = 0; i < n; i++) {
        lcm_all = lcm(lcm_all, m[i]);
    }
    for (int i = 0; i < n; i++) {
        int period = m[i] / gcd(m[i], r[i]);
        ans += (lcm_all / m[i]) * (period - r[i]);
    }
    cout << fixed << setprecision(6) << ans / lcm_all << endl;
    return 0;
}
