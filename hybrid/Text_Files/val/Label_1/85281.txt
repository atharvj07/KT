#include <cstdio>
#include <algorithm>
#define ll long long
using namespace std;

int main() {
	ll n, ans, p[4001], q[4001], r[4001];
	scanf("%lld", &n);
	for (ll i = 1; i <= n; ++i) { scanf("%lld%lld%lld", &r[i], &q[i], &p[i]); }
	sort(r + 1, r + n + 1);
	sort(q + 1, q + n + 1);
	sort(p + 1, p + n + 1);
	ans = 1;
	printf("%lld\n", ans);
	printf("%lld\n", p[1]);
	return 0;
}
