n, t = map(int, input().split())
a = input()
i = a.find('.')
while t > 0:
    if i+2 < n and a[i+2] >= '5':
        a = a[:i] + str(int(a[i:i+2]) + 1)
    elif i == n-1:
        a += '0'
        i += 1
    t -= 1
    i += 1
while a[len(a)-1] == '0':
    a = a[:len(a)-1]
if a[len(a)-1] == '.':
    a = a[:len(a)-1]
print(a)
