import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Permutation {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int[] q = new int[n];
            for (int j = 0; j < n; j++) {
                q[j] = scanner.nextInt();
            }
            List<Integer> permutation = findPermutation(q);
            if (permutation == null) {
                System.out.println("-1");
            } else {
                for (int j = 0; j < n; j++) {
                    System.out.print(permutation.get(j) + " ");
                }
                System.out.println();
            }
        }
    }

    public static List<Integer> findPermutation(int[] q) {
        int n = q.length;
        List<Integer> permutation = new ArrayList<>();
        boolean[] used = new boolean[n + 1];
        int currentMax = 0;
        for (int i = 0; i < n; i++) {
            if (q[i] > currentMax) {
                // Find the smallest unused number greater than or equal to q[i]
                int next = 0;
                for (int j = q[i]; j <= n; j++) {
                    if (!used[j]) {
                        next = j;
                        break;
                    }
                }
                if (next == 0) {
                    return null; // No suitable number found
                }
                used[next] = true;
                permutation.add(next);
                currentMax = next;
            } else {
                // Find the next unused number less than or equal to q[i]
                int next = 0;
                for (int j = q[i]; j > 0; j--) {
                    if (!used[j]) {
                        next = j;
                        break;
                    }
                }
                if (next == 0) {
                    return null; // No suitable number found
                }
                used[next] = true;
                permutation.add(next);
            }
        }
        return permutation;
    }
}
