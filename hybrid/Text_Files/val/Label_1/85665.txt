#include <bits/stdc++.h>

using namespace std;

const int MAXN = 3001;
int a[MAXN], cnt[MAXN];

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        memset(cnt, 0, sizeof(cnt));
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            cnt[a[i]]++;
        }
        long long ans = 0;
        for (int i = 1; i <= n; i++) {
            ans += 1LL * cnt[i] * (cnt[i] - 1) * (cnt[i] - 2);
        }
        cout << ans / 2 << "\n";
    }
    return 0;
}
