def make_int(str):
    i = 0
    num = 0
    while str[i] != '.':
        num *= 10
        num += int(str[i])
        i += 1
    i += 1
    mul = 10
    while i < len(str):
        num *= 10
        num += int(str[i])
        mul *= 10
        i += 1
    return num, mul

n, t = list(map(int, input().split()))
num, mul = make_int(input())

ans = num
for i in range(1, t + 1):
    cur = num // mul
    if (num % mul * 10 // mul) >= 5:
        cur += 1
    mul //= 10
    ans = cur * mul + ans % mul

print(ans / mul)
