#include <bits/stdc++.h>
#define rep(i, l, r) for(int i = l; i <= r; ++i)
using namespace std;
const int N = 1e5 + 5;
vector<int> g[N], sz(N, 1), g2[N], b[N], a[N];
int p[N], c[N], cnt = 0, id[N];
void dfs(int u) {
	for(int v : g[u]) if(v != p[u]) dfs(v), sz[u] += sz[v];
}
int centroid(int u) {
	int mx = -1;
	for(int v : g[u]) if(v != p[u]) mx = max(mx, sz[v]);
	mx = max(mx, sz[u]);
	if(mx <= (sz[u] >> 1)) return u;
	for(int v : g[u]) if(v != p[u]) {
		int res = centroid(v);
		if(res != -1) return res;
	}
	return -1;
}
void invalid() {
	cout << "-1\n";
	exit(0);
}
void dfs2(int u) {
	id[u] = cnt;
	g2[cnt].push_back(c[u]);
	b[c[u]].push_back(cnt);
	for(int v : a[u]) {
		b[v].push_back(cnt);
		if(!id[v]) {
			if(g[u][g[u].size() - 1] != v) invalid();
			dfs2(v);
		}
	}
}
int32_t main() {
	ios::sync_with_stdio(0);
	int n, m;
	cin >> n >> m;
	rep(_, 1, m) {
		int u, v;
		cin >> u >> v;
		g[u].push_back(v);
		g[v].push_back(u);
	}
	rep(i, 2, n) cin >> p[i], g[p[i]].erase(find(g[p[i]].begin(), g[p[i]].end(), i));
	dfs(1);
	int cur = 1;
	while(cur != -1) {
		int r = centroid(cur);
		if(r == -1) invalid();
		c[r] = ++cnt;
		for(int v : g[r]) if(v != p[r]) a[r].push_back(v);
		g[r].clear();
		cur = r;
	}
	cnt = 0;
	rep(i, 1, n) if(!id[i]) {
		dfs2(i);
		cnt++;
	}
	rep(i, 1, cnt) if(b[i].size() == 1) invalid();
	cout << cnt << '\n';
	rep(i, 1, cnt) if(b[i].size() > 1) {
		for(int v : b[i]) cout << c[v] << '\n';
	}
}
