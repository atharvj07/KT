import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        Arrays.sort(a);
        int height = 0;
        int count = 0;
        int i = 0;
        while (i < n) {
            int j = i + 1;
            while (j < n && a[j] == a[i]) {
                j++;
            }
            if (j - i > height) {
                height = j - i;
                count = 1;
            } else if (j - i == height) {
                count++;
            }
            i = j;
        }
        System.out.println(height + " " + count);
    }
}
