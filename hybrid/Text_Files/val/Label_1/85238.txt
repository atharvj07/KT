#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int h, m, H, D, C, N;
    cin >> h >> m >> H >> D >> C >> N;
    H -= (h * 60 + m) * D;
    if (H <= 0) {
        cout << "0.0000" << endl;
    } else if (20 * 60 + 0 <= h * 60 + m) {
        cout << fixed << setprecision(4) << H * (C * 0.8) / N << endl;
    } else {
        cout << fixed << setprecision(4) << H * C / N << endl;
    }
}
