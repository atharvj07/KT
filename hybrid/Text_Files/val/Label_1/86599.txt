import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int m = sc.nextInt();
        int t = sc.nextInt();
        int[] l = new int[t + 1];
        l[0] = n;
        for (int i = 1; i <= t; i++) {
            int op = sc.nextInt();
            int p = sc.nextInt();
            if (op == 1) {
                l[i] = l[i - 1];
                if (p <= k) {
                    k++;
                }
                l[i]++;
            } else {
                l[i] = l[i - 1];
                if (p < k) {
                    k--;
                } else if (p == k) {
                    k = l[i] - k;
                }
                l[i]--;
            }
            System.out.println(l[i] + " " + k);
        }
    }
}
