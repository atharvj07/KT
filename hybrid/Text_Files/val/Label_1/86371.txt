import java.util.Scanner;

public class Main {
    private static int n;
    private static int[] p, c;
    private static int[][] edges;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        p = new int[n + 2];
        c = new int[n + 2];
        edges = new int[n][2];
        for (int i = 0; i < n - 1; i++) {
            String s1 = sc.next();
            String s2 = sc.next();
            p[i] = s1.length();
            c[i] = s2.length();
        }
        if (!isValid()) {
            System.out.println(-1);
            return;
        }
        buildTree();
        for (int i = 0; i < n - 1; i++) {
            System.out.println((edges[i][0] + 1) + " " + (edges[i][1] + 1));
        }
    }

    private static boolean isValid() {
        int[] count = new int[n + 2];
        for (int i = 0; i < n - 1; i++) {
            count[p[i]]++;
            count[c[i]]++;
        }
        for (int i = 1; i <= n; i++) {
            if (count[i] % 2 != 0) {
                return false;
            }
        }
        return true;
    }

    private static void buildTree() {
        int[] deg = new int[n + 2];
        for (int i = 0; i < n - 1; i++) {
            deg[p[i]]++;
            deg[c[i]]++;
        }
        int root = -1;
        for (int i = 1; i <= n; i++) {
            if (deg[i] == 1) {
                root = i;
                break;
            }
        }
        int[] q = new int[n];
        int front = 0, rear = 0;
        q[rear++] = root;
        int[][] graph = new int[n + 2][n + 2];
        while (front < rear) {
            int u = q[front++];
            for (int i = 0; i < n - 1; i++) {
                if (p[i] == u) {
                    if (graph[u][c[i]] == 0) {
                        graph[u][c[i]] = 1;
                        graph[c[i]][u] = 1;
                        q[rear++] = c[i];
                    }
                } else if (c[i] == u) {
                    if (graph[u][p[i]] == 0) {
                        graph[u][p[i]] = 1;
                        graph[p[i]][u] = 1;
                        q[rear++] = p[i];
                    }
                }
            }
        }
        int k = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (graph[i][j] > 0) {
                    edges[k][0] = i - 1;
                    edges[k][1] = j - 1;
                    k++;
                }
            }
        }
    }
}
