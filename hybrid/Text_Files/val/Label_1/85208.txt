#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int &x : a) cin >> x;
        int c = a[0];
        int m = a[1] - c;
        if (m <= 0) {
            cout << -1 << endl;
            continue;
        }
        int valid = 1;
        for (int i = 1; i < n; i++) {
            if ((a[i] - c + m) % m != a[i - 1]) valid = 0;
        }
        if (!valid) {
            cout << -1 << endl;
            continue;
        }
        int s = a[0];
        if (c == 0)
            cout << 0 << endl;
        else
            cout << m << ' ' << c << endl;
    }
}
