template <typename T>
ostream& operator<<(ostream& os, const vector<T>& v) {
    if (!v.empty()) {
        os << '[';
        for (int i = 0; i < v.size() - 1; i++) { os << v[i] << ", "; }
        os << v[v.size() - 1] << ']';
    }
    return os;
}
template <typename T>
ostream& operator<<(ostream& os, const set<T>& s) {
    if (!s.empty()) {
        os << '{';
        for (auto elem : s) { os << elem << ", "; }
        os << '}';
    }
    return os;
}
int get_min(priority_queue<int>& pq, set<int>& st) {
    while (true) {
        if (pq.empty()) {
            return -1;
        }
        int top = pq.top();
        if (!st.count(top)) {
            return pq.top();
        }
        pq.pop();
    }
}
int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n;
    cin >> n;
    priority_queue<int> pq;
    set<int> st;
    while (n--) {
        string s;
        cin >> s;
        if (s == "insert") {
            int x;
            cin >> x;
            pq.push(x);
        } else if (s == "removeMin") {
            if (!pq.empty()) st.insert(pq.top()), pq.pop();
        } else {
            int x;
            cin >> x;
            int ans = get_min(pq, st);
            if (ans == -1 || ans != x) {
                if (!pq.empty()) {
                    auto top = pq.top();
                    pq.pop();
                    pq.push(x);
                    st.insert(top);
                } else
                    pq.push(x);
            }
        }
    }
    int ans = pq.size() + st.size();
    cout << ans << nl;
    while (pq.empty() == false) {
        cout << "insert " << pq.top() << nl;
        pq.pop();
    }
    for (auto& e : st) {
        cout << "removeMin" << nl;
        cout << "insert " << e << nl;
    }
}
