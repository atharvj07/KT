import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int m = in.nextInt();
        int t = in.nextInt();
        int[] len = new int[t + 1];
        int[] pos = new int[t + 1];
        len[0] = n;
        pos[0] = k;
        for (int i = 1; i <= t; i++) {
            int op = in.nextInt();
            int p = in.nextInt();
            if (op == 1) {
                len[i] = len[i - 1] + 1;
                if (p <= pos[i - 1]) {
                    pos[i] = pos[i - 1] + 1;
                } else {
                    pos[i] = pos[i - 1];
                }
            } else {
                len[i] = len[i - 1] - 1;
                if (p < pos[i - 1]) {
                    pos[i] = pos[i - 1] - 1;
                } else {
                    pos[i] = pos[i - 1];
                }
            }
            System.out.println(len[i] + " " + pos[i]);
        }
    }
}
