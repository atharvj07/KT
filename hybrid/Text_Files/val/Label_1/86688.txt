import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        int ops = 0;
        for (int i = 1; i < n; i++) {
            while (a[i] < a[i - 1]) {
                a[i] *= -2;
                ops++;
            }
        }
        if (a[n - 1] >= 0) {
            System.out.println(ops);
        } else {
            System.out.println(-1);
        }
    }
}
