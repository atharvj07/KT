import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] fingers = new int[n];
    for (int i = 0; i < n; i++) {
      fingers[i] = sc.nextInt();
    }
    int sum = 0;
    for (int finger : fingers) {
      sum += finger;
    }
    if (sum == 1 || sum % 2 == 0) {
      System.out.println((1 << n) - 1);
    } else {
      System.out.println(1 << (n - 1));
    }
  }
}
