#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

struct Rating {
  int start;
  int change;
};

bool cmp(Rating a, Rating b) { return a.start < b.start; }

int main() {
  int n, s;
  cin >> n >> s;
  vector<Rating> a(n);
  for (int i = 0; i < n; i++) cin >> a[i].start >> a[i].change;
  sort(a.begin(), a.end(), cmp);
  for (int i = 0; i < n; i++) {
    if (s < a[i].start) {
      cout << "NO" << endl;
      return 0;
    }
    s -= a[i].change;
    if (s < 0) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
  return 0;
}
