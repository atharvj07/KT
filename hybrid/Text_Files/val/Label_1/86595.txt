import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] arr = new int[m][3];
        for (int i = 0; i < m; i++) {
            arr[i][0] = sc.nextInt() - 1;
            arr[i][1] = sc.nextInt() - 1;
            arr[i][2] = sc.nextInt();
        }
        int[] dp = new int[n];
        for (int[] a : arr) {
            dp[a[0]] = Math.max(dp[a[0]], a[2]);
            dp[a[1]] = Math.max(dp[a[1]], a[2]);
        }
        int sum = 0;
        for (int a : dp) {
            sum += a;
        }
        System.out.println(sum);
    }
}
