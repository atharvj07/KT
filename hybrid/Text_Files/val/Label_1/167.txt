n, m, x, y, z, p = map(int, input().split())
x %= 4
y %= 2
z %= 4

for _ in range(p):
    x_k, y_k = map(int, input().split())
    for _ in range(x):
        x_k, y_k = y_k, n + 1 - x_k
        n, m = m, n
    for _ in range(y):
        y_k = m + 1 - y_k
    for _ in range(z):
        x_k, y_k = m + 1 - y_k, x_k
        n, m = m, n
    print(x_k, y_k)
