#include <bits/stdc++.h>
#define ll long long
using namespace std;

int main() {
    ll n,a;
    cin>>n;
    vector<ll>x(n);
    for(ll i=0;i<n;i++){
        cin>>x[i];
    }
    sort(x.begin(),x.end());
    ll l=0,h=n-1,ans=0,m=-1;
    while(l<h){
        ll d1=x[h]-x[l];
        ll d2=x[h-1]-x[l+1];
        if(d1<=d2){
            ans=ans+d1;
            m=x[h];
            h=h-1;
        }
        else{
            ans=ans+d2;
            m=x[l+1];
            l=l+1;
        }
        l=l+1;
        h=h-1;
    }
    if(n%2==0) ans=ans-m;
    cout<<ans<<endl;
    return 0;
}
