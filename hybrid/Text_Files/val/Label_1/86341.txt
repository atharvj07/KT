import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int tests = in.nextInt();
        while (tests-- > 0) {
            int n = in.nextInt();
            System.out.println(n % 2020 <= n % 2021 ? "YES" : "NO");
        }
    }
}
