#include <cstdio>
#include <vector>
int main() {
  long long int a, b, x, y, ans, c;
  scanf("%lld %lld %lld %lld", &a, &b, &x, &y);
  if (y % x == 0)
    c = y / x;
  else
    c = y;
  if (a + b <= c)
    ans = a + b;
  else if (a + b > c && (a > c || b > c))
    if (a + b > x * y)
      ans = a + b;
    else
      ans = x * y;
  else if (a + b % 2 == 0) {
    if (a + b % x == 0 && a + b % y != 0)
      ans = a + b;
    else if (a + b % y == 0 && a + b % x != 0)
      ans = a + b;
    else
      ans = a + b + 1;
  } else if (a + b % 2 == 1) {
    if (a + b % x != 0 && a + b % y != 0)
      ans = a + b;
    else if (a + b % x == 0 && a + b % y == 0)
      ans = a + b + 1;
    else
      ans = a + b;
  }
  printf("%lld\n", ans);
  return 0;
}
