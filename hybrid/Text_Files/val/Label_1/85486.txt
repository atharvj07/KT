#include<bits/stdc++.h>
using namespace std;
int n,cnt[3005],x[3005],f[3005],g[3005][3005];
vector<pair<int,int>>edge[3005];
void dfs(int pre,int now){
    for(auto temp:edge[now]){
        int next=temp.first,val=temp.second;
        if(pre==next)continue;
        dfs(now,next);
        f[now]=max(f[now],f[next]);
        g[now][next]=g[next][now]=val;
    }
    f[now]=max(f[now],g[now][pre]);
}
int main(){
    cin>>n;
    for(int i=1;i<n;++i){
        int a,b,c;
        cin>>a>>b>>c;
        edge[a].push_back({b,c});
        edge[b].push_back({a,c});
    }
    for(int i=1;i<=n;++i){
        cin>>x[i];
        cnt[x[i]]++;
    }
    for(int i=1;i<=n;++i){
        if(cnt[i]==1)dfs(0,i);
    }
    int ans=0;
    for(int i=1;i<=n;++i){
        if(f[i]==0)continue;
        if(cnt[i]==1)ans=max(ans,f[i]);
        else ans=max(ans,f[i]-1);
    }
    cout<<ans<<endl;
    return 0;
}
