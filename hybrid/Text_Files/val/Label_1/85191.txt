#include <bits/stdc++.h>
#define sz(c) (int)c.size()
using namespace std;
const int N = 3e5 + 5;
int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<int> p(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> p[i];
    }
    vector<pair<int, int>> ans;
    for (int i = 1; i <= n; i += 2) {
        int mn = i, indx = i;
        for (int j = i + 1; j <= n; j += 2) {
            if (p[mn] > p[j]) {
                mn = j;
                indx = j;
            }
        }
        if (mn == i)
            continue;
        ans.push_back({mn, indx});
        for (int j = 0; j < n / 2; j++) {
            int x = p[indx - j];
            p[indx - j] = p[mn - j];
            p[mn - j] = x;
        }
    }
    cout << sz(ans) << "\n";
    for (auto &i : ans) {
        cout << i.first << " " << i.second << "\n";
    }
}
