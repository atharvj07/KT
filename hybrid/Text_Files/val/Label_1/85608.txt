#include <iostream>

using namespace std;

const int MAX_N = 2e5 + 5;

int main() {
    int n, t, c;
    cin >> n >> t >> c;
    long long int a[MAX_N];
    for (int i = 0; i < n; i++) cin >> a[i];
    int pref[MAX_N];
    for (int i = 0; i < n; i++) {
        pref[i] = (a[i] <= t);
        if (i > 0) pref[i] += pref[i - 1];
    }
    int sum = 0, ans = 0;
    for (int i = 0; i < c; i++) sum += (a[i] <= t);
    if (sum == c) ans++;
    for (int i = c; i < n; i++) {
        sum += (a[i] <= t) - (a[i - c] <= t);
        if (sum == c) ans++;
    }
    cout << ans << endl;
    return 0;
}
