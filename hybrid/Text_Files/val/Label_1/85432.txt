#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main(){
    ll n,arr[400005],i,j,c=0;
    cin>>n;
    for(i=0;i<n;i++) cin>>arr[i];
    while(arr[0]%2==0) arr[0]/=2,c++;
    for(i=1;i<n;i++){
        bool chk=0;
        while(arr[i]%2==0){
            arr[i]/=2;
            if(!chk) c++,chk=1;
        }
        for(j=3;j*j<=arr[i];j+=2){
            chk=0;
            while(arr[i]%j==0){
                arr[i]/=j;
                if(!chk) c++,chk=1;
            }
        }
        if(arr[i]>1) c++;
    }
    cout<<c<<endl;
    return 0;
}
