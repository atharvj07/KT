#include <iostream>
#include <vector>
#include <iomanip>
using namespace std;
typedef long long ll;
int main() {
  int n;
  cin >> n;
  vector<ll> m(n), r(n);
  for (int i =0; i < n; i++) { cin >> m[i]; }
  for (int i =0; i < n; i++) { cin >> r[i]; }
  double ans = 0;
  for (int i =0; i < n; i++) {
    double x = (double) m[i];
    double y = (double) r[i];
    ans += y / x;
  }
  ans /= (double) n;
  cout << fixed << setprecision(6) << ans << '\n';
}
