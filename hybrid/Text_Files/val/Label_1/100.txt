n = int(input())
x1, x2 = map(int, input().split())
lines = [tuple(map(int, input().split())) for _ in range(n)]
lines.sort(key=lambda x: (x[0], x[1]))
for i in range(n):
    line = lines[i]
    if line[0] * x1 + line[1] >= line[0] * x2 + line[1]:
        continue
    for j in range(i + 1, n):
        t = lines[j]
        if t[0] * x1 + t[1] == t[0] * x2 + t[1]:
            continue
        y = (line[1] - t[1] + t[0] * x1 - line[0] * x1) / (line[0] - t[0])
        if x1 < y < x2:
            print("YES")
            exit()
print("NO")
