#include <bits/stdc++.h>
using namespace std;

int main() {
  long long n;
  cin >> n;
  long long l = 0, r = 1e9, ans = 0;
  while (l <= r) {
    long long mid = (l + r) / 2;
    if (mid * (mid + 1) / 2 <= n) {
      ans = mid;
      l = mid + 1;
    } else
      r = mid - 1;
  }
  cout << ans << endl;
  return 0;
}
