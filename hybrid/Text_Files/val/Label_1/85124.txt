#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a[200005];
    cin >> n;
    for (int i = 1; i <= n; ++i) cin >> a[i];
    sort(a + 1, a + n + 1);
    int ans = 1;
    for (int l = 1, r = 2; r <= n; ++r) if (a[r] - a[l] <= 5) ans = max(ans, r - l + 1); else ++l;
    cout << ans << endl;
}
