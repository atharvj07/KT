s = input().strip()
t = input().strip()

# Create two lists to store the earliest and latest positions
# of each character in t in s
earliest_positions = []
latest_positions = [0] * len(t)

# Pointer for t
t_pointer = 0

# Find earliest positions of characters in t in s
for i, char in enumerate(s):
    if t_pointer < len(t) and char == t[t_pointer]:
        earliest_positions.append(i)
        t_pointer += 1

# Reset pointer for t
t_pointer = len(t) - 1

# Find latest positions of characters in t in s (in reverse)
for i in range(len(s) - 1, -1, -1):
    if t_pointer >= 0 and s[i] == t[t_pointer]:
        latest_positions[t_pointer] = i
        t_pointer -= 1

# Calculate the maximum length of the substring that can be removed
max_length = 0
for i in range(len(t) - 1):
    # The gap between the current character's latest position
    # and the next character's earliest position.
    max_length = max(max_length, earliest_positions[i + 1] - latest_positions[i] - 1)

# Additionally, check the length by removing from the start to the first t's character earliest position
# and from the last t's character latest position to the end of s
max_length = max(max_length, earliest_positions[0], len(s) - latest_positions[-1] - 1)

print(max_length)
