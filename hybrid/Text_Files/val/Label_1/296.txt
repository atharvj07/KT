def parse_command_line(s):
    lexemes = []
    i = 0
    while i < len(s):
        if s[i] == '"':
            j = i + 1
            while j < len(s) and s[j] != '"':
                j += 1
            lexemes.append(s[i+1:j])
            i = j + 1
        elif s[i] != ' ':
            j = i
            while j < len(s) and s[j] != ' ':
                j += 1
            lexemes.append(s[i:j])
            i = j
        else:
            i += 1
    return lexemes

s = input()
lexemes = parse_command_line(s)
for lexeme in lexemes:
    print(f"<{lexeme}>")
