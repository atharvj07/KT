n = int(input())
exercises = list(map(int, input().split()))
chest = sum(exercises[i] for i in range(0, len(exercises), 3))
biceps = sum(exercises[i] for i in range(1, len(exercises), 3))
back = sum(exercises[i] for i in range(2, len(exercises), 3))
result = max([chest, biceps, back])
if result == chest: print("chest")
elif result == biceps: print("biceps")
elif result == back: print("back")
