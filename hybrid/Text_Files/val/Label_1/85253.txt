#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, cnt = 0;
    vector<pair<int, int>> ans;
    scanf("%d", &n);
    while (n >= 4) {
        ans.emplace_back(2, 4);
        n -= 2;
        ++cnt;
    }
    if (n == 3) {
        ans.emplace_back(2, 3);
        ++cnt;
    }
    printf("%d\n", cnt);
    for (auto x : ans) printf("%d %d\n", x.first, x.second);
    return 0;
}
