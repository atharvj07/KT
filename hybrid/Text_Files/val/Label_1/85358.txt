#include <bits/stdc++.h>
using namespace std;

const int MM = 100001, RT = 18;

int n, m, d, p[MM], dep[MM], anc[MM][RT], dist[MM];
vector<int> adj[MM];

void dfs(int u, int p) {
  for (int v : adj[u])
    if (v != p) {
      dep[v] = dep[u] + 1;
      anc[v][0] = u;
      dfs(v, u);
    }
}

void init() {
  for (int i = 1; i < RT; i++)
    for (int j = 1; j <= n; j++) anc[j][i] = anc[anc[j][i - 1]][i - 1];
}

int lca(int a, int b) {
  if (dep[a] < dep[b]) swap(a, b);
  int diff = dep[a] - dep[b];
  for (int i = 0; i < RT && diff; i++)
    if (diff & (1 << i))
      a = anc[a][i], diff ^= (1 << i);
  if (a == b) return a;
  for (int i = RT - 1; i >= 0; i--)
    if (anc[a][i] != anc[b][i]) a = anc[a][i], b = anc[b][i];
  return anc[a][0];
}

int get_dist(int a, int b) {
  return dep[a] + dep[b] - 2 * dep[lca(a, b)];
}

int main() {
  cin >> n >> m >> d;
  for (int i = 0; i < m; i++) cin >> p[i];
  for (int i = 0; i < n - 1; i++) {
    int a, b;
    cin >> a >> b;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  dfs(1, 1);
  init();
  bool h = 0;
  for (int i = 0; i < m; i++) {
    for (int j = i + 1; j < m; j++) {
      if (get_dist(p[i], p[j]) <= d) h = 1;
    }
  }
  for (int i = 0; i < m; i++)
    for (int a : adj[p[i]])
      if (get_dist(p[i], a) <= d && dep[a] > dep[p[i]]) dist[a]++;
  int ans = 0;
  for (int i = 1; i <= n; i++)
    if (dist[i]) ans++;
  ans += h;
  cout << ans << endl;
  return 0;
}
