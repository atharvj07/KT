#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, c[2] = {0}, a[100005], cnt = 0;
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a, a + n);
  for (int i = 0; i < n; i++) {
    if (a[i] != a[i + 1]) { cnt++; }
    c[a[i] % 2]++;
  }
  cout << cnt + min(c[0], c[1]) << endl;
}
