import java.util.Scanner;

public class OlympiadTeams {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] skills = new int[n];

        for (int i = 0; i < n; i++) {
            skills[i] = scanner.nextInt();
        }

        int[] countSkills = new int[3]; // 0 - programming, 1 - math, 2 - PE
        for (int skill : skills) {
            countSkills[skill - 1]++;
        }

        int w = Math.min(countSkills[0], Math.min(countSkills[1], countSkills[2]));

        System.out.println(w);

        for (int i = 0; i < w; i++) {
            int programmer = -1, mathematician = -1, sportsman = -1;

            for (int j = 0; j < n; j++) {
                if (skills[j] == 1 && programmer == -1) {
                    programmer = j + 1;
                    countSkills[0]--;
                } else if (skills[j] == 2 && mathematician == -1) {
                    mathematician = j + 1;
                    countSkills[1]--;
                } else if (skills[j] == 3 && sportsman == -1) {
                    sportsman = j + 1;
                    countSkills[2]--;
                }
            }

            System.out.println(programmer + " " + mathematician + " " + sportsman);
        }
    }
}
