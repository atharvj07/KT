import sys

def solve():
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    for _ in range(m):
        op, *args = map(int, input().split())
        if op == 1: x, v = args; a[x-1] = v
        elif op == 2: l, r = args; print(sum(a[l-1:r]))
        else: l, r, d = args; a[l-1:r] = [x+d for x in a[l-1:r]]

solve()
