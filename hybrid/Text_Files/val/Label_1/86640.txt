import java.io.*;
import java.util.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] inputs = br.readLine().split(" ");
        int n = Integer.parseInt(inputs[0]);
        int r1 = Integer.parseInt(inputs[1]);
        int r2 = Integer.parseInt(inputs[2]);
        int r3 = Integer.parseInt(inputs[3]);
        int d = Integer.parseInt(inputs[4]);

        inputs = br.readLine().split(" ");
        int[] monster = new int[n];
        for (int i = 0; i < n; i++) {
            monster[i] = Integer.parseInt(inputs[i]);
        }

        long total = 0;
        for (int i = 0; i < n; i++) {
            int type = 3;
            if (monster[i] > 2) {
                total += r2 + d * 2;
                monster[i] -= 2;
            } else if (monster[i] == 2) {
                total += r3 + d * 2;
                monster[i] = 0;
            } else {
                while (monster[i] >= 2) {
                    monster[i] -= 2;
                    total += r2;
                }

                while (monster[i] > 0) {
                    monster[i] -= 1;
                    total += r1;
                }
            }
        }

        System.out.println(total);
    }
}
