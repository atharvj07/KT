n, k, x = map(int, input().split())
a = list(map(int, input().split()))

# If it's impossible to repost the required number of pictures
if x < (n + k - 1) // k:
    print(-1)
else:
    # Initialize dynamic programming table
    dp = [[0] * (x + 1) for _ in range(n + 1)]

    # Fill the table
    for i in range(1, n + 1):
        for j in range(1, x + 1):
            # Consider all possible segments ending at i
            for l in range(1, min(k, i) + 1):
                dp[i][j] = max(dp[i][j], dp[i-l][j-1] + a[i-1])
    
    # Find the maximum value for the required reposts
    print(max(dp[n][x], 0))
