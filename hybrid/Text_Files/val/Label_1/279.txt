n = int(input())
path1 = input()
path2 = input()

def can_reach_end(n, path1, path2):
    # Convert the paths into their delta (changes in x and y)
    deltas = {'N': (0, 1), 'S': (0, -1), 'E': (1, 0), 'W': (-1, 0)}
    delta_path1 = [deltas[move] for move in path1]
    delta_path2 = [deltas[move] for move in path2]

    # Calculate the cumulative changes in x and y for both paths
    cum_path1 = [(0, 0)]
    cum_path2 = [(0, 0)]
    for dx, dy in delta_path1:
        cum_path1.append((cum_path1[-1][0] + dx, cum_path1[-1][1] + dy))
    for dx, dy in delta_path2:
        cum_path2.append((cum_path2[-1][0] + dx, cum_path2[-1][1] + dy))

    # Check if the net changes in x and y are the same for both paths
    # This means they can reach the end at the same time
    return cum_path1[-1] == cum_path2[-1]

print("YES" if can_reach_end(n, path1, path2) else "NO")
