#include <iostream>
#include <algorithm>
#include <vector>
#include <map>

int main() {
    std::string s;
    int n;
    std::cin >> s >> n;
    std::vector<std::map<char, int>> rules(n);
    for (int i = 0; i < n; i++) {
        std::string p;
        int l, r;
        std::cin >> p >> l >> r;
        for (const char& c : p) {
            rules[i][c]++;
        }
    }
    std::map<std::string, int> substrings;
    int ans = 0;
    for (int i = 0; i < s.size(); i++) {
        std::map<char, int> cur;
        bool good = true;
        for (int j = i; j < s.size(); j++) {
            cur[s[j]]++;
            for (int k = 0; k < n; k++) {
                if (!rules[k].empty()) {
                    int cnt = 0;
                    for (const auto& [c, x] : rules[k]) {
                        cnt += std::min(cur[c], x);
                    }
                    if (cnt < rules[k].size() || cnt > 2 * rules[k].size()) {
                        good = false;
                    }
                }
            }
            if (!good) {
                break;
            }
            substrings[s.substr(i, j - i + 1)]++;
        }
        ans += substrings[s.substr(i, 1)];
    }
    std::cout << ans << std::endl;
    return 0;
}
