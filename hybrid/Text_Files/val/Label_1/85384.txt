#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  string l, r;
  cin >> l >> r;
  vector<pair<int, int>> ans;
  for (int i = 0; i < n; i++) {
    if (l[i] == '?' || r[i] == '?') ans.emplace_back(i + 1, i + 1);
  }
  for (int i = 0; i < n; i++) {
    if (l[i] != '?' && r[i] != '?') ans.emplace_back(i + 1, i + 1);
  }
  for (int i = 0; i < n; i++) {
    if (l[i] != '?') {
      for (int j = 0; j < n; j++) {
        if (r[j] != '?' && l[i] == r[j]) {
          ans.emplace_back(i + 1, j + 1);
          break;
        }
      }
    }
  }
  cout << ans.size() << '\n';
  for (auto& p : ans) { cout << p.first << ' ' << p.second << '\n'; }
  return 0;
}
