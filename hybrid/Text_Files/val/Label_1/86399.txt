import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {

    private static int n, k;
    private static List<Book> books;

    private static class Book {
        int time;
        int alice;
        int bob;

        public Book(int time, int alice, int bob) {
            this.time = time;
            this.alice = alice;
            this.bob = bob;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] nk = reader.readLine().split(" ");
        n = Integer.parseInt(nk[0]);
        k = Integer.parseInt(nk[1]);
        books = new ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            String[] book = reader.readLine().split(" ");
            books.add(new Book(Integer.parseInt(book[0]), Integer.parseInt(book[1]), Integer.parseInt(book[2])));
        }
        int result = solve();
        System.out.println(result);
    }

    private static int solve() {
        int[][] dp = new int[n + 1][k + 1];
        for (int i = 1; i <= n; i++) {
            Book book = books.get(i - 1);
            for (int j = 0; j <= k; j++) {
                dp[i][j] = dp[i - 1][j];
                if (j > 0 && (book.alice > 0 || book.bob > 0)) {
                    dp[i][j] = Math.max(dp[i][j], dp[i - 1][j - 1] + book.time);
                }
            }
        }
        int max = 0;
        for (int i = 0; i <= k; i++) {
            max = Math.max(max, dp[n][i]);
        }
        return max == 0 ? -1 : max;
    }
}
