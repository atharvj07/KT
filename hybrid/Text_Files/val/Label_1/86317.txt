import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    Queue<int[]> q1 = new LinkedList<>();
    Queue<int[]> q2 = new LinkedList<>();
    Queue<int[]> q3 = new LinkedList<>();
    int k1 = in.nextInt(), k2 = in.nextInt(), k3 = in.nextInt();
    int t1 = in.nextInt(), t2 = in.nextInt(), t3 = in.nextInt();
    int n = in.nextInt();
    int time = 0;
    for (int i = 0; i < n; i++) {
      int enterTime = in.nextInt();
      q1.add(new int[] { enterTime, 0 });
    }
    int[] ids = new int[n];
    for (int i = 0; i < n; i++) {
      while (!q1.isEmpty() && q1.peek()[1] <= time) {
        ids[i] = q1.poll()[0];
      }
      q2.add(new int[] { ids[i], 0 });
      time += t1;
    }
    int[] moneys = new int[n];
    for (int i = 0; i < n; i++) {
      while (!q2.isEmpty() && q2.peek()[1] <= time) {
        moneys[i] = q2.poll()[0];
      }
      q3.add(new int[] { moneys[i], 0 });
      time += t2;
    }
    for (int i = 0; i < n; i++) {
      while (!q3.isEmpty() && q3.peek()[1] <= time) {
        q3.poll();
      }
      time += t3;
    }
    System.out.println(time);
  }
}
