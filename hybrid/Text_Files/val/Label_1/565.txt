MOD = 10**9 + 7

def count_ways(n, m):
    if m == 1:
        return (2 * n) % MOD
    if n == m:
        return pow(2, m, MOD)

    # Number of ways to assign seats with at least one empty seat between passengers
    empty_seats = n - m + 1
    ways = 2 * empty_seats * pow(2, m-1, MOD)

    # Number of ways to assign passengers to the remaining seats
    ways += (m - 1) * empty_seats * pow(2, m-2, MOD)
    return ways % MOD

n, m = map(int, input().split())
print(count_ways(n, m))
