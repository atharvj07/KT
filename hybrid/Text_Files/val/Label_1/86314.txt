import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        // Read input
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int m = input.nextInt();
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = input.nextInt();
        }
        // Sort the arrays
        Arrays.sort(a);
        Arrays.sort(b);
        // Initialize the count of pairs
        int count = 0;
        // Iterate through the arrays
        int i = 0, j = 0;
        while (i < n && j < m) {
            // If the difference between the dancing skills is at most 1, increment the count and move on
            if (Math.abs(a[i] - b[j]) <= 1) {
                count++;
                i++;
                j++;
            } else if (a[i] < b[j]) {
                i++;
            } else {
                j++;
            }
        }
        // Print the count of pairs
        System.out.println(count);
    }
}
