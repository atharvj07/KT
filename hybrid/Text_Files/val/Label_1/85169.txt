#include <iostream>
#include <cstring>
using namespace std;
int main() {
    int T;
    cin >> T;
    while (T--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        bool flag = true;
        for (int i = 0; i < n / 2; i++) {
            if ((s[i] == 'z' && s[n - i - 1] != 'y') || (s[i] == 'a' && s[n - i - 1] != 'b')) {
                flag = false;
                break;
            }
            if (s[i] != s[n - i - 1]) {
                if (s[i] == s[n - i - 1] + 1 || s[i] == s[n - i - 1] - 1)
                    continue;
                else {
                    flag = false;
                    break;
                }
            }
        }
        cout << (flag ? "YES\n" : "NO\n");
    }
    return 0;
}
