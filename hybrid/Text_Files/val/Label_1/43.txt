for _ in range(int(input())):
    a = list(map(int, input().split()))
    a.sort()
    if a[0] == a[-1]:
        print("YES")
    elif a[1] == a[-1] and a[2] >= a[1] * 2:
        print("YES")
    elif a[1] >= a[0] * 2 and a[2] >= a[1] * 2:
        print("YES")
    elif a[2] >= a[0] * 3:
        print("YES")
    else:
        print("NO")
