#include <vector>
#include <iostream>
#include <queue>
using namespace std;

int n, m;
vector<vector<pair<int, int>>> adj;
vector<int> dist, a;

void dijkstra(int s) {
    dist[s] = 0;
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    pq.push({0, s});
    while (!pq.empty()) {
        int u = pq.top().second;
        pq.pop();
        for (auto& edge : adj[u]) {
            int v = edge.first, w = edge.second;
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }
}

int main() {
    cin >> n >> m;
    adj.resize(n + 1);
    dist.resize(n + 1, 1e9);
    a.resize(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) dijkstra(i);
    for (int i = 1; i <= n; i++) {
        int ans = 1e9;
        for (int j = 1; j <= n; j++) ans = min(ans, 2 * dist[j] + a[j]);
        cout << ans << " ";
    }
    cout << endl;
    return 0;
}
