#include<bits/stdc++.h>
#define ld long double
#define sc scanf
#define pr printf
const ld pi=2*acos(0.0);
int main()
{
	ld x[5],y[5];
	int n,dd,i,j;
	sc("%d %Lf %Lf %Lf %Lf",&n,&x[0],&y[0],&x[4],&y[4]);
	char s[n+5];
	sc("%s",s);
	char d[]="ESWN";
	for(i=0;i<4;i++)
	{
		x[i+1]=x[i];
		y[i+1]=y[i];
		if(s[0]==d[i])
		{
			dd=i;
			break;
		}
	}
	for(i=0;i<n;i++)
	{
		dd=(dd+1)%4;
		x[dd+1]=x[dd]+(d[dd]=='E'?1:d[dd]=='W'?-1:0);
		y[dd+1]=y[dd]+(d[dd]=='S'?1:d[dd]=='N'?-1:0);
		for(j=0;j<5;j++)
		{
			if(x[dd+1]==x[j]&&y[dd+1]==y[j])
				goto out;
		}
		if(x[dd+1]==x[4]&&y[dd+1]==y[4])
		{
			pr("%d\n",i+1);
			exit(0);
		}
	}
	out:
	pr("-1\n");
}
