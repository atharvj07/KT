import java.util.Scanner;
import java.util.HashMap;
import java.util.Map;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();

        Map<Integer, Integer> digitCounts = new HashMap<>();
        for (int i = 0; i < n - 1; i++) {
            String[] edge = sc.nextLine().split(" ");
            int digitCount1 = edge[0].length();
            int digitCount2 = edge[1].length();

            digitCounts.put(digitCount1, digitCounts.getOrDefault(digitCount1, 0) + 1);
            digitCounts.put(digitCount2, digitCounts.getOrDefault(digitCount2, 0) + 1);
        }

        if (digitCounts.size() != n) {
            System.out.println("-1");
            return;
        }

        for (int digitCount : digitCounts.keySet()) {
            if (digitCounts.get(digitCount) != 1) {
                System.out.println("-1");
                return;
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 1; i < n; i++) {
            sb.append("1 ").append(i).append("\n");
        }

        System.out.println(sb.toString());
    }
}
