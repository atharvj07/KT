import java.io.*;
import java.util.*;

public class Main {

    static char[][] grid;
    static int n, m;

    static boolean valid() {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == '.') {
                    continue;
                }
                int white = 0, red = 0;
                if (i > 0 && grid[i - 1][j] == 'R') {
                    red++;
                } else if (i > 0 && grid[i - 1][j] == 'W') {
                    white++;
                }
                if (i < n - 1 && grid[i + 1][j] == 'R') {
                    red++;
                } else if (i < n - 1 && grid[i + 1][j] == 'W') {
                    white++;
                }
                if (j > 0 && grid[i][j - 1] == 'R') {
                    red++;
                } else if (j > 0 && grid[i][j - 1] == 'W') {
                    white++;
                }
                if (j < m - 1 && grid[i][j + 1] == 'R') {
                    red++;
                } else if (j < m - 1 && grid[i][j + 1] == 'W') {
                    white++;
                }
                if (grid[i][j] == 'R' && white > 0) {
                    return false;
                }
                if (grid[i][j] == 'W' && red > 0) {
                    return false;
                }
            }
        }
        return true;
    }

    static boolean solve(int i, int j) {
        if (i == n) {
            return valid();
        }
        if (j == m) {
            return solve(i + 1, 0);
        }
        if (grid[i][j] != '.') {
            return solve(i, j + 1);
        }
        grid[i][j] = 'R';
        if (solve(i, j + 1)) {
            return true;
        }
        grid[i][j] = 'W';
        if (solve(i, j + 1)) {
            return true;
        }
        grid[i][j] = '.';
        return false;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(in.readLine());
        for (int i = 0; i < t; i++) {
            StringTokenizer st = new StringTokenizer(in.readLine());
            n = Integer.parseInt(st.nextToken());
            m = Integer.parseInt(st.nextToken());
            grid = new char[n][m];
            for (int j = 0; j < n; j++) {
                String line = in.readLine();
                for (int k = 0; k < m; k++) {
                    grid[j][k] = line.charAt(k);
                }
            }
            if (solve(0, 0)) {
                System.out.println("YES");
                for (int j = 0; j < n; j++) {
                    System.out.println(new String(grid[j]));
                }
            } else {
                System.out.println("NO");
            }
        }
    }
}
