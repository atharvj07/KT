#include <bits/stdc++.h>
#define ll long long
using namespace std;
typedef pair<ll,ll> pi;

const ll MAXN = 1e5 + 5;
ll n;
vector<pi> vec;
vector<pi> res;

int main() {
    cin >> n;
    for (ll i = 0; i < n; ++i) {
        string s; ll x; cin >> s >> x;
        vec.push_back({x,s=="insert"});
    }

    for (ll i = 0; i < vec.size(); ++i) {
        if (vec[i].second) res.push_back(vec[i]);
        else {
            if (res.back().first > vec[i].first) return cout << "NO" << endl, 0;
            res.pop_back();
        }
    }
    cout << res.size() << endl;
    for (auto p: res) cout << (p.second?"insert":"removeMin") << " " << p.first << endl;
    return 0;
}
