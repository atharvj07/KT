#include <bits/stdc++.h>
#define ll long long
using namespace std;
ll p = 1e9 + 7;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n;
  cin >> n;
  ll ans = 0, pref[n];
  map<ll, ll> cnt;
  cin >> pref[0];
  cnt[pref[0]]++;
  for (int i = 1; i < n; i++) {
    cin >> pref[i];
    pref[i] += pref[i - 1];
    cnt[pref[i]]++;
  }
  for (int i = 0; i < n - 1; i++) {
    ll suf = 0;
    for (int j = i + 1; j < n; j++) {
      suf += pref[j] - (i > 0 ? pref[i - 1] : 0);
      ans += cnt[suf] - (i > 0 ? 1 : 0);
      cnt[suf]--;
      ans %= p;
    }
    cnt[pref[i]]++;
  }
  cout << ans << endl;
}
