import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (sc.hasNext()) {
            int n = sc.nextInt(), r = sc.nextInt();
            if (n == 0 && r == 0) break;
            int top = 1;
            for (int i = 0; i < r; i++) {
                int p = sc.nextInt(), c = sc.nextInt();
                top = (top + p - c - 1) % n + 1;
            }
            System.out.println(top);
        }
    }
}
