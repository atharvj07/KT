#include <iostream>
#include <set>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    multiset<int> s;
    for (int i = 0; i < 2 * n; i++) {
      int x;
      cin >> x;
      s.insert(x);
    }
    bool flag = true;
    for (int x : s) {
      if (x % 2 == 0) {
        flag = false;
        break;
      }
      s.erase(s.find(x));
      if (s.empty()) {
        break;
      }
      x = *s.begin();
      s.erase(s.find(x));
      if (x % 2 == 0) {
        flag = false;
        break;
      }
    }
    cout << (flag ? "Yes\n" : "No\n");
  }
}
