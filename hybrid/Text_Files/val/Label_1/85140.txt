#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 2e5 + 10;
const int maxm = 2e5 + 10;
int main() {
    int n, m;
    cin >> n >> m;
    string s;
    cin >> s;
    int vis1[maxn], vis2[maxn];
    while(m--) {
        int x1, y1, len;
        cin >> x1 >> y1 >> len;
        bool ok = 1;
        memset(vis1, 0, sizeof(vis1));
        memset(vis2, 0, sizeof(vis2));
        for(int i = 0; i< len; i++) {
            if(vis1[s[x1 + i - 1]] && vis1[s[x1 + i - 1]] != vis2[s[y1 + i - 1]] ||
                !vis1[s[x1 + i - 1]] && vis2[s[y1 + i - 1]]) ok = 0;
            vis1[s[x1 + i - 1]] = i + 1;
            vis2[s[y1 + i - 1]] = i + 1;
        }
        cout<<(ok ? "YES":"NO")<<endl;
    }
    return 0;
}
