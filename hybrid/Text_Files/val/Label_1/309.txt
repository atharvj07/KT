for _ in range(int(input())):
    n, c = input().split()
    s = input()
    if all(map(lambda x:x==s[0], s)):print(0);continue
    ans = []; i=2; x = [];
    while 1:
        cnt = 0;
        for j in range(0,n,i):
            if s[j]!=c:cnt+=1
        if cnt==0:break
        if i>n//2:ans.append(n);x.append(n);break
        if cnt*i<=n:ans.append(cnt);x.append(i);
        i+=1
    print(len(ans));print(*ans);
    print(*x)
