#include <iostream>
#include <vector>
#include <queue>
using namespace std;
#define MAXN 200005
#define INF 1e9
#define ll long long
int n, k, a[MAXN], ans = INF;
bool vis[MAXN];
vector<int> g[MAXN], ord;
void dfs(int u, int p) {
    vis[u] = 1;
    ord.push_back(u);
    if (ord.size() > k)
        ans = min(ans, a[u]);
    for (int v : g[u])
        if (v != p) dfs(v, u);
}
int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1, u, v; i < n; i++) {
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    for (int i = 1; i <= n; i++)
        if (!vis[i]) dfs(i, i);
    cout << ans << endl;
}
