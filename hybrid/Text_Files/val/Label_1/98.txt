n = int(input())
x1, x2 = map(int, input().split())

# Construct a map of lines in the form (k, b)
lines = {}
for _ in range(n):
    k, b = map(int, input().split())
    lines[k] = b

# Check if any two lines intersect inside the strip
for k1 in lines:
    for k2 in lines:
        if k1 != k2:
            x = (lines[k2] - lines[k1]) / (k1 - k2)
            if x1 < x < x2:
                print("YES")
                exit()

# If no intersection found, print "NO"
print("NO")
