#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(2 * n);
    for (int i = 0; i < 2 * n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    bool ok = 1;
    for (int i = 0; i < n; i++) { if (a[i] > a[n + i]) ok = 0; }
    cout << (ok ? "YES" : "NO") << '\n';
    return 0;
}
