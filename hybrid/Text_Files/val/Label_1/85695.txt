#include <iostream>
#include <string>

using namespace std;

int main() {
  string time;
  cin >> time;
  int h = stoi(time.substr(0, 2));
  int m = stoi(time.substr(3, 2));
  int ans = 0;
  if (m != 0) {
    ans = 60 - m;
    h++;
    if (h == 24) h = 0;
  }
  if (h < 10) {
    if (m % 10 == 0) ans += 10 - m;
    else ans += 10 + m;
  }
  cout << ans << endl;
  return 0;
}
