#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    if (n == 1) {
        cout << "Possible\n1" << endl;
    } else if (n == 2) {
        if (a[0] == a[1]) {
            cout << "Possible\n1 1" << endl;
        } else {
            cout << "Impossible" << endl;
        }
    } else {
        if (a[0] > n - 1 || a[1] > n - 1) {
            cout << "Impossible" << endl;
        } else {
            cout << "Possible\n";
            for (int i = 0; i < n; i++) {
                if (i == n - 1) {
                    cout << n - a[0] << endl;
                } else {
                    cout << n - a[0] - a[1] << " ";
                    a[1]--;
                }
            }
        }
    }
    return 0;
}
