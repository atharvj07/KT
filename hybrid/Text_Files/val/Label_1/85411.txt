#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, m; cin >> n >> m;
  string s, t; cin >> s >> t;
  vector<int> pos[26];
  for (int i = 0; i < n; i++)
    pos[s[i] - 'a'].push_back(i);
  int ans = 0;
  for (int i = 0; i < m; i++) {
    if (pos[t[i] - 'a'].empty()) {
      cout << -1 << endl;
      return 0;
    }
    int curr = *lower_bound(pos[t[i] - 'a'].begin(), pos[t[i] - 'a'].end(), ans + 1);
    ans = max(ans, curr);
  }
  cout << ans << endl;
  return 0;
}
