import sys
mod = int(1e9) + 7

n, k = map(int, sys.stdin.readline().split())
w = list(map(int, sys.stdin.readline().split()))

dp = [[0] * (k + 1) for _ in range(n + 1)]
dp[0][0] = 1
for i in range(1, n + 1):
    for j in range(k + 1):
        dp[i][j] = dp[i - 1][j] << 1
        if j > 0:
            dp[i][j] += dp[i - 1][j - 1] * w[i - 1]
        dp[i][j] %= mod
print(dp[n][k])
