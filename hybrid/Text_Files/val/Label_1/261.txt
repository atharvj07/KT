import sys

# Read input from stdin
n = int(sys.stdin.readline())
adj = [[int(x) for x in sys.stdin.readline().split()] for _ in range(n)]

# Create a matrix to store the number of ways to connect two points
dp = [[0 for _ in range(n)] for _ in range(n)]

# Base case: dp[i][i] = 1 for all i
for i in range(n):
    dp[i][i] = 1

# Iterate over the points
for i in range(n):
    for j in range(i+1, n):
        # If two points can be connected directly, then dp[i][j] = dp[i][j-1] + dp[j-1][j]
        if adj[i][j] == 1:
            dp[i][j] = (dp[i][j-1] + dp[j-1][j]) % (10**9 + 7)
        # If two points cannot be connected directly, then dp[i][j] = dp[i][j-1] + dp[j-1][j] - dp[i][j-1] - dp[j-1][j]
        else:
            dp[i][j] = (dp[i][j-1] + dp[j-1][j] - dp[i][j-1] - dp[j-1][j]) % (10**9 + 7)

# Print the number of ways to connect all points modulo 10^9 + 7
print(dp[0][n-1])
