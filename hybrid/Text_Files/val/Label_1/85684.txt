#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
typedef long long int lli;

const lli infinito = 1e12;

struct Arista { lli to, w; };
struct Town { vector<Arista> edges; };

vector<Town> G;
vector<lli> dist;
vector<bool> vis;

lli dijkstra(lli s, lli n) {
    dist = vector<lli>(n, infinito);
    vis = vector<bool>(n, false);
    priority_queue<pair<lli, lli>, vector<pair<lli, lli>>, greater<pair<lli, lli>>> pq;
    pq.push(make_pair(0, s));
    while (!pq.empty()) {
        lli u = pq.top().second;
        lli d = pq.top().first;
        pq.pop();
        if (vis[u]) continue;
        vis[u] = true;
        dist[u] = d;
        for ( auto v : G[u].edges) {
            if (dist[v.to] > dist[u] + v.w) {
                dist[v.to] = dist[u] + v.w;
                pq.push(make_pair(dist[v.to], v.to));
            }
        }
    }
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    lli n, m;
    cin >> n >> m;
    G = vector<Town>(n);
    for (lli i = 0; i < m; ++i) {
        lli a, b, c;
        cin >> a >> b >> c;
        --a; --b;
        G[a].edges.push_back({b, c});
        G[b].edges.push_back({a, c});
    }
    vector<lli> a(n);
    for (auto& ai : a) cin >> ai;
    vector<lli> ans(n);
    for (lli i = 0; i < n; ++i) {
        dijkstra(i, n);
        ans[i] = infinito;
        for (lli j = 0; j < n; ++j) ans[i] = min(ans[i], 2LL * dist[j] + a[j]);
    }
    for (auto& ans_i : ans) cout << ans_i << ' ';
}
