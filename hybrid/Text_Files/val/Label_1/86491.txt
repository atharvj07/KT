import java.util.*;

class Main {
    static int n, m;

    static class Pair implements Comparable<Pair> {
        int i, j;

        Pair(int i, int j) {
            this.i = i;
            this.j = j;
        }

        @Override
        public int compareTo(Pair p) {
            return this.i - p.i;
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        String[] s = new String[n];
        Set<Character>[] mp = new Set[m];
        for (int i = 0; i < m; i++) mp[i] = new HashSet<>();
        for (int i = 0; i < n; i++) {
            s[i] = in.next();
            for (int j = 0; j < m; j++) mp[j].add(s[i].charAt(j));
        }
        for (int j = 0; j < m; j++) {
            if (mp[j].size() == n) continue;
            PriorityQueue<Pair> pq = new PriorityQueue<>();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = in.nextInt();
                pq.add(new Pair(a[i], i));
            }
            int ans = 0;
            for (int i = 0; i < n; i++) {
                Pair p = pq.poll();
                char c = s[p.i].charAt(j);
                for (char d = 'a'; d <= 'z'; d++) {
                    if (d != c && !mp[j].contains(d)) {
                        ans += p.i;
                        s[p.i] = s[p.i].substring(0, j) + d + s[p.i].substring(j + 1);
                        break;
                    }
                }
            }
            System.out.println(ans);
            return;
        }
        System.out.println(0);
    }
}
