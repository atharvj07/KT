MOD = 1000000007

def power(x, y):
    res = 1
    while y > 0:
        if y % 2:
            res = res * x % MOD
        y //= 2
        x = x * x % MOD
    return res

def solve(k):
    dp = [[0] * 31 for _ in range(31)]
    dp[0][0] = 1
    for i in range(1, 31):
        for j in range(1, 31):
            dp[i][j] = (dp[i - 1][j] * j + dp[i - 1][j - 1]) % MOD
    
    ans = 0
    last = 0
    for i in range(30, -1, -1):
        if k & (1 << i):
            for j in range(1, 31):
                ans = (ans + dp[last + i][j]) % MOD
            last += 1
    ans = (ans + 1) % MOD
    return ans

k = int(input())
print(solve(k))
