import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int x = in.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextInt();
        }
        Arrays.sort(arr);
        int[] ans = new int[n + 1];
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] < x) {
                cnt++;
            }
            ans[cnt]++;
        }
        for (int i = 0; i < n + 1; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
