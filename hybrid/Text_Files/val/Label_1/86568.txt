import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] monsters = new int[n];
        for (int i = 0; i < n; i++) {
            monsters[i] = in.nextInt();
        }
        int[] special = new int[m];
        for (int i = 0; i < m; i++) {
            special[i] = in.nextInt();
        }
        Arrays.sort(monsters);
        Arrays.sort(special);
        int ans = 0;
        int l = 0;
        int r = 0;
        while (l < n && r < m) {
            if (monsters[l] <= special[r]) {
                ans++;
                l++;
                r++;
            } else {
                r++;
            }
        }
        System.out.println(ans);
    }
}
