import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Scanner;

public class Main {

    static int[] arr, len, id;

    static class Pair implements Comparable<Pair> {
        int l, r, idx, d;

        Pair(int l, int r, int idx, int d) {
            this.l = l;
            this.r = r;
            this.idx = idx;
            this.d = d;
        }

        @Override
        public int compareTo(Pair o) {
            return r - o.r;
        }
    }

    static void modify(int idx, int v) {
        for (; idx < len.length; idx = (idx + 1) << 1) {
            len[idx] = Math.max(len[idx], v);
            id[idx] = -1;
        }
    }

    static int get(int idx, int d) {
        int ans = -1;
        for (idx += len.length >> 1; idx > 0; idx >>= 1) {
            if (id[idx] == d) {
                ans = len[idx];
            } else if (id[idx] < 0) {
                id[idx] = d;
                len[idx] = 0;
            }
        }
        return ans;
    }

    public static void main(String[] args) throws IOException {
        Scanner in = new Scanner(new BufferedReader(new InputStreamReader(System.in)));
        char[] s = in.next().toCharArray();
        int n = s.length;

        int m = in.nextInt();
        int[][] queries = new int[m][2];

        for (int i = 0; i < m; i++) {
            queries[i][0] = in.nextInt();
            queries[i][1] = in.nextInt();
        }

        Arrays.sort(queries, (a, b) -> a[0] - b[0]);

        StringBuilder sb = new StringBuilder(new String(s));
        len = new int[n << 2];
        id = new int[n << 2];
        Arrays.fill(len, Integer.MAX_VALUE);
        Arrays.fill(id, -1);

        int qIdx = 0;
        int last = 0;
        for (int i = 0; i < n; i++) {
            while (qIdx < m && queries[qIdx][0] <= i) {
                int l = queries[qIdx][0];
                int r = l + queries[qIdx][0];
                int d = queries[qIdx++][1];
                if (l > 0) {
                    modify(get(l - 1, d) + 1, d);
                }
                modify(0, d);
            }

            int curD = get(i, last);
            if (curD < 0) {
                curD = last;
            }

            if (i > 0) {
                sb.setCharAt(i - curD, s[i]);
            }
            last = curD;
        }

        System.out.println(sb);
    }
}
