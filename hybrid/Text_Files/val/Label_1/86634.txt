import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        for (int i = 0; i < n; i++) {
            String s = scanner.next();
            boolean isNatural = true;
            int left = 0;
            int right = 0;
            for (char c : s.toCharArray()) {
                if (c == 'L') {
                    if (right == 1) {
                        isNatural = false;
                        break;
                    }
                    left = 1;
                    right = 0;
                } else if (c == 'R') {
                    if (left == 1) {
                        isNatural = false;
                        break;
                    }
                    left = 0;
                    right = 1;
                }
            }
            System.out.println(isNatural ? "Yes" : "No");
        }
    }
}
