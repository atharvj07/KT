#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll n, phi[500005], mx[500005], p[500005], ans[500005];
vector<ll> divv[500005];
int main() {
  cin >> n;
  for (ll i = 2; i <= n; i++) {
    if (!p[i]) {
      p[i] = i;
      phi[i] = i - 1;
      divv[i].push_back(i);
      for (ll j = i * i; j <= n; j += i) {
        p[j] = i;
        if (i == p[j / i]) phi[j] = phi[j / i] * i;
        divv[j].push_back(i);
      }
    }
  }
  for (ll i = 2; i <= n; i++) {
    ll tmp = 0;
    for (auto &x : divv[i]) { tmp = max(tmp, x); }
    mx[i] = tmp;
  }
  for (ll i = 2; i <= n; i++) {
    if (phi[i] == 0) continue;
    for (ll j = i + i; j <= n; j += i) {
      phi[j] -= phi[j] / i;
    }
  }
  for (ll i = 2; i <= n; i++) {
    if (i == 2) ans[i] = 1;
    if (i > 2) {
      ll tmp = phi[i];
      for (auto &x : divv[i]) {
        tmp -= phi[i - tmp / x];
        tmp += phi[i - tmp / mx[i - tmp / x]];
      }
      ans[i] = tmp;
    }
    cout << ans[i] << ' ';
  }
  cout << endl;
}
