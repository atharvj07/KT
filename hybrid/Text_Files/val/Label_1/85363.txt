#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m,k;
    cin>>n>>m>>k;
    if(2*k>n*m){
        puts("NO");
        return 0;
    }
    puts("YES");
    int a1=k,a2=k,a3=n*m-k;
    int b1=0,b2=m-k,b3=0;
    if(k*2==n*m){
        a1=a2=a3=n*m/3;
        b2=m-a2;
        if(m-a2>a3){
            b1=a3+1;
            b2-=b1;
            a3=0;
        }
    }
    cout<<a1<<" "<<b1<<'\n'<<a2<<" "<<b2<<'\n'<<a3<<" "<<b3<<'\n';
    return 0;
}
