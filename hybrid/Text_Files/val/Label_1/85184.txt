#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int M = 1e9+7;
int dp[105][105];
ll pow10[105];
int n, k, m;

ll solve(int pos, int rem) {
    if(pos == n) return (rem == 0);
    int &ret = dp[pos][rem];
    if(~ret) return ret;
    ret = 0;
    for(int d = 0; d <= 9; d++) {
        if((rem+d)%k == 0) ret = (ret + solve(pos+1, (rem+d)%k)) % M;
        else ret = (ret + solve(pos+1, (rem+d)%k)) % M;
    }
    return ret;
}

int main() {
    cin >> n >> k >> m;
    memset(dp, -1, sizeof dp);
    pow10[0] = 1;
    for(int i = 1; i <= 100; i++) pow10[i] = pow10[i-1]*10;
    cout << solve(0,0) << "\n";
}
