#include <iostream>
#include <cstdio>
#include <string>

using namespace std;

int main() {
    int m, s;
    cin >> m >> s;
    if (s > m * 9) {
        puts("-1 -1");
        return 0;
    }
    string ss(m, '0');
    int rest = s;
    for (int i = m - 1; i >= 0; i--) {
        int x = min(9, rest);
        ss[i] = '0' + x;
        rest -= x;
    }
    cout << ss << ' ';
    rest = s;
    ss.assign(m, '0');
    for (int i = 0; i < m; i++) {
        int x = min(9 - (ss[i] - '0'), rest);
        ss[i] = '0' + x;
        rest -= x;
    }
    cout << ss << endl;
    return 0;
}
