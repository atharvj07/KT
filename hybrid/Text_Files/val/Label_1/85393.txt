#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

const int MAXN = 5000, MAXQ = 100000;
int n, q;
ll a[MAXN+5];

struct SegTree {
    int l, r;
    ll val;
} tree[4*MAXN+5];

int build(int id, int l, int r) {
    tree[id] = {l, r, a[r]};
    if (l < r) {
        int mid = (l + r) >> 1;
        tree[id].val |= build(id<<1, l, mid);
        tree[id].val |= build(id<<1|1, mid+1, r);
    }
    return tree[id].val;
}

ll query(int id, int l, int r) {
    if (tree[id].l >= l && tree[id].r <= r) return tree[id].val;
    if (tree[id].r < l || tree[id].l > r) return 0LL;
    return tree[id].val | query(id<<1, l, r) | query(id<<1|1, l, r);
}

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    cin >> n;
    for (int i=1; i<=n; i++) cin >> a[i];
    build(1, 1, n);
    cin >> q;
    while (q--) {
        int l, r;
        cin >> l >> r;
        cout << query(1, l, r) << '\n';
    }
}
