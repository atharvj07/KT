#include <bits/stdc++.h>
using namespace std;
int a[50010], x[50010], ans = 0, n, b, c;
char s[50010], p[50010];
inline int read() {
  int f = 1, x = 0;
  char c;
  c = getchar();
  while (c < '0' || c > '9') {
    if (c == '-')
      f = -1;
    c = getchar();
  }
  while (c >= '0' && c <= '9') {
    x = x * 10 + c - '0';
    c = getchar();
  }
  return f * x;
}
int main() {
  cin >> s >> n;
  for (int i = 1; i <= n; i++) {
    cin >> p >> b >> c;
    for (int j = 0; j < strlen(p); j++) x[j]++;
    for (int j = strlen(p) - b; j < strlen(p) - c; j++) a[j]++;
  }
  int len = strlen(s);
  for (int i = 0; i < len; i++) {
    int l = i;
    int r = i;
    ans++;
    for (int j = i; j < len; j++) {
      x[s[j] - 'a']--;
      a[j]++;
      if (x[s[j] - 'a'] < 0) ans -= 2;
      if (x[s[j] - 'a'] >= 0 && a[j] > x[s[j] - 'a']) ans--;
      if (ans < 0) ans = 0;
    }
  }
  cout << ans << endl;
  return 0;
}
