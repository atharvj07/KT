import java.util.Scanner;
import java.math.*;
import java.util.Arrays;
public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int N = sc.nextInt();
    double[] A = new double[N];
    for (int i=0; i<N; i++) {
      A[i] = sc.nextDouble();
    }
    Arrays.sort(A);
    long ans = 0;
    for (int i=0; i<N; i++) {
      for (int j=i+1; j<N; j++) {
        if (Math.abs(A[i] * A[j] / Math.round(A[i] * A[j])) < 1e-9) ans++;
      }
    }
    System.out.println(ans);
  }
}
