#include<bits/stdc++.h>
using namespace std;
const int maxn = 3e5 + 10;
int n, m, k, q, a[maxn], lst[maxn], last[22], sum[maxn];
long long ans;
int main() {
    cin >> n >> m >> k;
    for(int i = 1; i <= n; i++) cin >> a[i];
    for(int i = 1; i <= n; i++) sum[i] = sum[i - 1] + a[i];
    memset(lst, 0x3f, sizeof(lst));
    lst[n + 1] = n + 1;
    for(int i = n; i >= 1; i--) {
        lst[i] = lst[i + 1];
        if((i + m - 1) / m < (i + 1 + m - 1) / m) lst[i] = i;
    }
    for(int i = 1; i <= n; i++) {
        ans = max(ans, 1ll * sum[i - 1] - 1ll * k * ((i + m - 1) / m));
        if(last[i / m] != -1) ans = max(ans, 1ll * (sum[i - 1] - sum[last[i / m]]) - 1ll * k * ((i - last[i / m]) / m));
        last[i / m] = i;
    }
    cout << ans << endl;
    return 0;
}
