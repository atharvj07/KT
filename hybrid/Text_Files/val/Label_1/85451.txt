#include<bits/stdc++.h>
using namespace std;
int t,n,i,a[1005],b[1005];
int main()
{
  cin>>n;
  for(i=1;i<=n;i++)
    cin>>a[i];
  for(i=1;i<=n;i++)
  {
    reverse(a,a+i+1);
    for(t=0;t<13;t++)
    {
      if(a[1]==0)
        for(int j=1;j<=n;j++)
          if(a[j]!=0){
            t=j-1;
            break;
          }
      for(int j=1;j<=n;j++)
        b[j]=a[j];
      for(int j=1;j<=t;j++)
      {
        b[j+1]=a[j];
        if(b[1]==0)
          for(int k=1;k<=n;k++)
            if(b[k]!=0){
              t=k-1;
              break;
            }
      }
      b[1]+=1;
      for(int j=1;j<=n;j++)
        if(b[j]==10)
        {
          b[j]=0;
          b[j+1]+=1;
        }
      bool f=0;
      for(int j=1;j<=n;j++)
        if(b[j]!=a[j])
          f=1;
      if(!f)
        break;
    }
    for(t=1;t<=n;t++)
      a[t]=b[t];
    reverse(a,a+n+1);
  }
  for(i=1;i<=n;i++)
    cout<<a[i];
  return 0;
}
