#include <bits/stdc++.h>
using namespace std;

int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  int n, m;
  cin >> n >> m;
  string s;
  cin >> s;
  while (m--) {
    int x, y, l;
    cin >> x >> y >> l;
    int ax[26] = {0}, ay[26] = {0};
    bool ok = 1;
    for (int i = 0; i < l && ok; i++) {
      if (ax[s[x + i - 1] - 'a'] != ay[s[y + i - 1] - 'a']) ok = 0;
      ax[s[x + i - 1] - 'a'] = i + 1;
      ay[s[y + i - 1] - 'a'] = i + 1;
    }
    cout << (ok ? "YES\n" : "NO\n");
  }
  return 0;
}
