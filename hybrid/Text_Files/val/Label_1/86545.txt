import java.util.Queue;
import java.util.LinkedList;
import java.util.Scanner;
public class Main {
  public static char[][] grid;
  public static int n, m;
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int t = sc.nextInt();
    while (t-- > 0) {
      n = sc.nextInt();
      m = sc.nextInt();
      grid = new char[n][m];
      for (int i = 0; i < n; i++)
        grid[i] = sc.next().toCharArray();
      Queue<Pair> q = new LinkedList<>();
      boolean fl = false;
      for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
          if (grid[i][j] == '.') {
            q.add(new Pair(i, j));
            fl = true;
          }
        }
      }
      if (!fl) {
        System.out.println("YES");
        for (int i = 0; i < n; i++)
          System.out.println(new String(grid[i]));
        continue;
      }
      Queue<Pair> q1 = new LinkedList<>();
      Queue<Pair> q2 = new LinkedList<>();
      boolean ok = true;
      while (!q.isEmpty()) {
        Pair p = q.poll();
        int x = p.x, y = p.y;
        if (x - 1 >= 0 && grid[x - 1][y] == 'R')
          q1.add(new Pair(x - 1, y));
        if (x + 1 < n && grid[x + 1][y] == 'W')
          q2.add(new Pair(x + 1, y));
        if (y - 1 >= 0 && grid[x][y - 1] == 'R')
          q1.add(new Pair(x, y - 1));
        if (y + 1 < m && grid[x][y + 1] == 'W')
          q2.add(new Pair(x, y + 1));
        ok &= !(x - 1 >= 0 && grid[x - 1][y] == 'W' ||
            x + 1 < n && grid[x + 1][y] == 'R' ||
            y - 1 >= 0 && grid[x][y - 1] == 'W' ||
            y + 1 < m && grid[x][y + 1] == 'R');
      }
      while (!q1.isEmpty())
        q.add(q1.poll());
      while (!q2.isEmpty())
        q.add(q2.poll());
      while (!q.isEmpty()) {
        Pair p = q.poll();
        int x = p.x, y = p.y;
        if (ok) {
          grid[x][y] = 'R';
          if (x - 1 >= 0 && grid[x - 1][y] == '.')
            q.add(new Pair(x - 1, y));
          if (x + 1 < n && grid[x + 1][y] == '.')
            q.add(new Pair(x + 1, y));
          if (y - 1 >= 0 && grid[x][y - 1] == '.')
            q.add(new Pair(x, y - 1));
          if (y + 1 < m && grid[x][y + 1] == '.')
            q.add(new Pair(x, y + 1));
          ok &= !(x - 1 >= 0 && grid[x - 1][y] == 'W' ||
              x + 1 < n && grid[x + 1][y] == 'R' ||
              y - 1 >= 0 && grid[x][y - 1] == 'W' ||
              y + 1 < m && grid[x][y + 1] == 'R');
        } else {
          grid[x][y] = 'W';
          if (x - 1 >= 0 && grid[x - 1][y] == '.')
            q.add(new Pair(x - 1, y));
          if (x + 1 < n && grid[x + 1][y] == '.')
            q.add(new Pair(x + 1, y));
          if (y - 1 >= 0 && grid[x][y - 1] == '.')
            q.add(new Pair(x, y - 1));
          if (y + 1 < m && grid[x][y + 1] == '.')
            q.add(new Pair(x, y + 1));
          ok &= !(x - 1 >= 0 && grid[x - 1][y] == 'R' ||
              x + 1 < n && grid[x + 1][y] == 'W' ||
              y - 1 >= 0 && grid[x][y - 1] == 'R' ||
              y + 1 < m && grid[x][y + 1] == 'W');
        }
      }
      if (ok) {
        System.out.println("YES");
        for (int i = 0; i < n; i++)
          System.out.println(new String(grid[i]));
      } else
        System.out.println("NO");
    }
  }
  static class Pair {
    int x, y;
    Pair(int x, int y) {
      this.x = x;
      this.y = y;
    }
  }
}
