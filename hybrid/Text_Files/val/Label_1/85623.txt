#include <bits/stdc++.h>
typedef long long ll;
using namespace std ;
int a[5005];
int main() {
    int n; cin >> n;
    int m = -1e9;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        if (i && a[i] < a[i - 1]) m = max(m, i - 1);
    }
    if (~m) {
        int s = 0;
        for (int i = 0; i < n; i++) s += a[i] > a[m];
        cout << s << ' ' << n - 1 - m << '\n';
    } else {
        cout << 0 << ' ' << 0 << '\n';
    }
    return 0 ;
}
