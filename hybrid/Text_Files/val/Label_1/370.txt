for t in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    i, s, x = 0, 0, 1
    while i < n:
        j, y = i, x
        while j < n and x == y:
            if x == 1:
                s = max(s, sum(a[i:j + 1]))
            j, y = j + 1, -y
        i = j
        x = -x
    print(s)
