import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.println(compare(input.nextLong(), input.nextLong()));
    }
    static String compare(long x, long y) {
        int xLen = (int) Math.log10(x) + 1;
        int yLen = (int) Math.log10(y) + 1;
        if (xLen > yLen) return ">";
        else if (yLen > xLen) return "<";
        long xPow = x, yPow = y;
        while (xPow > 0 && xPow != 1) {
            xPow = Math.floorDiv(xPow, 10);
            yPow *= 10;
        }
        if (xPow == 1 && yPow == 1) return "=";
        if (xPow > yPow) return ">";
        else return "<";
    }
}
