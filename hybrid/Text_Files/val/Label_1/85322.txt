#include <iostream>
#include <string>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;
    bool ok = true;
    for (int i = 0; i < n; i++) {
      if (s[i] != s[0] && s[i] != s[i - 1]) {
        ok = false;
        break;
      }
    }
    std::cout << (ok ? "YES" : "NO") << std::endl;
  }
  return 0;
}
