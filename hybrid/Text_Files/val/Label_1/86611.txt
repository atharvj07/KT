import java.util.*;
import java.io.*;

public class Main{
  static final int mod = (int) (1e9 + 7);
  
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    long n = sc.nextLong();
    long m = sc.nextLong();
    long ans;
    if (n == m) {
      ans = ((n + mod - 1) % mod * (n + mod - 2) % mod) % mod;
    } else {
      ans = (m + mod - 1) % mod * (n + mod - 1) % mod;
    }
    System.out.println(ans);
   }
}
