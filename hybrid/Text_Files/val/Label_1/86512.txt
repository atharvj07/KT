import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), q = in.nextInt();
        char[] s = in.next().toCharArray();
        SegmentTree st = new SegmentTree(n);
        for (int i = 0; i < n; i++) {
            st.set(i, s[i] == '<' ? -1 : 1);
        }
        for (int i = 0; i < q; i++) {
            int l = in.nextInt() - 1, r = in.nextInt() - 1;
            st.update(l, r);
            System.out.println(st.max);
        }
    }

    static class SegmentTree {
        int n, max;
        int[] t, lz;

        SegmentTree(int n) {
            this.n = n;
            t = new int[4 * n + 4];
            lz = new int[4 * n + 4];
            max = 0;
        }

        void set(int p, int v) {
            t[p + n] = lz[p + n] = v;
            if (v > 0) max = n;
        }

        void push(int p) {
            if (lz[p] > 0) {
                t[2 * p + 1] = lz[2 * p + 1];
                lz[2 * p + 1] = lz[p];
                t[2 * p + 2] = lz[2 * p + 2];
                lz[2 * p + 2] = lz[p];
                lz[p] = 0;
            }
        }

        void update(int a, int b) {
            update(a, b, 1, 0, n - 1);
        }

        void update(int a, int b, int p, int l, int r) {
            if (a <= l && r <= b) {
                t[p] = -t[p];
                lz[p] = -lz[p];
                max = Math.max(max, -max);
            } else {
                push(p);
                int mid = (l + r) / 2;
                if (a <= mid) update(a, b, 2 * p + 1, l, mid);
                if (b > mid) update(a, b, 2 * p + 2, mid + 1, r);
                t[p] = t[2 * p + 1] + t[2 * p + 2];
            }
        }
    }
}
