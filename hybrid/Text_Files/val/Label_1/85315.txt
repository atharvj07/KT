#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 500005;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(MAXN), b(MAXN);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < m; i++) cin >> b[i];
  int cnt = 0, day = 0;
  while (cnt < k) {
    day++;
    if (a[(day - 1) % n] != b[(day - 1) % m]) cnt++;
  }
  cout << day << endl;
  return 0;
}
