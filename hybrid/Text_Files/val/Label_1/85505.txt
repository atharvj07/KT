#include <iostream>
#include <cmath>
#include <iomanip>
using namespace std;

int main() {
    int k, q; cin >> k >> q;
    while(q--) {
        double p; cin >> p;
        double ans = log(1 - p / 2000) / log(1 - p / k);
        ans = ceil(ans); ans = max(ans, 1.0);
        cout << (int)ans << '\n';
    }
    return 0;
}
