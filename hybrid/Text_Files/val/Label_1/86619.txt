import java.util.Scanner;

public class Main {
    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = sc.nextInt();
        }

        long ans = 1;
        for (int i = 0; i < k; i++) {
            ans = (ans * comb(n, a[i])) % MOD;
        }

        ans = (ans * pow(n, a[0])) % MOD;
        for (int i = 1; i < k; i++) {
            ans = (ans * pow(n - a[i], a[i] - a[i - 1])) % MOD;
        }

        System.out.println(ans);
    }

    private static long comb(int n, int r) {
        long ans = 1;
        for (int i = n; i > n - r; i--) {
            ans = (ans * i) % MOD;
        }
        for (int i = 1; i <= r; i++) {
            ans = (ans * pow(i, MOD - 2)) % MOD;
        }
        return ans;
    }

    private static long pow(long x, long y) {
        long ans = 1;
        while (y > 0) {
            if ((y & 1) == 1) {
                ans = (ans * x) % MOD;
            }
            x = (x * x) % MOD;
            y >>= 1;
        }
        return ans;
    }
}
