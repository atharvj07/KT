#include<bits/stdc++.h>
using namespace std;
int n;
vector<int>p,q;
int32_t main(){
    cin>>n;
    if(n%2){cout<<"NO\nNO";return 0;}
    cout<<"YES\n";
    for(int i=1;i<=n;i++){
        if(i%2)p.push_back(i+1);
        else p.push_back(i-1);
    }
    for(int i=1;i<=n;i++){
        if(i%2)q.push_back(i+1);
        else q.push_back(i-2);
    }
    p[n-1]=n;
    q[n-1]=n+1;
    for(int i:p)cout<<i<<" ";cout<<"\n";
    for(int i:q)cout<<i<<" ";cout<<"\n";
}
