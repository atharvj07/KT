#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
    int k, q;
    cin >> k >> q;
    while (q--) {
        int p;
        cin >> p;
        double val = (double)k / p;
        double ans = log(1 - val) / log(0.5);
        ll day = ceil(ans);
        cout << day << endl;
    }
}
