#include <cstdio>
#include <vector>
using namespace std;
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n;
    scanf("%d", &n);
    vector<int> a(n);
    for (int i = 0; i < n; i++) { scanf("%d", &a[i]); }
    int ok = 1;
    vector<int> b(n);
    for (int i = 0; i < n; i++) {
      int r = (i + a[i % n] % n + n) % n;
      if (b[r] != 0) {
        ok = 0;
      }
      b[r] = 1;
    }
    printf("%s\n", ok ? "YES" : "NO");
  }
  return 0;
}
