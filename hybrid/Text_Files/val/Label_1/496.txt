import sys
input = sys.stdin.readline

def solve(a: int, b: int, m: int) -> tuple[bool, list[int]]:
    if (b - a) % m: return False, []
    r = (b - a) // m
    if r > m: return False, []
    l = [a]
    while l[-1] != b:
        l.append(l[-1] + r)
    return True, l

q = int(input())
for _ in range(q):
    a, b, m = map(int, input().split())
    ok, res = solve(a, b, m)
    if ok: print(len(res), *res)
    else: print(-1)
