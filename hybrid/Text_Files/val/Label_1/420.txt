from collections import deque
n = int(input())
s = input()
a = deque()
for i in s:
    if i == '(':
        a.append(i)
    else:
        if a:
            a.pop()
        else:
            a.append(')')
ans = len(a)
for i in s:
    if i == '(':
        ans -= 1
    elif a:
        a.pop()
        ans += 1
print(ans)
