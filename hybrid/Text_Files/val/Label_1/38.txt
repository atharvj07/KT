n, m = map(int, input().split())
a = list(map(int, input().split()))
ans = []
for _ in range(m):
    l, r = map(int, input().split())
    l -= 1
    r -= 1
    mn = 1e9
    for i in range(l, r + 1):
        for j in range(i + 1, r + 1):
            if a[i] == a[j]:
                mn = min(mn, j - i)
    if mn == 1e9:
        mn = -1
    ans.append(mn)
print(*ans, sep="\n")
