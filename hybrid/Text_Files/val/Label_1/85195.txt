#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> c(n + 1), pos[200005];
  for (int i = 1; i <= n; i++) {
    cin >> c[i];
    pos[c[i]].push_back(i);
  }
  int ans = n;
  for (int i = 1; i <= m; i++) {
    int l, r;
    cin >> l >> r;
    int cl = c[l], cr = c[r];
    if (cl == cr) continue;
    ans -= pos[cl].size() + pos[cr].size();
    for (int p : pos[cl]) c[p] = cr;
    pos[cl].clear();
    pos[cr].push_back(l);
    pos[cr].push_back(r);
  }
  cout << ans << endl;
}
