def dfs(v, d):
    if d == 0:
        return w[v][0], w[v][0]

    minimum = w[v][0]
    for u in range(n):
        if v != u and f[u] == v:
            cur, m = dfs(u, d - 1)
            s[v] += cur
            minimum = min(m, minimum)

    s[v] += w[v][0]
    return s[v], minimum

n, k = list(map(int, input().split()))
f = list(map(int, input().split()))
w = [[int(x) for x in input().split()] for i in range(n)]
s = [0] * n
for i in range(n):
    print(*dfs(i, k))
