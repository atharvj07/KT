for _ in range(int(input())):
    n,k=map(int,input().split());a=list(map(int,input().split()))
    a.sort(reverse=True)
    ans=sum(a[k:]);a=a[:k];b=[0]
    for i in a:
        l=0;r=len(b);
        while l!=r:
            m=(l+r)//2
            if b[m]<=i:l=m+1
            else:r=m-1
        b+=[i]
    ans+=sum(i-b[len(b)-1] for i in b);print(ans)
