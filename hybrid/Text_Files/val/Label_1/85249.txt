#include<bits/stdc++.h>
using namespace std;
int main() {
  int n; cin >> n;
  vector<int> a(n+1), g(n+1);
  for (int i=2;i<=n;i++) { cin >> a[i]; g[i] = i; }
  for (int i=2;i<=n;i++) for (int j=a[i];j>1;j--) if (!(i%j)) g[j] = i;
  set<pair<int,int>> s;
  for (int i=2;i<=n;i++) if (g[i] < g[a[i]]) s.insert({i, g[a[i]]});
  cout << s.size() << "\n";
  for (pair<int,int> i : s) { cout << i.first << " " << i.second << "\n"; }
}
