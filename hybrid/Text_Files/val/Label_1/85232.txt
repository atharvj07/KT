#include <iostream>
#include <vector>
#include <limits>
using ll = long long;

int main() {
    ll n; std::cin >> n;
    std::vector<ll> a(n);
    for (int i = 0; i < n; i++) std::cin >> a[i];
    ll ans = 0, sum = 0;
    for (int i = 0; i < n; i++) sum += a[i];
    sum += n - 1;
    ll avg = (sum + n - 1) / n;
    for (int i = 0; i < n; i++) ans = std::max(ans, (avg + a[i] - 1) / a[i]);
    std::cout << ans << std::endl;
    return 0;
}
