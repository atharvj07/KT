import sys

input = lambda: sys.stdin.readline()[:-1]

t = int(input())

for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))

    dp = [[0] * 2 for _ in range(n)]

    dp[0][0] = max(0, a[0])
    dp[0][1] = min(0, a[0])

    for i in range(1, n):
        dp[i][0] = max(dp[i - 1][0], dp[i - 1][1] + a[i])
        dp[i][1] = min(dp[i - 1][1], dp[i - 1][0] + a[i])

    print(max(dp[n - 1]))
