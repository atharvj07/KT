#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int i = 0; i < n; i++) {
    cin >> p[i];
  }
  int mn_swaps = n - 1;
  int cnt_swaps = 0;
  for (int i = 0; i < n; i++) {
    if (p[i] == i) {
      mn_swaps--;
    } else {
      cnt_swaps++;
    }
  }
  cout << mn_swaps << " " << cnt_swaps << endl;
  return 0;
}
