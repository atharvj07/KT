#include <bits/stdc++.h>

using namespace std;

const int maxn = 1e5 + 1;
const int INF = 1e9;

vector<pair<int, char>> adj[maxn];
int color[maxn];

int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    int u, v;
    char c;
    cin >> u >> v >> c;
    adj[u].push_back({v, c});
    adj[v].push_back({u, c});
  }
  int cnt[2] = {0, 0};
  queue<int> q;
  for (int i = 1; i <= n; i++) {
    if (color[i] == 0) {
      q.push(i);
      color[i] = 1;
      cnt[0]++;
    }
  }
  int ans = INF;
  vector<int> seq;
  while (!q.empty()) {
    int u = q.front();
    q.pop();
    int cur_color = color[u];
    int other_color = 1 - cur_color;
    vector<int> next_q;
    for (auto& edge : adj[u]) {
      int v = edge.first;
      char c = edge.second;
      if (color[v] == 0) {
        next_q.push_back(v);
        color[v] = other_color;
        cnt[other_color]++;
      } else if (color[v] == cur_color) {
        if (c == 'B') cnt[0]--;
        else cnt[1]--;
      }
    }
    if (cnt[0] == 0 || cnt[1] == 0) {
      if (ans > seq.size()) {
        ans = seq.size();
        seq = {u};
      }
      while (!q.empty()) {
        int v = q.front();
        q.pop();
        if (ans > seq.size()) {
          ans = seq.size();
          seq = {u, v};
        }
      }
    } else {
      seq.push_back(u);
      for (auto& v : next_q) q.push(v);
    }
  }
  if (ans == INF)
    cout << -1 << endl;
  else {
    cout << ans << endl;
    for (auto& v : seq) cout << v << ' ';
    cout << endl;
  }
  return 0;
}
