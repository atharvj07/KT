for _ in range(int(input())):
    x, y = map(int, input().split())
    ans = 0
    minx, miny, maxx, maxy = 0, 0, 0, 0
    if x < 0:
        ans += abs(x)
        minx = x
        maxx = 0
    else:
        maxx = x
    if y < 0:
        ans += abs(y)
        miny = y
        maxy = 0
    else:
        maxy = y
    ans += 2*max(abs(minx), abs(miny)) + abs(maxx - minx) + abs(maxy - miny)
    print(ans)
