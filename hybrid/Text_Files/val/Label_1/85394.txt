#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int N = 5e3 + 5;
ll a[N], f[N], mx[N];
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  int n, q;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = n; i >= 1; i--) {
    if (i == n)
      f[i] = a[n];
    else
      f[i] = a[i] ^ f[i + 1];
  }
  mx[n + 1] = -1;
  for (int i = n; i >= 1; i--) mx[i] = max(mx[i + 1], f[i]);
  cin >> q;
  while (q--) {
    int l, r;
    cin >> l >> r;
    cout << mx[l] << '\n';
  }
}
