#include <iostream>
#include <string>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        bool ok = true;
        char last = 'a';
        for (char c : s) {
            if (c < last) {
                ok = false;
                break;
            }
            last = c;
        }
        cout << (ok ? "YES" : "NO") << endl;
    }
}
