#include <bits/stdc++.h>

using namespace std;

int main() {
  string s;
  cin >> s;
  int h = 10 * (s[0] - '0') + s[1] - '0';
  int m = 10 * (s[3] - '0') + s[4] - '0';
  int ans = 0;
  while (h != m) {
    m++;
    ans++;
    if (m == 60) m = 0, h++;
    if (h == 24) h = 0;
  }
  cout << ans << endl;
  return 0;
}
