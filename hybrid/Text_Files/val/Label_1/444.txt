import sys

n, k = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))
e = [[] for _ in range(n+1)]
for _ in range(n-1):
    u, v = map(int, sys.stdin.readline().split())
    e[u] += [v]
    e[v] += [u]

ans = 1e9
vis = [False]*(n+1)
def dfs(node, par=-1):
    global ans
    vis[node] = True
    l = len(e[node])
    e[node].sort(key=lambda x: a[x-1])
    for i in range(len(e[node])):
        if not vis[e[node][i]] and e[node][i] != par:
            dfs(e[node][i], node)

dfs(1)
for i in range(1,n+1):
    if not vis[i]:
        dfs(i)
    l = len(e[i])
    e[i].sort(key=lambda x: a[x-1], reverse=True)
    ans = min(ans, min(a[i-1],a[e[i][l-1]-1]))

print(ans)
