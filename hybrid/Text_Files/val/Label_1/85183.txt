#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<char> status(n);
    for (int i = 0; i < n; i++) {
        cin >> status[i];
    }
    int ans = 0;
    bool allin = false;
    for (int i = 0; i < n; i++) {
        if (status[i] == 'F') {
            continue;
        }
        if (status[i] == 'A') {
            allin = true;
        }
        if (allin) {
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
