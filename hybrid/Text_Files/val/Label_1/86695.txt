import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] present = new int[n + 1];
        int count = 0;
        List<Integer> leaders = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            char sign = sc.next().charAt(0);
            int id = sc.nextInt();
            if (sign == '+') {
                if (present[id] == 0) {
                    count++;
                }
                present[id] = 1;
            } else {
                if (present[id] == 1) {
                    count--;
                }
                present[id] = 0;
            }
            if (count == 0) {
                leaders.clear();
            } else if (count > 0 && present[1] == 1) {
                leaders.add(1);
            }
        }
        System.out.println(leaders.size());
        for (int leader : leaders) {
            System.out.print(leader + " ");
        }
    }
}
