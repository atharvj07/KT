#include <bits/stdc++.h>
using namespace std;

long long gcd(long long a, long long b) { return b ? gcd(b, a % b) : a; }
int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (auto &x : a) cin >> x;
  long long ans = a[0];
  for (int i = 1; i < n; i++) ans = gcd(ans, a[i]);
  cout << ans << endl;
}
