#include <bits/stdc++.h>

using namespace std;

int main() {
    int w, h;
    cin >> w >> h;
    vector<vector<char>> a(h, vector<char>(w));
    for (int i = 0; i < h; i++) {
        for (int j = 0; j < w; j++) {
            cin >> a[i][j];
        }
    }
    vector<vector<char>> b(2 * h, vector<char>(2 * w));
    for (int i = 0; i < h; i++) {
        for (int j = 0; j < w; j++) {
            b[2 * i][2 * j] = b[2 * i + 1][2 * j] = b[2 * i][2 * j + 1] = b[2 * i + 1][2 * j + 1] = a[i][j];
        }
    }
    for (int i = 0; i < 2 * h; i++) {
        for (int j = 0; j < 2 * w; j++) {
            cout << b[i][j];
        }
        cout << endl;
    }
}
