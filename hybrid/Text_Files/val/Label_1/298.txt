def split_command_line(s):
  output = []
  quoted = False
  word = ''
  for c in s:
    if c == '"':
      quoted = not quoted
    elif c == ' ' and not quoted:
      if word: output.append('<%s>' % word)
      word = ''
    else: word += c
  if word: output.append('<%s>' % word)
  return output

s = input()
print('\n'.join(split_command_line(s)))
