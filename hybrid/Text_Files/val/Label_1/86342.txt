import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();

            if (n % 2020 == 0 || (n - 2020) % 2021 == 0)
                System.out.println("YES");
            else
                System.out.println("NO");
        }
    }
}
