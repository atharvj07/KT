#include<bits/stdc++.h>
using namespace std;
typedef long long LL;
const int N=5e5+5;
int n;
int p[N],sum[N];
int find(int x) {
	if(x!=p[x]) p[x]=find(p[x]);
	return p[x];
}
int main() {
	cin>>n;
	iota(p+1,p+1+n,1);
	for(int i=2; i<=n; i++) {
		for(int j=i+i; j<=n; j+=i) {
			int u=find(i),v=find(j);
			if(u!=v) {
				p[u]=v;
				sum[v]+=sum[u];
			}
		}
		sum[find(i)]++;
	}
	for(int i=2; i<=n; i++) {
		cout<<sum[find(i)]<<' ';
	}
	return 0;
}
