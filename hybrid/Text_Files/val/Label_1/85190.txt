#include <bits/stdc++.h>
using namespace std;

int n, p[300005];
int ans, a[300005], b[300005];

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> p[i];
  }
  int j = n;
  for (int i = 1; i <= n; i++) {
    if (p[i] != i) {
      if (p[j] == i) {
        a[++ans] = i;
        b[ans] = j;
        swap(p[i], p[j]);
      } else {
        a[++ans] = j;
        b[ans] = p[j];
        swap(p[j], p[p[j]]);
        a[++ans] = i;
        b[ans] = j;
        swap(p[i], p[j]);
      }
      j--;
    }
  }
  cout << ans << "\n";
  for (int i = 1; i <= ans; i++) {
    cout << a[i] << " " << b[i] << "\n";
  }
  return 0;
}
