inp = input().split()
n, m = map(int, inp)
bad = {tuple(map(int, input().split())) for _ in range(m)}
ans = 0
for i in range(1, n + 1):
    for j in range(1, n + 1):
        if (i, j) not in bad and ((i, j) != (1, 1) and (i, j) != (n, n)):
            if (i, j + 1) not in bad and (i, j - 1) not in bad and (j, i + 1) not in bad and (j, i - 1) not in bad:
                ans = max(ans, 1 + (n - 1 > 1))
print(ans)
