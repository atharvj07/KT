import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int n = scanner.nextInt();
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            if (n == 0 && a == 0 && b == 0) {
                break;
            }
            int count = 0;
            for (int i = 0; i <= n; i++) {
                if (!isRepresentable(i, a, b)) {
                    count++;
                }
            }
            System.out.println(count);
        }
    }

    private static boolean isRepresentable(int n, int a, int b) {
        for (int i = 0; i <= n / a; i++) {
            for (int j = 0; j <= n / b; j++) {
                if (a * i + b * j == n) {
                    return true;
                }
            }
        }
        return false;
    }
}
