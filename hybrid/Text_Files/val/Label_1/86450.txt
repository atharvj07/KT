import java.util.ArrayDeque;
import java.util.Queue;
import java.util.Scanner;

class Main {
    static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        Graph graph = new Graph(n);
        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt() - 1;
            int v = scanner.nextInt() - 1;
            graph.addEdge(u, v);
        }
        System.out.println(countWinningGraphs(graph));
    }

    static long countWinningGraphs(Graph graph) {
        int n = graph.n;
        long[][] dp = new long[1 << n][n];
        dp[1 << 1][1] = 1;
        for (int mask = 1 << 2; mask < 1 << n; mask++) {
            for (int i = 0; i < n; i++) {
                if ((mask & (1 << i)) == 0) {
                    continue;
                }
                for (int j = 0; j < n; j++) {
                    if ((mask & (1 << j)) != 0 || !graph.adj[i].contains(j)) {
                        continue;
                    }
                    dp[mask][i] = (dp[mask][i] + dp[mask ^ (1 << i)][j]) % MOD;
                }
            }
        }
        return dp[(1 << n) - 1][1];
    }

    static class Graph {
        int n;
        Queue<Integer>[] adj;

        Graph(int n) {
            this.n = n;
            adj = new Queue[n];
            for (int i = 0; i < n; i++) {
                adj[i] = new ArrayDeque<>();
            }
        }

        void addEdge(int u, int v) {
            adj[u].add(v);
        }
    }
}
