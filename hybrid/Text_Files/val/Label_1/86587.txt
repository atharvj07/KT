import java.util.Scanner;

class Main {

    static int maxPoints, maxMoves;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        char[][] board = new char[n][m];
        for (int i = 0; i < n; i++) {
            board[i] = sc.next().toCharArray();
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (board[i][j] != '.') {
                    int points = dfs(board, i, j);
                    if (points > maxPoints) {
                        maxPoints = points;
                        maxMoves = 1;
                    } else if (points == maxPoints) {
                        maxMoves++;
                    }
                }
            }
        }
        System.out.println(maxPoints + " " + maxMoves);
    }

    static int dfs(char[][] board, int i, int j) {
        if (i < 0 || i >= board.length || j < 0 || j >= board[0].length || board[i][j] == '.') {
            return 0;
        }
        int points = 1;
        char dir = board[i][j];
        board[i][j] = '.';
        switch (dir) {
            case 'L':
                points += dfs(board, i, j - 1);
                break;
            case 'R':
                points += dfs(board, i, j + 1);
                break;
            case 'U':
                points += dfs(board, i - 1, j);
                break;
            case 'D':
                points += dfs(board, i + 1, j);
                break;
        }
        return points;
    }
}
