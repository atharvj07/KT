import sys

def valid(a, b):
    for i in range(1, len(a) // 2 + 1):
        if a[i - 1] != b[len(a) - i]:
            return False
    return True

for i in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    print("Yes" if valid(a, b) else "No")
