import sys
input = sys.stdin.readline

def check(a, b):
    n = len(a)
    for i in range(1,n//2+1):
        if b[i-1] != a[n-i]:
            return False
    return True

for _ in range(int(input())):
    n = int(input())
    a = list(map(int,input().split()))
    b = list(map(int,input().split()))
    print("Yes" if check(a,b) else "No")
