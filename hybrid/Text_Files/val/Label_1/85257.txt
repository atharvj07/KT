#include <iostream>
using namespace std;
int main() {
    int n, sum = 1, last = 1, t;
    cin >> n;
    for (int i = 1; i < n; i++) {
        for (int j = 1; j <= i + 1; j++) {
            t = sum;
            sum += last;
            last = t;
        }
    }
    cout << sum << endl;
    return 0;
}
