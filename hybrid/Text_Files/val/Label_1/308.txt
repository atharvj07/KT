t = int(input().strip())
for _ in range(t):
    n, c = input().strip().split()
    n = int(n)
    s = input().strip()
    
    if s.count(c) == n:
        print(0)
        continue
    
    for x in range(2, n+1):
        if all(s[i] == c for i in range(x-1, n, x)):
            print(1)
            print(x)
            break
    else:
        print(2)
        print(n, n-1)
