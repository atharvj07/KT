import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long[] p = new long[n];
        long[] s = new long[n];
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        long[] inc = new long[m];
        long[] pref = new long[m];
        for (int i = 0; i < m; i++) {
            inc[i] = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            pref[i] = sc.nextInt();
        }
        int[] ans = new int[m];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (p[j] <= inc[i] && inc[i] <= s[j] && Math.abs(b[j] - pref[i]) <= (inc[i] - p[j])) {
                    ans[i]++;
                }
            }
        }
        for (int i = 0; i < m; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
