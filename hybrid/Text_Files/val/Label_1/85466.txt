#include <bits/stdc++.h>

using namespace std;

const int mod = 1e9 + 7;
const int maxn = 1e6 + 5;

char s[maxn];
char t[maxn];
int dp[maxn][2];
int n, m;
vector<int> a;

int power(int x, int y) {
  int ret = 1;
  while (y) {
    if (y & 1) ret = 1LL * ret * x % mod;
    x = 1LL * x * x % mod;
    y >>= 1;
  }
  return ret;
}

int main() {
  cin >> n >> m;
  cin >> t;
  for (int i = 0; i < m; i++) {
    int x;
    cin >> x;
    a.push_back(x);
  }
  dp[n][0] = dp[n][1] = 1;
  for (int i = n - 1; i >= 0; i--) {
    int f = 0;
    int h = 0;
    for (int j = i + 1; j <= n; j++) {
      if (s[j] == s[i]) f = 1;
      if (s[j] != s[i]) h = 1;
    }
    if (!f) {
      dp[i][0] = dp[i + 1][0];
      dp[i][1] = dp[i + 1][1];
    } else {
      dp[i][0] = 1LL * dp[i][0] * dp[i + 1][0] % mod;
      dp[i][0] = (dp[i][0] + 1LL * dp[i + 1][1] * h % mod) % mod;
      dp[i][1] = 1LL * dp[i][1] * dp[i + 1][1] % mod;
      dp[i][1] = (dp[i][1] + 1LL * dp[i + 1][0] * f % mod) % mod;
    }
  }
  for (int i = 0; i < a.size(); i++) {
    s[a[i]] = t[i];
  }
  int ans = dp[0][1];
  for (int i = 0; i < m; i++) {
    int x = a[i] - (i + 1);
    int y = n - a[i] - (m - (i + 1));
    ans = 1LL * ans * power(26, x % mod) % mod;
    ans = 1LL * ans * power(25, y % mod) % mod;
  }
  cout << ans << endl;
  return 0;
}
