n, k = map(int, input().split())
p = list(map(int, input().split()))
c = list(map(int, input().split()))

p.sort()
m = [len({j for j in range(i, n) if p[j] < p[i] and j > i}) for i in range(n)]
m.sort()
c.sort()
if k == 0:
    print(*c)
else:
    ans = []
    l = 0
    for i in m:
        if k == 0:
            break
        if i <= k:
            ans.append(c.pop())
            k -= i
            l += 1
        else:
            ans.append(c[0])
            k -= 1
            l += 1
    ans.extend(c[l:])
    print(*ans)
