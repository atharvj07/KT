#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int inf = 1e9 + 7;
char s[200005], t[200005];
int pos[200005];
int main() {
  int n, m, ans = 0;
  cin >> n >> m;
  if (n == m) { ans = n; }
  cin >> s >> t;
  for (int i = 0; i < m; i++) { pos[t[i]] = i; }
  vector<int> v;
  for (int i = 0; i < n; i++) {
    if (pos[s[i]] != 0) { v.push_back(i); }
  }
  for (int i = 0; i < int(v.size()) - 1; i++) { ans = max(ans, v[i + 1] - v[i]); }
  cout << ans << endl;
  return 0;
}
