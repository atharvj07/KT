import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    if n % 4 > 0:
        print("NO")
    else:
        print("YES")
        arr = [2 * i for i in range(1, n // 4 + 1)] + [2 * i + 1 for i in range(1, n // 4 + 1)] + [2 * i for i in range(n // 4 + 1, n // 2 + 1)] + [2 * i + 1 for i in range(n // 4 + 1, n // 2 + 1)]
        print(*arr)
