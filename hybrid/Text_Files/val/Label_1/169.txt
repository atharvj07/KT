n, m, x, y, z, p = map(int, input().split())
for _ in range(p):
    i, j = map(int, input().split())
    x = (x - z) % 4
    y = (y + z) % 4
    if x == 0:
        i, j = i, m - j + 1
    elif x == 1:
        i, j = n - j + 1, i
    elif x == 2:
        i, j = n - i + 1, j
    elif x == 3:
        i, j = j, i
    if y == 1:
        i, j = j, n - i + 1
    elif y == 2:
        i, j = n - i + 1, j
    elif y == 3:
        i, j = j, i
    print(i, j)
