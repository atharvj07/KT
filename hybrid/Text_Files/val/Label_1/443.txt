import queue
n,k=map(int,input().split())
a=[int(x) for x in input().split()]
g=[[] for i in range(n+1)]
for i in range(n-1):
    x,y=map(int,input().split())
    g[x].append(y)
    g[y].append(x)
q=queue.Queue()
ans=1000000007
vis=[0]*(n+1)
q.put(1)
while k>0 and not q.empty():
    v=q.get()
    vis[v]=1
    if k==0:break
    k-=1
    ans=min(ans,a[v-1])
    for u in g[v]:
        if vis[u]==0:q.put(u)
print(ans)
