n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
m = int(input())

# Merge and sort the sequences with identification of their origin (a or b)
merged_list = [(a[i], 'a') for i in range(n)] + [(b[i], 'b') for i in range(n)]
merged_list.sort()

# Count the duplicates and their occurrences
duplicates = 1
result = 1
for i in range(1, 2*n):
    if merged_list[i][0] == merged_list[i-1][0]:
        duplicates += 1
    else:
        # When no more duplicates, calculate the possibilities for the sequence of duplicates
        result = (result * duplicates) % m
        duplicates = 1
# Include the possibilities for the last set of duplicates
result = (result * duplicates) % m

print(result)
