import sys

# Read input
t = int(sys.stdin.readline())
for _ in range(t):
    x, y = map(int, sys.stdin.readline().split())

    # Check if x is even
    if x % 2 == 0:
        # Check if y is even and greater than x
        if y % 2 == 0 and y > x:
            # Petya can use the first spell to get y
            print("YES")
        # Check if y is odd and greater than x
        elif y % 2 == 1 and y > x:
            # Petya can use the first spell to get to an even number, then use the second spell to get to y
            print("YES")
        # Otherwise, Petya cannot get y from x
        else:
            print("NO")
    # If x is odd, Petya cannot get y from x
    else:
        print("NO")
