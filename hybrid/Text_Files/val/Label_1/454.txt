from collections import Counter, deque
from sys import stdin

n, m = map(int, stdin.readline().strip().split())
e = [[] for i in range(n + 1)]
for i in range(m):
    a, b = map(int, stdin.readline().strip().split())
    e[a].append(b)
    e[b].append(a)

vis = [False] * (n + 1)
color = [0] * (n + 1)
group = []

def dfs(u, c):
    vis[u] = True
    color[u] = c
    group[-1].append(u)
    for v in e[u]:
        if not vis[v]:
            dfs(v, c)

cc = 0
for i in range(1, n + 1):
    if not vis[i]:
        cc += 1
        group.append([])
        dfs(i, cc)

pos = [[] for i in range(cc + 1)]
for i in range(1, n + 1):
    pos[color[i]].append((i, 1))

cnt = Counter()
for i in range(1, cc + 1):
    pos[i].sort()
    pos[i] = [(x, y) for x, y in pos[i] if not cnt[x]]
    for x, y in pos[i]:
        cnt[x] += y

ans = []
for i in range(1, cc + 1):
    if not group[i - 1]:
        print(0)
    else:
        print(len(group[i - 1]))
        for x in group[i - 1]:
            x, y = pos[i].pop(0)
            print(x, y)
            ans.append((x, y))
