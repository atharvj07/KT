#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> m(n);
    for (int i = 0; i < n; i++) {
        cin >> m[i];
    }
    vector<int> a(n);
    a[0] = m[0];
    for (int i = 1; i < n; i++) {
        a[i] = min(m[i], max(a[i - 1], (i + 1 < n ? a[i + 1] : 0)));
    }
    for (int i = n - 2; i >= 0; i--) {
        a[i] = min(m[i], max(a[i + 1], (i - 1 >= 0 ? a[i - 1] : 0)));
    }
    for (int i = 0; i < n; i++) {
        cout << a[i] << " ";
    }
    cout << endl;
    return 0;
}
