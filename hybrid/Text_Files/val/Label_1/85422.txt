#include <iostream>
#include <vector>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n;
        std::cin >> n;
        std::vector<int> p(n);
        std::vector<bool> ans(n);
        for (int i = 0; i < n; i++) {
            std::cin >> p[i];
            p[i]--;
        }
        for (int i = 0; i < n; i++) {
            int mx = 0;
            for (int j = i; j < n; j++) {
                mx = std::max(mx, p[j]);
                if (mx == j - i) ans[mx] = true;
            }
        }
        for (int i = 0; i < n; i++) {
            std::cout << (ans[i] ? '1' : '0');
        }
        std::cout << std::endl;
    }
    return 0;
}
