#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int maxn=500500;
ll n,m,x,y,a,b,tmpx,tmpy,dis,cnt,nowans;
ll f[maxn],sum[maxn];
struct node{
	ll x;
	ll y;
}que[maxn];
bool cmp(const node&a,const node&b){
	return a.x<b.x;
}
ll gcd(ll a,ll b){
	if(b==0) return a;
	else return gcd(b,a%b);
}
int main(){
	cin>>n>>m>>x>>y>>a>>b;
	ll l,r,mid;
	for(ll i=0;i<=y;i++){
		if((x+i*(m-y))%a==0){
			l=x-i*(m-y);
			r=x+i*(m-y);
			if(l>=0&&r<=n){
				if(l<0) l=0;
				if(r>n) r=n;
				if((r-l)*b!=(m-y)*a) continue;
				ll tt=(l/a)*m+i;
				if(tt<nowans) continue;
				ll xx=(l/a)*a+a-1,yy=tt/a*b+b-1;
				que[++cnt].x=xx;
				que[cnt].y=yy;
			}
		}
	}
	sort(que+1,que+1+cnt,cmp);
	tmpx=que[1].x;
	tmpy=que[1].y;
	for(ll i=1;i<=cnt;i++){
		if(que[i].x==tmpx&&que[i].y==tmpy)
		sum[++dis]++;
		else tmpx=que[i].x,tmpy=que[i].y,dis++,sum[dis]++;
	}
	for(ll i=1;i<=dis;i++){
		if(sum[i]>sum[nowans])
		nowans=i;
		else if(sum[i]==sum[nowans]){
			if((que[dis-nowans+1].x-x)*(que[dis-nowans+1].x-x)+(que[dis-nowans+1].y-y)*(que[dis-nowans+1].y-y)<(que[dis-i+1].x-x)*(que[dis-i+1].x-x)+(que[dis-i+1].y-y)*(que[dis-i+1].y-y)){
				nowans=i;
			}
			else if((que[dis-nowans+1].x-x)*(que[dis-nowans+1].x-x)+(que[dis-nowans+1].y-y)*(que[dis-nowans+1].y-y)==(que[dis-i+1].x-x)*(que[dis-i+1].x-x)+(que[dis-i+1].y-y)*(que[dis-i+1].y-y)){
				if(que[dis-nowans+1].x-x>que[dis-i+1].x-x){
					nowans=i;
				}
				else if(que[dis-nowans+1].x-x==que[dis-i+1].x-x){
					if(que[dis-nowans+1].y-y>que[dis-i+1].y-y){
						nowans=i;
					}
				}
			}
		}
	}
	cout<<que[dis-nowans+1].x<<" "<<que[dis-nowans+1].y<<" "<<que[dis-nowans+1].x+(ll)(m-y)/b*a<<" "<<que[dis-nowans+1].y+(ll)(m-y)/b*b<<endl;
	return 0;
}
