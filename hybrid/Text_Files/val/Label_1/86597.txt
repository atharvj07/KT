import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    int n = scan.nextInt();
    int m = scan.nextInt();
    int a[] = new int[m];
    int b[] = new int[m];
    int w[] = new int[m];
    for (int i = 0; i < m; i++) {
      a[i] = scan.nextInt();
      b[i] = scan.nextInt();
      w[i] = scan.nextInt();
    }
    Arrays.sort(w);
    int ans = 0;
    for (int i = m - 1; i >= 0; i--) {
      if (a[i] > b[i]) {
        ans += w[i];
      }
    }
    System.out.println(ans);
  }
}
