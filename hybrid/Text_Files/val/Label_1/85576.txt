#include<bits/stdc++.h>
using namespace std;
int32_t main() {
	int t;
	cin >> t;
	for (int i = 1; i <= t; i++) {
		int n;
		cin >> n;
		vector<int> A(n);
		for (int j = 0; j < n; j++) {
			cin >> A[j];
		}
		sort(A.begin(), A.end());
		int ans = 1;
		for (int j = 0; j < n; j++) {
			if (A[j] <= ans) ans++;
		}
		cout << ans << endl;
	}
}
