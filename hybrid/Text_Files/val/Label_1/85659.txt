#include <bits/stdc++.h>
#define pb push_back
using namespace std;
#define ll long long
const ll N=2e5+5;
ll n,k,a[N],ans;
vector<ll>adj[N],vis(N);
void dfs(ll node, ll par, ll val){
    ans=min(ans,a[node]);
    for(auto u:adj[node])
        if(u!=par){
            if(!vis[u]){
                vis[u]=1;
                dfs(u,node,val);
            }
        }
}
int main() {
    cin>>n>>k;
    for(ll i=1;i<=n;i++) cin>>a[i];
    for(ll i=1;i<n;i++){
        ll u,v; cin>>u>>v;
        adj[u].pb(v); adj[v].pb(u);
    }
    ans=1e9;
    for(ll i=1;i<=n;i++){
        vis.assign(N,0);
        vis[i]=1;
        dfs(i,0,i);
    }
    cout<<ans<<endl;
    return 0;
}
