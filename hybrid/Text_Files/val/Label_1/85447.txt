#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll N=1e5+5;
vector<ll> adj[N];
ll seg[N],sz[N];
void dfs(ll u,ll p){
    sz[u]=1;
    for(auto v:adj[u])
        if(v!=p){
            dfs(v,u);
            if(seg[v]>seg[u])
                swap(seg[u],seg[v]),swap(sz[u],sz[v]);
            if(seg[v]==seg[u])
                sz[u]+=sz[v];
        }
    seg[u]+=sz[u];
}
int main(){
    ll n;cin>>n;
    for(ll i=1;i<n;i++){
        ll u,v;
        cin>>u>>v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1,0);
    cout<<seg[1];
}
