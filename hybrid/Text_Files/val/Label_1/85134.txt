#include <bits/stdc++.h>
using namespace std;

const int N = 2e3 + 10, K = 2e6 + 10;
int n, m, k, a[N], b[N], sum[N][N];

int main() {
  cin >> n >> m >> k;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 1; i <= m; i++) { cin >> b[i]; }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) { sum[i][j] = sum[i - 1][j] + sum[i][j - 1] - sum[i - 1][j - 1] + a[i] * b[j]; }
  }
  int ans = 0;
  for (int i1 = 1; i1 <= n; i1++) {
    for (int i2 = i1; i2 <= n; i2++) {
      int pre = 0;
      for (int j1 = 1; j1 <= m; j1++) {
        int cur = sum[i2][j1] - sum[i1 - 1][j1] - sum[i2][j1 - 1] + sum[i1 - 1][j1 - 1];
        int low = 0, high = k + 1;
        while (low < high) {
          int mid = (low + high) >> 1;
          if (mid * (j1 - pre) <= cur) {
            ans = max(ans, (i2 - i1 + 1) * (j1 - pre));
            pre = j1;
            low = mid + 1;
          } else {
            high = mid;
          }
        }
      }
    }
  }
  cout << ans << '\n';
}
