n, m = map(int, input().split())
boss = [0] * (n + 1)
packet = [0] * (n + 1)
for _ in range(m):
    t, *args = map(int, input().split())
    if t == 1:
        x, y = args
        boss[x] = y
    elif t == 2:
        x = args[0]
        packet[x] += 1
    else:
        x, i = args
        while x != 0 and packet[x] < i:
            x = boss[x]
        print("YES" if x != 0 else "NO")
