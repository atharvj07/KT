#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int n, a;
    cin >> n;
    vector<int> b(n + 1);
    for (int i = 0; i < n; i++) {
        cin >> a;
        b[a + 1]++;
    }
    int flg = 0;
    for (int i = 1; i <= n; i++) {
        if (b[i] < n - b[i] - 1) {
            cout << "Impossible" << endl;
            flg = 1;
            break;
        }
    }
    if (!flg) {
        cout << "Possible" << endl;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < n - b[i] - 1; j++) {
                cout << i << ' ';
            }
        }
        cout << endl;
    }
    return 0;
}
