from sys import stdin

input = stdin.readline

n, m = map(int, input().split())
a = list(map(int, input().split()))
for i in range(1, m + 1):
    t, *v = map(int, input().split())
    if t == 1:
        a[v[0] - 1] = v[1]
    elif t == 2:
        l, r = v
        print(sum(a[l - 1:r]))
    else:
        l, r, d = v
        a[l - 1:r] = [a[i] + d for i in range(l - 1, r)]
