#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<bool> vis(26, false);
    bool ok = true;
    for (int i = 0; i < n; i++) {
      int idx = s[i] - 'A';
      if (vis[idx]) {
        ok = false;
        break;
      }
      vis[idx] = true;
    }
    cout << (ok ? "YES" : "NO") << endl;
  }
}
