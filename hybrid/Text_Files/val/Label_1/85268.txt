#include <bits/stdc++.h>
using namespace std;
const int maxn=110,maxm=100010;
char a[maxn];
int b[maxn],c[maxn],n;
int main() {
  int t; cin>>t;
  while (t--) {
    cin>>n;
    cin>>a+1;
    b[0]=b[n+1]=c[0]=c[n+1]=0;
    for (int i=1;i<=n;i++) {
      b[i]=b[i-1];
      c[i]=c[i-1];
      if (a[i]=='>') ++b[i];
      if (a[i]=='<') ++c[i];
    }
    for (int i=n;i>=1;i--) {
      b[i]=max(b[i],b[i+1]);
      c[i]=max(c[i],c[i+1]);
    }
    int ans=n;
    for (int i=1;i<=n;i++) {
      ans=min(ans,max(b[i-1],c[i]));
    }
    cout<<ans<<endl;
  }
  return 0;
}
