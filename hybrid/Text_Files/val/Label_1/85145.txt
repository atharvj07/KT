#include <bits/stdc++.h>
using namespace std;
int n, a[106];
void dfs(int u, int cnt) {
    if (cnt > a[u]) return;
    if (u == n) for (int i = 1; i < n; i++)if (a[i] == 0) return;
    for (int i = u + 1; i <= n; i++) {
        a[i]--;
        cout << u << ' ' << i << '\n';
        dfs(i, cnt + 1);
        a[i]++;
    }
}
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    if (a[n] == 0) { cout << "-1\n"; return 0; }
    cout << n - 1 << '\n';
    for (int i = 2; i <= n; i++) { a[i]--; cout << 1 << ' ' << i << '\n'; }
    dfs(2, 1);
}
