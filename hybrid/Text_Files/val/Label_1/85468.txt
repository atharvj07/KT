#include <bits/stdc++.h>

using namespace std;
const int mod = 1e9 + 7;

vector<int> p, occ[28];
int n, m, dp[1000005];

int main() {
  cin >> n >> m;
  string x;
  cin >> x;
  for (int i = 0; i < (int)x.size(); i++) p.push_back(x[i] - 'a');
  for (int i = 0, k; i < n; i++) { cin >> k; occ[p[i]].push_back(k); }
  dp[0] = 1;
  for (int i = 1; i < n; i++) {
    int &ret = dp[i] = 0;
    for (auto &x : occ[p[i]]) {
      if (x + (int)p.size() - 1 <= i) ret = (ret + dp[x + (int)p.size() - 1 - 1]) % mod;
    }
  }
  cout << dp[n - 1];
  return 0;
}
