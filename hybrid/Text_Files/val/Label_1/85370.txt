# include <iostream>
# include <vector>
# include <algorithm>
using namespace std;

typedef long long LL;

int main() {
    int n;
    cin >> n;

    vector<LL> heights(n);
    vector<LL> lh(n), rh(n);

    for (int i = 0; i < n; i++) {
        cin >> heights[i];
    }

    LL ans = 0;

    lh[0] = heights[0];
    for (int i = 1; i < n; i++) {
        lh[i] = min(heights[i], lh[i - 1]);
    }

    rh[n - 1] = heights[n - 1];
    for (int i = n - 2; i >= 0; i--) {
        rh[i] = min(heights[i], rh[i + 1]);
    }

    for (int i = 0; i < n; i++) {
        ans += min({lh[i], rh[i], heights[i]});
    }

    cout << ans << endl;
    return 0;
}
