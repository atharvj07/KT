n = int(input().strip())
cards = list(map(int, input().split()))

sereja_points, dima_points = 0, 0
turn = 0 # 0 for Sereja, 1 for Dima

while cards:
    if cards[0] > cards[-1]:
        choice = cards.pop(0)
    else:
        choice = cards.pop()

    if turn == 0:
        sereja_points += choice
        turn = 1
    else:
        dima_points += choice
        turn = 0

print(sereja_points, dima_points)
