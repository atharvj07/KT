#include <bits/stdc++.h>
long long gcd_naive(long long a, long long b) {
    if (b == 0)
        return a;
    return gcd_naive(b, a % b);
}
long long solve(long long a, long long b) {
    long long ans = b;
    while (b != 1) {
        long long temp = gcd_naive(ans, b);
        ans = ans * b / temp;
        if (ans > 1e18)
            ans = 1e18;
        b /= temp;
    }
    return ans;
}

int main() {
    long long a, b;
    std::cin >> a >> b;
    std::cout << solve(a, b) << '\n';
    return 0;
}
