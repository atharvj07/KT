#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> a(n, vector<int>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      char c;
      cin >> c;
      a[i][j] = c - '0';
    }
  }
  int oddRow[n][m], oddCol[n][m];
  oddRow[n - 1][m - 1] = oddCol[n - 1][m - 1] = a[n - 1][m - 1] % 2;
  for (int i = n - 2; i >= 0; i--) {
    oddRow[i][m - 1] = oddRow[i + 1][m - 1];
    oddRow[i][m - 1] += a[i][m - 1] % 2;
  }
  for (int j = m - 2; j >= 0; j--) {
    oddCol[n - 1][j] = oddCol[n - 1][j + 1];
    oddCol[n - 1][j] += a[n - 1][j] % 2;
  }
  for (int i = n - 2; i >= 0; i--) {
    for (int j = m - 2; j >= 0; j--) {
      oddRow[i][j] = oddRow[i + 1][j];
      oddRow[i][j] += a[i][j] % 2;
      oddCol[i][j] = oddCol[i][j + 1];
      oddCol[i][j] += a[i][j] % 2;
    }
  }
  int ans = INT_MAX;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      int odd = (i % 2 == j % 2) ? 1 : 0;
      int cur = a[i][j];
      int need = 1 - (oddRow[i][j] + oddCol[i][j] + cur - odd) % 2;
      ans = min(ans, cur + need - 2 * need * cur);
    }
  }
  if (ans == INT_MAX) ans = -1;
  cout << ans << endl;
}
