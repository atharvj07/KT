#include<iostream>
#include<bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<string> v1(n), v2(n);
    for (auto &i : v1) cin >> i;
    for (auto &i : v2) cin >> i;
    bool ok = (v1 == v2);
    reverse(v2.begin(), v2.end());
    ok |= (v1 == v2);
    for (int i = 0; i < n; i++) {
        string t = "";
        for (int j = n - 1; j >= 0; j--) t += v2[j][i];
        v2[i] = t;
    }
    ok |= (v1 == v2);
    reverse(v2.begin(), v2.end());
    ok |= (v1 == v2);
    for (int i = 0; i < n; i++) {
        string t = "";
        for (int j = n - 1; j >= 0; j--) t += v2[i][j];
        v2[i] = t;
    }
    ok |= (v1 == v2);
    reverse(v2.begin(), v2.end());
    ok |= (v1 == v2);
    cout << (ok ? "Yes" : "No") << endl;
    return 0;
}
