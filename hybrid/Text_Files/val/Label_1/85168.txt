#include <iostream>
#include <string>

using namespace std;

bool can_palindrome(string s) {
  int n = s.size();
  for (int i = 0; i < n / 2; i++) {
    if (s[i] == s[n - i - 1]) continue;
    if (s[i] + 1 == s[n - i - 1] || s[i] - 1 == s[n - i - 1]) continue;
    return false;
  }
  return true;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    string s;
    cin >> n >> s;
    cout << (can_palindrome(s) ? "YES\n" : "NO\n");
  }
  return 0;
}
