from functools import reduce
from operator import xor

n,k=map(int,input().split())
a=[0]*n
for i in range(n):a[i]=int(input())
s=[0]*n
for i in range(n):s[i]=a[i]
for i in range(1,n):s[i]^=s[i-1]

ans=0
for i in range(n):
    for j in range(i,n):
        if i==0:t=s[j]
        else:t=s[j]^s[i-1]
        if t!=0:ans+=1
print(ans)
