#include <iostream>
#include <algorithm>
#include <vector>

int main() {
    int n;
    std::cin >> n;
    std::string s;
    std::cin >> s;
    std::vector<int> v(n);
    for (int i = 0; i < n; ++i) {
        v[i] = s[i] - '0';
    }
    std::sort(v.begin(), v.end());
    std::string ans;
    for (int i = 0; i < n; ++i) {
        if (v[i] == 0 && ans.empty()) {
            continue;
        }
        ans += std::to_string(v[i]);
    }
    if (ans.empty()) ans = "0";
    std::cout << ans << std::endl;
    return 0;
}
