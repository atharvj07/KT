import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextInt();
        }
        Arrays.sort(arr);
        int maxlen = 0;
        int num = 0;
        for (int i = 0; i < n; i++) {
            if (i == 0 || arr[i] != arr[i - 1]) {
                num++;
            }
            if (arr[i] > maxlen) {
                maxlen = arr[i];
            }
        }
        System.out.println(maxlen + " " + num);
    }
}
