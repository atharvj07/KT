from sys import *
setrecursionlimit(10**8)
for _ in range(int(input())):
    n = int(input())
    l = list(map(int,input().split()))
    r = [0]*n
    for i in range(n):
        r[l[i]-1] = i
    x = [0]*n
    for i in range(n):
        x[i]=r[i]-i
    cycle = 0
    flag = 0
    for i in range(n):
        if not x[i]:
            continue
        if flag:
            cycle += 1
        flag = 1
        j = i
        while x[j]>0:
            j += x[j]
            j %= n
        while x[j]<0:
            j += x[j]
            j %= n
        if j==i:
            flag = 0
    print(cycle)
