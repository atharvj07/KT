#include <vector>
#include <iostream>

int n, c[2005];
int f[2005][2005];

int main() {
    std::cin >> n;
    for (int i = 1; i <= n; ++i) std::cin >> c[i];
    f[0][0] = f[0][1] = 0;
    bool flag = 1;
    for (int i = 1; i <= n; ++i) {
        if (flag) {
            for (int j = 0; j < c[i]; ++j) {
                for (int k = 1; k <= i; ++k) {
                    if (f[i - k][j] + k <= c[i]) {
                        f[i][j + k] = f[i - k][j] + k;
                        char ch = 'a' + j;
                        for (int l = f[i - k][j]; l < f[i - k][j] + k; ++l)
                            std::cout << ch;
                        flag = 0;
                    }
                }
            }
        }
    }
    return 0;
}
