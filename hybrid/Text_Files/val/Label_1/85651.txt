#include<bits/stdc++.h>
using namespace std;
int s, m, a[102];
bool ok() {
  int sum = 0;
  for (int i = 1; i <= m; i++) {
    if (a[i] == 0 && i != m) return 0;
    sum += a[i];
  }
  return sum == s;
}
int main() {
  cin >> m >> s;
  if (s < m || s > 9 * m) {
    cout << "-1 -1" << endl;
    return 0;
  }
  int dd = s / m;
  int res = s - dd * m;
  for (int i = 1; i <= m; i++) {
    if (i == m) {
      a[i] = res;
    } else {
      a[i] = dd;
    }
  }
  for (int i = m; i; i--) {
    cout << a[i];
  }
  cout << ' ';
  memset(a, 0, sizeof a);
  for (int i = 1; i <= m; i++) {
    res++;
    if (res > 9) {
      res = 1;
      dd++;
    }
    a[i] = res;
  }
  for (int i = m; i; i--) {
    cout << a[i];
  }
  cout << endl;
}
