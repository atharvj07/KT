#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;
using ll = long long;

int main() {
    ll q, answer = 0;
    cin >> q;
    vector<ll> booksLeft, booksRight;
    while (q--) {
        char t;
        ll id;
        cin >> t >> id;
        switch (t) {
            case 'L':
                answer++;
                booksLeft.push_back(id);
                break;
            case 'R':
                answer++;
                booksRight.push_back(id);
                break;
            case '?':
                ll posLeft = lower_bound(booksLeft.begin(), booksLeft.end(), id) - booksLeft.begin();
                ll posRight = upper_bound(booksRight.begin(), booksRight.end(), id) - booksRight.begin();
                if (posLeft != booksLeft.size()) posLeft = booksLeft.size() - posLeft;
                else posLeft = 0;
                if (posRight != booksRight.size()) posRight = booksRight.size() - posRight;
                else posRight = 0;
                cout << min(posLeft, posRight) << endl;
                break;
        }
    }
    return 0;
}
