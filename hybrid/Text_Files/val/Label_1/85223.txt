#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

typedef long long ll;

const int MAXN = 100005;

int n, m;
int p[MAXN], c[MAXN];
vector<int> g[MAXN];

int find_set(int v) {
    if (p[v] == v) return v;
    return p[v] = find_set(p[v]);
}

void union_sets(int a, int b, int cc) {
    a = find_set(a);
    b = find_set(b);
    if (a != b) {
        p[a] = b;
        if (c[b] != cc) c[b] = 2;
        if (c[a] != cc) c[b] = 2;
    }
}

int main() {
    ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    cin >> n >> m;
    for (int i = 0; i <= n; i++) p[i] = i, c[i] = 0;
    for (int i = 0; i < m; i++) {
        int a, b;
        char clr;
        cin >> a >> b >> clr;
        if (clr == 'R') clr = 1;
        else clr = 0;
        g[a].push_back(b);
        g[b].push_back(a);
        union_sets(a, b, clr);
    }
    int dif = 0;
    for (int i = 1; i <= n; i++) {
        if (c[find_set(i)] == 2) dif = 1;
        if (c[find_set(i)] == 0) {
            cout << -1 << "\n";
            return 0;
        }
    }
    if (!dif) {
        cout << 0 << "\n";
        return 0;
    }
    vector<int> ans;
    for (int i = 1; i <= n; i++) {
        if (find_set(i) == i && c[i] == 2) {
            ans.push_back(i);
            for (int j : g[i]) {
                union_sets(i, j, 1 - c[find_set(i)]);
            }
        }
    }
    cout << ans.size() << "\n";
    for (int x : ans) cout << x << " ";
    cout << "\n";
    return 0;
}
