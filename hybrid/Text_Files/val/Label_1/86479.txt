import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int ans = 0;
        for (int i = 0, j = n - 1; i < j; i++, j--) {
            if (a[i] > 0 && a[j] == 0) {
                ++ans;
            } else if (a[i] == 0 && a[j] > 0) {
                ++ans;
            }
        }
        System.out.println(ans);
    }
}
