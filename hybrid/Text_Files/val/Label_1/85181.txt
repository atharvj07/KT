#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<char> status(n);
  for (int i = 0; i < n; ++i) {
    cin >> status[i];
  }
  int num_allin = 0;
  int num_in = 0;
  int num_can_show = 0;
  for (int i = 0; i < n; ++i) {
    if (status[i] == 'A') {
      ++num_allin;
    } else if (status[i] == 'I') {
      ++num_in;
    } else {
      ++num_can_show;
    }
  }
  if (num_allin == n - 1) {
    ++num_can_show;
  } else if (num_in == 1) {
    ++num_can_show;
  }
  cout << num_can_show << endl;
  return 0;
}
