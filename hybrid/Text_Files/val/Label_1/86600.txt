import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), k = scanner.nextInt(), m = scanner.nextInt(), t = scanner.nextInt();
        int[] len = new int[t + 1], pos = new int[t + 1];
        len[0] = n; pos[0] = k;
        for (int i = 1; i <= t; i++) {
            int op = scanner.nextInt(), idx = scanner.nextInt();
            if (op == 0) {
                if (pos[i - 1] > idx) {
                    len[i] = len[i - 1] - 1;
                    pos[i] = pos[i - 1] - 1;
                } else {
                    len[i] = len[i - 1];
                    pos[i] = pos[i - 1];
                }
            } else {
                if (idx <= pos[i - 1]) {
                    len[i] = len[i - 1] + 1;
                    pos[i] = pos[i - 1] + 1;
                } else {
                    len[i] = len[i - 1];
                    pos[i] = pos[i - 1];
                }
            }
            System.out.println(len[i] + " " + pos[i]);
        }
    }
}
