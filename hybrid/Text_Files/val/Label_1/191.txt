import sys
for t in range(int(input())):
    n, W = map(int,input().split())
    w = list(map(int,input().split()))
    w.sort()
    c = 0; ans = []; x = (W+1)//2
    for i in w:
        if i+c<=W and c+i>=x:
            c+=i; ans.append(i)
    if c<x: print(-1)
    else: print(len(ans)); print(*ans)
