#include<bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
int n, m, q, k, b[maxn], r[maxn], c[maxn], f[maxn], g[maxn], ans;
int main() {
    cin >> n >> m >> k >> q;
    for(int i = 1;i <= k;i++) {
        cin >> r[i] >> c[i];
        f[c[i]] = 1;
    }
    for(int i = 1;i <= q;i++) {
        cin >> b[i];
        g[b[i]] = 1;
    }
    for(int i = 1;i <= m;i++) {
        if(!f[i]) {
            if(g[i]) return puts("-1"), 0;
            continue;
        }
        if(g[i]) continue;
        for(int j = 1;j <= k;j++) {
            if(c[j] == i) {
                ans += r[j] - 1;
                c[j] = i + 1;
                r[j] = 1;
                break;
            }
        }
    }
    for(int i = 1;i <= k;i++) {
        if(!f[c[i]]) {
            return puts("-1"), 0;
        }
        ans += abs(c[i] - c[i - 1]);
    }
    cout << ans << endl;
    return 0;
}
