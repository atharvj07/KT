#include<bits/stdc++.h>
using namespace std;
int main(){
   int t;
   cin>>t;
   while(t--){
        int n,a,b;
        cin>>n>>a>>b;
        string s="";
        int j=0;
        char c='a';
        while(j<n){
            for(int i=0;i<a&&j<n;++i){
                s+=c;
                c++;
                if(c>'z') c='a';
            }
            j+=a;
        }
        cout<<s<<endl;
   }
}
