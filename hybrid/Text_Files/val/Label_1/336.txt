import sys

input = sys.stdin.readline

n, t, c = map(int, input().strip().split())
A = list(map(int, input().strip().split()))

dp = [0] * n

if A[0] <= t:
    dp[0] = 1

for i in range(1, n):
    if A[i] <= t:
        dp[i] = dp[i - 1] + 1
    if i - c >= 0:
        dp[i] -= dp[i - c]

print(dp[n - 1])
