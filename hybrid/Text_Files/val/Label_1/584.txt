n, m = map(int, input().split())
e = [[] for i in range(n + 1)]
f = [0] * (n + 1)
l = [0] * (n + 1)
for i in range(m):
    t, *a = map(int, input().split())
    if t == 1:
        e[a[1]].append(a[0])
    elif t == 2:
        l[a[0]] += 1
        f[a[0]] = l[a[0]]
    else:
        print('YES' if l[a[0]] >= f[a[1]] else 'NO')
