#include <iostream>
#include <vector>
using namespace std;

const int MAXN = 2e5;
int n, xs, ys, xe, ye, ans = -1, T;
vector<pair<int,int>> windTab = {{1, 0}, {0, -1}, {-1, 0}, {0, 1}};

int main() {
    cin >> n >> xs >> ys >> xe >> ye >> T;
    vector<pair<int,int>> windSeq(T);
    for (int i = 0; i < T; i++) {
        char dir;
        cin >> dir;
        if (dir == 'E') windSeq[i] = {1, 0};
        else if (dir == 'S') windSeq[i] = {0, -1};
        else if (dir == 'W') windSeq[i] = {-1, 0};
        else if (dir == 'N') windSeq[i] = {0, 1};
    }
    for (int step = 0; step < T; step++) {
        int x = xs, y = ys;
        for (int i = step; i < T; i++) {
            x += windSeq[i].first;
            y += windSeq[i].second;
            if (x == xe && y == ye) {
                ans = i - step + 1;
                break;
            }
        }
        if (ans != -1) break;
    }
    cout << ans <<endl;
    return 0;
}
