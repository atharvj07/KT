#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> digits(n);
    for (int i = 0; i < n; i++) { cin >> digits[i]; }
    while (digits[0] > 0) {
        digits.push_back(digits[0]);
        digits.erase(digits.begin());
        for (int i = 0; i < n; i++) { digits[i] = (digits[i] + 1) % 10; }
    }
    for (int digit : digits) { cout << digit; }
}
