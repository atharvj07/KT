#include<bits/stdc++.h>
#include<cstring>
using namespace std;
const int maxn=5e5+5,maxm=20,maxl=50005;
int n,m,fail[maxl],tr[maxl][26],cnt[maxl],ok[23][maxl],ch[maxl][26]=
    {0};
int a[maxm][3],go[maxl][3],vis[maxl];
char s[maxl];
int newnode(){
    memset(tr[cnt[0]],0,sizeof(tr[cnt[0]]));
    memset(ch[cnt[0]],-1,sizeof(ch[cnt[0]]));
    return cnt[0]++;
}
void insert(char *p){
    int len=strlen(p),now=0;
    for(int i=0;i<len;i++){
        int x=p[i]-'a';
        if(tr[now][x]==0) tr[now][x]=newnode();
        now=tr[now][x];
    }
}
void getfail(){
    queue<int> q;
    fail[0]=0;
    for(int i=0;i<26;i++){
        int x=tr[0][i];
        if(x) {fail[x]=0;q.push(x);}
    }
    while(!q.empty()){
        int now=q.front();q.pop();
        for(int i=0;i<26;i++){
            int x=tr[now][i];
            if(x){
                fail[x]=tr[fail[now]][i];
                q.push(x);
            }
            else tr[now][i]=tr[fail[now]][i];
        }
    }
}
void init(char *p,int x,int y,int z){
    insert(p);
    a[x][0]=z;a[x][1]=y;a[x][2]=strlen(p);
    for(int i=0;i<maxl;i++) ok[x][i]=0;
}
bool query(int id,int x,int y,int len){
    int now=0,cnt=0;
    while(x<=y){
        int t=s[x]-'a',tmp=now;
        if(ch[now][t]!=-1){
            now=ch[now][t];
            if(ok[id][now]) cnt++;
            else{
                cnt+=x>=a[id][1]?0:y-a[id][1]+1;
                go[now][0]=y-a[id][1]+1,go[now][1]=x,go[now][2]=id;
                ok[id][now]=1;
            }
        }
        else{
            now=tmp;
            while(now&&ch[now][t]==-1) now=fail[now];
            if(now){
                if(ok[id][now]) cnt++;
                else{
                    cnt+=x>=a[id][1]?0:y-a[id][1]+1;
                    go[now][0]=y-a[id][1]+1,go[now][1]=x,go[now][2]=id;
                    ok[id][now]=1;
                }
                now=ch[now][t];
            }
        }
        x++;
    }
    return cnt>=a[id][0];
}
bool check(int x,int y){
    for(int i=1;i<=m;i++)
        if(!query(i,x,y,y-x+1)) return 0;
    return 1;
}
int main(){
    cin>>s>>m;
    for(int i=1;i<=m;i++){
        char p[maxn];
        int l,r;
        cin>>p>>l>>r;
        init(p,i,l,r);
    }
    getfail();
    int ans=0;
    for(int i=0;i<strlen(s);i++)
        for(int j=i;j<strlen(s);j++){
            if(check(i,j)) ans++;
            else{
                for(int k=0;k<23;k++)
                    for(int l=0;l<maxl;l++)
                        ok[k][l]=0;
                break;
            }
        }
    cout<<ans<<endl;
    return 0;
}
