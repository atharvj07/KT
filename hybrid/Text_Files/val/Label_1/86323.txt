import java.util.*;
import java.lang.*;

class Main {
  static final int UNVISITED = -1;

  class Node {
    int x, y;
    Node(int x, int y) { this.x = x; this.y = y; }
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int N = sc.nextInt();
    int M = sc.nextInt();

    ArrayList<ArrayList<Integer>> G = new ArrayList<>();
    int[] paired = new int[N + 1];
    int[] deg = new int[N + 1];
    Arrays.fill(paired, UNVISITED);
    for (int i = 0; i <= N; i++) G.add(new ArrayList<>());
    for (int i = 0; i < M; i++) {
      int x = sc.nextInt();
      int y = sc.nextInt();
      G.get(x).add(y);
      G.get(y).add(x);
      deg[x]++;
      deg[y]++;
    }

    Queue<Integer> q = new LinkedList<>();
    for (int i = 1; i <= N; i++) {
      if (deg[i] == 1) {
        q.add(i);
      }
    }

    int[] ans = new int[N + 1];
    Arrays.fill(ans, UNVISITED);
    while (!q.isEmpty()) {
      int x = q.poll();
      if (x == 1) break;
      for (int y : G.get(x)) {
        if (paired[x] != y) {
          paired[x] = y;
          paired[y] = x;
          ans[x] = y;
          ans[y] = x;
          deg[y]--;
          if (deg[y] == 1) {
            q.add(y);
          }
          break;
        }
      }
    }

    if (paired[1] == UNVISITED) {
      System.out.println("No");
    } else {
      System.out.println("Yes");
      for (int i = 2; i <= N; i++) {
        System.out.println(ans[i]);
      }
    }
  }
}
