#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> a(n + 2);
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  int sum = 0;
  vector<int> r(n + 2, 0);
  for (int i = n; i >= 1; i--) {
    sum += a[i];
    r[i] = sum - k * ((sum + k - 1) / k);
    r[i] = max(r[i], r[i + 1]);
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    ans = max(ans, r[i]);
    sum += a[i - 1];
    r[i] = sum - k * ((sum - k) / k);
    r[i] = max(r[i], r[i + 1]);
  }
  cout << ans << "\n";
  return 0;
}
