#include <bits/stdc++.h>
using namespace std;
#define int long long
const int maxn = 1e5 + 5;
vector<int> p[maxn];
void divisors(int up) {
    for (int i = 1; i * i <= up; i++) {
        if (up % i == 0) {
            p[i].push_back(up / i);
            p[up / i].push_back(i);
        }
    }
}
int32_t main() {
    int n;
    cin >> n;
    for (int i = 2; i <= n; i++) divisors(i);
    vector<int> ans;
    for (int i = 2; i <= n; i++) {
        for (int j : p[i]) {
            if (j <= i) {
                ans.push_back(i);
                ans.push_back(j);
            }
        }
    }
    cout << ans.size() / 2 << endl;
    for (int i = 0; i < ans.size(); i += 2) {
        cout << ans[i] << " " << ans[i + 1] << endl;
    }
}
