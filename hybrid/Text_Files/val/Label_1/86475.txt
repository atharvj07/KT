import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        ArrayList<BigDecimal> a = new ArrayList<>(n);

        for (int i = 0; i < n; i++) {
            a.add(input.nextBigDecimal());
        }

        Collections.sort(a);
        int ans = 0;

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (a.get(i).multiply(a.get(j)).compareTo(BigDecimal.valueOf(Math.round(a.get(i).multiply(a.get(j)).doubleValue()))) == 0) {
                    ans++;
                }
            }
        }

        System.out.println(ans);
    }
}
