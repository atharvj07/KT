#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 5;
vector<int> g[N];
int dist[N], sub[N], up[N];

void dfs1(int v, int p) {
  dist[v] = dist[p] + 1;
  for (int u : g[v]) {
    if (u == p) continue;
    dfs1(u, v);
  }
}

void dfs2(int v, int p) {
  sub[v] = 1;
  for (int u : g[v]) {
    if (u == p) continue;
    dfs2(u, v);
    sub[v] += sub[u];
  }
}

void dfs3(int v, int p) {
  for (int u : g[v]) {
    if (u == p) continue;
    up[u] = up[v] + sub[v] - 2 * sub[u];
    dfs3(u, v);
  }
}

int main() {
  int n, m, d;
  cin >> n >> m >> d;
  vector<int> p(m);
  for (int i = 0; i < m; i++) cin >> p[i];
  for (int i = 0; i < n - 1; i++) {
    int a, b;
    cin >> a >> b;
    g[a].push_back(b);
    g[b].push_back(a);
  }

  dfs1(1, 0);
  dfs2(1, 0);
  dfs3(1, 0);
  int ans = 0;
  for (int i = 0; i < m; i++) {
    if (dist[p[i]] <= d || up[p[i]] >= m) ans++;
  }
  cout << ans << endl;
}
