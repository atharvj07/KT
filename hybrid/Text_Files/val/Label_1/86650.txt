import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int d1 = sc.nextInt();
        int d2 = sc.nextInt();
        int d3 = sc.nextInt();
        System.out.println(Math.min(d1 + d2 + d3, 2 * Math.min(d1 + d3, d2 + d3)));
    }
}
