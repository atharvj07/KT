#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(m);
  for (int i = 0; i < m; i++) cin >> a[i];
  vector<int> cnt(n + 1);
  for (int i = 0; i < m; i++) {
    cnt[a[i]]++;
    int ok = 1;
    for (int j = 1; j <= n; j++) {
      if (!cnt[j]) {
        ok = 0;
        break;
      }
    }
    cout << ok;
    if (ok)
      for (int j = 1; j <= n; j++) cnt[j]--;
  }
  cout << endl;
  return 0;
}
