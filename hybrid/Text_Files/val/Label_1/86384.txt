import java.util.*;

public class DenisAndNastya {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList<>());
        }

        for (int i = 0; i < n - 1; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            graph.get(u).add(v);
            graph.get(v).add(u);
        }

        int[] time = new int[n + 1];
        int[] parent = new int[n + 1];
        Arrays.fill(time, Integer.MAX_VALUE);

        dfs(1, 0, graph, time, parent);

        int start = 1;
        for (int i = 2; i <= n; i++) {
            if (time[i] < time[start]) {
                start = i;
            }
        }

        List<Pair> path = new ArrayList<>();
        dfs2(start, 0, graph, time, parent, path);

        System.out.println(path.size());
        for (Pair pair : path) {
            System.out.println(pair.vertex + " " + pair.time);
        }
    }

    private static void dfs(int vertex, int currentTime, List<List<Integer>> graph, int[] time, int[] parent) {
        time[vertex] = currentTime;
        for (int nextVertex : graph.get(vertex)) {
            if (nextVertex == parent[vertex]) {
                continue;
            }
            parent[nextVertex] = vertex;
            dfs(nextVertex, currentTime + 1, graph, time, parent);
        }
    }

    private static void dfs2(int vertex, int currentTime, List<List<Integer>> graph, int[] time, int[] parent, List<Pair> path) {
        path.add(new Pair(vertex, currentTime));
        for (int nextVertex : graph.get(vertex)) {
            if (nextVertex == parent[vertex]) {
                continue;
            }
            dfs2(nextVertex, currentTime + 1, graph, time, parent, path);
            path.add(new Pair(vertex, currentTime));
        }
    }

    private static class Pair {
        int vertex;
        int time;

        public Pair(int vertex, int time) {
            this.vertex = vertex;
            this.time = time;
        }
    }
}
