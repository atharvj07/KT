import java.util.*;
import java.lang.*;

class Main {
    static boolean[][] was;
    static int[] dx = {0, 0, -1, 1};
    static int[] dy = {1, -1, 0, 0};
    static int n, m;
    static int[][] a = new int[52][52];
    static int[][] b = new int[52][52];
    static Queue<int[]> q = new LinkedList<>();
    static int ans = 10800;
    static ArrayList<int[]> v = new ArrayList<>();

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        for (int i = 1; i <= m; i++) {
            a[in.nextInt()][in.nextInt()] = i;
        }
        for (int i = 1; i <= m; i++) {
            b[in.nextInt()][in.nextInt()] = i;
        }
        was = new boolean[52][52];
        q.add(new int[]{1, 1});
        was[1][1] = true;
        while (!q.isEmpty()) {
            int[] u = q.poll();
            if (a[u[0]][u[1]] != b[u[0]][u[1]]) {
                int zx = b[u[0]][u[1]];
                for (int i = 1; i <= n; i++) {
                    for (int j = 1; j <= n; j++) {
                        if (a[i][j] == zx && !was[i][j]) {
                            q.add(new int[]{i, j});
                            was[i][j] = true;
                            v.add(new int[]{u[0], u[1], i, j});
                            if (v.size() > 10800) {
                                System.out.println(-1);
                                return;
                            }
                        }
                    }
                }
            } else {
                ans--;
                if (ans < 0) {
                    System.out.println(-1);
                    return;
                }
            }
        }
        System.out.println(v.size());
        for (int[] i : v) {
            System.out.println(i[0] + " " + i[1] + " " + i[2] + " " + i[3]);
        }
    }
}
