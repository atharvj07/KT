import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        int m = in.nextInt();
        for (int i = 0; i < m; i++) {
            int k = in.nextInt();
            int d = in.nextInt();
            StringBuilder sb = new StringBuilder();
            for (int j = 0; j < k; j++) {
                for (int l = 0; l + j < s.length(); l += k) {
                    sb.append(s.charAt(l + j));
                }
            }
            s = sb.toString();
            System.out.println(s);
        }
    }
}
