n = int(input())
events = []

for _ in range(n):
    b, d = map(int, input().split())
    events.append((b, 1))  # +1 for birth
    events.append((d, -1))  # -1 for death

# Sort events by year and then by type of event (death before birth in the same year)
events.sort()

max_alive = 0
current_alive = 0
year = 0

for event in events:
    current_alive += event[1]
    if current_alive > max_alive:
        max_alive = current_alive
        year = event[0]

print(year, max_alive)
