import java.util.*;
import java.lang.*;

public class Main {
    static char[][] map = new char[5001][5001];
    static boolean[][] v = new boolean[5001][5001];

    static int dfs(int x, int y) {
        if (v[x][y] || map[x][y] == '.') {
            return 0;
        }
        v[x][y] = true;
        int ans = 1;
        if (map[x][y] == 'U') {
            for (int i = x - 1; i >= 0; i--) {
                if (map[i][y] != 'U') {
                    break;
                }
                v[i][y] = true;
                ans++;
            }
        } else if (map[x][y] == 'D') {
            for (int i = x + 1; i < map.length; i++) {
                if (map[i][y] != 'D') {
                    break;
                }
                v[i][y] = true;
                ans++;
            }
        } else if (map[x][y] == 'L') {
            for (int j = y - 1; j >= 0; j--) {
                if (map[x][j] != 'L') {
                    break;
                }
                v[x][j] = true;
                ans++;
            }
        } else {
            for (int j = y + 1; j < map[0].length; j++) {
                if (map[x][j] != 'R') {
                    break;
                }
                v[x][j] = true;
                ans++;
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt(), m = scan.nextInt();
        for (int i = 0; i < n; i++) {
            String s = scan.next();
            for (int j = 0; j < s.length(); j++) {
                map[i][j] = s.charAt(j);
            }
        }
        int ans1 = 0, ans2 = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int cur = dfs(i, j);
                if (cur > ans1) {
                    ans1 = cur;
                    ans2 = 1;
                } else if (cur == ans1) {
                    ans2++;
                }
            }
        }
        System.out.println(ans1 + " " + ans2);
    }
}
