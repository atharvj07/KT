import sys
path1 = sys.stdin.readline().strip()
path2 = sys.stdin.readline().strip()
if len(path1) != len(path2):
    print("NO")
else:
    i = 0
    while i < len(path1):
        if path1[i] == path2[i]:
            i += 1
        else:
            if i == 0:
                if path1[i] == path2[i+1] and path1[i+1] == path2[i]:
                    i += 2
                else:
                    break
            elif i == len(path1) - 1:
                if path1[i] == path2[i-1] and path1[i-1] == path2[i]:
                    i += 1
                else:
                    break
            else:
                if path1[i] == path2[i+1] and path1[i+1] == path2[i-1]:
                    i += 2
                else:
                    break
    if i == len(path1):
        print("YES")
    else:
        print("NO")
