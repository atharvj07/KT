#include<bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
typedef long long ll;
ll gcd(ll a, ll b){
    if(b == 0) return a;
    return gcd(b, a % b);
}
ll lcm(ll a, ll b){
    return (a / gcd(a, b)) * b;
}
long long bigmod(long long a,long long b,long long m){
    if(b == 0) return 1LL;
    if(b == 1) return a;
    long long int x = bigmod(a,b >> 1,m)%m;
    x = (x * x)%m;
    if(b % 2 == 1) return (x * a)%m;
    else return x;
}
int main(){
    ll n, m, k;
    cin >> n >> m >> k;
    ll l = lcm(n, m);
    ll ans = k % l + 1;
    cout << ans << endl;
    return 0;
}
