from collections import deque, defaultdict

n, k = map(int, input().split())
a = list(map(int, input().split()))
g = defaultdict(list)
for i in range(n - 1):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    g[u].append(v)
    g[v].append(u)
vis = [False] * n
mn = []
q = deque()
q.append((0, -1))
while q:
    u, p = q.popleft()
    mn.append(a[u])
    vis[u] = True
    for v in g[u]:
        if vis[v] or v == p:
            continue
        q.append((v, u))
mn.sort()
print(mn[k - 1])
