#include <bits/stdc++.h>

int main() {
    int n;
    std::cin >> n;
    std::unordered_map<int, std::pair<int, int>> obelisks, clues;
    for (int i = 0; i < n; ++i) {
        int x, y;
        std::cin >> x >> y;
        obelisks[x + y] = {x, y};
    }
    for (int i = 0; i < n; ++i) {
        int a, b;
        std::cin >> a >> b;
        clues[a + b] = {a, b};
    }
    std::pair<int, int> treasure = {0, 0};
    for (auto& [key, val] : obelisks) {
        if (clues.count(key)) {
            auto& [dx, dy] = clues[key];
            treasure.first += val.first + dx;
            treasure.second += val.second + dy;
        }
    }
    treasure.first /= 2;
    treasure.second /= 2;
    std::cout << treasure.first << ' ' << treasure.second << '\n';
    return 0;
}
