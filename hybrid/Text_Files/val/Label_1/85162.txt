#include <bits/stdc++.h>

using ll = long long;

int main() {
    int n;
    std::cin >> n;
    std::vector<std::vector<int>> a(n, std::vector<int>(n));
    ll sum = 0, xr = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            std::cin >> a[i][j];
            sum += a[i][j];
            xr ^= a[i][j];
        }
    }
    int r = -1, c = -1;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (a[i][j] == 0) {
                r = i, c = j;
                break;
            }
        }
    }
    if (r == -1) {
        std::cout << -1 << "\n";
        return 0;
    }
    int magic = sum / n;
    int mis = magic - std::accumulate(a[r].begin(), a[r].end(), 0LL);
    if (mis <= 0 || (mis & 1) || xr != mis) {
        std::cout << -1 << "\n";
    } else {
        std::cout << mis << "\n";
    }
    return 0;
}
