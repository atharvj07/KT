#include <bits/stdc++.h>
using namespace std;

const long long int MOD = 1e9+7;

long long int sum_rows[510];
long long int sum_columns[510];
long long int sum_diag1, sum_diag2;
long long int sum = 0;
int grid[510][510], r, c;

int main() {
    ios_base::sync_with_stdio(false); cin.tie(NULL);
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            cin >> grid[i][j];
            sum_rows[i] += grid[i][j];
            sum_columns[j] += grid[i][j];
            if (i == j) sum_diag1 += grid[i][j];
            if (i + j == n + 1) sum_diag2 += grid[i][j];
            sum += grid[i][j];
        }
    }
    for (int i = 1; i <= n; i++) {
        if (sum_rows[i] != sum_columns[i]) {
            cout << -1 << endl;
            return 0;
        }
    }
    if (sum_diag1 != sum_diag2 || sum_diag1 != sum_rows[1]) {
        cout << -1 << endl;
        return 0;
    }
    long long int target_sum = sum / n;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (grid[i][j] == 0) {
                r = i;
                c = j;
            }
        }
    }
    cout << target_sum - (sum_rows[r] - grid[r][c]) << endl;
    return 0;
}
