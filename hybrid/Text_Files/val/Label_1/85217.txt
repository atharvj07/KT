#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int INF = 1e9;
const int maxn = 5e5 + 5;
char s[maxn], t[maxn];
int n, k, f[maxn][2][2], g[maxn][2][2];
ll ans;
int main() {
    cin >> n >> k >> s + 1 >> t + 1;
    f[0][0][0] = 1;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= 1; j++) {
            for (int k = 0; k <= 1; k++) {
                if (j == 0 || (s[i] == 'a' && t[i] == 'b')) { f[i][j][k] |= f[i - 1][j][k]; }
                if (k == 0 || (s[i] == 'b' && t[i] == 'a')) { f[i][j][k] |= f[i - 1][j][k]; }
                if (j && k) { f[i][j][k] = 1; }
            }
        }
    }
    g[n + 1][0][0] = 1;
    for (int i = n; i >= 1; i--) {
        for (int j = 0; j <= 1; j++) {
            for (int k = 0; k <= 1; k++) {
                if (j == 0 || (s[i] == 'a' && t[i] == 'b')) { g[i][j][k] |= g[i + 1][j][k]; }
                if (k == 0 || (s[i] == 'b' && t[i] == 'a')) { g[i][j][k] |= g[i + 1][j][k]; }
                if (j && k) { g[i][j][k] = 1; }
            }
        }
    }
    for (int i = n; i >= 1; i--) {
        for (int j = 0; j <= 1; j++) {
            for (int k = 0; k <= 1; k++) {
                for (int c = 0; c <= 1; c++) {
                    for (int d = 0; d <= 1; d++) {
                        if (f[i - 1][j][c] && g[i + 1][k][d] && (j == k || ((c == 1 && d == 0) || (c == 0 && d == 1)))) { ans++; }
                    }
                }
            }
        }
    }
    ans++;
    cout << ans << '\n';
    return 0;
}
