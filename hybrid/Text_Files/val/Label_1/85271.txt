#include <bits/stdc++.h>

using namespace std;
using ll = long long;
using ld = long double;
using vi = vector<int>;
using vvi = vector<vector<int>>;
using vb = vector<bool>;
using pii = pair<int, int>;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  set<pii> st;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int h, m;
    cin >> h >> m;
    if (st.count({h, m})) continue;
    st.insert({h, m});
    if (st.size() > ans) ans = st.size();
    if (st.count({h, m + 1})) st.erase({h, m + 1});
    if (st.count({h, m + 2})) st.erase({h, m + 2});
  }
  cout << ans << "\n";
  return 0;
}
