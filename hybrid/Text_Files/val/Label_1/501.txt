from sys import stdin, stdout
n = int(stdin.readline())
for _ in range(n):
    a, b, m = map(int, stdin.readline().split())
    if a > b or (b - a + 1) % (m + 1) != 0:
        print(-1)
    else:
        d = (b - a) // (m + 1)
        k = (b - a) % (m + 1)
        r = [d for _ in range(k)] + [d + 1 for _ in range(m + 1 - k)]
        stdout.write(f'{m + 1}\n{" ".join(map(str, [a] + r + [b]))}\n')
