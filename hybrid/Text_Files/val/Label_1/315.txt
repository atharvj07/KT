n, m = map(int, input().split())
k = list(map(int, input().split()))
special_offers = [list(map(int, input().split())) for _ in range(m)]

special_offers_by_type = [[] for _ in range(n + 1)]
for d, t in special_offers:
    special_offers_by_type[t].append(d)

for offers in special_offers_by_type:
    offers.sort()

def min_days(n, m, k, special_offers_by_type):
    left, right = 1, sum(k) * 2
    while left < right:
        mid = (left + right) // 2
        money = mid
        cost = 0
        for i in range(1, n + 1):
            full_price = k[i - 1]
            discount_days = special_offers_by_type[i]
            for day in discount_days:
                if day <= mid:
                    full_price -= 1
            cost += full_price * 2
            discount = len(discount_days) - discount_days.count(mid)
            cost -= discount
        if cost <= money:
            right = mid
        else:
            left = mid + 1
    return left

print(min_days(n, m, k, special_offers_by_type))
