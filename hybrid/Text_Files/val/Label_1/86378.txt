import java.util.Scanner;

public class TreeInvariant {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt() - 1;
        }

        if (!isPermutation(p)) {
            System.out.println("NO");
            return;
        }

        int[][] tree = new int[n - 1][2];
        for (int i = 0; i < n - 1; i++) {
            tree[i][0] = i + 1;
            tree[i][1] = p[i] + 1;
        }

        System.out.println("YES");
        for (int i = 0; i < n - 1; i++) {
            System.out.println(tree[i][0] + " " + tree[i][1]);
        }
    }

    private static boolean isPermutation(int[] arr) {
        boolean[] seen = new boolean[arr.length];
        for (int x : arr) {
            if (x < 0 || x >= arr.length || seen[x]) {
                return false;
            }
            seen[x] = true;
        }
        return true;
    }
}
