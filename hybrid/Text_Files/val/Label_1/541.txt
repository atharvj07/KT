n, k = map(int, input().split())
powers = list(map(int, input().split()))
coins = list(map(int, input().split()))

# Preparing tuples of (power, coins, index), to sort knights by power
knights = sorted([(powers[i], coins[i], i) for i in range(n)])

# To store the results
max_coins = [0] * n

# Use a list to keep track of the smallest 'k' coins among the knights that can be killed
k_coins = []

for power, coin, idx in knights:
    # Sum the smallest 'k' coins and assign it to the corresponding knight
    max_coins[idx] = coin + sum(k_coins)
    # Add current knight's coins and keep the list sorted
    k_coins.append(coin)
    k_coins.sort(reverse=True)
    # If we have more than 'k' coins in the list, remove the smallest one (as knight can kill no more than k knights)
    if len(k_coins) > k:
        k_coins.pop()

# Output the max coins each knight can have
print(' '.join(map(str, max_coins)))
