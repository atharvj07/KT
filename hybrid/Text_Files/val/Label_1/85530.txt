#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, ans = 0;
  cin >> n;
  unordered_map<int, int> a;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) {
    if (a[i] == 0) continue;
    a[i - 1]++;
    a[i]++;
    a[i + 1]++;
    a[i] = 0;
    ans++;
  }
  cout << ans << endl;
  return 0;
}
