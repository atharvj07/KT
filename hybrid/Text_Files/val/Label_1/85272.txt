#include <bits/stdc++.h>
using namespace std;
pair<int, int> a[100001];
int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int h, m;
    cin >> h >> m;
    a[i] = make_pair(h * 60 + m, i);
  }
  sort(a, a + n);
  int ans = 1, cnt = 0, cur = a[0].first;
  for (int i = 1; i < n; i++) {
    if (a[i].first - cur >= 60) {
      cnt = 0;
    } else
      cnt++;
    ans = max(ans, cnt + 1);
    cur = a[i].first;
  }
  cout << ans << endl;
  return 0;
}
