#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  int h = stoi(s.substr(0, 2));
  int m = stoi(s.substr(3, 2));
  int min = 1440 - (h * 60 + m);
  for (int i = 1; i <= 59; i++) {
    if (i >= m) min = min < i - m ? min : i - m;
    if ((h + i / 60) % 10 == i % 10 && (h + i / 60) / 10 == i / 10) min = min < i - m ? min : i - m;
  }
  cout << min << endl;
  return 0;
}
