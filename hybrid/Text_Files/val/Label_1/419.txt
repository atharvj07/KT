n = int(input())
s = input()

# Counters for opening and closing brackets
open_count, close_count = 0, 0

# First pass to count the total number of each type of bracket
for bracket in s:
    if bracket == '(':
        open_count += 1
    else:
        close_count += 1

# If the difference between the number of opening and closing brackets is not 0 or 2, 
# it's impossible to make the sequence regular by changing just one bracket
if abs(open_count - close_count) != 2:
    print(0)
else:
    # Variables to keep track of the number of changes that could make the sequence regular
    change = 0
    balance = 0
    
    # The type of bracket that is more than the other
    target_bracket = '(' if open_count > close_count else ')'
    
    for bracket in s:
        if bracket == target_bracket:
            balance += 1
        else:
            balance -= 1
        
        # If balance is -2, it means changing the current bracket type could potentially make the sequence regular
        if balance == -2:
            change += 1
            balance = 0
    
    print(change)
