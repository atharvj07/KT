n,m,k=map(int,input().split())
a=[list(map(int,input().split()))for x in range(n)]
for x in a:
	if len(set(x))!=1:
		chk='.'
		break
	chk=x[0]
if chk=='.':
	print('-1')
	exit()
b=sorted([[x,y]for x in range(n)for y in range(m)if a[x][y]!=chk])
#print(b)
ans=0
for x in b:
	if k>=0:
		ans+=1
		k-=1
		if k==0:break
		a[x[0]][x[1]]=chk
	else:
		print('-1')
		exit()
print(ans)
