import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }
        int[][] queries = new int[q][2];
        for (int i = 0; i < q; i++) {
            queries[i][0] = sc.nextInt();
            queries[i][1] = sc.nextInt();
        }
        for (int[] query : queries) {
            int x = query[0];
            int y = query[1];
            if (dfs(x, y, a, new boolean[n + 1])) {
                System.out.println("Shi");
            } else {
                System.out.println("Fou");
            }
        }
    }
    static boolean dfs(int x, int y, int[] a, boolean[] visited) {
        if (x == y) {
            return true;
        }
        if (visited[x]) {
            return false;
        }
        visited[x] = true;
        if ((a[x] & a[x + 1]) > 0) {
            if (dfs(x + 1, y, a, visited)) {
                return true;
            }
        }
        visited[x] = false;
        return false;
    }
}
