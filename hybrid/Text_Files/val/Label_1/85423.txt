#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, k;
  cin >> n >> k;
  long long ans = 1;
  if (k == 1)
    ans = 0;
  else if (k == 2)
    ans = 1;
  else if (k % 2 == 0)
    ans = 2;
  else {
    k--;
    if (n == k)
      ans = n;
    else if ((k + 1) / 2 <= n - (k + 1))
      ans = k + 1;
    else if ((k + 1) > n - (k + 1))
      ans = (k + 1) ^ (n);
  }
  cout << ans << endl;
  return 0;
}
