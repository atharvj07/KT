import java.util.Arrays;
import java.util.Scanner;

public class Main {

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long[] arr = new long[n];
    for (int i = 0; i < n; i++) {
      arr[i] = sc.nextLong();
    }
    Arrays.sort(arr);
    int ans = 0;
    for (int i = 1; i < n; i++) {
      if (arr[i - 1] > arr[i]) {
        long diff = arr[i - 1] - arr[i];
        if (diff % 3 == 0) {
          ans += diff / 3;
        } else {
          ans = -1;
          break;
        }
      }
    }
    System.out.println(ans);
  }
}
