from collections import deque

def find_min_max_f(n, edges):
    # Convert edge list to adjacency list
    graph = [[] for _ in range(n + 1)]
    for a, b in edges:
        graph[a].append(b)
        graph[b].append(a)
    
    # Find all leaves
    leaves = [i for i in range(1, n + 1) if len(graph[i]) == 1]
    
    # Function to find the maximum depth from any node
    def max_depth(node):
        visited = [False] * (n + 1)
        queue = deque([(node, 0)])
        visited[node] = True
        max_d = 0
        while queue:
            current, depth = queue.popleft()
            max_d = max(max_d, depth)
            for neighbor in graph[current]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    queue.append((neighbor, depth + 1))
        return max_d
    
    # Finding maximum possible f by finding the maximum depth from any leaf
    max_f = max_depth(leaves[0]) + 1
    
    # Finding minimum possible f
    # Assigning 0 to all edges will satisfy the condition, so f_min is always 1 if the tree is a straight line
    # Otherwise, it's the number of nodes with degree >= 3
    degree_three_or_more = sum(1 for i in range(1, n + 1) if len(graph[i]) >= 3)
    min_f = 1 if degree_three_or_more == 0 else 3
    
    return min_f, max_f

n = int(input().strip())
edges = [tuple(map(int, input().strip().split())) for _ in range(n - 1)]
min_f, max_f = find_min_max_f(n, edges)
print(min_f, max_f)
