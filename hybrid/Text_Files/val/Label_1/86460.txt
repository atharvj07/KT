import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;
import java.util.StringTokenizer;

public class Main {

    static final int MAX_N = 4000;
    static final int INF = 1 << 30;

    static int N, M, Q;
    static ArrayList<Edge>[] adj;
    static long[] distS, distT;
    static int[] ans;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());

        N = Integer.parseInt(st.nextToken());
        M = Integer.parseInt(st.nextToken());

        adj = new ArrayList[N];
        for (int i = 0; i < N; i++) adj[i] = new ArrayList<>();

        for (int i = 0; i < M; i++) {
            st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken()) - 1;
            int b = Integer.parseInt(st.nextToken()) - 1;
            int c = Integer.parseInt(st.nextToken());

            // 重複のない辺のみ追加
            if (a > b) adj[b].add(new Edge(a, c));
            else adj[a].add(new Edge(b, c));
        }

        Q = Integer.parseInt(br.readLine());
        ans = new int[Q];

        for (int i = 0; i < Q; i++) {
            st = new StringTokenizer(br.readLine());
            int S = Integer.parseInt(st.nextToken()) - 1;
            int T = Integer.parseInt(st.nextToken()) - 1;

            // S からの最短距離を計算
            distS = dijkstra(S);
            // T からの最短距離を計算
            distT = dijkstra(T);

            // S か T から、各頂点に到達するための最短距離を計算
            int minDist = INF;
            for (int j = 0; j < N; j++) {
                long dist = Math.min(distS[j], distT[j]);
                if (dist < INF) minDist = Math.min(minDist, (int) dist);
            }

            ans[i] = minDist;
        }

        // 答えを出力
        for (int ans : ans) System.out.println(ans);
    }

    static long[] dijkstra(int start) {
        long[] dist = new long[N];
        Arrays.fill(dist, INF);
        dist[start] = 0L;

        Queue<Integer> que = new LinkedList<>();
        que.add(start);

        while (!que.isEmpty()) {
            int current = que.remove();

            for (Edge edge : adj[current]) {
                int next = edge.to;
                long cost = edge.cost;

                // より短い距離で到達できる場合のみ更新
                if (dist[next] > dist[current] + cost) {
                    dist[next] = dist[current] + cost;
                    que.add(next);
                }
            }
        }

        return dist;
    }

    static class Edge {
        int to;
        long cost;

        Edge(int to, long cost) {
            this.to = to;
            this.cost = cost;
        }
    }
}
