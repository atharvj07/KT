import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

class Main {
    static int id;
    static int[] dfsId;
    static int[] low;
    static boolean[] visited;
    static Map<Integer, List<Integer>> adj;

    public static void main(String[] args) throws Exception {
        id = 0;
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s = br.readLine().split(" ");
        int V = Integer.parseInt(s[0]);
        int E = Integer.parseInt(s[1]);
        dfsId = new int[V];
        low = new int[V];
        visited = new boolean[V];
        adj = new HashMap<>();
        for (int i = 0; i < V; i++) adj.put(i, new ArrayList<>());
        for (int i = 0; i < E; i++) {
            s = br.readLine().split(" ");
            int u = Integer.parseInt(s[0]);
            int v = Integer.parseInt(s[1]);
            adj.get(u).add(v);
            adj.get(v).add(u);
        }
        List<Integer> articulationPoints = new ArrayList<>();
        for (int i = 0; i < V; i++) {
            if (!visited[i]) dfs(i, -1, articulationPoints);
        }
        articulationPoints.stream().sorted().forEach(x -> System.out.println(x));
    }

    static void dfs(int u, int parent, List<Integer> articulationPoints) {
        visited[u] = true;
        dfsId[u] = low[u] = id++;
        int children = 0;
        boolean isArticulationPoint = false;
        for (int v : adj.get(u)) {
            if (!visited[v]) {
                children++;
                dfs(v, u, articulationPoints);
                low[u] = Math.min(low[u], low[v]);
                if (low[v] >= dfsId[u] && parent != -1) isArticulationPoint = true;
            } else if (v != parent) {
                low[u] = Math.min(low[u], dfsId[v]);
            }
        }
        if ((parent == -1 && children > 1) || isArticulationPoint) articulationPoints.add(u);
    }
}
