n=int(input())
lis=list(map(int,input().split()))
chest=0
back=0
biceps=0
for i in range(len(lis)):
    if (i+1)%3==1:
        chest+=lis[i]
    elif (i+1)%3==2:
        biceps+=lis[i]
    else:
        back+=lis[i]
if chest>back and chest>biceps:
    print("chest")
elif back>chest and back>biceps:
    print("back")
else:
    print("biceps")
