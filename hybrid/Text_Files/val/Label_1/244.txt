n, m = map(int, input().split())
x = list(map(int, input().split()))
t = list(map(int, input().split()))
pos = [i for i in range(len(x)) if t[i]]
ans = [0] * m
for i in range(n + m):
    if t[i]:
        j = pos.index(i)
        if j > 0:
            ans[j] -= 1
        if j < m - 1:
            ans[j + 1] += 1
for i in range(1, m):
    ans[i] += ans[i-1]
for a in ans:
    print(a + n)
