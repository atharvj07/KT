#include<bits/stdc++.h>
using namespace std;
int main(){
    int t; cin >> t;
    while (t--) {
        int n, a, b; cin >> n >> a >> b;
        string s (n, '0');
        for (int i = 0; i < n; i += a) {
            set<char>st;
            for (int j = 0; j < a && i + j < n; j++) {
                st.insert(s[i+j]);
            }
            if ((int)st.size() < b) {
                st.clear();
                for (int j = 0; j < a && i + j < n; j++) {
                    s[i+j] = 'a' + j % b;
                    st.insert(s[i+j]);
                }
                while ((int)st.size() < b && i + a < n) {
                    s[i+a] = 'a' + st.size();
                    st.insert(s[i+a]); i+=a;
                }
            }
        }
        cout << s << endl;
    }
    return 0;
}
