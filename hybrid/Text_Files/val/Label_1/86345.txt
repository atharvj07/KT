import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class NetworkPacketRouting {

    static class Router {
        int id;
        List<Integer> connectedRouters;

        public Router(int id) {
            this.id = id;
            this.connectedRouters = new ArrayList<>();
        }
    }

    static class Packet {
        int source;
        int destination;
        int ttl;

        public Packet(int source, int destination, int ttl) {
            this.source = source;
            this.destination = destination;
            this.ttl = ttl;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        Router[] routers = new Router[n + 1];
        for (int i = 1; i <= n; i++) {
            routers[i] = new Router(i);
        }

        for (int i = 1; i <= n; i++) {
            int r = scanner.nextInt();
            int k = scanner.nextInt();
            for (int j = 0; j < k; j++) {
                int t = scanner.nextInt();
                routers[r].connectedRouters.add(t);
            }
        }

        int p = scanner.nextInt();
        Packet[] packets = new Packet[p];
        for (int i = 0; i < p; i++) {
            int s = scanner.nextInt();
            int d = scanner.nextInt();
            int v = scanner.nextInt();
            packets[i] = new Packet(s, d, v);
        }

        for (Packet packet : packets) {
            int hops = findMinimumHops(routers, packet.source, packet.destination, packet.ttl);
            System.out.println(hops == Integer.MAX_VALUE ? "NA" : hops);
        }
    }

    static int findMinimumHops(Router[] routers, int source, int destination, int ttl) {
        if (ttl <= 0) {
            return Integer.MAX_VALUE;
        }

        if (source == destination) {
            return 0;
        }

        int minHops = Integer.MAX_VALUE;
        for (int connectedRouter : routers[source].connectedRouters) {
            int hops = findMinimumHops(routers, connectedRouter, destination, ttl - 1);
            if (hops != Integer.MAX_VALUE) {
                minHops = Math.min(minHops, hops + 1);
            }
        }

        return minHops;
    }
}
