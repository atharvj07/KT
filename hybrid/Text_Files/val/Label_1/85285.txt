#include <iostream>
#include <vector>
using namespace std;

typedef long long ll;

int main() {
    ll n, m; cin >> n >> m;
    vector<vector<bool>> a(n, vector<bool>(m));
    for (ll i = 0; i < n; i++) {
        string s; cin >> s;
        for (ll j = 0; j < m; j++) {
            a[i][j] = (s[j] == '1');
        }
    }
    ll ans = 0;
    for (ll i = 0; i < n; i++) {
        for (ll j = 0; j < m; j++) {
            ll cnt = 0;
            for (ll dx = 0; i + dx < n && j + dx < m; dx++) {
                cnt += (a[i + dx][j + dx] ^ ((i + j + dx) % 2));
            }
            for (ll dx = 0; i + dx < n && j - dx >= 0; dx++) {
                cnt += (a[i + dx][j - dx] ^ ((i + j + dx) % 2));
            }
            ans += cnt % 2;
        }
    }
    cout << ans << endl;
    return 0;
}
