#include <iostream>
using namespace std;
int main() {
  int t, sx, sy, ex, ey, x, y;
  cin >> t >> sx >> sy >> ex >> ey;
  char c;
  x = sx;
  y = sy;
  while (t--) {
    cin >> c;
    if (c == 'N') {
      y++;
    } else if (c == 'S') {
      y--;
    } else if (c == 'E') {
      x++;
    } else if (c == 'W') {
      x--;
    }
    if (x == ex && y == ey) {
      cout << t + 1 << endl;
      return 0;
    }
  }
  cout << "-1" << endl;
  return 0;
}
