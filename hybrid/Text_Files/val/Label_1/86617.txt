import java.util.*;
class Main {
  static final long MOD = (long)1e9 + 7;
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int N = scanner.nextInt();
    int K = scanner.nextInt();
    int[] a = new int[K];
    for (int i = 0; i < K; i++) {
      a[i] = scanner.nextInt();
    }
    long ans = 1;
    for (int i = 0; i < K; i++) {
      int b = N - a[i];
      long c = 1;
      for (int j = 1; j <= b; j++) {
        c = c * (N - j) % MOD;
        c = c * pow(j, MOD - 2) % MOD;
      }
      ans = ans * c % MOD;
    }
    System.out.println(ans);
  }

  static long pow(long x, long p) {
    if (p == 0) {
      return 1;
    }
    if (p == 1) {
      return x;
    }
    long t = pow(x, p / 2);
    return t * t % MOD * ((p % 2 == 1) ? x : 1) % MOD;
  }
}
