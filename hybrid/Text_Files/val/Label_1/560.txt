n, m, k = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
b = [[0] * m for _ in range(n)]
for i in range(n):
    for j in range(m):
        if a[i][j] == 1:
            if b[i][j] == 0:
                q = [(i, j)]
                while q:
                    x, y = q.pop()
                    b[x][y] = 1
                    for dx, dy in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
                        nx, ny = x + dx, y + dy
                        if 0 <= nx < n and 0 <= ny < m and a[nx][ny] == 1 and b[nx][ny] == 0:
                            q.append((nx, ny))
            if b[i][j] == 1:
                k -= 1
                a[i][j] = 0
print(0 if k >= 0 else -1)
