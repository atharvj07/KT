n, m = map(int, input().split())
ar = list(map(int, input().split()))
pos = {}
ans = [-1] * m
for i in range(n):
    if ar[i] in pos:
        ans[pos[ar[i]] - 1] = min(ans[pos[ar[i]] - 1], abs(pos[ar[i]] - i - 1) if pos[ar[i]] != i else float('inf'), i - pos[ar[i]] if i != pos[ar[i]] else float('inf'))
        ans[i - 1] = min(ans[i - 1], abs(pos[ar[i]] - i - 1) if pos[ar[i]] != i else float('inf'), i - pos[ar[i]] if i != pos[ar[i]] else float('inf'))
    pos[ar[i]] = i
for i in range(m):
    print(ans[i] if ans[i] != float('inf') else -1)
