for _ in range(int(input())):
    a, b = map(int, input().split())
    s = input()
    if a + b != len(s) or a < 0 or b < 0:
        print(-1)
        continue
    x = []
    for c in s:
        if c == '0': a -= 1; x += '0'
        elif c == '1': b -= 1; x += '1'
        else:
            if a > 0: a -= 1; x += '0'
            elif b > 0: b -= 1; x += '1'
            else: print(-1); break
    if a == 0 and b == 0: y = ''.join(x)
    else: print(-1); continue
    x.reverse()
    if y == ''.join(x): print(y)
    else: print(-1)
