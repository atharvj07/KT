#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod=1e9+7;
int main(){
    ios::sync_with_stdio(false);
    cin.tie(0); cout.tie(0);
    int n,m,k,q; ll ans=0;
    cin>>n>>m>>k>>q;
    vector<ll> col(m+1);
    for(int i=0;i<k;++i){
        int a,b; cin>>a>>b;
        if(!col[b]) ans+=a-1, col[b]=1;
    }
    for(int i=0;i<q;++i){
        int a; cin>>a;
        if(col[a]) ans+=n-1, col[a]=2;
    }
    cout<<ans+m-1<<'\n';
    return 0;
}
