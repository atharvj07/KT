#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int ans = 0;
    int flag = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'F') {
            continue;
        }
        if (s[i] == 'A') {
            flag = 1;
        } else {
            if (flag) {
                ans++;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
