#include<bits/stdc++.h>
using namespace std;
#define ll long long int
const ll mod=1e9+7;
const ll MAX=1e5+10;
map<ll,vector<ll>>adj;
vector<ll>vis(MAX);
vector<ll>dist(MAX);
ll n,m,d,p[MAX];
bool ok;
void dfs(ll nd, ll dis)
{
  if(vis[nd]==1) return;
  vis[nd]=1;
  dist[nd]=dis;
  if(dis>d) {ok=false;return;}
  for(auto x:adj[nd])
  {
      dfs(x,dis+1);
      if(!ok) return;
  }
}
int main()
{
    ll i,j,a,b;
    cin>>n>>m>>d;
    for(i=0;i<m;i++) cin>>p[i];
    for(i=0;i<n-1;i++)
    {
        cin>>a>>b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    ok=true;
    ll ans=0;
    for(i=0;i<m;i++)
    {
      dfs(p[i],0);
      if(!ok) break;
      if(!vis[p[i]]) ans++;
      for(j=0;j<MAX;j++) vis[j]=0,dist[j]=0;
    }
    cout<<ans<<endl;
    return 0;
}
