import java.util.Scanner;

public class TreeInvariant {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt() - 1;
        }

        boolean[][] adjMatrix = new boolean[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (p[i] == p[j]) {
                    adjMatrix[i][j] = adjMatrix[j][i] = true;
                }
            }
        }

        if (!isTree(adjMatrix)) {
            System.out.println("NO");
            return;
        }

        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (adjMatrix[i][j]) {
                    System.out.println((i + 1) + " " + (j + 1));
                }
            }
        }
    }

    private static boolean isTree(boolean[][] adjMatrix) {
        int n = adjMatrix.length;
        int[] degrees = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (adjMatrix[i][j]) {
                    degrees[i]++;
                }
            }
        }

        for (int i = 0; i < n; i++) {
            if (degrees[i] > 1) {
                return false;
            }
        }
        return true;
    }
}
