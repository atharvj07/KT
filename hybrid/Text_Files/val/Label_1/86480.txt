import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = in.nextInt();
        }

        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (a[i] > 0) {
                ans += i - 1;
                a[i] = 0;
                a[i - 1] = 1;
            }
        }
        System.out.println(ans);
    }
}
