#include<bits/stdc++.h>
using namespace std;
int main() {
  int w, h;
  cin >> w >> h;
  vector<string> pic(h);
  for (auto &i: pic) {
    cin >> i;
    reverse(begin(i), end(i));
  }
  int nw = 2 * h, nh = 2 * w;
  vector<string> pic2(nw, string(nh, '.'));
  for (int i = 0; i < h; i++) {
    for (int j = 0; j < w; j++) {
      for (int k = 0; k < 2; k++) {
        for (int l = 0; l < 2; l++) {
          pic2[i * 2 + k][j * 2 + l] = pic[i][j];
        }
      }
    }
  }
  for (auto &i: pic2) {
    cout << i << endl;
  }
}
