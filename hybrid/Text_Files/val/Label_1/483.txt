n=int(input());a=list(map(int,input().split()));b=list(map(int,input().split()));m=int(input());a.sort();b.sort();ans=0
for i in range(n):
    l=i;r=n-1;mid=-1
    while l<=r:
        tmp=(l+r)//2
        if a[i]<=b[tmp]:
            mid=tmp
            l=tmp+1
        else:r=tmp-1
    if mid==-1:continue
    ans+=mid+1
print(ans%m)
