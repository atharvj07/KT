#!/bin/python3

def populate_chessboard(n, m, chessboard):
    for row in range(n):
        for column in range(m):
            if chessboard[row][column] == ".":
                if (row == 0 or chessboard[row - 1][column] != chessboard[row][column]) and (column == 0 or chessboard[row][column - 1] != chessboard[row][column]):
                    chessboard[row][column] = 'B'
                elif (column == m - 1 or chessboard[row][column + 1] != chessboard[row][column]) and (row == n - 1 or chessboard[row + 1][column] != chessboard[row][column]):
                    chessboard[row][column] = 'W'
    return chessboard

[n, m] = map(int, input().split())
chessboard = [list(input()) for i in range(n)]
populated_chessboard = populate_chessboard(n, m, chessboard)
for row in range(n):
    print("".join(populated_chessboard[row]))
