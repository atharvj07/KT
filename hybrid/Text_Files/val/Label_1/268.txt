import heapq as hq

for _ in range(int(input())):
    n = int(input())
    arr = list(map(int, input().split()))
    arr.sort()
    hq.heappush(arr, arr[0])
    ans = arr[0]
    for e in arr[1:]:
        hq.heappush(arr, e)
        ans = min(ans, hq.nlargest(2, arr)[-1] - hq.nsmallest(2, arr)[0])
    print(ans)
