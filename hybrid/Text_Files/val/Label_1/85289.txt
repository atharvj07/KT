#include <bits/stdc++.h>
using namespace std;
int a[200005], ans[200005], cnt;
int main() {
  int n;
  cin >> n;
  while (n--) {
    char c;
    int x;
    cin >> c >> x;
    if (c == 'L') {
      a[cnt] = x;
      cnt++;
    } else if (c == 'R') {
      a[cnt++] = x;
      ans[cnt - 1] = cnt - 1;
    } else {
      cout << ans[x - 1] << "\n";
    }
  }
}
