#include <bits/stdc++.h>
#include <cstring>
using namespace std;
typedef long long ll;
#define mod 1000000007
const int N = 1e6;
int main() {
  int n, m;
  cin >> n >> m;
  string p;
  cin >> p;
  vector<int> v(m), mp(n, 0), vis(n, 0);
  for (int i = 0; i < m; i++) { cin >> v[i]; }
  for (int i = 0; i < m; i++) {
    int pos = v[i] - 1;
    int idx = pos;
    for (int j = 0; j < p.size(); j++) {
      if (mp[idx] == 1) break;
      mp[idx] = 1;
      idx++;
    }
    if (idx - pos == p.size()) vis[pos] = 1;
  }
  int ans = 1;
  bool f = 0;
  for (int i = 0; i < n; i++) {
    if (vis[i] == 1) {
      if (f == 1) ans = (1LL * ans * 26) % mod;
      f = 1;
    } else {
      f = 0;
    }
  }
  cout << ans << endl;
}
