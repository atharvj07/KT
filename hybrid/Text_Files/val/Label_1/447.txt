import sys

def valid(t, a, b):
    for i in range(len(a)):
        if (t[i] | t[i + 1]) != a[i] or (t[i] & t[i + 1]) != b[i]:
            return False
    return True

def generate(n, a, b):
    t = [0 for i in range(n + 1)]

    for i in range(1, n):
        t[i] = (a[i - 1] & ~b[i - 1]) | b[i - 1]

    t[n] = a[n - 2] & ~b[n - 2]

    if valid(t, a, b):
        return t
    else:
        return None

n = int(sys.stdin.readline())
a = [int(x) for x in sys.stdin.readline().split()]
b = [int(x) for x in sys.stdin.readline().split()]

t = generate(n, a, b)

if t is None:
    print("NO")
else:
    print("YES")
    print(' '.join(map(str, t)))
