MOD = 10**9 + 7
n, k = map(int, input().split())
w = list(map(int, input().split()))

total_weight = sum(w)
weight_sum = total_weight * k % MOD

factorials = [1] * (n + 1)
inverse = [1] * (n + 1)
for i in range(2, n + 1):
    factorials[i] = factorials[i - 1] * i % MOD
    inverse[i] = pow(factorials[i], MOD - 2, MOD)

def binom(n, k):
    return factorials[n] * inverse[k] % MOD * inverse[n - k] % MOD

def stirling_second(n, k):
    if n == k:
        return 1
    elif k == 1:
        return pow(k, n, MOD)
    else:
        res = 0
        for i in range(k + 1):
            if i % 2 == 0:
                res = (res + binom(k, i) * pow(k - i, n, MOD)) % MOD
            else:
                res = (res - binom(k, i) * pow(k - i, n, MOD)) % MOD
        return res * inverse[k] % MOD

partitions = stirling_second(n, k)
result = (partitions * weight_sum) % MOD

print(result)
