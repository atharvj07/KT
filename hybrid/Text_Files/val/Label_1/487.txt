n = int(input())
a = list(map(int,input().split()))
b = list(map(int,input().split()))
m = int(input())
a.sort()
b.sort()
ans = 0
for i in range(n):
    l = i
    r = n - 1
    while l <= r:
        mid = (l + r) // 2
        if a[i] <= b[mid]:
            ans = (ans + mid - i) % m
            l = mid + 1
        else:
            r = mid - 1
    l = i
    r = n - 1
    while l <= r:
        mid = (l + r) // 2
        if b[i] <= a[mid]:
            ans = (ans + mid - i) % m
            l = mid + 1
        else:
            r = mid - 1
print(ans)
