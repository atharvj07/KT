import polars as pl
import numpy as np

df_sim = pl.DataFrame({
    "daily_n": [1000, 2000, 3000, 4000],
    "prob": [0.5, 0.5, 0.5, 0.6],
    "size": [1, 1, 1, 1]
})

df_sim = df_sim.with_columns(
    pl.struct(["daily_n", "prob", "size"])
    .map_elements(lambda x: int(np.random.binomial(n=x["daily_n"], p=x["prob"], size=x["size"])))
    .alias("events")
)

print(df_sim)
