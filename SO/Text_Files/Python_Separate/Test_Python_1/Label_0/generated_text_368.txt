from icu import Collator, Locale

# Create Collators for different locales
collator1 = Collator.createInstance(Locale("en_US.UTF-8"))
collator2 = Collator.createInstance(Locale("fa_IR.UTF-8"))

# Define custom sort key function
def sortKey(s):
    return collator1.getSortKey(s), collator2.getSortKey(s)

# Test list with Persian and English strings
mylist = ["????", "hello", "????", "world"]

# Sort the list with the custom sort key
mylist.sort(key=sortKey)

print(mylist)
