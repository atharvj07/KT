# Test case to check if slice(a, b, None) always behaves like slice(a, b, 1)

lst = list(range(10))

assert lst[slice(2, 8, None)] == lst[slice(2, 8, 1)]
assert lst[slice(0, None, None)] == lst[slice(0, None, 1)]
assert lst[slice(None, None, None)] == lst[slice(None, None, 1)]
assert lst[slice(3, 7, None)] == lst[slice(3, 7, 1)]
assert lst[slice(5, 5, None)] == lst[slice(5, 5, 1)]
assert lst[slice(5, 5, None)] == []

print("All tests passed.")
