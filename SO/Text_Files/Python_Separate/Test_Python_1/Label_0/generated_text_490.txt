import matplotlib
import matplotlib.pyplot as plt
import numpy as np

matplotlib.rcParams.update({
    "font.family": "serif",
    "font.serif": ["Computer Modern Roman"],
    "font.weight": "bold",
    "mathtext.fontset": "cm",
    "axes.labelweight": "bold",
    "xtick.labelsize": 14,
    "ytick.labelsize": 14,
    "xtick.labelweight": "bold",
    "ytick.labelweight": "bold",
})

t = np.arange(0.0, 2.0, 0.01)
s = 1 + np.sin(2 * np.pi * t)

fig, ax = plt.subplots(figsize=(9,6))
ax.plot(t, s)

ax.set_xlabel(r'time (s)  $a_1$', fontsize=16, fontweight='bold')
ax.set_ylabel(r'voltage (mV)', fontsize=16, fontweight='bold')
ax.set_title(r'About as simple as it gets, folks', fontsize=18, fontweight='bold')

ax.grid()
plt.show()
