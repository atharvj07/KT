import asyncio

async def foo():
    return "foo_result"

async def bar():
    return "bar_result"

async def main():
    async with asyncio.TaskGroup() as tg:
        tasks = [tg.create_task(foo()), tg.create_task(bar())]
    res = [task.result() for task in tasks]
    print(res)  # Test case output: ['foo_result', 'bar_result']

asyncio.run(main())
