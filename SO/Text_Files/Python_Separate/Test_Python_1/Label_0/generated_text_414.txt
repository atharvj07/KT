import polars as pl

# Sample DataFrame
df = pl.DataFrame(
    {
        "Group": ["Group1", "Group2", "Group3"],
        "Weight": [100.0, 200.0, 300.0],
        "Vals": [[0.5, 0.5, 0.8], [0.5, 0.5, 0.8], [0.7, 0.9]],
    }
)

# Function to compute the weighted values for each group
def calculate_weighted(vals, weight):
    return [v * weight for v in vals]

# Add a 'Weighted' column to the DataFrame by applying the function
df = df.with_columns(
    pl.col("Vals").zip(
        pl.col("Weight").apply(lambda x: calculate_weighted(x[0], x[1]), return_dtype=pl.List(pl.Float64))
    ).alias("Weighted")
)

# Show the result
print(df)
