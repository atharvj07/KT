import polars as pl

# Sample DataFrame with datetime64
df = pl.DataFrame({
    'alertTime': ["2023-03-01", "2023-03-02", "2023-03-03"]
}).with_columns(pl.col("alertTime").str.strptime(pl.Datetime, "%Y-%m-%d"))

# Function to get day of the week
def get_day(dt_obj):
    days_of_week = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday', 'Sunday']
    return days_of_week[dt_obj.weekday()]

# Apply function to get day of the week
df = df.with_columns(
    pl.col('alertTime').apply(lambda dt: get_day(dt), return_dtype=pl.Utf8).alias('day_of_week')
)

# Show result
print(df)
