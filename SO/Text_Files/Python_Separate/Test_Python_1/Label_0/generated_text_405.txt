# Function to count 3-way conversations
def count_three_way_conversations(df):
    df = df.sort_values(['group_id', 'sent_time'])
    conversations = []

    for group_id, group in df.groupby('group_id'):
        red_messages = group[group['from_red'] == True]
        green_messages = group[group['from_red'] == False]
        
        for i in range(len(red_messages)-1):
            red_1 = red_messages.iloc[i]
            if red_1['touchpoint'] == 1:
                green_msg = green_messages[(green_messages['sent_time'] > red_1['sent_time']) & (green_messages['touchpoint'] == 2)]
                for _, green in green_msg.iterrows():
                    red_2 = red_messages[(red_messages['sent_time'] > green['sent_time']) & (red_messages['touchpoint'] == 3)]
                    for _, red in red_2.iterrows():
                        conversations.append((red_1['user_id'], green['user_id'], red['user_id']))
    
    return len(conversations)
