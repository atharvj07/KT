from tqdm import tqdm
import pandas as pd
import time

# Example data
data = {'textbody': ['test1', 'test2', 'test3', 'test4', 'test5']}
df = pd.DataFrame(data)

# Ensure tqdm is properly patched
tqdm.pandas(desc="Processing")

# Example function
def func1(text):
    time.sleep(1)  # Simulate processing time
    return text.upper()

# Use progress_apply directly on the series
df['result'] = df['textbody'].progress_apply(func1)
print(df)
