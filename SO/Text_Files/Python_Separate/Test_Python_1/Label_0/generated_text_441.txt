import pandas as pd

# Input DataFrame
data = {'FIELD_A': [123123, 123124, 123144],
        'FIELD_B': [8, 7, 99],
        'FIELD_C': ['a', 'c', 'x'],
        'FIELD_D': [23423, 6464, 234]}

df = pd.DataFrame(data)

# Melting the DataFrame
melted_df = df.melt(id_vars=["FIELD_A"], var_name="FIELD_NAME", value_name="VALUE")

# Creating conditional columns
melted_df['FIELD_STR_VALUE'] = melted_df['VALUE'].where(melted_df['VALUE'].apply(lambda x: isinstance(x, str)))
melted_df['FIELD_NUM_VALUE'] = melted_df['VALUE'].where(melted_df['VALUE'].apply(lambda x: isinstance(x, (int, float))))

# Dropping the VALUE column
melted_df = melted_df.drop(columns='VALUE')

# Re-indexing and renaming the output
melted_df['ID'] = melted_df.groupby('FIELD_A').cumcount() + 1
melted_df = melted_df[['ID', 'FIELD_A', 'FIELD_NAME', 'FIELD_STR_VALUE', 'FIELD_NUM_VALUE']]

# Test case output
print(melted_df)
