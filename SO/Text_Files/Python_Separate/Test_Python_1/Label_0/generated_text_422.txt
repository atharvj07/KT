import datetime as dt
from zoneinfo import ZoneInfo

# Midnight
d0 = dt.datetime(2020, 3, 29, 0, 0, tzinfo=ZoneInfo("Europe/Paris"))

# Before DST shift
d1 = d0 + dt.timedelta(hours=2)

# After DST shift
d2 = d0 + dt.timedelta(hours=3)

# Convert to UTC
d1u = d1.astimezone(dt.timezone.utc)
d2u = d2.astimezone(dt.timezone.utc)

# Compute timedeltas
print(d2 - d1)  # Expected: 1 hour, but it is 1 hour due to the DST shift
print(d2u - d1u)  # Expected: 0 hours, since both are at the same UTC time after DST shift

# Handling DST shift explicitly by adjusting time zone manually
d3 = d0 + dt.timedelta(hours=4)
print(d3.isoformat())  # '2020-03-29T04:00:00+02:00' because 3:00 AM (DST shift) happens at this time
