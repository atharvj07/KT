from abc import ABC, abstractmethod
from typing import Any


class FooBase(ABC):
    foo: str
    bar: str
    baz: int

    def __init__(self):
        self.bar = "bar"
        self.baz = 42

    @abstractmethod
    def __init__(self, foo: str):
        self.foo = foo


class Foo(FooBase):
    foo: str

    def __init__(self, foo: str):
        super().__init__(foo)


# Test Case
foo_instance = Foo("hello")  # Works fine
# foo_instance = Foo()  # Will raise an error in type checking because 'foo' is required
