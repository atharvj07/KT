class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y
        self._tuple = (x, y)
    
    @property
    def tuple(self):
        return self._tuple

    def __repr__(self):
        return f"Point({self.x}, {self.y})"

# Test Case
p = Point(4, 5)
print(p.x)        # Access x
print(p.tuple)    # Access tuple directly
