import numpy as np

def reverse_anti_diagonals(arr):
    rows, cols = arr.shape
    result = np.zeros_like(arr)
    
    for s in range(rows + cols - 1):
        anti_diag = [arr[i, s - i] for i in range(max(0, s - cols + 1), min(rows, s + 1))]
        anti_diag.reverse()
        for i, val in zip(range(max(0, s - cols + 1), min(rows, s + 1)), anti_diag):
            result[i, s - i] = val
    
    return result

# Test case
arr = np.array([
    [ 1,  2,  4],
    [ 3,  5,  7],
    [ 6,  8, 10],
    [ 9, 11, 13],
    [12, 14, 15]
])

result = reverse_anti_diagonals(arr)
print(result)
