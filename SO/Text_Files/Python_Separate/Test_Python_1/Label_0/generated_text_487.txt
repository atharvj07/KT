import matplotlib
matplotlib.use('TkAgg')  # Ensure proper backend
import matplotlib.pyplot as plt

# Read and process data
with open('notebook.txt', encoding='utf-8') as file:
    contents = [int(line.strip().split(':')[-1]) for line in file]

# Plot data
x = list(range(1, len(contents) + 1))
y = contents

plt.plot(x, y, 'r', label='Money Trend')
plt.xlabel('Month')
plt.ylabel('Money')
plt.legend()
plt.show()
