import numpy as np

def max_diff_partition(arr, a, b):
    n = len(arr)
    dp = [float('-inf')] * (n + 1)
    dp[0] = 0

    for i in range(1, n + 1):
        for j in range(a, b + 1):
            if i - j >= 0:
                subarray = arr[i - j:i]
                diff = max(subarray) - min(subarray)
                dp[i] = max(dp[i], dp[i - j] + diff)

    if dp[n] == float('-inf'):
        return []

    partitions = []
    i = n
    while i > 0:
        for j in range(a, b + 1):
            if i - j >= 0 and dp[i] == dp[i - j] + (max(arr[i - j:i]) - min(arr[i - j:i])):
                partitions.insert(0, arr[i - j:i])
                i -= j
                break

    return partitions

# Test cases
arr1 = [5, 8, 4, 5, 1, 3, 5, 1, 3, 1]
a1, b1 = 3, 7
result1 = max_diff_partition(arr1, a1, b1)
print(result1)  # Output: [[5, 8, 4], [5, 1, 3], [5, 1, 3, 1]]

arr2 = [1, 6, 2, 2, 5, 2, 8, 1, 5, 6]
a2, b2 = 3, 4
result2 = max_diff_partition(arr2, a2, b2)
print(result2)  # Output: [[1, 6, 2], [2, 5, 2, 8], [1, 5, 6]]

arr3 = [5, 8, 4, 5, 1, 3, 5, 1, 3, 1, 2]
a3, b3 = 4, 5
result3 = max_diff_partition(arr3, a3, b3)
print(result3)  # Output: []
