import pandas as pd

id = ['a','b','c','a','b','e','f','a','b','k']
value = [12,14,3,13,16,7,4,6,10,18]

df = pd.DataFrame({'id': id, 'value': value})

df["last_occurrence"] = ~df.duplicated("id", keep="last")
df["filtered_value"] = df["value"] * df["last_occurrence"]
df["desired_output"] = df["filtered_value"].cumsum()

df = df.drop(columns=["last_occurrence", "filtered_value"])

print(df)
