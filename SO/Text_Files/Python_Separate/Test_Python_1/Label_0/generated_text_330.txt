import numpy as np
from numpy.lib.stride_tricks import sliding_window_view

def find_largest_neighbor_direction(a):
    swv = sliding_window_view(a, (3, 3))  # (L-2) x (L-2) x 3 x 3
    center = swv[:, :, 1, 1]
    diffs = swv - center[:, :, None, None]
    directions = np.argmax(diffs[:, :, [0, 1, 2, 3, 5, 6, 7, 8]], axis=2).astype(np.uint8)
    return directions

# Test case
L = 1000
a = np.random.rand(L, L).astype(np.float32)
directions = find_largest_neighbor_direction(a)
assert directions.shape == (L-2, L-2)
