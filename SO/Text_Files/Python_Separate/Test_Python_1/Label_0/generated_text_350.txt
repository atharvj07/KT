import geopandas as gpd
import os
import requests
import zipfile

# Define the URL for the map
url = "https://www.naturalearthdata.com/downloads/110m-cultural-vectors/110m_admin_0_countries.zip"
file_name = "110m_admin_0_countries.zip"
extracted_folder = "world_map"

# Download the zip file
if not os.path.exists(file_name):
    r = requests.get(url)
    with open(file_name, 'wb') as f:
        f.write(r.content)

# Extract the zip file
if not os.path.exists(extracted_folder):
    with zipfile.ZipFile(file_name, 'r') as zip_ref:
        zip_ref.extractall(extracted_folder)

# Load the shapefile into geopandas
shapefile_path = os.path.join(extracted_folder, "ne_110m_admin_0_countries.shp")
world = gpd.read_file(shapefile_path)

# Test the output
print(world.head())
