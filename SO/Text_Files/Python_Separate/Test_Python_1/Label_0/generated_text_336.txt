from dataclasses import dataclass
from typing import TypeVar, Callable

T = TypeVar('T')

def something_else(klass):
    return klass

def my_dataclass(klass: Callable[[T], T]) -> Callable[[T], T]:
    return something_else(dataclass(klass))

@dataclass
class TestA:
    a: int
    b: str

TestA(0, "")

@my_dataclass
class TestB:
    a: int
    b: str

TestB(0, "")  # works correctly
