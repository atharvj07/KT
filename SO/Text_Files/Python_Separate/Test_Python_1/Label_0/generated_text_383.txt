import polars as pl

data = {
    "a": [[1], [2], [3, 4], [5, 6, 7]],
    "b": [[], [8], [9, 10], [11, 12]],
}

df = pl.DataFrame(data)

df_result = df.with_columns(
    pl.struct(["a", "b"])
    .map_elements(lambda row: (row["a"][: len(row["b"])], row["b"][: len(row["a"])]))
    .alias("truncated")
).select(pl.col("truncated").struct.field("a"), pl.col("truncated").struct.field("b")).explode(["a", "b"])

print(df_result)
