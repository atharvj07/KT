import numpy as np
import pandas as pd

df = pd.DataFrame({'column': [1, 2, 3, np.nan, 4, 5, 6]})

# np.cumsum with array
result_array = np.cumsum([1, 2, 3, np.nan, 4, 5, 6])

# np.cumsum with pandas DataFrame column
result_df = np.cumsum(df['column'])

print("np.cumsum with array:", result_array)
print("np.cumsum with DataFrame column:", result_df)
