import polars as pl

df = pl.DataFrame({
    "ABC": ["foo", "bar", "foo"],
    "A": [1, 2, 3],
    "B": [2, 1, None],
    "C": [1, 2, 3]
})

df = df.with_columns(
    pl.concat_list(pl.all().exclude("ABC"))  # Exclude non-numeric columns
    .list.eval(pl.element().drop_nulls().median())
    .alias("Horizontal Median")
)

print(df)
