import polars as pl
import numpy as np
import functools

# Small example
A = pl.DataFrame({
    "index": [0, 20, 40],
    "A": ['a', 'b', 'c']
})

B = pl.DataFrame({
    "index": [10, 20, 30],
    "B": ['d', 'e', 'f']
})

C = pl.DataFrame({
    "index": [20, 30],
    "C": ['g', 'h']
})

# Efficient join of multiple DataFrames
final_df = functools.reduce(
    lambda a, b: a.join(b, on='index', how='full', coalesce=True), [A, B, C]
).sort('index')

print(final_df)

# Test case for large DataFrame
N, C = 300000, 20
pls = []

for i in range(C):
    A = pl.DataFrame({
        "index": np.linspace(i, N * 3 - i, num=N, dtype=np.int32),
        f"A{i}": np.arange(N, dtype=np.float32),
    })
    pls.append(A)

%time F = functools.reduce(lambda a, b: a.join(b, on='index', how='full', coalesce=True), pls)
F.sort('index')
