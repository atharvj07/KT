import pydantic
from typing import Any
from random import randint
import string


class ExampleData(pydantic.BaseModel):
    a: int
    b: str = pydantic.Field(min_length=10, max_length=10)
    
    @staticmethod
    def example() -> dict[str, Any]:
        return {
            "a": randint(1, 100),
            "b": ''.join(randint(0, 9) for _ in range(10))
        }

# Test the example
example_data = ExampleData.example()
print(example_data)
