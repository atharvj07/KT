import os
import signal
import concurrent.futures
import multiprocessing

def _do_work(data):
    import os, signal
    # Ensure child exits if parent dies
    os.setpgrp()
    signal.signal(signal.SIGTERM, lambda signum, frame: exit())

    # Simulate work
    import time
    time.sleep(2)
    return data * 2

if __name__ == "__main__":
    n_workers = multiprocessing.cpu_count()
    data_list = list(range(10))

    with concurrent.futures.ProcessPoolExecutor(n_workers) as executor:
        result_list = [executor.submit(_do_work, data) for data in data_list]
        for id, future in enumerate(concurrent.futures.as_completed(result_list)):
            print(f'{id}: {future.result()}')
