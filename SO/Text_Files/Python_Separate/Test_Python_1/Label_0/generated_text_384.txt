import pandas as pd
import numpy as np

# Sample MultiIndex DataFrame
arrays = [
    ["A", "A", "B", "B", "C", "C"],
    ["Internet", "Other", "Internet", "Other", "Internet", "Other"],
]
index = pd.MultiIndex.from_tuples(list(zip(*arrays)), names=["Category", "Type"])
data = np.random.randint(1, 10, size=(5, 6))
end_df = pd.DataFrame(data, columns=index)

# Fix SettingWithCopyWarning and ensure correctness
filtered = end_df.loc[:, end_df.columns.get_level_values(1) == "Internet"].drop("site", axis=1, errors="ignore")
end_df = end_df.copy()  # Ensure no SettingWithCopyWarning
end_df["Soma Internet"] = filtered.sum(axis=1)

print(end_df)
