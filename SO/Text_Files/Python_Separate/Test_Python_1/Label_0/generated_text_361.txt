class Item_inference(BaseModel):
    text: str

@app.post("/retrieval")
async def Knowledge_recommendation(item: Item_inference, request: Request):
    model = request.app.state.model
    results = request.app.state.ai.ask_query(item.text, item.topN, model)
    return results
