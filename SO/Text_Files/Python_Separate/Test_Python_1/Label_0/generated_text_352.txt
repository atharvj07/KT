from pydantic import BaseModel
from typing import List, Optional

# Pydantic Base Model
class Model(BaseModel):
    a: int
    b: str
    c: List[str]
    d: Optional[float] = None  # Make 'd' optional

# Example response
response = {
    "a": 1,
    "b": "abc",
    "c": ["a", "b", "c"]
}

# Validating
model_instance = Model(**response)
print(model_instance)
