import asyncio
import rsa
from Crypto.PublicKey import RSA
from Crypto.Cipher import PKCS1_v1_5
from Crypto.Util.number import bytes_to_long

async def send_cmd_to_server(writer, cmd, args):
    data = f"{cmd}|{'|'.join(args)}".encode()
    writer.write(len(data).to_bytes(4, "big") + data)
    await writer.drain()

async def send_data_to_server(writer, data):
    writer.write(len(data).to_bytes(4, "big") + data)
    await writer.drain()

async def main(ip, port, ver, string1, string2):
    reader, writer = await asyncio.open_connection(ip, port)
    await send_cmd_to_server(writer, "auth", ver.split("."))

    modulus = await reader.read(256)
    exponent = await reader.read(3)

    pubkey = RSA.construct((bytes_to_long(modulus), bytes_to_long(exponent)))
    cipher = PKCS1_v1_5.new(pubkey)

    data = string1.encode() + string2.encode()
    encrypted_data = cipher.encrypt(data)

    await send_data_to_server(writer, encrypted_data)
    writer.close()
    await writer.wait_closed()

# Test Case
async def test_rsa_encryption():
    ip, port = "127.0.0.1", 12345
    ver = "1.0.0"
    string1, string2 = "test1", "test2"

    await main(ip, port, ver, string1, string2)

asyncio.run(test_rsa_encryption())
