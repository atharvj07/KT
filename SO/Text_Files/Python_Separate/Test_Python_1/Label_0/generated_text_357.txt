from pyspark.sql import SparkSession
from pyspark.sql import functions as F
from pyspark.sql.window import Window

# Initialize Spark session
spark = SparkSession.builder.master("local").appName("Window Example").getOrCreate()

# Sample data
data = [(1, "a", 10), (2, "b", 20), (3, "c", 30), (4, "d", 40)]
df = spark.createDataFrame(data, ["id", "name", "value"])

# Define a Window spec
window_spec = Window.orderBy("id")

# Example 1: Using Window.unboundedPreceding and Window.unboundedFollowing
df1 = df.withColumn("sum_all", F.sum("value").over(Window.unboundedPreceding))
df1.show()

# Example 2: Using Window.currentRow
df2 = df.withColumn("sum_current", F.sum("value").over(Window.rowsBetween(Window.unboundedPreceding, Window.currentRow)))
df2.show()

# Example 3: Using Window.rowsBetween with a range
df3 = df.withColumn("sum_range", F.sum("value").over(Window.rowsBetween(-1, 1)))  # sum of the previous, current, and next row
df3.show()
