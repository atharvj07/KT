from flask import Flask, render_template, jsonify
import map_plotting_test as mpt
import threading

app = Flask(__name__)

@app.route('/')
def render_the_map():
    # Run create_map in a separate thread to prevent blocking
    def generate_map():
        mpt.create_map()
    
    thread = threading.Thread(target=generate_map)
    thread.start()
    
    # Render the loading screen initially
    return render_template("loading.html")

@app.route('/map')
def map_ready():
    # Render the map after it's generated
    return render_template("map.html")

if __name__ == '__main__':
    app.run(debug=True)
