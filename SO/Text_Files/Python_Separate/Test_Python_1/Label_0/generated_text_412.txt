# Raspberry Pi Pico code (MicroPython)

import usb_cdc
import time

# Write data to USB serial
usb_cdc.data.write(b"Hello from Raspberry Pi Pico!")

# Read data from USB serial
while True:
    if usb_cdc.data.in_waiting:
        data = usb_cdc.data.read(64)
        print("Received:", data)
    time.sleep(1)

# PC-side code (Python)

import serial

# Open the serial connection
ser = serial.Serial('/dev/ttyACM0', 115200)  # Adjust the port if necessary

# Write data to Raspberry Pi Pico
ser.write(b"Hello from PC!")

# Read data from Raspberry Pi Pico
data = ser.readline()
print("Received from Pico:", data)

ser.close()
