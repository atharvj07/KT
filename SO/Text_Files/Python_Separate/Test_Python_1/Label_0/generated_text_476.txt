import hashlib
import sys

def get_hash(hash_type):
    filename = input("Enter the file name: ")
    hash_func = hashlib.sha256() if hash_type == "sha256" else hashlib.sha512()
    
    with open(filename, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            hash_func.update(byte_block)

    print(f"{hash_type} value has been calculated")
    return hash_func.hexdigest()

def verify_checksum(hash_type, expected_hash):
    calculated_hash = get_hash(hash_type)
    if calculated_hash == expected_hash:
        print("Checksum verified.")
    else:
        print("Checksum mismatch.")

def which_hash():
    sha_type = input("Which hash do you want to calculate: sha256 or sha512? \n").lower()
    if sha_type in ["sha256", "sha512"]:
        hash_value = get_hash(sha_type)
        print(f"{sha_type} hash: {hash_value}")
        
        verify = input("Do you want to verify a checksum? (yes/no): ").lower()
        if verify == "yes":
            expected_hash = input("Enter the expected hash: ")
            verify_checksum(sha_type, expected_hash)
    else:
        print("Invalid input. Program exiting.")
        sys.exit()

if __name__ == "__main__":
    which_hash()
