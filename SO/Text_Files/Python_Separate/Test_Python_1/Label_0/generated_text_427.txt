import polars as pl
import numpy as np
from numpy.linalg import norm

# Cosine similarity function
cosine_similarity = lambda a, b: (a @ b.T) / (norm(a) * norm(b))

# Sample dataframe
data = {
    "col1": ["a", "b", "c", "d"],
    "col2": [
        [-0.06066, 0.072485, 0.548874, 0.158507],
        [-0.536674, 0.10478, 0.926022, -0.083722],
        [-0.21311, -0.030623, 0.300583, 0.261814],
        [-0.308025, 0.006694, 0.176335, 0.533835],
    ]
}

# Create DataFrame
df = pl.DataFrame(data)

# Function to compute cosine similarities for each combination of col1
def compute_cosine_sim(df):
    result = []
    for i, row_i in df.iterrows():
        for j, row_j in df.iterrows():
            cos_sim = cosine_similarity(np.array(row_i["col2"]), np.array(row_j["col2"]))
            result.append({
                "col1_1": row_i["col1"], 
                "col1_2": row_j["col1"], 
                "cosine_similarity": cos_sim
            })
    return pl.DataFrame(result)

# Compute the cosine similarity
cosine_df = compute_cosine_sim(df)

# Show the result
print(cosine_df)
