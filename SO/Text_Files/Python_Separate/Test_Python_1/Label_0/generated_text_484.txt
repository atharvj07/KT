import numpy as np

v = np.array([0, 1, 4, 0, 5])
m = np.array([
    [True, True, False, False, False],
    [True, False, True, False, True],
    [True, True, True, True, True],
])

# Using np.ma (Masked Array)
masked_v = np.ma.array(v, mask=~m)
result = masked_v.mean(axis=1).filled(np.nan)

# Test case
assert np.allclose(result, [0.5, 3.0, 2.0])
print(result)
