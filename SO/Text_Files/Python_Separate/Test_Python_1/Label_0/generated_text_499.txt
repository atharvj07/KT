from pydantic import BaseModel, ValidationError, field_validator

class Salary(BaseModel):
    gross: int
    net: int
    tax: int

class Employee(BaseModel):
    name: str
    age: int
    salary: Salary

def validate_instance(model, data):
    errors = []
    try:
        model(**data)
    except ValidationError as e:
        errors.extend(e.errors())

    return errors

# Test Case
data = {
    "name": "Mattia",
    "age": "hello",
    "salary": {"gross": "hello", "net": 1000, "tax": 10}
}

errors = validate_instance(Employee, data)
print(errors)
