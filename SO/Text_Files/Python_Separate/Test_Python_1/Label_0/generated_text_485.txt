import numpy as np
import cv2
from PIL import Image

# Inputs
dpi = 1000
dpmm = 0.03937 * dpi
widthOfSampleMM = 50
heightOfSampleMM = 50
patternSizeMM = 0.1
density = 0.75
variation = 0.75

# Convert to pixels
widthOfSamplesPX = int(np.ceil(widthOfSampleMM * dpmm))
widthOfSamplesPX += 10 - widthOfSamplesPX % 10
heightOfSamplePX = int(np.ceil(heightOfSampleMM * dpmm))
heightOfSamplePX += 10 - heightOfSamplePX % 10
patternSizePX = patternSizeMM * dpmm

# Initialize empty image
im = 255 * np.ones((heightOfSamplePX, widthOfSamplesPX), dtype=np.uint8)

# Generate pattern
numPointsH = int(density * heightOfSamplePX / patternSizePX)
horizontal = [heightOfSamplePX // 2] if numPointsH == 1 else [int(i * heightOfSamplePX / (numPointsH + 1)) for i in range(1, numPointsH + 1)]
numPointsV = int(density * widthOfSamplesPX / patternSizePX)
vertical = [widthOfSamplesPX // 2] if numPointsV == 1 else [int(i * widthOfSamplesPX / (numPointsV + 1)) for i in range(1, numPointsV + 1)]

for i in vertical:
    for j in horizontal:
        iWithNoise = i + variation * np.random.randint(-2 * patternSizePX / density, 2 * patternSizePX / density)
        jWithNoise = j + variation * np.random.randint(-2 * patternSizePX / density, 2 * patternSizePX / density)
        patternSizePXWithNoise = patternSizePX + patternSizePX * variation * (np.random.rand() - 0.5) / 2
        cv2.circle(im, (int(iWithNoise), int(jWithNoise)), int(patternSizePXWithNoise // 2), 0, -1)

# Save as high-quality TIFF
image = Image.fromarray(im)
image.save("speckle_pattern.tiff", dpi=(dpi, dpi))

# Test case
assert image.size == (widthOfSamplesPX, heightOfSamplePX)
print("Pattern saved successfully.")
