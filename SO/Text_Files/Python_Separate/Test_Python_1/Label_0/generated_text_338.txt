# test_endpoint.py
import pytest
from fastapi.testclient import TestClient
from app import app, build_db
from pymongo import MongoClient

@pytest.fixture
def mock_db():
    def mock_close():
        return MongoClient()["mockDB"]
    return mock_close

def test_route_receive_notifications(mock_db):
    app.dependency_overrides[build_db] = mock_db
    
    with TestClient(app) as client:
        response = client.post("/notification/feed")
        assert response.status_code == 200
        assert response.json() == {"msg": "Notification received"}
