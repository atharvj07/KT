import numpy as np
import cython
from cython.parallel import prange
import time

# Create a large uint64 array
arr = np.random.randint(0, 2**64, int(3e8), dtype='uint64')

# Cython parallel sort using prange
def parallel_sort(arr):
    sorted_arr = np.copy(arr)
    sorted_arr.sort(kind='quicksort')
    return sorted_arr

# Measure time taken to sort using Cython
start = time.time()
sorted_arr = parallel_sort(arr)
end = time.time()

time_taken = end - start
time_taken
