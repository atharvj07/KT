import numpy as np
from sklearn.preprocessing import PolynomialFeatures
from sklearn.linear_model import LinearRegression

# Make or load an n x p data matrix X and n x 1 array y of the corresponding function values.
X = np.random.rand(10, 2)  # Example data matrix
y = np.random.rand(10, 1)  # Example target values

# Transform input data to include polynomial features
poly = PolynomialFeatures(degree=2)
Xp = poly.fit_transform(X)

# Fit linear model
model = LinearRegression()
model.fit(Xp, y)

# Gradient (first derivative)
def gradient(X, model, poly):
    Xp = poly.transform(X)
    return Xp @ model.coef_.T

# Hessian (second derivative)
def hessian(X, model, poly):
    Xp = poly.transform(X)
    return np.dot(Xp.T, Xp) * 2  # Hessian for polynomial regression

# Test
grad = gradient(X, model, poly)
hess = hessian(X, model, poly)

print("Gradient:", grad)
print("Hessian:", hess)
