import os
from fastapi import Security, HTTPException, Depends

API_KEY = "your-secret-key"
ENV = os.getenv("ENV", "prod")

async def api_key(api_key_header: str = Security(api_key_header_auth)):
    if ENV != "localhost" and api_key_header != API_KEY:
        raise HTTPException(
            status_code=401,
            detail="Invalid API Key",
        )

# Test case
import pytest
from fastapi.testclient import TestClient
from main import app

client = TestClient(app)

def test_api_key_auth_disabled():
    os.environ["ENV"] = "localhost"
    response = client.get("/v2/test/")
    assert response.status_code == 200

def test_api_key_auth_enabled():
    os.environ["ENV"] = "prod"
    response = client.get("/v2/test/", headers={"api_key_header": "wrong-key"})
    assert response.status_code == 401
