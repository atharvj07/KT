from typing import Optional
from pydantic import BaseSettings, Field


class SubSettings(BaseSettings):
    sub_field: Optional[str] = Field(None, env='SUB_FIELD')


class Settings(BaseSettings):
    field_one: Optional[str] = Field(None, env='FIELD_ONE')
    field_two: Optional[int] = Field(None, env='FIELD_TWO')
    sub_settings: SubSettings

    class Config:
        env_file = 'local.env'


settings = Settings()

print(settings.field_one)  # Should print the value of FIELD_ONE
print(settings.field_two)  # Should print the value of FIELD_TWO
print(settings.sub_settings.sub_field)  # Should print the value of SUB_FIELD
