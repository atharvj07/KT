import polars as pl

COL_VALUE = "value"

def compute(df):
    max_value = df[COL_VALUE].max()
    df = df.with_columns(
        (pl.col(COL_VALUE).cum_sum().shift(1).fill_none(0)).alias("sum_lower")
    )
    df = df.with_columns(
        (max_value - pl.col(COL_VALUE)).alias("max_other")
    )
    return df

if __name__ == "__main__":
    df = pl.DataFrame({COL_VALUE: [3, 7, 1, 9, 4]})
    df = compute(df)
    print(df)
