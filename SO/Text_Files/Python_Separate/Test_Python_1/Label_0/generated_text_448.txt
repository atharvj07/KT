dockerfile
FROM python:3.9

WORKDIR /project

COPY poetry.lock pyproject.toml /project/

# Install poetry only when poetry.lock or pyproject.toml changes
RUN pip install poetry

# Use a build argument for CI_JOB_TOKEN, but ignore it for cache
ARG CI_JOB_TOKEN
RUN echo "gitlab-ci-token:${CI_JOB_TOKEN}" > /tmp/token.txt && \
    poetry config http-basic.gitlab-pypi-repo $(cat /tmp/token.txt) && \
    rm /tmp/token.txt

# Install dependencies
RUN poetry install --no-interaction

# Add your project files
COPY . .

# Final step
CMD ["python", "app.py"]
