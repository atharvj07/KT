import polars as pl

df = pl.DataFrame({
    "MyDate": ["2020-01-01", "2021-06-15", "2019-12-31"]
}).with_columns(pl.col("MyDate").str.strptime(pl.Date, "%Y-%m-%d"))

date_str = "2020-01-01"
df_filtered = df.filter(pl.col("MyDate") >= pl.lit(date_str).str.strptime(pl.Date, "%Y-%m-%d"))

print(df_filtered)
