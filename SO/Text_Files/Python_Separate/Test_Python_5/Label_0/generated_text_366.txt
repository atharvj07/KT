import polars as pl

# Create a sample dataframe
df = pl.DataFrame(
    {
        "a": [1, 2, 3, 4],
        "b": [0.5, 4, 10, 13],
        "c": [True, True, False, True],
    }
)

# Create a copy of the dataframe using clone()
df_copy = df.clone()

# Print the original dataframe
print("Original DataFrame:")
print(df)

# Print the copied dataframe
print("\nCloned DataFrame:")
print(df_copy)
