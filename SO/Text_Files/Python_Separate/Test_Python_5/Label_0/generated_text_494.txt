import polars as pl

# Sample DataFrame
df = pl.DataFrame({
    "a": [1, 2, 3, 4, 5], 
    "b": [2, 3, 4, 5, 6], 
    "x": [1, 3, 5, 7, 9]
})

# Using pl.cut() within with_columns()
df = df.with_columns(
    pl.cut(pl.col("x"), bins=[0, 2, 4, 6, 8, 10]).alias("x_category")
)

print(df)
