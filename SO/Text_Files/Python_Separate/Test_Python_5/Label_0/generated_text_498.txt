import pandas as pd
import attr

@attr.s(auto_attribs=True)
class MyAttrsClass:
    i: int
    s: str

# Create a list of instances
arr = [MyAttrsClass("a", 1), MyAttrsClass("b", 2)]

# Convert to DataFrame
df = pd.DataFrame([attr.asdict(x) for x in arr])

print(df)


   i  s
0  a  1
1  b  2
