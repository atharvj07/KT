import polars as pl
import numpy as np

# Sample DataFrame
df = pl.DataFrame({
    'col1': [1, 2, 3, 4, 5],
    'col2': [10, 20, 30, 40, 50]
})

# Predictions
predictions = np.array([10, 20, 30, 40, 50])

# Adding the 'predictions' column using .with_columns()
df = df.with_columns(pl.from_numpy(predictions).alias('predictions'))

print(df)
