import re

s = "aaabbbbcca"
matches = re.findall(r'(.)\1*', s)

# Generate all possible substrings from the matched ones
substrings = []
for match in matches:
    for i in range(1, len(match) + 1):
        substrings.append(match[:i])

print(substrings)

python
['a', 'aa', 'aaa', 'b', 'bb', 'bbb', 'bbbb', 'c', 'cc', 'a']
