class Point:
 def __init__(self,x,y):
  self.x=x
  self.y=y
 def __iter__(self):
  yield self.x
  yield self.y
 def __getitem__(self,index):
  return (self.x,self.y)[index]
 def __repr__(self):
  return f"Point({self.x},{self.y})"

#Usage
p=Point(4,5)
my_x=p.x
my_tuple=tuple(p)
print(my_x)   # 4
print(my_tuple)   # (4,5)
