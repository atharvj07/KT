import polars as pl

# Create a lazy DataFrame
df = pl.LazyFrame({
    "col1": [1, None, 3, None, 5],
    "col2": [10, 20, 30, 40, 50]
})

# Fill null values in "col1" with the values from "col2" for the same row
df = df.with_columns(
    pl.col("col1").fill_none(pl.col("col2")).alias("col1")
)

# Collect the result
result = df.collect()

print(result)
