import polars as pl
import datetime as dt

# Create DataFrame with datetime column
df = pl.DataFrame({
    'time': [dt.datetime.now()]
})

# Use the `.dt.date()` method to extract the date directly
df = df.with_columns(
    pl.col("time").dt.date().alias("date")
)

# Show the result
print(df)
