import pandas as pd

# Sample data (single row dataframe)
data = {
    'Num': ['AA-24'],
    'TP1(USD)': [0],
    'TP2(USD)': [700],
    'TP3(USD)': [2100],
    'VReal1(USD)': [300],
    'VReal2(USD)': [1159],
    'VReal3(USD)': [2877],
    'TiV1 (EUR)': [30],
    'TiV2 (EUR)': [30],
    'TiV3 (EUR)': [47],
    'TR': [10],
    'TR-Tag': [5]
}

# Create the dataframe
df = pd.DataFrame(data)

# Extract relevant columns
TP_columns = df.filter(regex='TP').values[0]
VR_columns = df.filter(regex='VReal').values[0]
TV_columns = df.filter(regex='TiV').values[0]

# Create the output dataframe
result = pd.DataFrame({
    'ID': [1, 2, 3],
    'Price': TP_columns,
    'Net': VR_columns,
    'Range': TV_columns
})

# Display the result
print(result)
