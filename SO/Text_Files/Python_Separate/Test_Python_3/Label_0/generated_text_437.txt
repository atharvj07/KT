import matplotlib.pyplot as plt

def bar_chart_with_labels():
    data = [1000, 2000, 3000, 4000]
    labels = ['A', 'B', 'C', 'D']

    fig, ax = plt.subplots()
    bars = ax.bar(labels, data)

    for bar in bars:
        height = bar.get_height()
        bar.set_label(f'{height:,.2f}')  # Apply new style formatting
        ax.bar_label(bars, labels=[f'{height:,.2f}' for height in data])

    plt.show()

bar_chart_with_labels()
