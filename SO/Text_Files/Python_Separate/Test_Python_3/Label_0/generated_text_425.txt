r
install_tensorflow <- function() {
  # Install reticulate and keras
  devtools::install_github("rstudio/reticulate")
  devtools::install_github("rstudio/keras")
  
  # Load reticulate and configure conda environment
  library(reticulate)
  reticulate::install_miniconda()
  reticulate::use_miniconda("r-reticulate")
  
  # Create and activate conda environment
  system('conda create --name py38 python=3.8.0')
  system('conda activate py38')
  system('conda install tensorflow=2.4')
  
  # Use the environment in R
  reticulate::use_condaenv("/root/.conda/envs/py38", required = TRUE)
  reticulate::py_available(initialize = TRUE)
  
  # Install and configure TensorFlow in R
  library(tensorflow)
  tensorflow::tf_config()
}

# Run the function
install_tensorflow()
