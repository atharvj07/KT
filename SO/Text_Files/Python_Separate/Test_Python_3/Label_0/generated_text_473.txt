import pandas as pd

def find_overlapping_intervals(df):
    df['Start_tmp'] = pd.to_datetime(df['Start_tmp'])
    df['End_tmp'] = pd.to_datetime(df['End_tmp'])
    
    overlaps = []
    for date in df['date'].unique():
        day_df = df[df['date'] == date]
        for idx, row in day_df.iterrows():
            overlap = None
            for _, compare_row in day_df.iterrows():
                if row['Start_tmp'] < compare_row['End_tmp'] and row['End_tmp'] > compare_row['Start_tmp']:
                    overlap = max(row['Start_tmp'], compare_row['Start_tmp']), min(row['End_tmp'], compare_row['End_tmp'])
                    if overlap[0] < overlap[1]:
                        overlaps.append((overlap[1] - overlap[0]).seconds // 60)
    return overlaps
