from pyspark.sql import SparkSession
from pyspark.sql.window import Window
from pyspark.sql.functions import sum

def window_rows_between_example():
    spark = SparkSession.builder.appName("WindowExample").getOrCreate()
    
    data = [(1, "A", 10), (2, "A", 20), (3, "A", 30),
            (4, "B", 40), (5, "B", 50), (6, "B", 60)]
    
    df = spark.createDataFrame(data, ["id", "group", "value"])
    
    window_spec = Window.partitionBy("group").orderBy("id").rowsBetween(Window.unboundedPreceding, Window.currentRow)
    
    df = df.withColumn("cumulative_sum", sum("value").over(window_spec))
    
    df.show()

window_rows_between_example()
