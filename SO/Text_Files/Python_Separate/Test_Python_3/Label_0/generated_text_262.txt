import openai

# Upload a PDF file
file = openai.File.create(
  file=open("example.pdf", "rb"),
  purpose='answers'  # Use 'answers' for RAG
)

# Use the file ID in a query
response = openai.Completion.create(
  model="gpt-4",
  prompt="Use the PDF document to answer the following question: What is the capital of France?",
  file_ids=[file.id]
)
