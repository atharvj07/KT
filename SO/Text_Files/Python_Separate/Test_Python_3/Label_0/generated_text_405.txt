import pandas as pd

def count_three_way_conversations(df):
    df['sent_time'] = pd.to_datetime(df['sent_time'])
    
    result = []
    for group_id, group in df.groupby('group_id'):
        if group['from_red'].nunique() == 2:
            red_msgs = group[group['from_red']].sort_values('sent_time')
            green_msgs = group[~group['from_red']].sort_values('sent_time')
            
            for _, red_msg in red_msgs.iterrows():
                for _, green_msg in green_msgs.iterrows():
                    if red_msg['sent_time'] < green_msg['sent_time']:
                        third_msg = group[(group['sent_time'] > green_msg['sent_time']) & (group['from_red'])]
                        if not third_msg.empty:
                            third_msg_sorted = third_msg.sort_values('sent_time').iloc[0]
                            result.append([red_msg['user_id'], green_msg['user_id'], third_msg_sorted['user_id']])
    
    return pd.DataFrame(result, columns=['red', 'green', 'red_2'])
