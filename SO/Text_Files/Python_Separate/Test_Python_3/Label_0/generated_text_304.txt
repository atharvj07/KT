import numpy as np
import time
import matplotlib.pyplot as plt

def measure_execution_time(matrix_size=3000, num_iterations=50):
    execution_times1, execution_times2 = [], []

    for _ in range(num_iterations):
        matrix_a = np.random.rand(matrix_size, matrix_size)
        start = time.time()
        _ = np.matmul(matrix_a, matrix_a)
        execution_times1.append(time.time() - start)

    for _ in range(num_iterations):
        matrix_b = np.random.rand(matrix_size, matrix_size)
        start = time.time()
        _ = np.matmul(matrix_b, matrix_b.T)
        execution_times2.append(time.time() - start)

    plt.plot(range(num_iterations), execution_times1, label='A @ A')
    plt.plot(range(num_iterations), execution_times2, label='B @ B.T')
    plt.xlabel('Iteration')
    plt.ylabel('Execution Time')
    plt.title('Matrix Multiplication Execution Time Comparison')
    plt.legend()
    plt.show()

    return np.mean(execution_times1), np.mean(execution_times2)

measure_execution_time()
