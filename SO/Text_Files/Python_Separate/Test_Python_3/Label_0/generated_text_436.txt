# In your package's root directory, create a `setup.py` file for editable installation:
from setuptools import setup, find_packages

setup(
    name='your_package',
    version='0.1',
    packages=find_packages(),
    install_requires=[],
)

# Then run this command to install the package in "editable" mode:
# pip install -e .

# Now, you can freely import modules and sub-packages:
from your_package.submodule import your_function
