import pandas as pd

def unnest_route_set(df):
    route_set = df['Route_set'].iloc[0]
    routes = pd.DataFrame(route_set).T
    routes.columns = [f'route{i+1}' for i in range(routes.shape[1])]
    return routes

df = pd.DataFrame({'Generation': [0], 'Route_set': [[[20., 19., 47., 56.], [21., 34., 78., 34.]]]})
result = unnest_route_set(df)
print(result)
