import polars as pl

def reorder_columns(df):
    first_cols = ['col2', 'col1']
    remaining_cols = [col for col in df.columns if col not in first_cols]
    return df.select(first_cols + remaining_cols)

df = pl.DataFrame({'col1': [1, 2], 'col2': [3, 4], 'col3': [5, 6], 'col4': [7, 8], 'col5': [9, 10]})
df = reorder_columns(df)
