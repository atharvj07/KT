import polars as pl

def reshape_df(df):
    return df.select([
        pl.col("^a_.*").explode().alias("a"),
        pl.col("^b_.*").explode().alias("b")
    ])

# Example usage
df = pl.DataFrame({
    "a_0": [1, 2],
    "a_1": [3, 4],
    "a_2": [5, 6],
    "b_0": [7, 8],
    "b_1": [9, 10],
    "b_2": [11, 12]
})

reshaped_df = reshape_df(df)
print(reshaped_df)
