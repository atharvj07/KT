import numpy as np
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense

# Example data
x_data = np.random.rand(1000, 1209278)  # Replace with your actual data
y_data = np.random.rand(1000,)  # Replace with your actual labels

# Reshaping x_data to be 3D: (samples, timesteps, features)
x_data = x_data.reshape((x_data.shape[0], x_data.shape[1], 1))

# Model definition
model = Sequential()
model.add(LSTM(64, input_shape=(x_data.shape[1], 1)))
model.add(Dense(1))

model.compile(optimizer='adam', loss='mean_squared_error')

# Train the model
model.fit(x_data, y_data, epochs=10, batch_size=32)
