import pandas as pd

def custom_cumsum(df):
    last_values = df.loc[~df.duplicated("id", keep="last"), ["id", "value"]].set_index("id")["value"]
    return df["id"].map(last_values).fillna(0).cumsum()

id_list = ['a','b','c','a','b','e','f','a','b','k']
value_list = [12,14,3,13,16,7,4,6,10,18]
df = pd.DataFrame({'id': id_list, 'value': value_list})
df["cumsum_of_value"] = custom_cumsum(df)
print(df)
