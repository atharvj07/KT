import polars as pl

def calculate_ema_polars(values, span):
    df = pl.DataFrame({'value': values})
    df = df.with_columns(
        pl.col('value').ewm_mean(span=span, adjust=False).alias('ema_9')
    )
    return df

values = [143.15,143.1,143.06,143.01,143.03,143.09,143.14,143.18,143.2,143.2,143.2,143.31,143.38,143.35,143.34,143.25,143.33,143.3,143.33,143.36]
df = calculate_ema_polars(values, span=9)
print(df)
