import polars as pl

def group_by_sorted_names(df):
    return df.group_by("country").agg(
        sorted_names := pl.col("name").sort_by("age"),
        sorted_names.first().alias("age_sort_1"),
        sorted_names.get(2, default=None).alias("age_sort_2"),
        sorted_names.last().alias("age_sort_-1")
    )

df = pl.DataFrame(
    {"name": list("abcdef"), "age": [21, 31, 32, 53, 45, 26], "country": list("AABBBC")}
)

result = group_by_sorted_names(df)
