import pandas as pd
import numpy as np

def fill_na_with_pair(df):
    cols = [col for col in df.columns if not col.endswith('_a')]
    df[cols] = df[cols].fillna(df[[f"{col}_a" for col in cols]].values)
    return df

df = pd.DataFrame(
    {
        'x': [1, np.nan, 3, np.nan, 5],
        'y': [np.nan, 7, 8, 9, np.nan],
        'x_a': [1, 2, 3, 4, 5],
        'y_a': [6, 7, 8, 9, 10]
    }
)

print(fill_na_with_pair(df))
