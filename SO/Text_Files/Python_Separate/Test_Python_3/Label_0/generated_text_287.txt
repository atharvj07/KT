import sys
import os

def fix_vscode_import():
    python_path = sys.executable
    os.system(f'"{python_path}" -m pip install --upgrade langchain')
    os.system(f'"{python_path}" -m pip install --upgrade pip setuptools')
    os.environ["PYTHONPATH"] = os.path.dirname(python_path)
    print("Restart VS Code after running this function.")

fix_vscode_import()
