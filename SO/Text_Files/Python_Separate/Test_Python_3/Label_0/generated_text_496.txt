from Crypto.PublicKey import RSA
from Crypto.Cipher import PKCS1_v1_5
from Crypto.Util.Padding import pad

def encrypt_data(modulus, exponent, string1, string2):
    key = RSA.construct((int.from_bytes(modulus, 'big'), int.from_bytes(exponent, 'big')))
    cipher = PKCS1_v1_5.new(key)
    
    data = (string1 + string2).encode()
    encrypted_data = cipher.encrypt(pad(data, key.size_in_bytes()))
    
    return encrypted_data
