from tqdm import tqdm
from PIL import Image
import google.generativeai as genai

async def process_image_batch(model, prompt, image_paths):
    tasks = [asyncio.to_thread(process_image, model, prompt, path) for path in image_paths]
    return await asyncio.gather(*tasks)

def process_image(model, prompt, image_path):
