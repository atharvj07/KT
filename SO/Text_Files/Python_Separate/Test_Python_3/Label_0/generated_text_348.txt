from collections import deque

def partition_array(numbers, min_len, max_len):
    n = len(numbers)
    dp = [-1] * (n + 1)
    trace = [-1] * (n + 1)
    dp[0] = 0

    for i in range(min_len, n + 1):
        min_q, max_q = deque(), deque()
        for j in range(i, max(i - max_len, -1), -1):
            while min_q and numbers[min_q[-1]] >= numbers[j - 1]:
                min_q.pop()
            min_q.append(j - 1)
            while max_q and numbers[max_q[-1]] <= numbers[j - 1]:
                max_q.pop()
            max_q.append(j - 1)

            if j <= i - min_len + 1:
                while min_q[0] >= j + min_len - 1:
                    min_q.popleft()
                while max_q[0] >= j + min_len - 1:
                    max_q.popleft()
                diff = numbers[max_q[0]] - numbers[min_q[0]]
                if dp[j - 1] != -1 and dp[j - 1] + diff > dp[i]:
                    dp[i] = dp[j - 1] + diff
                    trace[i] = j - 1

    if dp[n] == -1:
        return []

    result = []
    idx = n
    while idx > 0:
        start = trace[idx]
        result.append(numbers[start:idx])
        idx = start

    return result[::-1]
