import pandas as pd

def aggregate_results(df):
    return df.assign(
        Low=(df["Value"] < -0.25).astype(int),
        Hit=((df["Value"] >= -0.25) & (df["Value"] <= 0.25)).astype(int),
        High=(df["Value"] > 0.25).astype(int)
    ).groupby("Type", as_index=False)[["Low", "Hit", "High"]].sum()

df = pd.DataFrame({"Type": ["RF", "RF", "RF", "MLP", "MLP", "MLP"], "Value": [-1.5, -0.1, 1.7, 0.2, -0.7, -0.6]})
result = aggregate_results(df)
