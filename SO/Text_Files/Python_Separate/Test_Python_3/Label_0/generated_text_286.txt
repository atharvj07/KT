import polars as pl
import numpy as np
import plotly.express as px

def plot_polars_bar(df: pl.DataFrame, x: str, y: str):
    fig = px.bar(df.to_dict(as_series=False), x=x, y=y)
    fig.show()

df = pl.DataFrame(
    {
        "nrs": [1, 2, 3, None, 5],
        "names": ["foo", "ham", "spam", "egg", None],
        "random": np.random.rand(5),
        "groups": ["A", "A", "B", "C", "B"],
    }
)

plot_polars_bar(df, x="names", y="random")
