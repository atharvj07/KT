from flask import Flask, render_template, jsonify
import map_plotting_test as mpt
import threading

app = Flask(__name__)

def create_map_in_background():
    mpt.create_map()

@app.route('/')
def render_the_map():
    # Start the map creation in a separate thread
    threading.Thread(target=create_map_in_background).start()
    return render_template("loading.html")

@app.route('/map_ready')
def map_ready():
    return render_template("map.html")

if __name__ == '__main__':
    app.run(debug=True)
