import polars as pl

def explode_with_nulls(df):
    max_len = df.select(pl.col("x").apply(lambda x: len(x) if x is not None else 0).max()).to_numpy()[0][0]
    df = df.with_columns([
        pl.col("x").apply(lambda x: x if x is not None else [None] * max_len).alias("x"),
        pl.col("y").apply(lambda y: y if y is not None else [None] * max_len).alias("y")
    ])
    return df.explode(["x", "y"])

# Example usage
df = pl.DataFrame({
    "x": [[1], [1, 2], [1, 2, 3]],
    "y": [[True], None, [True, False, True]]
})

result = explode_with_nulls(df)
print(result)
