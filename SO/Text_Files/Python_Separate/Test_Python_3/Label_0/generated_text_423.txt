from functools import partial
from inspect import signature

def bar(a, b):
    return a / b

def test_partial_behavior():
    f1 = partial(bar, 3)
    assert str(signature(f1)) == '(b)'
    assert f1(6) == 0.5

    try:
        f1(a=2, b=6)
    except TypeError as e:
        assert str(e) == "bar() got multiple values for argument 'a'"

    try:
        f1(c=2, b=6)
    except TypeError as e:
        assert str(e) == "bar() got an unexpected keyword argument 'c'"

    f2 = partial(bar, b=3)
    assert str(signature(f2)) == '(a, *, b=3)'
    assert f2(6) == 2.0
    assert f2(6, b=1) == 6.0

    f3 = partial(bar, a=3)
    assert str(signature(f3)) == '(*, a=3, b)'

if __name__ == "__main__":
    test_partial_behavior()
