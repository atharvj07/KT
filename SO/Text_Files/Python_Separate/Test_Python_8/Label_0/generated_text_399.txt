import polars as pl
import plotly.express as px

df = pl.DataFrame({
    "sub_id": [1, 2, 3],
    "engagement": ["one:one,two:two", "one:two,two:one", "one:one"],
    "total_duration": [123, 456, 789]
})

df = df.with_columns(pl.col("engagement").str.split(",")).explode("engagement")

df = df.with_columns(
    (pl.col("total_duration") / df.groupby("sub_id").count().select("engagement")).alias("adjusted_duration")
)

fig = px.bar(df, x="sub_id", y="adjusted_duration", color="engagement")
fig.show()
