import machine
import utime
import sys

usb = machine.UART(0, baudrate=115200)

def read_usb():
    if usb.any():
        return usb.read().decode('utf-8').strip()
    return None

def write_usb(data):
    usb.write(data + "\n")
    utime.sleep(0.1)

while True:
    msg = read_usb()
    if msg:
        write_usb("Received: " + msg)
