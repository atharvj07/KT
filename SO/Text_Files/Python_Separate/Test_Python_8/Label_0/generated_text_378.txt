from sklearn import datasets
import pandas pd

iris = pd.DataFrame(datasets.load_iris().data, columns=datasets.load_iris().feature_names)

def mutate_drop(df, new_col_name, expr):
    df[new_col_name] = expr(df)
    return df.drop(columns=[col for col in df.columns if col not in [new_col_name] + ['species']], errors='ignore')

iris = mutate_drop(iris, 'new_col', lambda x: x['sepal length (cm)'] + x['petal length (cm)'] * x['petal width (cm)'])
