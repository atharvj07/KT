install.packages("devtools")
devtools::install_github("rstudio/reticulate")
devtools::install_github("rstudio/keras")

library(reticulate)
reticulate::install_miniconda()
reticulate::use_miniconda("r-reticulate")

system("conda create --name r-tensor python=3.8 -y")
system("conda activate r-tensor && conda install tensorflow=2.4 keras -y")

reticulate::use_condaenv("r-tensor", required = TRUE)
reticulate::use_python("/root/.conda/envs/r-tensor/bin/python", required = TRUE)

Sys.setenv(LD_LIBRARY_PATH="/root/.conda/envs/r-tensor/lib")

library(tensorflow)
tensorflow::tf_config()
