import geopandas as gpd
import pooch

url = "https://naturalearth.s3.amazonaws.com/110m_cultural/ne_110m_admin_0_countries.zip"

path = pooch.retrieve(
    url,
    known_hash=None,
    processor=pooch.Unzip(),
)

shp_file = [f for f in path if f.endswith(".shp")][0]

world = gpd.read_file(shp_file)

print(world.head())
