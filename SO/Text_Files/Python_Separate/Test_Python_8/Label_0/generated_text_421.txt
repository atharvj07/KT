import click

@click.command()
@click.option('--value', default=1, show_default=True, help='a value.')
def hello(value):
    ctx = click.get_current_context()
    user_provided = ctx.get_parameter_source("value") == click.ParameterSource.COMMANDLINE
    print(f"Value: {value}, User provided: {user_provided}")

if __name__ == "__main__":
    hello()
