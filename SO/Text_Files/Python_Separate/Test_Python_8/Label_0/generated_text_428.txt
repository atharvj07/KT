import numpy as np

np.random.seed(7)
A = np.random.random(1000)

%timeit np.sum(A)
%timeit A.sum()

def check_sum_methods():
    arr = np.random.random(1000)
    
    sum_1 = np.sum(arr)
    sum_2 = arr.sum()
    
    assert np.isclose(sum_1, sum_2), "Sums should be identical"

check_sum_methods()

from numpy.core.multiarray import array as array_constructor

def custom_np_sum(arr):
    if not isinstance(arr, np.ndarray):
        arr = array_constructor(arr, dtype=float)
    
    return arr.sum(axis=None, dtype=None, out=None, keepdims=False, initial=0)

%timeit custom_np_sum(A)
