from math import factorial
from decimal import Decimal, getcontext
from joblib import Parallel, delayed
from tqdm import tqdm
import time


def calc(n_digits, pos, total):
    n = int(n_digits + 1 / 14.181647462725477)
    n = n if n >= 1 else 1

    getcontext().prec = n_digits + 1

    t = Decimal(0)
    pi = Decimal(0)
    deno = Decimal(0)

    progress = tqdm(total=n, position=pos, desc=f"Job {pos + 1} of {total}", leave=True)
    
    for k in range(n):
        t = ((-1) ** k) * (factorial(6 * k)) * (13591409 + 545140134 * k)
        deno = factorial(3 * k) * (factorial(k) ** 3) * (640320 ** (3 * k))
        pi += Decimal(t) / Decimal(deno)
        progress.update(1)
    
    progress.close()
    
    pi = pi * Decimal(12) / Decimal(640320 ** Decimal(1.5))
    pi = 1 / pi
    
    return pi


def parallel_with_joblib():
    n_cores = 3
    tasks = [1200, 1700, 900, 1400, 800, 600, 500]

    results = Parallel(n_jobs=n_cores, prefer="threads")(delayed(calc)(n, pos, len(tasks)) for pos, n in enumerate(tasks))


if __name__ == "__main__":
    parallel_with_joblib()
