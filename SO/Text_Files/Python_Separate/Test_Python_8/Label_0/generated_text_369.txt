import numpy as np
import concurrent.futures

arr = np.random.randint(0, 2**64, int(3e8), dtype=np.uint64)

def parallel_sort(arr, num_parts=8):
    chunk_size = len(arr) // num_parts
    chunks = [arr[i * chunk_size: (i + 1) * chunk_size] for i in range(num_parts)]

    with concurrent.futures.ProcessPoolExecutor() as executor:
        sorted_chunks = list(executor.map(np.sort, chunks))

    return np.concatenate(sorted_chunks)

sorted_arr = parallel_sort(arr)
