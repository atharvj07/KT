import polars as pl

df = pl.DataFrame({
    "x": [[1], [1, 2], [1, 2, 3]],
    "y": [[True], None, [True, False, True]]
})

df = df.with_columns(
    pl.when(df["y"].is_null())
    .then(df["x"].list.len().apply(lambda l: [None] * l))
    .otherwise(df["y"])
    .alias("y")
)

df.explode(["x", "y"])
