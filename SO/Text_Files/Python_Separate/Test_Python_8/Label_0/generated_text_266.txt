import polars as pl

df = pl.DataFrame({
    "id": [1, 2], 
    "event_date": ["27 July 2020", "31 December 2020"]
})

df = df.with_columns(
    pl.col("event_date").str.strptime(pl.Date, "%d %B %Y").alias("event_date")
)

df = df.with_columns(
    (pl.col("event_date").dt.month // 3 + 1).alias("quarter")
)
