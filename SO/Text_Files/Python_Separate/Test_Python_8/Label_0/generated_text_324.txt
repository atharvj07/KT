import pandas as pd
import numpy as np

def glimpse(df):
    print(f"Rows: {df.shape[0]}")
    print(f"Columns: {df.shape[1]}\n")
    for col in df.columns:
        dtype = df[col].dtype
        sample_values = ", ".join(map(str, df[col].iloc[:5].values))
        print(f"{col} <{dtype}> {sample_values}")

test = pd.DataFrame({'column_one':['A', 'B', 'C', 'D'],
                     'column_two':[1, 2, 3, 4],
                     'column_three': np.random.randn(4)})

glimpse(test)
