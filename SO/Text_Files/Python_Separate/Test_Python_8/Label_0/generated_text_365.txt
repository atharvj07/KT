import polars as pl

def read_and_process_json(path):
    df = pl.read_json(path)
    df = df.with_columns(
        pl.col("date").str.to_date(),
        pl.col("fillingDate").str.to_date(),
        pl.col("acceptedDate").str.to_date()
    )
    return df

quarterly_balance_df1 = read_and_process_json('../AAPL/single_statements/1985-09-30-quarterly_balance.json')
quarterly_balance_df2 = read_and_process_json('../AAPL/single_statements/1986-09-30-quarterly_balance.json')

quarterly_balance_df1 = quarterly_balance_df1.sort("date")
quarterly_balance_df2 = quarterly_balance_df2.sort("date")

test = pl.concat([quarterly_balance_df1, quarterly_balance_df2])
