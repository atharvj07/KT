import pandas as pd 

def count_three_way_conversations(df):
    results = []
    
    grouped = df.groupby("group_id")
    
    for group_id, group in grouped:
        group = group.sort_values("sent_time").reset_index(drop=True)
        
        for i in range(len(group) - 2):
            if (group.loc[i, "from_red"] and 
                not group.loc[i+1, "from_red"] and 
                group.loc[i+2, "from_red"] and
                group.loc[i, "user_id"] == group.loc[i+2, "user_id"]):
                results.append(group_id)

    return len(set(results))
