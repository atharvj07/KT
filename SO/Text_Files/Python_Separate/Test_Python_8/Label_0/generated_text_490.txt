import matplotlib
import matplotlib.pyplot as plt
import numpy as np

# Setting the font
font = {'family' : 'serif',
        'weight' : 'bold',
        'size'   : 12,
        'serif': 'cmr10'
        }

matplotlib.rc('font', **font)

# Data for plotting
t = np.arange(0.0, 2.0, 0.01)
s = 1 + np.sin(2 * np.pi * t)

fig, ax = plt.subplots(1, figsize=(9, 6))

ax.plot(t, s)

# Set labels and title
ax.set(xlabel=r'$\textbf{time (s)}$', ylabel=r'$\textbf{voltage (mV)}$',
       title='About as simple as it gets, folks')

# Set bold ticks
ax.tick_params(axis='both', which='both', labelsize=12, labelweight='bold')

ax.grid()

# Save and show the plot
fig.savefig("test.png")
plt.show()
