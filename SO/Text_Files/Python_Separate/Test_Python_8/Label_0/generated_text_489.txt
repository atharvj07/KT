from pydantic import root_validator
from typing import Callable

def root_unique_validator(field_name: str) -> Callable:
    def _unique_validator(cls, values):
        root_values = values.get('__root__')
        value_set = set()
        for value in root_values:
            if value.get(field_name) in value_set:
                raise ValueError(f'Duplicate {field_name.capitalize()}')
            else:
                value_set.add(value.get(field_name))
        return values
    return _unique_validator
