from abc import ABC, abstractmethod

class FooBase(ABC):
    foo: str
    bar: str
    baz: int

    def __init__(self):
        self.bar = "bar"
        self.baz = 42

    @abstractmethod
    def __init_subclass__(cls):
        if not hasattr(cls, 'foo'):
            raise TypeError(f"Class {cls.__name__} must define attribute 'foo'")

class Foo(FooBase):
    foo: str = "hello"


f = Foo()
print(f.foo, f.bar, f.baz)
