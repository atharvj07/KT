import polars as pl
from datetime import datetime

data = {
    "ID": [1, 2, 3, 4, 5],
    "Initial Date": ["2022-01-01", "2022-01-02", "2022-01-03", "2022-01-04", "2022-01-05"],
    "Final Date": ["2022-01-03", "2022-01-06", "2022-01-07", "2022-01-09", "2022-01-07"],
    "Value": [10, 20, 30, 40, 50]
}

df = pl.DataFrame(data).with_columns([
    pl.col("Initial Date").str.strptime(pl.Date, "%Y-%m-%d"),
    pl.col("Final Date").str.strptime(pl.Date, "%Y-%m-%d")
])

dates = pl.DataFrame({
    "date": pl.date_range(datetime(2022,1,1), datetime(2022,1,7), interval="1d")
})

result = dates.with_columns(
    pl.sum(
        pl.when(
            (df["Initial Date"] <= pl.col("date")) & (df["Final Date"] >= pl.col("date"))
        ).then(df["Value"]).otherwise(0)
    ).alias("Summed Value")
)

print(result)
