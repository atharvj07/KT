from collections import deque

def partition_array(numbers, min_len, max_len):
    n = len(numbers)
    dp = [-1] * (n + 1)
    path = [-1] * (n + 1)

    dp[0] = 0

    for i in range(n):
        if dp[i] == -1:
            continue

        min_q, max_q = deque(), deque()

        for j in range(i, min(i + max_len, n)):
            while min_q and numbers[min_q[-1]] > numbers[j]:
                min_q.pop()
            min_q.append(j)

            while max_q and numbers[max_q[-1]] < numbers[j]:
                max_q.pop()
            max_q.append(j)

            if j - i + 1 >= min_len:
                diff = numbers[max_q[0]] - numbers[min_q[0]]
                if dp[j + 1] < dp[i] + diff:
                    dp[j + 1] = dp[i] + diff
                    path[j + 1] = i

    if dp[n] == -1:
        return []

    result = []
    end = n
    while end > 0:
        start = path[end]
        result.append(numbers[start:end])
        end = start

    return result[::-1]

numbers = [5, 8, 4, 5, 1, 3, 5, 1, 3, 1]
a, b = 3, 7
result = partition_array(numbers, a, b)
print(result, sum(max(sub) - min(sub) for sub in result))
