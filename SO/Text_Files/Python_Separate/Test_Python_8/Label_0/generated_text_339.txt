import pandas as pd

data = {
    "A": ["a", "a", "a", "c", "c", "c", "c", "c", "e", "e", "e", "e"],
    "B": ["b", "b", "b", "d", "d", "d", "d", "d", "b", "b", "b", "b"],
    "C": [1, 2, 3, 7, 8, 5, 6, 3, 4, 3, 2, 1]
}

df = pd.DataFrame(data)

df_pivot = df.groupby(["A", "B"])["C"].apply(lambda x: x.tolist()).reset_index()
df_pivot = pd.DataFrame(df_pivot["C"].tolist(), index=df_pivot.set_index(["A", "B"]).index)

df_pivot.columns = [f"C{i+1}" for i in range(df_pivot.shape[1])]
df_pivot.reset_index(inplace=True)
