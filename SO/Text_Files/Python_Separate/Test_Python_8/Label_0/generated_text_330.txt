import numpy as np
from numpy.lib.stride_tricks import sliding_window_view

def largest_neighbor_direction(a):
    L = a.shape[0]
    swv = sliding_window_view(a, (3, 3))  
    swv[:, :, 1, 1] = -np.inf
    directions = swv.reshape(L-2, L-2, 9).argmax(axis=2).astype(np.uint8)
    return directions

a = np.random.rand(10000, 10000).astype(np.float32)
directions = largest_neighbor_direction(a)
