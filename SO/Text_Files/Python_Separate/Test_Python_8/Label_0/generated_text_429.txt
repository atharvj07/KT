from typing import Union, Literal
from pydantic import BaseModel, Field
from fastapi import FastAPI

app = FastAPI()

class Foo(BaseModel):
    type: Literal["Foo"] = Field("Foo", alias="_type", exclude=True)
    Name: str

class Bar(BaseModel):
    type: Literal["Bar"] = Field("Bar", alias="_type", exclude=True)
    Name: str

class Demo(BaseModel):
    example: Union[Foo, Bar] = Field(..., discriminator="type")

@app.post("/demo")
async def demo(foo: Foo):
    demo_instance = Demo(example=foo)
    return demo_instance
