import time
from itertools import repeat

def function():
    n = 10_000_000
    reps = repeat(1, n)
    now = time.time
    first = now()
    for _ in reps:
        second = now()
        assert first <= second, f"{first=} {second=} i={n - sum(reps)}"
        first = second
function()
