from flask import Flask, render_template
import threading
import map_plotting_test as mpt

app = Flask(__name__)

def create_map_async():
    mpt.create_map()

@app.route('/')
def render_map():
    threading.Thread(target=create_map_async).start()
    return render_template("loading.html")

if __name__ == '__main__':
    app.run(debug=True)
