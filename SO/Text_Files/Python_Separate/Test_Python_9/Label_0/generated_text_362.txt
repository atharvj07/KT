import asyncio
import httpx

async def req(cli, url):
    r = await cli.get(url)
    return r.text

async def run(urls):
    async with httpx.AsyncClient() as cli:
        res = await asyncio.gather(*(req(cli, u) for u in urls))
    return res

urls = ["https://ex.com", "https://ex.org"]
out = asyncio.run(run(urls))
print(out)
