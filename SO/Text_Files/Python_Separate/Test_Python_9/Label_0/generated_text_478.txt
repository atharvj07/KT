import polars as pl
import numpy as np

df_sim = pl.DataFrame({
   "daily_n": [1000, 2000, 3000, 4000],
   "prob": [.5, .5, .5, .6],
   "size": 1
})

df_sim = df_sim.with_columns(
    pl.struct(["daily_n", "prob", "size"])
    .map_elements(lambda x: np.random.binomial(n=x['daily_n'], p=x['prob'], size=x['size']))
    .cast(pl.Int32)
    .alias('events')
)

df_sim


df_sim2 = df_sim.with_columns(
    np.random.binomial(n=pl.col('daily_n'), p=pl.col('prob'), size=pl.col('size'))
    .alias('events')
)

df_sim2
