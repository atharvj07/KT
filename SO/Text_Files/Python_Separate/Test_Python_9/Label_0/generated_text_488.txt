import pytest
from unittest.mock import MagicMock
from db import PostgresqlApi


@pytest.fixture
def mock_db():
    db = MagicMock(PostgresqlApi)
    db.cursor.return_value = MagicMock()
    return db


def test_db_connection(mock_db):
    mock_db.add_event(event="Test Event")
    mock_db.events.return_value = [("Test Event",)]
    assert mock_db.events() == [("Test Event",)]


def test_add_event(mock_db):
    mock_db.add_event(event="Test Event")
    mock_db.cursor.execute.assert_called_with(
        "INSERT INTO history (event) VALUES (%s);", ("Test Event",)
    )


def test_read_currencies(mock_db):
    mock_db.events.return_value = [("Test Event",)]
    result = {'get events': 'ok'}
    assert result == {'get events': 'ok'}
