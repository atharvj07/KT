from pathlib import Path
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM

def load_mod(path):
    return AutoModelForCausalLM.from_pretrained(path, torch_dtype=torch.float32)

def load_tok(path):
    return AutoTokenizer.from_pretrained(path)

def infer(mod, tok, prompt):
    inputs = tok(prompt, return_tensors="pt")
    outputs = mod.generate(inputs['input_ids'])
    return tok.decode(outputs[0])

mod = load_mod(Path('path/to/gguf/model'))
tok = load_tok(Path('path/to/gguf/tokenizer'))
res = infer(mod, tok, "Hello, world!")
print(res)
