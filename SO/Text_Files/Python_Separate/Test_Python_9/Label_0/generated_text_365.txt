import polars as pl

def rd_json(pth: str):
    df = pl.read_json(pth)
    return df.with_columns(
        pl.col("date").str.to_date(),
        pl.col("fillingDate").str.to_date(),
        pl.col("acceptedDate").str.to_date()
    )

df1 = rd_json('../AAPL/single_statements/1985-09-30-quarterly_balance.json')
df2 = rd_json('../AAPL/single_statements/1986-09-30-quarterly_balance.json')

df = pl.concat([df1, df2]).sort("date")
