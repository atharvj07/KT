import pandas as pd

data1 = {
    'score': [1, 2],
    'seconds': [1140, 2100],
}

data2 = {
    'prize': [5.5, 14.5, 14.6, 21, 23, 24, 26, 38, 39, 40, 50],
    'seconds': [840, 1080, 1380, 1620, 1650, 1680, 1700, 1740, 2040, 2100, 2160],
}

df1 = pd.DataFrame.from_dict(data1)
df2 = pd.DataFrame.from_dict(data2)

def closest_match(s, df):
    idx = (df['seconds'] - s).abs().argsort()[:3]
    return df.iloc[idx]

result = pd.concat([closest_match(s, df2) for s in df1['seconds']]).reset_index(drop=True)
print(result)
