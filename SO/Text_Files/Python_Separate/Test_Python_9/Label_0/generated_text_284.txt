from dataclasses import dataclass
from typing import Dict, Callable

def mydec(cls):
    setattr(cls, 'save', lambda self: print('Saved'))
    setattr(cls, 'my_dict', {'a': 'str'})
    return cls

@mydec
@dataclass
class A:
    a: str = ""

def main():
    a = A()
    a.save()
    print(a.my_dict)

if __name__ == "__main__":
    main()
