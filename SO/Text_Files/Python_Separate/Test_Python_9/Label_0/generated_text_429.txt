from typing import Union, Literal
from fastapi import FastAPI
from pydantic import BaseModel, Field

app = FastAPI()

class Foo(BaseModel):
    typ: Literal["Foo"] = Field("Foo", alias="type")
    nam: str

    class Config:
        allow_population_by_field_name = True

class Bar(BaseModel):
    typ: Literal["Bar"] = Field("Bar", alias="type")
    nam: str

    class Config:
        allow_population_by_field_name = True

class Demo(BaseModel):
    exm: Union[Foo, Bar] = Field(..., discriminator="typ")

@app.post("/demo")
async def demo(foo: Foo):
    return Demo(exm=foo)
