import pandas as pd

inp = pd.DataFrame({'month':[1,1,1,1,1,2,2,2], 
                    'item':[10,10,10,10,20,20,20,20], 
                    'star':[1,2,1,3,3,2,2,3]})

df = inp.groupby(['month', 'item', 'star']).size().unstack(fill_value=0).reset_index()
df.columns = ['month', 'item', 'star_1_cnt', 'star_2_cnt', 'star_3_cnt']
print(df)
