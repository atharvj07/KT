import datetime as dt
from zoneinfo import ZoneInfo

def fix_add(d, h):
    return (d.astimezone(dt.timezone.utc) + dt.timedelta(hours=h)).astimezone(d.tzinfo)

z = ZoneInfo("Europe/Paris")
d0 = dt.datetime(2020, 3, 29, 0, 0, tzinfo=z)

d1 = fix_add(d0, 2)
d2 = fix_add(d0, 3)
d3 = fix_add(d0, 4)

print(d1.isoformat())
print(d2.isoformat())
print(d3.isoformat())
