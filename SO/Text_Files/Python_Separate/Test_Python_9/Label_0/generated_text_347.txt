import polars as pl

COL_VALUE = "value"

df = pl.DataFrame({COL_VALUE: [3, 7, 1, 9, 4]})

sum_lower = (
    df.with_columns(
        pl.col(COL_VALUE).rank(reverse=True, method="average")
        .alias("rank")
    )
    .join(df, on="rank", how="left")
    .with_columns(
        pl.col(COL_VALUE).apply(lambda x: sum(df[COL_VALUE] < x)).alias("sum_lower")
    )
)

max_other = df.with_columns(
    pl.col(COL_VALUE).apply(lambda x: max(df[COL_VALUE] - x)).alias("max_other")
)

df = df.join(sum_lower, on=COL_VALUE).join(max_other, on=COL_VALUE)

print(df)
