import torch
from transformers import AutoTokenizer, AutoModelForCausalLM

pth = "Meta-Llama-3.1-8B-Instruct"

tok = AutoTokenizer.from_pretrained(pth)
mdl = AutoModelForCausalLM.from_pretrained(pth, torch_dtype=torch.float16, device_map="auto")

txt = "Hello, how are you?"
inp = tok(txt, return_tensors="pt").to("cuda")
out = mdl.generate(**inp, max_length=50)

print(tok.decode(out[0], skip_special_tokens=True))
