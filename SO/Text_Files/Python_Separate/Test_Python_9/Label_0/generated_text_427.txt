import polars as pl
import numpy as np

dat = {
    "c1": ["a", "b", "c", "d"],
    "c2": [[-0.06066, 0.072485, 0.548874, 0.158507],
           [-0.536674, 0.10478, 0.926022, -0.083722],
           [-0.21311, -0.030623, 0.300583, 0.261814],
           [-0.308025, 0.006694, 0.176335, 0.533835]],
}

df = pl.DataFrame(dat)

def sim(x, y):
    return (x @ y.T) / (np.linalg.norm(x) * np.linalg.norm(y))

mat = np.array(df["c2"].to_list())

out = pl.DataFrame(
    [[df["c1"][i]] + [sim(mat[i], mat[j]) for j in range(len(mat))] for i in range(len(mat))],
    schema=["c1"] + df["c1"].to_list()
)

print(out)
