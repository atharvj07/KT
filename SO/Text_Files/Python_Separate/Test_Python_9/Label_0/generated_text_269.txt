import cv2 as cv
import numpy as np

src = cv.imread("boston_cooking_a.jpg", cv.IMREAD_GRAYSCALE)
thresh = cv.adaptiveThreshold(src, 255, cv.ADAPTIVE_THRESH_MEAN_C, cv.THRESH_BINARY, 55, 11)
dil = cv.dilate(thresh, cv.getStructuringElement(cv.MORPH_RECT, (3, 3)))
ero = cv.erode(dil, cv.getStructuringElement(cv.MORPH_RECT, (50, 3)))
grad = cv.Sobel(ero, cv.CV_8U, 0, 1, ksize=5)
lines = cv.HoughLinesP(grad, 1, np.pi / 180, threshold=100, minLineLength=200, maxLineGap=10)

for line in lines:
    x1, y1, x2, y2 = line[0]
    cv.line(src, (x1, y1), (x2, y2), (0, 0, 255), 2)

cv.imwrite("output.jpg", src)
cv.imshow("output", src)
cv.waitKey(0)
