# Faster equivalent of `np.random.choice(np.arange(len(p)), p=p)`
@nb.njit('(float64[::1],)')
def randomInt(p):
    assert p.size &gt; 0
    if p.size == 1:
        return 0
    tmp = np.empty(p.size, np.float64)
    # Cumulative sum
    s = 0.0
    for i in range(p.size):
        s += p[i]
        tmp[i] = s
    # Renormalization (to increase the accuracy)
    rnd = np.random.rand() * s
    idx = np.searchsorted(tmp, rnd)
    return idx
