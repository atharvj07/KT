import asyncio
import time

event = asyncio.Event()

async def light_job():
    print(&quot;hello &quot;)
    print(time.time())
    await asyncio.sleep(1)
    print(time.time())
    print(&quot;world!&quot;)
    event.set()


async def heavy_job():
    print(&quot;heavy start&quot;)
    time.sleep(3)
    print(&quot;heavy halt started&quot;)
    # await asyncio.sleep(0)
    await event.wait()
    print(&quot;heavy halt ended&quot;)
    time.sleep(3)
    print(&quot;heavy done&quot;)
    
async def main():
    await asyncio.gather(light_job(), heavy_job())

asyncio.run(main())
