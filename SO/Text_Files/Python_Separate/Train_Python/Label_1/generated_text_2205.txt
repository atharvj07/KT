from typing import Generic, TypeVar

T = TypeVar('T', int, float)

class Vector3(Generic[T]):
    def __init__(self, x: T, y: T, z: T) -&gt; None:
        self._x: T = x
        self._y: T = y
        self._z: T = z

    def __key(self) -&gt; tuple[T, T, T]:
        return (self._x, self._y, self._z)

    @property
    def x(self) -&gt; T:
        return self._x
