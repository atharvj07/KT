        entries.append([time.time_ns(), json.dumps(entry)])

        'streams': [{
            "stream": {
                "application": alb_log_data[0]['application'],
                "job": "aws-lambda",
                "level": "info",
                "namespace": "oauth"
            },
            "values": entries
        }]
    }
    payload = json.dumps(payload)
    logger.debug('Created Payload %s', payload)
    return payload
