# create a reference from target_list
ref = (pd.Series(target_list, name='Ko_EC')
         .str.split(r';\s*')
         .explode()
         .reset_index()
       )

# merge the exploded &quot;alls&quot; then aggregate
out = (
  ref.merge(alls.assign(Ko_EC=alls['Ko_EC'].str.split()).explode('Ko_EC'), how='left')
     .groupby('index')
     .agg({'Ko_EC': '; '.join,
           'FPKM': 'sum',
           'count': 'sum'})
 )
