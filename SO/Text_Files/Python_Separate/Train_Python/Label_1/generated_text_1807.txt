    id: int = Field(primary_key=True)

class SubAccount(SQLModel, table=True):
    __table_args__ = (
        UniqueConstraint(&quot;name&quot;, &quot;user_id&quot;, name=&quot;your_unique_constraint_name&quot;),
    )
    id: int = Field(primary_key=True)
    name: str
    description: str
    user_id: int = Field(foreign_key=&quot;user.id&quot;)
