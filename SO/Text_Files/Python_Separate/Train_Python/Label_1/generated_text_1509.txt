def get_test_output():
    filepath = './cypress/integration/myproject/myscript.js'

    res = subprocess.run(
        ['echo', filepath],
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
    )
    # In your case it will be:
    # res = subprocess.run(
    #     ['cypress', 'run', '--spec', filepath],
    #     stdout=subprocess.PIPE,
    #     stderr=subprocess.STDOUT,
    # )

    return res.stdout.decode()


if __name__ == '__main__':
