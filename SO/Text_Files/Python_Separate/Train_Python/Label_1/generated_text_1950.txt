import xgboost
import shap 

X, y = shap.datasets.boston()
bst = xgboost.train({&quot;learning_rate&quot;: 0.01}, xgboost.DMatrix(X, label=y), 100)

explainer = shap.TreeExplainer(bst)
shap_values = explainer.shap_values(X)

shap.force_plot(
    explainer.expected_value, 
    shap_values[0,:], 
    X.iloc[0,:], 
    matplotlib=True # &lt;--
)
