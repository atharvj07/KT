def cdr_out(dictionary_1,x):
    count=0
    for i in range(len(x)-5):
        rider=x[i:i+6]
        # print(rider)
        if rider[2]=='R':
            # print(rider)
            for i in range(len(dictionary_1['cdr1_out'])):
                if rider[i]!=dictionary_1['cdr1_out'][i]:
                    count+=1
            if count&lt;dictionary_1['max_pin_out']:
                # print(rider)
                return x.index(rider)
