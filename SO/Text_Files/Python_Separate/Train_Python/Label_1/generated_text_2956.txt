import plotly.graph_objs as go
fig = go.Figure()
trace_color = "#636EFA" ## default plotly blue
fig.add_trace(go.Scatter(
    x = [0, 1, 2, 3], y = [1, 2, 3, 4],
    mode = 'lines+markers',
    name = "Trace 0",
    marker = dict(color = trace_color),
    showlegend = False,
    legendgroup = "Trace",
))
fig.add_trace(go.Scatter(
    x = [5,6,7,8], y = [1, 2, 3, 4],
    mode = 'lines+markers',
    name = "Trace",
    marker = dict(color = trace_color),
    legendgroup = "Trace",
))
fig.show()
