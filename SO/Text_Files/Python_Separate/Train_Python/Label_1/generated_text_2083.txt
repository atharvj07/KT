import polars as pl
from polars import col

df = pl.DataFrame({'a': [1, 2, 3]}) # .lazy()

df = df.with_columns([
    col('a'),
    pl.Series('empty lists', [[]], dtype=pl.List),
    pl.lit(None).alias('null column'),
])
print(df) # print(df.collect()) (in case of LazyFrame)
