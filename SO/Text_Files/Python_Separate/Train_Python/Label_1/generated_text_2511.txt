    y = s[s[-1]]
    y ^= (y >> 11);
    y ^= (y << 7) & 0x9d2c5680;
    y ^= (y << 15) & 0xefc60000;
    y ^= (y >> 18);
    print('predict:', y)
    print('actual: ', random.getrandbits(32))
    print()
