from random import shuffle
from heapq import merge
from itertools import pairwise

# Create test data
a = list(range(1000))
shuffle(a)

def mysort_steps(a):
    steps = []
    while any(x &gt; y for x, y in pairwise(a)):
        it = iter(a)
        a = list(merge(it, it))
        steps.append(a[:])
    return steps

def bubblesort_steps(a):
    steps = []
    n = len(a)
    for i in range(n - 1):
        swapped = False
        for j in range(n - 1 - i):
            if a[j] &gt; a[j+1]:
                a[j], a[j+1] = a[j+1], a[j]
                swapped = True
        if not swapped:
            break
        steps.append(a[:])
    return steps

print(mysort_steps(a[:]) == bubblesort_steps(a[:]))
