def QR(M):
    Q=np.zeros_like(M)
    R=np.zeros_like(M)
    for i in range(len(M)):
        ei=M[:,i].copy()
        for j in range(i):
            R[j,i] = Q[:,j]@M[:,i]
            ei -= R[j,i]/(Q[:,j]@Q[:,j]) * Q[:,j]
        R[i,i] = np.sqrt(ei@ei)
        Q[:,i] = ei / R[i,i]
    return Q,R
