values = [
    1,
    &quot;hello&quot;,
    True,
]

for v in values:
    match v:
        case str():
            print(&quot;It is a string!&quot;)
        case bool():
            print(&quot;It is a boolean!&quot;)
        case int():
            print(&quot;It is an integer!&quot;)
        case _:
            print(f&quot;It is a {type(v)}!&quot;)
