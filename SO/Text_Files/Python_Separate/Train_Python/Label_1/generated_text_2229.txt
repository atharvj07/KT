import itertools

data = [
    {&quot;ID&quot;: &quot;A&quot;, &quot;RelationIDs&quot;: [&quot;B&quot;, &quot;C&quot;, &quot;D&quot;,&quot;F&quot;, &quot;E&quot;]},
    {&quot;ID&quot;: &quot;B&quot;, &quot;RelationIDs&quot;: [&quot;A&quot;, &quot;E&quot;]},
    {&quot;ID&quot;: &quot;C&quot;, &quot;RelationIDs&quot;: [&quot;D&quot;]},
    {&quot;ID&quot;: &quot;D&quot;, &quot;RelationIDs&quot;: [&quot;C&quot;]},
    {&quot;ID&quot;: &quot;E&quot;, &quot;RelationIDs&quot;: [&quot;A&quot;, &quot;B&quot;]},
    {&quot;ID&quot;: &quot;F&quot;, &quot;RelationIDs&quot;: [&quot;A&quot;, &quot;B&quot;, &quot;C&quot;]},
    {&quot;ID&quot;: &quot;G&quot;, &quot;RelationIDs&quot;: []},
    {&quot;ID&quot;: &quot;H&quot;, &quot;RelationIDs&quot;: []},
]

def is_valid_pair(a, b):
    return (b in a[&quot;RelationIDs&quot;] or not a[&quot;RelationIDs&quot;]) and \
           (a in b[&quot;RelationIDs&quot;] or not b[&quot;RelationIDs&quot;])

def find_optimal_pairs(data):
    unmatched = data.copy()
    pairs = []

    while unmatched:
        found_match = False
        for a, b in itertools.combinations(unmatched, 2):
            if is_valid_pair(a, b):
                pairs.append((a[&quot;ID&quot;], b[&quot;ID&quot;]))
                unmatched = [x for x in unmatched if x[&quot;ID&quot;] != a[&quot;ID&quot;] and x[&quot;ID&quot;] != b[&quot;ID&quot;]]
                found_match = True
                break

        if not found_match:
            print(f&quot;No match found for {unmatched[0]['ID']}&quot;)
            return pairs
        
    return pairs

pairs = find_optimal_pairs(data)

for a, b in pairs:
    print(f&quot;{a} is reserved with {b}&quot;)
