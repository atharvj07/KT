df = pl.DataFrame({
    &quot;a&quot;: [1, 2, 3],
    &quot;b&quot;: [True, False, None]
}).with_row_count(&quot;row_nr&quot;)

print(df)

# filter on condition
df_a = df.filter(pl.col(&quot;row_nr&quot;) != 1)

# stack two slices
df_b = df[:1].vstack(df[2:])

# or via explicit slice syntax
df_b = df.slice(0, 1).vstack(df.slice(2, -1))

assert df_a.frame_equal(df_b)

print(df_a)
