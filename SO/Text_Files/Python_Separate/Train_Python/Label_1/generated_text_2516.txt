class C:
    _immutable = False
    def __init__(self, immutable=False):
        self._immutable = immutable
    def __getattribute__(self, name):
        if name == "__dict__":
            raise TypeError("Can't access class dict.")
        return super().__getattribute__(name)
    def __setattr__(self, name, value):
        if self._immutable:
