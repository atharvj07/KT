from typing import Callable


class BaseC:
    def __init__(self) -&gt; None:
        pass

class InheritC(BaseC):
    def __init__(self) -&gt; None:
        super().__init__()

    @classmethod
    def validate(cls, c:'InheritC') -&gt;bool:
        return False

class AggrC:
    def register_validate_fn(self, fn: Callable[[BaseC], bool])-&gt; None:
        self.validate_fn = fn

ac = AggrC()
ic = InheritC()
ac.register_validate_fn(ic.validate)
