import polars as pl

df = (
    pl.DataFrame({"id": [1, 1, 1], "data": ["a", "b", "c"]})
    .group_by("id")
    .agg(pl.col("data"))
    .with_columns(structure=pl.col("data").list.to_struct())
    .unnest("structure")
    .drop("data")
)

print(df)
