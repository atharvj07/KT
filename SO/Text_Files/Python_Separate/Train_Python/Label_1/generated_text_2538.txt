def find_all_hidden_imports(directory_path: Path) -> set:
    imports_set = set()
    for file_path in directory_path.rglob('*.py'):
        if ".venv" not in str(file_path):
            imports_set.update(get_imports_of_file(file_path))
    return imports_set
