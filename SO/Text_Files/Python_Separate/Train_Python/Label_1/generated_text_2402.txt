#include &lt;cstdint&gt;
#include &lt;cstdlib&gt;
#include &lt;vector&gt;

int64_t count_in_range(const std::vector&lt;double&gt;&amp; arr, double min_value, double max_value)
{
    int64_t count = 0;

    for(int64_t i=0 ; i&lt;arr.size() ; ++i)
    {
        double a = arr[i];

        if (min_value &lt; a &amp;&amp; a &lt; max_value)
        {
            count += 1;
        }
    }

    return count;
}
