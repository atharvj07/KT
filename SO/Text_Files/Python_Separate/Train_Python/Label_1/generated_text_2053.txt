        entry: bash -c 'cd backend &amp;&amp; mypy -p api_v1 &quot;$@&quot;' --

    # and then repeat ...
    -   id: mypy
        pass_filenames: false
        files: ^backend/api_v2/
        entry: bash -c 'cd backend &amp;&amp; mypy -p api_v2 &quot;$@&quot;' --
