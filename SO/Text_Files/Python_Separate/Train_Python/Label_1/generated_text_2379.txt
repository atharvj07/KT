import pandas as pd
df1 = pd.DataFrame({&quot;a&quot;:[1,2,3,4,5],&quot;b&quot;:[2,3,4,5,6],&quot;c&quot;:[3,4,5,6,7]})


def t(df, row1, row2, shift_len):
    return df[row1] + df[row2] * shift_len, df[row2] - shift_len


df1[[&quot;d&quot;, &quot;e&quot;]] = df1.apply(t, args=(&quot;a&quot;, &quot;b&quot;, 3), axis=1, result_type=&quot;expand&quot;)
print(df1)
