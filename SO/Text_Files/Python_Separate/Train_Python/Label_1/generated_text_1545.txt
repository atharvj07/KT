import time

def a_very_expensive_function_1():
    time.sleep(1)
    return [0,1,2]

def a_very_expensive_function_2():
    time.sleep(1)
    return [3,4,5]

def find(n, funcs):
    for e in funcs:
        r = e()
        if n in r:
            return r
    assert False, 'Not found!'

my_int = 1

print(find(my_int, [
    a_very_expensive_function_1,
    a_very_expensive_function_2,
]) + [10])
