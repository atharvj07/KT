def plus_prev_times_2(col):
    x = 2 ** pl.int_range(pl.len() - 1).reverse()
    y = 2 ** pl.int_range(1, pl.len())
    cs = (x * col.slice(1)).cum_sum()
    return cs / x + col.first() * y

     .append(pl.col.some_col.pipe(plus_prev_times_2))
     .alias("plus_prev_times_2")
