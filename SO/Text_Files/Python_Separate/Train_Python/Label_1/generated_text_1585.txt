import datetime
from tkinter import *

start_time = datetime.datetime.now()


def update_timer():
   current_time = datetime.datetime.now()
   timer_label.config(text=f'{current_time - start_time}')
   root.after(1000, update_timer)


root = Tk()
timer_label = Label(text='0')
timer_label.pack()

update_timer()
root.mainloop()
