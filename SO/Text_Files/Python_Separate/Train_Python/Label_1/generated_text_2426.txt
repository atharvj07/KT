def gen_both(gen1, gen2):
    first = next(gen1, None)
    second = next(gen2, None)
    
    while True:
        if first is None and second is None:
            break
        elif first is None:
            yield second
            second = next(gen2, None)
        elif second is None:
            yield first
            first = next(gen1, None)
        else:
            if first &lt; second:
                yield first
                first = next(gen1, None)
            elif first == second:  
                yield first, second
                first = next(gen1, None)
                second = next(gen2, None)            
            else:
                yield second
                second = next(gen2, None)
