def concated_diagonal(ar, col):
    ar = np.asarray(ar).ravel()
    size = ar.size
    ret = np.zeros((col * size, col), ar.dtype)
    for i in range(col):
        ret[i::col, i] = ar
    return ret

def concated_diagonal(ar, col):
    ar = np.asarray(ar).reshape(-1, 1)
    size = ar.size
    ret = np.zeros((col * size, col), ar.dtype)
    ret.reshape(size, -1)[:, ::col + 1] = ar
    return ret
