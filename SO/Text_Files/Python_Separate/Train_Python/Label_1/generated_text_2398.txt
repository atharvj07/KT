from fastapi import FastAPI, Depends
from pydantic import BaseModel
from typing import Literal

app = FastAPI()
    
class Quiz(BaseModel):
    question: str
    subject: str
    choice: Literal['eu', 'us', 'cn', 'ru'] = 'us'

@app.post('/submit')
def post_data(data: Quiz = Depends()):
    return data
