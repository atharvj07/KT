import dataclasses


@dataclasses.dataclass
class Data:
    data_id: int
    attribute: str


@dataclasses.dataclass
class User:
    user_id: int
    name: str
    fullname: str
    data: list[Data]


user_1 = User(
    user_id=167,
    name="john",
    fullname="John Fresno",
    data=[
        Data(data_id=538, attribute='Privileged'),
        Data(data_id=529, attribute='User')
    ]
)

users: list[User] = [user_1]
