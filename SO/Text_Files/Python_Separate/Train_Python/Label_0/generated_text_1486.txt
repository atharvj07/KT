import matplotlib.pyplot as plt
import wandb

wandb.init(project="your_project")

fig, ax = plt.subplots()
ax.plot([1, 2, 3], [4, 5, 6])

wandb.log({"chart": wandb.Image(fig)})  # Log as an image

fig.savefig("plot.png")  # Save the figure
wandb.log({"chart": wandb.Image("plot.png")})  # Log as an image
