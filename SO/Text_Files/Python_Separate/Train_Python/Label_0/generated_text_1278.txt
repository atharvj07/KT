import asyncio

async def task():
    await asyncio.sleep(10)

async def run() -> None:
    try:
        async with asyncio.TaskGroup() as tg:
            t1 = tg.create_task(task())
            t2 = tg.create_task(task())
    except KeyboardInterrupt:
        print("Stopped")
        # Explicitly re-raise the KeyboardInterrupt to propagate it properly.
        raise

asyncio.run(run())
