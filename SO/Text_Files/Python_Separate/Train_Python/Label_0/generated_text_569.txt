import turtle

captured_pieces = []

def bmove(x, y):
    black.goto(x, y)
    if black.distance(blue) < 30:
        captured_pieces.append("BlC")
        print(captured_pieces)
        check()

def blmove(x, y):
    blue.goto(x, y)
    if blue.distance(black) < 30:
        captured_pieces.append("BC")
        print(captured_pieces)
        check()

def check():
    if "BlC" in captured_pieces and "BC" in captured_pieces:
        print("Both captured")
        turtle.done()

blue = turtle.Turtle()
black = turtle.Turtle()
screen = turtle.Screen()

blue.penup()
black.penup()

blue.shape('square')
black.shape('triangle')

blue.setpos(100, 100)
black.setpos(-100, -100)

blue.color('blue')
black.color('black')

blue.ondrag(bmove)
black.ondrag(blmove)

screen.mainloop()
