def match_pattern(lst):
    pattern = ["foo", "bar"]
    if len(lst) >= len(pattern) and lst[:len(pattern)] == pattern:
        print("matched!")

# Test cases
l1 = ["foo", "bar"]
l2 = [{1, 2}, "foo", "bar"]
l3 = ["foo", "bar", 5]
l4 = [{1, 2}, "foo", "bar", 5, 6]

match_pattern(l1)
match_pattern(l2)
match_pattern(l3)
match_pattern(l4)
