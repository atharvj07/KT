import numpy as np

# Your array
A = np.array([[0.33703753, 3.],
              [0.90115394, 5.],
              [0.91172016, 5.],
              [0.93230994, 3.],
              [0.08084283, 3.],
              [0.71531777, 2.],
              [0.07880787, 3.],
              [0.03501083, 4.],
              [0.69253184, 4.],
              [0.62214452, 3.],
              [0.26953094, 1.],
              [0.4617873 , 3.],
              [0.6495549 , 0.],
              [0.84531478, 4.],
              [0.08493308, 5.]])

# Sort the array by y-values
A_sorted = A[A[:, 1].argsort()]

# Initialize variables
current_y = A_sorted[0, 1]
total_x = A_sorted[0, 0]
count = 1
result = []

# Calculate the means
for i in range(1, len(A_sorted)):
    if A_sorted[i, 1] == current_y:
        total_x += A_sorted[i, 0]
        count += 1
    else:
        result.append([total_x / count, current_y])
        current_y = A_sorted[i, 1]
        total_x = A_sorted[i, 0]
        count = 1

# Add the last mean
result.append([total_x / count, current_y])

# Convert result to a numpy array
result_array = np.array(result)

print(result_array)
