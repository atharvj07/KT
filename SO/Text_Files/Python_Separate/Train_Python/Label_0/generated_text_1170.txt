import polars as pl

data = {
    "ID": [1, 1, 2, 2, 3, 3],
    "Action": ["A", "A", "B", "B", "A", "A"],
    "Where": ["Office", "Home", "Home", "Office", "Home", "Home"],
    "Value": [1, 2, 3, 4, 5, 6]
}

df = pl.DataFrame(data)

# Compute max value per [ID, Action]
result = (
    df.with_columns(
        TOP=pl.col("Value").max().over(["ID", "Action"])
    )
    .filter(pl.col("Value") == pl.col("TOP"))  # Keep only max value rows
)

print(result)
