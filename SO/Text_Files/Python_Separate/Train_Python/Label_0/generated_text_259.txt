import datetime
import time
import sys

class Bug(object):

    def __init__(self):
        print_meta_path()

    def __del__(self):
        print_meta_path()
        try:
            try_date('time')
            try_date('datetime')
        except ImportError:
            print("ImportError occurred during cleanup.")

def print_meta_path():
    print(f'meta_path: {sys.meta_path}')

def try_date(date_type):
    try:
        print('----------------------------------------------')
        print(date_type)
        if date_type == 'time':
            print(datetime.date.fromtimestamp(time.time()))
        if date_type == 'datetime':
            print(datetime.date.today())
    except Exception as ex:
        print(ex)

if __name__ == '__main__':
    print(sys.version)
    bug = Bug()
