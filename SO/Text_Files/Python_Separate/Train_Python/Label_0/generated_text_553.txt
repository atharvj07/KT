import math
import random

def generate_normal(mu=0, sigma=1):
    u1 = random.random()
    u2 = random.random()
    
    z0 = math.sqrt(-2 * math.log(u1)) * math.cos(2 * math.pi * u2)
    z1 = math.sqrt(-2 * math.log(u1)) * math.sin(2 * math.pi * u2)
    
    return mu + sigma * z0  # z1 can also be used

# Example usage
for _ in range(10):
    print(generate_normal(0, 1))
