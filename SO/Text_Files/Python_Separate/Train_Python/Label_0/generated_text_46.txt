import numpy as np

np.random.seed(10)

data = np.random.binomial(size=215, n=1, p=0.3)

sample_mean = []

for i in range(1000):
    np.random.seed(i)  # Reset seed for each iteration
    sample = np.random.choice(data, size=215, replace=True)
    mean = np.mean(sample)
    sample_mean.append(mean)

print(np.mean(sample_mean))
