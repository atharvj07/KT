import numpy as np

def create_matrix(N):
    dimension = int(np.ceil((np.sqrt(1 + 8*N) - 1) / 2))
    matrix = np.zeros((dimension, dimension), dtype=int)
    x, y = 0, 0
    num = 1
    for i in range(dimension):
        for j in range(i, dimension):
            matrix[y][x] = num
            num += 1
            x += 1
            if j != dimension - 1:
                y += 1
        y = i + 1
        x = 0
    return matrix

N = 13
matrix = create_matrix(N)
print(matrix)
