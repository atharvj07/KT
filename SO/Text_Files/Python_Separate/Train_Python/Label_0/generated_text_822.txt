skip = [1, 2]
a = [[1, None, 2], [3, 4, 5], [1, None, 7], [8, 9, 10], [11, None, 12]]
b = 0
d = {}
for i in range(len(a)):
    if a[i][1] is None:
        if b in skip:
            b += 1
        if a[i][0] in d:
            a[i][1] = d[a[i][0]]
        else:
            a[i][1] = b
            d[a[i][0]] = b
            b += 1
print(a)
