import polars as pl

# Input DataFrame
df = pl.DataFrame({
    "a": [1, 15, None, 20, None]
})

# Compute the difference with the previous non-null value in column "a"
df = df.with_columns(
    pl.col("a")
    .fill_none(strategy="forward")  # Fill None with the last valid value forward
    .diff()  # Subtract the previous value
    .alias("b")
)

# Result
print(df)
