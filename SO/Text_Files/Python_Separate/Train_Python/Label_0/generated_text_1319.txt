import pandas as pd

# Sample dataframe
df = pd.DataFrame([
    ['A','X',2000,5,3],
    ['A','X',2001,6,2],
    ['B','X',2000,6,3],
    ['B','X',2001,7,2],
    ['C','Y',2000,10,4],
    ['C','Y',2001,12,4],
    ['D','Y',2000,11,2],
    ['D','Y',2001,15,1]
], columns=['ctry','rgn','year','val1','val2'])

# Group by 'rgn' and 'year' and calculate the required 'value' column
new_df = df.groupby(['rgn', 'year']).agg(
    value=('val1', lambda x: (x * df.loc[x.index, 'val2']).sum() / df.loc[x.index, 'val2'].sum())
).reset_index()

# Display the result
print(new_df)
