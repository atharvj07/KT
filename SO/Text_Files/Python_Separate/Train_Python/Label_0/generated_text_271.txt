import numpy as np

def find_sublist(numbers, tolerance):
    numbers.sort()
    result = []
    n = len(numbers)
    for i in range(n):
        left, right = i + 1, n - 1
        while left < right:
            total = numbers[i] + numbers[left] + numbers[right]
            if abs(total) <= tolerance:
                result.append([numbers[i], numbers[left], numbers[right]])
                left += 1
                right -= 1
            elif total < -tolerance:
                left += 1
            else:
                right -= 1
    return result
