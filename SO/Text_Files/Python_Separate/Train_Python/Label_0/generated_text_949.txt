import numpy as np

# Define matrices a and b
a = np.array([[1, 4, 7, -2],
              [3, 0, -2, -1],
              [-4, 2, 1, 0],
              [-8, -3, -1, 2]])

b = np.array([[-4, 2, 1, 0],
              [3, 0, -2, -1],
              [-8, -3, -1, 2],
              [1, 4, 7, -2]])

# Find the permutation matrix P
P = np.linalg.inv(a) @ b

# Round the elements to nearest integer, as they should be either 0 or 1
P = np.round(P).astype(int)

print("Permutation matrix P:")
print(P)
