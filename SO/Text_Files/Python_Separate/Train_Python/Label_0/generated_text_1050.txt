import polars as pl
import numpy as np

df = pl.DataFrame({
    "a": np.arange(5),
    "b": np.arange(5)[::-1]
})

# Compute elementwise maximum inside a Polars expression
df = df.with_columns(
    pl.col("a").zip_with(pl.col("a") > pl.col("b"), pl.col("a"), pl.col("b")).alias("max(a, b)")
)

print(df)

df = df.with_columns(
    pl.struct(["a", "b"]).map_elements(lambda x: max(x["a"], x["b"])).alias("max(a, b)")
)
