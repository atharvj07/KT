import asyncio

async def coro1():
    await asyncio.sleep(1)
    return 'Result from coro1'

async def coro2():
    await asyncio.sleep(2)
    return 'Result from coro2'

async def main():
    tasks = [coro1(), coro2()]
    results = await asyncio.gather(*tasks)
    print(results)

asyncio.run(main())
