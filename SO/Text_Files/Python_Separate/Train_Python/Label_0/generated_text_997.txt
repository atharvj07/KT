from dataclasses import dataclass, field

@dataclass
class Person:
    name: str
    age: str

    def __post_init__(self):
        self.name = self.Name()
        self.age = self.Age()

    def Name(self):
        return f'My name is {self.name[0]} {self.name[1]}.'

    def Age(self):
        return f'I am {self.age} years old.'

# Creating an instance of Person
person = Person(('John', 'Smith'), '100')

# Printing the result
print(person)
