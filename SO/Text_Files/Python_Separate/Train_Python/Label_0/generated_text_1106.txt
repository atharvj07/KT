template = {    
    'a': ['A', 'AA'],
    'b': ['B', 'BB'],
    'c': ['C', 'CC'],
    'D1': [d.get('D1') for d in d_values],
    'DD1': [d.get('DD1') for d in d_values]
}

df_polars = pl.DataFrame(template)
print(df_polars)
