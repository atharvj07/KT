def count(t):
    if not t:
        return 0
    
    count = 0
    max_left = t[0]
    
    for i in range(1, len(t)):
        if t[i] >= max_left:
            max_left = t[i]
        else:
            count += 1
    
    return count + 1  # Add 1 to account for the initial split

# Example usage:
lst = [2, 1, 2, 5, 7, 6, 9]
result = count(lst)
print(result)  # Output: 3
