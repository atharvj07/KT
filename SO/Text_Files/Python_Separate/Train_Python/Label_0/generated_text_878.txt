import numpy as np

a = np.array([1, 2, 3, 4])
b = np.array([5, 6])

# Create an array to hold the result
rows = len(a) + 1
cols = len(a) + len(b)
res = np.empty((rows, cols), dtype=int)

# Fill the result array using broadcasting and slicing
for i in range(rows):
    res[i, i:i+len(b)] = b
    res[i, len(b)+i:] = a[i:]
    res[i, :i] = a[:i]

print(res)
