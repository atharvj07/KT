import polars as pl

# Sample DataFrame
df = pl.DataFrame({
    "Column 1": ["A", "B"],
    "Column 2": [1, 2],
    "Column 3": ["W", "X"],
    "Column 4": ["C", "D"],
    "Column 5": [3, 4],
    "Column 6": ["Y", "Z"]
})

# Convert to a single Series and reshape
reshaped = (
    df.to_series()
    .explode()
    .reshape((-1, 3))
)

# Create new DataFrame with correct column names
new_df = pl.DataFrame(reshaped, schema=["Column 1", "Column 2", "Column 3"])

print(new_df)
