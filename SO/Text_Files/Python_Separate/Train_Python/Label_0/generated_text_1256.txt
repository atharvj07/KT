import pandas as pd
import numpy as np

# Sample DataFrame
df = pd.DataFrame({
    'start': [3, 11, 9, 19, 22],
    'end': [10, 17, 21, 25, 30]
})

# Initialize an array to hold the x values
x_values = np.full(len(df), np.nan)

# Keep track of the max 'end' encountered so far
max_end = -np.inf

for i, row in df.iterrows():
    if row['start'] > max_end:
        max_end = row['end']
        x_values[i] = max_end

# Assign the result to the 'x' column
df['x'] = x_values

print(df)
