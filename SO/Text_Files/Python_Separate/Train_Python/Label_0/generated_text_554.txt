from pydantic import BaseModel
from datetime import date

class User(BaseModel):
    id: int
    name: str = 'John Doe'
    sex: str
    money: float = None
    dt: date

data = [1, 'Tike Myson', 'male', None, '2022-01-20']
data_dict = dict(zip(User.__annotations__.keys(), data))
user = User(**data_dict)
