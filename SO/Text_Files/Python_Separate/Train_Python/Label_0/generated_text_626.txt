from typing import TypeVar, Tuple, Generic, TypeVarTuple, dataclass

T = TypeVar("T")
Ts = TypeVarTuple("Ts")

@dataclass
class S(Generic[T]):
    data: T

def data_from_s(*structs: Ts) -> Tuple:
    return tuple(x.data for x in structs)

a = data_from_s(S(1), S("3"))  # a is type Tuple[int, str]
print(a)
