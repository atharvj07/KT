import numpy as np

          'y' : [10, 1, 5,  np.nan, 47, np.nan, 8, 5, 100, 3]}
df = pd.DataFrame(mydata)

df["z"] = np.where((df["x"] >= 50) & (df["y"] <= 20), 1, np.nan)

print(df)
