import asyncio

async def hel():
    return 4

def i():
    loop = asyncio.new_event_loop()  # Create a new event loop
    asyncio.set_event_loop(loop)  # Set it as the current event loop
    x = asyncio.run(hel())  # Run the async function in the new event loop
    print(x)  # Print the result

async def h():
    i()

asyncio.run(h())
