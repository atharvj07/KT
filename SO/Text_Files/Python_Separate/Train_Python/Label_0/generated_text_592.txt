class ReusableIterable:
    def __init__(self, generator_func):
        self.generator_func = generator_func

    def __iter__(self):
        return self.generator_func()

def reusable_iterable(generator_func):
    return ReusableIterable(generator_func)

@reusable_iterable
def myrange(n):
    for i in range(n):
        yield i

x = myrange(3)
print(list(x))  # [0, 1, 2]
print(list(x))  # [0, 1, 2]
