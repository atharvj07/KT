import polars as pl
import numpy as np

np.random.seed(0)
ragged_arrays = [np.random.randint(10, size=np.random.choice(range(10))).tolist() for _ in range(5)]

df = pl.DataFrame({'values': ragged_arrays})

df = df.with_columns(
    pl.col("values").list.eval(pl.element().diff()).alias("diff_values")
)

print(df)
