def gen_even():
    for x in range(0, 11, 2):
        yield x

def gen_odd():
    for x in sorted(list(range(1, 15, 2)) + [6]):
        yield x

def gen_both(gen1, gen2):
    try:
        first = next(gen1)
    except StopIteration:
        first = float('inf')  # Set a large value if gen1 is exhausted
    try:
        second = next(gen2)
    except StopIteration:
        second = float('inf')  # Set a large value if gen2 is exhausted
    
    while True:
        if first < second:
            yield first
            try:
                first = next(gen1)
            except StopIteration:
                first = float('inf')
        elif first == second:
            yield first, second
            try:
                first = next(gen1)
            except StopIteration:
                first = float('inf')
            try:
                second = next(gen2)
            except StopIteration:
                second = float('inf')
        else:
            yield second
            try:
                second = next(gen2)
            except StopIteration:
                second = float('inf')

gen1 = gen_even()
gen2 = gen_odd()

gen = gen_both(gen1, gen2)

for i in gen:
    print(i)
