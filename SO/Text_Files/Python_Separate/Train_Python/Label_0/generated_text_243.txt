from datetime import datetime
from random import randint

arr = [randint(-1000, 1000) for _ in range(1000)]

def brute_force(a):
    max_sum = 0
    for i in range(len(a)):
        max_sum = max(max_sum, max([sum(a[i:j]) for j in range(i+1, len(a)+1)]))
    return max_sum

start = datetime.now()
brute_force(arr)
end = datetime.now()

print(format(end - start))
