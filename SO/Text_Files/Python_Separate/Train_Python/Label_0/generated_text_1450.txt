def all_combinations_sum_to_n(n, start=1, path=None):
    if path is None:
        path = []
        
    if n == 0:
        print(path)  # Print the valid combination
        return
    
    for i in range(start, n + 1):  # Ensure increasing order
        all_combinations_sum_to_n(n - i, i, path + [i])
