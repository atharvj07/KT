import re

pattern = r"^(?P<prefix>[^A-Z]+-)?(?P<sequence>[^A-Z]*[A-Z])(?P<suffix>-[^A-Z]+)?$"
test_strings = [
    "foo-ABcmCD-bar",
    "DEF",
    "WHATEVER-foo",
    "cmC-foo",
    "ac-cmC-foo",
]

for test_str in test_strings:
    result = re.findall(pattern, test_str)
    print(result)
