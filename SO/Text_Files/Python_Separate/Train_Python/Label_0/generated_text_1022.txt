fun gen(): Sequence<Int> = sequence {
    var x = 0
    while (true) {
        yield(x)
        x += 1
    }
}

fun main() {
    val generator = gen().iterator() // Creates an iterator for the sequence

    println(generator.next()) // 0
    println(generator.next()) // 1
    println(generator.next()) // 2
}

class StatefulGen {
    private var x = 0

    fun next(): Int {
        return x++
    }
}

fun main() {
    val generator = StatefulGen()

    println(generator.next()) // 0
    println(generator.next()) // 1
    println(generator.next()) // 2
}
