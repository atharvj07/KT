import numpy as np

def add(A, B):
    # Extract alpha and normalize to [0,1]
    alpha = A[..., 3:4] / 255.0  

    # Perform blending using efficient broadcasting
    B[..., :3] = ((A[..., :3] * alpha) + (B[..., :3] * (1 - alpha))).astype(np.uint8)

    return B
