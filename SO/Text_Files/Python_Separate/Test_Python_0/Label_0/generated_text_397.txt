import polars as pl

df = pl.DataFrame(
    {
        "col1": ["a", "b", "a", "b", "c"],
        "col2": ["val1", "val2", "val1", "val3", "val3"]
    }
)

result = df.group_by("col1").agg(
    col2_g = pl.concat_str("col2", separator=",")
)

print(result)
