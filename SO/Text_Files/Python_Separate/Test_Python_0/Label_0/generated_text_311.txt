from typing import TypedDict

class A(TypedDict):
    a: int
    b: int

def create_a() -> A:
    return A(a=0, b=1)

# Usage
a_instance = create_a()
print(a_instance)  # {'a': 0, 'b': 1}

from collections import defaultdict

class A(TypedDict):
    a: int
    b: int

def create_a() -> A:
    return defaultdict(lambda: 0, {'a': 0, 'b': 1})

# Usage
a_instance = create_a()
print(a_instance['a'])  # 0
print(a_instance['b'])  # 1
