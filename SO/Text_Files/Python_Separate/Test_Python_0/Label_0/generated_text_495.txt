import polars as pl

df = pl.DataFrame({"x": [1.2, 1.3, 3.4, 3.5]})

df = df.with_columns(
    y = pl.col("x").rolling(
        index_column=pl.int_range(0, df.height, eager=True),
        period="4i",
        offset="-3i"
    )
)

print(df)
