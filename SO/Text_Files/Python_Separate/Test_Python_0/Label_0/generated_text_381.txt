import polars as pl

# Sample dataframe
df = pl.DataFrame({
    "x": [[1], [1, 2], [1, 2, 3]],
    "y": [[True], None, [True, False, True]]
})

# Fill the nulls with a list of None values of matching length
df = df.with_columns(
    pl.when(pl.col("y").is_null())
    .then(pl.repeat(None, pl.col("x").list.len()))
    .otherwise(pl.col("y"))
    .alias("y")
)

# Explode both columns
result = df.explode(["x", "y"])

print(result)
