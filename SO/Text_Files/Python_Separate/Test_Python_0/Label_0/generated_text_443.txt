import polars as pl

df = pl.DataFrame(
    {
        "a": [1, 8, 3],
        "b": [4, 5, None],
    }
)

df.with_columns(
    max_col=pl.arg_max("a", "b").map_elements(lambda idx: ["a", "b"][idx])
)
