import typing

def func():
    class B:
        pass
    class A:
        some_b: "B"
        
    # Pass the local namespace to resolve forward references correctly
    print(typing.get_type_hints(A, localns=locals()))

func()
# Output: {'some_b': <class '__main__.func.<locals>.B'>}
