import pandas as pd
import attr

@attr.define
class MyAttrsClass:
    i: int
    s: str

# Create a list of MyAttrsClass instances
data = [MyAttrsClass(i=1, s="a"), MyAttrsClass(i=2, s="b")]

# Convert the list of attrs objects to a DataFrame
df = pd.DataFrame([attr.asdict(obj) for obj in data])

print(df)

   i  s
0  1  a
1  2  b

s: str
