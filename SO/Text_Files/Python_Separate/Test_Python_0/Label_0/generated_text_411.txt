np.cumsum([1, 2, 3, np.nan, 4, 5, 6])
# Output:
# array([ 1.,  3.,  6., nan, nan, nan, nan])

df = pd.DataFrame({'column': [1, 2, 3, np.nan, 4, 5, 6]})
np.cumsum(df['column'])
# Output:
# 0     1.0
# 1     3.0
# 2     6.0
# 3     NaN
# 4    10.0
# 5    15.0
# 6    21.0

np.nancumsum([1, 2, 3, np.nan, 4, 5, 6])
# Output:
# array([ 1.,  3.,  6.,  6., 10., 15., 21.])
