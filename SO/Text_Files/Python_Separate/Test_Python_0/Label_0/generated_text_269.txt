from pydantic import BaseModel, validator

class Window(BaseModel):
    size: tuple[int, int]

    @validator('size', pre=True)
    def transform(cls, raw: str) -> tuple[int, int]:
        x, y = raw.split('x')
        return int(x), int(y)

# Usage
fields = {'size': '1920x1080'}
window = Window(**fields)

print(window)
