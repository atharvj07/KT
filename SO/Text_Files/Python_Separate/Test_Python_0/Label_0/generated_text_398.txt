import polars as pl

data = pl.DataFrame({
    'ColumnA': ['apple', 'banana', 'cherry']
})

# Use .str.slice()
result = data.with_columns(
    pl.col('ColumnA').str.slice(0, 3).alias('sliced')
)

print(result)
