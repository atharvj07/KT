import polars as pl
import numpy as np
import functools

A = pl.DataFrame({ "index": [0, 20, 40], "A": ['a', 'b', 'c'] })
B = pl.DataFrame({ "index": [10, 20, 30], "B": ['d', 'e', 'f'] })
C = pl.DataFrame({ "index": [20, 30], "C": ['g', 'h'] })

df = functools.reduce(lambda left, right: left.join(right, on='index', how='outer'), [A, B, C])
df = df.sort("index")
