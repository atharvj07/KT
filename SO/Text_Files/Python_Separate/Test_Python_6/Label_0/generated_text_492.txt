import polars as pl

df = pl.DataFrame({
    "session_id": [1, 1, 2, 2, 3, 3],
    "fqid": ["gramps", "intro", "gramps", "teddy", "teddy", "intro"],
    "elapsed_time": [1323, 831, 1147, 5197, 6180, 7014]
})

# Group by session_id, then get value counts of fqid
value_counts = df.groupby("session_id").agg(
    pl.col("fqid").value_counts().alias("fqid_counts")
)

# Create a new column for each unique fqid value
fqid_cols = value_counts.select([
    pl.col("fqid_counts").arr.get(i).alias(f"fqid_{i}") for i in range(3)
])

# Merge with original dataframe
result = value_counts.join(fqid_cols, on="session_id")

print(result)
