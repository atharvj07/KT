from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser
from langchain.chains import LLMChain

# Try setting verbose directly in the LLMChain
prompt = ChatPromptTemplate.from_template("tell me a joke about {topic}")
model = ChatOpenAI(verbose=True)  # Verbose flag should be passed here
output_parser = StrOutputParser()

chain = LLMChain(prompt=prompt, llm=model, output_parser=output_parser)

result = chain.invoke({"topic": "ice cream"})
print(result)
