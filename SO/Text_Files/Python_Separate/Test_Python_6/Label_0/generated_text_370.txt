import polars as pl

df = pl.DataFrame({"alertTime": pl.datetime_range("2024-03-01", "2024-03-07", interval="1d")})

days_of_week = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday', 'Sunday']

df = df.with_columns(
    pl.col("alertTime").dt.weekday().map_elements(lambda x: days_of_week[x]).alias("day_of_week")
)
