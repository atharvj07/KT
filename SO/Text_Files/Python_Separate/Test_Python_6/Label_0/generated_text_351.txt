from fastapi import FastAPI

def lifespan(_):
    print("startup")
    yield
    print("shutdown")

app = FastAPI(lifespan=lifespan)

@app.get("/")
async def root():
    return {"message": "Hello World"}

sub_app = FastAPI()

@sub_app.get("/")
async def sub_root():
    return {"message": "Hello Sub World"}

async def sub_startup():
    print("sub startup")

async def sub_shutdown():
    print("sub shutdown")

sub_app.router.on_startup.append(sub_startup)
sub_app.router.on_shutdown.append(sub_shutdown)

app.mount("/sub", sub_app)
