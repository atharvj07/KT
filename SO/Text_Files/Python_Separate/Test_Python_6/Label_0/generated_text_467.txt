from flask import Flask, render_template, jsonify
import map_plotting_test as mpt
import threading

app = Flask(__name__)

def create_map_task():
    mpt.create_map()

@app.route('/')
def render_the_map():
    thread = threading.Thread(target=create_map_task)
    thread.start()
    return render_template("loading.html")

@app.route('/map')
def render_the_actual_map():
    return render_template("map.html")

if __name__ == '__main__':
    app.run(debug=True)
