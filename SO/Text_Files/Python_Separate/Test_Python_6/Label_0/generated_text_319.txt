import os
from tqdm import tqdm
from PIL import Image
import genai

model = genai.GenerativeModel('gemini-pro-vision')

image_dir = "path_to_images"
images_to_process = [os.path.join(image_dir, image) for image in os.listdir(image_dir)][:107]

prompt = """Carefully scan these images: if it has text, extract all the text and return it. If the image does not have text return '<000>'."""

batch_size = 60
batched_images = [images_to_process[i: i + batch_size] for i in range(0, len(images_to_process), batch_size)]

for batch in tqdm(batched_images):
    inputs = [[prompt, Image.open(img)] for img in batch]
    outputs = model.generate_content(inputs)
    
    for output in outputs:
        print(output.text)
