from dataclasses import dataclass
from typing import Protocol

class HasA(Protocol):
    a: int

class HasHasA(Protocol):
    hasa: HasA

@dataclass
class Inner(HasA):  
    a: int

@dataclass
class Outer:
    hasa: HasA  

def func(b: HasHasA): ...

o = Outer(Inner(0))
func(o)
