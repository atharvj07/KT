import cv2 as cv
import numpy as np

image = cv.imread("boston_cooking_a.jpg", cv.IMREAD_GRAYSCALE)
if image is None:
    print("Error loading image")
    exit()

image = cv.GaussianBlur(image, (5, 5), 0)
thresh = cv.adaptiveThreshold(image, 255, cv.ADAPTIVE_THRESH_GAUSSIAN_C, cv.THRESH_BINARY_INV, 51, 9)

kernel = np.ones((2, 2), np.uint8)
morph = cv.morphologyEx(thresh, cv.MORPH_CLOSE, kernel, iterations=2)

edges = cv.Canny(morph, 50, 150)

lines = cv.HoughLinesP(edges, 1, np.pi / 180, 100, minLineLength=30, maxLineGap=5)
result = cv.cvtColor(image, cv.COLOR_GRAY2BGR)

if lines is not None:
    for line in lines:
        x1, y1, x2, y2 = line[0]
        cv.line(result, (x1, y1), (x2, y2), (0, 255, 0), 2)

cv.imwrite("baseline_result.jpg", result)
cv.imshow("Result", result)
cv.waitKey(0)
cv.destroyAllWindows()
