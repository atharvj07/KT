import numpy as np
from sklearn.preprocessing import PolynomialFeatures
from sklearn.linear_model import LinearRegression

def fit_polynomial(X, y, degree=2):
    poly = PolynomialFeatures(degree=degree)
    Xp = poly.fit_transform(X)
    model = LinearRegression()
    model.fit(Xp, y)
    return model, poly

def compute_gradient_hessian(model, poly, X):
    coefs = model.coef_
    n_features = X.shape[1]
    
    gradient = coefs[1:n_features+1]
    
    hessian = np.zeros((n_features, n_features))
    index = n_features + 1
    for i in range(n_features):
        for j in range(i, n_features):
            hessian[i, j] = coefs[index]
            hessian[j, i] = coefs[index]
            index += 1
            
    return np.array(gradient), np.array(hessian)

X = np.random.rand(10, 2)
y = np.random.rand(10)

model, poly = fit_polynomial(X, y, degree=2)
grad, hess = compute_gradient_hessian(model, poly, X)
print("Gradient:", grad)
print("Hessian:", hess)
