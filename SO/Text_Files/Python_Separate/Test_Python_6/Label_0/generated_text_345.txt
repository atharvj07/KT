from itertools import combinations
from functools import lru_cache

@lru_cache(None)
def generate_groups(letters, k, n):
    if k == 1:
        return [[tuple(sorted(letters))]]
    
    all_groups = []
    for group in combinations(letters, n):
        remaining = tuple(sorted(set(letters) - set(group)))
        sub_groups = generate_groups(remaining, k - 1, n)
        for sg in sub_groups:
            all_groups.append([tuple(sorted(group))] + sg)
    
    unique_combos = {tuple(sorted(g)) for g in all_groups}
    return [list(g) for g in unique_combos]

letters = tuple("ABCDEFGHIJKL")
k, n = 4, 3
combinations = generate_groups(letters, k, n)

print(combinations[:5])
