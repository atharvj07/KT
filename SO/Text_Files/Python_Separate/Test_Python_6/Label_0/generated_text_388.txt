import inspect
import ast

def fct1():
    frame = inspect.currentframe().f_back
    try:
        source_lines, _ = inspect.getsourcelines(frame)
        caller_line = source_lines[frame.f_lineno - 1].strip()
        if "=" in caller_line:
            var_name = caller_line.split("=")[0].strip()
            return {"name": var_name}
    finally:
        del frame

    return None

a = fct1()
