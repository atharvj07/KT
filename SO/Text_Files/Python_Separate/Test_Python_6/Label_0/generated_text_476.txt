import hashlib
import sys
from color import Color

def get_hash(algorithm):
    filename = input("Enter the file name: ")
    hash_obj = hashlib.new(algorithm)
    with open(filename, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            hash_obj.update(byte_block)
    print(f"{algorithm} value has been calculated")
    color_reset()

def verify_checksum(algorithm):
    print(f"Verifying {algorithm} checksum...")

def which_hash():
    sha256_or_sha512 = input("Which hash do you want to calculate: sha256 or sha512? \n")
    if sha256_or_sha512 in ["sha256", "sha512"]:
        get_hash(sha256_or_sha512)
        verify_checksum(sha256_or_sha512)
    else:
        print("Type either sha256 or sha512. If you type anything else the program will close...like this.")
        sys.exit()

if __name__ == "__main__":
    which_hash()
