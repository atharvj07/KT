from dataclasses import dataclass
from typing import Callable, Dict, Any, Type

def mydecorator(cls: Type[Any]) -> Type[Any]:
    def save(self) -> None:
        print(f"Saving {self}")

    setattr(cls, "save", save)
    setattr(cls, "my_fields_dict", {"a": str})

    cls.__annotations__["save"] = Callable[[], None]
    cls.__annotations__["my_fields_dict"] = Dict[str, Any]

    return cls

@mydecorator
@dataclass
class A:
    a: str = ""

a = A()
a.save()
print(a.my_fields_dict)
