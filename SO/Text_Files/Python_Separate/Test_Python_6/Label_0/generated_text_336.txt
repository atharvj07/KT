from dataclasses import dataclass
from typing import Type, TypeVar

T = TypeVar("T")

def something_else(klass: Type[T]) -> Type[T]:
    return klass

def my_dataclass(klass: Type[T]) -> Type[T]:
    return something_else(dataclass(klass))

@my_dataclass
class TestB:
    a: int
    b: str

TestB(0, "")  # should work
