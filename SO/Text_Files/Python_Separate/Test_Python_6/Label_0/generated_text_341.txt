import polars as pl

df = pl.DataFrame(
    {
        "id": ["NY", "TK", "FD"], 
        "eat2000": [1, None, 3], 
        "eat2001": [-2, None, 4],
        "eat2002": [None, None, None],
        "eat2003": [-9, None, 8],
        "eat2004": [None, None, 8]
    }
)

cols = [c for c in df.columns if c.startswith("eat")]

df = df.filter(~df.select(cols).is_null().all(axis=1))
