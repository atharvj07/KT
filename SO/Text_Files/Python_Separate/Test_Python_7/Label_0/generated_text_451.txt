from airflow.sensors.sql import SqlSensor
import pendulum
from airflow.decorators import task, dag

@dag(
    dag_id="db_mon",
    schedule="*/10 * * * *",
    start_date=pendulum.datetime(2023, 7, 16, 21, 0, tz="UTC"),
    catchup=False
)
def pipe():
    chk = SqlSensor(
        task_id="chk_db",
        conn_id="evergreen",
        sql="SELECT pg_is_in_recovery()",
        success=lambda x: x == False,
        poke_interval=60,
        mode="reschedule"
    )

    @task()
    def alert():
        import requests
        data = {
            "@key": "kkll",
            "@version": "alertapi-0.1",
            "@type": "ALERT",
            "object": "Testobject",
            "severity": "MINOR",
            "text": "Former primary instance is in recovery"
        }
        requests.post("https://httpevents.systems/api/sendAlert", verify=False, data=data)

    chk >> alert()

dag = pipe()
