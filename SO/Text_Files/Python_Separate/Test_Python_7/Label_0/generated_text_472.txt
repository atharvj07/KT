import pandas as pd
import requests
from bs4 import BeautifulSoup

page = requests.get('https://www.bmv.com.mx/es/mercados/capitales')
soup = BeautifulSoup(page.text, 'lxml')

thead = soup.find('thead')
tr = thead.find_all('tr')
hdr = [t.get_text().strip().replace('\n', ',').split(',') for t in tr][0]

df = pd.DataFrame(columns=hdr)

tbody = soup.find('tbody')
rows = tbody.find_all('tr')

for row in rows:
    cols = row.find_all('td')
    link = cols[0].find('a')['href']
    em = cols[0].get_text().strip()
    serie = cols[1].get_text().strip()
    hora = cols[2].get_text().strip()
    ultimo = cols[3].get_text().strip()
    ppp = cols[4].get_text().strip()
    ant = cols[5].get_text().strip()
    max_ = cols[6].get_text().strip()
    min_ = cols[7].get_text().strip()
    vol = cols[8].get_text().strip()
    imp = cols[9].get_text().strip()
    ops = cols[10].get_text().strip()
    var_pts = cols[11].get_text().strip()
    var_pct = cols[12].get_text().strip()
    
    row_data = [em, serie, hora, ultimo, ppp, ant, max_, min_, vol, imp, ops, var_pts, var_pct]
    
    df.loc[len(df)] = row_data
