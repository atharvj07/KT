import polars as pl
import numpy as np
import functools

A = pl.DataFrame({ "idx": [0, 20, 40], "A": ['a', 'b', 'c'] })
B = pl.DataFrame({ "idx": [10, 20, 30], "B": ['d', 'e', 'f'] })
C = pl.DataFrame({ "idx": [20, 30], "C": ['g', 'h'] })

F = functools.reduce(lambda x, y: x.join(y, on="idx", how="outer"), [A, B, C])
print(F.sort("idx"))

N, C = 300000, 20
lst = [pl.DataFrame({ "idx": np.linspace(i, N*3-i, num=N, dtype=np.int32), f"A{i}": np.arange(N, dtype=np.float32) }) for i in range(C)]
res = functools.reduce(lambda x, y: x.join(y, on="idx", how="outer"), lst)
print(res.sort("idx"))
