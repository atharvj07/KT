import polars as pl

def exp(df, x, y):
    df = df.with_columns(pl.map_zip([x, y], lambda a, b: list(zip(a, b))).alias("z"))
    return df.explode("z").select(pl.col("z").list.first().alias(x), pl.col("z").list.last().alias(y))

data = {
    "a": [[1], [2], [3, 4], [5, 6, 7]],
    "b": [[], [8], [9, 10], [11, 12]],
}

df = pl.DataFrame(data)
res = exp(df, "a", "b")
print(res)
