import numpy as np

def ijRev(i, j, rows, cols):
    r = i + j
    if r >= cols:
        return rows - 1 - (r - cols), cols - 1 - j
    return i, j

def rev_anti_diag(arr):
    rows, cols = arr.shape
    rev_arr = np.empty_like(arr)
    for i in range(rows):
        for j in range(cols):
            rev_i, rev_j = ijRev(i, j, rows, cols)
            rev_arr[rev_i, rev_j] = arr[i, j]
    return rev_arr

srcArr = np.array([[ 1,  2,  4],
                   [ 3,  5,  7],
                   [ 6,  8, 10],
                   [ 9, 11, 13],
                   [12, 14, 15]])

rev_arr = rev_anti_diag(srcArr)
print(rev_arr)
