import itertools

def get_combos(x, k, n):
    let = [chr(i) for i in range(65, 65 + x)]
    groups = list(itertools.combinations(let, n))
    return [list(comb) for comb in itertools.combinations(groups, k) if len(set(itertools.chain(*comb))) == x]

combs = get_combos(12, 4, 3)
print(combs)
