import os
from fastapi import FastAPI, Depends, HTTPException, Security
from fastapi.security.api_key import APIKeyHeader
from fastapi.routing import APIRouter

API_KEY = "your-api-key"
api_key_header_auth = APIKeyHeader(name="X-API-KEY")
env = os.getenv("ENV", "localhost")

async def api_key_func(api_key_header: str = Security(api_key_header_auth)):
    if env != "localhost" and api_key_header != API_KEY:
        raise HTTPException(
            status_code=401,
            detail="Invalid API Key",
        )

router = APIRouter(
    prefix="/v2/test",
    tags=["helloWorld"],
    dependencies=[Depends(api_key_func)],
    responses={404: {"description": "Not found"}},
)

app = FastAPI()
app.include_router(router)
