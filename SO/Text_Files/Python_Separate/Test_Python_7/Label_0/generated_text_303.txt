import bpy

inp = 'mushroom-shelve-1-merged.obj'

# Load the triangle mesh OBJ file
bpy.ops.wm.open_mainfile(filepath='your_blend_file.blend')

# Import the OBJ file as a mesh object

obj = bpy.context.selected_objects[0]
