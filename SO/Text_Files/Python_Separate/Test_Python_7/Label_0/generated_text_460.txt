import numpy as np
from collections import Counter


def bal(array: np.ndarray):
    cls = [(c, cnt) for c, cnt in Counter(array).items()]
    m = len(cls)
    n = len(array)
    H = -sum([(cnt / n) * np.log((cnt / n)) for c, cnt in cls])
    return H / np.log(m)


def wt(days: int, eps: float = 0.005):
    return np.exp(-1 * days * eps)


def most_balanced(opps, days, eps=0.005):
    res = {}
    for opp in set(opps):
        a = np.array([days[i] for i, p in enumerate(opps) if p == opp])
        bal_val = bal(a)
        weighted = sum([wt(d) for d in a])
        res[opp] = {'bal': bal_val, 'weighted': weighted}
    
    return min(res.items(), key=lambda x: (x[1]['bal'], -x[1]['weighted']))


opps = ['C', 'C', 'D', 'F', 'F', 'F', 'G', 'C', 'C', 'D', 'F', 'F', 'F', 'G', 'G', 'G', 'G', 'G', 'G', 'G', 'G']
days = [1, 2, 10, 100, 101, 102, 1, 1, 2, 10, 100, 101, 102, 1, 2, 3, 4, 5, 6, 7, 8]

print(most_balanced(opps, days))
