from sqlalchemy import Table, Column, Integer, TIMESTAMP, MetaData
from sqlalchemy.sql import text

Meta = MetaData()

user = Table(
    "users",
    Meta,
    Column("id", Integer, primary_key=True),
    Column(
        "datemodified",
        TIMESTAMP,
        server_default=text("CURRENT_TIMESTAMP"),
        onupdate=text("CURRENT_TIMESTAMP")
    ),
)
