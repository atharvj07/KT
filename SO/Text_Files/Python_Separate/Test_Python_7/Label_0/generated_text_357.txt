from pyspark.sql import SparkSession
from pyspark.sql import functions as F
from pyspark.sql.window import Window

spark = SparkSession.builder.master("local").appName("WindowExample").getOrCreate()

data = [(1, "A", 10), (2, "B", 20), (3, "C", 30), (4, "D", 40)]
df = spark.createDataFrame(data, ["id", "name", "value"])

window_spec = Window.orderBy("id")

df_with_sum = df.withColumn("sum_value", F.sum("value").over(window_spec.rowsBetween(Window.unboundedPreceding, Window.currentRow)))
df_with_sum.show()

df_with_sum_all = df.withColumn("sum_value_all", F.sum("value").over(window_spec.rowsBetween(Window.unboundedPreceding, Window.unboundedFollowing)))
df_with_sum_all.show()
