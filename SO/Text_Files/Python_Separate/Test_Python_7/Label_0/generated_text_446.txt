import pandas as pd

df = pd.DataFrame({
    "": [0,1,2,3,4,5,6],
    "p1": ["ryan", "delta", "delta", "delta","bravo","alpha","ryan"], 
    "p2": ["delta", "ryan", "alpha", "bravo","delta","ryan","alpha"], 
    "msg": [1, 1, 2, 3,3,9,9]
})

df["key"] = df.apply(lambda x: tuple(sorted([x["p1"], x["p2"]])), axis=1)
df = df.drop_duplicates(subset=["key", "msg"]).drop(columns=["key"])
print(df)
