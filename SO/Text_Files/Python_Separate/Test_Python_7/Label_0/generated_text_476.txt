import hashlib
import sys

def get_hash(alg):
    filename = input("Enter the file name: ")
    hash_obj = hashlib.new(alg)
    with open(filename, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            hash_obj.update(byte_block)
    print(f"{alg} value has been calculated")
    return hash_obj.hexdigest()

def verify_checksum(alg):
    checksum = input(f"Enter the {alg} checksum: ")
    calc_hash = get_hash(alg)
    if calc_hash == checksum:
        print(f"{alg} checksum verified!")
    else:
        print(f"{alg} checksum does not match.")

def which_hash():
    alg = input("Which hash do you want to calculate: sha256 or sha512? \n")
    if alg == "sha256" or alg == "sha512":
        calc = get_hash(alg)
        verify_checksum(alg)
    else:
        print("Type either sha256 or sha512. If you type anything else the program will close...like this.")
        sys.exit()

if __name__ == "__main__":
    which_hash()
