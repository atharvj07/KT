import polars as pl
import numpy as np

def main():
    df = pl.DataFrame({
        'col1': [1, 2, 3, 4, 5],
        'col2': [5, 4, 3, 2, 1]
    })
    preds = np.array([10, 20, 30, 40, 50])
    df = df.with_columns(pl.Series('predictions', preds))
    print(df)

if __name__ == "__main__":
    main()
