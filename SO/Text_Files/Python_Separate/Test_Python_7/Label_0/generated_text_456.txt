from math import factorial
from decimal import Decimal, getcontext
from joblib import Parallel, delayed
from tqdm import trange
import time

def calc(n, p, t):
    n = int(n + 1 / 14.181647462725477)
    n = n if n >= 1 else 1
    getcontext().prec = n + 1
    t = Decimal(0)
    pi = Decimal(0)
    deno = Decimal(0)
    for k in trange(n, position=p, desc=f"Job {p + 1} of {t}", leave=True, ncols=100):
        t = ((-1)**k) * (factorial(6 * k)) * (13591409 + 545140134 * k)
        deno = factorial(3 * k) * (factorial(k) ** 3) * (640320 ** (3 * k))
        pi += Decimal(t) / Decimal(deno)
    pi = pi * Decimal(12) / Decimal(640320 ** Decimal(1.5)))
    pi = 1 / pi
    return pi

def par_job():
    n_cores = 3
    jobs = [1200, 1700, 900, 1400, 800, 600, 500]
    results = Parallel(n_jobs=n_cores)(delayed(calc)(n, p, len(jobs)) for p, n in enumerate(jobs))

if __name__ == "__main__":
    par_job()
