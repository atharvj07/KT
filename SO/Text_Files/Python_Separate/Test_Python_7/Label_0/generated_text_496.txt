from Crypto.PublicKey import RSA
from Crypto.Cipher import PKCS1_v1_5
from Crypto.Util.Padding import pad
import socket

ip = 'server_ip'
port = 12345
modulus = b'...'  # server modulus
exponent = b'...'  # server exponent
string1 = 'data1'
string2 = 'data2'

pubkey = RSA.construct((
    int.from_bytes(modulus, byteorder='big'),
    int.from_bytes(exponent, byteorder='big')
))

cipher = PKCS1_v1_5.new(pubkey)

data = (string1 + string2).encode()
encrypted_data = cipher.encrypt(data)

client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
client.connect((ip, port))

with client.makefile('wb') as stream:
    # Send auth command
    stream.write(f"auth\n".encode())
    # Send encrypted data
    stream.write(encrypted_data)
    stream.flush()

client.close()
