import numpy as np

def nrng():
    nums = np.arange(1, 10.5, 0.5)
    i = 0
    while True:
        yield nums[i % len(nums)]
        i += 1

rng = lambda : np.random.randint(2,20)//2
n_rng = nrng()

for j in range(10):
    print('random number', rng())
    print('non-random number', next(n_rng))
