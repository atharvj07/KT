import numpy as np
from time import time

def cmp(do_cache: bool, arr_sz: int, n_arr: int, n_iter: int):
    arrs = {i: np.random.rand(arr_sz) for i in range(n_arr)}

    if do_cache:
        cache = {}

    for _ in range(n_iter):
        i, j = np.random.randint(n_arr, size=2)

        if do_cache and (i, j) in cache:
            _ = cache[(i, j)]
        else:
            _ = arrs[i] + arrs[j]
            if do_cache:
                cache[(i, j)] = _

%timeit cmp(do_cache=False, arr_sz=10000, n_arr=100, n_iter=1000)
%timeit cmp(do_cache=True, arr_sz=10000, n_arr=100, n_iter=1000)
