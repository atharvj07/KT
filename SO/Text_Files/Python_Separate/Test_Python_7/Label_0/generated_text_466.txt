class BST:
    def __init__(self, val):
        self.value = val
        self.left = None
        self.right = None

    def add(self, val):
        if self.value:
            if val < self.value:
                if self.left == None:
                    self.left = BST(val)
                else:
                    self.left.add(val)
            else:
                if val > self.value:
                    if self.right == None:
                        self.right = BST(val)
                    else:
                        self.right.add(val)
        else:
            self.value = val

    def traverse_pre(self, result=None):
        if result is None:
            result = []
        result.append(self.value)
        if self.left:
            self.left.traverse_pre(result)
        if self.right:
            self.right.traverse_pre(result)
        return result


tree = BST(5)
tree.add(10)
tree.add(8)
tree.add(2)
tree.add(4)
tree.add(7)

print(tree.traverse_pre())
