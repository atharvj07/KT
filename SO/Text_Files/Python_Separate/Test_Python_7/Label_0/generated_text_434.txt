from typing import Optional
from pydantic import BaseSettings, Field

class Sub(BaseSettings):
    sub_f: Optional[str] = Field(None, env='SUB_FIELD')

    class Config:
        env_file = 'local.env'

class Set(BaseSettings):
    f_one: Optional[str] = Field(None, env='FIELD_ONE')
    f_two: Optional[int] = Field(None, env='FIELD_TWO')
    sub: Sub = Sub()

    class Config:
        env_file = 'local.env'

s = Set()
