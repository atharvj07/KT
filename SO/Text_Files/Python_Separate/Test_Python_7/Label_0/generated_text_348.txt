from collections import deque

class Sub:
    def __init__(self, prev=None, s=0, e=0, diff=0):
        self.prev = prev
        self.s = s
        self.e = e
        self.diff = diff

def part(nums, mn, mx):
    n = len(nums)
    dp = [None] * (n + 1)
    dp[0] = Sub()

    for i in range(1, n + 1):
        for j in range(mn, mx + 1):
            if i - j >= 0:
                sub = dp[i - j]
                if sub:
                    diff = max(nums[i - j:i]) - min(nums[i - j:i])
                    new_diff = sub.diff + diff
                    if dp[i] is None or new_diff > dp[i].diff:
                        dp[i] = Sub(sub, i - j, i - 1, new_diff)

    return dp[n]
