import pandas as pd

def glmp(df):
    print(f"Rows: {len(df)}")
    print(f"Columns: {len(df.columns)}")
    for col in df.columns:
        dtype = df[col].dtype
        head = df[col].head().tolist()
        print(f"$ {col} <{dtype}> {head}")

test = pd.DataFrame({'column_one':['A', 'B', 'C', 'D'],
                     'column_two':[1, 2, 3, 4]})

glmp(test)
