def get_user(db: Session, uid: int):
    return db.query(models.User).filter(models.User.id == uid).first()

def create_user_item(db: Session, itm: schemas.ItemCreate, uid: int):
    db_itm = models.Item(**itm.dict(), owner_id=uid)
    db.add(db_itm)
    db.commit()
    db.refresh(db_itm)
    return schemas.Item.from_orm(db_itm)
