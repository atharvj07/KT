import re
from io import StringIO
from itertools import batched

file = StringIO('''a\t  b c\td _ e
a   b c\td _ e
   a   b c\td _ e
a   b_c\td _ e\t
abcd''')

for line in file:
    print(list(batched(re.split(r"([\s_]+)", line.rstrip('\r\n')) + [''], 2)))

[('a', '\t  '), ('b', ' '), ('c', '\t'), ('d', ' _ '), ('e', '')]
[('a', '   '), ('b', ' '), ('c', '\t'), ('d', ' _ '), ('e', '')]
[('', '   '), ('a', '   '), ('b', ' '), ('c', '\t'), ('d', ' _ '), ('e', '')]
[('a', '   '), ('b', '_'), ('c', '\t'), ('d', ' _ '), ('e', '\t'), ('', '')]
[('abcd', '')]
