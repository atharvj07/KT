from pyspark.sql import SparkSession
from pyspark.sql import functions as F
from pyspark.sql.window import Window

# Initialize Spark session
spark = SparkSession.builder.master("local").appName("Window Example").getOrCreate()

# Sample data
data = [(1, "A", 10), (2, "B", 20), (3, "C", 30), (4, "D", 40), (5, "E", 50)]
columns = ["id", "category", "value"]

# Create DataFrame
df = spark.createDataFrame(data, columns)

# Window specification example 1: Rows between unbounded preceding and current row
window1 = Window.orderBy("id").rowsBetween(Window.unboundedPreceding, Window.currentRow)

# Window specification example 2: Rows between current row and unbounded following
window2 = Window.orderBy("id").rowsBetween(Window.currentRow, Window.unboundedFollowing)

# Window specification example 3: Rows between unbounded preceding and unbounded following
window3 = Window.orderBy("id").rowsBetween(Window.unboundedPreceding, Window.unboundedFollowing)

# Window operation examples

# Calculate cumulative sum up to the current row (Window 1)
df_with_cumulative_sum = df.withColumn("cumulative_sum", F.sum("value").over(window1))

# Calculate the sum of values from current row to the end (Window 2)
df_with_forward_sum = df.withColumn("forward_sum", F.sum("value").over(window2))

# Calculate the sum of all rows (Window 3)
df_with_full_sum = df.withColumn("full_sum", F.sum("value").over(window3))

# Show results
df_with_cumulative_sum.show()
df_with_forward_sum.show()
df_with_full_sum.show()
