import polars as pl
from datetime import datetime
import pytz

def convert_timezone_dynamic():
    df = pl.DataFrame({
        "time": [datetime(2023, 4, 3, 2), datetime(2023, 4, 4, 3), datetime(2023, 4, 5, 4)],
        "tzone": ["Asia/Tokyo", "America/Chicago", "Europe/Paris"]
    }).with_columns(pl.col("time").dt.replace_time_zone("UTC"))

    df = df.with_columns(
        pl.struct(["time", "tzone"]).apply(
            lambda x: x["time"].convert_time_zone(x["tzone"]).hour
        ).alias("local_hour")
    )

    return df

convert_timezone_dynamic()
