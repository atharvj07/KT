import numpy as np
import cv2
import matplotlib.pyplot as plt
from PIL import Image

def generate_speckle_pattern(dpi=1000, widthOfSampleMM=50, heightOfSampleMM=50, patternSizeMM=0.1, density=0.75, variation=0.75, output_file="speckle_pattern.png"):
    dpmm = 0.03937 * dpi
    widthOfSamplesPX = int(np.ceil(widthOfSampleMM * dpmm))
    widthOfSamplesPX = widthOfSamplesPX + 10 - widthOfSamplesPX % 10
    heightOfSamplePX = int(np.ceil(heightOfSampleMM * dpmm))
    heightOfSamplePX = heightOfSamplePX + 10 - heightOfSamplePX % 10
    patternSizePX = patternSizeMM * dpmm
    
    im = 255 * np.ones((heightOfSamplePX, widthOfSamplesPX), dtype=np.uint8)
    
    numPoints = int(density * heightOfSamplePX / patternSizePX)
    horizontal = [heightOfSamplePX // 2] if numPoints == 1 else [int(i * heightOfSamplePX / (numPoints + 1)) for i in range(1, numPoints + 1)]
    
    numPoints = int(density * widthOfSamplesPX / patternSizePX)
    vertical = [widthOfSamplesPX // 2] if numPoints == 1 else [int(i * widthOfSamplesPX / (numPoints + 1)) for i in range(1, numPoints + 1)]
    
    for i in vertical:
        for j in horizontal:
            iWithNoise = i + variation * np.random.randint(-2 * patternSizePX / density, +2 * patternSizePX / density)
            jWithNoise = j + variation * np.random.randint(-2 * patternSizePX / density, +2 * patternSizePX / density)
            patternSizePXWithNoise = patternSizePX + patternSizePX * variation * (np.random.rand() - 0.5) / 2
            cv2.circle(im, (int(iWithNoise), int(jWithNoise)), int(patternSizePXWithNoise // 2), 0, -1)
    
    img = Image.fromarray(im)
    img.save(output_file, dpi=(dpi, dpi))

generate_speckle_pattern()
