def fizzbuzz(n):
    arr = []
    for i in range(1, n + 1):
        if (three := (i % 3 == 0)) | (five := (i % 5 == 0)):
            arr.append("FizzBuzz" if three and five else "Fizz" if three else "Buzz")
        else:
            arr.append(str(i))
    return arr
