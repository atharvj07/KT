import openai

def upload_pdf_for_rag(pdf_path, api_key):
    client = openai.OpenAI(api_key=api_key)
    file = client.files.create(file=open(pdf_path, "rb"), purpose="assistants")
    assistant = client.beta.assistants.create(
        instructions="Use the uploaded PDF for answering questions accurately.",
        model="gpt-4-turbo",
        tools=[{"type": "retrieval"}],
        file_ids=[file.id]
    )
    return assistant.id

# Example usage
api_key = "your_openai_api_key"
pdf_path = "example.pdf"
assistant_id = upload_pdf_for_rag(pdf_path, api_key)
print("Assistant ID:", assistant_id)
