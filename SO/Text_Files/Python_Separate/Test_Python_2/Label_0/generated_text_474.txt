import polars as pl
import duckdb
import sqlalchemy

def write_polars_to_duckdb(df, db_path, table_name):
    engine = sqlalchemy.create_engine(f"duckdb:///{db_path}")
    df.write_database(table_name=table_name, connection=engine, engine="sqlalchemy")

pldf = pl.DataFrame({'mynum': [1,2,3,4]})
write_polars_to_duckdb(pldf, "scratch.db", "test_table")
