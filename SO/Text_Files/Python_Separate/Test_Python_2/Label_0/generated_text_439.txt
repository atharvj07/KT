import requests
from bs4 import BeautifulSoup
import cloudscraper

def scrape_etherscan_tokens():
    scraper = cloudscraper.create_scraper()  
    url = "https://etherscan.io/tokens"
    
    response = scraper.get(url)
    if response.status_code != 200:
        raise Exception(f"Failed to fetch page, status code: {response.status_code}")

    soup = BeautifulSoup(response.text, "html.parser")
    
    tokens = []
    for row in soup.select("table tbody tr"):
        columns = row.find_all("td")
        if len(columns) >= 2:
            token_name = columns[0].get_text(strip=True)
            token_link = columns[0].find("a")["href"] if columns[0].find("a") else None
            tokens.append({"name": token_name, "link": f"https://etherscan.io{token_link}" if token_link else None})

    return tokens

tokens_data = scrape_etherscan_tokens()
print(tokens_data)
