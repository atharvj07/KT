import sys
import subprocess

def fix_langchain_import():
    python_executable = sys.executable
    try:
        subprocess.check_call([python_executable, "-m", "pip", "install", "--upgrade", "langchain"])
    except subprocess.CalledProcessError:
        print("Failed to install langchain. Please check your environment.")
    else:
        print("Langchain installed successfully.")

    print(f"Using Python executable: {python_executable}")
    print(f"Python sys.path: {sys.path}")

fix_langchain_import()
