import pandas as pd
import json

def analyze_json_parsing():
    data_records = [
        {"device": "rtr1", "dc": "London", "vendor": "Cisco"},
        {"device": "rtr2", "dc": "London", "vendor": "Cisco"},
        {"device": "rtr3", "dc": "London", "vendor": "Cisco"},
    ]

    data_index = {
        "rtr1": {"dc": "London", "vendor": "Cisco"},
        "rtr2": {"dc": "London", "vendor": "Cisco"},
        "rtr3": {"dc": "London", "vendor": "Cisco"},
    }

    df1 = pd.read_json(json.dumps(data_records))
    df2 = pd.read_json(json.dumps(data_index))
    df3 = pd.read_json(json.dumps(data_index), orient="index")

    return df1, df2, df3
