import polars as pl

def reshape_polars_dataframe(df):
    return df.melt(id_vars=[], variable_name="col").with_columns(
        pl.col("col").str.extract(r"([a-z])_").alias("group")
    ).pivot(values="value", index="row_nr", columns="group").drop("row_nr")

# Example usage
df = pl.DataFrame({
    "a_0": [1, 4], "a_1": [2, 5], "a_2": [3, 6],
    "b_0": [7, 10], "b_1": [8, 11], "b_2": [9, 12]
})
reshape_polars_dataframe(df)
