import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import matplotlib.patches as patches
from scipy.spatial import Voronoi, voronoi_plot_2d
from shapely.geometry import Polygon, MultiPolygon, Point

def plot_phase_diagram(df):
    points = df[['x', 'y']].values
    labels = df['label'].values
    colors = {label: color for label, color in zip(df['label'], df['color'])}

    vor = Voronoi(points)

    fig, ax = plt.subplots(figsize=(8, 8))
    ax.set_xlim(df['x'].min(), df['x'].max())
    ax.set_ylim(df['y'].min(), df['y'].max())
    ax.set_xlabel('Log pO2(g)')
    ax.set_ylabel('Log pSO2(g)')

    for region in vor.regions:
        if not region or -1 in region:
            continue
        polygon = [vor.vertices[i] for i in region]
        centroid = np.mean(polygon, axis=0)
        point_label = labels[np.argmin(np.linalg.norm(points - centroid, axis=1))]
        poly_patch = patches.Polygon(polygon, closed=True, facecolor=colors[point_label], edgecolor='black', alpha=0.5)
        ax.add_patch(poly_patch)

    plt.show()
