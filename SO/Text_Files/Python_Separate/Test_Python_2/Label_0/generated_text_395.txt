import importlib
import sys
from pathlib import Path

def run_entry_point(script_name: str):
    sys.path.insert(0, str(Path(__file__).parent))
    module_name = f"run.{script_name}"
    try:
        module = importlib.import_module(module_name)
        if hasattr(module, "main"):
            module.main()
        else:
            raise AttributeError(f"Module '{module_name}' has no 'main' function")
    except ModuleNotFoundError:
        print(f"Script '{script_name}' not found.")

if __name__ == "__main__":
    import argparse

    parser = argparse.ArgumentParser(description="Run an entry point script.")
    parser.add_argument("script", help="The script to run (e.g., 'main1')")
    args = parser.parse_args()

    run_entry_point(args.script)
