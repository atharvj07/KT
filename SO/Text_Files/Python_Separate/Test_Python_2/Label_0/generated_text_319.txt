import os
import asyncio
from tqdm import tqdm
from PIL import Image
import google.generativeai as genai

# Gemini Model
model = genai.GenerativeModel('gemini-pro-vision')

async def process_image(image_path, prompt):
    img = Image.open(image_path)
    output = await asyncio.to_thread(model.generate_content, [prompt, img])
    return output.text

async def process_images_in_batches(image_paths, batch_size, prompt):
    results = []
    for i in tqdm(range(0, len(image_paths), batch_size)):
        batch = image_paths[i:i + batch_size]
        tasks = [process_image(image, prompt) for image in batch]
        batch_results = await asyncio.gather(*tasks)
        results.extend(batch_results)
        await asyncio.sleep(1)  # Rate limit compliance
    return results

async def main():
    image_dir = "path_to_images"
    images_to_process = [os.path.join(image_dir, image_name) for image_name in os.listdir(image_dir)]
    prompt = """Carefully scan this images: if it has text, extract all the text and return the text from it. If the image does not have text return '<000>'."""
    batch_size = 10  # Ensuring we stay within the API limit

    results = await process_images_in_batches(images_to_process, batch_size, prompt)
    
    for result in results:
        print(result)

asyncio.run(main())
