import polars as pl

def idxmax_polars(df: pl.DataFrame, cols: list[str]) -> pl.DataFrame:
    return df.with_columns(
        max_col=pl.struct(cols)
        .map_elements(lambda row: max(row, key=lambda k: row[k]) if any(v is not None for v in row.values()) else None)
    )

df = pl.DataFrame({"a": [1, 8, 3], "b": [4, 5, None]})
result = idxmax_polars(df, ["a", "b"])
print(result)
