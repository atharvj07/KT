import pandas as pd

def sum_filtered_columns(end_df):
    mask = end_df.columns.get_level_values(1) == 'Internet'
    cols_to_sum = end_df.loc[:, mask].drop(columns='site', errors='ignore')
    end_df = end_df.copy()
    end_df.loc[:, 'Soma Internet'] = cols_to_sum.sum(axis=1)
    return end_df
