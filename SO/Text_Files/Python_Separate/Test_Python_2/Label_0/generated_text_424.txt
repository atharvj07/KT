install_tensorflow_rstudio <- function() {
  library(reticulate)
  
  # Create a new conda environment
  system("conda create --name py38 python=3.8.0 -y")
  system("conda activate py38 && conda install tensorflow=2.4 -y")

  # Use the new environment in R
  use_condaenv("py38", required = TRUE)
  use_python("/root/.conda/envs/py38/bin/python3.8", required = TRUE)

  # Set LD_LIBRARY_PATH to ensure correct libraries are used
  Sys.setenv("LD_LIBRARY_PATH" = "/root/.conda/envs/py38/lib")

  # Verify installation
  py_available(initialize = TRUE)
  tf <- import("tensorflow")
  
  return(tf$__version__)
}

install_tensorflow_rstudio()
