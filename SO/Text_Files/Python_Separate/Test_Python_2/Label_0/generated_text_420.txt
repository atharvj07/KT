import click

@click.command()
@click.option('--value', default=1, show_default=True, help='a value.', is_eager=True)
def hello(value):
    ctx = click.get_current_context()
    value_from_user = '--value' in ctx.args
    print(f"Value: {value}, From User: {value_from_user}")

if __name__ == "__main__":
    hello()
