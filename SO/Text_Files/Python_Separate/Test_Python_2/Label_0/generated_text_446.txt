import polars as pl
import numpy as np
import functools

def efficient_sparse_dataframe(N, C):
    pls = [pl.DataFrame({"index": np.linspace(i, N*3-i, num=N, dtype=np.int32), f"A{i}": np.arange(N, dtype=np.float32)}) for i in range(C)]
    return functools.reduce(lambda a, b: a.join(b, on='index', how='outer'), pls).sort(by='index')

N, C = 300000, 20
F = efficient_sparse_dataframe(N, C)
print(F)
