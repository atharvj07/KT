from llama_cpp import Llama

def run_gguf_inference(model_path: str, prompt: str, max_tokens: int = 256, temperature: float = 0.7, n_threads: int = 4) -> str:
    llm = Llama(model_path=model_path, n_threads=n_threads)
    response = llm(
        prompt,
        max_tokens=max_tokens,
        temperature=temperature,
        echo=True
    )
    return response["choices"][0]["text"]

if __name__ == '__main__':
    model_file = "path/to/your/model.gguf"
    user_prompt = "Explain the theory of relativity."
    result = run_gguf_inference(model_file, user_prompt)
    print(result)
