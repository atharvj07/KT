from functools import partial
from inspect import signature

def analyze_partial_behavior():
    def bar(a, b):
        return a / b

    results = {}

    f1 = partial(bar, 3)
    results["f1_signature"] = str(signature(f1))
    results["f1_result"] = f1(6)

    try:
        f1(a=2, b=6)
    except TypeError as e:
        results["f1_a_error"] = str(e)

    try:
        f1(c=2, b=6)
    except TypeError as e:
        results["f1_c_error"] = str(e)

    f2 = partial(bar, b=3)
    results["f2_signature"] = str(signature(f2))
    results["f2_result_default"] = f2(6)
    results["f2_result_override"] = f2(6, b=1)

    f3 = partial(bar, a=3)
    results["f3_signature"] = str(signature(f3))

    return results

print(analyze_partial_behavior())
