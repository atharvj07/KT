import pandas as pd

def transform_dataframe(df):
    df_filtered = df.filter(regex=r'TP\d|VReal\d|TiV\d')
    df_melted = df_filtered.melt(var_name="Category", value_name="Value")
    df_melted["ID"] = df_melted["Category"].str.extract(r'(\d)').astype(int)
    df_melted["Column"] = df_melted["Category"].str.extract(r'([A-Za-z]+)')
    
    df_pivot = df_melted.pivot(index="ID", columns="Column", values="Value")
    df_pivot = df_pivot.rename(columns={"TP": "Price", "VReal": "Net", "TiV": "Range"}).reset_index()
    
    return df_pivot[["ID", "Price", "Net", "Range"]]

# Example usage:
df = pd.DataFrame({
    "Num": ["AA-24"], "TP1(USD)": [0], "TP2(USD)": [700], "TP3(USD)": [2100],
    "VReal1(USD)": [300], "VReal2(USD)": [1159], "VReal3(USD)": [2877],
    "TiV1 (EUR)": [30], "TiV2 (EUR)": [30], "TiV3 (EUR)": [47], "TR": [10], "TR-Tag": [5]
})

transformed_df = transform_dataframe(df)
