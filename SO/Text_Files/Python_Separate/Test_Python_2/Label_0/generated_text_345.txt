import itertools

def generate_combinations(letters, k, n):
    all_combinations = set()
    
    for groups in itertools.combinations(itertools.combinations(letters, n), k):
        sorted_groups = tuple(sorted(''.join(group) for group in groups))
        all_combinations.add(sorted_groups)
    
    return [list(group) for group in all_combinations]

letters = ['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H', 'I', 'J', 'K', 'L']
result = generate_combinations(letters, 4, 3)
print(result)
