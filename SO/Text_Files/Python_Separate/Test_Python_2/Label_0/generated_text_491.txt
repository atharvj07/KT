import polars as pl

def transform_fqid_counts(df: pl.DataFrame) -> pl.DataFrame:
    return (
        df.group_by("session_id")
        .agg(pl.col("fqid").value_counts(sort=True))
        .explode("fqid")
        .unnest("fqid")
        .pivot(values="counts", index="session_id", columns="fqid")
        .fill_null(0)
        .rename({col: f"fqid_{col}" for col in df["fqid"].unique().to_list()})
    )
