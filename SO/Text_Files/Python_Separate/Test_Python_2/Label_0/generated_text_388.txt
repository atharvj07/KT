import inspect
import re

def get_caller_variable():
    frame = inspect.currentframe().f_back
    code_context = inspect.getsource(frame)
    caller_line = code_context.splitlines()[frame.f_lineno - 1]
    match = re.match(r"(\w+)\s*=\s*fct1\(\)", caller_line)
    return match.group(1) if match else None

def fct1():
    return {"name": get_caller_variable()}

a = fct1()
