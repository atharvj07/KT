import numpy as np

def max_partition_sum_diff(nums, a, b):
    n = len(nums)
    dp = [-float('inf')] * (n + 1)
    dp[0] = 0
    prev_index = [-1] * (n + 1)

    for i in range(1, n + 1):
        for j in range(i - b, i - a + 1):
            if j >= 0:
                subarray = nums[j:i]
                diff = max(subarray) - min(subarray)
                if dp[j] + diff > dp[i]:
                    dp[i] = dp[j] + diff
                    prev_index[i] = j

    if dp[n] == -float('inf'):
        return []

    partitions = []
    i = n
    while i > 0:
        start = prev_index[i]
        partitions.append(nums[start:i])
        i = start

    partitions.reverse()
    return partitions

numbers = [5, 8, 4, 5, 1, 3, 5, 1, 3, 1]
a = 3
b = 7
result = max_partition_sum_diff(numbers, a, b)
print(result)
