import torch
from transformers import LlamaForCausalLM, LlamaTokenizer, LlamaConfig

def load_llama_model(model_path):
    tokenizer = LlamaTokenizer.from_pretrained(model_path)
    
    config = LlamaConfig(
        hidden_size=4096,
        num_hidden_layers=32,
        num_attention_heads=32,
        intermediate_size=int(4096 * 1.3),
        vocab_size=128256,
        use_scaled_rope=True
    )

    model = LlamaForCausalLM(config=config)
    model.load_state_dict(torch.load(f"{model_path}/consolidated.00.pth", map_location=torch.device("cpu")))
    model.eval()

    return model, tokenizer

def generate_text(model, tokenizer, input_text):
    inputs = tokenizer(input_text, return_tensors="pt")
    outputs = model.generate(inputs["input_ids"])
    return tokenizer.decode(outputs[0], skip_special_tokens=True)
