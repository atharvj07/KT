import polars as pl

def truncate_explode(df, cols):
    df = df.with_columns([
        pl.struct(cols).map_elements(lambda row: list(zip(*row))).alias("zipped")
    ])
    return df.explode("zipped").unnest("zipped")

# Example usage:
data = {
    "a": [[1], [2], [3, 4], [5, 6, 7]],
    "b": [[], [8], [9, 10], [11, 12]],
}
df = pl.DataFrame(data)

result = truncate_explode(df, ["a", "b"])
