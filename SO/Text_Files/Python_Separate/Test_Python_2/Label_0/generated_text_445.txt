import pandas as pd

def remove_duplicate_pairs(df):
    df["_pair"] = df.apply(lambda row: tuple(sorted([row["person1"], row["person2"]])), axis=1)
    finaldf = df.drop_duplicates(subset=["_pair", "messages"]).drop(columns=["_pair"])
    return finaldf

df = pd.DataFrame({
    "": [0,1,2,3,4,5,6],
    "person1": ["ryan", "delta", "delta", "delta","bravo","alpha","ryan"], 
    "person2": ["delta", "ryan", "alpha", "bravo","delta","ryan","alpha"], 
    "messages": [1, 1, 2, 3,3,9,9]
})

finaldf = remove_duplicate_pairs(df)
print(finaldf)
