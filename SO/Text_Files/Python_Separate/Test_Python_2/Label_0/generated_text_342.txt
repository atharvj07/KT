class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y
        self._tuple = (x, y)

    @property
    def tuple(self):
        return self._tuple

    def __repr__(self):
        return f"Point({self.x}, {self.y})"

    def __str__(self):
        return f"({self.x}, {self.y})"

    def __getattr__(self, name):
        if name == 'tuple':
            return self._tuple
        raise AttributeError(f"'{type(self).__name__}' object has no attribute '{name}'")
