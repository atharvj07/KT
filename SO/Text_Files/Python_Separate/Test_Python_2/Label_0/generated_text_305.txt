import re
from itertools import zip_longest

def split_with_delimiters(text, delimiter_pattern=r"[\s_]+"):
    split_result = re.split(f"({delimiter_pattern})", text)
    return list(zip_longest(*[iter(split_result)]*2, fillvalue=""))

def process_file(filename, delimiter_pattern=r"[\s_]+"):
    with open(filename, "r", encoding="utf-8") as f:
        return [split_with_delimiters(line.rstrip("\r\n"), delimiter_pattern) for line in f]

# Example usage
filename = "f.txt"
result = process_file(filename)
for line in result:
    print(line)
