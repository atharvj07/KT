def compare_slices(sequence, start, stop):
    slice_none = sequence[slice(start, stop, None)]
    slice_one = sequence[slice(start, stop, 1)]
    return slice_none == slice_one

# Example usage
sequence = list(range(10))
test_cases = [(0, 5), (2, 8), (None, None), (-3, None), (None, -2)]
results = {case: compare_slices(sequence, *case) for case in test_cases}
print(results)
