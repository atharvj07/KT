import pandas as pd

def compute_float_price(df):
    fprice = df.groupby(['StartDate', 'Commodity', 'DealType'], group_keys=False).apply(
        lambda group: -(group['MTMValue'].sum() - (group['FixedPriceStrike'] * group['Quantity']).sum()) / group['Quantity'].sum()
    ).reset_index(name='FloatPrice')
    
    return df.merge(fprice, on=['StartDate', 'Commodity', 'DealType'], how='left')

# Sample usage
data = {
    'TradeID': ['aaa', 'bbb', 'ccc'],
    'TradeDate': ['01/01/2024'] * 3,
    'Commodity': ['(com1,com2)'] * 3,
    'StartDate': ['01/01/2024'] * 3,
    'ExpiryDate': ['01/01/2024'] * 3,
    'FixedPriceStrike': [10] * 3,
    'Quantity': [10] * 3,
    'MTMValue': [100.00] * 3
}

df = pd.DataFrame(data)
df = compute_float_price(df)
print(df)
