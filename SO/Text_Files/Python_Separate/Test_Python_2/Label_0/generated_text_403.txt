import pandas as pd
import numpy as np

def unnest_route_set(df):
    route_array = np.array(df["Route_set"].iloc[0])
    df_out = pd.DataFrame(route_array, columns=[f"route{i+1}" for i in range(route_array.shape[0])])
    return df_out
