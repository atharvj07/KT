import pandas as pd

def transform_dataframe(A):
    A = A.melt(id_vars=["FIELD_A"], var_name="FIELD_NAME", value_name="VALUE")
    A["FIELD_NAME"] = A["FIELD_NAME"].str[-1]  # Extracting last character
    A["FIELD_STR_VALUE"] = A["VALUE"].where(A["VALUE"].apply(lambda x: isinstance(x, str)), None)
    A["FIELD_NUM_VALUE"] = A["VALUE"].where(A["VALUE"].apply(lambda x: isinstance(x, (int, float))), None)
    A.drop(columns=["VALUE"], inplace=True)
    A.insert(0, "ID", range(1, len(A) + 1))
    return A

A = pd.DataFrame({
    "FIELD_A": [123123, 123124, 123144],
    "FIELD_B": [8, 7, 99],
    "FIELD_C": ["a", "c", "x"],
    "FIELD_D": [23423, 6464, 234]
})

B = transform_dataframe(A)
print(B)
