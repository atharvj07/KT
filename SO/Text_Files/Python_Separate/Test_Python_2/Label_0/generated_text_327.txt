import pandas as pd

def calculate_hit_low_high(df):
    bins = [-float('inf'), -0.25, 0.25, float('inf')]
    labels = ['Low', 'Hit', 'High']
    df['Category'] = pd.cut(df['Value'], bins=bins, labels=labels, right=True)
    return df.groupby('Type')['Category'].value_counts().unstack(fill_value=0).reset_index()

df = pd.DataFrame({"Type":["RF", "RF", "RF", "MLP", "MLP", "MLP"], "Value":[-1.5,-0.1,1.7,0.2,-0.7,-0.6]})
result = calculate_hit_low_high(df)
