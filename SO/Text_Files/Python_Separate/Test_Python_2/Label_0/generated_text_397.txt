import polars as pl

def concatenate_grouped_strings():
    df = pl.DataFrame(
        {
            "col1": ["a", "b", "a", "b", "c"],
            "col2": ["val1", "val2", "val1", "val3", "val3"]
        }
    )
    
    result = df.group_by("col1").agg(
        pl.col("col2").str.concat(",")
    )
    
    return result

concatenated_df = concatenate_grouped_strings()
