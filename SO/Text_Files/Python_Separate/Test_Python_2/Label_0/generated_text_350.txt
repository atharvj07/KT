import geopandas as gpd
import requests
import zipfile
import os

def load_world_map():
    url = "https://www.naturalearthdata.com/downloads/110m-cultural-vectors/110m_admin_0_countries.zip"
    zip_path = "world_map.zip"
    extract_dir = "world_map"

    # Download the zip file
    response = requests.get(url)
    with open(zip_path, 'wb') as f:
        f.write(response.content)

    # Extract the zip file
    with zipfile.ZipFile(zip_path, 'r') as zip_ref:
        zip_ref.extractall(extract_dir)

    # Load the shapefile using geopandas
    shapefile_path = os.path.join(extract_dir, "ne_110m_admin_0_countries.shp")
    world = gpd.read_file(shapefile_path)

    # Clean up the downloaded zip file
    os.remove(zip_path)

    return world

world = load_world_map()
print(world.head())
