class FooMeta(type):
    def __len__(cls):
        return 9000

class GoodBar(metaclass=FooMeta):
    def __len__(self):
        return 9001

class BadBar(metaclass=FooMeta):
    @classmethod
    def __len__(cls):
        return 9002

BadBar.__class__.__len__ = lambda cls: cls.__len__()

print(len(BadBar))  # 9002
print(len(BadBar()))  # 9002
