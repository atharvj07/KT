def demonstrate_format_behavior():
    results = {}
    try:
        results["index"] = '{x[1]}'.format(x="asd")
    except Exception as e:
        results["index"] = str(e)
    
    try:
        results["slice"] = '{x[1:3]}'.format(x="asd")
    except Exception as e:
        results["slice"] = str(e)
    
    return results

print(demonstrate_format_behavior())
