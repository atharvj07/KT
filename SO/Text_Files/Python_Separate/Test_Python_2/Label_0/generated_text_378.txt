import pandas as pd
from sklearn import datasets

def mutate_keep_unused(df, new_col_name, expr):
    temp_df = df.copy()
    temp_df[new_col_name] = expr(temp_df)
    used_columns = list(expr.__code__.co_names)
    existing_columns = set(df.columns)
    drop_columns = [col for col in used_columns if col in existing_columns]
    return temp_df.drop(columns=drop_columns)

iris = pd.DataFrame(datasets.load_iris().data, columns=datasets.load_iris().feature_names)

result = mutate_keep_unused(
    iris, 
    "new_col", 
    lambda x: x['sepal length (cm)'] + x['petal length (cm)'] * x['petal width (cm)']
)

print(result.head())
