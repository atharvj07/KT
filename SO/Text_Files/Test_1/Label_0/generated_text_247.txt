@Mapper
public interface MyMapper {

    @Mapping(target = ".", source = "wealthProdAccountInfoDTO", qualifiedByName = "nullIfAllFieldsAreNull")
    AgentInfo wealthProdAccountInfoDTOToAgentInfo(WealthProdAccountInfoDTO wealthProdAccountInfoDTO);

    @Named("nullIfAllFieldsAreNull")
    default AgentInfo mapToAgentInfo(WealthProdAccountInfoDTO dto) {
        if (dto == null) {
            return null;
        }
        if (dto.getAgentName() == null &&
            dto.getAgentIdentityType() == null &&
            dto.getAgentIdentityNo() == null &&
            dto.getAgentIdentityExpireAt() == null) {
            return null;
        }
        return new AgentInfo(
            dto.getAgentName(),
            dto.getAgentIdentityType(),
            dto.getAgentIdentityNo(),
            dto.getAgentIdentityExpireAt()
        );
    }
}

java
@Test
void testMappingWhenAllFieldsAreNull() {
    WealthProdAccountInfoDTO dto = new WealthProdAccountInfoDTO();
    AgentInfo agentInfo = myMapper.wealthProdAccountInfoDTOToAgentInfo(dto);
    assertNull(agentInfo);
}

@Test
void testMappingWhenSomeFieldsAreNonNull() {
    WealthProdAccountInfoDTO dto = new WealthProdAccountInfoDTO();
    dto.setAgentName("John");
    AgentInfo agentInfo = myMapper.wealthProdAccountInfoDTOToAgentInfo(dto);
    assertNotNull(agentInfo);
    assertEquals("John", agentInfo.getAgentName());
}
