# Step 1: Install the required system dependencies
sudo yum install gcc-c++ libstdc++-devel

# Step 2: Create and activate the Conda environment
conda create --name py38 python=3.8.0
conda activate py38
conda install tensorflow=2.4

# Step 3: Install R packages
devtools::install_github("rstudio/reticulate")
devtools::install_github("rstudio/keras")

# Step 4: Set up reticulate in RStudio
library(reticulate)
reticulate::use_condaenv("/root/.conda/envs/py38", required = TRUE)
reticulate::use_python("/root/.conda/envs/py38/bin/python3.8", required = TRUE)

# Step 5: Set the library path in RStudio to point to the correct libstdc++
Sys.setenv("LD_LIBRARY_PATH" = "/root/.conda/envs/py38/lib:$LD_LIBRARY_PATH")

# Step 6: Test TensorFlow setup
library(tensorflow)
tensorflow::tf_config()
