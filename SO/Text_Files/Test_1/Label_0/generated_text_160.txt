import javax.xml.bind.JAXBElement;
import javax.xml.ws.BindingProvider;
import javax.xml.ws.WebServiceFeature;
import javax.xml.ws.soap.MTOMFeature;
import javax.xml.ws.soap.SOAPBinding;

public class MdfeClient {

    public static void main(String[] args) throws Exception {
        ObjectFactory of = new ObjectFactory();
        String soapBody = "<soap:Envelope xmlns:soap=\"http://www.w3.org/2003/05/soap-envelope\" xmlns:mdf=\"http://www.portalfiscal.inf.br/mdfe/wsdl/MDFeRecepcaoSinc\">" +
                "<soap:Header/>" +
                "<soap:Body>" +
                "<mdf:mdfeDadosMsg>" +
                "<![CDATA[<your_soap_body_here>]]>" +
                "</mdf:mdfeDadosMsg>" +
                "</soap:Body>" +
                "</soap:Envelope>";
        
        JAXBElement<String> jaxb = of.createMdfeDadosMsg(soapBody);

        MDFeRecepcaoSinc recepcao = new MDFeRecepcaoSinc();
        MDFeRecepcaoSincSoap12 soap = recepcao.getMDFeRecepcaoSincSoap12(new MTOMFeature(true));

        // Set up the digital certificate and SSL context here (as per your certificate settings)
        BindingProvider bindingProvider = (BindingProvider) soap;
        bindingProvider.getRequestContext().put(BindingProvider.ENDPOINT_ADDRESS_PROPERTY, "https://mdfe.svrs.rs.gov.br/ws/MDFeRecepcaoSinc/MDFeRecepcaoSinc.asmx");

        MDFeRecepcaoResult result = soap.mdfeRecepcao(jaxb.getValue());

        System.out.println(result.getRetMDFe());  // Adjust this if needed to print out specific parts
    }
}
