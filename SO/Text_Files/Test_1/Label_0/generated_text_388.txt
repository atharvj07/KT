import inspect
import ast

def get_caller_line():
    frame = inspect.currentframe().f_back
    source = inspect.getsource(frame.f_code)
    lines = source.splitlines()
    caller_line = lines[frame.f_lineno - 1].strip()
    return caller_line

def fct1():
    caller_line = get_caller_line()
    var_name = caller_line.split("=")[0].strip()
    return {"name": var_name}

# Test case
a = fct1()
print(a)  # Expected output: {'name': 'a'}
