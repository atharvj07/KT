import polars as pl

df = pl.DataFrame({"x": [1.2, 1.3, 3.4, 3.5]})

df = df.with_columns(
    y=pl.int_range(0, pl.count()).map_elements(lambda i: df["x"].slice(i - 2, 3))
)

print(df)

# Test Case
def test_rolling():
    expected = [[1.2, 1.3], [1.2, 1.3, 3.4], [1.2, 1.3, 3.4, 3.5], [1.3, 3.4, 3.5]]
    assert df["y"].to_list() == expected

test_rolling()
