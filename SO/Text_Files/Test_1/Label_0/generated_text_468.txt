from http.server import SimpleHTTPRequestHandler, HTTPServer
import os

class CustomHTTPRequestHandler(SimpleHTTPRequestHandler):
    def translate_path(self, path):
        # Override to serve files from the specified directory
        return os.path.join(os.getcwd(), 'your_directory', path.lstrip('/'))

def run(server_class=HTTPServer, handler_class=CustomHTTPRequestHandler, addr='127.0.0.1', port=8000, directory='your_directory'):
    os.chdir(directory)  # Change the working directory
    server_address = (addr, port)
    httpd = server_class(server_address, handler_class)
    print(f"Serving HTTP on {addr} port {port}...")
    httpd.serve_forever()

run(addr='127.0.0.1', port=8000, directory='/path/to/your/directory')
