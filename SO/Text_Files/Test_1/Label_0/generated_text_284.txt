from dataclasses import dataclass
from typing import Callable, Dict, Type, TypeVar

T = TypeVar("T", bound=Type)

def mydecorator(cls: T) -> T:
    cls.save = lambda self: print("Saved!")
    cls.my_fields_dict = {field.name: field.type for field in cls.__dataclass_fields__.values()}  # type: ignore
    return cls

@mydecorator
@dataclass
class A:
    a: str = ""

# Test case
a = A()
a.save()
print(a.my_fields_dict)
