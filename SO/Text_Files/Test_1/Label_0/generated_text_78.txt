import java.lang.foreign.MemorySegment;
import java.lang.foreign.Arena;
import java.lang.foreign.ValueLayout;

public class LongArray {
    private final MemorySegment segment;
    private static final ValueLayout.OfLong LAYOUT = ValueLayout.JAVA_LONG;

    public LongArray(long size) {
        if (size < 0) throw new IllegalArgumentException("Size must be non-negative");
        this.segment = Arena.global().allocate(size * LAYOUT.byteSize());
    }

    public void set(long index, long value) {
        segment.set(LAYOUT, index * LAYOUT.byteSize(), value);
    }

    public long get(long index) {
        return segment.get(LAYOUT, index * LAYOUT.byteSize());
    }

    public void close() {
        segment.scope().close();
    }

    public static void main(String[] args) {
        LongArray array = new LongArray(10);

        for (long i = 0; i < 10; i++) {
            array.set(i, i * 2);
        }

        for (long i = 0; i < 10; i++) {
            System.out.println("Value at index " + i + " = " + array.get(i));
        }

        array.close();
    }
}
