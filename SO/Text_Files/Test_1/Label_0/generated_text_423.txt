from functools import partial
from inspect import signature

def bar(a, b):
    return a / b

# Test case 1: Fixing 'a' at 3
f = partial(bar, 3)
assert str(signature(f)) == '(b)'
assert f(6) == 0.5 == f(b=6)

# Test case 2: Providing an alternate value for 'a'
try:
    f(a=2, b=6)
except TypeError as e:
    assert str(e) == "bar() got multiple values for argument 'a'"

try:
    f(c=2, b=6)
except TypeError as e:
    assert str(e) == "bar() got an unexpected keyword argument 'c'"

# Test case 3: Fixing 'b' at 3
f = partial(bar, b=3)
assert str(signature(f)) == '(a, *, b=3)'
assert f(6) == 2.0 == f(6, b=3)
assert f(6, b=1) == 6.0

# Test case 4: Default argument following non-default argument
f = partial(bar, a=3)
assert str(signature(f)) == '(*, a=3, b)'
