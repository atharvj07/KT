import pandas as pd
import json

# Test with data_records
data_records = [
    {
        "device": "rtr1",
        "dc": "London",
        "vendor": "Cisco",
    },
    {
        "device": "rtr2",
        "dc": "London",
        "vendor": "Cisco",
    },
    {
        "device": "rtr3",
        "dc": "London",
        "vendor": "Cisco",
    },
]

df_records = pd.read_json(json.dumps(data_records))
print(df_records)

# Test with data_index
data_index = {
    "rtr1": {"dc": "London", "vendor": "Cisco"},
    "rtr2": {"dc": "London", "vendor": "Cisco"},
    "rtr3": {"dc": "London", "vendor": "Cisco"},
}

df_index_default = pd.read_json(json.dumps(data_index))
print(df_index_default)

df_index_orient = pd.read_json(json.dumps(data_index), orient="index")
print(df_index_orient)
