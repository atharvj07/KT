import polars as pl
import duckdb

pldf = pl.DataFrame({'mynum': [1, 2, 3, 4]})

# Write Polars DataFrame to DuckDB using DuckDB's Python API
with duckdb.connect("scratch.db") as con:
    con.execute("CREATE TABLE IF NOT EXISTS test_table (mynum INTEGER);")
    con.execute("INSERT INTO test_table SELECT * FROM ?", [pldf.to_pandas()])

# Verify data
with duckdb.connect("scratch.db") as con:
    result = con.execute("SELECT * FROM test_table").fetchall()
    print(result)
