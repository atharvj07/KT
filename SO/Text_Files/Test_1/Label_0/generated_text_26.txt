/** Superclass */
public class DemoSecurityProblem {
    public DemoSecurityProblem() {
        if (true) {
            throw new RuntimeException("Constructor exception");
        }
    }

    @Override
    protected void finalize() throws Throwable {
        System.out.println("DemoSecurityProblem finalize called");
    }
}

/** Malicious Subclass */
public class MaliciousSubClass extends DemoSecurityProblem {
    private static DemoSecurityProblem reference;

    @Override
    protected void finalize() throws Throwable {
        reference = this; // Save reference to prevent GC
        System.out.println("MaliciousSubClass finalize called");
    }

    public static void main(String[] args) {
        try {
            new MaliciousSubClass(); // Exception thrown here
        } catch (Exception e) {
            System.out.println("Exception caught: " + e.getMessage());
        }

        System.gc(); // Force garbage collection to trigger finalize()

        if (reference != null) {
            System.out.println("Exploited object resurrected!");
        } else {
            System.out.println("Object was not resurrected.");
        }
    }
}
