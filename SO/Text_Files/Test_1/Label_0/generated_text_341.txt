import polars as pl

# Define the DataFrame
df = pl.DataFrame(
    {
        "id": ["NY", "TK", "FD"],
        "eat2000": [1, None, 3],
        "eat2001": [-2, None, 4],
        "eat2002": [None, None, None],
        "eat2003": [-9, None, 8],
        "eat2004": [None, None, 8]
    }
)

# Select columns that start with "eat"
col_list = [word for word in df.columns if word.startswith("eat")]

# Apply the filter for selected columns
df_filtered = (
    df
    .with_columns(
        [
            pl.col(col).filter(~pl.col(col).is_null()).alias(col)
            for col in col_list
        ]
    )
)

# Show the resulting DataFrame
print(df_filtered)
