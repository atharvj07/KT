import pandas as pd
import numpy as np

# Using int64 to prevent overflow in int32
result = pd.Series([2**31-1], dtype='int32') + 1
print(result)

# Solution with explicit type casting and handling overflow
def safe_cast_to_int32(val):
    try:
        return np.int32(val)
    except OverflowError:
        print("OverflowError: Value exceeds int32 limits")
        return np.nan  # or handle it appropriately

# Test cases
series = pd.Series([2**31-1], dtype='int32')
print(series + 1)  # This will overflow and give incorrect result
print(safe_cast_to_int32(2**31))  # Will print OverflowError
