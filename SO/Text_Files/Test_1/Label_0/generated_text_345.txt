import itertools

def generate_combinations(letters, k, n):
    # Generate all possible combinations of n letters from the set
    all_combinations = list(itertools.combinations(letters, n))
    
    # Generate all k-sized groups without caring about order
    valid_combos = []
    for combo in itertools.combinations(all_combinations, k):
        # Flatten the selected groups and check if they are unique
        combined = sorted([''.join(sorted(group)) for group in combo])
        if len(set(combined)) == len(combined):  # Check for uniqueness
            valid_combos.append(combined)
    
    return valid_combos

# Test with x = 4, k = 2, n = 2
letters = ['A', 'B', 'C', 'D']
result = generate_combinations(letters, 2, 2)
print(result)
