# my_app.modules.box.schemas.py
from pydantic import BaseModel
from typing import List, TYPE_CHECKING

if TYPE_CHECKING:
    from my_app.modules.toy.schemas import ToyBase

class BoxBase(BaseModel):
    id: int

class BoxResponse(BoxBase):
    toys: List["ToyBase"]


# my_app.modules.toy.schemas.py
from pydantic import BaseModel
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from my_app.modules.box.schemas import BoxBase

class ToyBase(BaseModel):
    id: int

class ToyResponse(ToyBase):
    box: "BoxBase"


# FastAPI test endpoints
from fastapi import FastAPI
from my_app.modules.box.schemas import BoxResponse
from my_app.modules.toy.schemas import ToyResponse

app = FastAPI()

@app.get("/box/{box_id}", response_model=BoxResponse)
async def get_box(box_id: int):
    return BoxResponse(id=box_id, toys=[ToyResponse(id=1, box=BoxResponse(id=box_id))])

@app.get("/toy/{toy_id}", response_model=ToyResponse)
async def get_toy(toy_id: int):
    return ToyResponse(id=toy_id, box=BoxResponse(id=1, toys=[]))
