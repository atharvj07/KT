import numpy as np
from scipy.optimize import linprog

def find_guaranteed_variables(A_eq, b_eq, num_vars):
    guaranteed_zeros = set()
    guaranteed_ones = set()

    for i in range(num_vars):
        c = np.zeros(num_vars)
        c[i] = 1
        
        res_min = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=[(0, 1)] * num_vars, method='highs-ipm')
        res_max = linprog(-c, A_eq=A_eq, b_eq=b_eq, bounds=[(0, 1)] * num_vars, method='highs-ipm')

        if res_min.success and res_max.success:
            if np.isclose(res_min.x[i], 1) and np.isclose(res_max.x[i], 1):
                guaranteed_ones.add(i)
            elif np.isclose(res_min.x[i], 0) and np.isclose(res_max.x[i], 0):
                guaranteed_zeros.add(i)

    return guaranteed_zeros, guaranteed_ones

# Example system
A_eq = np.array([
    [1, 1, 1, 0],  # x1 + x2 + x3 = 2
    [1, 0, 0, 1],  # x1 + x4 = 1
    [1, 1, 0, 0]   # x1 + x2 = 1
])
b_eq = np.array([2, 1, 1])
num_vars = 4

# Find guaranteed variables
zeros, ones = find_guaranteed_variables(A_eq, b_eq, num_vars)
print("Guaranteed Zeros:", zeros)
print("Guaranteed Ones:", ones)
