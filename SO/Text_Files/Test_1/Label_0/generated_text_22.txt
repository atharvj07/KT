import java.util.*;
import java.util.concurrent.*;

public class FindMultiples {

    public int getSum(int n) {
        Set<Integer> set = ConcurrentHashMap.newKeySet();

        ExecutorService executor = Executors.newFixedThreadPool(3);

        executor.submit(() -> {
            for (int i = 3; i <= n; i += 3) set.add(i);
        });
        executor.submit(() -> {
            for (int i = 5; i <= n; i += 5) set.add(i);
        });
        executor.submit(() -> {
            for (int i = 7; i <= n; i += 7) set.add(i);
        });

        executor.shutdown();
        try {
            executor.awaitTermination(Long.MAX_VALUE, TimeUnit.NANOSECONDS);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        return set.stream().mapToInt(Integer::intValue).sum();
    }

    public static void main(String[] args) {
        FindMultiples fm = new FindMultiples();
        System.out.println(fm.getSum(10));      // Output: 40
        System.out.println(fm.getSum(1000));    // Output: 272066
        System.out.println(fm.getSum(76293));   // Output: 1365433807
    }
}
