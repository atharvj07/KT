class Instanceof {
    static void doesNotWork(Object o) {
        if (o == null || !(o instanceof String s)) {
            throw new Error();
        }
        System.out.println(s);
    }

    public static void main(String[] args) {
        try {
            doesNotWork("Test"); // Should print "Test"
            doesNotWork(null);   // Should throw Error
            doesNotWork(123);    // Should throw Error
        } catch (Error e) {
            System.out.println("Caught Error");
        }
    }
}
