import xarray as xr
import numpy as np

# Sample dataset
lon = np.linspace(0, 10, 11)
lat = np.linspace(0, 10, 11)
data = np.random.rand(11, 11)
DS = xr.DataArray(data, coords=[lat, lon], dims=["latitude", "longitude"])

# Coordinates to extract
x = [2, 4, 6, 8]
y = [3, 5, 7, 9]

# Extract values efficiently
subset = DS.sel(longitude=xr.DataArray(x, dims="points"), 
                latitude=xr.DataArray(y, dims="points"), 
                method="nearest")

print(subset)
