import pandas as pd
import numpy as np

# Sample DataFrame
df = pd.DataFrame({
    "Generation": [0],
    "Route_set": [np.array([[20, 19, 47, 56], [21, 34, 78, 34]])]
})

# Unnest and create new DataFrame
routes = np.column_stack(df["Route_set"].iloc[0])
df_out = pd.DataFrame(routes, columns=[f"route{i+1}" for i in range(routes.shape[1])])

print(df_out)
