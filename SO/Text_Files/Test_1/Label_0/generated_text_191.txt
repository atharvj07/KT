import java.lang.module.ModuleFinder;
import java.lang.module.ModuleReference;
import java.util.HashMap;
import java.util.Map;

public class CoreNonCoreModules {
    public static void main(String[] args) {
        Map<String, String> moduleClassLoaders = new HashMap<>();

        ModuleFinder.ofSystem().findAll().forEach(moduleReference -> {
            String moduleName = moduleReference.descriptor().name();
            try {
                Class<?> cls = Class.forName(moduleName + ".module-info");
                ClassLoader cl = cls.getClassLoader();
                moduleClassLoaders.put(moduleName, cl == null ? "BootLoader" : cl.getClass().getSimpleName());
            } catch (ClassNotFoundException ignored) {
                moduleClassLoaders.put(moduleName, "Unknown");
            }
        });

        moduleClassLoaders.forEach((module, classLoader) ->
            System.out.println(module + " -> " + classLoader)
        );
    }
}
