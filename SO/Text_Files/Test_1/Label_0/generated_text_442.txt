# Test case to verify resources are packaged
import myscript

def test_resources():
    from importlib.resources import files
    resource_folder = files('myscript.resources')
    assert 'artifacts' in resource_folder.parts
    assert 'recipe.template.yaml' in resource_folder.parts
    assert 'confs.yaml' in resource_folder.parts

test_resources()
