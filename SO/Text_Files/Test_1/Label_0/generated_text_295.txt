import pandas as pd
import numpy as np

df = pd.DataFrame(
    {
        'x': [1, np.nan, 3, np.nan, 5],
        'y': [np.nan, 7, 8, 9, np.nan],
        'x_a': [1, 2, 3, 4, 5],
        'y_a': [6, 7, 8, 9, 10]
    }
)

for col in df.columns:
    if col.endswith('_a'):
        base_col = col[:-2]
        if base_col in df.columns:
            df[base_col] = df[base_col].fillna(df[col])

print(df)
