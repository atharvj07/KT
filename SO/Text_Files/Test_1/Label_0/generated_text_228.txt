import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.TextArea;
import javafx.scene.text.Font;
import javafx.stage.Stage;

import java.io.File;
import java.net.MalformedURLException;

public class EmojiApp extends Application {
    @Override
    public void start(Stage stage) throws MalformedURLException {
        String emojiFontPath = new File("/path/to/NotoColorEmoji-Regular.ttf").toURI().toURL().toExternalForm();
        Font emojiFont = Font.loadFont(emojiFontPath, 18);
        
        TextArea ta = new TextArea("?? ?? ?? ?? ?? ??");
        ta.setFont(emojiFont);

        Scene scene = new Scene(ta, 400, 200);
        stage.setScene(scene);
        stage.setTitle("Emoji Test");
        stage.show();
    }

    public static void main(String[] args) {
        System.setProperty("prism.text", "t2k");
        System.setProperty("prism.lcdtext", "false");
        launch(args);
    }
}
