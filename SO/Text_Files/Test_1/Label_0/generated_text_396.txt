from typing import Type
import contextlib

class ErrorA(Exception): pass
class ErrorB(Exception): pass
class ErrorC(Exception): pass

class Validator:
    def validate(self, something, *, suppress: list[Type[Exception]] = []) -> None:
        if something == "a":
            self._raise(ErrorA(), suppress)
        if something == "b":
            self._raise(ErrorB(), suppress)
        if something == "c":
            self._raise(ErrorC(), suppress)

    def _raise(self, e: Exception, suppress: list[Type[Exception]]) -> None:
        with contextlib.suppress(*suppress):
            raise e

# Test cases
def test_validator():
    v = Validator()

    try:
        v.validate("a")
    except ErrorA:
        print("Caught ErrorA")  # Expected

    try:
        v.validate("b")
    except ErrorB:
        print("Caught ErrorB")  # Expected

    try:
        v.validate("c")
    except ErrorC:
        print("Caught ErrorC")  # Expected

    try:
        v.validate("b", suppress=[ErrorB])
    except Exception as e:
        print(f"Unexpected exception: {e}")
    else:
        print("ErrorB suppressed successfully")  # Expected

    try:
        v.validate("c", suppress=[ErrorB])
    except ErrorC:
        print("Caught ErrorC")  # Expected

test_validator()
