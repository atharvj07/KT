import pandas as pd
import requests
from bs4 import BeautifulSoup

# Send request to the website
page = requests.get('https://www.bmv.com.mx/es/mercados/capitales')

# Parse the HTML content with BeautifulSoup
soup = BeautifulSoup(page.text, 'lxml')

# Find the table rows inside tbody
rows = soup.find('tbody').find_all('tr')

# Define the columns that we are expecting
columns = [
    'Stock', 'Series', 'Time', 'Price', 'Open', 'High', 'Low', 'Close',
    'Volume', 'Value', 'Change', 'Percentage Change'
]

# Create an empty list to hold row data
data = []

# Loop through each row and extract the data
for row in rows:
    cols = row.find_all('td')
    
    # Extract href value
    stock_name = cols[0].find('a').text.strip() if cols[0].find('a') else ''
    
    # Extract series (if present) and other values
    series = cols[1].text.strip() if len(cols) > 1 else ''
    time = cols[2].text.strip() if len(cols) > 2 else ''
    price = cols[3].text.strip() if len(cols) > 3 else ''
    open_price = cols[4].text.strip() if len(cols) > 4 else ''
    high = cols[5].text.strip() if len(cols) > 5 else ''
    low = cols[6].text.strip() if len(cols) > 6 else ''
    close = cols[7].text.strip() if len(cols) > 7 else ''
    volume = cols[8].text.strip() if len(cols) > 8 else ''
    value = cols[9].text.strip() if len(cols) > 9 else ''
    change = cols[10].text.strip() if len(cols) > 10 else ''
    percentage_change = cols[11].text.strip() if len(cols) > 11 else ''

    # Append the data for this row
    data.append([
        stock_name, series, time, price, open_price, high, low, close, 
        volume, value, change, percentage_change
    ])

# Create a DataFrame
df = pd.DataFrame(data, columns=columns)

# Show the dataframe
print(df)
