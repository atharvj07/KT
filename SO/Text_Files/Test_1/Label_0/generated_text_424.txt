arr_1 = [1, [2, 3], 4, 5]

# Test case 2
arr_2 = [1, [2, 3], [[4, 5]]]
flattened_arr_2 = [item for sublist in arr_2 for item in (sublist if isinstance(sublist, list) else [sublist])]
assert flattened_arr_2 == [1, 2, 3, 4, 5]
