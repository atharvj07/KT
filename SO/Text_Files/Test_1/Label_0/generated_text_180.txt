import java.util.*;

public class OverlappingSetsFinder {
    public static void main(String[] args) {
        Map<String, Set<Long>> myMap = new HashMap<>();
        myMap.put("a", Set.of(1L, 2L, 3L));
        myMap.put("b", Set.of(4L, 5L));
        myMap.put("c", Set.of(1L));
        myMap.put("d", Set.of(2L, 3L));
        myMap.put("e", Set.of(5L));
        myMap.put("f", Set.of(6L));

        findOverlappingKeys(myMap).forEach(System.out::println);
    }

    public static List<List<String>> findOverlappingKeys(Map<String, Set<Long>> myMap) {
        List<Map.Entry<String, Set<Long>>> entries = new ArrayList<>(myMap.entrySet());
        List<List<String>> result = new ArrayList<>();

        for (int i = 0; i < entries.size(); i++) {
            for (int j = i + 1; j < entries.size(); j++) {
                if (entries.get(j).getValue().containsAll(entries.get(i).getValue()) || 
                    entries.get(i).getValue().containsAll(entries.get(j).getValue())) {
                    result.add(List.of(entries.get(i).getKey(), entries.get(j).getKey()));
                }
            }
        }
        return result;
    }
}
