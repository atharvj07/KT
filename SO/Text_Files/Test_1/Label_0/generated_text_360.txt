import polars as pl

df = pl.DataFrame(
    {"name": list("abcdef"), "age": [21, 31, 32, 53, 45, 26], "country": list("AABBBC")}
)

# Sorting names by age within each country and handling out-of-bounds with nulls
df = df.with_columns(
    pl.col("name").sort_by("age").over("country").alias("name_sorted")
)

df.group_by("country").agg(
    pl.col("name_sorted").first().alias("age_sort_1"),
    pl.col("name_sorted").get(2).alias("age_sort_2"),  # Will return null if out of bounds
    pl.col("name_sorted").last().alias("age_sort_-1")
)

# Test case
print(df)
