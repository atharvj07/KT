public class MyFragment extends Fragment {

    private LinearLayout parentLayout;
    private Button addProduct;

    @Nullable
    @Override
    public View onCreateView(@NonNull LayoutInflater inflater, @Nullable ViewGroup container, @Nullable Bundle savedInstanceState) {
        View root = inflater.inflate(R.layout.fragment_layout, container, false);

        parentLayout = root.findViewById(R.id.ll_out);
        addProduct = root.findViewById(R.id.btn_prd);

        addProduct.setOnClickListener(v -> addNewProduct());

        return root;
    }

    private void addNewProduct() {
        LinearLayout productLayout = (LinearLayout) getLayoutInflater().inflate(R.layout.product_row, parentLayout, false);

        AutoCompleteTextView tvProduct = productLayout.findViewById(R.id.tv_product);
        EditText prodQty = productLayout.findViewById(R.id.prod_qty);
        EditText prodPrice = productLayout.findViewById(R.id.prod_price);
        EditText prodSpecs = productLayout.findViewById(R.id.prod_specs);
        Button deleteButton = new Button(getContext());
        deleteButton.setText("X");
        deleteButton.setOnClickListener(v -> parentLayout.removeView(productLayout));

        productLayout.addView(deleteButton);
        parentLayout.addView(productLayout);
    }
}
