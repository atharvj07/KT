import polars as pl

def reorder_columns(df, first_cols):
    remaining_cols = [col for col in df.columns if col not in first_cols]
    return df.select(first_cols + remaining_cols)

# Test Case
df = pl.DataFrame({
    "col1": [1, 2, 3],
    "col2": [4, 5, 6],
    "col3": [7, 8, 9],
    "col4": [10, 11, 12],
    "col5": [13, 14, 15],
    "col6": [16, 17, 18]
})

first_cols = ["col2", "col1", "col5"]
df_reordered = reorder_columns(df, first_cols)
print(df_reordered)
