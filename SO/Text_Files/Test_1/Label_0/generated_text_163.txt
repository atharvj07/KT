import java.util.Random;

class Card {
    String value;
    String suit;

    Card(String value, String suit) {
        this.value = value;
        this.suit = suit;
    }

    @Override
    public String toString() {
        return value + " of " + suit;
    }
}

public class CardShuffle {
    public static void main(String[] args) {
        Card[] unshuffledDeck = new Card[]{
            new Card("2", "Hearts"),
            new Card("3", "Hearts"),
            new Card("4", "Hearts"),
            new Card("5", "Hearts"),
            new Card("6", "Hearts")
        };

        Card[] shuffledDeck = shuffleDeck(unshuffledDeck);

        for (Card card : shuffledDeck) {
            System.out.println(card);
        }
    }

    public static Card[] shuffleDeck(Card[] unshuffledDeck) {
        Random shuffleRandom = new Random();
        Card[] shuffledDeck = new Card[unshuffledDeck.length];
        boolean[] picked = new boolean[unshuffledDeck.length];

        for (int i = 0; i < unshuffledDeck.length; i++) {
            int j;
            do {
                j = shuffleRandom.nextInt(unshuffledDeck.length);
            } while (picked[j]);
            shuffledDeck[i] = unshuffledDeck[j];
            picked[j] = true;
        }

        return shuffledDeck;
    }
}
