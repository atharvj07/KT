import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.canvas.Canvas;
import javafx.scene.canvas.GraphicsContext;
import javafx.scene.input.KeyCode;
import javafx.scene.layout.StackPane;
import javafx.scene.paint.Color;
import javafx.stage.Stage;

public class DynamicDrawMode extends Application {

    private boolean drawMode = true; // true = line, false = rectangle

    @Override
    public void start(Stage stage) {
        Canvas canvas = new Canvas(800, 600);
        GraphicsContext gc = canvas.getGraphicsContext2D();

        canvas.setOnMouseDragged(e -> {
            gc.setFill(Color.LIGHTBLUE);
            gc.setStroke(Color.BLACK);
            gc.setLineWidth(2);

            if (drawMode) {
                gc.strokeLine(e.getX(), e.getY(), e.getX() + 1, e.getY() + 1);
            } else {
                gc.fillRect(e.getX(), e.getY(), 20, 20);
            }
        });

        canvas.setOnKeyPressed(e -> {
            if (e.getCode() == KeyCode.SPACE) {
                drawMode = !drawMode;
            }
        });

        StackPane root = new StackPane(canvas);
        Scene scene = new Scene(root, 800, 600);
        canvas.requestFocus();

        stage.setTitle("Dynamic Draw Mode");
        stage.setScene(scene);
        stage.show();
    }

    public static void main(String[] args) {
        launch();
    }
}
