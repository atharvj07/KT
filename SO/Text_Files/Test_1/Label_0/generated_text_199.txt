import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

import java.util.Optional;
import java.util.UUID;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;

public class PriceServiceTest {
    @Mock
    private PriceRepository priceRepository;

    @InjectMocks
    private PriceService priceService;

    private UUID priceUuid;
    private Price price;
    private PriceRequest request;

    @BeforeEach
    public void setup() {
        MockitoAnnotations.openMocks(this);
        priceUuid = UUID.randomUUID();
        price = new Price();
        price.setUuid(priceUuid);
        price.setPriceAmount(100.0);

        request = new PriceRequest();
        request.setPriceAmount(150.0);
    }

    @Test
    public void testUpdate() {
        when(priceRepository.findByUuid(priceUuid)).thenReturn(Optional.of(price));
        when(priceRepository.saveAndFlush(any(Price.class))).thenAnswer(invocation -> invocation.getArgument(0));

        PriceDTO updatedDto = priceService.update(priceUuid, request);

        assertNotNull(updatedDto);
        assertEquals(150.0, updatedDto.getPriceAmount());
        verify(priceRepository, times(1)).saveAndFlush(price);
    }
}
