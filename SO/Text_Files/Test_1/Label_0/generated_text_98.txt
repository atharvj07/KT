// Test case to ensure each controller is a unique instance
@SpringBootTest
public class ListItemTest {

    @Autowired
    private ApplicationContext applicationContext;

    @Test
    public void testNewControllerInstance() throws IOException {
        FXMLLoader fxmlLoader = new FXMLLoader(getClass().getResource("/path/to/your/fxml"));
        fxmlLoader.setControllerFactory(applicationContext::getBean);
        VBox listItem1 = fxmlLoader.load();
        VBox listItem2 = fxmlLoader.load();

        ListItemController controller1 = fxmlLoader.getController();
        ListItemController controller2 = fxmlLoader.getController();

        assertNotSame(controller1, controller2);
    }
}
