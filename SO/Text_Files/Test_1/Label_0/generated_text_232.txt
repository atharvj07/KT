String leftText = "Left";
String rightText = "Right";

SpannableStringBuilder builder = new SpannableStringBuilder();
builder.append(leftText);
int spaceCount = Math.max(1, 40 - leftText.length() - rightText.length()); // Adjust spacing as needed
builder.append(new String(new char[spaceCount]).replace("\0", "\u00A0")); // Non-breaking space
builder.append(rightText);

textView.setText(builder);
