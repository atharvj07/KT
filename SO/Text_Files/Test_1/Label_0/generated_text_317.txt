import pandas as pd

inp = pd.DataFrame({'month':[1,1,1,1,1,2,2,2], 
                    'item':[10,10,10,10,20,20,20,20], 
                    'star':[1,2,1,3,3,2,2,3]})

out = (inp
       .groupby(['month', 'item', 'star'])
       .size()
       .unstack(fill_value=0)
       .rename(columns={1: 'star_1_cnt', 2: 'star_2_cnt', 3: 'star_3_cnt'})
       .reset_index())

print(out)
