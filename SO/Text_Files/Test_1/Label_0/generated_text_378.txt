import pandas as pd
from sklearn import datasets

# Load iris dataset
iris = pd.DataFrame(datasets.load_iris().data, columns=datasets.load_iris().feature_names)

# Create new column and drop used columns
iris = (iris.assign(new_col=iris['sepal length (cm)'] + iris['petal length (cm)'] * iris['petal width (cm)'])
            .loc[:, ~iris.columns.isin(['sepal length (cm)', 'petal length (cm)', 'petal width (cm)'])])

print(iris.head())
