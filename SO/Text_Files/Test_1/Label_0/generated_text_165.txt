public Mono<Void> updateAnalysisWithResults(String uuidString, String results) {
    return findByUUID(uuidString)
            .doOnNext(analysis -> analysis.setResults(results))
            .flatMap(computeSCARepository::save)
            .then();
}

@PatchMapping("compute/{uuid}/results")
public Mono<Void> patchAnalysisWithResults(@PathVariable String uuid, @RequestBody String results) {
    return computeSCAService.updateAnalysisWithResults(uuid, results);
}

// Test Case
@Test
public void testUpdateAnalysisWithResults() {
    String uuid = "test-uuid";
    String results = "new-results";

    Analysis analysis = new Analysis();
    analysis.setUuid(uuid);

    when(computeSCARepository.save(any(Analysis.class))).thenReturn(Mono.just(analysis));
    when(computeSCAService.findByUUID(uuid)).thenReturn(Mono.just(analysis));

    StepVerifier.create(computeSCAService.updateAnalysisWithResults(uuid, results))
            .verifyComplete();

    verify(computeSCARepository).save(any(Analysis.class));
    assertEquals("new-results", analysis.getResults());
}
