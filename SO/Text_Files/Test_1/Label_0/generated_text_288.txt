from dataclasses import dataclass
from omegaconf import OmegaConf
from pydantic import BaseModel, ValidationError, conint

class SomeConfigValidator(BaseModel):
    param: conint(ge=1, le=10)  # Example: Ensure param is between 1 and 10

@dataclass
class SomeConfig:
    param: int

    def __post_init__(self):
        try:
            SomeConfigValidator(param=self.param)
        except ValidationError as e:
            raise ValueError(e)

cfg = OmegaConf.structured(SomeConfig(param=5))  # Valid
print(cfg)

cfg_invalid = OmegaConf.structured(SomeConfig(param=15))  # Raises ValueError
