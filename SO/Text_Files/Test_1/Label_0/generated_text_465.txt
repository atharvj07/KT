import pandas as pd

df = pd.read_csv('XBT_60.csv', index_col='date', parse_dates=True)
df['date'] = pd.to_datetime(df['date'])
df.set_index('date', inplace=True)
df = df.asfreq('H')

# Test case for seasonal decomposition
from statsmodels.tsa.seasonal import seasonal_decompose
result = seasonal_decompose(df['column_name'], model='additive', period=24)  # adjust column_name and period as needed
print(result)
