import hudson.AbortException
import org.mockito.invocation.InvocationOnMock
import org.mockito.stubbing.Answer
import static org.mockito.ArgumentMatchers.any
import static org.mockito.ArgumentMatchers.eq
import static org.mockito.Mockito.*

class DSLMock {

    def mock

    DSLMock() {
        this.mock = mock(DSL.class)

        doAnswer(new Answer<String>() {
            @Override
            String answer(InvocationOnMock invocation) throws Throwable {
                throw new AbortException((String) invocation.getArguments()[1][0])
            }
        }).when(mock).invokeMethod(eq("error"), any())
    }
}

groovy
import org.junit.jupiter.api.BeforeEach
import org.junit.jupiter.api.Test
import static org.mockito.Mockito.*

class DSLMockTest {

    private DSLMock dslMock

    @BeforeEach
    void setUp() {
        dslMock = new DSLMock()
    }

    @Test
    void testInvokeMethodThrowsAbortException() {
        assertThrows(AbortException.class, () -> {
            dslMock.mock.invokeMethod("error", ["Test error"])
        })
    }
}
