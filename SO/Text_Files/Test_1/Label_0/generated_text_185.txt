import java.util.regex.*;

public class AzurePathParser {
    private static final Pattern AZURE_REGEX = Pattern.compile(
        "abfss?://([^/]+)@([^\\.]+)(\\.[^/]+)?/?((.+)?)"
    );
    private static final Pattern AZURITE_REGEX = Pattern.compile(
        "abfss?://([^/]+)@([^:]+:\\d+)/([^/]+)/?((.+)?)"
    );

    public static void parsePath(String path) {
        Matcher matcher = AZURE_REGEX.matcher(path);
        if (matcher.find()) {
            String fileSystem = matcher.group(1);
            String accountName = matcher.group(2);
            String accountSuffix = matcher.group(3) != null ? matcher.group(3) : "";
            String relativePath = matcher.group(4) != null ? matcher.group(4) : "";
            printResult(path, fileSystem, accountName, accountSuffix, relativePath);
            return;
        }

        matcher = AZURITE_REGEX.matcher(path);
        if (matcher.find()) {
            String fileSystem = matcher.group(1);
            String accountName = matcher.group(3);
            String accountSuffix = "";
            String relativePath = matcher.group(4) != null ? matcher.group(4) : "";
            printResult(path, fileSystem, accountName, accountSuffix, relativePath);
        }
    }

    private static void printResult(String path, String fileSystem, String accountName, String accountSuffix, String relativePath) {
        System.out.println("Path: " + path);
        System.out.println("FileSystem: " + fileSystem);
        System.out.println("AccountName: " + accountName);
        System.out.println("AccountSuffix: " + accountSuffix);
        System.out.println("RelativePath: " + relativePath);
        System.out.println("------------");
    }

    public static void main(String[] args) {
        parsePath("abfs://storage@myaccount.dfs.core.windows.net/selim/test.csv");
        parsePath("abfs://storage@127.0.0.1:10000/devstoreaccount1/selim/test.csv");
    }
}
