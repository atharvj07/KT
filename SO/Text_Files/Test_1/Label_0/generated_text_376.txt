from dataclasses import dataclass
from typing import Protocol

class HasA(Protocol):
    a: int

class HasHasA(Protocol):
    hasa: HasA

@dataclass
class Inner:
    a: int

@dataclass
class Outer:
    hasa: HasA  # Change this to HasA, not Inner

def func(b: HasHasA): ...

o = Outer(Inner(0))
func(o)
