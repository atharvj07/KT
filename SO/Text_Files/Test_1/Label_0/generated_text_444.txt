import numpy as np

arr = np.arange(1, 499875841)

# Split array into chunks of 4 elements every 8 elements
reshaped = arr[:len(arr) // 8 * 8].reshape(-1, 8)
result = reshaped[:, :4]

# Test case
assert np.array_equal(result[0], np.array([1, 2, 3, 4]))
assert np.array_equal(result[1], np.array([5, 6, 7, 8]))
