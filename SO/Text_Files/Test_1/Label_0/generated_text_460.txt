import numpy as np
from collections import Counter

def balanceness(array: np.ndarray):
    classes = [(c, cnt) for c, cnt in Counter(array).items()]
    m = len(classes)
    n = len(array)
    H = -sum([(cnt / n) * np.log((cnt / n)) for c, cnt in classes])
    return H / np.log(m)

def weight(days_ago: int, epilson: float=0.005) -> float:
    return np.exp(-1 * days_ago * epilson)

def weighted_balanceness(player_A: np.ndarray, player_B: np.ndarray, days_ago: np.ndarray, epilson: float=0.005):
    weight_A = np.array([weight(d) for d in days_ago[0]])
    weight_B = np.array([weight(d) for d in days_ago[1]])
    
    weighted_A = player_A * weight_A
    weighted_B = player_B * weight_B
    
    combined = np.concatenate([weighted_A, weighted_B])
    return balanceness(combined)

# Test data
player_A = np.array(['C', 'C', 'D', 'F', 'F', 'F', 'G'])
player_B = np.array(['C', 'C', 'D', 'F', 'F', 'F', 'G', 'G', 'G', 'G', 'G', 'G', 'G', 'G'])
days_ago_A = np.array([1, 2, 10, 100, 101, 102, 1])
days_ago_B = np.array([1, 2, 10, 100, 101, 102, 1, 2, 3, 4, 5, 6, 7, 8])

# Calculate weighted balanced opponent
opponents = np.unique(np.concatenate([player_A, player_B]))
opponent_scores = {}
for opponent in opponents:
    a_matches = player_A == opponent
    b_matches = player_B == opponent
    days_ago_a = days_ago_A[a_matches]
    days_ago_b = days_ago_B[b_matches]
    all_days_ago = np.concatenate([days_ago_a, days_ago_b])
    
    score = weighted_balanceness(a_matches, b_matches, [days_ago_a, days_ago_b])
    opponent_scores[opponent] = score

# Find the most recent balanced opponent
most_recent_balanced_opponent = min(opponent_scores, key=opponent_scores.get)
print(most_recent_balanced_opponent)
