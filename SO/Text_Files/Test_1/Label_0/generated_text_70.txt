import com.google.gson.Gson;
import org.bson.Document;
import com.mongodb.client.model.InsertOneModel;

import java.util.List;
import java.util.stream.Collectors;

public class ParallelStreamExample {
    private static final Gson gson = new Gson();

    public static void main(String[] args) {
        List<Puzzle> puzzles = List.of(new Puzzle("Puzzle1"), new Puzzle("Puzzle2"), new Puzzle("Puzzle3"));

        List<InsertOneModel<Document>> bulkWrites = puzzles.parallelStream()
                .map(gson::toJson)
                .map(Document::parse)
                .map(InsertOneModel::new)
                .collect(Collectors.toList());

        bulkWrites.forEach(System.out::println);
    }

    static class Puzzle {
        String name;

        Puzzle(String name) {
            this.name = name;
        }
    }
}

import org.junit.jupiter.api.Test;
import org.bson.Document;
import com.mongodb.client.model.InsertOneModel;
import java.util.List;

import static org.junit.jupiter.api.Assertions.*;

class ParallelStreamExampleTest {

    @Test
    void testParallelStream() {
        List<ParallelStreamExample.Puzzle> puzzles = List.of(
                new ParallelStreamExample.Puzzle("Puzzle1"),
                new ParallelStreamExample.Puzzle("Puzzle2"),
                new ParallelStreamExample.Puzzle("Puzzle3")
        );

        List<InsertOneModel<Document>> bulkWrites = puzzles.parallelStream()
                .map(ParallelStreamExample.gson::toJson)
                .map(Document::parse)
                .map(InsertOneModel::new)
                .toList();

        assertEquals(3, bulkWrites.size());
        assertEquals("Puzzle1", bulkWrites.get(0).getDocument().getString("name"));
    }
}
