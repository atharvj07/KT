import cv2 as cv
import numpy as np

def find_baseline(image_path, output_path):
    src = cv.imread(image_path, cv.IMREAD_GRAYSCALE)
    _, binary = cv.threshold(src, 0, 255, cv.THRESH_BINARY_INV + cv.THRESH_OTSU)

    # Get horizontal projections
    projection = np.sum(binary, axis=1)
    
    # Smooth projection to remove noise
    projection = cv.GaussianBlur(projection, (5, 5), 0)
    
    # Find peaks in projection (baseline candidates)
    peaks = np.argwhere(projection > np.percentile(projection, 90)).flatten()
    
    # Draw baselines
    output = cv.cvtColor(src, cv.COLOR_GRAY2BGR)
    for y in peaks:
        cv.line(output, (0, y), (output.shape[1], y), (0, 0, 255), 1)
    
    cv.imwrite(output_path, output)

# Test case
find_baseline("boston_cooking_a.jpg", "test_output.jpg")
