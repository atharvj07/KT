import pandas as pd

# Sample data
data = {'dttime': [240313153045, 240313120500, 240313091530, 240313210000]}
df = pd.DataFrame(data)

# Convert to datetime
df['dttime'] = pd.to_datetime(df['dttime'].astype(str), format='%y%m%d%H%M%S')

# Sort correctly
df = df.sort_values(by='dttime', ascending=True, ignore_index=True)

# Output result
print(df)
