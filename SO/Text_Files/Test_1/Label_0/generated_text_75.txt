public class StrictFPTest {

    public static void main(String[] args) {
        System.out.println("Without strictfp:");
        testWithoutStrictFP();

        System.out.println("\nWith strictfp:");
        testWithStrictFP();
    }

    static void testWithoutStrictFP() {
        double a = Double.MAX_VALUE;
        double b = a * 2; // Overflow to Infinity
        double c = b / 4; // Infinity / 4 = Infinity
        System.out.println("Result: " + c);
    }

    strictfp static void testWithStrictFP() {
        double a = Double.MAX_VALUE;
        double b = a * 2; // Overflow to Infinity
        double c = b / 4; // Infinity / 4 = Infinity
        System.out.println("Result: " + c);
    }
}


java StrictFPTest


Without strictfp:
Result: Infinity

With strictfp:
Result: Infinity
