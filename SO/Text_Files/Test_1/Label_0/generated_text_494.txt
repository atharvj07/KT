import polars as pl

df = pl.DataFrame({"a": [1, 2, 3, 4, 5], "b": [2, 3, 4, 5, 6], "x": [1, 3, 5, 7, 9]})

df = df.with_columns(pl.col("x").cut(bins=[0, 2, 4, 6, 8, 10]).alias("x_binned"))

print(df)

# Test Case
def test_cut():
    assert "x_binned" in df.columns
    assert df["x_binned"].to_list() == ["(0.0, 2.0]", "(2.0, 4.0]", "(4.0, 6.0]", "(6.0, 8.0]", "(8.0, 10.0]"]

test_cut()
