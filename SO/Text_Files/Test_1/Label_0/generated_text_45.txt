import java.util.concurrent.ConcurrentHashMap;

public class AtomicComputeIfAbsent {

    private final ConcurrentHashMap<String, Integer> map = new ConcurrentHashMap<>();
    private volatile int sharedField = 0;

    public int computeValue(String key) {
        return map.computeIfAbsent(key, k -> {
            synchronized (this) {
                sharedField++;
                return sharedField;
            }
        });
    }

    public static void main(String[] args) {
        AtomicComputeIfAbsent example = new AtomicComputeIfAbsent();

        Runnable task = () -> {
            for (int i = 0; i < 1000; i++) {
                String key = "key" + (i % 10);
                int value = example.computeValue(key);
                System.out.println(Thread.currentThread().getName() + " computed " + key + " = " + value);
            }
        };

        Thread t1 = new Thread(task);
        Thread t2 = new Thread(task);

        t1.start();
        t2.start();

        try {
            t1.join();
            t2.join();
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        System.out.println("Final sharedField value: " + example.sharedField);
    }
}
