import pandas as pd

df = pd.DataFrame({"Type":["RF", "RF", "RF", "MLP", "MLP", "MLP"], "Value":[-1.5,-0.1,1.7,0.2,-0.7,-0.6]})

result = df.assign(
    Low=(df["Value"] < -0.25).astype(int),
    Hit=(df["Value"].between(-0.25, 0.25)).astype(int),
    High=(df["Value"] > 0.25).astype(int)
).groupby("Type", as_index=False)[["Low", "Hit", "High"]].sum()

# Test case
expected = pd.DataFrame({"Type":["RF", "MLP"], "Low":[1,2], "Hit":[1,1], "High":[1,0]})
pd.testing.assert_frame_equal(result, expected)
