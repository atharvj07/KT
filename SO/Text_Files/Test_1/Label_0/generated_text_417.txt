from pyspark.sql import functions as F

def apply_schema(df, schema):
    for field in schema.fields:
        df = df.withColumn(field.name, df[field.name].cast(field.dataType))
    return df

# Example usage
df1 = spark.createDataFrame([(None, 1, "a")], ["A", "B", "C"]).withColumn("A", F.col("A").cast("date"))
df2 = spark.createDataFrame([("2021-01-01", "2", "b")], ["A", "B", "C"])

df2 = apply_schema(df2, df1.schema)

df2.printSchema()  # Should match df1 schema
df2.show()         # Should show df2 with df1's schema applied
