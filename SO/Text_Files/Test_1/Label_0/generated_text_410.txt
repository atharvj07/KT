import numpy as np
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense

# Example data
x_data = np.random.rand(1000, 1209278)  # Shape (samples, time_steps)
y_data = np.random.rand(1000)  # Shape (samples,)

# Reshaping x_data to 3D shape (samples, time_steps, features)
x_data = x_data.reshape(x_data.shape[0], x_data.shape[1], 1)

# Model
model = Sequential()
model.add(LSTM(64, input_shape=(x_data.shape[1], 1)))
model.add(Dense(1))

model.compile(optimizer='adam', loss='mean_squared_error')

# Training
model.fit(x_data, y_data, epochs=10, batch_size=32)

# Test case
x_test = np.random.rand(1, 1209278).reshape(1, 1209278, 1)
y_pred = model.predict(x_test)
print(y_pred)
