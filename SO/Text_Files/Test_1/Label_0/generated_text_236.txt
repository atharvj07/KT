private void bfs(String source) {
    level = new int[nodes.size()];
    Arrays.fill(level, -1);
    int sourceIndex = nodes.indexOf(source);
    level[sourceIndex] = 0;

    Queue<Integer> queue = new LinkedList<>();
    queue.offer(sourceIndex);

    while (!queue.isEmpty()) {
        int u = queue.poll();
        String uNode = nodes.get(u);
        for (Map.Entry<String, Integer> entry : network.get(uNode).entrySet()) {
            String vNode = entry.getKey();
            int capacity = entry.getValue();
            int v = nodes.indexOf(vNode);
            if (capacity > 0 && level[v] == -1) {
                level[v] = level[u] + 1;
                queue.offer(v);
            }
        }
    }
}

private int dfs(int u, int flow, int sinkIndex) {
    if (u == sinkIndex) {
        return flow;
    }

    String uNode = nodes.get(u);
    for (Map.Entry<String, Integer> entry : network.get(uNode).entrySet()) {
        String vNode = entry.getKey();
        int capacity = entry.getValue();
        int v = nodes.indexOf(vNode);

        if (capacity > 0 && level[u] < level[v]) {
            int sent = dfs(v, Math.min(flow, capacity), sinkIndex);
            if (sent > 0) {
                network.get(uNode).put(vNode, capacity - sent);
                network.get(vNode).put(uNode, network.get(vNode).getOrDefault(uNode, 0) + sent);
                return sent;
            }
        }
    }
    return 0;
}

public int calculator(Map<T, Set<T>> graph) {
    nodes = new ArrayList<>();
    nodes.add("src");
    nodes.add("sink");
    network.clear();

    for (T t : graph.keySet()) {
        String tKey = "A" + t.toString();
        addEdge("src", tKey, 1);
        addEdge("B" + t, "sink", 1);
        Set<String> visitedSubset = new HashSet<>();
        for (String u : reach(graph, t, visitedSubset)) {
            addEdge(tKey, u, 1);
        }
    }

    int maxFlow = 0;
    int sourceIndex = nodes.indexOf("src");
    int sinkIndex = nodes.indexOf("sink");

    while (true) {
        bfs("src");
        if (level[sinkIndex] == -1) {
            break;
        }

        int flow;
        while ((flow = dfs(sourceIndex, Integer.MAX_VALUE, sinkIndex)) > 0) {
            maxFlow += flow;
        }
    }

    return graph.size() - maxFlow;
}
