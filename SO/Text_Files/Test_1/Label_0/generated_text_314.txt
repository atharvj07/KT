import numpy as np
from time import time

def numpy_comparison(do_cache: bool, array_size: int, num_arrays: int, num_iter: int):
    arrays = np.random.rand(num_arrays, array_size)

    if do_cache:
        cache = {}

    for _ in range(num_iter):
        i, j = np.random.randint(num_arrays, size=2)

        if do_cache:
            key = (i, j) if i <= j else (j, i)  # Ensure consistent order in cache keys
            if key in cache:
                a = cache[key]
            else:
                a = arrays[i] + arrays[j]
                cache[key] = a
        else:
            a = arrays[i] + arrays[j]

if __name__ == "__main__":
    import timeit
    num_iter = 1000
    print(timeit.timeit(lambda: numpy_comparison(do_cache=False, array_size=10000, num_arrays=100, num_iter=num_iter), number=10))
    print(timeit.timeit(lambda: numpy_comparison(do_cache=True, array_size=10000, num_arrays=100, num_iter=num_iter), number=10))
