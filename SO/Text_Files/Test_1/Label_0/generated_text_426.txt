import polars as pl

# Sample dataframe
df = pl.DataFrame({
    "a_0": [1, 4],
    "a_1": [2, 5],
    "a_2": [3, 6],
    "b_0": [7, 10],
    "b_1": [8, 11],
    "b_2": [9, 12]
})

# Reshape the dataframe
df_long = df.select(
    [pl.col(col).alias("a") for col in ["a_0", "a_1", "a_2"]] +
    [pl.col(col).alias("b") for col in ["b_0", "b_1", "b_2"]]
).melt()

# Display the result
print(df_long)
