import click

@click.command()
@click.option('--value', default=1, help='a value.')
def hello(value, ctx):
    if ctx.params['value'] == 1 and 'value' not in ctx.resilient_parsing:
        print(f"Using default value: {value}")
    else:
        print(f"Using user input value: {value}")

if __name__ == "__main__":
    hello()
