python
import re
from itertools import zip_longest

def gawk_split(s, regex):
    parts = re.split(f"({regex})", s)
    return list(zip_longest(*[iter(parts)]*2, fillvalue=""))

# Test case
test_str = "a   b c    d _ e"
print(gawk_split(test_str, r"[\s_]+"))

perl
use strict;
use warnings;
use Data::Dumper;

sub gawk_split {
    my ($str, $regex) = @_;
    my @parts = split /($regex)/, $str;
    my @result;
    push @result, [ shift(@parts), shift(@parts) // '' ] while @parts;
    return \@result;
}

# Test case
my $test_str = "a   b c    d _ e";
print Dumper(gawk_split($test_str, qr/[\s_]+/));
