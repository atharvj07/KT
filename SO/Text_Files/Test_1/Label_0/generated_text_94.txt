import java.lang.annotation.*;
import java.lang.reflect.*;

@Target(ElementType.FIELD)
@Retention(RetentionPolicy.RUNTIME)
@Documented
public @interface Annotation {
    Class<?> className() default Object.class;
}

class Main {
    @Annotation
    private AnotherClass annotatedAttribute;

    @Annotation(className = YetAnotherClass.class)
    private AnotherClass anotherAnnotatedAttribute;

    public static void main(String[] args) throws Exception {
        for (Field field : Main.class.getDeclaredFields()) {
            Annotation annotation = field.getAnnotation(Annotation.class);
            if (annotation != null) {
                Class<?> value = annotation.className() == Object.class ? field.getType() : annotation.className();
                System.out.printf("Field: %s, Annotated class: %s%n", field.getName(), value.getName());
            }
        }
    }
}

class AnotherClass {}
class YetAnotherClass {}
