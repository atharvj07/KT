import polars as pl

# Create the dataframe
df = pl.DataFrame({
    "x": [[1], [1, 2], [1, 2, 3]],
    "y": [[True], None, [True, False, True]]
})

# Fix the null by repeating it to the correct length
df = df.with_columns(
    pl.col("y").apply(lambda s: s if s is not None else [None] * len(df), return_dtype=pl.List(pl.Boolean))
)

# Explode the columns
df = df.explode(["x", "y"])

print(df)
