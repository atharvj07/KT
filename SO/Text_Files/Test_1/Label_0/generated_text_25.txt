import java.time.temporal.ChronoUnit;
import java.time.Year;
import java.time.YearWeek;

public class WeekOperations {
    public static void main(String[] args) {
        YearWeek weekA = YearWeek.of(2012, 48);
        YearWeek weekB = YearWeek.of(2013, 3);

        System.out.println(weekA.isBefore(weekB)); // true
        System.out.println(ChronoUnit.WEEKS.between(weekA.atDay(1), weekB.atDay(1))); // 6
    }
}

class YearWeek implements Comparable<YearWeek> {
    private final int year;
    private final int week;

    private YearWeek(int year, int week) {
        this.year = year;
        this.week = week;
    }

    public static YearWeek of(int year, int week) {
        if (week < 1 || week > Year.of(year).length() / 7 + (Year.of(year).length() % 7 > 0 ? 1 : 0)) {
            throw new IllegalArgumentException("Invalid week: " + week + " for year: " + year);
        }
        return new YearWeek(year, week);
    }

    public boolean isBefore(YearWeek other) {
        return this.compareTo(other) < 0;
    }

    public boolean isAfter(YearWeek other) {
        return this.compareTo(other) > 0;
    }

    @Override
    public int compareTo(YearWeek other) {
        if (this.year != other.year) {
            return Integer.compare(this.year, other.year);
        }
        return Integer.compare(this.week, other.week);
    }

    public java.time.LocalDate atDay(int dayOfWeek) {
        return java.time.LocalDate.ofYearDay(year, 1)
                .with(java.time.temporal.WeekFields.ISO.weekOfYear(), week)
                .with(java.time.temporal.WeekFields.ISO.dayOfWeek(), dayOfWeek);
    }

    @Override
    public String toString() {
        return String.format("%d-W%02d", year, week);
    }
}
