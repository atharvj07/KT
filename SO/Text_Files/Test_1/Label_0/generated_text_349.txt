from transformers import pipeline, AutoTokenizer, AutoModelForSequenceClassification
import torch

# Define custom pipeline to return logits
class LogitsSentimentAnalysisPipeline(pipeline.Pipeline):
    def _sanitize_parameters(self, *args, **kwargs):
        return {}, {}

    def _call(self, inputs, *args, **kwargs):
        model_inputs = self.tokenizer(inputs, return_tensors="pt", padding=True, truncation=True)
        with torch.no_grad():
            logits = self.model(**model_inputs).logits
        return logits

# Initialize model and tokenizer
selected_model = "distilbert-base-uncased-finetuned-sst-2-english"
tokenizer = AutoTokenizer.from_pretrained(selected_model)
model = AutoModelForSequenceClassification.from_pretrained(selected_model)

# Create the custom pipeline
logits_classifier = LogitsSentimentAnalysisPipeline(model=model, tokenizer=tokenizer)

# Test data
texts = ["this is the first sentence", "of my data.", "In fact, thats not true,", "but we are going to assume it", "is"]

# Get logits
logits = logits_classifier(texts)
print(logits)
