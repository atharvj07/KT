import attr
import pandas as pd

@attr.s(auto_attribs=True)
class MyAttrsClass:
    i: int
    s: str

def attrs_to_dataframe(objects):
    return pd.DataFrame([attr.asdict(obj) for obj in objects])

# Test Case
df = attrs_to_dataframe([MyAttrsClass(1, "a"), MyAttrsClass(2, "b")])
print(df)
