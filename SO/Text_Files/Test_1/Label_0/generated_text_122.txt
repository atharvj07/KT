import javafx.application.Application;
import javafx.beans.property.DoubleProperty;
import javafx.beans.property.SimpleDoubleProperty;
import javafx.scene.Scene;
import javafx.scene.control.Spinner;
import javafx.scene.control.SpinnerValueFactory;
import javafx.scene.layout.VBox;
import javafx.scene.text.Text;
import javafx.stage.Stage;

import java.math.BigDecimal;
import java.text.DecimalFormat;
import java.util.regex.Pattern;
import java.util.function.UnaryOperator;

public class TestPane extends Application {

    @Override
    public void start(Stage primaryStage) {
        Spinner<Double> spinner = createSpinner();

        VBox root = new VBox(10, new Text("Spinner"), spinner);
        Scene scene = new Scene(root, 300, 200);
        primaryStage.setTitle("Spinner Example");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    private Spinner<Double> createSpinner() {
        Spinner<Double> spinner = new Spinner<>();
        spinner.setEditable(true);

        SpinnerValueFactory<Double> valueFactory = new SpinnerValueFactory.DoubleSpinnerValueFactory(-20.0, 20.0, 0.0, 0.25);
        spinner.setValueFactory(valueFactory);

        // Custom TextFormatter for spinner editor
        UnaryOperator<TextFormatter.Change> filter = c -> {
            String newText = c.getControlNewText();
            Pattern validText = Pattern.compile("[+-]?\\d{0,2}(\\,\\d{0,2})?");
            if (validText.matcher(newText).matches()) {
                if (newText.length() > 0 && !newText.startsWith("+") && !newText.startsWith("-")) {
                    c.setText("+" + newText);
                }
                return c;
            } else {
                return null;
            }
        };
        TextFormatter<Double> textFormatter = new TextFormatter<>(filter);
        spinner.getEditor().setTextFormatter(textFormatter);

        spinner.setValueFactory(valueFactory);

        return spinner;
    }

    public static void main(String[] args) {
        launch(args);
    }
}
