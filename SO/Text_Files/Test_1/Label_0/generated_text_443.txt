import polars as pl

df = pl.DataFrame(
    {
        "a": [1, 8, 3],
        "b": [4, 5, None],
    }
)

df.with_columns(
    max_column=pl.max_horizontal(["a", "b"]),
    max_column_name=pl.when(pl.col("a") == pl.col("max_column")).then("a")
    .when(pl.col("b") == pl.col("max_column")).then("b")
)

# Test case
assert df.select("max_column_name").to_dict() == {"max_column_name": ["b", "a", "a"]}
