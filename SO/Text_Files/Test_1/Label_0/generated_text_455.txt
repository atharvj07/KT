#!/bin/bash

# Increment version
increment_version() {
    local version_file="uv.config.json"
    local current_version=$(jq -r '.version' $version_file)
    local major minor patch
    IFS='.' read -r major minor patch <<< "$current_version"
    
    patch=$((patch + 1)) # Increment patch version
    new_version="$major.$minor.$patch"

    # Update version in uv.config.json
    jq --arg version "$new_version" '.version = $version' $version_file > tmp.$$.json && mv tmp.$$.json $version_file

    echo "Updated version to $new_version"
}

# Test Case: Run the increment_version function
increment_version
