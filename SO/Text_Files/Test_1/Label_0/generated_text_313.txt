import numpy as np

def calculate_Fourier(points, num_coefficients):
    complex_points = np.array([p[0] + 1j * p[1] for p in points])

    N = len(complex_points)
    t = np.arange(N) / N

    c_k = np.zeros(num_coefficients, dtype=np.complex128)

    for k in range(-num_coefficients//2, num_coefficients//2):
        c_k[k + num_coefficients//2] = np.sum(
            complex_points * np.exp(-2j * np.pi * k * t)
        ) / N
    
    return c_k

def reconstruct_curve(coefficients, num_points):
    t = np.linspace(0, 1, num_points)
    curve = np.zeros(num_points, dtype=np.complex128)

    num_coefficients = len(coefficients)
    for k in range(-num_coefficients//2, num_coefficients//2):
        curve += coefficients[k + num_coefficients//2] * np.exp(2j * np.pi * k * t)

    return curve

import matplotlib.pyplot as plt
import imageio
from Contour_Classifier import contours

png_files = []
num_points = len(contours[0])

for i in range(1, num_points + 1):
    coefficients = calculate_Fourier(contours[0], i)
    curve = reconstruct_curve(coefficients, num_points)

    plt.figure(figsize=(8, 8))
    plt.plot(curve.real, curve.imag, label="Trajectory", color="blue")
    plt.scatter(0, 0, color="black", label="Origin")
    plt.axis("equal")
    plt.title(f"Fourier Series with {i} Coefficients")
    plt.xlabel("Real Part (X)")
    plt.ylabel("Imaginary Part (Y)")
    plt.legend()
    plt.text(-0.5, -0.5, f"Using {i} coefficients", fontsize=12, color="red")

    filename = f"fourier_{i}_coefficients.png"
    plt.savefig(filename)
    plt.close()

    png_files.append(filename)

gif_filename = "fourier_series.gif"
with imageio.get_writer(gif_filename, mode='I', duration=0.5) as writer:
    for filename in png_files:
        image = imageio.imread(filename)
        writer.append_data(image)
