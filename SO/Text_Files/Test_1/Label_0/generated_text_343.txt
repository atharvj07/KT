import numpy as np
from numba import njit

@njit
def compute_y_coordinates(z_coordinates, x_size, z_size):
    y_coordinates = np.empty((x_size, z_size), dtype=np.uint16)
    for i in range(z_size):
        for x in range(x_size):
            y_coordinates[x, i] = np.argmin(np.abs(z_coordinates[x] - i))
    return y_coordinates

# Test case
x_size = 2000
y_size = 2500
z_size = 400
rng = np.random.default_rng(123)
z_coordinates = np.linspace(0, z_size, y_size) + rng.laplace(0, 1, (x_size, y_size))

y_coordinates = compute_y_coordinates(z_coordinates, x_size, z_size)
