import javax.swing.*;
import java.awt.*;
import java.awt.image.BufferedImage;
import java.util.ArrayList;

public class BulletGame extends JPanel {
    private final ArrayList<Bullet> bullets = new ArrayList<>();
    private final BufferedImage bulletImage;

    public BulletGame() {
        bulletImage = new BufferedImage(16, 16, BufferedImage.TYPE_INT_ARGB);
        Graphics2D g2d = bulletImage.createGraphics();
        g2d.setColor(Color.RED);
        g2d.fillOval(0, 0, 16, 16);
        g2d.dispose();

        for (int i = 0; i < 10000; i++) {
            bullets.add(new Bullet(100 + i % 800, 100 + i % 600, i * 0.1, bulletImage));
        }

        Timer timer = new Timer(16, e -> {
            for (Bullet bullet : bullets) {
                bullet.update();
            }
            repaint();
        });
        timer.start();
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        Graphics2D g2d = (Graphics2D) g;
        for (Bullet bullet : bullets) {
            bullet.draw(g2d);
        }
    }

    private static class Bullet {
        int x, y;
        double angle;
        BufferedImage img;

        Bullet(int x, int y, double angle, BufferedImage originalImg) {
            this.x = x;
            this.y = y;
            this.angle = angle;
            this.img = originalImg;
        }

        void update() {
            angle += 2;
        }

        void draw(Graphics2D g) {
            int w = img.getWidth();
            int h = img.getHeight();
            g.rotate(Math.toRadians(angle), x + w / 2, y + h / 2);
            g.drawImage(img, x, y, null);
            g.rotate(-Math.toRadians(angle), x + w / 2, y + h / 2);
        }
    }

    public static void main(String[] args) {
        JFrame frame = new JFrame("Touhou-esque Bullet Game");
        BulletGame game = new BulletGame();
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setSize(800, 600);
        frame.add(game);
        frame.setVisible(true);
    }
}
