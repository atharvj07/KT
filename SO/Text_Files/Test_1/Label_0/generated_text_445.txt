class FooMeta(type):
    def __len__(cls):
        return 9000


class GoodBar(metaclass=FooMeta):
    def __len__(self):
        return 9001


class BadBar(metaclass=FooMeta):
    @classmethod
    def __len__(cls):
        return 9002

    @classmethod
    def __getattr__(cls, name):
        if name == "__len__":
            return cls.__len__
        raise AttributeError


# Test cases
assert len(GoodBar) == 9000
assert len(GoodBar()) == 9001
assert GoodBar.__len__() == 9001
assert len(BadBar) == 9002
assert len(BadBar()) == 9002
assert BadBar.__len__() == 9002
