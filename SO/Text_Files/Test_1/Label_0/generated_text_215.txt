public class CircularSearch {

    public static boolean search(int[][] mat, int num) {
        int n = mat.length;
        int row = 0, col = n - 1;

        while (row < n && col >= 0) {
            if (mat[row][col] == num) return true;
            if (mat[row][col] > num) col--;
            else row++;
        }

        return false;
    }

    public static void main(String[] args) {
        int[][] mat = {
            {10, 20, 30, 40},
            {15, 25, 35, 45},
            {27, 29, 37, 48},
            {32, 33, 39, 50}
        };
        System.out.println(search(mat, 22)); // true
        System.out.println(search(mat, 23)); // false
    }
}
