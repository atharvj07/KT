from typing import Callable, TypeVar, Protocol
from functools import wraps

T = TypeVar("T", bound=Callable[[int], int])


class MemoizedFunction(Protocol):
    memory: dict[int, int]

    def __call__(self, n: int) -> int: ...


def remember(f: T) -> MemoizedFunction:
    @wraps(f)
    def wrapper(n: int) -> int:
        if n not in wrapper.memory:
            wrapper.memory[n] = f(n)
        return wrapper.memory[n]

    wrapper.memory = {}
    return wrapper  # type: ignore


@remember
def fib(n: int) -> int:
    if n < 2:
        return 1
    return fib(n - 2) + fib(n - 1)


print(fib(100))  # 573147844013817084101
print(fib.memory[50])  # 20365011074
