public void copyDatabaseFromAssets() {
    Context context = requireContext();
    AssetManager assetManager = context.getAssets();
    
    File databasesFolder = new File(context.getDataDir(), "databases");
    if (!databasesFolder.exists()) {
        databasesFolder.mkdirs();
    }

    try {
        String[] files = assetManager.list("database/test");
        if (files != null) {
            for (String filename : files) {
                File outFile = new File(databasesFolder, filename);
                if (!outFile.exists()) {
                    try (InputStream in = assetManager.open("database/test/" + filename);
                         OutputStream out = new FileOutputStream(outFile)) {
                        byte[] buffer = new byte[1024];
                        int read;
                        while ((read = in.read(buffer)) != -1) {
                            out.write(buffer, 0, read);
                        }
                        out.flush();
                    }
                }
            }
        }
    } catch (IOException e) {
        Log.e("DatabaseCopy", "Failed to copy database file", e);
    }
}

// Test case
public void testDatabaseCopy() {
    copyDatabaseFromAssets();
    File dbFile = new File(requireContext().getDataDir(), "databases/test.db");
    if (dbFile.exists()) {
        Log.i("Test", "Database copied successfully");
    } else {
        Log.e("Test", "Database copy failed");
    }
}
