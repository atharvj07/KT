import typing

def func():
    class B:
        pass
    class A:
        some_b: "B" 
    # Use typing.get_type_hints in the function's scope with globalns for proper evaluation
    print(typing.get_type_hints(A, globalns=globals())) 

func()  # Should not raise NameError, prints {'some_b': <class '__main__.B'>}
