import pytest
from databases import Database
from sqlalchemy.ext.asyncio import AsyncSession, create_async_engine
from sqlalchemy.orm import sessionmaker
from factory import Factory, Faker
from yourapp.models import YourModel  # Replace with your actual model

DATABASE_URL = "postgresql+asyncpg://user:password@localhost/dbname"  # Adjust your DB URL

engine = create_async_engine(DATABASE_URL, echo=True)
SessionLocal = sessionmaker(
    bind=engine, class_=AsyncSession, expire_on_commit=False
)

@pytest.fixture
async def async_session():
    async with SessionLocal() as session:
        yield session

class AsyncFactory(Factory):
    class Meta:
        model = YourModel
    field1 = Faker('name')
    field2 = Faker('email')

@pytest.mark.asyncio
async def test_create_model(async_session):
    instance = AsyncFactory.build()
    async_session.add(instance)
    await async_session.commit()
    await async_session.refresh(instance)
    assert instance.field1 is not None
    assert instance.field2 is not None
