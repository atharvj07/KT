import polars as pl
from datetime import datetime
import pytz

df = pl.DataFrame({
    "time": [datetime(2023, 4, 3, 2), datetime(2023, 4, 4, 3), datetime(2023, 4, 5, 4)],
    "tzone": ["Asia/Tokyo", "America/Chicago", "Europe/Paris"]
}).with_columns(pl.col("time").dt.replace_time_zone("UTC"))

def convert_timezone(time, timezone):
    return time.replace(tzinfo=pytz.utc).astimezone(pytz.timezone(timezone)).hour

df = df.with_columns(
    pl.struct(["time", "tzone"]).map_elements(lambda x: convert_timezone(x["time"], x["tzone"])).alias("hour")
)

print(df)
