import polars as pl

# Sample data
df = pl.DataFrame({
    "col1": [1, None, 3, None, 5],
    "col2": [10, 20, 30, 40, 50]
})

# Fill null values in 'col1' with values from 'col2'
df_lazy = df.lazy()
result = df_lazy.with_columns([
    pl.col("col1").fill_none(pl.col("col2")).alias("col1_filled")
])

# Collect the result
result.collect()
