from typing import TypedDict

class A(TypedDict):
    a: int
    b: int

    @classmethod
    def create(cls, a: int = 0, b: int = 1) -> A:
        return A(a=a, b=b)

a = A.create(a=4)
# {"a": 4, "b": 1}

from dataclasses import dataclass

@dataclass
class A:
    a: int = 0
    b: int = 1

from dataclasses import asdict

a = A(a=4)
asdict(a)  # {"a": 4, "b": 1}
