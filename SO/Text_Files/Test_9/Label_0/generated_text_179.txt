import java.io.*;
import org.apache.poi.hssf.usermodel.*;
import org.apache.poi.ss.usermodel.*;

public class ExpXls {
    public static void main(String[] args) {
        try (OutputStream os = new FileOutputStream("test.xls");
             HSSFWorkbook wb = new HSSFWorkbook()) {
            Sheet sh = wb.createSheet("test");
            Row r = sh.createRow(0);
            HSSFCellStyle[] stl = new HSSFCellStyle[50];

            for (int i = 0; i < 50; i++) {
                stl[i] = wb.createCellStyle();
                stl[i].setFillForegroundColor(IndexedColors.LIGHT_BLUE.getIndex());
                stl[i].setFillPattern(FillPatternType.SOLID_FOREGROUND);
            }

            for (int i = 0; i < 50; i++) {
                Cell c = r.createCell(i);
                c.setCellValue(i + 1);
                c.setCellStyle(stl[i]);
            }

            wb.write(os);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
