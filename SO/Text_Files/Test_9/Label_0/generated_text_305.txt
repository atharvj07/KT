import numpy as np
import time

s = 3000

et1 = []
et2 = []

n = 50
for _ in range(n):
    a = np.random.rand(s, s)
    st = time.time()
    r1 = np.matmul(a, a)
    et1.append(time.time() - st)

for _ in range(n):
    b = np.random.rand(s, s)
    st = time.time()
    r2 = np.matmul(b, b.T)
    et2.append(time.time() - st)

avg1 = np.mean(et1)
avg2 = np.mean(et2)

print(f"Avg time A @ A: {avg1}")
print(f"Avg time B @ B.T: {avg2}")

import matplotlib.pyplot as plt
plt.plot(range(n), et1, label='A @ A')
plt.plot(range(n), et2, label='B @ B.T')
plt.xlabel('Iteration')
plt.ylabel('Execution Time')
plt.title('Matrix Multiplication Time Comparison')
plt.legend()
plt.show()

np.show_config()
