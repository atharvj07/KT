public class FxTest extends Application {
    public static class Fp {
        private final String p;
        public Fp(String p) { this.p = p; }
        public String getP() { return p; }
    }

    @Override
    public void start(Stage st) {
        TableView<Fp> tv = new TableView<>();
        TableColumn<Fp, String> tc = new TableColumn<>("Path");
        tc.setCellValueFactory(new PropertyValueFactory<>("p"));
        tc.setPrefWidth(200);
        tc.setCellFactory(c -> new TableCell<>() {
            private final Text t = new Text();
            @Override
            protected void updateItem(String i, boolean e) {
                super.updateItem(i, e);
                if (e || i == null) { setGraphic(null); }
                else {
                    int m = 20;
                    String s = i.length() > m ? ".." + i.substring(i.length() - m) : i;
                    t.setText(s);
                    setGraphic(t);
                }
            }
        });
        tv.getColumns().add(tc);
        tv.getItems().addAll(
            new Fp("/usr/local/bin/someverylongdirectoryname/file.txt"),
            new Fp("/usr/local/bin/someverylongdirectoryname/file2.txt")
        );
        st.setScene(new Scene(tv, 300, 200));
        st.show();
    }

    public static void main(String[] args) { launch(args); }
}
