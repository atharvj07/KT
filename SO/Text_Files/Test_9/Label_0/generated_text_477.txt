import polars as pl

df = pl.DataFrame({
    'col1': [1, 2, 3],
    'col2': [4, 5, 6],
    'col3': [7, 8, 9],
    'col4': [10, 11, 12],
    'col5': [13, 14, 15],
    'col6': [16, 17, 18],
    'col7': [19, 20, 21]
})

cols = df.columns
first_cols = ['col2', 'col1', 'col3', 'col4', 'col5']
rest_cols = [col for col in cols if col not in first_cols]

df_new = df.select(first_cols + rest_cols)
print(df_new)
