(MockitoExtension.class)
class TestCls {
    @Mock Repo repo;
    @Mock Cvt cvt;
    @InjectMocks Svc svc;
    @Mock User usr;
    @Mock Ctx ctx;
    @Mock Req req;
    @Mock Spc spc;
    @Mock Pg pg;
    @Mock Res res;
    @Mock Pageable pgbl;
    @Mock Page<Res> pgr;
    @MockConstruction(PageImpl.class) MockedConstruction<PageImpl<Res>> pgMock;

    @Test
    void get_pg() {
        when(cvt.toSpc(usr, req)).thenReturn(spc);
        when(PageRequest.of(1, 100)).thenReturn(pgbl);
        when(repo.findAll(spc, pgbl)).thenReturn(pgr);
        when(cvt.toRes(ctx)).thenReturn(res);
        
        pgMock = Mockito.mockConstruction(PageImpl.class, 
            (m, c) -> {
                when(m.getTotalElements()).thenReturn(123456L);
                when(m.getTotalPages()).thenReturn(1);
                when(m.getContent()).thenReturn(List.of(res));
            });

        Page<Res> act = svc.getPg(usr, req);

        Assertions.assertThat(act.getTotalElements()).isEqualTo(123456L);
        Assertions.assertThat(act.getTotalPages()).isEqualTo(1);
        Assertions.assertThat(act.getContent()).isEqualTo(List.of(res));
    }
}
