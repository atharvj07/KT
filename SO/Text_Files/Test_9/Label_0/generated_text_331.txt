from sqlmodel import SQLModel, Field, Relationship
from sqlalchemy.orm import relationship

class Node(SQLModel, table=True):
    id: int = Field(default=None, primary_key=True)
    data: str
    parent_id: int = Field(default=None, foreign_key="node.id")
    
    parent: "Node" = Relationship(back_populates="children", sa_relationship_kwargs={"remote_side": [id]})
    children: list["Node"] = Relationship(back_populates="parent")

def create_node(session, data, parent=None):
    node = Node(data=data, parent=parent)
    session.add(node)
    session.commit()
    session.refresh(node)
    return node
