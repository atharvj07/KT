import polars as pl

df = pl.DataFrame(
    {"name": list("abcdef"), "age": [21, 31, 32, 53, 45, 26], "cty": list("AABBBC")}
)

df.group_by("cty").agg(
    (srt := pl.col("name").sort_by("age")),
    srt.first().alias("srt_1"),
    srt.slice(2, 1).fill_null(None).alias("srt_2"),
    srt.last().alias("srt_-1")
)
