import polars as pl
from datetime import datetime

d1 = pl.DataFrame({
    "id": [1, 2, 3, 4, 5],
    "ini": ["2022-01-01", "2022-01-02", "2022-01-03", "2022-01-04", "2022-01-05"],
    "fin": ["2022-01-03", "2022-01-06", "2022-01-07", "2022-01-09", "2022-01-07"],
    "val": [10, 20, 30, 40, 50]
}).with_columns(pl.col("ini").str.to_date(), pl.col("fin").str.to_date())

d2 = pl.datetime_range(
    start=datetime(2022, 1, 1),
    end=datetime(2022, 1, 7),
    interval="1d",
    eager=True
).to_frame("dat")

res = d2.with_columns(
    pl.sum(
        pl.when(
            (d1["ini"] <= pl.col("dat")) & (d1["fin"] >= pl.col("dat"))
        ).then(d1["val"]).otherwise(0)
    ).alias("sum")
)

print(res)
