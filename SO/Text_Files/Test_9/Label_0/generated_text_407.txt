import pprint
from langchain_community.vectorstores import FAISS
from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain.docstore.document import Document
from tqdm import tqdm

mod = "sentence-transformers/multi-qa-MiniLM-L6-cos-v1"
emb = HuggingFaceEmbeddings(model_name=mod)

def main():
    d1 = Document(page_content="The sky is blue.", metadata={"doc_id": "10"})
    d2 = Document(page_content="The forest is green", metadata={"doc_id": "62"})
    docs = [d1, d2]

    for d in tqdm(docs, desc="Proc"):
        d.metadata['sum'] = 'hello'

    pprint.pprint(docs)
    db = FAISS.from_documents(docs, emb)
    db.save_local("faiss_idx")
    new_db = FAISS.load_local("faiss_idx", emb)

    q = "Which color is the sky?"
    res = new_db.similarity_search_with_score(q)
    print('Got:', res)
    print('Meta:', res[0][0].metadata)

if __name__ == '__main__':
    main()
