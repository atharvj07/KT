import org.junit.jupiter.api.extension.BeforeAllCallback;
import org.junit.jupiter.api.extension.ExtensionContext;
import org.mockito.MockedStatic;
import org.mockito.Mockito;

public class MockUtilExt implements BeforeAllCallback, ExtensionContext.Store.CloseableResource {

    private static MockedStatic<UtilCls> mock;

    @Override
    public void beforeAll(ExtensionContext ctx) {
        if (mock == null) {
            mock = Mockito.mockStatic(UtilCls.class);
            mock.when(UtilCls::genId).thenReturn("mock-id");
        }
    }

    @Override
    public void close() {
        if (mock != null) {
            mock.close();
            mock = null;
        }
    }
}
