 class ResEnv<T> {
    @Schema
    private boolean sts;
    @Schema
    private T res;

    public ResEnv(T res) {
        this.sts = true;
        this.res = res;
    }

    public boolean isSts() {
        return sts;
    }

    public T getRes() {
        return res;
    }
}

@RestController
public class UsrCtl {
    private final UsrSrv srv;

    public UsrCtl(final UsrSrv srv) {
        this.srv = srv;
    }

    @PostMapping("/usr")
    @PreAuthorize("hasAuthority('create:user')")
    public ResponseEntity<ResEnv<UsrDto>> crt(final @RequestBody @Valid CrtUsr cmd) {
        return ResponseEntity
                .status(HttpStatus.CREATED)
                .body(new ResEnv<>(srv.crt(cmd)));
    }
}
