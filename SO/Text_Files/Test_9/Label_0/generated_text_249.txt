 class TMPFilter implements WebFilter {
    private long reqTime = System.nanoTime();

    @Override
    public Mono<Void> filter(ServerWebExchange ex, WebFilterChain chain) {
        long deltaTime = System.nanoTime() - reqTime;
        ex.getResponse().getHeaders().add("server-timing-test", "test");
        System.out.println(deltaTime);
        return chain.filter(ex)
                .doOnRequest(req -> {
                    reqTime = System.nanoTime();
                })
                .doOnSuccess(arg -> {
                    long delta = System.nanoTime() - reqTime;
                    ex.getResponse().getHeaders().set("server-timing-success", Long.toString(delta));
                })
                .doOnError((arg) -> {
                    long delta = System.nanoTime() - reqTime;
                    ex.getResponse().getHeaders().set("server-timing-error", Long.toString(delta));
                })
                .doFinally((arg) -> {
                    long delta = System.nanoTime() - reqTime;
                    ex.getResponse().getHeaders().set("server-timing-finally", Long.toString(delta));
                });
    }
}
