class MainAct : FlutterActivity() {
    private val res = registerForActivityResult(PianoIdAuthResultContract()) { r ->
        when (r) {
            null -> {}
            is PianoIdAuthSuccessResult -> {
                val tkn = r.token
                val isNew = r.isNewUser
                if (tkn?.emailConfirmationRequired == true) {}
            }
            is PianoIdAuthFailureResult -> {
                val err = r.exception
            }
        }
    }

    override fun onCreate(s: Bundle?) {
        super.onCreate(s)
        MethodChannel(flutterEngine!!.dartExecutor.binaryMessenger, "signInChannel").setMethodCallHandler { call, res ->
            if (call.method == "testSignIn") {
                try {
                    res.launch(PianoId.signIn())
                } catch (e: Exception) {
                    Toast.makeText(this, e.message, Toast.LENGTH_SHORT).show()
                }
            }
        }
    }
}
