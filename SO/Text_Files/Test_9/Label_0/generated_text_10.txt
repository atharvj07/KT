import java.util.*;

public class ZigZag {
    public static int sol(List<Integer> lst) {
        List<Integer> up = new ArrayList<>();
        List<Integer> dn = new ArrayList<>();
        List<Integer> up2 = new ArrayList<>();

        TreeSet<Integer> set = new TreeSet<>(lst);
        for (int x : set) up.add(x);

        for (int i = up.size() - 1; i >= 0; i--) dn.add(up.get(i));

        for (int x : set) {
            if (x > dn.get(dn.size() - 1)) up2.add(x);
        }

        return up.size() + dn.size() + up2.size() - 2;
    }

    public static void main(String[] a) {
        System.out.println(sol(List.of(1, 3, 5, 4, 2, 6, 8, 7, 9)));
        System.out.println(sol(List.of(5, 5, 2, 1, 3, 4, 5)));
        System.out.println(sol(List.of(1, 100)));
        System.out.println(sol(List.of(2, 1, 3, 3, 1, 2, 1, 2, 3)));
    }
}
