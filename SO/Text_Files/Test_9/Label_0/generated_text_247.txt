 interface MMapper {

    @Mapping(target = "agentName", source = "agentName")
    @Mapping(target = "agentIdentityType", source = "agentIdentityType")
    @Mapping(target = "agentIdentityNo", source = "agentIdentityNo")
    @Mapping(target = "agentIdentityExpireAt", source = "agentIdentityExpireAt")
    AgentInfo toAgentInfo(WealthProdAccountInfoDTO dto);

    default AgentInfo toAgentInfoWithNullCheck(WealthProdAccountInfoDTO dto) {
        if (dto == null) {
            return null;
        }
        if (dto.getAgentName() == null && dto.getAgentIdentityType() == null && 
            dto.getAgentIdentityNo() == null && dto.getAgentIdentityExpireAt() == null) {
            return null;
        }
        return toAgentInfo(dto);
    }
}
