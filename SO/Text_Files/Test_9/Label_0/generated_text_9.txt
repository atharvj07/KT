import io.opentelemetry.api.trace.*;
import io.opentelemetry.context.*;

public class MsgTrc {

    private final Tracer trc;

    public MsgTrc(Tracer trc) {
        this.trc = trc;
    }

    public void procMsg(String tid) {
        SpanContext ctx = SpanContext.createFromRemoteParent(
            tid, "0000000000000000", TraceFlags.getSampled(), TraceState.getDefault()
        );
        Context c = Context.current().with(Span.wrap(ctx));

        try (Scope s = c.makeCurrent()) {
            Span sp = trc.spanBuilder("consumeMsg").setParent(c).startSpan();
            sp.end();
        }
    }
}
