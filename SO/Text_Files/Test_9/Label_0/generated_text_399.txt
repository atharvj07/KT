import polars as pl
import plotly.express as px

df = pl.DataFrame({
    "sid": [1, 2, 3],
    "eng": ["one:one,two:two", "one:two,two:one", "one:one"],
    "dur": [123, 456, 789]
})

df = df.with_columns(pl.col("eng").str.split(",")).explode("eng")
df = df.with_columns(pl.col("dur") / df.groupby("sid").count().select("eng"))

fig = px.bar(df, x="sid", y="dur", color="eng")
fig.show()
