import java.util.*;

public class Gen {

    public static void main(String[] a) {
        String[] p1 = {"A", "B", "C"};
        String[] p2 = {"1", "2", "3"};
        String[] p3 = {"X", "Y", "Z"};

        List<String> out = gen(p1, p2, p3);
        out.forEach(System.out::println);
    }

    static List<String> gen(String[]... p) {
        List<String> res = new ArrayList<>();
        loop(p, 0, "", res);
        return res;
    }

    static void loop(String[][] p, int idx, String cur, List<String> res) {
        if (idx == p.length) {
            res.add(cur);
            return;
        }
        for (String s : p[idx]) {
            loop(p, idx + 1, cur + s, res);
        }
    }
}
