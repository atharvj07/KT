 class StrmCfg {

    @Bean
    public Function<KStream<String, Inp>, Map<String, KStream<String, Outp>>> myfn() {
        return inp -> {
            Map<String, KStream<String, Outp>> out = new HashMap<>();
            List<String> tnts = List.of("A", "B");
            for (String tnt : tnts) {
                KStream<String, Inp> flt = inp.filter((k, v) -> k.startsWith(tnt));
                out.put(tnt, flt.map((k, v) -> KeyValue.pair(k, Outp.mk(v))));
            }
            return out;
        };
    }
}
