public class PatternSearch {

    public static void main(String[] args) {
        int[][] P = {
            {2, 3, 0},
            {0, 1, 5}
        };

        int[][] T = {
            {2, 3, 4, 3, 6},
            {4, 1, 5, 7, 8},
            {9, 1, 2, 3, 1},
            {2, 4, 5, 1, 5},
            {3, 1, 9, 0, 2}
        };

        PatternSearch ps = new PatternSearch();
        ps.search(T, P);
    }

    public void search(int[][] T, int[][] P) {
        int m = T.length, n = T[0].length;
        int u = P.length, v = P[0].length;
        
        for (int i = 0; i <= m - u; i++) {
            for (int j = 0; j <= n - v; j++) {
                if (match(T, P, i, j, u, v)) {
                    System.out.println("Pattern found at (" + i + "," + j + ")");
                }
            }
        }
    }
