from pathlib import Path
from yolov5 import YOLOv5

def load_model(path):
    return YOLOv5(path)

def train_model(model, data_path, imgsz):
    model.train(data=data_path, imgsz=imgsz)

def detect(model, img_path):
    return model(img_path)

model = load_model('yolov5s.pt')
train_model(model, Path('data/custom'), 640)
detect(model, 'path/to/image.jpg')
