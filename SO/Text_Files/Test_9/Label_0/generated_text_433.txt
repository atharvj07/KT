import numpy as np
from sklearn.preprocessing import PolynomialFeatures
from sklearn.linear_model import LinearRegression

def fit_poly(x, y, d=2):
    p = PolynomialFeatures(d)
    xp = p.fit_transform(x)
    m = LinearRegression()
    m.fit(xp, y)
    return m, p

def grd_hes(m, p, x):
    j = p.transform(x)
    c = m.coef_
    g = c[:, 1:].reshape(-1, x.shape[1])
    h = np.zeros((x.shape[1], x.shape[1]))
    idx = 1
    for i in range(x.shape[1]):
        h[i, i] = c[0, idx + i]
        idx += x.shape[1] - i
    return g, h

x = np.random.rand(10, 2)
y = np.random.rand(10, 1)
m, p = fit_poly(x, y)
g, h = grd_hes(m, p, x[:1])
