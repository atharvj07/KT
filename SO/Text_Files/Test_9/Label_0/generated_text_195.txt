import org.apache.poi.ss.formula.*;
import org.apache.poi.ss.formula.ptg.*;
import org.apache.poi.ss.util.*;
import java.util.*;

public class FmSh {
    public static void main(String[] args) {
        String fm = "=(A2+B2)";
        String org = "C2";
        String des = "D2";
        System.out.println(shf(fm, org, des));
    }

    public static String shf(String fm, String org, String des) {
        Ptgs ptg = FormulaParser.parse(fm, null, FormulaType.CELL, 0);
        CellRef o = new CellRef(org);
        CellRef d = new CellRef(des);
        int dr = d.getRow() - o.getRow();
        int dc = d.getCol() - o.getCol();
        
        for (Ptg p : ptg) {
            if (p instanceof RefPtgBase) {
                RefPtgBase r = (RefPtgBase) p;
                if (!r.isColumnAbsolute()) r.setColumn(r.getColumn() + dc);
                if (!r.isRowAbsolute()) r.setRow(r.getRow() + dr);
            }
        }
        
        return FormulaRenderer.toFormulaString(null, ptg);
    }
}

class CellRef {
    private final CellReference ref;

    public CellRef(String addr) {
        this.ref = new CellReference(addr);
    }

    public int getRow() {
        return ref.getRow();
    }

    public int getCol() {
        return ref.getCol();
    }
}
