public void pC(Graphics g) {
    super.pC(g);
    if (A.getImg() != null) {
        float wF = getP().getW() / 1280F;
        float hF = getP().getH() / 859F;
        int w, h, x, y;
        if (wF < hF) {
            h = getP().getH();
            w = (int) (1280F * hF);
            x = getP().getW() / 2 - w / 2;
            y = 0;
        } else {
            w = getP().getW();
            h = (int) (859F * wF);
            x = 0;
            y = getP().getH() / 2 - h / 2;
        }
        g.dI(A.getImg(), x, y, w, h, this);
    }
}
