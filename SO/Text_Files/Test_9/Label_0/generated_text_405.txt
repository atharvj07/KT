import pandas as pd

def cnt_3w(df):
    df = df.sort_values(["group_id", "sent_time"])
    res = 0
    for g, d in df.groupby("group_id"):
        d = d[["user_id", "touchpoint"]].values
        for i in range(len(d) - 2):
            if d[i][1] == 1 and d[i+1][1] == 2 and d[i+2][1] == 3 and d[i][0] == d[i+2][0] and d[i][0] != d[i+1][0]:
                res += 1
    return res
