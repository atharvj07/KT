class Rect {
    constructor(l, w) {
        this.name = "Rect";
        this.l = l;
        this.w = w;
    }

    shout() {
        console.log(
            `I AM A ${this.name.toUpperCase()} AND MY AREA IS ${this.l * this.w}`
        );
    }

    hello() {
        return "Rect: hello";
    }
}

class Squa extends Rect {
    constructor(l) {
        super(l, l);
        this.name = "Squa";
    }

    sqHello() {
        const h = super.hello();
        return "Squa:" + h.split(":")[1];
    }
}

const r = new Rect(6, 4);
r.shout();

const s = new Squa(5);
s.shout();

console.log(s.sqHello());
