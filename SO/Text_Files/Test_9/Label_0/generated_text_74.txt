public interface Obj {
    void act();
}

@Component
@Scope("prototype")
public class A implements Obj {
    private final int x, y;

    public A(int x, int y) {
        this.x = x;
        this.y = y;
    }

    public void act() {
        System.out.println("A: " + x + ", " + y);
    }
}

@Component
@Scope("prototype")
public class B implements Obj {
    private final int a, b, c;

    public B(int a, int b, int c) {
        this.a = a;
        this.b = b;
        this.c = c;
    }

    public void act() {
        System.out.println("B: " + a + ", " + b + ", " + c);
    }
}

@Component
@Scope("prototype")
public class C implements Obj {
    private final int n;

    public C(int n) {
        this.n = n;
    }

    public void act() {
        System.out.println("C: " + n);
    }
}

@Component
public class ObjFac {
    private final ApplicationContext ctx;

    @Autowired
    public ObjFac(ApplicationContext ctx) {
        this.ctx = ctx;
    }

    public Obj make(String key, int... vals) {
        switch (key) {
            case "A": return ctx.getBean(A.class, vals[0], vals[1]);
            case "B": return ctx.getBean(B.class, vals[0], vals[1], vals[2]);
            case "C": return ctx.getBean(C.class, vals[0]);
            default: throw new IllegalArgumentException("No bean for " + key);
        }
    }
}

public class App {
    public static void main(String[] args) {
        AnnotationConfigApplicationContext ctx = new AnnotationConfigApplicationContext("pkg");
        ObjFac fac = ctx.getBean(ObjFac.class);

        Obj a = fac.make("A", 1, 2);
        Obj b = fac.make("B", 3, 4, 5);
        Obj c = fac.make("C", 1);

        a.act();
        b.act();
        c.act();

        ctx.close();
    }
}
