public class EqDiv {
    public static boolean chk(int[] arr) {
        if (arr.length % 2 != 0) return false;
        return chk(arr, 0, 0, 0, 0);
    }

    private static boolean chk(int[] arr, int idx, int s1, int c1, int c2) {
        if (idx == arr.length) return c1 == c2 && s1 * 2 == sum(arr, 0);
        return chk(arr, idx + 1, s1 + arr[idx], c1 + 1, c2) ||
               chk(arr, idx + 1, s1, c1, c2 + 1);
    }

    private static int sum(int[] arr, int idx) {
        return idx == arr.length ? 0 : arr[idx] + sum(arr, idx + 1);
    }

    public static void main(String[] args) {
        System.out.println(chk(new int[]{1, 5, 11, 5}));
        System.out.println(chk(new int[]{1, 2, 3, 5}));
    }
}
