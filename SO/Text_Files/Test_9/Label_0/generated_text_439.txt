import numpy.typing as npt
import numpy as np
from typing import Sequence, Union

T = Union[npt.NDArray, Sequence]

def f(a: T) -> int:
    return len(a)  

def g(a: T) -> Union[int, float, complex]:
    return a[0]  

print(f(np.array([0, 1])), g(np.array([0, 1])))
print(f([0, 1]), g([0, 1]))
