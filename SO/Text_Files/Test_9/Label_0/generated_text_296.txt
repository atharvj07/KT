import pandas as pd
import numpy as np

id = ['a','b','c','a','b','e','f','a','b','k']
value = [12,14,3,13,16,7,4,6,10,18]

df = pd.DataFrame({'id':id, 'value':value})

df['cum'] = df.groupby('id')['value'].apply(lambda x: x[::-1].cumsum()[::-1])
df['desired'] = df['cum'].where(~df['id'].duplicated(), np.nan)

df['desired'] = df['desired'].fillna(method='ffill')
print(df)
