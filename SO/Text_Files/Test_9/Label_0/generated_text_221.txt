import java.time.YearMonth;
import java.time.format.DateTimeFormatter;
import java.util.*;

class D {
    private String m;
    private int c;
    public D(String m, int c) { this.m = m; this.c = c; }
    public String gM() { return m; }
    public int gC() { return c; }
    public String toString() { return "Feature: " + m + " Count: " + c; }
}

public class S {
    public static void main(String[] a) {
        List<D> l = new ArrayList<>();
        l.add(new D("october", 46));
        l.add(new D("april", 46));
        l.add(new D("march", 46));
        l.add(new D("november", 30));
        l.add(new D("may", 46));
        l.add(new D("january", 53));
        l.add(new D("december", 30));

        YearMonth t = YearMonth.now();
        List<String> mL = new ArrayList<>();
        for (int i = 0; i < 6; i++) {
            mL.add(t.minusMonths(i).format(DateTimeFormatter.ofPattern("MMMM")).toUpperCase());
        }
        l.sort(Comparator.comparingInt(o -> mL.indexOf(o.gM().toUpperCase())));
        Collections.reverse(l);
        l.forEach(System.out::println);
    }
}
