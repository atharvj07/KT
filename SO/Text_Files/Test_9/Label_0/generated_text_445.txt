class FooMeta(type):
    def __len__(cls):
        return 9000

class GoodBar(metaclass=FooMeta):
    def __len__(self):
        return 9001

class BadBar(metaclass=FooMeta):
    @classmethod
    def __len__(cls):
        return 9002

setattr(BadBar, "__len__", classmethod(lambda cls: 9002))

print(len(BadBar))
print(len(BadBar()))
