import org.springframework.web.bind.annotation.*;
import org.springframework.scheduling.annotation.*;
import org.springframework.http.*;
import java.util.*;
import java.util.concurrent.*;

@RestController
@RequestMapping("/run")
public class RunCtl {

    private final Map<Integer, String> sts = new ConcurrentHashMap<>();
    private final ExecutorService ex = Executors.newSingleThreadExecutor();

    @PostMapping
    public ResponseEntity<String> strt() {
        ex.submit(this::runAll);
        return ResponseEntity.ok("Started");
    }

    @GetMapping
    public ResponseEntity<Map<Integer, String>> chk() {
        return ResponseEntity.ok(sts);
    }

    private void runAll() {
        for (int i = 1; i <= 30; i++) {
            sts.put(i, "Running");
            try {
                Thread.sleep(300000);
                sts.put(i, "Done");
            } catch (Exception e) {
                sts.put(i, "Fail");
            }
        }
    }
}
