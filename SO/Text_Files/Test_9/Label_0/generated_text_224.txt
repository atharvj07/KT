import javax.servlet.*;
import javax.servlet.http.*;

public class MainSrv extends HttpServlet {
    public void init(ServletConfig cfg) throws ServletException {
        super.init(cfg);
        try {
            conn();
        } catch (Exception e) {
            throw new UnavailableException("Fail to conn");
        }
    }

    void conn() throws Exception {
        if (!isUp()) throw new Exception("No conn");
    }

    boolean isUp() {
        return false;
    }
}
