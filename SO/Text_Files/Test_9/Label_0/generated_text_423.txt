from functools import partial
from inspect import signature

def bar(a, b):
    return a / b

def chk(f):
    return str(signature(f))

p1 = partial(bar, 3)
p2 = partial(bar, b=3)
p3 = partial(bar, a=3)

print(chk(p1))
print(chk(p2))
print(chk(p3))

print(p1(6))
print(p2(6))
print(p2(6, b=1))
print(p3(b=6))
