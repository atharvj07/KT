import matplotlib.pyplot as plt
import numpy as np

fig, ax = plt.subplots()
vals = [1000, 2500, 3000, 4500]
x = np.arange(len(vals))
bars = ax.bar(x, vals)

for bar in bars:
    ax.text(
        bar.get_x() + bar.get_width() / 2,
        bar.get_height(),
        f"{bar.get_height():,.2f}",
        ha="center",
        va="bottom",
    )

plt.show()
