package com.auto.ex;

import javafx.application.Application;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import javafx.scene.Scene;
import javafx.scene.control.TextField;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;
import org.controlsfx.control.textfield.AutoCompletionBinding;
import org.controlsfx.control.textfield.TextFields;

public class Main extends Application {

    private static final ObservableList<String> lst1 = FXCollections.observableArrayList(
            "Alice", "Adam", "Alfred", "Amon", "Alfredo", "Al", "Albert");

    private static final ObservableList<String> lst2 = FXCollections.observableArrayList(
            "Bob", "Conner", "Robin", "Fred", "Freddy", "Edward", "Fredward", "Mariam");

    private AutoCompletionBinding<String> bind;

    @Override
    public void start(Stage stg) {
        TextField txt = new TextField();
        txt.setOnKeyTyped(e -> {
            String in = txt.getText().toUpperCase();
            if (bind != null) bind.dispose();
            if (in.length() == 2) {
                bind = TextFields.bindAutoCompletion(txt, in.startsWith("A") ? lst1 : lst2);
                bind.setOnAutoCompleted(v -> System.out.println("You picked " + v.getCompletion()));
            }
        });

        VBox box = new VBox(10, txt);
        stg.setScene(new Scene(box, 300, 200));
        stg.setTitle("AutoComplete");
        stg.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
