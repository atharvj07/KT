import org.web3j.crypto.*;
import java.math.BigInteger;
import java.nio.charset.StandardCharsets;

public class SigVer {
    public boolean chk(String adr, String sig, String nc) {
        try {
            String pre = "\u0019Ethereum Signed Message:\n" + nc.length() + nc;
            byte[] msg = pre.getBytes(StandardCharsets.UTF_8);
            byte[] msgH = Hash.sha3(msg);
            Sign.SignatureData sigD = sigDat(sig);
            BigInteger pubK = Sign.signedMessageToKey(msgH, sigD);
            String recAdr = "0x" + Keys.getAddress(pubK);
            return recAdr.equalsIgnoreCase(adr);
        } catch (Exception e) {
            return false;
        }
    }

    private Sign.SignatureData sigDat(String sig) {
        byte[] sigB = Numeric.hexStringToByteArray(sig);
        byte v = sigB[64];
        if (v < 27) v += 27;
        return new Sign.SignatureData(v, 
                Arrays.copyOfRange(sigB, 0, 32), 
                Arrays.copyOfRange(sigB, 32, 64));
    }
}
