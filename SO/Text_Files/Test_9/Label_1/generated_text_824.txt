def glimpse(df):
    print(f"Rows: {df.shape[0]}")
    print(f"Columns: {df.shape[1]}")
    for col in df.columns:
        print(f"$ {col} <{df[col].dtype}> {df[col].head().values}")

import pandas as pd

df = pd.DataFrame(
    {"column_one": ["A", "B", "C", "D"], "column_two": [1, 2, 3, 4]}
)

glimpse(df)

# Output
Rows: 4
Columns: 2
$ column_one <object> ['A' 'B' 'C' 'D']
$ column_two <int64> [1 2 3 4]
