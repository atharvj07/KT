import matplotlib.pyplot as plt

with open('notebook.txt', encoding='utf-8') as file:
    contents = file.readlines()

newList = []
for content in contents:
    newContent = content.replace('\n', '')
    money = newContent.split(':')[-1]
    newList.append(int(money))

x = [1, 2, 3, 4, 5, 6]
y = newList
plt.plot(x, y, 'r')
plt.xlabel('month')
plt.ylabel('money')
plt.legend()
plt.show()
