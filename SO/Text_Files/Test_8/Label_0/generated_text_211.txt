import com.fasterxml.jackson.core.JsonProcessingException;
import jakarta.persistence.EntityManager;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.io.PrintWriter;
import java.util.Iterator;
import java.util.stream.Stream;

public class EmployeeStreamWriter {

    private final EmployeeRepository empRepo;
    private final EntityManager entityManager;

    public EmployeeStreamWriter(EmployeeRepository empRepo, EntityManager entityManager) {
        this.empRepo = empRepo;
        this.entityManager = entityManager;
    }

    public void writeEmployees(HttpServletResponse response) {
        try (Stream<Employee> dataStream = empRepo.findAllStream()) {
            response.setHeader("Content-Type", "application/json");
            PrintWriter respWriter = response.getWriter();
            Iterator<Employee> iterator = dataStream.iterator();

            respWriter.write("[");
            if (iterator.hasNext()) {
                Employee first = iterator.next();
                respWriter.write(jsonSerialize(first));
                entityManager.detach(first);
            }
            while (iterator.hasNext()) {
                Employee data = iterator.next();
                respWriter.write(",");
                respWriter.write(jsonSerialize(data));
                entityManager.detach(data);
            }
            respWriter.write("]");
            respWriter.flush();
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

        return "{}"; // placeholder for actual serialization logic
    }
}
