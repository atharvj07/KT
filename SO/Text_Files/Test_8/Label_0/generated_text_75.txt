public class StrictFpExample {
    public static void main(String[] args) {
        double a = Double.MAX_VALUE;
        double b = 2.0;
        double c = a * b;
        double d = c / 4;  

        System.out.println("Without strictfp: " + d);

        StrictFpClass strictFpClass = new StrictFpClass();
        strictFpClass.runStrictFp();
    }
}

strictfp class StrictFpClass {
    public void runStrictFp() {
        double a = Double.MAX_VALUE;
        double b = 2.0;
        double c = a * b;
        double d = c / 4;  

        System.out.println("With strictfp: " + d);
    }
}
