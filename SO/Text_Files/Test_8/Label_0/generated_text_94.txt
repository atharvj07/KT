import java.lang.annotation.*;
import java.lang.reflect.Field;

@Target(ElementType.FIELD)
@Retention(RetentionPolicy.RUNTIME)
@Documented
@interface Annotation {
    Class<?> className() default Object.class;
}

class AnotherClass {}

class Main {
    @Annotation
    private AnotherClass annotatedAttribute;

    @Annotation(className = String.class)
    private String explicitAttribute;
}

public class AnnotationProcessor {
    public static void main(String[] args) throws Exception {
        for (Field field : Main.class.getDeclaredFields()) {
            Annotation annotation = field.getAnnotation(Annotation.class);
            if (annotation != null) {
                Class<?> clazz = annotation.className() == Object.class ? field.getType() : annotation.className();
                System.out.println("Field: " + field.getName() + ", Class: " + clazz.getSimpleName());
            }
        }
    }
}
