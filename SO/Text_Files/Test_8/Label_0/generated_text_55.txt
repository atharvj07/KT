import java.util.*;

public class StringCombinations {

    public static void main(String[] args) {
        String[] first  = {"A", "B", "C"};
        String[] second = {"1", "2", "3"};
        String[] third  = {"X", "Y", "Z"};

        List<String> result = createStrings(first, second, third);
        result.forEach(System.out::println);
    }

    static List<String> createStrings(String[]... strs) {
        List<String> result = new ArrayList<>();
        generateCombinations(strs, 0, "", result);
        return result;
    }

    static void generateCombinations(String[][] arrays, int index, String current, List<String> result) {
        if (index > arrays.length) {
            result.add(current);
            return;
        }

        if (index == arrays.length) return;

        for (String s : arrays[index]) {
            generateCombinations(arrays, index + 1, current + s, result);
        }
    }
}
