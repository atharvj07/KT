import click
import boto3
import os
from pathlib import Path
from importlib_resources import files

s3_client = boto3.client('s3')

@click.command()
@click.argument('bucket_name')
def upload_files(bucket_name):
    base_path = Path(__file__).parent / "../resources/artifacts"  
    if not base_path.exists():
        click.echo("Resources folder not found, check installation.")
        return

    for file_path in base_path.iterdir():
        if file_path.is_file():
            s3_client.upload_file(str(file_path), bucket_name, file_path.name)
            click.echo(f"Uploaded {file_path.name}")

if __name__ == "__main__":
    upload_files()
