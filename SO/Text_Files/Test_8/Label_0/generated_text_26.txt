import java.util.concurrent.atomic.AtomicReference;

public class DemoSecurityProblem {
    static final AtomicReference<DemoSecurityProblem> instance = new AtomicReference<>();

    public DemoSecurityProblem() {
        if (true) {
            throw new RuntimeException("Constructor failed!");
        }
    }

    @Override
    protected void finalize() throws Throwable {
        instance.set(this);
    }

    public void doSomething() {
        System.out.println("Executing dangerous operation!");
    }

    public static void main(String[] args) {
        try {
            new MaliciousSubClass();
        } catch (Exception ignored) { }

        System.gc();

        if (instance.get() != null) {
            instance.get().doSomething();
        }
    }
}

class MaliciousSubClass extends DemoSecurityProblem { }
