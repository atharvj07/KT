import com.google.auto.service.AutoService;
import javax.annotation.processing.*;
import javax.lang.model.SourceVersion;
import javax.lang.model.element.*;
import java.util.*;

@SupportedAnnotationTypes("com.company.generator.GenerateAnnotation")
@SupportedSourceVersion(SourceVersion.RELEASE_21)
@AutoService(Processor.class)
public class GenerateAnnotationProcessor extends AbstractProcessor {
    
    @Override
    public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment roundEnv) {
        for (Element element : roundEnv.getElementsAnnotatedWith(GenerateAnnotation.class)) {
            if (element.getKind() == ElementKind.ENUM) {
                String enumName = element.getSimpleName().toString();
                String packageName = processingEnv.getElementUtils().getPackageOf(element).toString();
                
                String annotationCode = "package " + packageName + ";\n\n" +
                        "public @interface " + enumName + "Annotation {\n" +
                        "    " + enumName + " value();\n" +
                        "}";

                try {
                    JavaFileObject file = processingEnv.getFiler().createSourceFile(packageName + "." + enumName + "Annotation");
                    file.openWriter().write(annotationCode);
                    file.openWriter().close();
                } catch (Exception e) {
                    e.printStackTrace();
                }
            }
        }
        return false;
    }
}
