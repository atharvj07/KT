import javax.swing.*;
import java.awt.*;
import java.awt.image.BufferedImage;
import java.util.ArrayList;
import java.util.List;

public class BulletHell extends JPanel {
    private final List<Bullet> bullets = new ArrayList<>();
    private final BufferedImage bulletSprite;

    public BulletHell() {
        bulletSprite = new BufferedImage(16, 16, BufferedImage.TYPE_INT_ARGB);
        Graphics g = bulletSprite.getGraphics();
        g.setColor(Color.RED);
        g.fillOval(0, 0, 16, 16);
        g.dispose();
        
        for (int i = 0; i < 10000; i++) {
            bullets.add(new Bullet(Math.random() * 800, Math.random() * 600, Math.random() * 360, bulletSprite));
        }

        Timer timer = new Timer(16, e -> {
            bullets.forEach(Bullet::update);
            repaint();
        });
        timer.start();
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        Graphics2D g2d = (Graphics2D) g;
        bullets.forEach(b -> b.draw(g2d));
    }

    public static void main(String[] args) {
        JFrame frame = new JFrame();
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setSize(800, 600);
        frame.add(new BulletHell());
        frame.setVisible(true);
    }
}

class Bullet {
    double x, y, angle;
    BufferedImage sprite;

    public Bullet(double x, double y, double angle, BufferedImage sprite) {
        this.x = x;
        this.y = y;
        this.angle = angle;
        this.sprite = sprite;
    }

    public void update() {
        x += Math.cos(Math.toRadians(angle)) * 2;
        y += Math.sin(Math.toRadians(angle)) * 2;
    }

    public void draw(Graphics2D g) {
        AffineTransform old = g.getTransform();
        g.translate(x, y);
        g.rotate(Math.toRadians(angle), sprite.getWidth() / 2.0, sprite.getHeight() / 2.0);
        g.drawImage(sprite, 0, 0, null);
        g.setTransform(old);
    }
}
