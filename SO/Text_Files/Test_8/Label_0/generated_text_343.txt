import numpy as np

x_size = 2000
y_size = 2500
z_size = 400
rng = np.random.default_rng(123)
z_coordinates = np.linspace(0, z_size, y_size) + rng.laplace(0, 1, (x_size, y_size))

y_coordinates = np.empty((x_size, z_size), dtype=np.uint16)

z_range = np.arange(z_size)
for i in range(x_size):
    y_coordinates[i] = np.argmin(np.abs(z_coordinates[i][:, None] - z_range), axis=0)
