import numpy as np

rng = lambda: np.random.randint(2, 20) // 2

class NRNG:
    def __init__(self):
        self.numbers = iter(np.arange(1, 10.5, 0.5))

    def __call__(self):
        try:
            return next(self.numbers)
        except StopIteration:
            self.numbers = iter(np.arange(1, 10.5, 0.5))
            return next(self.numbers)

nrng = NRNG()

for j in range(10):
    print('random number', rng())
    print('non-random number', nrng())
