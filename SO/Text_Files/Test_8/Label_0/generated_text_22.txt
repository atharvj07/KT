import java.util.*;
import java.util.concurrent.*;

public class FindMultiples {

    private static final Set<Integer> uniqueDividends = Collections.synchronizedSet(new HashSet<>());

    static class DivisorTask implements Runnable {
        private int divisor, n;

        public DivisorTask(int n, int divisor) {
            this.n = n;
            this.divisor = divisor;
        }

        @Override
        public void run() {
            for (int i = 1; i <= n; i++) {
                if (i % divisor == 0) {
                    uniqueDividends.add(i);
                }
            }
        }
    }

    public int getSum(int n) {
        int sum = 0;
        ExecutorService executor = Executors.newFixedThreadPool(3);
        
        executor.execute(new DivisorTask(n, 3));
        executor.execute(new DivisorTask(n, 5));
        executor.execute(new DivisorTask(n, 7));

        executor.shutdown();
        try {
            executor.awaitTermination(1, TimeUnit.SECONDS);
        } catch (InterruptedException e) {
        }

        for (int num : uniqueDividends) {
            sum += num;
        }

        return sum;
    }

    public static void main(String[] args) {
        FindMultiples finder = new FindMultiples();
        int n = 1000;
        System.out.println("Sum: " + finder.getSum(n));
    }
}
