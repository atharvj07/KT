import numpy as np
from numpy.typing import NDArray

def example1(arr: NDArray[np.float64]):
    return arr * 2

def example2(arr: np.ndarray[np.float32, np.dtype[np.int32]]): 
    return arr.astype(np.float64)

arr1 = np.array([1, 2, 3], dtype=np.float64)
arr2 = np.array([[1, 2], [3, 4]], dtype=np.float32)

print(example1(arr1))
print(example2(arr2))

T = np.ndarray[float]
print(T)
