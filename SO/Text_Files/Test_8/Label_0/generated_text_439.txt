import numpy.typing as npt
import numpy as np
from typing import Any, cast, Sequence

def f(a: npt.ArrayLike) -> int:
    arr = cast(Sequence, a)
    return len(arr)

def g(a: npt.ArrayLike) -> Any:
    arr = cast(Sequence, a)
    return arr[0]

print(f(np.array([0, 1])), g(np.array([0, 1])))
print(f([0, 1]), g([0, 1]))
