import org.apache.poi.ss.formula.FormulaParser;
import org.apache.poi.ss.formula.FormulaRenderingWorkbook;
import org.apache.poi.ss.usermodel.WorkbookFactory;
import org.apache.poi.ss.usermodel.Workbook;
import org.apache.poi.ss.formula.ptg.Ptg;
import java.io.File;
import java.io.FileInputStream;
import java.util.List;

public class ExcelFormulaTransformer {
    public static void main(String[] args) {
        try {
            FileInputStream fis = new FileInputStream(new File("example.xlsx"));
            Workbook workbook = WorkbookFactory.create(fis);
            String formula = "IF(A2=A3,A4,A5)";
            String transformedFormula = transformFormula(formula, "A6", "C6", workbook);
            System.out.println("Transformed Formula: " + transformedFormula);
        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
        }
    }

    public static String transformFormula(String formula, String originalAddress, String destinationAddress, Workbook workbook) {
        Ptg[] tokens = FormulaParser.parse(formula, (FormulaRenderingWorkbook) workbook, 0, 0);
        for (Ptg token : tokens) {
            if (token instanceof org.apache.poi.ss.formula.ptg.RefPtgBase) {
                org.apache.poi.ss.formula.ptg.RefPtgBase ref = (org.apache.poi.ss.formula.ptg.RefPtgBase) token;
                int rowOffset = parseRow(destinationAddress) - parseRow(originalAddress);
                int colOffset = parseColumn(destinationAddress) - parseColumn(originalAddress);
                ref.setRow(ref.getRow() + rowOffset);
                ref.setColumn(ref.getColumn() + colOffset);
            }
        }
        return FormulaParser.toFormulaString((FormulaRenderingWorkbook) workbook, tokens);
    }

    private static int parseColumn(String cell) {
        int col = 0;
        for (char ch : cell.toCharArray()) {
            if (Character.isLetter(ch)) {
                col = col * 26 + (ch - 'A' + 1);
            }
        }
        return col - 1;
    }

    private static int parseRow(String cell) {
        String num = cell.replaceAll("\\D+", "");
        return num.isEmpty() ? -1 : Integer.parseInt(num) - 1;
    }
}
