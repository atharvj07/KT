public class CircularSearch {
    public static boolean search(int[][] mat, int num) {
        int n = mat.length;
        int row = 0, col = n - 1;

        while (row < n && col >= 0) {
            if (mat[row][col] == num) {
                return true;
            } else if (mat[row][col] > num) {
                col--;
            } else {
                row++;
            }
        }
        return false;
    }

    public static void main(String[] args) {
        int[][] mat = {
            { 10, 12, 15, 20 },
            { 25, 30, 35, 40 },
            { 50, 55, 60, 65 },
            { 70, 75, 80, 85 }
        };

        System.out.println(search(mat, 22)); 
        System.out.println(search(mat, 15)); 
    }
}
