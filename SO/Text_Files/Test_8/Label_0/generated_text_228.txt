import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.TextArea;
import javafx.scene.text.Font;
import javafx.stage.Stage;

public class EmojiSupportApp extends Application {
    @Override
    public void start(Stage primaryStage) {
        String emojiText = "?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ??";
        TextArea textArea = new TextArea(emojiText);

        String fontUrl = getClass().getResource("NotoColorEmoji-Regular.ttf").toExternalForm();
        Font font = Font.loadFont(fontUrl, 14);
        if (font != null) {
            textArea.setFont(font);
        }

        Scene scene = new Scene(textArea, 400, 300);
        primaryStage.setScene(scene);
        primaryStage.setTitle("Emoji TextArea");
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
