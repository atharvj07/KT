class FooMeta(type):
    def __len__(cls):
        return 9000

class GoodBar(metaclass=FooMeta):
    def __len__(self):
        return 9001

class BadBar(metaclass=FooMeta):
    @classmethod
    def __len__(cls):
        return 9002

    def __class_getitem__(cls, item):
        return cls.__len__()

BadBar.__len__ = classmethod(lambda cls: 9002)

len(GoodBar), len(GoodBar()), len(BadBar), len(BadBar())
