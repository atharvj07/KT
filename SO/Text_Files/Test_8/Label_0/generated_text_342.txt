class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __iter__(self):
        return iter((self.x, self.y))
    
    def __getitem__(self, index):
        return (self.x, self.y)[index]

    def __repr__(self):
        return f"Point({self.x}, {self.y})"

    def __eq__(self, other):
        if isinstance(other, tuple):
            return (self.x, self.y) == other
        return False
