from typing import Generic, TypeVar

T = TypeVar("T")

class MyType(Generic[T]):
    def __init__(self, value: T):
        self.value = value

    def __repr__(self):
        return f"MyType({repr(self.value)})"
    
    def __eq__(self, other):
        if isinstance(other, MyType):
            return self.value == other.value
        return False

# Now when you use MyType[int], it will act like int
a = MyType(10)
b = MyType(10)

print(a == b)  # Should be True
