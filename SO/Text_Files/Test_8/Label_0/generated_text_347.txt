import polars as pl

COL_VALUE = "value"

df = pl.DataFrame({COL_VALUE: [3, 7, 1, 9, 4]})

sum_lower = (
    df.with_columns(pl.col(COL_VALUE).rank("ordinal").alias("rank"))
    .sort("rank")
    .with_columns(pl.col(COL_VALUE).cumsum().shift(1).fill_null(0).alias("sum_lower"))
    .sort(COL_VALUE)
    .select("sum_lower")
)

max_other = (
    df.select(pl.max(COL_VALUE).alias("max_val"))
    .join(df, how="cross")
    .with_columns((pl.col("max_val") - (pl.col("max_val") == pl.col(COL_VALUE)) * pl.col(COL_VALUE)).alias("max_other"))
    .select("max_other")
)

df = df.with_columns(sum_lower, max_other)
print(df)
