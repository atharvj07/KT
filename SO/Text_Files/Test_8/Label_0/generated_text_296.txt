import pandas as pd
import numpy as np

id = ['a','b','c','a','b','e','f','a','b','k']
value = [12,14,3,13,16,7,4,6,10,18]

df = pd.DataFrame({'id':id, 'value':value})

df['duped'] = np.where(df['id'].duplicated(keep='last'), 0, 1)
df['value_duped'] = df['value'] * df['duped']

df['cumsum_of_value'] = df['value_duped'].cumsum()
df['desired_output'] = df.groupby('id')['value'].transform(lambda x: x.cumsum().where(x.notna(), 0))

print(df)
