import com.opencsv.CSVWriter;
import lombok.AllArgsConstructor;
import lombok.Data;
import lombok.NoArgsConstructor;

import java.io.StringWriter;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;

@Data
@AllArgsConstructor
@NoArgsConstructor
public class DmnTestCaseResult {
    private Map<String, Object> testInput;
    private Map<String, Object> expectedOutput;
    private List<Map<String, Object>> testOutput;
    private String errorMessage;
}

public class CsvConverter {
    public String convertDmnRuleTestResultToCsv(List<DmnTestCaseResult> results) {
        try (StringWriter sw = new StringWriter();
             CSVWriter writer = new CSVWriter(sw, ';', CSVWriter.NO_QUOTE_CHARACTER, CSVWriter.NO_ESCAPE_CHARACTER, "\r\n")) {

            writer.writeNext(new String[]{"Test Input", "Expected Output", "Test Output", "Error Message"});

            for (DmnTestCaseResult result : results) {
                List<String[]> rows = new ArrayList<>();
                for (Map<String, Object> output : result.getTestOutput()) {
                    rows.add(new String[]{
                            result.getTestInput().toString(),
                            result.getExpectedOutput().toString(),
                            output.toString(),
                            result.getErrorMessage()
                    });
                }
                if (rows.isEmpty()) {
                    rows.add(new String[]{
                            result.getTestInput().toString(),
                            result.getExpectedOutput().toString(),
                            "No output",
                            result.getErrorMessage()
                    });
                }
                writer.writeAll(rows);
            }
            return sw.toString();
        } catch (Exception ex) {
            throw new RuntimeException("Error while writing CSV: " + ex.getMessage());
        }
    }
}
