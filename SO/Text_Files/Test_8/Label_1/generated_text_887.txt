from pydantic import BaseModel, validator


class Foo(BaseModel):
    x: int

    @validator("x", pre=True)
    def do_stuff(cls, v: object) -> object:
        if v is None:
            return 0
        return v


print(Foo(x=None))  # x=0

from pydantic import BaseModel, field_validator


class Foo(BaseModel):
    x: int

    @field_validator("x", mode="before")
    def do_stuff(cls, v: object) -> object:
        if v is None:
            return 0
        return v


print(Foo(x=None))  # x=0
