import java.io.RandomAccessFile;

public class OverwriteFile {
public static void main(String[] args) {
String filePath = "test.txt";
try (RandomAccessFile raf = new RandomAccessFile(filePath, "rw")) {
raf.seek(0);
raf.write("hello".getBytes());
} catch (Exception e) {
e.printStackTrace();
}
}
}
