public class CircularMatrixSearch {
public static boolean search(int[][] mat, int num) {
int n = mat.length;
int row = 0, col = n - 1;

while (row < n && col >= 0) {
if (mat[row][col] == num) {
return true;
} else if (mat[row][col] > num) {
col--;
} else {
row++;
}
}
return false;
}

public static void main(String[] args) {
int[][] mat = {
{10, 12, 14, 16},
{22, 24, 26, 28},
{34, 36, 38, 40},
{50, 52, 54, 56}
};

System.out.println(search(mat, 22)); // true
System.out.println(search(mat, 23)); // false
}
}
