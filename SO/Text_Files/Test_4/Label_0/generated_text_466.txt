class BST:
    def __init__(self, val):
        self.value = val
        self.left = None
        self.right = None

    def add_child(self, val):
        if self.value:
            if val < self.value:
                if self.left == None:
                    self.left = BST(val)
                else:
                    self.left.add_child(val)
            else:
                if val > self.value:
                    if self.right == None:
                        self.right = BST(val)
                    else:
                        self.right.add_child(val)
        else:
            self.value = val

    def traverse_pre(self, result=None):
        if result is None:
            result = []

        result.append(self.value)  # Add the current node value to the result list

        if self.left:
            self.left.traverse_pre(result)  # Recursively traverse the left subtree

        if self.right:
            self.right.traverse_pre(result)  # Recursively traverse the right subtree

        return result


Tree = BST(5)
Tree.add_child(10)
Tree.add_child(8)
Tree.add_child(2)
Tree.add_child(4)
Tree.add_child(7)

result = Tree.traverse_pre()  # This will return the pre-order traversal as a list
print(result)
