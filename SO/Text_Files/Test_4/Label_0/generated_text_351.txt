from fastapi import FastAPI

# --- main app ---
def lifespan(_):
 print("startup")
 yield
 print("shutdown")

app=FastAPI(lifespan=lifespan)

@app.get("/")
async def root():
 return {"message":"Hello World"}

# --- sub app ---
def sub_lifespan(_):
 print("sub startup")
 yield
 print("sub shutdown")

sub_app=FastAPI(lifespan=sub_lifespan)

@sub_app.get("/")
async def sub_root():
 return {"message":"Hello Sub World"}

app.mount("/sub",sub_app)
