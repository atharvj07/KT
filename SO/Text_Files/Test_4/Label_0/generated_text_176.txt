interface Show1 {
default void display() {
System.out.println("hello");
}
}

interface Show2 {
void display();
}

public class Person implements Show1, Show2 {
@Override
public void display() {
Show1.super.display(); // Explicitly calling the default method
}

public static void main(String args[]) {
Person p = new Person();
p.display();
}
}
