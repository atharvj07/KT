import numpy as np
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense

x_data = [np.random.rand(np.random.randint(100, 200)) for _ in range(50)]
y_data = np.random.rand(50).astype(np.float32)

max_len = max(len(seq) for seq in x_data)
x_padded = np.array([np.pad(seq, (0, max_len - len(seq))) for seq in x_data])

x_padded = np.expand_dims(x_padded, axis=-1)

model = Sequential([
    LSTM(64, return_sequences=False, input_shape=(max_len, 1)),
    Dense(1, activation="linear")
])

model.compile(optimizer="adam", loss="mse")
model.fit(x_padded, y_data, epochs=10, batch_size=8)
