import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Paths;
import com.google.protobuf.DescriptorProtos.FileDescriptorSet;
import com.google.protobuf.DescriptorProtos.FileDescriptorProto;

public class ProtoParser {
    public static void main(String[] args) throws IOException {
        byte[] protoBytes = Files.readAllBytes(Paths.get("example.proto"));

        FileDescriptorSet descriptorSet = FileDescriptorSet.parseFrom(protoBytes);

        for (FileDescriptorProto descriptorProto : descriptorSet.getFileList()) {
            System.out.println("Proto Name: " + descriptorProto.getName());
            System.out.println("Package: " + descriptorProto.getPackage());
        }
    }
}
