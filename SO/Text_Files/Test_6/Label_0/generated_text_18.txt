import org.yaml.snakeyaml.Yaml;
import java.io.FileInputStream;
import java.io.InputStream;
import java.util.List;
import java.util.Map;

public class YamlLoader {
    public static void main(String[] args) {
        try (InputStream input = new FileInputStream("config.yaml")) {
            Yaml yaml = new Yaml();
            Map<String, Object> yamlConfig = yaml.load(input);

            Object interestingObject = yamlConfig.get("interesting-setup");
            Map<String, Object> interestingMap = (Map<String, Object>) interestingObject;

            InterestingSetup setup = new InterestingSetup();
            setup.port = (int) interestingMap.get("port");
            setup.validation = (boolean) interestingMap.get("validation");
            setup.parts = (List<Map<String, String>>) interestingMap.get("parts");

            System.out.println(setup);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

class InterestingSetup {
    int port;
    boolean validation;
    List<Map<String, String>> parts;

    @Override
    public String toString() {
        return "InterestingSetup{" +
                "port=" + port +
                ", validation=" + validation +
                ", parts=" + parts +
                '}';
    }
}
