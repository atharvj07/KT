import polars as pl

df = pl.DataFrame(
    pl.Series("sector_exposure", [
        "Technology=0.207;Financials=0.090;Health Care=0.084;Consumer Discretionary=0.069", 
        "Financials=0.250;Health Care=0.200;Consumer Staples=0.150;Industrials=0.400"
    ])
)

sector_names = ["Technology", "Financials", "Health Care", "Consumer Discretionary", "Consumer Staples", "Industrials"]

df = df.with_columns(
    pl.col("sector_exposure")
    .str.split(";", inclusive=True)
    .arr.eval(pl.col().str.split("="), parallel=True)
    .arr.eval(pl.struct([pl.element().arr.get(1).cast(pl.Float64).alias(name) for name in sector_names]), parallel=True)
)

print(df)
