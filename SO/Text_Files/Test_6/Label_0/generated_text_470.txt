import base64
import requests
import numpy as np
import json
from matplotlib.pyplot import imread
from skimage.transform import resize

if __name__ == '__main__':
    path_to_img = "my image path"

    image = imread(path_to_img)
    image = resize(image, (512, 512), mode='reflect')

    image_byte = base64.b64encode(image.tobytes())
    data = {"shape": image.shape, "image": image_byte.decode()}

    response = requests.post('http://127.0.0.1:8000/myapp/v1/filter/a', data=json.dumps(data))
