import pandas as pd

data = {
    "TradeID": ["aaa", "bbb", "ccc"],
    "TradeDate": ["01/01/2024"] * 3,
    "Commodity": ["(com1,com2)"] * 3,
    "StartDate": ["01/01/2024"] * 3,
    "ExpiryDate": ["01/01/2024"] * 3,
    "FixedPrice": [10] * 3,
    "Quantity": [10] * 3,
    "MTMValue": [100.00] * 3
}

df = pd.DataFrame(data)

fprice = df.groupby(['StartDate', 'Commodity', 'DealType'], group_keys=False) \
    .apply(lambda group: -(group['MTMValue'].sum() - (group['FixedPrice'] * group['Quantity']).sum()) / 
           (group['Quantity'].sum() if group['Quantity'].sum() != 0 else 1)) \
    .reset_index(name='FloatPrice')

df = df.merge(fprice, on=['StartDate', 'Commodity'], how='left')

print(df)
