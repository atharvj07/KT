import pandas as pd

data = {
    "FIELD_A": [123123, 123124, 123144],
    "FIELD_B": [8, 7, 99],
    "FIELD_C": ["a", "c", "x"],
    "FIELD_D": [23423, 6464, 234]
}

df = pd.DataFrame(data)

df_melted = df.melt(id_vars=["FIELD_A"], var_name="FIELD_NAME", value_name="FIELD_VALUE")

df_melted["FIELD_NAME"] = df_melted["FIELD_NAME"].str[-1]

df_melted["FIELD_STR_VALUE"] = df_melted["FIELD_VALUE"].apply(lambda x: x if isinstance(x, str) else "")
df_melted["FIELD_NUM_VALUE"] = df_melted["FIELD_VALUE"].apply(lambda x: x if isinstance(x, int) else None)

df_melted = df_melted.drop(columns=["FIELD_VALUE"])

df_melted.insert(0, "ID", range(1, len(df_melted) + 2))  

print(df_melted)
