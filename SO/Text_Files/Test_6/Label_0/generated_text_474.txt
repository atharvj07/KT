import polars as pl

df = pl.DataFrame(
    pl.Series("sector_exposure", [
        "Technology=0.207;Financials=0.090;Health Care=0.084;Consumer Discretionary=0.069", 
        "Financials=0.250;Health Care=0.200;Consumer Staples=0.150;Industrials=0.400"
    ])
)

sector_columns = [
    "Technology", "Financials", "Health Care", 
    "Consumer Discretionary", "Consumer Staples", "Industrials"
]

df = df.with_columns(
    pl.col("sector_exposure")
    .str.split(";")
    .apply(lambda x: {entry.split('=')[0]: float(entry.split('=')[1]) for entry in x}, return_dtype=pl.Object)
    .alias("sector_struct")
)

df = df.with_columns([
    pl.col("sector_struct").arr.get(i).alias(sector_columns[i]) for i in range(len(sector_columns))
])

df = df.select(["sector_exposure"] + sector_columns)

df
