import polars as pl
import numpy as np

data = {
    "col1": ["a", "b", "c", "d"],
    "col2": [[-0.06066, 0.072485, 0.548874, 0.158507],
             [-0.536674, 0.10478, 0.926022, -0.083722],
             [-0.21311, -0.030623, 0.300583, 0.261814],
             [-0.308025, 0.006694, 0.176335, 0.533835]],
}

df = pl.DataFrame(data)

def cosine_sim(a, b):
    return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))

labels = df["col1"].to_list()
vectors = np.array(df["col2"].to_list())

result = []

for i in range(len(vectors)):
    row = []
    for j in range(len(vectors)):
        row.append(cosine_sim(vectors[i], vectors[j]))
    result.append(row)

df_sim = pl.DataFrame({labels[i]: result[i] for i in range(len(labels))})
df_sim = df_sim.with_columns(pl.Series("col1_col2", labels))
df_sim = df_sim.select(["col1_col2"] + labels)
