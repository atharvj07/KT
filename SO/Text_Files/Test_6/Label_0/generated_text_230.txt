import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

import com.inventory.server.domain.Item;
import com.inventory.server.domain.Categorie;
import com.inventory.server.dto.CreateItemData;
import com.inventory.server.exception.ItemAlreadyCreatedException;
import com.inventory.server.repository.ItemRepository;
import com.inventory.server.service.ItemService;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.*;
import org.mockito.junit.jupiter.MockitoExtension;
import org.springframework.web.util.UriComponentsBuilder;

import java.math.BigDecimal;
import java.util.Optional;

@ExtendWith(MockitoExtension.class)
class ItemServiceTest {

    @InjectMocks
    ItemService itemService;

    @Mock
    ItemRepository itemRepository;

    @Mock
    CategorieRepository categorieRepository;

    @Mock
    UriComponentsBuilder uriBuilder;

    MockItem input;

    @BeforeEach
    void setUp() {
        input = new MockItem();
        MockitoAnnotations.openMocks(this);
    }

    @Test
    void testItemAlreadyExists() {
        Item item = input.mockEntity();
        CreateItemData data = input.mockDTO();

        when(itemRepository.findByItemNameIgnoreCase(anyString())).thenReturn(Optional.ofNullable(item));

        Exception exception = assertThrows(ItemAlreadyCreatedException.class, () -> {
            itemService.createItem(data, uriBuilder);
        });

        assertEquals("There is an item created with this name", exception.getMessage());

        verify(itemRepository, times(1)).findByItemNameIgnoreCase(any());
    }
}
