import io.micronaut.context.annotation.Prototype;
import io.micronaut.core.type.Argument;
import io.micronaut.serde.ObjectMapper;
import jakarta.inject.Inject;
import java.io.IOException;
import java.lang.invoke.MethodHandles;
import java.util.Optional;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

@Prototype
public class Scratch {
    private static final Logger LOG = LoggerFactory.getLogger(MethodHandles.lookup().lookupClass());
    private final ObjectMapper objectMapper;

    @Inject
    public Scratch(ObjectMapper objectMapper) {
        this.objectMapper = objectMapper;
    }

    private <T> Optional<CommonResponse<T>> getCommonResponse(final String entry, Class<T> type) {
        try {
            Argument<CommonResponse<T>> argument = Argument.of(CommonResponse.class, type);
            return Optional.of(objectMapper.readValue(entry, argument));
        } catch (IOException e) {
            LOG.warn("Could not deserialize, skipping entry: '{}'", entry, e);
        }
        return Optional.empty();
    }
}
