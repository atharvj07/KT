import javafx.application.Application;
import javafx.beans.property.ObjectProperty;
import javafx.beans.property.SimpleObjectProperty;
import javafx.stage.Stage;

public class UserDataListenerExample extends Application {

    @Override
    public void start(Stage stage) {
        ObjectProperty<Object> userDataProperty = new SimpleObjectProperty<>(stage.getUserData());
        
        stage.setUserData("InitialData");

        userDataProperty.addListener((observable, oldValue, newValue) -> {
            System.out.println("New userData: " + newValue);
        });

        stage.setUserData("Test1");
        userDataProperty.set(stage.getUserData());

        stage.setUserData("Test2");
        userDataProperty.set(stage.getUserData());
    }

    public static void main(String[] args) {
        launch(args);
    }
}
