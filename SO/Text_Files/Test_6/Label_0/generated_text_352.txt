import json
from typing import List, Optional
from pydantic import BaseModel

response = json.loads('{"a": 1, "b": "abc", "c": ["a", "b", "c"]}')

class Model(BaseModel):
    a: int
    b: str
    c: List[str]
    d: Optional[float] = None

validated_data = Model(**response)

print(validated_data)
