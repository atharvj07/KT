import org.junit.jupiter.api.extension.*;

@Suite
@SelectClasses({FirstTest.class, SecondTest.class})
public class JUnit5Suite {
    public static final JUnit5Server MY_SERVER = new JUnit5Server();
    
    @BeforeAll
    public static void setup() {
        MY_SERVER.beforeAll(null);
    }

    @AfterAll
    public static void teardown() {
        MY_SERVER.afterAll(null);
    }
}

public class JUnit5Server implements BeforeAllCallback, AfterAllCallback {
    @Override
    public void beforeAll(ExtensionContext context) throws Exception {
        System.out.println("Starting server...");
    }

    @Override
    public void afterAll(ExtensionContext context) throws Exception {
        System.out.println("Stopping server...");
    }
}
