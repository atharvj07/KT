from fastapi import FastAPI, Depends
from pymongo import MongoClient
from motor.motor_asyncio import AsyncIOMotorClient
from typing import Callable

app = FastAPI()

def build_db(name: str) -> Callable[[], AsyncIOMotorClient]:
    def get_db():
        return AsyncIOMotorClient("mongodb://localhost:27017")[name]
    return get_db

@app.post("/notification/feed")
async def route_receive_notifications(db=Depends(build_db("someDB"))):
    return {"db_name": db.name}

def mock_db():
    return AsyncIOMotorClient("mongodb://localhost:27017")["testDB"]

app.dependency_overrides[build_db("someDB")] = lambda: mock_db()
