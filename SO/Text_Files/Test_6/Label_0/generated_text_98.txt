import javafx.fxml.FXMLLoader;
import javafx.scene.layout.VBox;
import org.springframework.context.ApplicationContext;
import org.springframework.stereotype.Component;
import java.io.IOException;

@Component
public class ListItemFactory {

    private final ApplicationContext applicationContext;

    public ListItemFactory(ApplicationContext applicationContext) {
        this.applicationContext = applicationContext;
    }

    public VBox createListItem() {
        FXMLLoader fxmlLoader = new FXMLLoader(getClass().getResource("/path/to/fxml"));
        fxmlLoader.setControllerFactory(clazz -> {
            if (clazz.isAnnotationPresent(Component.class)) {
                return applicationContext.getAutowireCapableBeanFactory().createBean(clazz);
            }
            try {
                return clazz.getDeclaredConstructor().newInstance();
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        });

        try {
            return fxmlLoader.load();
        } catch (IOException e) {
            e.printStackTrace();
            return null;
        }
    }
}
import javafx.fxml.FXML;
import javafx.scene.layout.VBox;
import org.springframework.stereotype.Component;

@Component
public class MainController {

    @FXML
    private VBox contentArea;

    private final ListItemFactory listItemFactory;

    public MainController(ListItemFactory listItemFactory) {
        this.listItemFactory = listItemFactory;
    }

    public void addItem() {
        VBox listItem = listItemFactory.createListItem();
        if (listItem != null) {
            contentArea.getChildren().add(listItem);
        }
    }
