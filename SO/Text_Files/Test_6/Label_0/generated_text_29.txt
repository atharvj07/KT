public class EqualSplit {
    public static boolean equalSplit(int[] arr) {
        if (arr.length % 2 != 0) 
            return false;
        return equalSplit(arr, 0, 0, 0, arr.length / 2);
    }

    private static boolean equalSplit(int[] arr, int index, int sum1, int sum2, int count1) {
        if (index == arr.length) 
            return sum1 == sum2 && count1 == 0;
        
        if (count1 > 0 && equalSplit(arr, index + 1, sum1 + arr[index], sum2, count1 - 1)) 
            return true;
        
        return equalSplit(arr, index + 1, sum1, sum2 + arr[index], count1);
    }

    public static void main(String[] args) {
        int[] arr = {1, 2, 3, 4, 6, 8};
        System.out.println(equalSplit(arr)); 
    }
}
