import java.util.concurrent.*;

public class ComputeIfAbsentExample {
    private final ConcurrentHashMap<String, StringBuilder> map = new ConcurrentHashMap<>();
    private volatile int counter = 0;  

    public StringBuilder getValue(String key) {
        return map.computeIfAbsent(key, k -> {
            counter++;  
            StringBuilder sb = new StringBuilder();
            sb.append("Value: ").append(counter);
            return sb;
        });
    }

    public static void main(String[] args) {
        ComputeIfAbsentExample example = new ComputeIfAbsentExample();
        ExecutorService executor = Executors.newFixedThreadPool(3);

        Runnable task1 = () -> System.out.println(example.getValue("key1"));
        Runnable task2 = () -> System.out.println(example.getValue("key2"));
        Runnable task3 = () -> System.out.println(example.getValue("key1"));

        executor.execute(task1);
        executor.execute(task2);
        executor.execute(task3);
        
        executor.shutdown();
    }
}
