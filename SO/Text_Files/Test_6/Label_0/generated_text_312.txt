import polars as pl

df = pl.DataFrame(
    {
        "A": ["foo", "ham", "spam", "egg"],
        "L": ["A54", "A12", "B84", "C12"],
        "G": ["X34", "C84", "G96", "L6"],
    }
)

df = df.with_columns(
    pl.when((pl.col("A") == "foo") | (pl.col("A") == "spam"))
    .then(pl.lit("XX"))
    .otherwise(pl.col("L"))
    .alias("L"),
    pl.when((pl.col("A") == "foo") | (pl.col("A") == "spam"))
    .then(pl.lit("XX"))
    .otherwise(pl.col("G"))
    .alias("G")
)

print(df)
