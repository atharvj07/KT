import numpy as np

def ijIndicesToSourceArray(i, j, rows, cols):
    return (rows - j - 1, cols - i - 1)

arr = np.array([[ 1,  2,  4],
                [ 3,  5,  7],
                [ 6,  8, 10],
                [ 9, 11, 13],
                [12, 14, 15]])

rows, cols = arr.shape
new_arr = np.zeros((rows, cols), dtype=arr.dtype)

for i in range(rows):
    for j in range(cols):
        ni, nj = ijIndicesToSourceArray(i, j, rows, cols)
        new_arr[i, j] = arr[ni, nj]

print(new_arr)
