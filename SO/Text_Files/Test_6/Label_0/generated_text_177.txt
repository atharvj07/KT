abstract class MySuperClass {
    private final Object mySuperField;

    MySuperClass(Object myField) {
        if (myField == null) {
            throw new IllegalArgumentException("mySuperField cannot be null");
        }
        this.mySuperField = myField;
    }

    public Object getMySuperField() {
        return mySuperField;
    }
}

public class MySubClass extends MySuperClass {
    private final Object mySubField;

    public MySubClass(MySubClass toCopy) {
        super(toCopy != null ? toCopy.getMySuperField() : new Object());
        if (toCopy == null) {
            throw new IllegalArgumentException("Cannot copy from a null instance");
        }
        this.mySubField = toCopy.mySubField;
    }
}
