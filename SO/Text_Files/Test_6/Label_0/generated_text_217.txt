import javafx.geometry.Rectangle2D;
import javafx.scene.Scene;
import javafx.scene.image.ImageView;
import javafx.stage.Screen;
import javafx.stage.Stage;
import javafx.stage.StageStyle;

public class CrosshairScene {

    private Stage stage;
    private ImageView crosshairImage;
    private Stage primary;

    private void setStageProperties() {
        Screen screen = Screen.getPrimary();
        Rectangle2D bounds = screen.getBounds();

        stage.setWidth(bounds.getWidth());
        stage.setHeight(bounds.getHeight());

        Scene scene = new Scene(this);
        scene.setFill(null);
        scene.setOnMouseClicked(e -> e.consume());
        stage.setScene(scene);
        stage.setAlwaysOnTop(true);

        primary = new Stage();
        primary.initStyle(StageStyle.UTILITY);
        primary.setOpacity(0);
        primary.setHeight(1);
        primary.setWidth(1);
        primary.show();

        stage.initOwner(primary);
        stage.initStyle(StageStyle.TRANSPARENT);

        double centerX = bounds.getMinX() + bounds.getWidth() / 2;
        double centerY = bounds.getMinY() + bounds.getHeight() / 2;

        stage.setX(centerX - 10);
        stage.setY(centerY - 10);
    }

    public CrosshairScene() {
        this.stage = new Stage();
        this.crosshairImage = new ImageView("crosshair.png");
        this.crosshairImage.setPickOnBounds(false);
        this.crosshairImage.setMouseTransparent(true);
        this.setMouseTransparent(true);
        this.setCenter(crosshairImage);
        this.setStageProperties();
        this.setStyle("-fx-background-color: transparent;");
    }
}
