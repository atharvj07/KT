from flask import Flask, render_template, jsonify
import map_plotting_test as mpt
import threading

app = Flask(__name__)

def generate_map():
    mpt.create_map()

@app.route('/')
def render_the_map():
    # Show loading screen while the map is being generated
    threading.Thread(target=generate_map).start()
    return render_template("loading.html")

@app.route('/map')
def show_map():
    return render_template("map.html")

if __name__ == '__main__':
    app.run(debug=True)
