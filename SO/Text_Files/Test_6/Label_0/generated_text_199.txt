import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

import java.util.Optional;
import java.util.UUID;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;

public class PriceServiceTest {

    @Mock
    private PriceRepository priceRepository;

    @InjectMocks
    private PriceService priceService;

    private UUID priceUuid;
    private Price existingPrice;
    private PriceRequest priceRequest;

    @BeforeEach
    void setUp() {
        MockitoAnnotations.initMocks(this);

        priceUuid = UUID.randomUUID();
        existingPrice = new Price();
        existingPrice.setUuid(priceUuid);
        existingPrice.setPriceAmount(100.0);

        priceRequest = new PriceRequest();
        priceRequest.setPriceAmount(150.0);
    }

    @Test
    void testUpdate() {
        when(priceRepository.findByUuid(priceUuid)).thenReturn(Optional.of(existingPrice));
        when(priceRepository.saveAndFlush(any())).thenAnswer(invocation -> invocation.getArgument(0));

        PriceDTO result = priceService.update(priceUuid, priceRequest);

        assertNotNull(result);
        assertEquals(150.0, result.getPriceAmount());
        verify(priceRepository, times(1)).saveAndFlush(existingPrice);
    }
}
