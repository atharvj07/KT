from fastapi import FastAPI, Request, Response

app = FastAPI()

@app.get("/")
def read_root():
    return {"Hello": "World"}

@app.post("/input")
def input_request(request: Request):
    body = request.body  # This is actually a coroutine, need to call it properly
    raw_body = body()  # Incorrect way, this will cause an error

    return Response(content=raw_body, media_type="application/xml")
