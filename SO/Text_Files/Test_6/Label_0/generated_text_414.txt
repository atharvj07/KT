import polars as pl

df = pl.DataFrame(
    {
        "Group": ["Group1", "Group2", "Group3"],
        "Weight": [100.0, 200.0, 300.0],
        "Vals": [[0.5, 0.5, 0.8], [0.5, 0.5, 0.8], [0.7, 0.9]]
    }
)

df = df.with_columns(
    pl.col("Vals").arr.eval(pl.element() * pl.first().over("Group", "Weight")).alias("Weighted")
)

print(df)
