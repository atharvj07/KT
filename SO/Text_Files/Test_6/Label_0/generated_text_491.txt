import polars as pl
from datetime import datetime
from polars import col as c

df = pl.DataFrame({
    "time": [datetime(2023, 4, 3, 2), datetime(2023, 4, 4, 3), datetime(2023, 4, 5, 4)],
    "tzone": ["Asia/Tokyo", "America/Chicago", "Europe/Paris"]
})

# Dynamically applying timezone
df = df.with_columns(
    c.time.dt.replace_time_zone("UTC"),
    c.tzone.apply(lambda tz: c.time.dt.convert_time_zone(tz).dt.hour(), return_dtype=pl.Int64).alias("time_in_tzone")
)

print(df)
