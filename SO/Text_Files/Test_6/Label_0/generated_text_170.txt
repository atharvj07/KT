# First, find the PID of the Java process running Tomcat
jps | grep "Bootstrap" | awk '{print $1}'

# Assuming the PID is 1234, list open file descriptors for the process
# (You may need to adjust the command depending on your OS)
handle -p 1234 | findstr "C:\\path\\to\\your\\file"

# If using Linux or WSL, an alternative way is:
ls -l /proc/1234/fd | grep "your-file-name"

# Another approach: Use jcmd to dump all open file descriptors
jcmd 1234 VM.native_memory summary | findstr "C:\\path\\to\\your\\file"

# If needed, check the references in the JVM
jcmd 1234 GC.class_histogram | findstr "FileInputStream"
