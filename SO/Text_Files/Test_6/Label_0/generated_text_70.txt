import com.mongodb.client.model.InsertOneModel;

public class ParallelStreamTest {
    public static void main(String[] args) {
        List<String> puzzles = List.of("puzzle1", "puzzle2", "puzzle3");
        Gson gson = new Gson();

        List<InsertOneModel<Document>> bulkWrites1 = puzzles.parallelStream()
                .map(gson::toJson)
                .map(Document::parse)
                .map(InsertOneModel::new)
                .collect(Collectors.toList());

        List<InsertOneModel<Document>> bulkWrites2 = puzzles.parallelStream()
                .map(puzzle -> new InsertOneModel<>(Document.parse(gson.toJson(puzzle))))
                .collect(Collectors.toList());
    }
}
