public class Test {
    public static void go(Short n) {
        System.out.println("Short");
    }

    public static void go(int n) {
        System.out.println("int");
    }

    public static void main(String[] args) {
        short x = 11;
        go(Short.valueOf(x));  // Explicit boxing to call go(Short)
    }
}
