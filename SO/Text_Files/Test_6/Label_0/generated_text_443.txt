import polars as pl

df = pl.DataFrame(
    {
        "a": [1, 8, 3],
        "b": [4, 5, None],
    }
)

df = df.with_columns(
    pl.concat_list(["a", "b"]).alias("values"),
    pl.lit(["a", "b"]).alias("cols")
)

df = df.with_columns(
    pl.struct(["values", "cols"]).map_elements(
        lambda x: x["cols"][x["values"].arg_max()]
    ).alias("max")
)

df.select("a", "b", "max")
