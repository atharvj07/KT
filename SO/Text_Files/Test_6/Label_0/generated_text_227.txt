public class Solution {
    static int i = 0;

    public static void f() {
        System.out.println(i++);
        try {
            f();
        } catch (StackOverflowError e) {
            System.out.println("Stack Overflow occurred: " + e);
            f();
        }
    }

    public static void main(String[] args) {
        f();
    }
}
