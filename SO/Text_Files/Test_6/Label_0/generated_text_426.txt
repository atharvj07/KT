import polars as pl 

df = pl.DataFrame({
    "a_0": [1, 4], "a_1": [2, 5], "a_2": [3, 6], 
    "b_0": [7, 10], "b_1": [8, 11], "b_2": [9, 12]
})

df_melted = df.melt(id_vars=[], value_vars=df.columns)

df_melted = df_melted.with_columns([
    pl.col("variable").str.replace(r"_[0-9]+", "").alias("category"),
    pl.col("value").alias("values")
])

df_final = df_melted.pivot(values="values", index=None, columns="category")
