import org.springframework.context.ApplicationContext;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.stereotype.Component;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;

interface CustomClass {
    void execute();
}

@Component("A")
class ClassA implements CustomClass {
    private final int param1;
    private final int param2;
    public ClassA(int param1, int param2) {
        this.param1 = param1;
        this.param2 = param2;
    }
    public void execute() {
        System.out.println("Executing ClassA with params: " + param1 + ", " + param2);
    }
}

@Component("B")
class ClassB implements CustomClass {
    private final int param1;
    public ClassB(int param1) {
        this.param1 = param1;
    }
    public void execute() {
        System.out.println("Executing ClassB with param: " + param1);
    }
}

public class DependencyInjectionExample {
    private static final ApplicationContext context = new AnnotationConfigApplicationContext("your.package.name");

    public static void main(String[] args) {
        String input = "A 1 2";
        String[] parts = input.split(" ");
        String className = parts[0];
        int[] params = Arrays.stream(Arrays.copyOfRange(parts, 1, parts.length))
                             .mapToInt(Integer::parseInt)
                             .toArray();
        
        CustomClass instance = createInstance(className, params);
        if (instance != null) {
            instance.execute();
        }
    }

    private static CustomClass createInstance(String className, int[] params) {
        try {
            return (CustomClass) context.getBean(className, params);
        } catch (Exception e) {
            e.printStackTrace();
            return null;
        }
    }
}
