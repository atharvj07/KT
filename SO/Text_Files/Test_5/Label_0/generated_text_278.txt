from dataclasses import dataclass, fields

@dataclass
class MyDataClass:
    a: float
    b: float
    c: float

    def __iter__(self):
        return (getattr(self, field.name) for field in fields(self))

# Example usage
thing = MyDataClass(1, 2, 3)

for i in thing:
    print(i)

from dataclasses import dataclass, asdict

@dataclass
class MyDataClass:
    a: float
    b: float
    c: float

    def __iter__(self):
        yield from asdict(self).values()

# Example usage
thing = MyDataClass(1, 2, 3)

for i in thing:
    print(i)
