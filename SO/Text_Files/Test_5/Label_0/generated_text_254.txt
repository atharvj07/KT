def safe_dict(**kwargs):
    seen = set()
    for key in kwargs:
        if key in seen:
            raise TypeError(f"Duplicate key: {key}")
        seen.add(key)
    return dict(**kwargs)

# Example usage
try:
    raise ValueError("foo")
except:
    safe_dict(id=1, **{'id': 2})  # Raises TypeError as expected
