import polars as pl
import datetime as dt

# Create a DataFrame with a datetime column
df = pl.DataFrame({
    "time": [dt.datetime.now()]
})

# Extract the date directly using `.dt.date()` (simpler and cleaner)
df = df.with_columns(
    pl.col("time").dt.date().alias("date")
)

print(df)
