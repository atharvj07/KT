import pytest
import time

class Test:
    a = None  # Class attribute to store the value

    @pytest.fixture(scope="class", autouse=True)
    def set_a(cls, request):
        print("Setting a...")
        time.sleep(5)
        cls.a = 1
        request.cls.a = cls.a  # Assign to class-level attribute

    def test_1(self):
        print("TEST 1")
        assert self.a == 1

    def test_2(self):
        print("TEST 2")
        assert self.a == 1
