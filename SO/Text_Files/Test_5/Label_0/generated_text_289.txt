def get_spouse(name):
    match name:
        case "John":
            return "Jane"
        case "David":
            return "Alice"
        case _:
            raise ValueError("Unknown name")

spouse = get_spouse("John")
print(spouse)

python
spouse = {
    "John": "Jane",
    "David": "Alice"
}.get(name, "Unknown")
print(spouse)

python
spouse = (
    "Jane" if name == "John" else
    "Alice" if name == "David" else
    "Unknown"
)
print(spouse)
