import polars as pl

# Create the DataFrame
df = pl.DataFrame(
    {
        "col1": ["a", "b", "a", "b", "c"],
        "col2": ["val1", "val2", "val1", "val3", "val3"]
    }
)

# Group by col1 and concatenate col2 values
out = df.group_by("col1").agg(
    pl.col("col2").str.concat(",")
)

# Display the result
print(out)
