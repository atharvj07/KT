import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.chart.LineChart;
import javafx.scene.chart.NumberAxis;
import javafx.scene.chart.XYChart;
import javafx.stage.Stage;
import javafx.scene.paint.Color;
import javafx.scene.shape.Path;
import java.util.HashMap;
import java.util.Map;

public class CustomLineChart extends Application {

    @Override
    public void start(Stage stage) {
        NumberAxis xAxis = new NumberAxis();
        NumberAxis yAxis = new NumberAxis();
        LineChart<Number, Number> lineChart = new LineChart<>(xAxis, yAxis);

        XYChart.Series<Number, Number> series1 = new XYChart.Series<>();
        series1.setName("Series 1");
        series1.getData().add(new XYChart.Data<>(1, 5));
        series1.getData().add(new XYChart.Data<>(2, 7));
        series1.getData().add(new XYChart.Data<>(3, 6));

        XYChart.Series<Number, Number> series2 = new XYChart.Series<>();
        series2.setName("Series 2");
        series2.getData().add(new XYChart.Data<>(1, 3));
        series2.getData().add(new XYChart.Data<>(2, 8));
        series2.getData().add(new XYChart.Data<>(3, 5));

        lineChart.getData().addAll(series1, series2);
        Scene scene = new Scene(lineChart, 800, 600);

        Map<String, String> colorMap = new HashMap<>();
        colorMap.put("Series 1", "red");
        colorMap.put("Series 2", "blue");

        lineChart.getData().forEach(series -> {
            String color = colorMap.getOrDefault(series.getName(), "black");
            series.getNode().lookup(".chart-series-line").setStyle("-fx-stroke: " + color + ";");
            for (XYChart.Data<Number, Number> data : series.getData()) {
                data.getNode().setStyle("-fx-background-color: " + color + ", white;");
            }
        });

        stage.setScene(scene);
        stage.setTitle("Custom Line Chart");
        stage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
