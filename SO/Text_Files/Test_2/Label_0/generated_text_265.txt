import polars as pl

def convert_and_add_quarter_column(df: pl.DataFrame) -> pl.DataFrame:
    return df.with_columns(
        pl.col("event_date")
        .str.to_date("%d %B %Y")
        .alias("event_date")
    ).with_columns(
        (pl.col("event_date").dt.month().cast(pl.Int64) // 4 + 1)
        .alias("quarter")
    )

df = pl.DataFrame({
    "id": [1, 2], 
    "event_date": ["27 July 2020", "31 December 2020"]
})

df = convert_and_add_quarter_column(df)
