import numpy as np
from numba import njit

@njit(parallel=True)
def find_closest_y(z_coordinates, z_size):
    x_size, y_size = z_coordinates.shape
    y_coordinates = np.empty((x_size, z_size), dtype=np.uint16)
    for i in range(z_size):
        for x in range(x_size):
            closest_y = np.argmin(np.abs(z_coordinates[x] - i))
            y_coordinates[x, i] = closest_y
    return y_coordinates

x_size = 2000
y_size = 2500
z_size = 400
rng = np.random.default_rng(123)
z_coordinates = np.linspace(0, z_size, y_size) + rng.laplace(0, 1, (x_size, y_size))

y_coordinates = find_closest_y(z_coordinates, z_size)
