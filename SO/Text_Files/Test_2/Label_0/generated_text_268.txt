import cv2 as cv
import numpy as np

def find_text_baselines(image_path):
    img = cv.imread(image_path, cv.IMREAD_GRAYSCALE)
    img = cv.GaussianBlur(img, (5, 5), 0)
    _, binary = cv.threshold(img, 0, 255, cv.THRESH_BINARY_INV + cv.THRESH_OTSU)
    
    horizontal_kernel = cv.getStructuringElement(cv.MORPH_RECT, (50, 3))
    dilated = cv.dilate(binary, horizontal_kernel, iterations=1)
    
    contours, _ = cv.findContours(dilated, cv.RETR_EXTERNAL, cv.CHAIN_APPROX_SIMPLE)
    lines = []
    
    for cnt in contours:
        x, y, w, h = cv.boundingRect(cnt)
        baseline_y = y + h - 1
        lines.append((x, baseline_y, x + w, baseline_y))
    
    result = cv.cvtColor(img, cv.COLOR_GRAY2BGR)
    for line in lines:
        cv.line(result, (line[0], line[1]), (line[2], line[3]), (0, 255, 0), 2)
    
    return result

# Example usage:
result_image = find_text_baselines("boston_cooking_a.jpg")
cv.imshow("Baselines", result_image)
cv.waitKey(0)
cv.destroyAllWindows()
