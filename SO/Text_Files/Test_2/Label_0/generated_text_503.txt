import matplotlib.pyplot as plt
import numpy as np
import matplotlib

def plot_with_computer_modern():
    plt.rcParams.update({
        "font.family": "serif",
        "font.serif": ["Computer Modern Roman"],
        "text.usetex": True,
        "axes.labelweight": "bold",
        "xtick.labelsize": 12,
        "ytick.labelsize": 12,
        "xtick.major.width": 1.2,
        "ytick.major.width": 1.2,
        "axes.titlesize": 14,
        "axes.labelsize": 14,
        "legend.fontsize": 12,
    })

    t = np.arange(0.0, 2.0, 0.01)
    s = 1 + np.sin(2 * np.pi * t)

    fig, ax = plt.subplots(figsize=(9, 6))
    ax.plot(t, s)

    ax.set_xlabel(r'\textbf{Time (s)}')
    ax.set_ylabel(r'\textbf{Voltage (mV)}')
    ax.set_title(r'\textbf{A Simple Plot}')
    
    ax.tick_params(axis='both', which='major', labelsize=12, width=1.2)
    
    ax.grid()
    fig.savefig("test.png")
    plt.show()

plot_with_computer_modern()
