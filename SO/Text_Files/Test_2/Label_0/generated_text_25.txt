import java.time.LocalDate;
import java.time.temporal.WeekFields;
import java.util.Locale;

public class IsoWeek {
    private final int year;
    private final int week;

    public IsoWeek(int year, int week) {
        this.year = year;
        this.week = week;
    }

    public static IsoWeek parse(String isoWeek) {
        String[] parts = isoWeek.split("-W");
        return new IsoWeek(Integer.parseInt(parts[0]), Integer.parseInt(parts[1]));
    }

    public boolean isBefore(IsoWeek other) {
        return this.year < other.year || (this.year == other.year && this.week < other.week);
    }

    public int weeksBetween(IsoWeek other) {
        LocalDate startDate = getFirstDayOfWeek();
        LocalDate endDate = other.getFirstDayOfWeek();
        return (int) java.time.temporal.ChronoUnit.WEEKS.between(startDate, endDate);
    }

    private LocalDate getFirstDayOfWeek() {
        return LocalDate.of(year, 1, 4)
                .with(WeekFields.of(Locale.getDefault()).weekOfWeekBasedYear(), week)
                .with(WeekFields.of(Locale.getDefault()).dayOfWeek(), 1);
    }

    @Override
    public String toString() {
        return String.format("%d-W%02d", year, week);
    }

    public static void main(String[] args) {
        IsoWeek weekA = IsoWeek.parse("2012-W48");
        IsoWeek weekB = IsoWeek.parse("2013-W03");

        System.out.println("Week A: " + weekA);
        System.out.println("Week B: " + weekB);
        System.out.println("Is Week A before Week B? " + weekA.isBefore(weekB));
        System.out.println("Weeks between Week A and Week B: " + weekA.weeksBetween(weekB));
    }
}
