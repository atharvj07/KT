import subprocess

def update_python3_symlink():
    try:
        subprocess.run(["sudo", "ln", "-sf", "/usr/bin/python3.9", "/usr/bin/python3"], check=True)
        subprocess.run(["sudo", "ln", "-sf", "/usr/bin/pip3.9", "/usr/bin/pip3"], check=True)
        return "Python3 and pip3 symlinks updated successfully."
    except subprocess.CalledProcessError as e:
        return f"Error: {e}"

print(update_python3_symlink())
