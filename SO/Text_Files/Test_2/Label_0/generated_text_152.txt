@Bean
public OpenAPI customOpenAPI() {
    return new OpenAPI()
        .components(new Components().addSchemas("ResponseEnvelope", 
            new Schema<ResponseEnvelope>()
                .type("object")
                .addProperty("success", new BooleanSchema())
                .addProperty("result", new ObjectSchema())))
        .info(new Info().title("API Documentation").version("1.0"));
}

@Operation(summary = "Create user", description = "Creates a new user and returns wrapped response")
@ApiResponses(value = {
    @ApiResponse(responseCode = "201", description = "User created successfully",
        content = @Content(mediaType = "application/json",
            schema = @Schema(implementation = ResponseEnvelope.class)))
})
@PostMapping("/users")
@PreAuthorize("hasAuthority('create:user')")
public ResponseEntity<ResponseEnvelope<UserDto>> create(@RequestBody @Valid CreateUserCommand command) {
    return ResponseEntity.status(HttpStatus.CREATED)
            .body(new ResponseEnvelope<>(true, userService.create(command)));
}
