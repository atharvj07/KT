import numpy as np

def masked_mean(v, m):
    masked_array = np.ma.array(v, mask=~m)
    return masked_array.mean(axis=1).filled(np.nan)

v = np.array([0, 1, 4, 0, 5])
m = np.array([
    [True, True, False, False, False],
    [True, False, True, False, True],
    [True, True, True, True, True],
])

result = masked_mean(v, m)
print(result)
