import pandas as pd

def count_three_way_conversations(df):
    df = df.sort_values(by=["group_id", "sent_time"])
    three_way_count = 0

    for group_id, group_df in df.groupby("group_id"):
        red_first = group_df[(group_df["from_red"]) & (group_df["touchpoint"] == 1)]
        green_reply = group_df[~group_df["from_red"] & (group_df["touchpoint"] == 2)]
        red_second = group_df[(group_df["from_red"]) & (group_df["touchpoint"] == 3)]

        if not red_first.empty and not green_reply.empty and not red_second.empty:
            first_time = red_first["sent_time"].min()
            green_after_first = green_reply[green_reply["sent_time"] > first_time]

            if not green_after_first.empty:
                green_time = green_after_first["sent_time"].min()
                red_after_green = red_second[red_second["sent_time"] > green_time]

                if not red_after_green.empty:
                    three_way_count += 1

    return three_way_count
