import polars as pl
import numpy as np

def simulate_binomial(df):
    return df.with_columns(
        pl.struct(["daily_n", "prob", "size"])
        .map_elements(lambda x: np.random.binomial(n=x["daily_n"], p=x["prob"], size=x["size"]))
        .cast(pl.Int32)
        .alias("events")
    )
