import pandas as pd
import numpy as np

def handle_large_ints_in_series(values, dtype='int32'):
    # Check the max and min limits for the given dtype
    dtype_limits = {
        'int8': (-128, 127),
        'int16': (-32768, 32767),
        'int32': (-2147483648, 2147483647),
        'int64': (-9223372036854775808, 9223372036854775807)
    }
    
    min_val, max_val = dtype_limits.get(dtype, (-float('inf'), float('inf')))
    
    # Check for any number outside the dtype limits
    for value in values:
        if value < min_val or value > max_val:
            raise ValueError(f"Value {value} is outside the range of {dtype}.")

    # Create the pandas Series with the specified dtype
    return pd.Series(values, dtype=dtype)

def fix_integer_overflow(series, dtype='int32'):
    # Check for potential overflow and correct the values
    min_val, max_val = {
        'int8': (-128, 127),
        'int16': (-32768, 32767),
        'int32': (-2147483648, 2147483647),
        'int64': (-9223372036854775808, 9223372036854775807)
    }[dtype]
    
    corrected_series = series.clip(min=min_val, max=max_val)
    return corrected_series

# Example usage:
try:
    series = handle_large_ints_in_series([2**31], dtype='int32')
except ValueError as e:
    print(e)

# Example of overflow case
overflow_series = pd.Series([2**31-1], dtype='int32') + 1
fixed_series = fix_integer_overflow(overflow_series, dtype='int32')
print(fixed_series)
