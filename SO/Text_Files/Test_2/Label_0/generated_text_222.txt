public void addNewProductRow(LinearLayout parentLayout, Context context) {
    LinearLayout newProductRow = new LinearLayout(context);
    newProductRow.setOrientation(LinearLayout.HORIZONTAL);
    newProductRow.setLayoutParams(new LinearLayout.LayoutParams(
            LinearLayout.LayoutParams.MATCH_PARENT,
            LinearLayout.LayoutParams.WRAP_CONTENT
    ));

    AutoCompleteTextView productName = new AutoCompleteTextView(context);
    productName.setLayoutParams(new LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1));
    productName.setHint("Enter Product");

    EditText productQty = new EditText(context);
    productQty.setLayoutParams(new LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1));
    productQty.setHint("Enter Quantity");
    productQty.setInputType(InputType.TYPE_CLASS_NUMBER);

    EditText productPrice = new EditText(context);
    productPrice.setLayoutParams(new LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1));
    productPrice.setHint("Prod Price");
    productPrice.setFocusable(false);

    EditText productSpecs = new EditText(context);
    productSpecs.setLayoutParams(new LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1));
    productSpecs.setHint("Prod Specs");
    productSpecs.setFocusable(false);

    Button removeButton = new Button(context);
    removeButton.setLayoutParams(new LinearLayout.LayoutParams(
            LinearLayout.LayoutParams.WRAP_CONTENT,
            LinearLayout.LayoutParams.WRAP_CONTENT
    ));
    removeButton.setText("X");
    removeButton.setOnClickListener(v -> parentLayout.removeView(newProductRow));

    newProductRow.addView(productName);
    newProductRow.addView(productQty);
    newProductRow.addView(productPrice);
    newProductRow.addView(productSpecs);
    newProductRow.addView(removeButton);

    parentLayout.addView(newProductRow);
}
