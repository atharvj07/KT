import polars as pl
from pathlib import Path

def load_and_concat_jsons(json_paths):
    def process_json(json_path):
        df = pl.read_json(json_path)
        return df.with_columns(
            pl.col("date").str.to_date(),
            pl.col("fillingDate").str.to_date(),
            pl.col("acceptedDate").str.to_date()
        )

    dfs = [process_json(path) for path in json_paths]
    return pl.concat(dfs).sort("date")

json_files = [
    "../AAPL/single_statements/1985-09-30-quarterly_balance.json",
    "../AAPL/single_statements/1986-09-30-quarterly_balance.json"
]

result_df = load_and_concat_jsons(json_files)
