import bpy

def convert_triangles_to_quads(input_path, output_path):
    bpy.ops.wm.read_factory_settings(use_empty=True)
    
    bpy.ops.import_scene.obj(filepath=input_path, use_smooth_groups=False, use_image_search=False)
    
    obj = bpy.context.selected_objects[0]
    bpy.context.view_layer.objects.active = obj
    
    bpy.ops.object.mode_set(mode='EDIT')
    bpy.ops.mesh.select_all(action='SELECT')
    bpy.ops.mesh.tris_convert_to_quads(beauty=True)
    bpy.ops.object.mode_set(mode='OBJECT')
    
    bpy.ops.export_scene.obj(filepath=output_path)

convert_triangles_to_quads('mushroom-shelve-1-merged.obj', 'quad_mesh.obj')
