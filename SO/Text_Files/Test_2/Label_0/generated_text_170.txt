#!/bin/bash

function find_java_fd() {
    local file_path="$1"
    local pid=$(jps -l | grep "org.apache.catalina.startup.Bootstrap" | awk '{print $1}')
    
    if [[ -z "$pid" ]]; then
        echo "Tomcat process not found."
        return 1
    fi

    local fd_list=$(ls -l /proc/$pid/fd 2>/dev/null | grep "$file_path")

    if [[ -z "$fd_list" ]]; then
        echo "No open file descriptor found for $file_path in Tomcat process."
        return 2
    fi

    echo "Open file descriptor(s) for $file_path in Tomcat process $pid:"
    echo "$fd_list"
}

find_java_fd "/path/to/your/temp/file"
