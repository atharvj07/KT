import polars as pl

def rolling_window_df(df: pl.DataFrame, column: str, period: int, offset: int):
    return df.with_columns(
        pl.col(column).rolling(index_column=pl.int_range(0, df.height), period=f"{period}i", offset=f"{offset}i")
        .alias(f"{column}_rolling")
    )

df = pl.DataFrame({"x": [1.2, 1.3, 3.4, 3.5]})
df = rolling_window_df(df, "x", period=4, offset=-3)
print(df)
