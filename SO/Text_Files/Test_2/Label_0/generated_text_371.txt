import pandas as pd
import re

def markdown_to_dataframe(md_table):
    lines = md_table.strip().split("\n")
    lines = [line for line in lines if not re.match(r"^\|[-\s]+[-\|]+\|$", line)]
    df = pd.read_csv(pd.io.common.StringIO("\n".join(lines)), sep="|", skipinitialspace=True)
    df = df.iloc[:, 1:-1].apply(lambda x: x.str.strip() if x.dtype == "object" else x)
    return df
