from typing import Any, TypeVar
import numpy as np
import numpy.typing as npt

T = TypeVar("T", bound=np.generic)

def explain_ndarray_annotations(
    arr: npt.NDArray[T]
) -> tuple[str, str, str]:
    dtype_info = f"Dtype: {arr.dtype} (Type: {type(arr.dtype)})"
    shape_info = f"Shape: {arr.shape}"
    annotation_info = (
        "Annotations:\n"
        " - np.ndarray[Any, np.dtype[Any]]: Generic array, dtype unspecified.\n"
        " - np.ndarray[Any, np.dtype[T]]: Generic array, dtype restricted.\n"
        " - np.ndarray[T]: Short form for dtype-only specification."
    )
    return dtype_info, shape_info, annotation_info

arr_example = np.array([[1.0, 2.0], [3.0, 4.0]], dtype=np.float64)
result = explain_ndarray_annotations(arr_example)
print("\n".join(result))
