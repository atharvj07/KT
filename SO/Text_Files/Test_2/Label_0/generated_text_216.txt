import java.io.*;
import java.util.ArrayList;
import java.util.List;
import org.apache.poi.openxml4j.opc.OPCPackage;
import org.apache.poi.xwpf.usermodel.*;
import org.openxmlformats.schemas.wordprocessingml.x2006.main.*;

public class WordMerge {

    private final OutputStream result;
    private final List<InputStream> inputs;
    private XWPFDocument first;

    public WordMerge(OutputStream result) {
        this.result = result;
        this.inputs = new ArrayList<>();
    }

    public void add(InputStream stream) throws Exception {
        inputs.add(stream);
        OPCPackage srcPackage = OPCPackage.open(stream);
        XWPFDocument srcDocument = new XWPFDocument(srcPackage);

        if (inputs.size() == 1) {
            first = srcDocument;
        } else {
            addPageBreak(first);
            appendDocument(first, srcDocument);
        }
    }

    private void addPageBreak(XWPFDocument document) {
        XWPFParagraph pageBreak = document.createParagraph();
        pageBreak.setPageBreak(true);
    }

    private void appendDocument(XWPFDocument target, XWPFDocument source) throws Exception {
        for (IBodyElement elem : source.getBodyElements()) {
            if (elem instanceof XWPFParagraph) {
                XWPFParagraph p = (XWPFParagraph) elem;
                XWPFParagraph newPara = target.createParagraph();
                newPara.getCTP().set(p.getCTP().copy());
            } else if (elem instanceof XWPFTable) {
                XWPFTable table = (XWPFTable) elem;
                XWPFTable newTable = target.createTable();
                newTable.getCTTbl().set(table.getCTTbl().copy());
            }
        }
    }

    public void doMerge() throws Exception {
        first.write(result);
    }

    public void close() throws Exception {
        result.flush();
        result.close();
        for (InputStream input : inputs) {
            input.close();
        }
    }

    public static void main(String[] args) throws Exception {
        FileOutputStream faos = new FileOutputStream("/home/victor/result.docx");
        WordMerge wm = new WordMerge(faos);
        wm.add(new FileInputStream("/home/victor/001.docx"));
        wm.add(new FileInputStream("/home/victor/002.docx"));
        wm.doMerge();
        wm.close();
    }
}
