@Configuration
public class WebFluxConfig implements WebFluxConfigurer {

    @Override
    public void configureHttpMessageReaders(List<HttpMessageReader<?>> readers) {
        readers.add(0, new Jaxb2XmlDecoder());
    }

    @Override
    public void configureHttpMessageWriters(List<HttpMessageWriter<?>> writers) {
        writers.add(0, new Jaxb2XmlEncoder());
    }
}
