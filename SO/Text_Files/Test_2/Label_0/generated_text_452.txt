import numpy.typing as npt
import numpy as np
from typing import Any, Union, Sequence

ArrayLike = Union[npt.NDArray[Any], Sequence[Any]]

def f(a: ArrayLike) -> int:
    return len(a)  # type: ignore[arg-type]

def g(a: ArrayLike) -> Any:
    return a[0]  # type: ignore[index]

print(f(np.array([0, 1])), g(np.array([0, 1])))
print(f([0, 1]), g([0, 1]))
