def fix_fastapi_sqlmodel_imports():
    from typing import List, Optional, TYPE_CHECKING
    from sqlmodel import SQLModel, Field, Relationship

    if TYPE_CHECKING:
        from item import Item

    class User(SQLModel, table=True):
        id: int = Field(default=None, primary_key=True)
        age: Optional[int] = None
        bought_items: List["Item"] = Relationship(back_populates="owner")

# item.py
def define_item_model():
    from typing import Optional
    from sqlmodel import SQLModel, Field, Relationship

    class Item(SQLModel, table=True):
        id: int = Field(default=None, primary_key=True)
        price: float
        name: str
        owner_id: Optional[int] = Field(default=None, foreign_key="user.id")
        owner: Optional["User"] = Relationship(back_populates="bought_items")

# main.py
def create_fastapi_app():
    from fastapi import FastAPI
    from user import User

    app = FastAPI()

    @app.get("/", response_model=User)
    def main():
        return {"message": "working just fine"}

    return app
