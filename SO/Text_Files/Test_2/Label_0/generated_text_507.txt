import polars as pl

def add_cut_column(df: pl.DataFrame, column: str, bins: list, labels: list = None):
    return df.with_columns(
        pl.col(column).cut(bins=bins, labels=labels).alias(f"{column}_binned")
    )

df = pl.DataFrame({"a": [1, 2, 3, 4, 5], "b": [2, 3, 4, 5, 6], "x": [1, 3, 5, 7, 9]})

df = add_cut_column(df, "x", bins=[0, 2, 4, 6, 8, 10], labels=["A", "B", "C", "D", "E"])
print(df)
