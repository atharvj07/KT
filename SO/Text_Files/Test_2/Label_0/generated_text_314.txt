import numpy as np
from time import time

def optimized_numpy_comparison(do_cache: bool, array_size: int, num_arrays: int, num_iter: int):
    arrays = {i: np.random.rand(array_size) for i in range(num_arrays)}
    
    if do_cache:
        cache = {}

    for _ in range(num_iter):
        i, j = np.random.randint(num_arrays, size=2)

        if do_cache:
            key = (i, j) if i < j else (j, i)
            if key in cache:
                a = cache[key]
            else:
                a = np.add(arrays[i], arrays[j], dtype=np.float64)
                cache[key] = a
        else:
            a = np.add(arrays[i], arrays[j], dtype=np.float64)

    return arrays
