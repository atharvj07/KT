import com.zaxxer.hikari.HikariDataSource;
import org.mockito.Mockito;
import javax.sql.DataSource;
import java.sql.Connection;
import java.sql.SQLException;
import java.util.Random;

public class DataSourceMocker {
    public static DataSource createUnstableDataSource(DataSource realDataSource) {
        HikariDataSource hikariDataSource = (HikariDataSource) realDataSource;
        HikariDataSource spyDataSource = Mockito.spy(hikariDataSource);

        Mockito.doAnswer(invocation -> {
            if (new Random().nextInt(10) < 3) {
                throw new SQLException("Simulated connection failure");
            }
            return invocation.callRealMethod();
        }).when(spyDataSource).getConnection();

        return spyDataSource;
    }
}
