import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import com.google.protobuf.TextFormat;
import com.google.protobuf.DescriptorProtos.FileDescriptorProto;

public class ProtoParser {
    public static FileDescriptorProto parseProtoFile(String filePath) {
        try {
            Path path = Paths.get(filePath);
            String protoContent = new String(Files.readAllBytes(path));
            FileDescriptorProto.Builder builder = FileDescriptorProto.newBuilder();
            TextFormat.getParser().merge(protoContent, builder);
            return builder.build();
        } catch (IOException e) {
            throw new RuntimeException("Failed to parse proto file", e);
        }
    }
}
