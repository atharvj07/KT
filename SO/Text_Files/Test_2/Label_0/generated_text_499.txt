import polars as pl
import numpy as np

def horizontal_median(df: pl.DataFrame) -> pl.DataFrame:
    return df.with_columns(
        pl.struct(df.select(pl.col(pl.Int64))).map_elements(
            lambda row: float(np.nanmedian([v for v in row.values() if v is not None]))
        ).alias("Horizontal Median")
    )

df = pl.DataFrame({
    "ABC": ["foo", "bar", "foo"],
    "A": [1, 2, 3],
    "B": [2, 1, None],
    "C": [1, 2, 3]
})

print(horizontal_median(df))
