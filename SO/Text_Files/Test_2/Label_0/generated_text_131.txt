import javax.swing.*;
import java.awt.event.ActionEvent;
import javax.swing.text.DefaultEditorKit;

public class TabToSpacesTextArea {
    public static JTextArea createTextAreaWithSpacesTab() {
        JTextArea textArea = new JTextArea();
        Action tabAction = new AbstractAction() {
            @Override
            public void actionPerformed(ActionEvent e) {
                textArea.replaceSelection("    ");
            }
        };
        textArea.getInputMap().put(KeyStroke.getKeyStroke("TAB"), "insert-spaces");
        textArea.getActionMap().put("insert-spaces", tabAction);
        return textArea;
    }

    public static void createAndShowGUI() {
        JFrame frame = new JFrame("JTextArea Tab to Spaces");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        JTextArea textArea = createTextAreaWithSpacesTab();
        frame.add(new JScrollPane(textArea));
        frame.setSize(400, 300);
        frame.setVisible(true);
    }

    public static void main(String[] args) {
        SwingUtilities.invokeLater(TabToSpacesTextArea::createAndShowGUI);
    }
}
