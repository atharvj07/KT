import numpy as np
from timeit import timeit

def benchmark_matmul_vs_dot():
    N = 1300
    xx = np.random.randn(N, N) + 1j
    yy = np.random.randn(N, N) + 1j

    x = np.real(xx).copy()
    y = np.real(yy).copy()

    dot_time = timeit('np.dot(x, y)', number=10, globals=globals())
    matmul_time = timeit('np.matmul(x, y)', number=10, globals=globals())

    print('time for np.matmul: ', matmul_time)
    print('time for np.dot: ', dot_time)

benchmark_matmul_vs_dot()
