import numpy as np
import inspect

def analyze_numpy_sum():
    A = np.random.random(1000)
    
    np_sum_source = inspect.getsource(np.sum)
    array_sum_source = inspect.getsource(A.sum)
    
    np_sum_dis = None
    array_sum_dis = None
    
    try:
        import dis
        np_sum_dis = dis.Bytecode(np.sum).dis()
        array_sum_dis = dis.Bytecode(A.sum).dis()
    except:
        pass
    
    return {
        "np_sum_source": np_sum_source,
        "array_sum_source": array_sum_source,
        "np_sum_dis": np_sum_dis,
        "array_sum_dis": array_sum_dis
    }

analyze_numpy_sum()
