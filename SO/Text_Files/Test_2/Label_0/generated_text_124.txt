import javafx.beans.property.ObjectProperty;
import javafx.beans.property.SimpleObjectProperty;
import javafx.stage.Stage;

public class StageUserDataListener {
    private final ObjectProperty<Object> userDataProperty = new SimpleObjectProperty<>();

    public StageUserDataListener(Stage stage) {
        userDataProperty.addListener((observable, oldValue, newValue) -> 
            System.out.println("New userData: " + newValue)
        );

        stage.userDataProperty().addListener((observable, oldValue, newValue) -> 
            userDataProperty.set(newValue)
        );
    }

    public static void main(String[] args) {
        Stage stage = new Stage();
        StageUserDataListener listener = new StageUserDataListener(stage);

        stage.setUserData(2);
        stage.setUserData(3);
    }
}
