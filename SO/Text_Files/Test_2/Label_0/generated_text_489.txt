import hashlib
import sys

class Color:
    DARKCYAN = '\033[36m'
    GREEN = '\033[32m'
    RESET = '\033[0m'

def color_reset():
    print(Color.RESET, end="")

def get_hash(algorithm):
    filename = input("Enter the file name: ")
    hash_obj = hashlib.new(algorithm)
    with open(filename, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            hash_obj.update(byte_block)
    print(Color.DARKCYAN + f"{algorithm} value has been calculated")
    color_reset()
    return hash_obj.hexdigest()

def verify_checksum(algorithm, hash_value):
    expected_hash = input(f"Enter the expected {algorithm} checksum: ")
    if hash_value == expected_hash:
        print(Color.GREEN + f"{algorithm} checksum verified!")
    else:
        print("Checksum verification failed.")
    color_reset()

def which_hash():
    algorithm = input("Which hash do you want to calculate: sha256 or sha512? \n")
    if algorithm in ["sha256", "sha512"]:
        hash_value = get_hash(algorithm)
        verify_checksum(algorithm, hash_value)
    else:
        print("Type either sha256 or sha512. If you type anything else the program will close...like this.")
        sys.exit()

if __name__ == "__main__":
    which_hash()
