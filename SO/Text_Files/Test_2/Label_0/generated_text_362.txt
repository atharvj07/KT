import asyncio
import httpx

async def fetch_url(url: str, client: httpx.AsyncClient) -> str:
    response = await client.get(url)
    return response.text

async def fetch_all_urls(urls: list[str]) -> list[str]:
    async with httpx.AsyncClient() as client:
        tasks = [fetch_url(url, client) for url in urls]
        return await asyncio.gather(*tasks)

def fetch_multiple_urls(urls: list[str]) -> list[str]:
    result = asyncio.run(fetch_all_urls(urls))
    return result
