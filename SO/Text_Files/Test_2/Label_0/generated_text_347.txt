import polars as pl

def compute_columns(df):
    max_value = df[COL_VALUE].max()

    sum_lower = df.select(pl.col(COL_VALUE)).to_series().apply(
        lambda x: df.filter(pl.col(COL_VALUE) < x).select(pl.sum(COL_VALUE)).to_numpy()[0]
    )

    max_other = df[COL_VALUE].apply(
        lambda x: max_value if x != max_value else df.filter(pl.col(COL_VALUE) != x).select(pl.col(COL_VALUE)).max()
    )

    return df.with_columns(
        pl.Series("sum_lower", sum_lower),
        pl.Series("max_other", max_other),
    )

df = pl.DataFrame({COL_VALUE: [3, 7, 1, 9, 4]})
df = compute_columns(df)
print(df)
