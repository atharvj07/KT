import matplotlib.pyplot as plt

def format_bar_labels(ax, format_func):
    for container in ax.containers:
        labels = [format_func(p.get_height()) for p in container.patches]
        ax.bar_label(container, labels=labels)

# Example usage
fig, ax = plt.subplots()
bars = ax.bar(["A", "B", "C"], [1000, 2500, 5000])

format_bar_labels(ax, lambda x: f"{x:,.2f}")

plt.show()
