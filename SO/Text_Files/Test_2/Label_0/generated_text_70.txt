import com.google.gson.Gson;
import com.mongodb.client.model.InsertOneModel;
import org.bson.Document;
import java.util.List;

public class ParallelStreamProcessor {

    private static final Gson gson = new Gson();

    public static List<InsertOneModel<Document>> processPuzzles(List<Object> puzzles) {
        return puzzles.parallelStream()
                .map(puzzle -> new InsertOneModel<>(Document.parse(gson.toJson(puzzle))))
                .toList();
    }
}
