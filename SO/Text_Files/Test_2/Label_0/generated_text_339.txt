import pandas as pd
import numpy as np

def transform_dataframe(df):
    df['C'] = df.groupby(['A', 'B']).cumcount() + 1
    result = df.pivot_table(index=['A', 'B'], columns='C', values='C', aggfunc='first')
    result.columns = [f'C{i}' for i in result.columns]
    result.reset_index(inplace=True)
    return result

df = pd.DataFrame({
    'A': ['a', 'a', 'a', 'c', 'c', 'c', 'c', 'c', 'e', 'e', 'e', 'e'],
    'B': ['b', 'b', 'b', 'd', 'd', 'd', 'd', 'd', 'b', 'b', 'b', 'b'],
    'C': [1, 2, 3, 7, 8, 5, 6, 3, 4, 3, 2, 1]
})

transformed_df = transform_dataframe(df)
print(transformed_df)
