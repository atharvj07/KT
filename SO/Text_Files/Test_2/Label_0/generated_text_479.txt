class BST:
    def __init__(self, val):
        self.value = val
        self.left = None
        self.right = None

    def add_child(self, val):
        if self.value:
            if val < self.value:
                if self.left is None:
                    self.left = BST(val)
                else:
                    self.left.add_child(val)
            elif val > self.value:
                if self.right is None:
                    self.right = BST(val)
                else:
                    self.right.add_child(val)
        else:
            self.value = val

    def traverse_pre(self):
        result = []
        def helper(node):
            if node:
                result.append(node.value)
                helper(node.left)
                helper(node.right)
        helper(self)
        return result

Tree = BST(5)
Tree.add_child(10)
Tree.add_child(8)
Tree.add_child(2)
Tree.add_child(4)
Tree.add_child(7)

print(Tree.traverse_pre())
