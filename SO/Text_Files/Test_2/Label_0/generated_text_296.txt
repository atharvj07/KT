import pandas as pd

def cumulative_sum_last_occurrences(df):
    df["last_value"] = df.groupby("id")["value"].transform("last")
    df["cumsum_of_value"] = df["last_value"].cumsum()
    return df.drop(columns=["last_value"])

id = ['a','b','c','a','b','e','f','a','b','k']
value = [12,14,3,13,16,7,4,6,10,18]

df = pd.DataFrame({'id': id, 'value': value})
df = cumulative_sum_last_occurrences(df)
print(df)
