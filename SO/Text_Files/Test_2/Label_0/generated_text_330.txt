import numpy as np
from numpy.lib.stride_tricks import sliding_window_view

def find_largest_neighbor_direction(a: np.ndarray) -> np.ndarray:
    L = a.shape[0]
    swv = sliding_window_view(a, (3, 3))  # (L-2) x (L-2) x 3 x 3
    flattened = swv[:, :, [0, 1, 2, 3, 5, 6, 7, 8]]  # Exclude center element
    directions = flattened.argmax(axis=2).astype(np.uint8)
    return directions
