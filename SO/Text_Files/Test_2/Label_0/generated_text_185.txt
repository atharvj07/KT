import java.util.regex.*;

public class AzurePathParser {
    private static final Pattern AZURE_REGEX = Pattern.compile(
            "abfss?://([^/]+)@([^\\.]+)(\\.[^/]+)/?([^?]*)"
    );

    private static final Pattern AZURITE_REGEX = Pattern.compile(
            "abfss?://([^/]+)@([\\d\\.]+):\\d+/([^/]+)/?([^?]*)"
    );

    public static void parsePath(String path) {
        Matcher azureMatcher = AZURE_REGEX.matcher(path);
        Matcher azuriteMatcher = AZURITE_REGEX.matcher(path);

        String fileSystem = "";
        String accountName = "";
        String accountSuffix = "";
        String relativePath = "";

        if (azureMatcher.find()) {
            fileSystem = azureMatcher.group(1);
            accountName = azureMatcher.group(2);
            accountSuffix = azureMatcher.group(3);
            relativePath = azureMatcher.group(4);
        } else if (azuriteMatcher.find()) {
            fileSystem = azuriteMatcher.group(1);
            accountName = azuriteMatcher.group(3);
            accountSuffix = "";
            relativePath = azuriteMatcher.group(4);
        }

        System.out.println("File System: " + fileSystem);
        System.out.println("Account Name: " + accountName);
        System.out.println("Account Suffix: " + accountSuffix);
        System.out.println("Relative Path: " + relativePath);
    }

    public static void main(String[] args) {
        parsePath("abfs://storage@myaccount.dfs.core.windows.net/selim/test.csv");
        parsePath("abfs://storage@127.0.0.1:10000/devstoreaccount1/selim/test.csv");
    }
}
