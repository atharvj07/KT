import pandas as pd

def count_star_combinations(inp):
    pivot_df = inp.pivot_table(index=['month', 'item'], columns='star', aggfunc='size', fill_value=0)
    pivot_df.columns = [f'star_{col}_cnt' for col in pivot_df.columns]
    return pivot_df.reset_index()

inp = pd.DataFrame({'month':[1,1,1,1,1,2,2,2], 
                    'item':[10,10,10,10,20,20,20,20], 
                    'star':[1,2,1,3,3,2,2,3]})

result = count_star_combinations(inp)
print(result)
