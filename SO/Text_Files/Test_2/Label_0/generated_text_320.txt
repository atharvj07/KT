import numpy as np

def ijIndicesToSourceArray_gettingValueOfSourceArrayWithReversedRightLeftAntiDiagonalsAt(i, j, arrShapeRows, arrShapeColumns):
    return (i + j - min(arrShapeRows - 1, i + j), min(arrShapeRows - 1, i + j) - i)

def reverse_antidiagonals(arr):
    arrShapeRows, arrShapeColumns = arr.shape
    result = np.empty_like(arr)
    
    for i in range(arrShapeRows):
        for j in range(arrShapeColumns):
            new_i, new_j = ijIndicesToSourceArray_gettingValueOfSourceArrayWithReversedRightLeftAntiDiagonalsAt(i, j, arrShapeRows, arrShapeColumns)
            result[i, j] = arr[new_i, new_j]
    
    return result

# Example usage
srcArr = np.array([[ 1,  2,  4],
                   [ 3,  5,  7],
                   [ 6,  8, 10],
                   [ 9, 11, 13],
                   [12, 14, 15]])

resultArr = reverse_antidiagonals(srcArr)
print(resultArr)
