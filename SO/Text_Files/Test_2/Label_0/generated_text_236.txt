package org.apache.misc.alg.dag;

import java.util.*;

public class OptimizedDinicCalculator<T> implements MaxAntichainCalculator<T> {

    private final Map<String, Map<String, Integer>> network = new HashMap<>();
    private final Map<String, Integer> nodeIndex = new HashMap<>();
    private final List<String> nodes = new ArrayList<>();
    private int[] level;

    public OptimizedDinicCalculator() {
        addNode("src");
        addNode("sink");
    }

    private void addNode(String node) {
        if (!nodeIndex.containsKey(node)) {
            nodeIndex.put(node, nodes.size());
            nodes.add(node);
            network.put(node, new HashMap<>());
        }
    }

    private void addEdge(String from, String to, int capacity) {
        addNode(from);
        addNode(to);
        network.get(from).put(to, capacity);
        network.get(to).put(from, 0);
    }

    private void bfs(String source) {
        level = new int[nodes.size()];
        Arrays.fill(level, -1);
        level[nodeIndex.get(source)] = 0;

        Queue<String> queue = new LinkedList<>();
        queue.offer(source);

        while (!queue.isEmpty()) {
            String u = queue.poll();
            int uIndex = nodeIndex.get(u);
            for (Map.Entry<String, Integer> entry : network.get(u).entrySet()) {
                String v = entry.getKey();
                int capacity = entry.getValue();
                if (capacity > 0 && level[nodeIndex.get(v)] == -1) {
                    level[nodeIndex.get(v)] = level[uIndex] + 1;
                    queue.offer(v);
                }
            }
        }
    }

    private int dfs(String u, int flow, String sink) {
        if (u.equals(sink)) {
            return flow;
        }
        int uIndex = nodeIndex.get(u);
        for (Map.Entry<String, Integer> entry : network.get(u).entrySet()) {
            String v = entry.getKey();
            int capacity = entry.getValue();
            int vIndex = nodeIndex.get(v);
            if (capacity > 0 && level[uIndex] < level[vIndex]) {
                int sent = dfs(v, Math.min(flow, capacity), sink);
                if (sent > 0) {
                    network.get(u).put(v, capacity - sent);
                    network.get(v).put(u, network.get(v).getOrDefault(u, 0) + sent);
                    return sent;
                }
            }
        }
        return 0;
    }

    private Set<String> reach(Map<T, Set<T>> graph, T t, Set<String> visited) {
        Queue<T> queue = new LinkedList<>();
        queue.add(t);
        while (!queue.isEmpty()) {
            T current = queue.poll();
            String currentKey = "A" + current.toString();
            visited.add(currentKey);
            for (T neighbor : graph.get(current)) {
                String neighborKey = "B" + neighbor.toString();
                if (!visited.contains(neighborKey)) {
                    queue.add(neighbor);
                    visited.add(neighborKey);
                }
            }
        }
        return visited;
    }

    public int calculator(Map<T, Set<T>> graph) {
        for (T t : graph.keySet()) {
            addEdge("src", "A" + t, 1);
            addEdge("B" + t, "sink", 1);
            Set<String> visitedSubset = new HashSet<>();
            for (String u : reach(graph, t, visitedSubset)) {
                addEdge("A" + t, u, 1);
            }
        }

        int maxFlow = 0;
        while (true) {
            bfs("src");
            if (level[nodeIndex.get("sink")] == -1) break;
            int flow;
            while ((flow = dfs("src", Integer.MAX_VALUE, "sink")) > 0) {
                maxFlow += flow;
            }
        }
        return graph.size() - maxFlow;
    }
}
