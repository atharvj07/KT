import java.util.*;
import java.util.stream.Collectors;

public class RecordProcessor {

    public static List<String> calculatePercentageDistribution(List<Record> records) {
        return records.stream()
                .collect(Collectors.groupingBy(
                        r -> r.getWeekIndex() + "|" + r.getSourceNode(),
                        Collectors.summingInt(r -> Integer.parseInt(r.getQuantity()))
                ))
                .entrySet().stream()
                .flatMap(entry -> {
                    String[] keys = entry.getKey().split("\\|");
                    String week = keys[0], source = keys[1];
                    int totalQuantity = entry.getValue();

                    return records.stream()
                            .filter(r -> r.getWeekIndex().equals(week) && r.getSourceNode().equals(source))
                            .map(r -> {
                                double percentage = (Double.parseDouble(r.getQuantity()) / totalQuantity) * 100;
                                return String.format("%s  %s  %s   %.2f%%", week, source, r.getDestinationNode(), percentage);
                            });
                })
                .distinct()
                .collect(Collectors.toList());
    }
}
