from pydantic import BaseModel, ValidationError, field_validator
from typing import List

class Salary(BaseModel):
    gross: int
    net: int
    tax: int

class Employee(BaseModel):
    name: str
    age: int
    salary: Salary

def validate_instance(model_class, data):
    errors = []
    try:
        model_class(**data)
    except ValidationError as e:
        errors.extend(e.errors())

    return errors

salary_data = {"gross": "hello", "net": 1000, "tax": 10}
employee_data = {"name": "Mattia", "age": "hello", "salary": salary_data}

errors = validate_instance(Employee, employee_data)
print(errors)
