from abc import ABC, abstractmethod

class FooBase(ABC):
    bar: str
    baz: int

    def __init__(self):
        self.bar = "bar"
        self.baz = "baz"

    @property
    @abstractmethod
    def foo(self) -> str:
        pass


class Foo(FooBase):
    @property
    def foo(self) -> str:
        return "hello"
