import java.util.*;

class Edge {
    int toVertex, weight;

    Edge(int v, int w) {
        toVertex = v;
        weight = w;
    }
}

class Solver {
    public static int solve(List<List<Edge>> T, long L) {
        int N = T.size();
        int minEdges = Integer.MAX_VALUE;
        Map<Long, Integer> prefixMap = new HashMap<>();

        for (int i = 0; i < N; i++) {
            minEdges = Math.min(minEdges, dfs(T, L, i, -1, 0, 0, prefixMap));
        }

        return minEdges == Integer.MAX_VALUE ? -1 : minEdges;
    }

    private static int dfs(List<List<Edge>> T, long L, int node, int parent, long sum, int edges, Map<Long, Integer> prefixMap) {
        int minEdges = Integer.MAX_VALUE;
        if (sum == L) {
            return edges;
        }

        if (prefixMap.containsKey(sum - L)) {
            minEdges = Math.min(minEdges, edges - prefixMap.get(sum - L));
        }

        prefixMap.put(sum, edges);

        for (Edge edge : T.get(node)) {
            if (edge.toVertex != parent) {
                minEdges = Math.min(minEdges, dfs(T, L, edge.toVertex, node, sum + edge.weight, edges + 1, prefixMap));
            }
        }

        prefixMap.remove(sum);
        return minEdges;
    }
}

public class Main {
    static void putEdge(List<List<Edge>> T, int vertex1, int vertex2, int weight) {
        T.get(vertex1).add(new Edge(vertex2, weight));
        T.get(vertex2).add(new Edge(vertex1, weight));
    }

    public static void main(String[] args) {
        List<List<Edge>> T = new ArrayList<>();
        int N = 8;
        for (int i = 0; i < N; i++) T.add(new ArrayList<>());

        putEdge(T, 0, 1, 2);
        putEdge(T, 1, 2, 1);
        putEdge(T, 1, 3, 2);
        putEdge(T, 2, 6, 1);
        putEdge(T, 6, 7, 1);
        putEdge(T, 3, 4, 1);
        putEdge(T, 3, 5, 4);

        System.out.println(Solver.solve(T, 5L));
    }
}
