import polars as pl

def slice_polars_column():
    df = pl.DataFrame(
        {
            "ColumnA": ["abcd", "efgh", "ijkl", "mnop"]
        }
    )
    
    result = df.with_columns(
        pl.col("ColumnA").str.slice(0, 2).alias("ColumnA_Sliced")
    )
    
    return result

sliced_df = slice_polars_column()
