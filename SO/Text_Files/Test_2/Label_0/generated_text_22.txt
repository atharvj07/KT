import java.util.HashSet;
import java.util.Set;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.Future;

public class FindMultiples {

    public int getSum(int n) {
        Set<Integer> dividends = new HashSet<>();
        ExecutorService executor = Executors.newFixedThreadPool(3);

        Future<?> future1 = executor.submit(() -> {
            for (int i = 3; i <= n; i += 3) {
                synchronized (dividends) {
                    dividends.add(i);
                }
            }
        });

        Future<?> future2 = executor.submit(() -> {
            for (int i = 5; i <= n; i += 5) {
                synchronized (dividends) {
                    dividends.add(i);
                }
            }
        });

        Future<?> future3 = executor.submit(() -> {
            for (int i = 7; i <= n; i += 7) {
                synchronized (dividends) {
                    dividends.add(i);
                }
            }
        });

        try {
            future1.get();
            future2.get();
            future3.get();
        } catch (Exception e) {
            e.printStackTrace();
        }

        executor.shutdown();

        int sum = 0;
        for (int num : dividends) {
            sum += num;
        }

        return sum;
    }

    public static void main(String[] args) {
        FindMultiples fm = new FindMultiples();
        System.out.println(fm.getSum(10));
        System.out.println(fm.getSum(1000));
        System.out.println(fm.getSum(76293));
    }
}
