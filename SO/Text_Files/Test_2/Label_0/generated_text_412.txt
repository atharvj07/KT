import serial
import time

def pico_usb_communication(port='/dev/ttyACM0', baudrate=115200, message="Hello, Pico!"):
    try:
        with serial.Serial(port, baudrate, timeout=1) as ser:
            time.sleep(2)  # Allow time for the connection to establish
            ser.write((message + "\n").encode())  # Send data
            time.sleep(0.1)  # Give Pico time to process
            response = ser.readline().decode().strip()  # Read response
            return response
    except serial.SerialException as e:
        return f"Error: {e}"

# Example usage:
response = pico_usb_communication()
print("Response from Pico:", response)
