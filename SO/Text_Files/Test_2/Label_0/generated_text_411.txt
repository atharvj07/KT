import numpy as np
import pandas as pd

def custom_cumsum(arr):
    if isinstance(arr, pd.Series):
        return arr.cumsum()
    else:
        mask = np.isnan(arr)
        cumsum = np.cumsum(np.where(mask, 0, arr))
        cumsum[mask] = np.nan
        return cumsum

df = pd.DataFrame({'column': [1, 2, 3, np.nan, 4, 5, 6]})
result = custom_cumsum(df['column'])
