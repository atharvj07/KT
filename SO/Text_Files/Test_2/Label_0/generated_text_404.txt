import pytest
import json
from starlette.types import Scope, Receive, Send, Message
from starlette.requests import Request
from starlette.responses import Response
from starlette.testclient import TestClient
from fastapi import FastAPI
from typing import Callable
from mymiddleware import MyMiddleware  # Replace with actual import path

async def mock_receive() -> Message:
    return {
        "type": "http.request",
        "body": json.dumps({"key": "value"}).encode("utf-8"),
        "more_body": False,
    }

async def mock_send(message: Message):
    pass

@pytest.mark.asyncio
async def test_middleware_direct():
    app = MyMiddleware(lambda scope, receive, send: Response("OK")(scope, receive, send))
    scope: Scope = {"type": "http", "method": "POST"}
    await app(scope, mock_receive, mock_send)
    
@pytest.fixture
def client():
    app = FastAPI()

    @app.post("/")
    async def test_endpoint(request: Request):
        return await request.json()

    app.add_middleware(MyMiddleware)
    return TestClient(app)

def test_middleware_with_client(client):
    response = client.post("/", json={"key": "value"})
    assert response.status_code == 200
    assert response.json() == {"key": "value", "some_field": "foobar"}
