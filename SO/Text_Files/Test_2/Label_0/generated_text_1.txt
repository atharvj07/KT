import org.springframework.http.ResponseEntity;
import org.springframework.web.client.RestTemplate;
import org.springframework.web.util.UriComponentsBuilder;
import java.net.URI;
import java.util.Map;

public class RestTemplateClient {

    private final RestTemplate restTemplate;

    public RestTemplateClient() {
        this.restTemplate = new RestTemplate();
    }

    public ResponseEntity<String> makeRequest(String baseUrl, String path, Map<String, String> queryParams) {
        URI uri = UriComponentsBuilder.fromHttpUrl(baseUrl)
                .path(path)
                .queryParams(CollectionUtils.toMultiValueMap(queryParams))
                .build()
                .toUri();

        return restTemplate.getForEntity(uri, String.class);
    }

    public ResponseEntity<String> postRequest(String baseUrl, String path, Object requestBody) {
        URI uri = UriComponentsBuilder.fromHttpUrl(baseUrl)
                .path(path)
                .build()
                .toUri();

        return restTemplate.postForEntity(uri, requestBody, String.class);
    }

    public ResponseEntity<String> putRequest(String baseUrl, String path, Object requestBody) {
        URI uri = UriComponentsBuilder.fromHttpUrl(baseUrl)
                .path(path)
                .build()
                .toUri();

        restTemplate.put(uri, requestBody);
        return ResponseEntity.noContent().build();
    }

    public ResponseEntity<String> deleteRequest(String baseUrl, String path) {
        URI uri = UriComponentsBuilder.fromHttpUrl(baseUrl)
                .path(path)
                .build()
                .toUri();

        restTemplate.delete(uri);
        return ResponseEntity.noContent().build();
    }
}
