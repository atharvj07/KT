import numpy as np
from sklearn.preprocessing import PolynomialFeatures
from sklearn.linear_model import LinearRegression

def fit_polynomial_and_compute_derivatives(X, y, degree=2):
    poly = PolynomialFeatures(degree)
    Xp = poly.fit_transform(X)
    model = LinearRegression()
    model.fit(Xp, y)

    def gradient_and_hessian(x):
        x_transformed = poly.transform(x.reshape(1, -1))
        coefs = model.coef_.flatten()
        features = poly.get_feature_names_out()
        
        grad = np.zeros(X.shape[1])
        hessian = np.zeros((X.shape[1], X.shape[1]))
        
        for i, feature in enumerate(features[1:], start=1):
            powers = np.array([int(p) for p in feature.split(' ') if p.isdigit()])
            if len(powers) < X.shape[1]:
                powers = np.pad(powers, (0, X.shape[1] - len(powers)))
            
            grad += coefs[i] * np.where(powers > 0, powers * x ** (powers - 1), 0)
            
            for j in range(X.shape[1]):
                second_deriv = np.where(powers > 1, powers * (powers - 1) * x ** (powers - 2), 0)
                hessian[:, j] += coefs[i] * second_deriv

        return grad, hessian

    return model, gradient_and_hessian
