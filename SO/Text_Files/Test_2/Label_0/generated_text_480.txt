from flask import Flask, render_template, Response
import threading
import time
import map_plotting_test as mpt

app = Flask(__name__)

def generate_loading():
    yield "<html><head><meta http-equiv='refresh' content='5'></head><body>"
    yield "<h1>Loading map...</h1>"
    yield "</body></html>"

@app.route('/')
def render_loading():
    threading.Thread(target=create_map_background).start()
    return Response(generate_loading(), content_type='text/html')

def create_map_background():
    mpt.create_map()
    time.sleep(1)  

@app.route('/map')
def render_the_map():
    return render_template("map.html")

if __name__ == '__main__':
    app.run(debug=True)
