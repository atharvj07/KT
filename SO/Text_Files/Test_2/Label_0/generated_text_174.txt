@Override
public boolean isValid(ProductExportFilters productExportFilters, ConstraintValidatorContext constraintValidatorContext) {
    if (productExportFilters == null) {
        constraintValidatorContext.disableDefaultConstraintViolation();
        constraintValidatorContext.buildConstraintViolationWithTemplate("Product export filters cannot be null.")
                .addConstraintViolation();
        return false;
    }
    try {
        DateTimeFormatter dateTimeFormatter = DateTimeFormatter.ISO_DATE_TIME;
        LocalDateTime startDate = null;
        LocalDateTime endDate = null;

        if (productExportFilters.getStartDate() != null) {
            startDate = LocalDateTime.from(dateTimeFormatter.parse(productExportFilters.getStartDate()));
        }
        if (productExportFilters.getEndDate() != null) {
            endDate = LocalDateTime.from(dateTimeFormatter.parse(productExportFilters.getEndDate()));
        }
        if (startDate == null && endDate == null) {
            return true;
        }
        if (startDate != null && endDate == null) {
            constraintValidatorContext.disableDefaultConstraintViolation();
            constraintValidatorContext.buildConstraintViolationWithTemplate("End date must be provided if start date is set.")
                    .addConstraintViolation();
            return false;
        }
        if (startDate == null && endDate != null) {
            constraintValidatorContext.disableDefaultConstraintViolation();
            constraintValidatorContext.buildConstraintViolationWithTemplate("Start date must be provided if end date is set.")
                    .addConstraintViolation();
            return false;
        }
        if (!startDate.equals(endDate) && startDate.isAfter(endDate)) {
            constraintValidatorContext.disableDefaultConstraintViolation();
            constraintValidatorContext.buildConstraintViolationWithTemplate("Start date must be before or equal to end date.")
                    .addConstraintViolation();
            return false;
        }
        return true;
    } catch (DateTimeException e) {
        constraintValidatorContext.disableDefaultConstraintViolation();
        constraintValidatorContext.buildConstraintViolationWithTemplate("Invalid date format. Expected ISO date-time format.")
                .addConstraintViolation();
        return false;
    }
}
