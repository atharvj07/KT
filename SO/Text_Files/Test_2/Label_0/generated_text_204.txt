public void copyDatabaseFromAssets(String assetFolder, String outputFolder) {
    Executors.newSingleThreadExecutor().execute(() -> {
        Context myContext = requireContext();
        AssetManager assetManager = myContext.getAssets();
        File databasesFolder = new File(myContext.getExternalFilesDir(null).getParent(), outputFolder);
        if (!databasesFolder.exists() && !databasesFolder.mkdirs()) {
            Log.e("DatabaseCopy", "Failed to create database folder.");
            return;
        }
        try {
            String[] files = assetManager.list(assetFolder);
            if (files != null) {
                for (String filename : files) {
                    File outFile = new File(databasesFolder, filename);
                    if (!outFile.exists()) {
                        try (InputStream in = assetManager.open(assetFolder + "/" + filename);
                             OutputStream out = new FileOutputStream(outFile)) {
                            byte[] buffer = new byte[4096];
                            int bytesRead;
                            while ((bytesRead = in.read(buffer)) != -1) {
                                out.write(buffer, 0, bytesRead);
                            }
                            out.flush();
                        }
                    }
                }
            } else {
                Log.e("DatabaseCopy", "No files found in assets folder: " + assetFolder);
            }
        } catch (IOException e) {
            Log.e("DatabaseCopy", "Error copying database files", e);
        }
    });
}
