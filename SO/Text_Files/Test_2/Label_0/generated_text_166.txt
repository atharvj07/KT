import java.awt.*;
import java.awt.font.FontRenderContext;
import java.awt.geom.Line2D;
import java.awt.image.BufferedImage;
import javax.swing.*;

public class Main {
    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            JFrame jf = new JFrame("Demo");
            Container cp = jf.getContentPane();
            MyCanvas tl = new MyCanvas();
            cp.add(tl);
            jf.setSize(500, 250);
            jf.setVisible(true);
            jf.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        });
    }
}

class MyCanvas extends JComponent {
    @Override
    protected void paintComponent(Graphics g) {
        if (g instanceof Graphics2D g2) {
            g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            g2.setFont(Font.decode("Times New Roman-26"));
            g2.drawString("The properly scaled cake is a lie.", 70, 40);
            g2.setStroke(new BasicStroke(2.3f));
            g2.draw(new Line2D.Double(420, 10, 425, 70));

            Image I = createScalingAwareBuffer(this, 500, 150);
            Graphics2D g2_ = (Graphics2D) I.getGraphics();
            g2_.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            g2_.setFont(Font.decode("Times New Roman-26"));
            g2_.drawString("The properly scaled cake is a lie.", 70, 40);
            g2_.setStroke(new BasicStroke(2.3f));
            g2_.draw(new Line2D.Double(420, 10, 425, 70));
            g2_.dispose();
            g2.drawImage(I, 0, 130, null);
        }
    }

    private Image createScalingAwareBuffer(JComponent component, int width, int height) {
        Graphics2D g2d = (Graphics2D) component.getGraphics();
        if (g2d == null) return new BufferedImage(width, height, BufferedImage.TYPE_INT_ARGB);

        GraphicsConfiguration config = g2d.getDeviceConfiguration();
        double scaleX = g2d.getFontRenderContext().getTransform().getScaleX();
        double scaleY = g2d.getFontRenderContext().getTransform().getScaleY();
        g2d.dispose();

        int scaledWidth = (int) Math.ceil(width * scaleX);
        int scaledHeight = (int) Math.ceil(height * scaleY);
        BufferedImage image = config.createCompatibleImage(scaledWidth, scaledHeight, Transparency.TRANSLUCENT);

        Graphics2D g2 = image.createGraphics();
        g2.scale(scaleX, scaleY);
        g2.dispose();

        return image;
    }
}
