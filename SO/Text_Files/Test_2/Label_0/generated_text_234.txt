import com.opencsv.CSVWriter;
import java.io.StringWriter;
import java.util.List;
import java.util.Map;

public String convertDmnRuleTestResultToCsv(DmnRuleTestResult result) {
    List<DmnTestCaseResult> results = result.getTestCases();
    try (StringWriter sw = new StringWriter();
         CSVWriter writer = new CSVWriter(sw, CSVWriter.DEFAULT_SEPARATOR,
                 CSVWriter.NO_QUOTE_CHARACTER, CSVWriter.NO_ESCAPE_CHARACTER, CSVWriter.DEFAULT_LINE_END)) {
        
        String[] header = {"TestInput", "ExpectedOutput", "TestOutput", "ErrorMessage"};
        writer.writeNext(header);

        for (DmnTestCaseResult testCase : results) {
            String testInput = testCase.getTestInput().toString();
            String expectedOutput = testCase.getExpectedOutput().toString();
            String errorMessage = testCase.getErrorMessage();

            if (testCase.getTestOutput() != null && !testCase.getTestOutput().isEmpty()) {
                for (Map<String, Object> testOutput : testCase.getTestOutput()) {
                    writer.writeNext(new String[]{
                            testInput,
                            expectedOutput,
                            testOutput.toString(),
                            errorMessage
                    });
                }
            } else {
                writer.writeNext(new String[]{
                        testInput,
                        expectedOutput,
                        "No Output",
                        errorMessage
                });
            }
        }
        return sw.toString();
    } catch (Exception ex) {
        throw new CsvParseException(ex.getMessage());
    }
}
