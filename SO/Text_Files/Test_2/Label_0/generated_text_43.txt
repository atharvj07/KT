import hudson.AbortException
import org.mockito.invocation.InvocationOnMock
import org.mockito.stubbing.Answer

import static org.mockito.ArgumentMatchers.*
import static org.mockito.Mockito.*

class DSLMock {

    def mock

    DSLMock() {
        this.mock = mock(DSL.class)

        when(mock.invokeMethod(eq("error"), argThat { it instanceof Object[] && it.length > 0 && it[0] instanceof String }))
                .thenAnswer(new Answer<String>() {
                    @Override
                    String answer(InvocationOnMock invocation) throws Throwable {
                        throw new AbortException((String) ((Object[]) invocation.getArguments()[1])[0])
                    }
                })
    }
}
