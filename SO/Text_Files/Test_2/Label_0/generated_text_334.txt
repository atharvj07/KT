import subprocess
import sys
import os

def install_package_with_different_python(package_name, python_version='3.8'):
    # Check if the specified Python version is installed
    python_path = f'/usr/bin/python{python_version}'
    
    if not os.path.exists(python_path):
        print(f"Python {python_version} is not installed.")
        return

    # Use pip from the specified Python version
    subprocess.check_call([python_path, '-m', 'pip', 'install', package_name])

install_package_with_different_python('your_package_name', '3.8')
