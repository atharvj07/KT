import numpy as np
import concurrent.futures

def parallel_sort(arr, num_workers=None):
    if num_workers is None:
        num_workers = min(8, len(arr) // int(1e7))  # Adjust based on array size

    split_arrays = np.array_split(arr, num_workers)

    with concurrent.futures.ThreadPoolExecutor(max_workers=num_workers) as executor:
        sorted_splits = list(executor.map(np.sort, split_arrays))

    return np.concatenate(sorted_splits)
