public static void sortFeatureAnalyzeDTOResults(List<FeatureAnalyzeDTOResult> list) {
    YearMonth thisMonth = YearMonth.now();
    List<String> monthList = new ArrayList<>();
    DateTimeFormatter formatter = DateTimeFormatter.ofPattern("MMMM", Locale.ENGLISH);

    for (int i = 0; i < 6; i++) {
        monthList.add(thisMonth.minusMonths(i).format(formatter).toUpperCase());
    }

    list.sort(Comparator.comparingInt(o -> {
        int index = monthList.indexOf(o.getMonth().toUpperCase());
        return index == -1 ? Integer.MAX_VALUE : index;
    }));
}
