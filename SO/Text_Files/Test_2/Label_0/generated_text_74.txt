import org.springframework.context.ApplicationContext;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.ComponentScan;
import org.springframework.context.annotation.Configuration;
import org.springframework.stereotype.Component;
import java.util.Arrays;
import java.util.Map;

interface InputProcessor {
    void process(int... args);
}

@Component("A")
class ClassA implements InputProcessor {
    public ClassA() {}
    @Override
    public void process(int... args) {
        System.out.println("ClassA created with " + Arrays.toString(args));
    }
}

@Component("B")
class ClassB implements InputProcessor {
    public ClassB() {}
    @Override
    public void process(int... args) {
        System.out.println("ClassB created with " + Arrays.toString(args));
    }
}

@Component("C")
class ClassC implements InputProcessor {
    public ClassC() {}
    @Override
    public void process(int... args) {
        System.out.println("ClassC created with " + Arrays.toString(args));
    }
}

@Configuration
@ComponentScan
class AppConfig {
    @Bean
    public Factory factory(ApplicationContext context) {
        return new Factory(context);
    }
}

class Factory {
    private final Map<String, InputProcessor> beans;

    public Factory(ApplicationContext context) {
        this.beans = context.getBeansOfType(InputProcessor.class);
    }

    public void create(String input) {
        String[] parts = input.split(" ");
        String key = parts[0];
        int[] args = Arrays.stream(parts, 1, parts.length).mapToInt(Integer::parseInt).toArray();
        if (beans.containsKey(key)) {
            beans.get(key).process(args);
        } else {
            System.out.println("No matching class found for key: " + key);
        }
    }
}

public class Main {
    public static void main(String[] args) {
        ApplicationContext context = new AnnotationConfigApplicationContext(AppConfig.class);
        Factory factory = context.getBean(Factory.class);

        factory.create("A 1 2");
        factory.create("B 3 4 5");
        factory.create("C 1");
    }
}
