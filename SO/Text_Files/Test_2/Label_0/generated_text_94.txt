import java.lang.annotation.*;
import java.lang.reflect.Field;

@Target(ElementType.FIELD)
@Retention(RetentionPolicy.RUNTIME)
@Documented
@interface Annotation {
    Class<?> className() default Void.class;
}

class AnotherClass {}

class Main {
    @Annotation
    private AnotherClass annotatedAttribute;
    
    @Annotation(className = String.class)
    private String explicitlyAnnotated;

    public static void main(String[] args) {
        extractAndPrintClassName(Main.class);
    }

    public static void extractAndPrintClassName(Class<?> clazz) {
        for (Field field : clazz.getDeclaredFields()) {
            Annotation annotation = field.getAnnotation(Annotation.class);
            if (annotation != null) {
                Class<?> defaultClass = annotation.className() == Void.class ? field.getType() : annotation.className();
                System.out.println("Field: " + field.getName() + " -> Class: " + defaultClass.getSimpleName());
            }
        }
    }
}
