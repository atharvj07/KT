import pandas as pd
import numpy as np

def fill_missing_values(df):
    for col in df.columns:
        if col + '_a' in df.columns:
            df[col] = df[col].fillna(df[col + '_a'])
    return df

df = pd.DataFrame(
    {
        'x': [1, np.nan, 3, np.nan, 5],
        'y': [np.nan, 7, 8, 9, np.nan],
        'x_a': [1, 2, 3, 4, 5],
        'y_a': [6, 7, 8, 9, 10]
    }
)

df = fill_missing_values(df)
print(df)
