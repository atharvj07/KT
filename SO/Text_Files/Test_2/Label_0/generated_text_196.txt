import java.io.RandomAccessFile;

public class FileOverwrite {
    public static void overwriteFile(String filePath, String content, long position) {
        try (RandomAccessFile raf = new RandomAccessFile(filePath, "rw")) {
            raf.seek(position);
            raf.write(content.getBytes());
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public static void main(String[] args) {
        overwriteFile("test.txt", "123456789", 0);
        overwriteFile("test.txt", "hello", 0);
    }
}
