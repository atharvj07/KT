import polars as pl

def replace_values(df: pl.DataFrame) -> pl.DataFrame:
    return df.with_columns(
        pl.when(pl.col("A").is_in(["foo", "spam"]))
        .then(pl.lit("XX"))
        .otherwise(pl.col("L"))
        .alias("L"),
        pl.when(pl.col("A").is_in(["foo", "spam"]))
        .then(pl.lit("XX"))
        .otherwise(pl.col("G"))
        .alias("G")
    )

df = pl.DataFrame(
    {
        "A": ["foo", "ham", "spam", "egg"],
        "L": ["A54", "A12", "B84", "C12"],
        "G": ["X34", "C84", "G96", "L6"],
    }
)

df = replace_values(df)
print(df)
