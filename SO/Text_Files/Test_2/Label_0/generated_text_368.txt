from icu import Collator, Locale

def create_multi_locale_sorter(primary_locale="en_US.UTF-8", secondary_locale="fa_IR.UTF-8"):
    collator1 = Collator.createInstance(Locale(primary_locale))
    collator2 = Collator.createInstance(Locale(secondary_locale))

    def sort_key(s):
        primary_key = collator1.getSortKey(s)
        secondary_key = collator2.getSortKey(s) if not s.isascii() else b""
        return primary_key, secondary_key

    return sort_key
