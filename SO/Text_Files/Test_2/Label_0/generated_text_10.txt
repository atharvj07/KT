import java.util.*;

public class MountainSequence {

    public static int solve(List<Integer> list) {
        int n = list.size();
        List<Integer> uniqueSorted = new ArrayList<>(new TreeSet<>(list));
        
        int maxLen = 0;
        
        for (int peak = 1; peak < uniqueSorted.size() - 1; peak++) {
            int peakValue = uniqueSorted.get(peak);

            List<Integer> increasing = new ArrayList<>();
            List<Integer> decreasing = new ArrayList<>();
            List<Integer> increasingAgain = new ArrayList<>();

            for (int num : list) {
                if (num < peakValue) {
                    if (decreasing.isEmpty()) {
                        if (increasing.isEmpty() || num > increasing.get(increasing.size() - 1)) {
                            increasing.add(num);
                        }
                    }
                } else if (num == peakValue) {
                    if (!increasing.isEmpty()) {
                        decreasing.add(num);
                    }
                } else {
                    if (!decreasing.isEmpty()) {
                        if (increasingAgain.isEmpty() || num > increasingAgain.get(increasingAgain.size() - 1)) {
                            increasingAgain.add(num);
                        }
                    }
                }
            }

            if (!increasing.isEmpty() && !decreasing.isEmpty() && !increasingAgain.isEmpty()) {
                maxLen = Math.max(maxLen, increasing.size() + decreasing.size() + increasingAgain.size());
            }
        }

        return maxLen;
    }

    public static void main(String[] args) {
        System.out.println(solve(List.of(1, 3, 5, 4, 2, 6, 8, 7, 9))); // Expected output: 9
        System.out.println(solve(List.of(5, 5, 2, 1, 3, 4, 5))); // Expected output: 6
        System.out.println(solve(List.of(1, 100))); // Expected output: 2
        System.out.println(solve(List.of(2, 1, 3, 3, 1, 2, 1, 2, 3))); // Expected output: 7
    }
}
