import polars as pl

def explode_with_null_expansion(df: pl.DataFrame) -> pl.DataFrame:
    return df.with_columns(
        pl.when(pl.col("y").is_null())
        .then(pl.lit(None).repeat_by(pl.col("x").list.len()))
        .otherwise(pl.col("y"))
        .alias("y")
    ).explode(["x", "y"])
