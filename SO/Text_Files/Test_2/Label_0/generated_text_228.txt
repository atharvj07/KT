import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.TextArea;
import javafx.scene.layout.VBox;
import javafx.scene.text.Font;
import javafx.stage.Stage;
import java.io.File;
import java.net.MalformedURLException;

public class EmojiTextArea extends Application {
    @Override
    public void start(Stage primaryStage) throws MalformedURLException {
        String emojiText = "?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ??";
        
        TextArea textArea = new TextArea(emojiText);
        textArea.setWrapText(true);

        File fontFile = new File("NotoColorEmoji-Regular.ttf"); // Ensure correct path
        String fontUrl = fontFile.toURI().toURL().toExternalForm();
        Font emojiFont = Font.loadFont(fontUrl, 16);

        if (emojiFont != null) {
            textArea.setFont(emojiFont);
        } else {
            textArea.setStyle("-fx-font-family: 'Segoe UI Emoji', 'Apple Color Emoji', 'Noto Color Emoji'; -fx-font-size: 16px;");
        }

        VBox root = new VBox(textArea);
        Scene scene = new Scene(root, 400, 200);
        primaryStage.setScene(scene);
        primaryStage.setTitle("Emoji TextArea");
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
