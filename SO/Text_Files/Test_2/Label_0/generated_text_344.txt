from ultralytics import YOLO

def get_class_data():
    model = YOLO("yolov8n.pt")
    results = model.predict(source="0")
    class_data = []
    for result in results:
        for obj in result.pred[0]:
            class_data.append(result.names[int(obj[5])])
    return class_data

class_data = get_class_data()
