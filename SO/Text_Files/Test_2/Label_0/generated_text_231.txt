import org.springframework.core.io.FileSystemResource;
import org.springframework.http.ContentDisposition;
import org.springframework.http.HttpHeaders;
import org.springframework.http.MediaType;
import org.springframework.http.client.MultipartBodyBuilder;
import org.springframework.web.multipart.MultipartFile;
import org.springframework.web.multipart.support.StandardMultipartHttpServletRequest;

import java.io.File;
import java.nio.file.Files;
import java.util.Collections;

public MultipartFile createMultipartFile(File file, String mimeType) {
    try {
        MultipartBodyBuilder builder = new MultipartBodyBuilder();
        builder.part("file", new FileSystemResource(file))
               .header(HttpHeaders.CONTENT_DISPOSITION, ContentDisposition.formData().name("file").filename(file.getName()).build().toString())
               .contentType(MediaType.parseMediaType(mimeType));

        StandardMultipartHttpServletRequest request = new StandardMultipartHttpServletRequest(null);
        request.setMultipartFiles(Collections.singletonMap("file", builder.build().get("file")));

        return request.getFile("file");
    } catch (Exception e) {
        throw new RuntimeException("Cannot build MultipartFile", e);
    }
}
