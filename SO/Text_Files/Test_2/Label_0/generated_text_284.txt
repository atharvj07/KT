from dataclasses import dataclass
from typing import Callable, Dict, Type, TypeVar

T = TypeVar("T", bound=Type)

def mydecorator(cls: T) -> T:
    def save(self):
        print(f"Saving {self}")

    cls.save = save
    cls.my_fields_dict: Dict[str, str] = {field.name: field.type for field in cls.__dataclass_fields__.values()}  # type: ignore
    return cls

@mydecorator
@dataclass
class A:
    a: str = ""

# Example usage
a = A()
a.save()
print(a.my_fields_dict)
