import numpy as np
import pandas as pd
from collections import Counter

def balanceness(array: np.ndarray) -> float:
    classes = [(c, cnt) for c, cnt in Counter(array).items()]
    m = len(classes)
    n = len(array)

    H = -sum([(cnt / n) * np.log((cnt / n)) for _, cnt in classes])

    return H / np.log(m)

def weight(days_ago: int, epsilon: float = 0.005) -> float:
    return np.exp(-1 * days_ago * epsilon)

def find_most_recent_balanced_opponent(df: pd.DataFrame) -> str:
    df["weight"] = df["days ago"].apply(weight)

    grouped = df.groupby("opponent").agg(
        balanceness=("player", lambda x: balanceness(x.to_numpy())),
        weighted_n_matches=("weight", "sum"),
        recent_balanceness=("weight", lambda x: balanceness(x.to_numpy()))
    )

    grouped["score"] = grouped["balanceness"] * grouped["recent_balanceness"] * grouped["weighted_n_matches"]

    return grouped["score"].idxmax()

df = pd.DataFrame({
    "player": ["A", "A", "A", "A", "A", "A", "A", "B", "B", "B", "B", "B", "B", "B", "B", "B", "B", "B", "B", "B", "B"],
    "opponent": ["C", "C", "D", "F", "F", "F", "G", "C", "C", "D", "F", "F", "F", "G", "G", "G", "G", "G", "G", "G", "G"],
    "days ago": [1, 2, 10, 100, 101, 102, 1, 1, 2, 10, 100, 101, 102, 1, 2, 3, 4, 5, 6, 7, 8]
})

most_recent_balanced_opponent = find_most_recent_balanced_opponent(df)
print(most_recent_balanced_opponent)
