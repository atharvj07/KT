package app.controllers;

import javafx.fxml.FXML;
import javafx.fxml.FXMLLoader;
import javafx.scene.layout.VBox;
import java.io.IOException;
import app.components.Custom;

public class Menu {

    @FXML
    private VBox ccContainer;

    private Custom cc;

    @FXML
    public void initialize() {
        try {
            FXMLLoader loader = new FXMLLoader(getClass().getResource("/app/components/Custom.fxml"));
            cc = loader.load();
            ccContainer.getChildren().add(cc);
            cc.newText("Test");
        } catch (IOException e) {
            throw new RuntimeException(e);
        }
    }
}
