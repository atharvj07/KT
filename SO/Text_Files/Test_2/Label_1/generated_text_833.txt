def mirror(a):
    m, n = a.shape
    if m == n:
        return a.T.copy()
    if m > n:
        return mirror(a.T).T

    # Shear
    v = a.flatten()
    w = v[:-m].reshape((m, n-1))
    
    # Flip the parallelogram
    w[:, m-1:] = w[::-1, m-1:]

    # Flip the triangles
    t = np.vstack((w[:, :m-1].reshape((m-1, m)), v[-m:]))
    t = t.T
    w[:, :m-1] = t[:-1].reshape((m, m-1))

    # Write flipped parts back and unshear
    v[:-m] = w.ravel()
    v[-m:] = t[-1]
    return v.reshape((m, n))
