import java.util.*;

public class LongestMountainSequence {

    public static int solve(List<Integer> list) {
        int n = list.size();
        if (n < 2) return n;

        List<Integer> uniqueSorted = new ArrayList<>(new TreeSet<>(list));

        int maxLen = 0;
        for (int i = 1; i < uniqueSorted.size() - 1; i++) {
            for (int j = i + 1; j < uniqueSorted.size(); j++) {
                List<Integer> increasing = new ArrayList<>();
                List<Integer> decreasing = new ArrayList<>();
                List<Integer> increasingAgain = new ArrayList<>();

                for (int num : list) {
                    if (num <= uniqueSorted.get(i) && (increasing.isEmpty() || num > increasing.get(increasing.size() - 1))) {
                        increasing.add(num);
                    } else if (num <= uniqueSorted.get(j) && (decreasing.isEmpty() || num < decreasing.get(decreasing.size() - 1))) {
                        decreasing.add(num);
                    } else if (!decreasing.isEmpty() && num > decreasing.get(decreasing.size() - 1)) {
                        increasingAgain.add(num);
                    }
                }

                if (!increasing.isEmpty() && !decreasing.isEmpty() && !increasingAgain.isEmpty()) {
                    maxLen = Math.max(maxLen, increasing.size() + decreasing.size() + increasingAgain.size());
                }
            }
        }

        return maxLen;
    }

    public static void main(String[] args) {
        System.out.println(solve(List.of(1, 3, 5, 4, 2, 6, 8, 7, 9)));
        System.out.println(solve(List.of(5, 5, 2, 1, 3, 4, 5)));
        System.out.println(solve(List.of(1, 100)));
        System.out.println(solve(List.of(2, 1, 3, 3, 1, 2, 1, 2, 3)));
    }
}
