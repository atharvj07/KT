class Instanceof {
    static void doesNotWork(Object o) {
        if (o == null) {
            throw new Error();
        } else if (!(o instanceof String s)) {
            throw new Error();
        }   
        System.out.println(s);
    }

    static void doesWork(Object o) {
        if (o == null) {
            throw new Error();
        } else if (!(o instanceof String s)) {
            throw new Error();
        } else {
            System.out.println(s);
        }
    }

    static void doesWorkAlternative(Object o) {
        if (o == null) {
            throw new Error();
        }
        if (!(o instanceof String s)) {
            throw new Error();
        }
        System.out.println(s);
    }

    static void doesWorkSingleIf(Object o) {
        if (o == null || !(o instanceof String s)) {
            throw new Error();
        }
        System.out.println(s);
    }
}
