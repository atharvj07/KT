import com.fasterxml.jackson.core.JsonGenerator;
import com.fasterxml.jackson.databind.SerializerProvider;
import com.fasterxml.jackson.dataformat.xml.ser.ToXmlGenerator;
import com.fasterxml.jackson.databind.ser.std.StdSerializer;
import java.io.IOException;
import java.util.Iterator;

public class ItemSerializer extends StdSerializer<Item> {

    public ItemSerializer() {
        this(null);
    }

    public ItemSerializer(Class<Item> t) {
        super(t);
    }

    @Override
    public void serialize(Item item, JsonGenerator gen, SerializerProvider provider) throws IOException {
        ToXmlGenerator xmlGen = (ToXmlGenerator) gen;
        xmlGen.writeStartObject();
        
        xmlGen.writeStringField("SNO", item.getSNo());
        xmlGen.writeNumberField("Weight", item.getWeight());

        Iterator<String> aIter = item.getAList().iterator();
        Iterator<String> bIter = item.getBList().iterator();

        while (aIter.hasNext() || bIter.hasNext()) {
            if (aIter.hasNext()) xmlGen.writeStringField("A", aIter.next());
            if (bIter.hasNext()) xmlGen.writeStringField("B", bIter.next());
        }

        xmlGen.writeEndObject();
    }
}
