import java.awt.Graphics;
import java.awt.Image;
import java.awt.image.BufferedImage;
import javax.swing.JPanel;

public class ScaledBackgroundPanel extends JPanel {
    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        BufferedImage image = ApplicationImages.getImage();
        if (image != null) {
            float factorWidth = (float) getWidth() / 1280;
            float factorHeight = (float) getHeight() / 859;
            int width, height, x = 0, y = 0;

            if (factorWidth < factorHeight) {
                width = (int) (1280 * factorWidth);
                height = (int) (859 * factorWidth);
                y = (getHeight() - height) / 2;
            } else {
                width = (int) (1280 * factorHeight);
                height = (int) (859 * factorHeight);
                x = (getWidth() - width) / 2;
            }

            Image scaledImage = image.getScaledInstance(width, height, Image.SCALE_SMOOTH);
            g.drawImage(scaledImage, x, y, this);
        }
    }
}
