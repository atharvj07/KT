import java.lang.foreign.MemorySegment;
import java.lang.foreign.MemorySession;
import java.lang.foreign.ValueLayout;
import java.util.stream.LongStream;

public class LargeLongArray {
    private final MemorySegment segment;
    private final long size;

    public LargeLongArray(long size) {
        this.size = size;
        this.segment = MemorySegment.allocateNative(size * Long.BYTES, MemorySession.global());
    }

    public void set(long index, long value) {
        if (index < 0 || index >= size) throw new IndexOutOfBoundsException();
        segment.set(ValueLayout.JAVA_LONG, index * Long.BYTES, value);
    }

    public long get(long index) {
        if (index < 0 || index >= size) throw new IndexOutOfBoundsException();
        return segment.get(ValueLayout.JAVA_LONG, index * Long.BYTES);
    }

    public long sum() {
        return LongStream.range(0, size).map(this::get).sum();
    }

    public void close() {
        segment.close();
    }

    public static void main(String[] args) {
        LargeLongArray array = new LargeLongArray(1_000_000L);
        array.set(999_999, 42);
        System.out.println(array.get(999_999));
        System.out.println(array.sum());
        array.close();
    }
}
