import reactor.core.publisher.Flux;
import reactor.core.publisher.Mono;

import java.util.List;
import java.util.Map;

public class ReactiveExample {
    public static void main(String[] args) {
        List<A> list = List.of(new A("one"), new A("two"));
        
        Map<String, String> result = Flux.fromIterable(list)
                .flatMap(a -> functionReturningMonoOfC(a)
                        .map(C::url)
                        .map(url -> Map.entry(a.getName(), url))
                )
                .collectMap(Map.Entry::getKey, Map.Entry::getValue)
                .block();
    }
    
    static Mono<C> functionReturningMonoOfC(A a) {
        return Mono.just(new C("http://example.com/" + a.getName()));
    }

    static class A {
        private final String name;
        A(String name) { this.name = name; }
        String getName() { return name; }
    }

    static class C {
        private final String url;
        C(String url) { this.url = url; }
        String url() { return url; }
    }
}
