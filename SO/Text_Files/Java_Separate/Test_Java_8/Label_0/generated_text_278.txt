import java.util.concurrent.Callable;

class InferenceExample {
    static <T, E extends Exception> T execute(Callable<T> callable) throws E {
        try {
            return callable.call();
        } catch (Exception e) {
            throw (E) e; // unchecked cast, can be unsafe
        }
    }

    public static void main(String[] args) {
        try {
            execute(() -> {
                if (true) throw new IllegalArgumentException("Unchecked"); 
                return "OK";
            }); // No compilation error

            execute(() -> {
                if (true) throw new Exception("Checked"); 
                return "OK";
            }); // Compilation error: unhandled exception
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
