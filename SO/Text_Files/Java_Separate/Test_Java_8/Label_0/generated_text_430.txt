import java.util.*;

public class OverlappingSets {
    public static void main(String[] args) {
        Map<String, Set<Long>> myMap = new HashMap<>();
        myMap.put("a", new HashSet<>(Arrays.asList(1L, 2L, 3L)));
        myMap.put("b", new HashSet<>(Arrays.asList(4L, 5L)));
        myMap.put("c", new HashSet<>(Collections.singletonList(1L)));
        myMap.put("d", new HashSet<>(Arrays.asList(2L, 3L)));
        myMap.put("e", new HashSet<>(Collections.singletonList(5L)));
        myMap.put("f", new HashSet<>(Collections.singletonList(6L)));

        List<String> overlappingKeys = new ArrayList<>();

        List<Map.Entry<String, Set<Long>>> entries = new ArrayList<>(myMap.entrySet());

        for (int i = 0; i < entries.size(); i++) {
            for (int j = i + 1; j < entries.size(); j++) {
                if (!Collections.disjoint(entries.get(i).getValue(), entries.get(j).getValue())) {
                    overlappingKeys.add(entries.get(i).getKey() + " and " + entries.get(j).getKey());
                }
            }
        }

        overlappingKeys.forEach(System.out::println);
    }
}
