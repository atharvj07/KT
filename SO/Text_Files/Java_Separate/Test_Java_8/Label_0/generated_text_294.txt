import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;
import javafx.stage.Popup;

public class MenuUpwardsApp extends Application {
    @Override
    public void start(Stage stage) {
        MenuBar menuBar = new MenuBar();
        menuBar.getMenus().add(createMenu("Options"));

        VBox root = new VBox();
        root.getChildren().add(menuBar);
        root.setStyle("-fx-background-color: lightgray;");
        
        Scene scene = new Scene(root, 400, 300);
        stage.setScene(scene);
        stage.show();
        
        menuBar.setTranslateY(250);  
    }

    private Menu createMenu(String title) {
        Menu menu = new Menu(title);
        menu.getItems().addAll(new MenuItem("Item 1"), new MenuItem("Item 2"));

        menu.setOnShowing(e -> {
            Popup popup = new Popup();
            popup.getContent().add(new Label("Temporary Workaround"));
            popup.setAutoHide(true);
            popup.show(menu.getParentPopup(), 200, 100);
        });

        return menu;
    }

    public static void main(String[] args) {
        launch(args);
    }
}
