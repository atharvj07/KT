import java.util.*;

class Edge {
    int to, weight;
    Edge(int to, int weight) {
        this.to = to;
        this.weight = weight;
    }
}

class Solver {
    public static int solve(List<List<Edge>> T, long L) {
        int[] minEdges = {Integer.MAX_VALUE};
        for (int i = 0; i < T.size(); i++) {
            boolean[] visited = new boolean[T.size()];
            dfs(T, L, i, -1, 0, 0, minEdges, visited);
        }
        return minEdges[0] == Integer.MAX_VALUE ? -1 : minEdges[0];
    }

    private static void dfs(List<List<Edge>> T, long L, int node, int parent, long sum, int edges, int[] minEdges, boolean[] visited) {
        if (sum == L) {
            minEdges[0] = Math.min(minEdges[0], edges);
            return;
        }
        if (sum > L) return;
        
        visited[node] = true;
        for (Edge edge : T.get(node)) {
            if (!visited[edge.to] && edge.to != parent) {
                dfs(T, L, edge.to, node, sum + edge.weight, edges + 1, minEdges, visited);
            }
        }
        visited[node] = false;
    }
}

public class Main {
    static void putEdge(List<List<Edge>> T, int v1, int v2, int weight) {
        T.get(v1).add(new Edge(v2, weight));
        T.get(v2).add(new Edge(v1, weight));
    }

    public static void main(String[] args) {
        int N = 8;
        List<List<Edge>> T = new ArrayList<>();
        for (int i = 0; i < N; i++) T.add(new ArrayList<>());

        putEdge(T, 0, 1, 2);
        putEdge(T, 1, 2, 1);
        putEdge(T, 1, 3, 2);
        putEdge(T, 2, 6, 1);
        putEdge(T, 6, 7, 1);
        putEdge(T, 3, 4, 1);
        putEdge(T, 3, 5, 4);

        System.out.println(Solver.solve(T, 5L)); 
    }
}
