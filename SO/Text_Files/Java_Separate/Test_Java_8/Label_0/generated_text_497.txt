(componentModel = "spring")
public interface AgentInfoMapper {

    default AgentInfo wealthProdAccountInfoDTOToAgentInfo(WealthProdAccountInfoDTO wealthProdAccountInfoDTO) {
        if (wealthProdAccountInfoDTO == null) {
            return null;
        }

        String agentName = wealthProdAccountInfoDTO.getAgentName();
        String agentIdentityType = wealthProdAccountInfoDTO.getAgentIdentityType();
        String agentIdentityNo = wealthProdAccountInfoDTO.getAgentIdentityNo();
        String agentIdentityExpireAt = wealthProdAccountInfoDTO.getAgentIdentityExpireAt();

        if (agentName == null && agentIdentityType == null && agentIdentityNo == null && agentIdentityExpireAt == null) {
            return null;
        }

        return new AgentInfo(agentName, agentIdentityType, agentIdentityNo, agentIdentityExpireAt);
    }
}
