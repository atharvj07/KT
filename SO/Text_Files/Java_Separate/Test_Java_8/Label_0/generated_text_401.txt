package org.springframework.cloud.sleuth.instrument.async;

import java.util.concurrent.ScheduledThreadPoolExecutor;
import java.util.concurrent.RunnableScheduledFuture;
import java.lang.reflect.Method;
import org.springframework.util.ReflectionUtils;

public class LazyTraceScheduledThreadPoolExecutor extends ScheduledThreadPoolExecutor {

    public LazyTraceScheduledThreadPoolExecutor(int corePoolSize) {
        super(corePoolSize);
    }

    @Override
    protected RunnableScheduledFuture decorateTask(Runnable runnable, RunnableScheduledFuture task) {
        try {
            Method decorateMethod = ScheduledThreadPoolExecutor.class.getDeclaredMethod("decorateTask", Runnable.class, RunnableScheduledFuture.class);
            decorateMethod.setAccessible(true); // Making the protected method accessible
            return (RunnableScheduledFuture) decorateMethod.invoke(this, runnable, task);
        } catch (Exception e) {
            throw new RuntimeException("Error accessing decorateTask", e);
        }
    }
}
