import android.content.ContentResolver;
import android.content.ContentValues;
import android.content.Context;
import android.net.Uri;
import android.os.Build;
import android.os.Environment;
import android.util.Log;
import java.io.File;
import java.io.FileOutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.OutputStream;
import java.util.Objects;

public void saveFileToStorage(String fileName, InputStream inputStream) {
    OutputStream outputStream = null;
    try {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            ContentResolver contentResolver = requireContext().getContentResolver();
            ContentValues contentValues = new ContentValues();
            contentValues.put("_display_name", fileName);
            contentValues.put("relative_path", Environment.DIRECTORY_DOCUMENTS);
            Uri fileUri = contentResolver.insert(Uri.parse("content://media/external_primary/document"), contentValues);
            if (fileUri != null) {
                outputStream = contentResolver.openOutputStream(fileUri);
                byte[] buffer = new byte[8192];
                int length;
                while ((length = inputStream.read(buffer)) != -1) {
                    outputStream.write(buffer, 0, length);
                }
            }
        }
    } catch (IOException e) {
        Log.e("FileSave", "Error saving file: " + fileName, e);
    } finally {
        try {
            if (outputStream != null) outputStream.close();
            inputStream.close();
        } catch (IOException ignored) {}
    }
}

public void copyDatabaseFiles() {
    try {
        String[] files = requireContext().getAssets().list("database/test");
        if (files != null) {
            for (String filename : files) {
                InputStream in = requireContext().getAssets().open("database/test/" + filename);
                saveFileToStorage(filename, in);
            }
        } else {
            Log.e("FileCopy", "No files found");
        }
    } catch (IOException e) {
        Log.e("FileCopy", "Error copying database files", e);
    }
}
