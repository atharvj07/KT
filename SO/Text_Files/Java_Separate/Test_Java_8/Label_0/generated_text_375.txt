import org.hibernate.annotations.Type;
import org.locationtech.jts.geom.Polygon;
import org.locationtech.jts.geom.GeometryFactory;
import org.locationtech.jts.geom.Coordinate;
import javax.persistence.*;

@Entity
@Table(name = "range")
public class Range {
    
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(name = "shape", columnDefinition = "MDSYS.SDO_GEOMETRY")
    @Type(type = "org.hibernate.spatial.GeometryType")
    private Polygon shape;

    public void setShape(Polygon shape) {
        this.shape = shape;
    }

    public Polygon getShape() {
        return shape;
    }
}

@Service
public class RangeService {
    
    @Autowired
    private RangeRepository rangeRepository;

    public Range saveRange(List<Coordinate> coordinates) {
        GeometryFactory geometryFactory = new GeometryFactory();
        if (coordinates.size() > 2) {
            coordinates.add(coordinates.get(0));
        }
        Polygon polygon = geometryFactory.createPolygon(coordinates.toArray(new Coordinate[coordinates.size()]));
        Range range = new Range();
        range.setShape(polygon);
        return rangeRepository.save(range);
    }
}

@Repository
public interface RangeRepository extends JpaRepository<Range, Long> {
}
