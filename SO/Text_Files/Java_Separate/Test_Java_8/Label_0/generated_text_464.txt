import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.ApplicationContext;
import org.springframework.stereotype.Component;
import org.springframework.stereotype.Service;

@Component
public class UtilFactory {

    @Autowired
    private ApplicationContext context;

    public Util getUtil(String type) {
        if ("two".equalsIgnoreCase(type)) {
            return context.getBean(UtilTwo.class);
        }
        return context.getBean(UtilOne.class);
    }
}

@Service
public class DemoService {

    @Autowired
    private UtilFactory utilFactory;

    private volatile Util util;

    public void updateUtil(String type) {
        this.util = utilFactory.getUtil(type);
    }

    public void useUtil() {
        if (util == null) {
            util = utilFactory.getUtil("one");
        }
        util.getClient();
    }
}
