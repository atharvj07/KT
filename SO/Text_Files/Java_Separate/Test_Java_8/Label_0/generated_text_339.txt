package application;

import javafx.animation.KeyFrame;
import javafx.animation.Timeline;
import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.StackPane;
import javafx.scene.paint.Color;
import javafx.scene.text.Font;
import javafx.scene.text.Text;
import javafx.stage.Stage;
import javafx.util.Duration;

public class Main extends Application {
    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        Scene scene = new Scene(root, 400, 400);
        StackPane stack = new StackPane();
        
        Text text = new Text("Hello there StackOverflow, how are you? (:");
        
        Color color1 = Color.RED;
        Color color2 = Color.BLUE;
        
        double[] r = {color1.getRed()};
        double[] g = {color1.getGreen()};
        double[] b = {color1.getBlue()};
        
        text.setFont(Font.font(16));
        text.setFill(color1);
        
        Timeline timeline = new Timeline(
            new KeyFrame(Duration.millis(20), event -> {
                r[0] += (color2.getRed() - color1.getRed()) / 50;
                g[0] += (color2.getGreen() - color1.getGreen()) / 50;
                b[0] += (color2.getBlue() - color1.getBlue()) / 50;
                
                if (r[0] > color2.getRed()) r[0] = color2.getRed();
                if (g[0] > color2.getGreen()) g[0] = color2.getGreen();
                if (b[0] > color2.getBlue()) b[0] = color2.getBlue();
                
                text.setFill(new Color(r[0], g[0], b[0], 1.0));
            })
        );
        
        timeline.setCycleCount(50);
        timeline.play();
        
        stack.getChildren().add(text);
        root.getChildren().add(stack);
        
        primaryStage.setScene(scene);
        primaryStage.show();
    }
    
    public static void main(String[] args) {
        launch(args);
    }
}
