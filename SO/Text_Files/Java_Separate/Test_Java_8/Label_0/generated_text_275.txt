import java.time.*;
import java.time.format.DateTimeFormatter;
import java.time.temporal.ChronoUnit;

public class IsoWeek {
    private Year year;
    private int week;

    private static final DateTimeFormatter formatter = DateTimeFormatter.ofPattern("YYYY-'W'ww");

    public IsoWeek(String weekString) {
        LocalDate date = LocalDate.parse(weekString + "-1", formatter);
        this.year = Year.of(date.getYear());
        this.week = date.get(ChronoField.ALIGNED_WEEK_OF_YEAR);
    }

    public boolean isBefore(IsoWeek other) {
        return this.year.getValue() < other.year.getValue() ||
               (this.year.getValue() == other.year.getValue() && this.week < other.week);
    }

    public long weeksBetween(IsoWeek other) {
        LocalDate start = LocalDate.ofYearDay(this.year.getValue(), 1)
                                   .with(ChronoField.ALIGNED_WEEK_OF_YEAR, this.week);
        LocalDate end = LocalDate.ofYearDay(other.year.getValue(), 1)
                                 .with(ChronoField.ALIGNED_WEEK_OF_YEAR, other.week);
        return ChronoUnit.WEEKS.between(start, end);
    }

    @Override
    public String toString() {
        return year + "-W" + (week < 10 ? "0" + week : week);
    }

    public static void main(String[] args) {
        IsoWeek weekA = new IsoWeek("2012-W48");
        IsoWeek weekB = new IsoWeek("2013-W03");

        System.out.println(weekA.isBefore(weekB)); 
        System.out.println(weekA.weeksBetween(weekB)); 
    }
}
