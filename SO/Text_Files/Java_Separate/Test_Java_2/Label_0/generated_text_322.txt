import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.util.Map;
import java.util.stream.Collectors;

public class MapToJsonConverter {

    public static String convertMapToJsonWithCamelCase(Map<String, Object> fields) throws JsonProcessingException {
        ObjectMapper mapper = new ObjectMapper();
        Map<String, Object> transformedMap = fields.entrySet().stream()
                .collect(Collectors.toMap(
                        entry -> toCamelCase(entry.getKey()),
                        Map.Entry::getValue
                ));
        return mapper.writeValueAsString(transformedMap);
    }

    private static String toCamelCase(String underscore) {
        String[] parts = underscore.split("_");
        if (parts.length == 0) return underscore;
        StringBuilder camelCase = new StringBuilder(parts[0]);
        for (int i = 1; i < parts.length; i++) {
            camelCase.append(Character.toUpperCase(parts[i].charAt(0))).append(parts[i].substring(1));
        }
        return camelCase.toString();
    }
}
