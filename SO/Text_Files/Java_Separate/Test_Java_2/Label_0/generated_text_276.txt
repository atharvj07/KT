public class DemoSecurityProblem {
    static DemoSecurityProblem survivor;

    public DemoSecurityProblem() throws Exception {
        if (true) { // Simulating a failure
            throw new Exception("Constructor failed!");
        }
    }

    @Override
    protected void finalize() throws Throwable {
        super.finalize();
        survivor = this; // Resurrect the object
        System.out.println("Finalizer invoked, object resurrected!");
    }

    public void doSomething() {
        System.out.println("Malicious operation executed!");
    }
}

public class MaliciousSubClass extends DemoSecurityProblem {
    public MaliciousSubClass() throws Exception {
        super();
    }

    @Override
    protected void finalize() throws Throwable {
        super.finalize();
        System.out.println("Malicious finalizer invoked!");
    }

    public static void main(String[] args) {
        try {
            new MaliciousSubClass();
        } catch (Exception e) {
            System.out.println("Exception caught, object should be dead.");
        }

        System.gc(); // Encourage finalization

        try {
            Thread.sleep(2000); // Give time for finalizer to run
        } catch (InterruptedException e) {
            e.printStackTrace();
        }

        if (DemoSecurityProblem.survivor != null) {
            DemoSecurityProblem.survivor.doSomething();
        } else {
            System.out.println("Object was correctly garbage collected.");
        }
    }
}
