import java.lang.reflect.Constructor;
import java.lang.reflect.Field;
import java.lang.reflect.InvocationTargetException;
import java.lang.reflect.Modifier;

public class ReflectionUtil {
    public static Object createParentInstance() {
        try {
            Constructor<Parent> constructor = Parent.class.getDeclaredConstructor();
            constructor.setAccessible(true);
            removeFinalModifier(constructor);
            return constructor.newInstance();
        } catch (NoSuchMethodException | InstantiationException | IllegalAccessException | InvocationTargetException e) {
            throw new RuntimeException("Failed to create Parent instance", e);
        }
    }

    private static void removeFinalModifier(Constructor<?> constructor) {
        try {
            Field modifiersField = Constructor.class.getDeclaredField("modifiers");
            modifiersField.setAccessible(true);
            modifiersField.setInt(constructor, constructor.getModifiers() & ~Modifier.PRIVATE);
        } catch (NoSuchFieldException | IllegalAccessException ignored) {
        }
    }
}
