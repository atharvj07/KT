import com.sun.source.tree.*;
import com.sun.source.util.Trees;
import javax.lang.model.type.TypeMirror;
import javax.lang.model.util.Types;
import javax.lang.model.element.Element;
import javax.lang.model.element.TypeElement;
import javax.lang.model.element.VariableElement;
import javax.annotation.processing.ProcessingEnvironment;

public class SwitchTypeResolver {

    private final Trees trees;
    private final Types types;

    public SwitchTypeResolver(ProcessingEnvironment processingEnv) {
        this.trees = Trees.instance(processingEnv);
        this.types = processingEnv.getTypeUtils();
    }

    public TypeMirror getSwitchExpressionType(MethodTree methodTree) {
        BlockTree blockTree = methodTree.getBody();

        for (StatementTree statementTree : blockTree.getStatements()) {
            if (statementTree.getKind() == Tree.Kind.SWITCH) {
                SwitchTree switchTree = (SwitchTree) statementTree;
                ExpressionTree switchTreeExpression = switchTree.getExpression();

                Element element = trees.getElement(trees.getPath(trees.getCompilationUnit(methodTree), switchTreeExpression));
                if (element instanceof VariableElement || element instanceof TypeElement) {
                    return element.asType();
                }
            }
        }
        return null;
    }
}
