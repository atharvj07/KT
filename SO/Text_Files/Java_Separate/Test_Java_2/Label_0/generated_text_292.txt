import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.Label;
import javafx.scene.control.TextField;
import javafx.scene.control.TitledPane;
import javafx.scene.layout.ColumnConstraints;
import javafx.scene.layout.GridPane;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class MyGridPanes extends Application {

    @Override
    public void start(Stage stage) {
        GridPane gridPane1 = createGridPane();
        GridPane gridPane2 = createGridPane();

        Label label1 = new Label("One two three");
        TextField textField1 = new TextField();
        gridPane1.add(label1, 0, 0);
        gridPane1.add(textField1, 1, 0);

        Label label2 = new Label("Four five six seven");
        TextField textField2 = new TextField();
        gridPane2.add(label2, 0, 0);
        gridPane2.add(textField2, 1, 0);

        TitledPane titledPane1 = new TitledPane("Top", gridPane1);
        titledPane1.setCollapsible(false);

        TitledPane titledPane2 = new TitledPane("Bottom", gridPane2);
        titledPane2.setCollapsible(false);

        double maxLabelWidth = Math.max(label1.prefWidth(-1), label2.prefWidth(-1));

        gridPane1.getColumnConstraints().get(0).setMinWidth(maxLabelWidth);
        gridPane2.getColumnConstraints().get(0).setMinWidth(maxLabelWidth);

        Scene scene = new Scene(new VBox(titledPane1, titledPane2), 400, 200);
        stage.setScene(scene);
        stage.show();
    }

    private GridPane createGridPane() {
        GridPane gridPane = new GridPane();
        gridPane.setHgap(20);
        ColumnConstraints labelColumn = new ColumnConstraints();
        ColumnConstraints controlColumn = new ColumnConstraints();
        controlColumn.setPercentWidth(100);
        gridPane.getColumnConstraints().addAll(labelColumn, controlColumn);
        return gridPane;
    }

    public static void main(String[] args) {
        launch(args);
    }
}
