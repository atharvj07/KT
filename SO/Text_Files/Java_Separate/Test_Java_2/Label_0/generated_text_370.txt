import java.awt.*;  
import java.awt.event.*;
import javax.swing.*;  

class Surface extends JPanel implements ActionListener, KeyListener {
    private final int DELAY = 8;
    private Timer timer;
    private Image image;
    private int x, y;
    private final int MOVE_AMOUNT = 5;
    private boolean upPressed, downPressed, leftPressed, rightPressed;

    public Surface() {
        setDoubleBuffered(true);
        initTimer();
        loadImage();
        setFocusable(true);
        requestFocusInWindow();
        addKeyListener(this);
        x = 200;
        y = 200;
    }

    private void initTimer() {
        timer = new Timer(DELAY, this);
        timer.start();
    }

    private void loadImage() {
        image = new ImageIcon("Basic_red_dot.png").getImage().getScaledInstance(75, 75, Image.SCALE_SMOOTH);
    }

    public Rectangle getRedDotBounds() {
        return new Rectangle(x, y, 75, 75);
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        g.drawImage(image, x, y, this);
    }

    public void actionPerformed(ActionEvent e) {
        updatePosition();
        repaint();
    }

    private void updatePosition() {
        if (leftPressed) x = Math.max(x - MOVE_AMOUNT, 0);
        if (rightPressed) x = Math.min(x + MOVE_AMOUNT, getWidth() - 75);
        if (upPressed) y = Math.max(y - MOVE_AMOUNT, 0);
        if (downPressed) y = Math.min(y + MOVE_AMOUNT, getHeight() - 75);
    }

    @Override
    public void keyPressed(KeyEvent e) {
        switch (e.getKeyCode()) {
            case KeyEvent.VK_LEFT -> leftPressed = true;
            case KeyEvent.VK_RIGHT -> rightPressed = true;
            case KeyEvent.VK_UP -> upPressed = true;
            case KeyEvent.VK_DOWN -> downPressed = true;
        }
    }

    @Override
    public void keyReleased(KeyEvent e) {
        switch (e.getKeyCode()) {
            case KeyEvent.VK_LEFT -> leftPressed = false;
            case KeyEvent.VK_RIGHT -> rightPressed = false;
            case KeyEvent.VK_UP -> upPressed = false;
            case KeyEvent.VK_DOWN -> downPressed = false;
        }
    }

    @Override
    public void keyTyped(KeyEvent e) {}
}

class BlueDot extends JPanel implements ActionListener {
    private int x, y;
    private Image image;
    private final int FALL_SPEED = 1;
    private Timer timer;

    public BlueDot() {
        setOpaque(false);
        x = (int) (Math.random() * 500);
        y = 0;
        loadImage();
        timer = new Timer(10, this);
        timer.start();
    }

    private void loadImage() {
        image = new ImageIcon("Basic_blue_dot.png").getImage().getScaledInstance(10, 10, Image.SCALE_SMOOTH);
    }

    public Rectangle getBlueDotBounds() {
        return new Rectangle(x, y, 10, 10);
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        g.drawImage(image, x, y, this);
    }

    public void actionPerformed(ActionEvent e) {
        y += FALL_SPEED;
        if (y > getParent().getHeight()) {
            x = (int) (Math.random() * 500);
            y = 0;
        }
        repaint();
    }
}

public class MyProgram {
    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            JFrame frame = new JFrame("Image Display");
            frame.setSize(500, 500);
            frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
            frame.setLayout(null);

            Surface surface = new Surface();
            surface.setBounds(0, 0, 500, 500);
            frame.add(surface);

            BlueDot blueDot = new BlueDot();
            blueDot.setBounds(0, 0, 500, 500);
            frame.add(blueDot);

            frame.setLocationRelativeTo(null);
            frame.setVisible(true);
            surface.requestFocusInWindow();

            Timer collisionTimer = new Timer(10, e -> {
                if (surface.getRedDotBounds().intersects(blueDot.getBlueDotBounds())) {
                    System.out.println("Collision!!!!");
                }
            });
            collisionTimer.start();
        });
    }
}
