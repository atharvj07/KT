import java.util.concurrent.Callable;

public class InferenceThrowsExample {
    public static <T extends Exception> void throwAsUnchecked(Exception e) throws T {
        throw (T) e;
    }

    public static <T extends Exception> void execute(Callable<Void> callable) throws T {
        try {
            callable.call();
        } catch (Exception e) {
            throwAsUnchecked(e);
        }
    }

    public static void main(String[] args) {
        try {
            execute(() -> {
                throw new IllegalArgumentException("Unchecked Exception");
            });
        } catch (Exception e) {
            System.out.println("Caught: " + e);
        }

        try {
            execute(() -> {
                throw new Exception("Checked Exception");
            });
        } catch (Exception e) {
            System.out.println("Caught: " + e);
        }
    }
}
