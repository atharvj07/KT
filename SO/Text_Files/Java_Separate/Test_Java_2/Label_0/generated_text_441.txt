import java.lang.module.ModuleFinder;
import java.lang.module.ModuleReference;
import java.util.HashSet;
import java.util.Set;

public class CoreNonCoreModules {
    public static void main(String[] args) {
        Set<String> bootModules = getBootModules();
        Set<String> platformModules = getPlatformModules();

        System.out.println("Core (Boot) Modules:");
        bootModules.forEach(System.out::println);

        System.out.println("\nNon-Core (Platform) Modules:");
        platformModules.forEach(System.out::println);
    }

    private static Set<String> getBootModules() {
        Set<String> bootModules = new HashSet<>();
        for (Module module : ModuleLayer.boot().modules()) {
            if (isBootClassLoader(module)) {
                bootModules.add(module.getName());
            }
        }
        return bootModules;
    }

    private static Set<String> getPlatformModules() {
        Set<String> platformModules = new HashSet<>();
        for (ModuleReference ref : ModuleFinder.ofSystem().findAll()) {
            String moduleName = ref.descriptor().name();
            if (!getBootModules().contains(moduleName)) {
                platformModules.add(moduleName);
            }
        }
        return platformModules;
    }

    private static boolean isBootClassLoader(Module module) {
        try {
            Class<?> moduleInfoClass = Class.forName(module.getName() + ".module-info");
            return moduleInfoClass.getClassLoader() == null;
        } catch (ClassNotFoundException e) {
            return false;
        }
    }
}
