import java.util.ArrayList;
import java.util.List;

public class StringGenerator {
    public static void main(String[] args) {
        String[][] input = {
            {"A", "B", "C"},
            {"1", "2", "3"},
            {"X", "Y", "Z"}
        };

        List<String> result = generateStrings(input);
        result.forEach(System.out::println);
    }

    public static List<String> generateStrings(String[][] inputs) {
        List<String> result = new ArrayList<>();
        generateRecursive(inputs, 0, new StringBuilder(), result);
        return result;
    }

    private static void generateRecursive(String[][] inputs, int index, StringBuilder current, List<String> result) {
        if (index == inputs.length) {
            result.add(current.toString());
            return;
        }
        for (String s : inputs[index]) {
            int len = current.length();
            current.append(s);
            generateRecursive(inputs, index + 1, current, result);
            current.setLength(len);
        }
    }
}
