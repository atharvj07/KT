import java.util.*;

public class MapOverlapChecker {
    public static List<Set<String>> findOverlappingEntries(Map<String, Set<Long>> myMap) {
        List<Set<String>> result = new ArrayList<>();
        List<String> keys = new ArrayList<>(myMap.keySet());
        int size = keys.size();

        for (int i = 0; i < size; i++) {
            for (int j = i + 1; j < size; j++) {
                String key1 = keys.get(i);
                String key2 = keys.get(j);
                Set<Long> set1 = myMap.get(key1);
                Set<Long> set2 = myMap.get(key2);

                if (set1.containsAll(set2) || set2.containsAll(set1)) {
                    result.add(new HashSet<>(Arrays.asList(key1, key2)));
                }
            }
        }
        return result;
    }

    public static void main(String[] args) {
        Map<String, Set<Long>> myMap = new HashMap<>();
        myMap.put("a", new HashSet<>(Arrays.asList(1L, 2L, 3L)));
        myMap.put("b", new HashSet<>(Arrays.asList(4L, 5L)));
        myMap.put("c", new HashSet<>(Collections.singletonList(1L)));
        myMap.put("d", new HashSet<>(Arrays.asList(2L, 3L)));
        myMap.put("e", new HashSet<>(Collections.singletonList(5L)));
        myMap.put("f", new HashSet<>(Collections.singletonList(6L)));

        List<Set<String>> overlappingEntries = findOverlappingEntries(myMap);
        System.out.println(overlappingEntries);
    }
}
