import javafx.application.Application;
import javafx.scene.Group;
import javafx.scene.PerspectiveCamera;
import javafx.scene.Scene;
import javafx.scene.image.Image;
import javafx.scene.paint.PhongMaterial;
import javafx.scene.shape.MeshView;
import javafx.scene.shape.Sphere;
import javafx.scene.shape.TriangleMesh;
import javafx.scene.transform.Rotate;
import javafx.stage.Stage;

public class Earth3D extends Application {

    private static final double EARTH_RADIUS = 150;

    private MeshView createTexturedSphere(double radius, int divisions) {
        TriangleMesh mesh = new TriangleMesh();

        for (int y = 0; y <= divisions; y++) {
            double v = (double) y / divisions;
            double latitude = Math.PI * (v - 0.5);
            double sinLat = Math.sin(latitude);
            double cosLat = Math.cos(latitude);

            for (int x = 0; x <= divisions; x++) {
                double u = (double) x / divisions;
                double longitude = 2 * Math.PI * u;
                float xPos = (float) (radius * cosLat * Math.cos(longitude));
                float yPos = (float) (radius * sinLat);
                float zPos = (float) (radius * cosLat * Math.sin(longitude));

                mesh.getPoints().addAll(xPos, yPos, zPos);
                mesh.getTexCoords().addAll((float) u, (float) v);
            }
        }

        for (int y = 0; y < divisions; y++) {
            for (int x = 0; x < divisions; x++) {
                int p0 = y * (divisions + 1) + x;
                int p1 = p0 + 1;
                int p2 = p0 + divisions + 1;
                int p3 = p2 + 1;

                mesh.getFaces().addAll(p0, p0, p2, p2, p1, p1);
                mesh.getFaces().addAll(p1, p1, p2, p2, p3, p3);
            }
        }

        MeshView meshView = new MeshView(mesh);
        PhongMaterial material = new PhongMaterial();
        material.setDiffuseMap(new Image(getClass().getResourceAsStream("/images/earth2.jpg")));
        meshView.setMaterial(material);
        return meshView;
    }

    @Override
    public void start(Stage primaryStage) {
        MeshView earthSphere = createTexturedSphere(EARTH_RADIUS, 64);
        earthSphere.setRotationAxis(Rotate.Y_AXIS);
        earthSphere.setRotate(90);

        Group root = new Group(earthSphere);
        Scene scene = new Scene(root, 800, 600, true);
        scene.setCamera(new PerspectiveCamera());

        primaryStage.setTitle("Textured Earth Sphere");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
