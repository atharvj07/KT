public class StrictFpExample {
    public static void main(String[] args) {
        nonStrictFpCalculation();
        strictFpCalculation();
    }

    static void nonStrictFpCalculation() {
        double a = Double.MAX_VALUE;
        double b = a * 2;
        double c = b / 4;
        System.out.println("Non-strictfp result: " + c);
    }

    static strictfp void strictFpCalculation() {
        double a = Double.MAX_VALUE;
        double b = a * 2;
        double c = b / 4;
        System.out.println("Strictfp result: " + c);
    }
}
