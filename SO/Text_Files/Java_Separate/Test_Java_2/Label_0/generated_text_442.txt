import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.time.format.DateTimeFormatterBuilder;
import java.time.format.TextStyle;
import java.time.temporal.ChronoField;
import java.util.HashMap;
import java.util.Locale;
import java.util.Map;

public class Main {
    public static void main(String[] args) {
        System.out.println("Hello world! " + parseSpanishDate("24ENE1982"));
    }

    private static LocalDate parseSpanishDate(String dateString) {
        Map<String, String> monthAbbreviations = new HashMap<>();
        for (int i = 1; i <= 12; i++) {
            String month = LocalDate.of(2000, i, 1)
                    .getMonth()
                    .getDisplayName(TextStyle.SHORT, new Locale("es", "ES"))
                    .toUpperCase();
            monthAbbreviations.put(month, String.format("%02d", i));
        }

        for (Map.Entry<String, String> entry : monthAbbreviations.entrySet()) {
            if (dateString.contains(entry.getKey())) {
                dateString = dateString.replace(entry.getKey(), entry.getValue());
                break;
            }
        }

        DateTimeFormatter formatter = new DateTimeFormatterBuilder()
                .appendPattern("ddMMyyyy")
                .toFormatter(Locale.ROOT);

        return LocalDate.parse(dateString, formatter);
    }
}
