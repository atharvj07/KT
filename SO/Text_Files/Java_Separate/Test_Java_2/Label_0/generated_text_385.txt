import com.fasterxml.jackson.annotation.JsonProperty;
import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;

public class Response {
    @JsonProperty("count")
    private int totalCount;

    @JsonProperty("respCode")
    private int responseCode;

    @JsonProperty("payLoad")
    private JsonNode transactionsList;

    public String[] getTransactionsList() throws JsonProcessingException {
        ObjectMapper mapper = new ObjectMapper();
        return mapper.writeValueAsString(transactionsList).split("\n");
    }

    public void setTransactionsList(JsonNode transactionsList) {
        this.transactionsList = transactionsList;
    }
    
    public static void transactionsReceived(String jsonString) throws JsonProcessingException {
        ObjectMapper objectMapper = new ObjectMapper();
        Response responseRcvd = objectMapper.readValue(jsonString, Response.class);
    }
}
