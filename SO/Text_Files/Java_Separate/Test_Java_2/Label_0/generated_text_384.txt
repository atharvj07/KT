public static String getMacChromedriverZipName(boolean chromeForTesting) {
    String osName = System.getProperty("os.name", "").toLowerCase();
    String arch = System.getProperty("os.arch", "").toLowerCase();

    if (osName.contains("darwin") || osName.contains("mac")) {
        if (arch.contains("aarch64") || arch.contains("arm64")) {
            return chromeForTesting ? "chromedriver-mac-arm64.zip" : "chromedriver_mac64_m1.zip";
        } else {
            return chromeForTesting ? "chromedriver-mac-x64.zip" : "chromedriver_mac64.zip";
        }
    }
    return null;
}
