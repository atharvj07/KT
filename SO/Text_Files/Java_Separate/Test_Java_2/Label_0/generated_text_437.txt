import java.util.function.Function;
import java.util.stream.DoubleStream;

public enum AggregationType {
    MIN(DoubleStream::min),
    MAX(DoubleStream::max),
    AVERAGE(DoubleStream::average);

    private final Function<DoubleStream, java.util.OptionalDouble> operation;

    AggregationType(Function<DoubleStream, java.util.OptionalDouble> operation) {
        this.operation = operation;
    }

    public java.util.OptionalDouble apply(DoubleStream stream) {
        return operation.apply(stream);
    }
}

public class AggregationService {
    public Float someFunction(AggregationType e) {
        return (float) e.apply(provides.stream().mapToDouble(this::someFunc)).orElse(-1);
    }
}
