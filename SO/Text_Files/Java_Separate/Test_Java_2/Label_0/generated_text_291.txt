import javafx.application.Application;
import javafx.geometry.Bounds;
import javafx.scene.Scene;
import javafx.scene.input.MouseEvent;
import javafx.scene.layout.Pane;
import javafx.scene.paint.Color;
import javafx.scene.shape.Rectangle;
import javafx.scene.control.Label;
import javafx.stage.Stage;
import javafx.scene.text.TextAlignment;

public class MyApp extends Application {
    @Override
    public void start(Stage stage) {
        Pane root = new Pane();
        root.setStyle("-fx-background-color: black;");
        Label testLabel = createTestLabel();
        root.getChildren().add(testLabel);
        Scene scene = new Scene(root, 600, 400);
        scene.getStylesheets().add(getClass().getResource("/ExampleStylesheet.css").toExternalForm());
        stage.setScene(scene);
        stage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }

    public Label createTestLabel() {
        Label testLabel = new Label("Placeholder\nText");
        testLabel.getStyleClass().add("test-label");
        testLabel.setTextAlignment(TextAlignment.CENTER);
        testLabel.setLayoutX(200);
        testLabel.setLayoutY(100);
        testLabel.setOnMouseEntered(event -> doSomething());
        testLabel.setOnMouseExited(event -> doSomethingElse());
        testLabel.addEventFilter(MouseEvent.MOUSE_MOVED, event -> {
            Bounds bounds = testLabel.getBoundsInParent();
            if (event.getX() < bounds.getMinX() || event.getX() > bounds.getMaxX() ||
                event.getY() < bounds.getMinY() || event.getY() > bounds.getMaxY()) {
                event.consume();
            }
        });
        return testLabel;
    }

    public void doSomething() {
        System.out.println("Inside label.");
    }

    public void doSomethingElse() {
        System.out.println("Outside label.");
    }
}
