import java.util.ArrayList;
import java.util.List;

public class PatternSearch {
    public static List<int[]> findPattern(int[][] T, int[][] P) {
        int m = T.length, n = T[0].length;
        int u = P.length, v = P[0].length;
        List<int[]> result = new ArrayList<>();

        for (int i = 0; i <= m - u; i++) {
            for (int j = 0; j <= n - v; j++) {
                if (matches(T, P, i, j)) {
                    result.add(new int[]{i, j});
                }
            }
        }
        return result;
    }

    private static boolean matches(int[][] T, int[][] P, int startX, int startY) {
        int u = P.length, v = P[0].length;
        for (int i = 0; i < u; i++) {
            for (int j = 0; j < v; j++) {
                if (P[i][j] != 0 && P[i][j] != T[startX + i][startY + j]) {
                    return false;
                }
            }
        }
        return true;
    }

    public static void main(String[] args) {
        int[][] T = {
                {2, 3, 4, 3, 6},
                {4, 1, 5, 7, 8},
                {9, 1, 2, 3, 1},
                {2, 4, 5, 1, 5},
                {3, 1, 9, 0, 2}
        };

        int[][] P = {
                {2, 3, 0},
                {0, 1, 5},
                {9, 0, 2}
        };

        List<int[]> positions = findPattern(T, P);
        for (int[] pos : positions) {
            System.out.println("(" + pos[0] + ", " + pos[1] + ")");
        }
    }
}
