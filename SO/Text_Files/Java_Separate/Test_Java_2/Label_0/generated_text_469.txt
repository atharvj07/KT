# application.yml (default properties)
spring:
  profiles:
    active: dev
  config:
    import: optional:classpath:.env[.properties]

---
# application-dev.yml (for dev profile)
spring:
  config:
    activate:
      on-profile: dev
    import: optional:classpath:.env-dev[.properties]

---
# application-prod.yml (for prod profile)
spring:
  config:
    activate:
      on-profile: prod
    import: optional:classpath:.env-prod[.properties]

bash
# .env (default environment variables)
DB_NAME=default_db
DB_USERNAME=default_user
DB_PASSWORD=default_password

# .env-dev (for dev profile)
DB_NAME=dev_db
DB_USERNAME=dev_user
DB_PASSWORD=dev_password

# .env-prod (for prod profile)
DB_NAME=prod_db
DB_USERNAME=prod_user
DB_PASSWORD=prod_password

bash
# Maven run command with environment variables
mvn spring-boot:run \
  -Dspring-boot.run.profiles=dev \
  -Dspring-boot.run.jvmArguments="-Xdebug" \
  -Dspring-boot.run.arguments="DB_NAME=employee_db DB_USERNAME=postgres DB_PASSWORD=******"
