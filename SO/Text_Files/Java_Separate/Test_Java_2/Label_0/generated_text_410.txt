import javax.xml.namespace.QName;
import javax.xml.ws.BindingProvider;
import javax.xml.ws.Service;
import java.net.URL;
import java.util.Map;

public class MDFeClient {
    public static String sendRequest(String xmlData) throws Exception {
        URL wsdlURL = new URL("https://mdfe-homologacao.svrs.rs.gov.br/ws/MDFeRecepcaoSinc/MDFeRecepcaoSinc.asmx?wsdl");
        QName serviceName = new QName("http://www.portalfiscal.inf.br/mdfe/wsdl/MDFeRecepcaoSinc", "MDFeRecepcaoSinc");
        QName portName = new QName("http://www.portalfiscal.inf.br/mdfe/wsdl/MDFeRecepcaoSinc", "MDFeRecepcaoSincSoap12");

        Service service = Service.create(wsdlURL, serviceName);
        MDFeRecepcaoSincSoap12 port = service.getPort(portName, MDFeRecepcaoSincSoap12.class);

        Map<String, Object> requestContext = ((BindingProvider) port).getRequestContext();
        requestContext.put(BindingProvider.ENDPOINT_ADDRESS_PROPERTY, "https://mdfe-homologacao.svrs.rs.gov.br/ws/MDFeRecepcaoSinc/MDFeRecepcaoSinc.asmx");
        requestContext.put("com.sun.xml.internal.ws.transport.http.client.HttpTransportPipe.dump", true);

        MdfeRecepcaoResult response = port.mdfeRecepcao(xmlData);
        return response != null ? response.getContent().toString() : "No Response";
    }
}
