import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.io.IOException;
import java.util.Iterator;
import java.util.Map;

public class JsonParser {
    public static void main(String[] args) {
        String json = "{ \"name\" : \"sahal\", \"address\" : [ { \"state\" : \"FL\" }, { \"country\" : \"FL\", \"city\" : { \"type\" : \"municipality\", \"value\" : \"California City\" } }, { \"pin\" : \"87876\" } ] }";
        try {
            ObjectMapper mapper = new ObjectMapper();
            JsonNode rootNode = mapper.readTree(json);
            parseJson(rootNode, "");
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    private static void parseJson(JsonNode node, String path) {
        if (node.isObject()) {
            Iterator<Map.Entry<String, JsonNode>> fields = node.fields();
            while (fields.hasNext()) {
                Map.Entry<String, JsonNode> field = fields.next();
                parseJson(field.getValue(), path.isEmpty() ? field.getKey() : path + "." + field.getKey());
            }
        } else if (node.isArray()) {
            for (int i = 0; i < node.size(); i++) {
                parseJson(node.get(i), path + "[" + i + "]");
            }
        } else {
            System.out.println(path + " : " + node.asText());
        }
    }
}
