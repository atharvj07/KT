import javafx.application.Application;
import javafx.geometry.Side;
import javafx.scene.Scene;
import javafx.scene.control.ContextMenu;
import javafx.scene.control.Menu;
import javafx.scene.control.MenuBar;
import javafx.scene.control.MenuItem;
import javafx.scene.control.MenuButton;
import javafx.scene.layout.BorderPane;
import javafx.stage.Stage;

public class UpwardMenuBar extends Application {

    @Override
    public void start(Stage stage) {
        MenuButton menuButton = new MenuButton("File");
        menuButton.getItems().addAll(
                new MenuItem("New"),
                new MenuItem("Open"),
                new MenuItem("Save"),
                new MenuItem("Exit")
        );

        menuButton.showingProperty().addListener((obs, wasShowing, isNowShowing) -> {
            if (isNowShowing) {
                ContextMenu contextMenu = menuButton.getContextMenu();
                if (contextMenu != null) {
                    contextMenu.setAnchorLocation(javafx.stage.PopupWindow.AnchorLocation.WINDOW_TOP_RIGHT);
                    contextMenu.show(menuButton, Side.TOP, 0, -menuButton.getHeight());
                }
            }
        });

        MenuBar menuBar = new MenuBar();
        menuBar.getMenus().add(new Menu("Edit"));
        menuBar.getMenus().add(new Menu("View"));

        BorderPane root = new BorderPane();
        root.setBottom(menuButton);
        root.setTop(menuBar);

        Scene scene = new Scene(root, 400, 300);
        stage.setScene(scene);
        stage.setTitle("Upward Opening Menu");
        stage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
