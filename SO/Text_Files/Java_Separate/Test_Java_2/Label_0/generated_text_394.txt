import javafx.application.Application;
import javafx.beans.value.ChangeListener;
import javafx.beans.value.ObservableValue;
import javafx.scene.Scene;
import javafx.scene.control.Spinner;
import javafx.scene.control.SpinnerValueFactory;
import javafx.scene.control.TextFormatter;
import javafx.scene.layout.HBox;
import javafx.stage.Stage;

public class SpinnerRestrictionApp extends Application {

    @Override
    public void start(Stage primaryStage) {
        Spinner<Integer> spinnerMin = new Spinner<>(2, 6, 2);
        Spinner<Integer> spinnerMax = new Spinner<>(2, 6, 6);

        disableButtons(spinnerMin, spinnerMax);

        spinnerMin.valueProperty().addListener((obs, oldVal, newVal) -> {
            spinnerMax.setValueFactory(new SpinnerValueFactory.IntegerSpinnerValueFactory(newVal, 6, spinnerMax.getValue()));
            disableButtons(spinnerMin, spinnerMax);
        });

        spinnerMax.valueProperty().addListener((obs, oldVal, newVal) -> {
            spinnerMin.setValueFactory(new SpinnerValueFactory.IntegerSpinnerValueFactory(2, newVal, spinnerMin.getValue()));
            disableButtons(spinnerMin, spinnerMax);
        });

        HBox root = new HBox(10, spinnerMin, spinnerMax);
        primaryStage.setScene(new Scene(root, 300, 100));
        primaryStage.setTitle("Spinner Restriction");
        primaryStage.show();
    }

    private void disableButtons(Spinner<Integer> spinnerMin, Spinner<Integer> spinnerMax) {
        boolean minAtMax = spinnerMin.getValue().equals(spinnerMax.getValue());
        boolean maxAtMin = spinnerMax.getValue().equals(spinnerMin.getValue());

        spinnerMin.lookup(".increment-arrow-button").setDisable(minAtMax);
        spinnerMax.lookup(".decrement-arrow-button").setDisable(maxAtMin);
    }

    public static void main(String[] args) {
        launch(args);
    }
}
