import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.TextArea;
import javafx.scene.cursor.Cursor;
import javafx.scene.layout.StackPane;
import javafx.stage.Stage;

public class TextAreaCursorExample extends Application {
    @Override
    public void start(Stage primaryStage) {
        TextArea numTextArea = new TextArea();
        numTextArea.setStyle("-fx-cursor: none;"); // Hide cursor using CSS

        StackPane root = new StackPane(numTextArea);
        Scene scene = new Scene(root, 400, 300);

        primaryStage.setScene(scene);
        primaryStage.setTitle("TextArea Cursor Example");
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
