import org.yaml.snakeyaml.Yaml;
import org.yaml.snakeyaml.constructor.Constructor;

import java.io.FileInputStream;
import java.io.IOException;
import java.util.Map;

public class YamlLoader {

    public static InterestingSetup loadInterestingSetup(String filePath) throws IOException {
        Yaml yaml = new Yaml();
        try (FileInputStream inputStream = new FileInputStream(filePath)) {
            Map<String, Object> yamlConfig = yaml.load(inputStream);
            Object interestingObject = yamlConfig.get("interesting-setup");
            return convertToInterestingSetup(interestingObject);
        }
    }

    @SuppressWarnings("unchecked")
    private static InterestingSetup convertToInterestingSetup(Object object) {
        if (object instanceof Map<?, ?> map) {
            Yaml yaml = new Yaml(new Constructor(InterestingSetup.class));
            return yaml.load(yaml.dump(map));
        }
        throw new IllegalArgumentException("Invalid YAML structure for InterestingSetup");
    }
}
