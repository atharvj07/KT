@Configuration
public class StreamProcessorConfig {

    @Bean
    public Map<String, Function<KStream<String, InputType>, KStream<String, OutputType>>> tenantProcessors(
            @Value("${tenants}") List<String> tenants) {
        return tenants.stream().collect(Collectors.toMap(
                tenant -> tenant + "Processor",
                tenant -> inputTypeStream -> inputTypeStream.map((String k, InputType v) ->
                        KeyValue.pair(k, OutputType.createFrom(v)))
        ));
    }
}

yaml
tenants: A,B,C

spring.cloud.function.definition: AProcessor;BProcessor;CProcessor

spring.cloud.stream.kafka.streams.binder.functions.AProcessor.applicationId: ${spring.application.name}-AProcessor
spring.cloud.stream.kafka.streams.binder.functions.BProcessor.applicationId: ${spring.application.name}-BProcessor
spring.cloud.stream.kafka.streams.binder.functions.CProcessor.applicationId: ${spring.application.name}-CProcessor

spring.cloud.stream.bindings.AProcessor-in-0.destination: A-input
spring.cloud.stream.bindings.AProcessor-out-0.destination: A-output

spring.cloud.stream.bindings.BProcessor-out-0.destination: B-output
