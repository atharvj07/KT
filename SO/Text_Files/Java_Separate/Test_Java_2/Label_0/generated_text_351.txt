public String transform(List<CustomObject> listOfObjects) {
    return listOfObjects.stream()
        .filter(object -> object.objectType().equals("BUSINESS"))
        .min(Comparator.comparing(CustomObject::url))
        .orElseThrow(() -> new RuntimeException("NO BUSINESS OBJECT FOUND"))
        .let(firstBusiness -> listOfObjects.stream()
            .filter(object -> object.objectType().equals("BUSINESS") && object.objectStatus().equals("ACTIVE"))
            .min(Comparator.comparing(CustomObject::url))
            .orElse(firstBusiness))
        .let(firstActive -> listOfObjects.stream()
            .filter(object -> object.objectType().equals("BUSINESS") && object.objectStatus().equals("ACTIVE") && object.locale().equals("en-US"))
            .min(Comparator.comparing(CustomObject::url))
            .orElse(firstActive))
        .getUrl();
}
