import java.util.Random;
import java.lang.reflect.Constructor;

public class Game {
    private Random random = new Random();
    private int sizeX;
    private int sizeY;
    private Field field;
    private Player player;

    public <T extends Fieldable> void positionEntity(Class<T> clazz) {
        int x, y;
        boolean entityCreated = false;
        while (!entityCreated) {
            x = random.nextInt(sizeX);
            y = random.nextInt(sizeY);
            if (field.getFieldable(x, y) instanceof Empty) {
                try {
                    Constructor<T> constructor = clazz.getConstructor(
                            int.class, int.class, int.class, int.class, 
                            Field.class, Player.class, Game.class
                    );
                    T entity = constructor.newInstance(x, y, sizeX, sizeY, field, player, this);
                    field.setFieldable(x, y, entity);
                    entityCreated = true;
                } catch (Exception e) {
                    throw new RuntimeException("Failed to create entity: " + clazz.getSimpleName(), e);
                }
            }
        }
    }
}
