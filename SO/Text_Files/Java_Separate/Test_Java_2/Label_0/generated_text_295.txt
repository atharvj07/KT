import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

public class ConcurrentComputeIfAbsent {
    private final ConcurrentHashMap<String, Integer> map = new ConcurrentHashMap<>();
    private volatile int sharedField = 0;

    public int computeValue(String key) {
        return map.computeIfAbsent(key, k -> {
            synchronized (this) {
                sharedField++;
                return sharedField * 10;
            }
        });
    }

    public static void main(String[] args) {
        ConcurrentComputeIfAbsent instance = new ConcurrentComputeIfAbsent();
        ExecutorService executor = Executors.newFixedThreadPool(2);

        executor.submit(() -> System.out.println("Thread 1: " + instance.computeValue("A")));
        executor.submit(() -> System.out.println("Thread 2: " + instance.computeValue("B")));

        executor.shutdown();
    }
}
