import javax.servlet.ServletConfig;
import javax.servlet.ServletException;
import javax.servlet.UnavailableException;
import javax.servlet.annotation.WebListener;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpSessionEvent;
import javax.servlet.http.HttpSessionListener;

public class MainServlet extends HttpServlet {
    @Override
    public void init(ServletConfig config) throws ServletException {
        super.init(config);
        try {
            connectToBackendServer();
        } catch (Exception e) {
            throw new UnavailableException("Cannot connect to Backend Server");
        }
    }

    private void connectToBackendServer() throws Exception {
        throw new NullPointerException("Backend connection failed");
    }
}

@WebListener
class DeploymentValidator implements HttpSessionListener {
    @Override
    public void sessionCreated(HttpSessionEvent se) {
        if (!isBackendAvailable()) {
            throw new RuntimeException("Deployment Aborted: Backend Server Unreachable");
        }
    }

    @Override
    public void sessionDestroyed(HttpSessionEvent se) {}

    private boolean isBackendAvailable() {
        try {
            new MainServlet().connectToBackendServer();
            return true;
        } catch (Exception e) {
            return false;
        }
    }
}
