@Bean
public TaskScheduler taskScheduler() {
    ThreadPoolTaskScheduler taskScheduler = new ThreadPoolTaskScheduler();
    taskScheduler.setPoolSize(10);
    taskScheduler.setThreadNamePrefix("custom-scheduler-");
    taskScheduler.initialize();
    return taskScheduler;
}

@Bean
public Executor taskExecutor() {
    ThreadPoolTaskExecutor executor = new ThreadPoolTaskExecutor();
    executor.setCorePoolSize(10);
    executor.setMaxPoolSize(20);
    executor.setQueueCapacity(500);
    executor.setThreadNamePrefix("custom-executor-");
    executor.initialize();
    return executor;
}

@Bean
public MeterBinder taskExecutorMetrics(TaskExecutor taskExecutor, MeterRegistry meterRegistry) {
    if (taskExecutor instanceof ThreadPoolTaskExecutor) {
        return new ExecutorServiceMetrics(((ThreadPoolTaskExecutor) taskExecutor).getThreadPoolExecutor(),
                "custom.executor", emptyList(), emptyList());
    }
    return registry -> {};
}

@Bean
public MeterBinder taskSchedulerMetrics(TaskScheduler taskScheduler, MeterRegistry meterRegistry) {
    if (taskScheduler instanceof ThreadPoolTaskScheduler) {
        return new ExecutorServiceMetrics(((ThreadPoolTaskScheduler) taskScheduler).getScheduledThreadPoolExecutor(),
                "custom.scheduler", emptyList(), emptyList());
    }
    return registry -> {};
}
