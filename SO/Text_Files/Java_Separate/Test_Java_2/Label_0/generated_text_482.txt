import android.text.SpannableString;
import android.text.Spanned;
import android.text.style.TabStopSpan;
import android.widget.TextView;

public void setAlignedText(TextView textView, String leftText, String rightText) {
    int tabStop = textView.getWidth() - textView.getPaddingLeft() - textView.getPaddingRight();
    SpannableString spannableString = new SpannableString(leftText + "\t" + rightText);
    spannableString.setSpan(new TabStopSpan.Standard(tabStop), leftText.length() + 1, leftText.length() + 2, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE);
    textView.setText(spannableString);
}
