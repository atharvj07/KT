import java.util.*;

public class Solution {
    public static List<Integer> findSubsetWithSum(int[] nums, int target, int limit) {
        Arrays.sort(nums);
        List<Integer> result = new ArrayList<>();
        if (findSubset(nums, nums.length - 1, target, limit, result)) {
            return result;
        }
        return Collections.emptyList();
    }

    private static boolean findSubset(int[] nums, int index, int target, int limit, List<Integer> result) {
        if (target == 0 && result.size() == limit) {
            return true;
        }
        if (index < 0 || result.size() >= limit) {
            return false;
        }
        result.add(nums[index]);
        if (findSubset(nums, index - 1, target - nums[index], limit, result)) {
            return true;
        }
        result.remove(result.size() - 1);
        return findSubset(nums, index - 1, target, limit, result);
    }

    public static void main(String[] args) {
        int[] nums = {2, 7, 11, 15};
        int target = 20;
        int limit = 3;
        System.out.println(findSubsetWithSum(nums, target, limit));
    }
}
