public class EqualSplit {
    public static boolean equalSplit(int[] arr) {
        if (arr.length % 2 != 0) {
            return false;
        }
        int totalSum = arraySum(arr, 0);
        if (totalSum % 2 != 0) {
            return false;
        }
        return equalSplit(arr, 0, 0, 0, arr.length / 2, totalSum / 2);
    }

    private static boolean equalSplit(int[] arr, int index, int sum, int count, int halfSize, int targetSum) {
        if (count == halfSize) {
            return sum == targetSum;
        }
        if (index == arr.length) {
            return false;
        }
        return equalSplit(arr, index + 1, sum + arr[index], count + 1, halfSize, targetSum) ||
               equalSplit(arr, index + 1, sum, count, halfSize, targetSum);
    }

    private static int arraySum(int[] arr, int index) {
        return index == arr.length ? 0 : arr[index] + arraySum(arr, index + 1);
    }

    public static void main(String[] args) {
        int[] arr1 = {1, 2, 3, 4, 5, 6};
        int[] arr2 = {1, 1, 1, 1, 2, 2};
        int[] arr3 = {1, 2, 3, 5, 5, 6};
        System.out.println(equalSplit(arr1)); // false
        System.out.println(equalSplit(arr2)); // true
        System.out.println(equalSplit(arr3)); // true
    }
}
