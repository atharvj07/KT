import org.springframework.util.StringUtils;
import org.springframework.web.servlet.LocaleResolver;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import java.util.Arrays;
import java.util.List;
import java.util.Locale;

public class RestrictedLocaleResolver implements LocaleResolver {
    private static final List<Locale> ALLOWED_LOCALES = Arrays.asList(Locale.ENGLISH, new Locale("hi"));

    @Override
    public Locale resolveLocale(HttpServletRequest request) {
        String localeParam = request.getParameter("locale");
        if (StringUtils.hasText(localeParam)) {
            Locale requestedLocale = Locale.forLanguageTag(localeParam);
            if (ALLOWED_LOCALES.contains(requestedLocale)) {
                return requestedLocale;
            } else {
                throw new IllegalArgumentException("Invalid locale: " + localeParam);
            }
        }
        return Locale.ENGLISH;
    }

    @Override
    public void setLocale(HttpServletRequest request, HttpServletResponse response, Locale locale) {
        if (!ALLOWED_LOCALES.contains(locale)) {
            throw new IllegalArgumentException("Invalid locale: " + locale);
        }
    }
}
