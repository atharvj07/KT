import java.awt.*;
import java.awt.image.*;
import java.util.*;
import javax.swing.*;

class Bul {
    double x, y, d;
    BufferedImage img;
    static Map<Double, BufferedImage> rotCache = new HashMap<>();

    Bul(double x, double y, double d, BufferedImage img) {
        this.x = x;
        this.y = y;
        this.d = d;
        this.img = rot(d, img);
    }

    BufferedImage rot(double d, BufferedImage img) {
        return rotCache.computeIfAbsent(d, k -> {
            int w = img.getWidth(), h = img.getHeight();
            BufferedImage out = new BufferedImage(w, h, BufferedImage.TYPE_INT_ARGB);
            Graphics2D g = out.createGraphics();
            g.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);
            g.rotate(Math.toRadians(k), w / 2.0, h / 2.0);
            g.drawImage(img, 0, 0, null);
            g.dispose();
            return out;
        });
    }
}

class Pan extends JPanel {
    List<Bul> bls = new ArrayList<>();
    BufferedImage img;

    Pan(BufferedImage img) {
        this.img = img;
        for (int i = 0; i < 10000; i++) bls.add(new Bul(Math.random() * 800, Math.random() * 600, Math.random() * 360, img));
        new Timer(16, e -> repaint()).start();
    }

    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        for (Bul b : bls) g.drawImage(b.img, (int) b.x, (int) b.y, null);
    }
}

public class Gam {
    public static void main(String[] a) throws Exception {
        BufferedImage img = new BufferedImage(16, 16, BufferedImage.TYPE_INT_ARGB);
        Graphics2D g = img.createGraphics();
        g.setColor(Color.RED);
        g.fillOval(0, 0, 16, 16);
        g.dispose();
        JFrame f = new JFrame();
        f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        f.setSize(800, 600);
        f.add(new Pan(img));
        f.setVisible(true);
    }
}
