public class PAdp extends RecyclerView.Adapter<PAdp.VH> {
    private final Context ctx;
    private final List<PMod> lst;
    private final int dCol, sCol;
    private final Set<Integer> sel;

    private static final String P_N = "PqItm";
    private static final String S_K = "PqSel";

    public PAdp(Context ctx, List<PMod> lst) {
        this.ctx = ctx;
        this.lst = lst;
        this.sel = new HashSet<>();

        Resources res = ctx.getResources();
        int m = res.getConfiguration().uiMode & Configuration.UI_MODE_NIGHT_MASK;
        dCol = ContextCompat.getColor(ctx, (m == Configuration.UI_MODE_NIGHT_YES) ? R.color.dD : R.color.dL);
        sCol = ContextCompat.getColor(ctx, (m == Configuration.UI_MODE_NIGHT_YES) ? R.color.sD : R.color.sL);

        SharedPreferences p = ctx.getSharedPreferences(P_N, Context.MODE_PRIVATE);
        for (String i : p.getStringSet(S_K, new HashSet<>())) sel.add(Integer.parseInt(i));
    }

    @NonNull
    @Override
    public VH onCreateViewHolder(@NonNull ViewGroup p, int v) {
        return new VH(LayoutInflater.from(p.getContext()).inflate(R.layout.pq_l, p, false));
    }

    @Override
    public void onBindViewHolder(@NonNull VH h, int p) {
        final PMod t = lst.get(p);
        h.cv.setCardBackgroundColor(sel.contains(p) ? sCol : dCol);
        h.cv.setOnClickListener(v -> {
            SharedPreferences e = ctx.getSharedPreferences(P_N, Context.MODE_PRIVATE);
            sel.add(p);
            e.edit().putStringSet(S_K, sel.stream().map(String::valueOf).collect(Collectors.toSet())).apply();
            notifyItemChanged(p);
            Intent i = new Intent(h.cv.getContext(), VPdf.class);
            i.putExtra("pN", t.getN());
            i.putExtra("pU", t.getU());
            i.setFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            h.cv.getContext().startActivity(i);
        });
        h.t.setText(t.getN());
    }

    @Override
    public int getItemCount() {
        return lst.size();
    }

    public static class VH extends RecyclerView.ViewHolder {
        CardView cv;
        TextView t;

        public VH(View i) {
            super(i);
            cv = i.findViewById(R.id.pqC);
            t = i.findViewById(R.id.pqT);
        }
    }
}
