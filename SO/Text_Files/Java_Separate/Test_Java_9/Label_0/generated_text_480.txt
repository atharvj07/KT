public interface IRep extends JpaRepository<ITm, Long> {
    Optional<ITm> findByNIgnoreCase(String n);
}

@Service
public class ISvc {

    private final IRep rep;
    private final CRep crep;
    private final IDMap map;
    private final IImg img;

    public ISvc(IRep rep, CRep crep, IDMap map, IImg img) {
        this.rep = rep;
        this.crep = crep;
        this.map = map;
        this.img = img;
    }

    @Transactional
    public CRU create(CData d, UriComponentsBuilder ub) {
        Optional<ITm> f = rep.findByNIgnoreCase(d.n());

        if (f.isPresent()) throw new ICrtEx("There is an item created with this name");

        return new CRU();
    }
}
