import jakarta.annotation.Resource;
import jakarta.inject.Inject;
import jakarta.jms.*;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.io.PrintWriter;

public class MsgSrv extends HttpServlet {
    @Inject
    @JMSConnectionFactory("java:/jms/remoteCF")
    private JMSContext ctx;

    @Resource(lookup = "java:/queue/testQueue")
    private Queue q;

    private static final int CNT = 5;

    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse res) throws ServletException, IOException {
        res.setContentType("text/html");
        try (PrintWriter out = res.getWriter()) {
            out.write("<p>Sending msgs to <em>" + q + "</em></p>");
            out.write("<h2>Msgs sent and rec:</h2>");
            for (int i = 0; i < CNT; i++) {
                String txt = "Msg " + (i + 1);
                ctx.createProducer().send(q, txt);
                out.write("Sent (" + i + "): " + txt + "<br/>");
            }
        }
    }
}
