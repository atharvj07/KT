public class GenTemp {
    private static final AtomicLong CNT = new AtomicLong(0);

    public static String genTemp() {
        long val = CNT.getAndIncrement();
        return String.format("%04X-7%03X-%04X", (val >> 32) & 0xFFFF, (val >> 16) & 0xFFF, val & 0xFFFF).toLowerCase();
    }

    public static void main(String[] args) {
        for (int i = 0; i < 3; i++) {
            System.out.println(genTemp());
        }
    }
}
