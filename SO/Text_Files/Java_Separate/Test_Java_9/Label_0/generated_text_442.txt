import java.time.*;
import java.time.format.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        DateTimeFormatter fmt = new DateTimeFormatterBuilder()
            .parseCaseInsensitive()
            .appendPattern("ddMMMyyyy")
            .toFormatter(Locale.of("es", "ES"));

        Map<String, String> rep = Map.of(
            "ENE", "JAN", "FEB", "FEB", "MAR", "MAR",
            "ABR", "APR", "MAY", "MAY", "JUN", "JUN",
            "JUL", "JUL", "AGO", "AUG", "SEP", "SEP",
            "OCT", "OCT", "NOV", "NOV", "DIC", "DEC"
        );

        String txt = "24ENE1982";
        for (var e : rep.entrySet()) {
            txt = txt.replace(e.getKey(), e.getValue());
        }

        LocalDate dt = LocalDate.parse(txt, DateTimeFormatter.ofPattern("ddMMMyyyy", Locale.ENGLISH));
        System.out.println("Hello world! " + dt);
    }
}
