import java.io.*;
import java.nio.charset.StandardCharsets;
import java.security.*;
import java.security.spec.*;
import javax.crypto.*;
import javax.crypto.spec.*;
import org.bouncycastle.openssl.*;
import org.bouncycastle.openssl.jcajce.*;

public class KeyUtil {
    public static PrivateKey getKey(String path, String pwd) throws Exception {
        PEMParser p = new PEMParser(new FileReader(path));
        Object o = p.readObject();
        p.close();
        JcaPEMKeyConverter c = new JcaPEMKeyConverter().setProvider("BC");
        if (o instanceof PEMEncryptedKeyPair) {
            PEMDecryptorProvider d = new JcePEMDecryptorProviderBuilder().build(pwd.toCharArray());
            o = ((PEMEncryptedKeyPair) o).decryptKeyPair(d);
        }
        return c.getKeyPair((PEMKeyPair) o).getPrivate();
    }

    public static byte[] enc(PrivateKey k, byte[] d) throws Exception {
        Cipher c = Cipher.getInstance("RSA");
        c.init(Cipher.ENCRYPT_MODE, k);
        return c.doFinal(d);
    }

    public static byte[] dec(PrivateKey k, byte[] d) throws Exception {
        Cipher c = Cipher.getInstance("RSA");
        c.init(Cipher.DECRYPT_MODE, k);
        return c.doFinal(d);
    }

    public static void main(String[] a) throws Exception {
        PrivateKey k = getKey("key.pem", "pass");
        byte[] msg = "Hello".getBytes(StandardCharsets.UTF_8);
        byte[] en = enc(k, msg);
        byte[] de = dec(k, en);
        System.out.println(new String(de, StandardCharsets.UTF_8));
    }
}
