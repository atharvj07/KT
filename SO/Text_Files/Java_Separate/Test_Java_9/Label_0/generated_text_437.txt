import java.util.function.Function;
import java.util.stream.DoubleStream;

public enum Agg {
    MIN(DoubleStream::min),
    MAX(DoubleStream::max),
    AVG(DoubleStream::average);

    private final Function<DoubleStream, java.util.OptionalDouble> fn;

    Agg(Function<DoubleStream, java.util.OptionalDouble> fn) {
        this.fn = fn;
    }

    public double app(DoubleStream ds) {
        return fn.apply(ds).orElse(-1);
    }
}

public class Cal {
    public float calc(Agg a) {
        return (float) a.app(provides.stream().mapToDouble(this::someFunc));
    }
}
