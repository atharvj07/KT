public class Tst {
    public static void prs(String pth) throws Exception {
        try (CSVReader rdr = new CSVReaderBuilder(
                new InputStreamReader(Objects.requireNonNull(Tst.class.getResourceAsStream(pth))))
                .withCSVParser(new CSVParserBuilder().withSeparator('\t').build())
                .build()) {
            String[] ln;
            while ((ln = rdr.readNext()) != null) {
                System.out.println(ln[0] + " " + ln[1]);
            }
        }
    }

    public static void main(String[] args) {
        try {
            prs("pth-tsv");
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }
}
