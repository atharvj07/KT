import javax.xml.transform.*;
import javax.xml.transform.stream.*;
import java.io.*;

public class App {
    public static void main(String[] a) throws Exception {
        var f = new File("D:\\Demo\\src\\test\\java\\StringXml\\_ignoreText.xsl");
        var s = "<?xml version='1.0' encoding='UTF-8'?>"
              + "<root><n>123</n><t>hello</t></root>";
        var tf = TransformerFactory.newInstance();
        var t = tf.newTransformer(new StreamSource(f));
        var r = new StringWriter();
        t.transform(new StreamSource(new StringReader(s)), new StreamResult(r));
        System.out.println(r);
    }
}
