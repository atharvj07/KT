import org.timefold.solver.core.api.score.stream.*;
import java.util.*;

public class TimConProv implements ConstraintProvider {
    private static Set<String> selCon = new HashSet<>();

    public static void upd(Set<String> con) {
        selCon = con;
    }

    @Override
    public Constraint[] defineConstraints(ConstraintFactory cf) {
        List<Constraint> lst = new ArrayList<>();
        if (selCon.contains("room")) lst.add(room(cf));
        if (selCon.contains("teach")) lst.add(teach(cf));
        if (selCon.contains("stud")) lst.add(stud(cf));
        if (selCon.contains("stab")) lst.add(stab(cf));
        if (selCon.contains("eff")) lst.add(eff(cf));
        if (selCon.contains("var")) lst.add(var(cf));
        return lst.toArray(new Constraint[0]);
    }
