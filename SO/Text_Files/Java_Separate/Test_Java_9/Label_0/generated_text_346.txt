import com.sun.source.tree.*;
import com.sun.source.util.Trees;
import javax.lang.model.type.TypeMirror;
import javax.lang.model.util.Elements;

public class SwitchTypeFinder {

    private Trees trees;
    private Elements elements;

    public SwitchTypeFinder(Trees trees, Elements elements) {
        this.trees = trees;
        this.elements = elements;
    }

    public TypeMirror getSwitchExpressionType(MethodTree methodTree) {
        BlockTree blockTree = methodTree.getBody();

        for (StatementTree statementTree : blockTree.getStatements()) {
            if (statementTree.getKind() == Tree.Kind.SWITCH) {
                SwitchTree switchTree = (SwitchTree) statementTree;
                ExpressionTree switchTreeExpression = switchTree.getExpression();
                return trees.getTypeMirror(new TreePath(trees.getPath(methodTree), switchTreeExpression));
            }
        }
        return null;
    }
}
