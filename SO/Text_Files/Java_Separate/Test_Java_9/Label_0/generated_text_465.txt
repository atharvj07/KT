public class Src {
    public static boolean search(int[][] mat, int num) {
        int n = mat.length, r = 0, c = n - 1;
        while (r < n && c >= 0) {
            if (mat[r][c] == num) return true;
            if (mat[r][c] > num) c--;
            else r++;
        }
        return false;
    }
}
