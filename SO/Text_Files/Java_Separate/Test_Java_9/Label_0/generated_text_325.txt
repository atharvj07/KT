public class FpTest {
    public static void main(String[] args) {
        run();
        runStrict();
    }

    static void run() {
        double a = Double.MAX_VALUE;
        double b = a * 2;
        double c = b / 4;
        System.out.println("Norm: " + c);
    }

    strictfp static void runStrict() {
        double a = Double.MAX_VALUE;
        double b = a * 2;
        double c = b / 4;
        System.out.println("Strict: " + c);
    }
}
