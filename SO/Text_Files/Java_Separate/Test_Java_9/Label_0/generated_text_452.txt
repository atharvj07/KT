public class StsTypeHndlr extends BaseTypeHandler<Task.Status> {
    @Override
    public void setNonNullParameter(PreparedStatement ps, int i, Task.Status p, JdbcType jt) throws SQLException {
        ps.setInt(i, p.getId());
    }

    @Override
    public Task.Status getNullableResult(ResultSet rs, String col) throws SQLException {
        return fromId(rs.getInt(col));
    }

    @Override
    public Task.Status getNullableResult(ResultSet rs, int i) throws SQLException {
        return fromId(rs.getInt(i));
    }

    @Override
    public Task.Status getNullableResult(CallableStatement cs, int i) throws SQLException {
        return fromId(cs.getInt(i));
    }

    private Task.Status fromId(int id) {
        return Arrays.stream(Task.Status.values())
                .filter(v -> v.getId().equals(id))
                .findFirst()
                .orElse(null);
    }
}
