import javafx.application.*;
import javafx.scene.*;
import javafx.scene.control.*;
import javafx.scene.layout.*;
import javafx.stage.*;

public class App extends Application {
    @Override
    public void start(Stage p) {
        ComboBox<String> cb = new ComboBox<>();
        cb.getItems().addAll("One", "Two", "Three");
        cb.getStyleClass().add("yel");
        cb.setOnShowing(e -> cb.lookup(".list-view").getStyleClass().add("yel"));

        VBox v = new VBox(cb);
        Scene s = new Scene(v, 400, 300);
        s.getStylesheets().add(App.class.getResource("test.css").toExternalForm());
        p.setScene(s);
        p.show();
    }

    public static void main(String[] a) {
        launch(a);
    }
}
