import java.util.*;
import java.util.concurrent.*;

class MltSum {

    public int getSum(int n) {
        Set<Integer> res = ConcurrentHashMap.newKeySet();
        ExecutorService ex = Executors.newFixedThreadPool(3);

        ex.execute(() -> addMul(n, 3, res));
        ex.execute(() -> addMul(n, 5, res));
        ex.execute(() -> addMul(n, 7, res));

        ex.shutdown();
        try { ex.awaitTermination(Long.MAX_VALUE, TimeUnit.NANOSECONDS); } 
        catch (InterruptedException e) {}

        return res.stream().mapToInt(Integer::intValue).sum();
    }

    private void addMul(int n, int d, Set<Integer> res) {
        for (int i = d; i <= n; i += d) res.add(i);
    }

    public static void main(String[] args) {
        MltSum obj = new MltSum();
        System.out.println(obj.getSum(10));
        System.out.println(obj.getSum(1000));
        System.out.println(obj.getSum(76293));
    }
}
