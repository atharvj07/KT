import java.util.*;

class Brd {
    public String gss;

    public String tke() {
        Scanner sc = new Scanner(System.in);
        String gss;

        System.out.println("choose a word, pick carefully: ");
        gss = sc.next();
        
        chck(gss);
        
        return gss;
    }

    public void chck(String gss) {
        boolean isOk = true;

        if (gss.length() != 5)
            isOk = false;

        if (!gss.matches("[a-zA-Z]+"))
            isOk = false;

        if (!isOk) {
            System.out.println("make sure your guess consists of 5 english letters, try again.");
            gss = tke();
        }
    }
}

public class Main {
    public static void main(String[] args) {
        Brd brd = new Brd();
        String guess = brd.tke();
    }
}
