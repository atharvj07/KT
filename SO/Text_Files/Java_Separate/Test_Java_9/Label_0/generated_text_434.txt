import java.util.*;

public class SumLim {
    public List<Integer> getSet(int[] arr, int tgt, int lim) {
        List<Integer> res = new ArrayList<>();
        if (dfs(arr, tgt, lim, 0, res)) return res;
        return Collections.emptyList();
    }

    private boolean dfs(int[] arr, int tgt, int lim, int idx, List<Integer> res) {
        if (tgt == 0 && res.size() == lim) return true;
        if (res.size() >= lim || idx >= arr.length) return false;
        res.add(arr[idx]);
        if (dfs(arr, tgt - arr[idx], lim, idx + 1, res)) return true;
        res.remove(res.size() - 1);
        return dfs(arr, tgt, lim, idx + 1, res);
    }

    public static void main(String[] args) {
        SumLim obj = new SumLim();
        System.out.println(obj.getSet(new int[]{2, 7, 11, 15}, 20, 3));
    }
}
