import org.springframework.context.annotation.*;
import org.springframework.security.config.annotation.web.messaging.*;
import org.springframework.security.config.annotation.web.builders.*;
import org.springframework.security.config.http.*;
import org.springframework.security.web.*;

@Configuration
@EnableWebSecurity
public class SecCfg {

    @Bean
    SecFil security(HttpSecurity h) throws Exception {
        return h.csrf(c -> c.ignoringRequestMatchers("/ws/**"))
                .cors(c -> c.disable())
                .authorizeHttpRequests(a -> a
                        .requestMatchers("/auth").permitAll()
                        .requestMatchers("/chat").authenticated()
                        .requestMatchers("/pub/**").permitAll()
                        .requestMatchers("/prv/**").permitAll()
                        .requestMatchers("/ws/**").authenticated()
                        .requestMatchers("/reg").permitAll()
                        .requestMatchers("/csrf").authenticated()
                        .requestMatchers("/favEmp").hasAnyAuthority("EMP", "ADM")
                        .requestMatchers("/favChat").hasAnyAuthority("EMP", "ADM")
                        .requestMatchers("/home").hasAnyAuthority("EMP", "ADM"))
                .build();
    }
}

@Configuration
@EnableWebSocketSecurity
public class WsSecCfg extends AbsSecWebMsgBroCfg {

    @Bean
    AutMan<Message<?>> msgAuthMan(MsgMatDelAutMan.Builder m) {
        m.anyMessage().permitAll();
        return m.build();
    }

    @Override
    protected boolean sameOriginDisabled() {
        return true;
    }
}
