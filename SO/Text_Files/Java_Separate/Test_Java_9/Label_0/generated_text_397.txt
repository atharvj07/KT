import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.io.File;
import java.util.Iterator;
import java.util.Map;

public class Jsn {
    public static void main(String[] a) {
        try {
            ObjectMapper m = new ObjectMapper();
            JsonNode n = m.readTree(new File("data.json"));
            prc(n, "");
        } catch (Exception e) {
            System.out.println(e);
        }
    }

    private static void prc(JsonNode n, String p) {
        if (n.isObject()) {
            Iterator<Map.Entry<String, JsonNode>> i = n.fields();
            while (i.hasNext()) {
                Map.Entry<String, JsonNode> e = i.next();
                prc(e.getValue(), p.isEmpty() ? e.getKey() : p + "." + e.getKey());
            }
        } else if (n.isArray()) {
            for (int i = 0; i < n.size(); i++) {
                prc(n.get(i), p + "[" + i + "]");
            }
        } else {
            System.out.println(p + " : " + n.asText());
        }
    }
}
