import java.util.concurrent.*;
import java.util.concurrent.locks.*;

public class Run {
    public static void main(String[] args) {
        Exe svc = new Exe(5);
        svc.run("hi");
        svc.run("h");
        svc.run("u");
        svc.run("y");
        svc.run("hi");
        svc.run("p");
        svc.run("o");
        svc.run("bb");
        svc.shut();
    }
}

class Exe {
    private final ExecutorService ex;
    private final ConcurrentMap<String, Lock> map = new ConcurrentHashMap<>();

    public Exe(int n) {
        this.ex = Executors.newFixedThreadPool(n);
    }

    public void run(String s) {
        ex.submit(() -> {
            Lock l = map.computeIfAbsent(s, k -> new ReentrantLock());
            if (l.tryLock()) {
                try {
                    System.out.println(s);
                    Thread.sleep(5000);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                } finally {
                    l.unlock();
                }
            }
        });
    }

    public void shut() {
        ex.shutdown();
    }
}
