public String toCsv(DmnRuleTestResult res) {
    List<DmnTestCaseResult> lst = res.getTestCases();
    try (StringWriter sw = new StringWriter(); 
         CSVWriter wr = new CSVWriter(sw, ',', CSVWriter.NO_QUOTE_CHARACTER, CSVWriter.NO_ESCAPE_CHARACTER, "\n")) {
        
        wr.writeNext(new String[]{"testInput", "expectedOutput", "testOutput", "errorMessage"});
        
        for (DmnTestCaseResult r : lst) {
            for (Map<String, Object> out : r.getTestOutput()) {
                wr.writeNext(new String[]{
                    r.getTestInput().toString(),
                    r.getExpectedOutput().toString(),
                    out.toString(),
                    r.getErrorMessage()
                });
            }
        }
        return sw.toString();
    } catch (Exception e) {
        throw new CsvParseException(e.getMessage());
    }
}
