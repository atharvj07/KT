import java.time.*;
import java.time.format.*;
import java.util.*;

public class DteFmt {
    public static void main(String[] args) {
        String in = "18-FEB-24 10.02.33.578000000 AM";
        DateTimeFormatter inf = DateTimeFormatter.ofPattern("dd-MMM-yy hh.mm.ss.SSSSSSSSS a", Locale.ENGLISH);
        DateTimeFormatter outf = DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss.SSSSSS");
        LocalDateTime dte = LocalDateTime.parse(in, inf);
        System.out.println(dte.format(outf));
    }
}
