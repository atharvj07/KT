package enc.me;

import javax.crypto.Cipher;
import javax.crypto.spec.IvParameterSpec;
import javax.crypto.spec.SecretKeySpec;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.util.Base64;

public class Enc {
    public static void main(String[] args) throws Exception {
        byte[] key = Files.readAllBytes(Paths.get("/home/debashishd/Downloads/s_key"));
        byte[] fixKey = new byte[32];
        System.arraycopy(key, 0, fixKey, 0, Math.min(key.length, 32));

        SecretKeySpec sec = new SecretKeySpec(fixKey, "AES");
        IvParameterSpec iv = new IvParameterSpec(new byte[16]);

        Cipher en = Cipher.getInstance("AES/CBC/PKCS5Padding");
        en.init(Cipher.ENCRYPT_MODE, sec, iv);
        String msg = "Hello this is Roxane";
        String enc = Base64.getEncoder().encodeToString(en.doFinal(msg.getBytes(StandardCharsets.UTF_8)));

        System.out.println("Encrypted value: " + enc);

        Cipher de = Cipher.getInstance("AES/CBC/PKCS5Padding");
        de.init(Cipher.DECRYPT_MODE, sec, iv);
        String dec = new String(de.doFinal(Base64.getDecoder().decode(enc)), StandardCharsets.UTF_8);

        System.out.println("Decrypted value: " + dec);
    }
}
