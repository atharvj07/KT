import android.annotation.SuppressLint;
import android.content.ClipData;
import android.content.ClipboardManager;
import android.content.Context;
import android.os.Bundle;
import android.widget.Toast;
import androidx.annotation.NonNull;
import androidx.appcompat.app.AppCompatActivity;
import com.google.android.gms.ads.*;

public class Hm extends AppCompatActivity {
    int clkN = 0, clkC = 0, trgN = 7, trgC = 3;
    InterstitialAd adN, adC;

    protected void onCreate(Bundle s) {
        super.onCreate(s);
        setContentView(R.layout.home_activity);
        MobileAds.initialize(this, st -> {});
        AdView adV = findViewById(R.id.adView);
        AdRequest rq = new AdRequest.Builder().build();
        adV.loadAd(rq);
        adV.setAdListener(new AdListener() {
            public void onAdFailedToLoad(@NonNull LoadAdError e) {
                adV.loadAd(rq);
            }
        });
        lAdN();
        lAdC();
    }

    @SuppressLint("SetTextI18n")
    private void nxt() {
        clkN++;
        if (adN != null && clkN >= trgN) shN();
    }

    private void cpy() {
        clkC++;
        ClipboardManager cm = (ClipboardManager) getSystemService(Context.CLIPBOARD_SERVICE);
        if (cm != null) cm.setPrimaryClip(ClipData.newPlainText("txt", "data"));
        Toast.makeText(getApplicationContext(), "Copied", Toast.LENGTH_SHORT).show();
        if (adC != null && clkC >= trgC) shC();
    }

    private void shN() {
        adN.show(Hm.this);
        adN.setFullScreenContentCallback(new FullScreenContentCallback() {
            public void onAdDismissedFullScreenContent() {
                adN = null;
                lAdN();
                clkN = 0;
            }
        });
    }

    private void shC() {
        adC.show(Hm.this);
        adC.setFullScreenContentCallback(new FullScreenContentCallback() {
            public void onAdDismissedFullScreenContent() {
                adC = null;
                lAdC();
                clkC = 0;
            }
        });
    }

    private void lAdN() {
        InterstitialAd.load(this, "ca-app-pub-3940256099942544/1033173712", new AdRequest.Builder().build(),
                new InterstitialAdLoadCallback() {
                    public void onAdLoaded(@NonNull InterstitialAd a) {
                        adN = a;
                    }

                    public void onAdFailedToLoad(@NonNull LoadAdError e) {
                        adN = null;
                    }
                });
    }

    private void lAdC() {
        InterstitialAd.load(this, "ca-app-pub-3940256099942544/8691691433", new AdRequest.Builder().build(),
                new InterstitialAdLoadCallback() {
                    public void onAdLoaded(@NonNull InterstitialAd a) {
                        adC = a;
                    }

                    public void onAdFailedToLoad(@NonNull LoadAdError e) {
                        adC = null;
                    }
                });
    }
}
