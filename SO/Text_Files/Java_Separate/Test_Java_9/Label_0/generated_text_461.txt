try (Stream<Employee> s = empRepo.findAllStream()) {
    response.setHeader("content-type", "application/json");
    PrintWriter w = response.getWriter();
    Iterator<Employee> i = s.iterator();
    w.write("[");
    if (i.hasNext()) {
        w.write(jsonSerialize(i.next()));
        while (i.hasNext()) {
            w.write(",");
            w.write(jsonSerialize(i.next()));
        }
    }
    w.write("]");
    w.flush();
} catch (IOException e) {
    log(e);
}
