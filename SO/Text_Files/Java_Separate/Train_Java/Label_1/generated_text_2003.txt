class Hotel {
    private String city;
    private  int numberOfStart;
    private List&lt;Room&gt; rooms ;
    
    public Hotel(String city, int numberOfStart,List&lt;Room&gt; rooms) {
        
       rooms.forEach(s-&gt;s.setRefHotel(this))
        this.city = city;
        this.numberOfStart = numberOfStart;
    }
}


class Room  {
    private Hotel refHotel;

    public void setRefHotel(Hotel refHotel) {
        this.refHotel = refHotel;
    }

    public Hotel getRefHotel() {
        return refHotel;
    }

}
