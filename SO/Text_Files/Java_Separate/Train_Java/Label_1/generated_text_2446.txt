public static int[] solve(int[] A, int[][] commands) {
    int max = Integer.MIN_VALUE;
    boolean[] initialized = new boolean[A.length]; // note: initialized to false
    int[] items = new int[A.length];
    for (int[] command : Arrays.asList(commands).reversed()) {
        if (command[0] == 1) {
            // set(i, v)
            int i = command[1];
            if (initialized[i])
                continue; // overwritten by a later set operation
            int v = command[2];
            items[i] = Math.max(max, v); // apply clamping of later clamp commands
            initialized[i] = true;
        } else {
            // clamp(min)
            assert command[0] == 2;
            int min = command[2];
            max = Math.max(max, min); // remember most restrictive clamping
        }
    }
    for (int i = 0; i &lt; A.length; i++)
        if (!initialized[i])
            items[i] = Math.max(max, A[i]); // clamped &quot;default&quot; value
    return items;
}
