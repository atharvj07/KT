@javax.persistence.Entity
@Data
public class Entity {
    @Id
    private Long id;

    @ManyToOne
    private Relation relation;
}

@javax.persistence.Entity
public class Relation {
    @Id
    private Long id;

    @ManyToMany
    private Set&lt;DistantRelation&gt; distantRelation;
}

@Entity
public class DistantRelation {
    @Id
    private Long id;

    private String name;

    @ManyToMany
    private Set&lt;Relation&gt; relation;
}
