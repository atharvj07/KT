public class DuplicateEntry {

    public DuplicateEntry(Event evt1, int differenceScore, Event evt2) {
        this.evt1 = evt1;
        this.differenceScore = differenceScore;
        this.evt2 = evt2;
    }

    @Getter @Setter
    private Event evt1;

    @Getter @Setter
    private int differenceScore;

    @Getter @Setter
    private Event evt2;

}
