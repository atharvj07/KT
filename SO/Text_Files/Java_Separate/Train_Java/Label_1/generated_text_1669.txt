@Data
class Human{
    private long id;

    @ToString.Exclude
    @EqualsAndHashCode.Exclude
    private Pet pet;

    @ToString.Include
    public long petId() { return pet.getId(); }

    //other properties...
}

@Data
class Pet{
    private long id;

    @ToString.Exclude
    @EqualsAndHashCode.Exclude
    private Human owner;

    @ToString.Include
    public long ownerId() { return owner.getId(); }
}
