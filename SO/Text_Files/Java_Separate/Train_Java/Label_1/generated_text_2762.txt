public interface NullSafeMap<K,V> extends Map<K,V>
{
    public default <A> void getAndDo( final K key, final A argument, final BiConsumer<V,A> process )
    {
        final var value = get( key );
        if( nonNull( value ) ) process.accept( value, argument );
    }

    public default Optional<V> optionalGet( final K key )
    {
        return Optional.ofNullable( get( key ) );
    }

    public default <A> Optional<V> optionalGetAndDo( final K key, final A argument, final BiConsumer<V,A> process )
    {
        final var value = optionalGet( key );
        value.ifPresent( v -> process.accept( v, argument ) );
        return value;
    }

    //---* The method you suggested *-------------------------------
    public default void getAndUse( final K key, final Consumer<V> process )
    {
        final var value = get( key );
        if( nonNull( value ) ) process.accept( value );
    }
}

public class NullSafeHashMap<K,V> extends HashMap<K,V> implements NullSafeMap<K,V>
{
    public NullSafeHashMap() { super(); }
}
