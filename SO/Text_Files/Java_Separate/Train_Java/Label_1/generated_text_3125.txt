public static String getGroups(String[][] matrix, int groupSize) {
    String output = "";
    int matrixWidth = matrix.length;
    int matrixHeight = matrix[0].length;
    
    int k = 0;
    for (int i = 0; i < matrixHeight; i++) {
        for (int j = 0; j < matrixWidth; j++) {
            k++;
            output += matrix[j][i];
            if(k % groupSize == 0) output += " ";
        }
    }
    
    return output;
}
