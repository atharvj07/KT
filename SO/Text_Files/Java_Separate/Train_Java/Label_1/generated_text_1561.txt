class PointAndColor { 
     Point thePoint;
     Color theColor;
     PointAndColor () { 
         theColor = Color.black;
         thePoint = new Point (0,0);            
     }
        
     PointAndColor (Point point, Color color) { 
          theColor = color;
          thePoint = point;
     }
        
     Color getColor () { return theColor; }
     Point getPoint () { return thePoint; }
 }
