public void PrintLL() {
        Node n = head;
        while(n!=null){
            System.out.print(n.data+ " --> ");
            n= n.next;
        }
        System.out.println("NULL");
    }
    
    public void RemoveNthNode(int nth){
        if(head == null || head.next==null){
            head = null;
            return;
        }
        int size=0;
        Node curNode=head;
        while(curNode!=null) {
            curNode = curNode.next;
            size++;
        }

        if(nth==size){
            head = head.next;
            return;
        }
         
        Node prevnode = head;
        int i=1;
        while(i<size-nth){
            prevnode=prevnode.next;
            i++;
        }
        prevnode.next = prevnode.next.next;
    }
    
    public static void main(String[] args) {
        Main ll = new Main();

        ll.addFirst(45);
        
        ll.PrintLL();
      
        ll.RemoveNthNode(3);
        ll.PrintLL();
    }
