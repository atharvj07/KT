public static int pairsum(int arr[], int sum,int j){
    int count=0;
    int k = arr.length - 1;
    while (j &lt; k) {
        int nj = 1; // Same values at lower bound.
        while (j &lt; k &amp;&amp; str[j + nj] == str[j]) {
            ++nj;
        }
        int nk = 1; // Same values at upper bound.
        while (j &lt; k &amp;&amp; str[k - nj] == str[k]) {
            ++nk;
        }
        
        if (arr[j] + arr[k] == sum) {
            if (arr[j] == arr[k]) { // Or k - j + 1 == nj
                count++; // binom(k - jÂ´+ 1, 2)! For 1 - 4: 12, 13, 14; 23, 24; 34
                j = k; // Or break
            } else {
                count++; // nj, nk
                j = k;
            }
        }
        else if (arr[j] + arr[k] &gt; sum){
            k -= nk;
        }
        else {
            j += nj;
            //k = arr.length - 1;//occurence of error
        }           
    }
    return count;
}

/**
 * @return the number of triplets summing up to num.
 */
public static int tripletSum(int[] arr, int num) {
    Arrays.sort(arr);
    int count = 0;
    for (int i = 0; i &lt; arr.length - 2; i++) {
        int sum = num - arr[i];
        count += pairsum(arr, sum, i+1);
    }
    return count;
}
