import Data.Bits
import Data.List (tails)
import Data.Maybe (listToMaybe)
import Control.Monad (guard)

cookedWords :: [Int]
cookedWords = [0..]

match :: [Int] -&gt; Maybe (Int, Int, Int, Int, Int)
match words =
  listToMaybe $ do
    a : as &lt;- tails cookedWords
    b : bs &lt;- tails as
    guard $ a .&amp;. b == 0
    let ab = a .|. b
    c : cs &lt;- tails bs
    guard $ ab .&amp;. c == 0
    let abc = ab .|. c
    d : ds &lt;- tails cs
    guard $ abc .&amp;. d == 0
    let abcd = abc .|. d
    e : es &lt;- tails ds
    guard $ abcd .&amp;. e == 0
    pure (a, b, c, d, e)
