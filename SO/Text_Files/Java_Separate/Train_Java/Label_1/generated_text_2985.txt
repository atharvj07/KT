public class AlteredRequestHandler extends Handler.Wrapper
{
    @Override
    public boolean handle(Request request, Response response, Callback callback) throws Exception
    {
        HttpURI alteredHttpURI = HttpURI.build(request.getHttpURI())
            .query("a=b");
        String alteredMethod = "COPY";
        
        Request alteredRequest = new Request.Wrapper(request)
        {
            @Override
            public HttpURI getHttpURI()
            {
                return alteredHttpURI;
            }

            @Override
            public String getMethod()
            {
                return alteredMethod;
            }
        };
        
        return super.handle(alteredRequest, response, callback);
    }
}
