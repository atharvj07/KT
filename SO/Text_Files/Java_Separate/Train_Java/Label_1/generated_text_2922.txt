class apples{
 public static void main(String[] args) {
  int bucky[]={3,4,5,6,7};

  bucky = change(bucky);
  for(int b:bucky) {
        System.out.println(b);
    }
  }

  public static void change(int x[]){
    for (int i = 0; i < x.length; i++) {
        x[i]=x[i]+5;
    }
    return x;
  }
}
