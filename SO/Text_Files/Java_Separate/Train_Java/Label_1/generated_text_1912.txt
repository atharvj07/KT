List&lt;String&gt; listA = new ArrayList&lt;&gt;(List.of(&quot;us&quot;,&quot;au&quot;,&quot;in&quot;,&quot;gb&quot;));
List&lt;String&gt; listB = new ArrayList&lt;&gt;(List.of(&quot;au&quot;,&quot;us&quot;,&quot;in&quot;));
        
Map&lt;String, Integer&gt; order = IntStream.range(0, listB.size())
    .boxed()
    .collect(Collectors.toMap(

        Function.identity()
    ));
