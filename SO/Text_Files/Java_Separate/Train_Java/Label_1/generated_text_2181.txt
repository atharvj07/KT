public static int getOccurrences(String s) {
    String lower = s.toLowerCase();
    int occurrences = 0;
    
    for (int i = 0; i &lt; lower.length() - 3; i++) {
        if (lower.charAt(i) == 'l' &amp;&amp;
            lower.charAt(i + 2) == 'n' &amp;&amp;
            lower.charAt(i + 3) == 'e') {

            occurrences++; // incrementing the number of occurrences
            i += 3;        // advancing the index
        }
    }
    return occurrences;
}
