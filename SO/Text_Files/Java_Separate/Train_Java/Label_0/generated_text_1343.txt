import org.yaml.snakeyaml.nodes.NodeTuple;
import org.yaml.snakeyaml.DumperOptions;
import org.yaml.snakeyaml.Yaml;

import java.util.List;

public class MyYamlRepresenter extends Representer {

    public MyYamlRepresenter() {
        super();
        this.representers.put(Norwich.class, new NorwichRepresenter());
    }

    private class NorwichRepresenter implements Represent {
        @Override
        public Node representData(Object o) {
            Norwich norwich = (Norwich) o;
            // Check if the "disabled" flag is true
            if (norwich.getDisabled() != null && norwich.getDisabled()) {
                // Skip this object by returning a null representation
                return nullRepresenter.representData((Object) null);
            }
            // Otherwise, represent the Norwich object as a JavaBean
            return representJavaBean(getProperties(Norwich.class), norwich);
        }
    }

    @Override
    protected NodeTuple representJavaBeanProperty(Object javaBean, Property property, Object propertyValue, Tag customTag) {
        if (javaBean == null || propertyValue == null) {
            return null;  // Skip null values
        }

        // Skip the "disabled" property from serialization
        if (property.getName().equals("disabled")) {
            return null;
        }

        // If the property is a List, we need to filter out disabled children
        if (propertyValue instanceof List<?>) {
            List<?> list = (List<?>) propertyValue;
            list.removeIf(item -> item instanceof Norwich && ((Norwich) item).getDisabled() != null && ((Norwich) item).getDisabled());
        }

        return super.representJavaBeanProperty(javaBean, property, propertyValue, customTag);
    }
}
