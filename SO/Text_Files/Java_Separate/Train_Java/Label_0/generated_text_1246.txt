import javafx.application.Application;
import javafx.scene.Group;
import javafx.scene.PerspectiveCamera;
import javafx.scene.Scene;
import javafx.scene.paint.Color;
import javafx.scene.paint.PhongMaterial;
import javafx.scene.shape.MeshView;
import javafx.scene.shape.TriangleMesh;
import javafx.scene.transform.Rotate;
import javafx.stage.Stage;
import javafx.scene.input.KeyEvent;
import javafx.scene.input.KeyCode;
import javafx.scene.layout.Pane;
import javafx.scene.shape.Sphere;

public class ModelLoader extends Application {

    @Override
    public void start(Stage primaryStage) {
        // Load the 3D model
        MeshView model = loadModel("model.obj");

        // Apply correction transformation for Blender's coordinate system
        model.getTransforms().add(new Rotate(90, Rotate.X_AXIS)); // Fix orientation

        // Set up the scene
        Group root = new Group(model);
        Scene scene = new Scene(root, 800, 600, true);
        scene.setFill(Color.LIGHTGRAY);
        
        // Camera setup
        PerspectiveCamera camera = new PerspectiveCamera(true);
        camera.setTranslateZ(-500);
        scene.setCamera(camera);

        primaryStage.setTitle("Blender to JavaFX 3D");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    private MeshView loadModel(String filename) {

        meshView.setMaterial(new PhongMaterial(Color.BLUE));

    }

    public static void main(String[] args) {
        launch(args);
    }
}
