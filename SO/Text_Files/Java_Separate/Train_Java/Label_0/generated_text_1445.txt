import org.objectweb.asm.ClassReader;
import org.objectweb.asm.ClassVisitor;
import org.objectweb.asm.Opcodes;
import org.objectweb.asm.util.CheckClassAdapter;
import java.io.IOException;

public class ASMVerifier {

    public static void verifyBytecode(byte[] bytes) throws IOException {
        ClassReader classReader = new ClassReader(bytes);
        
        try {
            classReader.accept(new CheckClassAdapter(new ClassVisitor(Opcodes.ASM9) {
                @Override
                public void visitEnd() {
                    super.visitEnd();
                    throw new IllegalStateException("Bytecode verification failed.");
                }
