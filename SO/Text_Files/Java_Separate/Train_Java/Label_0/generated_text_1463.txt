import org.testng.annotations.Test;

public class MainImpClass {

    int a;

    public MainImpClass(int a) {
        this.a = a;
    }

    public int getA() {
        return a;
    }

    @Test(priority = 1)
    public void getValue1() {
        System.out.println("Value from getValue1: " + a);
    }

    @Test(priority = 2)
    public void getValue2() {
        System.out.println("Value from getValue2: " + a);
    }

    @Test(priority = 3)
    public void getValue3() {
        System.out.println("Value from getValue3: " + a);
    }
}

import org.testng.IMethodInstance;
import org.testng.IMethodInterceptor;
import org.testng.ITestContext;

import java.util.*;

public class CustomMethodInterceptor implements IMethodInterceptor {

    @Override
    public List<IMethodInstance> intercept(List<IMethodInstance> methods, ITestContext context) {
        // TreeMap to store test methods in ascending order based on 'a' value
        TreeMap<Integer, List<IMethodInstance>> sortedMethods = new TreeMap<>();

        for (IMethodInstance methodInstance : methods) {
            MainImpClass testInstance = (MainImpClass) methodInstance.getInstance();
            sortedMethods.computeIfAbsent(testInstance.getA(), k -> new ArrayList<>()).add(methodInstance);
        }

        // Flatten the sorted TreeMap into an ordered list
        List<IMethodInstance> orderedMethods = new ArrayList<>();
        for (List<IMethodInstance> methodList : sortedMethods.values()) {
            orderedMethods.addAll(methodList);
        }

        return orderedMethods; // Return ordered test execution
    }
}
