public class ItemChecker {
    private String userInput;
    private String validity;

    // Setter for user input
    public void setUserInput(String userInput) {
        this.userInput = userInput;
    }

    // Compute validity
    public void computeValidity() {
        if (userInput.length() != 13) {  // Ensure the length is exactly 13
            validity = "Invalid: Incorrect length.";
            return;
        }

        // Check first three characters (digits)
        for (int i = 0; i < 3; i++) {
            if (!Character.isDigit(userInput.charAt(i))) {
                validity = "Invalid: First three characters must be digits.";
                return;
            }
        }

        // Check fourth and seventh characters (dashes)
        if (userInput.charAt(3) != '-' || userInput.charAt(6) != '-') {
            validity = "Invalid: Dashes must be at positions 4 and 7.";
            return;
        }

        // Check fifth and sixth characters (letters)
        if (!Character.isLetter(userInput.charAt(4)) || !Character.isLetter(userInput.charAt(5))) {
            validity = "Invalid: Fifth and sixth characters must be letters.";
            return;
        }

        // Check last six characters (digits)
        for (int i = 7; i < 13; i++) {
            if (!Character.isDigit(userInput.charAt(i))) {
                validity = "Invalid: Last six characters must be digits.";
                return;
            }
        }

        // If all checks pass
        validity = "Valid Item.";
    }

    // Getter for validity result
    public String getValidity() {
        return validity;
    }

    // Main method for testing
    public static void main(String[] args) {
        ItemChecker checker = new ItemChecker();
        
        checker.setUserInput("222-DN-105555"); // Change this input to test
        checker.computeValidity();
        
        System.out.println(checker.getValidity());
    }
}
