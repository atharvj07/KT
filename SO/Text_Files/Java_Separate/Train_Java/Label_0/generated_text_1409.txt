public interface Translator<T> {
    String translate(T object);
}

public class ArgumentTranslator implements Translator<Argument> {
    @Override
    public String translate(Argument argument) {
        // Implement translation logic for Argument
        return "Translated Argument";
    }
}

public class TypeATranslator implements Translator<TypeA> {
    @Override
    public String translate(TypeA object) {
        // Implement translation logic for TypeA
        return "Translated TypeA";
    }
}

public class TranslatorFactory {
    public static <T> Translator<T> getTranslator(Class<T> clazz) {
        if (clazz == Argument.class) {
            return (Translator<T>) new ArgumentTranslator();
        } else if (clazz == TypeA.class) {
            return (Translator<T>) new TypeATranslator();
        }
        // Add more conditions as needed
        throw new IllegalArgumentException("No translator found for " + clazz.getName());
    }
}

protected <T> String translate(final List<T> list) {
    if (list == null || list.isEmpty()) {
        return "";
    }

    return list.stream()
            .map(item -> TranslatorFactory.getTranslator(item.getClass()).translate(item))
            .collect(Collectors.joining(", "));
}
