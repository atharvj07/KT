public class TestPanel extends ScrollPane {

    final int SPACING = 5;
    final int ROW_MAX = 6;

    public TestPanel(ArrayList<Item> items) {

        VBox root = new VBox(); 
        root.setSpacing(SPACING);
        HBox row = null;

        int count = 0;
        for (Item item: items) {
            if (count == ROW_MAX || row == null) {
                row = new HBox();
                row.setSpacing(SPACING);
                root.getChildren().add(row);
                count = 0;
            }

            CustomBox box = new CustomBox(item);
            row.getChildren().add(box);

            // Make sure the box can grow within the HBox
            HBox.setHgrow(box, Priority.ALWAYS);

            count++;
        }

        setFitToWidth(true);
        setContent(root);
    }
}

public class CustomBox extends StackPane {

    private Item item;
    private Rectangle square;

    private int size = 20; // Irrelevant for resizing

    public CustomBox(Item item) {
        this.item = item;

        setPadding(new Insets(5, 5, 5, 5));

        // Initialize the rectangle
        square = new Rectangle();

        // Bind both width and height to the StackPane's width and height
        square.widthProperty().bind(widthProperty());
        square.heightProperty().bind(heightProperty());

        // Maintain the aspect ratio if needed, here it's square so width == height
        minHeightProperty().bind(widthProperty()); // Ensures it stays square

        // Add the rectangle to the StackPane
        getChildren().add(square);
    }
}
