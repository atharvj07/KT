import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.time.format.DateTimeParseException;
import java.util.Locale;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        // Use Locale.ENGLISH to make month parsing case-insensitive
        DateTimeFormatter format1 = DateTimeFormatter.ofPattern("MMM/dd/yyyy", Locale.ENGLISH);

        System.out.print("Please enter the first date: ");
        String input = sc.nextLine();
        
        try {
            LocalDate firstDate = LocalDate.parse(input, format1);
            System.out.println("Parsed date: " + firstDate);
        } catch (DateTimeParseException e) {
            System.out.println("Invalid date format! Please use 'MMM/dd/yyyy' (e.g., Dec/12/1999).");
        }

        sc.close();
    }
}
