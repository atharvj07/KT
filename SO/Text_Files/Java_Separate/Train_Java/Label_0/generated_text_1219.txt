import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.ScrollPane;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.Pane;
import javafx.stage.Stage;

public class ScrollEventIssueApplication extends Application {
    private boolean isFocused = true;

    @Override
    public void start(Stage primaryStage) {
        BorderPane borderPane = new BorderPane();
        borderPane.setPrefWidth(600);
        borderPane.setPrefHeight(600);

        Pane content = new Pane();
        content.setPrefWidth(1000);
        content.setPrefHeight(1800);
        ScrollPane scrollPane = new ScrollPane(content);

        scrollPane.setPrefWidth(700);
        scrollPane.setPrefHeight(700);

        content.setOnScroll(event -> {
            if (isFocused) { // Only handle scroll when window is focused
                System.out.println("Scroll event received: " + event.getDeltaY());
            } else {
                System.out.println("Ignoring stale scroll event...");
            }
        });

        primaryStage.focusedProperty().addListener((obs, oldVal, newVal) -> {
            isFocused = newVal;
        });

        borderPane.setCenter(scrollPane);

        Scene scene = new Scene(borderPane, 1800, 900);
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}

import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.ScrollPane;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.Pane;
import javafx.stage.Stage;

import java.util.Timer;
import java.util.TimerTask;

public class ScrollEventIssueApplication extends Application {
    private volatile boolean ignoreScroll = false;

    @Override
    public void start(Stage primaryStage) {
        BorderPane borderPane = new BorderPane();
        borderPane.setPrefWidth(600);
        borderPane.setPrefHeight(600);

        Pane content = new Pane();
        content.setPrefWidth(1000);
        content.setPrefHeight(1800);
        ScrollPane scrollPane = new ScrollPane(content);

        scrollPane.setPrefWidth(700);
        scrollPane.setPrefHeight(700);

        content.setOnScroll(event -> {
            if (!ignoreScroll) {
                System.out.println("Scroll event received: " + event.getDeltaY());
            } else {
                System.out.println("Ignoring stale scroll event...");
            }
        });

        primaryStage.focusedProperty().addListener((obs, oldVal, newVal) -> {
            if (newVal) {
                ignoreScroll = true;
                new Timer().schedule(new TimerTask() {
                    @Override
                    public void run() {
                        ignoreScroll = false;
                    }
                }, 200); // Ignore scroll events for 200ms after focus
            }
        });

        borderPane.setCenter(scrollPane);

        Scene scene = new Scene(borderPane, 1800, 900);
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
