private Observable<Boolean> applyAdaptersRecursively(boolean isConnected, 
                                                     List<Function<Boolean, Observable<Boolean>>> adaptersList, 
                                                     int index) {
    if (index >= adaptersList.size()) {
        return Observable.just(isConnected); // No more fallbacks, return final result
    }

    return adaptersList.get(index).apply(isConnected)
            .concatMap(nextIsConnected -> 
                applyAdaptersRecursively(nextIsConnected, adaptersList, index + 1));
}
