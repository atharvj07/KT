import reactor.core.publisher.Flux;
import reactor.core.publisher.Mono;

public class FluxExample {
    public static void main(String[] args) {
        Flux<Integer> fluxA = Flux.just(1, 2, 3, 4, 5);

        Flux<Integer> fluxC = fluxA.concatWith(
            fluxA.last()
                .flatMapMany(FluxExample::createFluxB)
        );

        fluxC.subscribe(System.out::println);
    }

    private static Flux<Integer> createFluxB(int lastA) {
        return Flux.just(lastA * 10, lastA * 20, lastA * 30);
    }
}
