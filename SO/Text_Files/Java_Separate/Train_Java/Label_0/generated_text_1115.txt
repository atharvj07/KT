import com.maxmind.geoip2.DatabaseReader;
import com.maxmind.geoip2.exception.GeoIp2Exception;
import com.maxmind.geoip2.model.CityResponse;
import com.maxmind.geoip2.record.Country;
import java.io.File;
import java.io.IOException;
import java.io.InputStream;
import java.net.InetAddress;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;

class MyUDFHandler {
    public static String isoCode(String ip_address) throws Exception {
        // Use the staged location in AWS_CSV_STAGE directly
        String databasePath = "/mnt/AWS_CSV_STAGE/geodata/GeoLite2-City.mmdb"; // Specify the full path here

        // Create a File object for the database file
        File file = new File(databasePath);
        if (!file.exists()) {
            throw new IOException("GeoLite2-City.mmdb file not found at: " + databasePath);
        }

        // Use DatabaseReader to load the file
        DatabaseReader reader = new DatabaseReader.Builder(file).build();

        // Get the city response based on the IP address
        CityResponse response = reader.city(InetAddress.getByName(ip_address));

        // Return the ISO code of the country
        return response.getCountry().getIsoCode();
    }
}
