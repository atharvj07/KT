public class LetterMatrix {
    public static void main(String[] args) {
        String str = "i love my mother very big and love car";

        // Step 1: Remove spaces
        str = str.replaceAll(" ", "");

        // Step 2: Determine number of columns (groups of 5)
        int groupSize = 5;
        int numColumns = (int) Math.ceil((double) str.length() / groupSize);

        // Step 3: Create the matrix (2 rows, numColumns columns)
        String[][] matrix = new String[2][numColumns];

        // Step 4: Fill the matrix
        int index = 0;
        for (int col = 0; col < numColumns; col++) {
            for (int row = 0; row < 2; row++) {
                if (index < str.length()) {
                    matrix[row][col] = String.valueOf(str.charAt(index));
                    index++;
                } else {
                    matrix[row][col] = ""; // Fill with empty if needed
                }
            }
        }

        // Step 5: Read the matrix column-wise and build the output
        StringBuilder result = new StringBuilder();
        for (int col = 0; col < numColumns; col++) {
            for (int row = 0; row < 2; row++) {
                result.append(matrix[row][col]);
            }
        }

        // Print the transformed string
        System.out.println("Transformed string: " + result.toString());
    }
}
