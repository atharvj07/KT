class Spaceship extends Actor {
    Sprite sprite;

    public Spaceship(Texture texture) {
        sprite = new Sprite(texture);
        sprite.setPosition(getX(), getY());
    }

    @Override
    public void draw(Batch batch, float parentAlpha) {
        sprite.setPosition(getX(), getY());  // Ensure sprite position matches actor position
        sprite.draw(batch);
    }

    @Override
    public void act(float delta) {
        super.act(delta);
        // Update the actor's position if necessary
        setPosition(sprite.getX(), sprite.getY());
    }

    // Update position for spaceship movement logic
    public void moveBy(float x, float y) {
        sprite.translate(x, y);
        setPosition(sprite.getX(), sprite.getY()); // Keep actor's position in sync with sprite
    }
}

class SpaceshipLayer extends Group {
    Spaceship spaceship;
    float screenWidth;

    public SpaceshipLayer(Texture spaceshipTexture, float screenWidth) {
        spaceship = new Spaceship(spaceshipTexture);
        this.screenWidth = screenWidth;
        addActor(spaceship); // Add spaceship to group
    }

    @Override
    public void draw(Batch batch, float parentAlpha) {
        if (isRightKeyPressed) {
            spaceship.moveBy(5, 0); // Move spaceship right
            if (spaceship.getX() > screenWidth) {
                // When spaceship moves off screen, reset its position to the left side
                spaceship.setX(-spaceship.sprite.getWidth());  // Make sure spaceship fully moves off the right side
            }
        }
        super.draw(batch, parentAlpha);
    }
}

class MainScreen implements Screen {
    SpaceshipLayer spaceshipLayer;
    Stage stage;
    float screenWidth;

    public MainScreen() {
        stage = new Stage();
        screenWidth = Gdx.graphics.getWidth();  // Get the screen width dynamically
        spaceshipLayer = new SpaceshipLayer(new Texture("spaceship.png"), screenWidth);
        stage.addActor(spaceshipLayer);
    }

    @Override
    public void render(float delta) {
        Gdx.gl.glClear(GL20.GL_COLOR_BUFFER_BIT);
        stage.act(Math.min(Gdx.graphics.getDeltaTime(), 1 / 30f));
        stage.draw();
    }

    @Override
    public void resize(int width, int height) {
        stage.getViewport().update(width, height, true);
    }

    @Override
    public void hide() {}

    @Override
    public void show() {}

    @Override
    public void pause() {}

    @Override
    public void resume() {}
}
