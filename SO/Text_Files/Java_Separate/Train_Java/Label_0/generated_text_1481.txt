import java.awt.*;
import java.awt.event.MouseAdapter;
import java.awt.event.MouseEvent;
import java.util.ArrayList;
import javax.swing.*;

public class SandSimulation {
    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            MainFrame frame = new MainFrame();
            frame.init();
        });
    }
}

// Main JFrame
class MainFrame extends JFrame {
    private SandPanel sandPanel;

    public void init() {
        setTitle("Sandsim");
        setSize(600, 600);
        setDefaultCloseOperation(WindowConstants.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);

        sandPanel = new SandPanel();
        setContentPane(sandPanel);

        setVisible(true);
    }
}

// Custom JPanel to draw grid and sand
class SandPanel extends JPanel {
    private final int GRID_SIZE = 10;
    private final ArrayList<Point> sandParticles = new ArrayList<>();

    public SandPanel() {
        setBackground(Color.WHITE);

        // Mouse listener to drop sand at mouse position
        addMouseListener(new MouseAdapter() {
            @Override
            public void mousePressed(MouseEvent e) {
                int x = (e.getX() / GRID_SIZE) * GRID_SIZE; // Snap to grid
                int y = (e.getY() / GRID_SIZE) * GRID_SIZE;
                sandParticles.add(new Point(x, y));
                repaint();
            }
        });
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        drawGrid(g);
        drawSand(g);
    }

    private void drawGrid(Graphics g) {
        g.setColor(Color.LIGHT_GRAY);
        for (int x = 0; x < getWidth(); x += GRID_SIZE) {
            g.drawLine(x, 0, x, getHeight());
        }
        for (int y = 0; y < getHeight(); y += GRID_SIZE) {
            g.drawLine(0, y, getWidth(), y);
        }
    }

    private void drawSand(Graphics g) {
        g.setColor(Color.YELLOW);
        for (Point p : sandParticles) {
            g.fillRect(p.x, p.y, GRID_SIZE, GRID_SIZE);
        }
    }
}

public static void main(String[] args) {
    SwingUtilities.invokeLater(() -> {
        MainFrame frame = new MainFrame();
        frame.init();
    });
}
