@Override
protected void paintComponent(Graphics g) {
    super.paintComponent(g);
    Graphics2D g2d = (Graphics2D) g.create();
    
    if (zoomer) {
        // Set up the AffineTransform for scaling and translating
        AffineTransform at = new AffineTransform();

        Dimension center = getPreferredScrollableViewportSize();
        double xRel = center.getWidth() / 2;
        double yRel = center.getHeight() / 2;

        double zoomDiv = zoomFactor / prevZoomFactor;

        // Adjust offsets for zooming to ensure content remains centered
        xOffset = (zoomDiv) * (xOffset) + (1 - zoomDiv) * xRel;
        yOffset = (zoomDiv) * (yOffset) + (1 - zoomDiv) * yRel;

        at.translate(xOffset, yOffset);
        at.scale(zoomFactor, zoomFactor);
        prevZoomFactor = zoomFactor;

        g2d.transform(at);
        zoomer = false;
    }

    // Draw the string (which will be scaled based on zoom)
    g2d.drawString("1234567890", 500, 200);

    g2d.dispose();
}

@Override
public Dimension getPreferredSize() {
    // Dynamically adjust the preferred size based on the zoom factor
    return new Dimension((int) (2000 * zoomFactor), (int) (750 * zoomFactor));
}

@Override
public Dimension getPreferredScrollableViewportSize() {
    return new Dimension(1550, 750);
}

@Override
public void actionPerformed(ActionEvent e) {
    if (e.getSource() == in) {
        zoomer = true;
        zoomFactor *= 1.1;
        revalidate(); // Recalculate the layout after zooming
        repaint();
    }

    if (e.getSource() == out) {
        zoomer = true;
        zoomFactor /= 1.1;
        revalidate(); // Recalculate the layout after zooming
        repaint();
    }
}
