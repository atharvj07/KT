import org.simpleframework.xml.Attribute;

import org.simpleframework.xml.Path;

@Root(name = "note")
public class Note {
    
    @Element(name = "to")
    private String to;

    @Element(name = "from")
    private Sender from;

    @Element(name = "heading")
    private String heading;

    @Element(name = "body")
    private String body;

    // Getters and Setters (if needed)
}

@Root(name = "from")
public class Sender {

    @Attribute(name = "name", required = false)  // Attribute in <from>
    private String attribute;

    @Path("from")  // Path to the text content inside <from>
    @Element(name = "from", required = false) 
    private String content;

    // Getters and Setters (if needed)
}

import org.simpleframework.xml.transform.Transform;

public class SenderConverter implements Transform<Sender> {
    @Override
    public Sender read(String value) {
        Sender sender = new Sender();
        sender.setContent(value);
        return sender;
    }

    @Override
    public String write(Sender sender) {
        return sender.getContent();
    }
}

import org.simpleframework.xml.Convert;
