import java.util.*;

public class Main {
    public static void main(String[] args) {
        // Create a List of Lists
        List<List<Integer>> list = new ArrayList<>();
        list.add(Arrays.asList(5, 10));
        list.add(Arrays.asList(2, 11));
        list.add(Arrays.asList(1, 12));
        list.add(Arrays.asList(8, 13));

        // Sorting in Ascending Order (Based on the first element)
        Collections.sort(list, Comparator.comparingInt(innerList -> innerList.get(0)));

        // Print the sorted list
        System.out.println("Ascending order:");
        for (List<Integer> sublist : list) {
            System.out.println(sublist);
        }

        // Sorting in Descending Order
        Collections.sort(list, (a, b) -> Integer.compare(b.get(0), a.get(0)));

        // Print the sorted list
        System.out.println("\nDescending order:");
        for (List<Integer> sublist : list) {
            System.out.println(sublist);
        }
    }
}
