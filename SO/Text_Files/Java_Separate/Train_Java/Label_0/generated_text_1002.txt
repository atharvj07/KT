import java.lang.annotation.*;
import java.lang.reflect.*;

@Retention(RetentionPolicy.RUNTIME)
@interface SomeAnnotation {
    String value();
}

interface Foo {
    @SomeAnnotation("Hello from Foo!")
    String foo();
}

class DynamicProxyHandler implements InvocationHandler {
    private final Class<?> interfaceClass;

    public DynamicProxyHandler(Class<?> interfaceClass) {
        this.interfaceClass = interfaceClass;
    }

    @Override
    public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
        // Get annotation from method
        SomeAnnotation annotation = method.getAnnotation(SomeAnnotation.class);
        if (annotation != null) {
            return "Intercepted: " + annotation.value();
        }
        return null;
    }
}

public class DynamicProxyExample {
    public static void main(String[] args) {
        Foo foo = (Foo) Proxy.newProxyInstance(
            Foo.class.getClassLoader(),
            new Class[]{Foo.class},
            new DynamicProxyHandler(Foo.class)
        );

        System.out.println(foo.foo());  // Output: Intercepted: Hello from Foo!
    }
}

import net.bytebuddy.ByteBuddy;
import net.bytebuddy.implementation.bind.annotation.Origin;
import net.bytebuddy.implementation.bind.annotation.RuntimeType;
import net.bytebuddy.implementation.bind.annotation.SuperCall;
import net.bytebuddy.matcher.ElementMatchers;

import java.lang.annotation.*;
import java.lang.reflect.Method;
import java.util.concurrent.Callable;

@Retention(RetentionPolicy.RUNTIME)
@interface SomeAnnotation {
    String value();
}

abstract class Foo {
    @SomeAnnotation("Hello from Foo!")
    abstract String foo();
}

public class ByteBuddyExample {
    public static void main(String[] args) throws Exception {
        Foo foo = new ByteBuddy()
            .subclass(Foo.class)
            .method(ElementMatchers.isAnnotatedWith(SomeAnnotation.class))
            .intercept(net.bytebuddy.implementation.MethodDelegation.to(new Interceptor()))
            .make()
            .load(ByteBuddyExample.class.getClassLoader())
            .getLoaded()
            .getDeclaredConstructor()
            .newInstance();

        System.out.println(foo.foo());  // Output: Intercepted: Hello from Foo!
    }

    public static class Interceptor {
        @RuntimeType
        public static Object intercept(@Origin Method method, @SuperCall Callable<?> callable) throws Exception {
            SomeAnnotation annotation = method.getAnnotation(SomeAnnotation.class);
            if (annotation != null) {
                return "Intercepted: " + annotation.value();
            }
            return callable != null ? callable.call() : null;
        }
    }
}
