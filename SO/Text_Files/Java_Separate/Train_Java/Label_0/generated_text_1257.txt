import com.sun.jna.platform.win32.Advapi32;
import com.sun.jna.platform.win32.Advapi32Util;
import com.sun.jna.platform.win32.WinNT;
import com.sun.jna.platform.win32.WinBase;
import com.sun.jna.ptr.PointerByReference;

public class WindowsEventLogger {
    public static void main(String[] args) {
        String sourceName = "MyJavaApp"; // Event source name
        String logMessage = "This is a test event log entry from Java!";

        // Register the event source
        WinNT.HANDLE eventLogHandle = Advapi32.INSTANCE.RegisterEventSource(null, sourceName);
        if (eventLogHandle == null) {
            System.err.println("Failed to register event source. Error: " + Advapi32.INSTANCE.GetLastError());
            return;
        }

        // Event type (EVENTLOG_INFORMATION_TYPE = 0x0004)
        int eventType = WinNT.EVENTLOG_INFORMATION_TYPE;
        int eventCategory = 0; // Category (unused)
        int eventID = 1000; // Custom event ID
        String[] messages = { logMessage };

        // Report the event
        boolean success = Advapi32.INSTANCE.ReportEvent(eventLogHandle, eventType, eventCategory, eventID, null, messages.length, 0, messages, null);
        if (!success) {
            System.err.println("Failed to write event log. Error: " + Advapi32.INSTANCE.GetLastError());
        } else {
            System.out.println("Event logged successfully.");
        }

        // Close the event source handle
        Advapi32.INSTANCE.DeregisterEventSource(eventLogHandle);
    }
}
