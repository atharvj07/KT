import java.util.*;

public class Solution {
    public static int solve(List<Integer> A, List<Integer> B) {
        int n = A.size();
        final int MOD = 1_000_000_007;

        // Step 1: Pair the elements of A and B and sort based on A[i]
        List<int[]> pairs = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            pairs.add(new int[]{A.get(i), B.get(i)});
        }
        pairs.sort((x, y) -> Integer.compare(x[0], y[0]));

        // Step 2: Process the sorted pairs
        long max = 0;
        long sumB = 0;

        for (int i = 0; i < n; i++) {
            int currentA = pairs.get(i)[0];
            int currentB = pairs.get(i)[1];
            
            // Update the running sum of B[i]
            sumB = (sumB + currentB) % MOD;

            // Calculate min * sum for this subset with the current element as min
            long currentProduct = (long) currentA * sumB % MOD;

            // Update the max value
            max = Math.max(max, currentProduct);
        }

        return (int) max;
    }

    public static void main(String[] args) {
        List<Integer> A = Arrays.asList(1, 1, 3);
        List<Integer> B = Arrays.asList(1, 2, 2);
        
        System.out.println(solve(A, B)); // Expected output: 6
    }
}
