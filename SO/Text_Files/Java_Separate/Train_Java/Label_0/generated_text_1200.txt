import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        String file = "studentData.txt";

        // Read the student data into an ArrayList
        ArrayList<Student> users = readFile(file);

        Scanner input = new Scanner(System.in);

        // Ask user to enter an ID number
        System.out.println("Enter ID Number:");
        int idNum = input.nextInt();

        // Search for the student by ID and print their age
        boolean found = false;
        for (Student s : users) {
            if (s.getIdNum() == idNum) {
                System.out.println("Age: " + s.getAge());
                found = true;
                break; // Stop searching once we find a match
            }
        }

        if (!found) {
            System.out.println("Student with ID " + idNum + " not found.");
        }
    }

    // Method for reading the file and creating Student objects
    public static ArrayList<Student> readFile(String file) throws IOException {
        Scanner reader = new Scanner(new File(file));

        ArrayList<Student> list = new ArrayList<>(); // List to store Student objects

        while (reader.hasNext()) { // Read and split text in the file
            String lName = reader.next();
            int idNum = reader.nextInt();
            int age = reader.nextInt();

            Student student = new Student(lName, idNum, age);
            list.add(student);
        }

        reader.close(); // Close the file scanner
        return list;
    }
}
