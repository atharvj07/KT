import java.awt.event.KeyEvent;
import java.awt.event.KeyListener;
import javax.swing.JFrame;

public class KeyPressExample extends JFrame implements KeyListener {
    private boolean keyHeld = false; // Track key state

    public KeyPressExample() {
        this.setTitle("Key Press Example");
        this.setSize(300, 200);
        this.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        this.addKeyListener(this);
        this.setVisible(true);
    }

    @Override
    public void keyPressed(KeyEvent e) {
        if (!keyHeld) { // Prevent multiple triggers
            keyHeld = true;
            System.out.println("Key Pressed: " + KeyEvent.getKeyText(e.getKeyCode()));
        }
    }

    @Override
    public void keyReleased(KeyEvent e) {
        keyHeld = false; // Reset on key release
    }

    @Override
    public void keyTyped(KeyEvent e) {
        // Not needed
    }

    public static void main(String[] args) {
        new KeyPressExample();
    }
}

@Override
public void keyPressed(KeyEvent e) {
    if (e.isActionKey()) { // Prevent repeated key events
        System.out.println("Key Pressed: " + KeyEvent.getKeyText(e.getKeyCode()));
    }
}
