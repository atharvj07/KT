import javax.swing.*;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;

public class Cronometro {
    private int secondi;
    private int minuti;
    private int ore;
    private boolean running = false;
    private Timer timer;

    public Cronometro(JLabel s, JLabel m, JLabel h) {
        // Timer that updates every 1000ms (1 second)
        timer = new Timer(1000, new ActionListener() {
            @Override
            public void actionPerformed(ActionEvent e) {
                secondi++;
                if (secondi >= 60) {
                    secondi = 0;
                    minuti++;
                }
                if (minuti >= 60) {
                    minuti = 0;
                    ore++;
                }

                // Update UI on the Event Dispatch Thread
                SwingUtilities.invokeLater(() -> {
                    s.setText(String.format(" : %02d", secondi));
                    m.setText(String.format(" : %02d", minuti));
                    h.setText(String.format("%02d", ore));
                });
            }
        });
    }

    public void start() {
        if (!running) {
            timer.start();
            running = true;
        }
    }

    public void stop() {
        timer.stop();
        running = false;
    }

    public void reset() {
        timer.stop();
        running = false;
        secondi = 0;
        minuti = 0;
        ore = 0;

        // Update UI
        SwingUtilities.invokeLater(() -> {
            // Reset labels
        });
    }
}

import javax.swing.*;
import java.awt.*;

public class StopwatchApp {
    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            JFrame frame = new JFrame("Stopwatch");
            frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
            frame.setSize(300, 200);
            frame.setLayout(new GridLayout(2, 1));

            JPanel panel = new JPanel();
            JLabel h = new JLabel("00");
            JLabel m = new JLabel(" : 00");
            JLabel s = new JLabel(" : 00");

            JButton startButton = new JButton("Start");
            JButton stopButton = new JButton("Stop");
            JButton resetButton = new JButton("Reset");

            Cronometro stopwatch = new Cronometro(s, m, h);

            startButton.addActionListener(e -> stopwatch.start());
            stopButton.addActionListener(e -> stopwatch.stop());
            resetButton.addActionListener(e -> stopwatch.reset());

            panel.add(h);
            panel.add(m);
            panel.add(s);
            panel.add(startButton);
            panel.add(stopButton);
            panel.add(resetButton);

            frame.add(panel);
            frame.setVisible(true);
        });
    }
}
