import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.time.ZonedDateTime;
import java.time.ZoneOffset;

public class DateFormatter {
    public static void main(String[] args) {
        // Define the formatter for "Thu, 3 Nov 2022 06:00:00 +0100"
        String firstDateString = "Thu, 3 Nov 2022 06:00:00 +0100";
        DateTimeFormatter formatter1 = DateTimeFormatter.ofPattern("EEE, d MMM yyyy HH:mm:ss Z");

        // Parse the first date string with formatter1
        ZonedDateTime zonedDateTime1 = ZonedDateTime.parse(firstDateString, formatter1);

        // Format it to "DD.MM.YYYY"
        String formattedDate1 = zonedDateTime1.format(DateTimeFormatter.ofPattern("dd.MM.yyyy"));
        System.out.println("Formatted Date 1: " + formattedDate1);  // Output: "03.11.2022"

        // Define the formatter for "01.11.2022 20:00:00"
        String secondDateString = "01.11.2022 20:00:00";
        DateTimeFormatter formatter2 = DateTimeFormatter.ofPattern("dd.MM.yyyy HH:mm:ss");

        // Parse the second date string with formatter2
        LocalDateTime localDateTime2 = LocalDateTime.parse(secondDateString, formatter2);

        // Format it to "DD.MM.YYYY"
        String formattedDate2 = localDateTime2.format(DateTimeFormatter.ofPattern("dd.MM.yyyy"));
        System.out.println("Formatted Date 2: " + formattedDate2);  // Output: "01.11.2022"
    }
}
