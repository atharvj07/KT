import java.util.*;
import java.util.stream.Collectors;

public class PrefixGrouping {
    public static Map<String, List<String>> groupByPrefix(Map<String, List<String>> input) {
        return input.entrySet().stream()
            .collect(Collectors.groupingBy(
                entry -> entry.getKey().split("\\.")[0], // Extract the prefix from the key
                Collectors.flatMapping(entry -> entry.getValue().stream(), Collectors.toList()) // Flatten and collect all lists
            ));
    }

    public static void main(String[] args) {
        // Example input map
        Map<String, List<String>> input = new HashMap<>();
        input.put("a.01", Arrays.asList("l1"));
        input.put("a.02", Arrays.asList("l2"));
        input.put("b.01", Arrays.asList("l3"));
        input.put("b.02", Arrays.asList("l4"));

        // Group by prefix and aggregate lists
        Map<String, List<String>> output = groupByPrefix(input);

        // Print the result
        output.forEach((prefix, values) -> System.out.println(prefix + ": " + values));
    }
}
