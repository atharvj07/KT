@Override
protected void onDraw(Canvas canvas) {
    if (_strokeWidth > 0) {
        // Save the current paint state
        Paint p = getPaint();
        
        // Draw outer stroke (using shadow layer)
        p.setStyle(Paint.Style.FILL); // Ensure we're filling the shape
        p.setShadowLayer(_strokeWidth, 0, 0, _strokeColor); // Set shadow layer as outer stroke
        setLayerType(LAYER_TYPE_SOFTWARE, p); // Disable hardware acceleration for shadow layer

        // Draw the filled text (with shadow layer for stroke)
        super.onDraw(canvas);
        
        // Clear the shadow layer after the outer stroke
        p.clearShadowLayer();

        // Set text color to normal color and remove shadow
        int currentTextColor = getCurrentTextColor();
        setTextColor(_strokeColor);

        // Draw the actual text (without shadow or outer stroke)
        super.onDraw(canvas);

        // Revert text color to original
        setTextColor(currentTextColor);

    } else {
        // If no stroke width, just draw the text normally
        super.onDraw(canvas);
    }
}
