public class Student {
    String name;
    int roll_no;

    // Method to set student details
    public void getDetails(String Name, int roll) {
        name = Name;
        roll_no = roll;
    }
}

public class StudentRun {
    public static void main(String[] args) {
        Student student = new Student();  // Create an instance of Student
        String n = "John";  // Name of the student
        int r = 2;  // Roll number of the student

        // Call the getDetails method to set student details
        student.getDetails(n, r);

        // You can print the details to verify (optional)
        System.out.println("Student Name: " + student.name);
        System.out.println("Roll Number: " + student.roll_no);
    }
}
