import com.machinezoo.sourceafis.FingerprintImageOptions;

import java.nio.file.Files;
import java.io.File;

public class FingerprintProcessor {

    public static void main(String[] args) throws Exception {
        // Load image from file
        File file = new File("path/to/fingerprint/image.jpg");
        FingerprintImage image = new FingerprintImage(
            Files.readAllBytes(file.toPath()), 
            new FingerprintImageOptions().dpi(500)
        );

        // Create the template
        FingerprintTemplate template = new FingerprintTemplate(image);

        // Optionally, print the template as bytes
        System.out.println("Template (bytes): " + new String(template.toByteArray()));

        // Create a matcher
        FingerprintMatcher matcher = new FingerprintMatcher();
        matcher.setDatabase(new FingerprintTemplate[] { template });

        // Match the template with another template (you need to load or create the second template)
        FingerprintTemplate queryTemplate = new FingerprintTemplate(new FingerprintImage(
            Files.readAllBytes(new File("path/to/second/image.jpg").toPath()), 
            new FingerprintImageOptions().dpi(500)
        ));

        // Perform matching
        double score = matcher.match(queryTemplate);
        System.out.println("Matching score: " + score);
    }
}
