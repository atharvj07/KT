public class Manager {
    public Todo doSomething() {
        return new Todo();
    }
}

public class Todo {
    public String getDescription() {
        return "This is a task description!";
    }
}

WebEngine engine = webView.getEngine();
JSObject window = (JSObject) engine.executeScript("window");

Manager manager = new Manager();
window.setMember("manager", manager);

<script>
    function onclick() {
        var todo = manager.doSomething();  // Calling the doSomething() method
        var desc = todo.getDescription();  // Calling getDescription() on the returned Todo object
        document.getElementById("description-label").innerHTML = desc;
    }
</script>

public class TodoJSObject extends JSObject {
    private final Todo todo;

    public TodoJSObject(Todo todo) {
        this.todo = todo;
    }

    public String getDescription() {
        return todo.getDescription();
    }
}

public class Manager {
    public JSObject doSomething() {
        return new TodoJSObject(new Todo());
    }
}
