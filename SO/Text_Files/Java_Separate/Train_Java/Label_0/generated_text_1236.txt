public void setWakeAlarm(int secondsFromNow) {
    try {
        Process process = Runtime.getRuntime().exec("su");
        OutputStream os = process.getOutputStream();
        os.write(("echo $(($(date +%s) + " + secondsFromNow + ")) > /sys/class/rtc/rtc0/wakealarm\n").getBytes());
        os.flush();
        os.close();
        process.waitFor();
    } catch (Exception e) {
        e.printStackTrace();
    }
}
