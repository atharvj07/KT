import com.vaadin.flow.component.textfield.TextField;
import com.vaadin.flow.formatters.NumeralFieldFormatter;

public class CurrencyTextField extends VerticalLayout {

    public CurrencyTextField() {
        // Create a TextField for currency
        TextField currencyField = new TextField("Amount");

        // Apply NumeralFieldFormatter
        NumeralFieldFormatter currencyFormatter = new NumeralFieldFormatter.Builder()
                .decimalMark(",")  // Use ',' as the decimal separator
                .decimalScale(2)   // 2 decimal places
                .delimiter(".")    // Use '.' as the thousand separator
                .thousandsGroupStyle(NumeralFieldFormatter.ThousandsGroupStyle.THOUSAND)
                .build();

        // Extend the TextField with the currency formatter
        currencyFormatter.extend(currencyField);

        add(currencyField);
    }
}
