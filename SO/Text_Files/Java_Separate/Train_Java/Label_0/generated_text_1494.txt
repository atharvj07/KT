import javax.swing.*;
import java.awt.*;

public class DrawCircle extends JComponent {
    private int x = 30; // Initial X position
    private int y = 30; // Initial Y position
    private final int diameter = 50; // Circle size
    private final int moveSpeed = 10; // Movement speed

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        g.drawOval(x, y, diameter, diameter);
    }

    public void moveUp() {
        if (y - moveSpeed >= 0) {
            y -= moveSpeed;
            repaint();
        }
    }

    public void moveDown() {
        if (y + moveSpeed + diameter <= getHeight()) {
            y += moveSpeed;
            repaint();
        }
    }

    public void moveRight() {
        if (x + moveSpeed + diameter <= getWidth()) {
            x += moveSpeed;
            repaint();
        }
    }

    public void moveLeft() {
        if (x - moveSpeed >= 0) {
            x -= moveSpeed;
            repaint();
        }
    }
}

import javax.swing.*;
import java.awt.*;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;

public class Main {
    public static void main(String[] args) {
        JFrame frame = new JFrame();
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setSize(500, 400);
        frame.setLayout(new BorderLayout());

        // Create a DrawCircle instance
        final DrawCircle c = new DrawCircle();
        frame.add(c, BorderLayout.CENTER);

        // Create buttons
        JButton buttonRight = new JButton("Right");
        JButton buttonLeft = new JButton("Left");
        JButton buttonUp = new JButton("Up");
        JButton buttonDown = new JButton("Down");

        // Create control panel
        JPanel panel = new JPanel();
        panel.setLayout(new GridLayout(2, 2));
        panel.add(buttonLeft);
        panel.add(buttonRight);
        panel.add(buttonUp);
        panel.add(buttonDown);
        frame.add(panel, BorderLayout.SOUTH);

        // Add event listeners
        buttonUp.addActionListener(e -> c.moveUp());
        buttonDown.addActionListener(e -> c.moveDown());
        buttonRight.addActionListener(e -> c.moveRight());
        buttonLeft.addActionListener(e -> c.moveLeft());

        frame.setVisible(true);
    }
}
