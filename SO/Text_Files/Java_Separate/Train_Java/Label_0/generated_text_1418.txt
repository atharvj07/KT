import java.time.*;
import java.util.*;

public class Appointment {
    private Long epochTime;      // UTC time in epoch (seconds)
    private String timezone;     // Timezone, e.g., "America/New_York"
    private Calendar localAppointmentTime; // Local time in the specified timezone

    public Appointment(Long epochTime, String timezone) {
        this.epochTime = epochTime;
        this.timezone = timezone;
    }

    public void convertEpochToLocalTime() {
        // Step 1: Convert epoch time to Instant (UTC)
        Instant instant = Instant.ofEpochSecond(epochTime);
        
        // Step 2: Convert Instant to ZonedDateTime using the provided timezone
        ZonedDateTime zonedDateTime = instant.atZone(ZoneId.of(timezone));
        
        // Step 3: Optionally, you can convert to a Calendar object if needed (not mandatory)
        localAppointmentTime = Calendar.getInstance();
        localAppointmentTime.setTime(Date.from(zonedDateTime.toInstant()));
        
        // Print results
        System.out.println("Local Appointment Time: " + zonedDateTime);
    }

    public static void main(String[] args) {
        // Example with an epoch time (e.g., 1649075419) and timezone "America/New_York"
        Appointment appointment = new Appointment(1649075419L, "America/New_York");
        appointment.convertEpochToLocalTime();
    }
}
