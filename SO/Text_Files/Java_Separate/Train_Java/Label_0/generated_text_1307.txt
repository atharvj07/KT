import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Listy {
    public static void main(String[] args) {
        // Creating a list of string arrays
        List<String[]> words = new ArrayList<>();
        words.add(new String[] {"Pat"});
        words.add(new String[] {"Micha?"});

        // Call the method to get the sum of lengths of all strings in the arrays
        System.out.println(getSumOfLengths(words));
    }

    // Method to get the sum of lengths of strings in each array
    public static int getSumOfLengths(List<String[]> words) {
        int sum = 0;
        
        // Loop through each array in the list and add their lengths
        for (String[] currentArray : words) {
            for (String word : currentArray) {
                sum += word.length(); // Add the length of each word to the sum
            }
        }
        return sum;
    }
}
