public String toString() {
    StringBuilder sb = new StringBuilder();
    int count = 0;

    // Loop through each row of the grid
    for (int row = 0; row < 5; row++) { // Assuming 5 rows in the grid
        for (int i = 0; i < square.size(); i++) {
            // Print the row of the current grid
            for (int col = 0; col < 5; col++) { // Assuming 5 columns in the grid
                // Format the numbers with leading space
                sb.append(String.format("%3d", square.get(i).getGrid()[row][col]));
                if (col < 4) {
                    sb.append(" "); // Space between elements in the same grid
                }
            }
            if (i < square.size() - 1) {
                sb.append(" | "); // Separate grids with a vertical bar
            }

            count++;
            // After every 6 grids, print a newline and reset the count
            if (count == 6) {
                sb.append("\n");
                count = 0;
            }
        }
        sb.append("\n"); // After every row, print a newline to move to the next row
    }

    return sb.toString();
}
