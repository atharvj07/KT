import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.List;

public class BigDecimalDifference {
    public static void main(String[] args) {
        List<BigDecimal> numbers = List.of(
            new BigDecimal("76.2372"),
            new BigDecimal("0E-4"),  // Considered zero
            new BigDecimal("80.2318"),
            new BigDecimal("82.1111"),
            new BigDecimal("88.0937")
        );

        List<BigDecimal> result = calculateDifferences(numbers);
        System.out.println(result);
    }

    public static List<BigDecimal> calculateDifferences(List<BigDecimal> numbers) {
        List<BigDecimal> differences = new ArrayList<>();
        BigDecimal previousNonZero = null;

        for (BigDecimal num : numbers) {
            if (num.compareTo(BigDecimal.ZERO) != 0) {
                if (previousNonZero != null) {
                    differences.add(num.subtract(previousNonZero).abs());
                }
                previousNonZero = num;
            }
        }

        return differences;
    }
}
