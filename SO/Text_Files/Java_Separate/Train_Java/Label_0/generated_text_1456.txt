import java.time.*;
import java.time.format.DateTimeFormatter;
import java.time.format.DateTimeParseException;
import java.util.Locale;

public class DateTimeUtils {
    private static final ZoneId DEFAULT_ZONE = ZoneId.of("America/New_York"); // Default to Eastern Time

    public static Instant parseBirthDateTime(String birthdate) {
        if (birthdate == null || birthdate.isEmpty()) {
            return Instant.parse("0001-01-01T00:00:00.00Z");
        }

        DateTimeFormatter dateTimeFormatter = DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss")
                .withLocale(Locale.ENGLISH);

        DateTimeFormatter dateFormatter = DateTimeFormatter.ofPattern("dd-MMM-yyyy")
                .withLocale(Locale.ENGLISH);

        try {
            // First attempt: Parse full date-time
            LocalDateTime localDateTime = LocalDateTime.parse(birthdate, dateTimeFormatter);
            return localDateTime.atZone(DEFAULT_ZONE).toInstant();
        } catch (DateTimeParseException e) {
            // Try parsing just the date format
            try {
                LocalDate localDate = LocalDate.parse(birthdate, dateFormatter);
                return localDate.atStartOfDay(DEFAULT_ZONE).toInstant(); // Start of the day in the specified timezone
            } catch (DateTimeParseException ex) {
                return Instant.parse("0001-01-01T00:00:00.00Z"); // Fallback for unparseable formats
            }
        }
    }

    public static void main(String[] args) {
        System.out.println(parseBirthDateTime("2024-03-17 15:30:45")); // With time
        System.out.println(parseBirthDateTime("17-Mar-1985")); // Only date
        System.out.println(parseBirthDateTime("invalid-date")); // Invalid case
    }
}
