public static int daysForYears(int year) {
    int days = 0;

    for (int i = 0; i < year; i++) {
        if ((i % 4 == 0 && (i % 100 != 0)) || (i % 400 == 0)) {
            days += 366; // Leap year
        } else {
            days += 365; // Non-leap year
        }
    }

    return days;
}
