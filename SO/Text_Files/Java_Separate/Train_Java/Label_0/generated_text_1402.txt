import javax.swing.*;

public class HorizontalJListExample {
    public static void main(String[] args) {
        JFrame frame = new JFrame("JList Example");

        // Sample items for the JList
        String[] items = {"Item 1", "Item 2", "Item 3", "Item 4", "Item 5", "Item 6", "Item 7", "Item 8", "Item 9", "Item 10"};
        
        // Create the JList
        JList<String> list = new JList<>(items);

        // Set the JList to be horizontally wrapped
        list.setLayoutOrientation(JList.HORIZONTAL_WRAP);

        // Set the number of visible items
        list.setVisibleRowCount(1);  // This ensures there's only one row of items in the JList

        // Create a JScrollPane and add the JList to it
        JScrollPane scrollPane = new JScrollPane(list);
        
        // Set the horizontal scrollbar policy
        scrollPane.setHorizontalScrollBarPolicy(JScrollPane.HORIZONTAL_SCROLLBAR_ALWAYS);
        
        // Set the preferred size of the JScrollPane to allow only 3 items visible at once
        scrollPane.setPreferredSize(new java.awt.Dimension(400, 60));  // Adjust width as needed
        
        // Add the JScrollPane to the JFrame
        frame.add(scrollPane);

        // Basic JFrame setup
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.pack();
        frame.setVisible(true);
    }
}
