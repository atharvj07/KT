public interface StatusMapper {
    // Mapping State to Status
    Status stateToStatus(State state);
    
    // Mapping Status to State (optional, if you need reverse mapping)
    State statusToState(Status status);
}

public class StatusMapperImpl implements StatusMapper {

    @Override
    public Status stateToStatus(State state) {
        if (state == null) {
            return null;
        }
        switch (state) {
            case PENDING:
                return Status.PENDING;
            case CANCELLED:
                return Status.CANCELLED;
            default:
                throw new IllegalArgumentException("Unknown state: " + state);
        }
    }

    @Override
    public State statusToState(Status status) {
        if (status == null) {
            return null;
        }
        switch (status) {
            case PENDING:
                return State.PENDING;
            case CANCELLED:
                return State.CANCELLED;
            default:
                throw new IllegalArgumentException("Unknown status: " + status);
        }
    }
}

public interface EntityMapper {
    // Mapping Entity to Domain
    Domain entityToDomain(Entity entity);
}

public class EntityMapperImpl implements EntityMapper {
    
    private final StatusMapper statusMapper;
    
    // Constructor to inject the StatusMapper
    public EntityMapperImpl(StatusMapper statusMapper) {
        this.statusMapper = statusMapper;
    }

    @Override
    public Domain entityToDomain(Entity entity) {
        if (entity == null) {
            return null;
        }

        // Map the State to Status using the statusMapper
        Status status = statusMapper.stateToStatus(entity.getState());
        
        // Create a Domain object and return it
        return new Domain(entity.getName(), status);
    }
}

public class Entity {
    private String name;
    private State state;

    // Getters and Setters
    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public State getState() {
        return state;
    }

    public void setState(State state) {
        this.state = state;
    }
}

public class Domain {
    private String name;
    private Status status;

    // Constructor, Getters, Setters
    public Domain(String name, Status status) {
        this.name = name;
        this.status = status;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public Status getStatus() {
        return status;
    }

    public void setStatus(Status status) {
        this.status = status;
    }
}

public class Main {
    public static void main(String[] args) {
        // Create the StatusMapper and EntityMapper
        StatusMapper statusMapper = new StatusMapperImpl();
        EntityMapper entityMapper = new EntityMapperImpl(statusMapper);

        // Create an example Entity
        Entity entity = new Entity();
        entity.setName("Name");
        entity.setState(State.PENDING);

        // Convert Entity to Domain
        Domain domain = entityMapper.entityToDomain(entity);

        // Print the result
        System.out.println("Domain name: " + domain.getName());
        System.out.println("Domain status: " + domain.getStatus());
    }
