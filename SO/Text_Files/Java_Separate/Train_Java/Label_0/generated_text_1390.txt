import javax.swing.*;
import java.awt.*;
import java.awt.event.KeyEvent;
import java.awt.event.KeyListener;

public class Game {
    public static void main(String[] args) {
        JFrame frame = new JFrame();
        final int FRAME_WIDTH = 1000;
        final int FRAME_HEIGHT = 600;
        frame.setSize(FRAME_WIDTH, FRAME_HEIGHT);
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        final Map map = new Map();
        frame.add(map, BorderLayout.CENTER);
        frame.setVisible(true);
        map.startAnimation();
    }
}

class Map extends JComponent implements Runnable, KeyListener {
    private int speed = 5;
    private int xcenter = 500; // starts on player
    private int ycenter = 300;
    private double angle = 0.0;

    private int[] xcords = {xcenter + 10, xcenter, xcenter + 20};
    private int[] ycords = {ycenter - 10, ycenter + 20, ycenter + 20};

    private boolean moveNorth = false;
    private boolean moveEast = false;
    private boolean moveSouth = false;
    private boolean moveWest = false;

    public Map() {
        addKeyListener(this);
        setFocusable(true);
        setFocusTraversalKeysEnabled(false);
    }

    public void startAnimation() {
        Thread t = new Thread(this);
        t.start();
    }

    public void paintComponent(Graphics g) {
        Graphics2D g2d = (Graphics2D) g;
        
        // Clear previous frame
        super.paintComponent(g);

        // Move the map and rotate the world around the player
        g2d.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        // Center the map on the player by translating the whole world
        g2d.translate(getWidth() / 2 - xcenter, getHeight() / 2 - ycenter);

        // Rotate the world around the player based on the angle
        g2d.rotate(angle, xcenter, ycenter);

        // Draw the grid and player (or your racing track)
        for (int i = -10; i < 21; i++) {
            g2d.drawLine(i * 50, -1000, i * 50, 1000);
            g2d.drawLine(-1000, i * 50, 1000, i * 50);
        }
        g2d.fillPolygon(xcords, ycords, 3);

        // Draw a player as a simple circle for now
        g2d.fillOval(xcenter - 15, ycenter - 15, 30, 30);
    }

    public void run() {
        while (true) {
            try {
                if (moveNorth || moveEast || moveSouth || moveWest) {
                    repaint();
                }
                Thread.sleep(10);
            } catch (InterruptedException e) {
                e.printStackTrace();
            }
        }
    }

    public void keyPressed(KeyEvent e) {
        if (e.getExtendedKeyCode() == 68) { // d
            moveEast = true;
        } else if (e.getExtendedKeyCode() == 87) { // w
            moveNorth = true;
        } else if (e.getExtendedKeyCode() == 65) { // a
            moveWest = true;
        } else if (e.getExtendedKeyCode() == 83) { // s
            moveSouth = true;
        }
    }

    public void keyReleased(KeyEvent e) {
        moveNorth = false;
        moveEast = false;
        moveSouth = false;
        moveWest = false;
    }

    public void keyTyped(KeyEvent e) {
    }

    // Update the player position based on key input
    private void updatePlayerPosition() {
        if (moveNorth) {
            ycenter -= speed;
        } else if (moveSouth) {
            ycenter += speed;
        } else if (moveEast) {
            xcenter += speed;
        } else if (moveWest) {
            xcenter -= speed;
        }
    }
}
