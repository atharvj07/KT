public Flux<SoloPreguntaImp> getPreguntas(String perfil, int limit) {
    return getIdEntrevista(perfil)
        .concatMap(idEntrevista -> 
            interviewTestDao.getPreguntas(idEntrevista.getId(), limit)
                .collectList()
                .filter(list -> !list.isEmpty()) // Only keep non-empty results
                .flatMapMany(Flux::fromIterable) // Convert to Flux if non-empty
        )
        .take(1); // Stop after the first valid non-empty result
}
