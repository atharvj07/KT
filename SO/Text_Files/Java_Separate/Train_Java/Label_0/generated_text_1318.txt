   import org.bouncycastle.jce.provider.BouncyCastleProvider;
   import org.bouncycastle.openssl.PEMParser;
   import org.bouncycastle.pkcs.PKCS8EncryptedPrivateKeyInfo;
   import org.bouncycastle.pkcs.PKCS8PrivateKeyInfo;
   import org.bouncycastle.util.io.pem.PemObject;
   import org.bouncycastle.util.io.pem.PemReader;

   import java.io.File;
   import java.io.FileReader;
   import java.io.FileOutputStream;
   import java.io.IOException;
   import java.security.PrivateKey;
   import java.security.Security;

   public class PemToPkcs8Converter {

       public static void main(String[] args) {
           try {
               // Add BouncyCastle provider
               Security.addProvider(new BouncyCastleProvider());

               // Read the PEM private key file
               File pemFile = new File("secret.pem");
               FileReader fileReader = new FileReader(pemFile);
               PemReader pemReader = new PemReader(fileReader);

               PemObject pemObject = pemReader.readPemObject();
               byte[] pemContent = pemObject.getContent();

               // Decode the PEM content into the private key
               PEMParser pemParser = new PEMParser(new StringReader(new String(pemContent)));
               Object pemObjectParsed = pemParser.readObject();

               PrivateKey privateKey = null;

               if (pemObjectParsed instanceof PKCS8PrivateKeyInfo) {
                   privateKey = ((PKCS8PrivateKeyInfo) pemObjectParsed).getPrivateKey();
               } else {
                   System.out.println("Unable to read PEM object");
               }

               // Write the private key to a PKCS#8 DER file
               if (privateKey != null) {
                   FileOutputStream fos = new FileOutputStream("secret.key");
                   fos.write(privateKey.getEncoded());
                   fos.close();
                   System.out.println("Private key successfully written to secret.key in PKCS#8 format.");
               }

           } catch (Exception e) {
               e.printStackTrace();
           }
       }
   }
