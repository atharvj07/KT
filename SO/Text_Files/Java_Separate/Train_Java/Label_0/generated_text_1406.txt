public interface MyInterface {
    void doSomething();

    // Static factory method
    static MyInterface create(int value) {
        return new MyImplementation(value);
    }
}

    private int value;

    public MyImplementation(int value) {
        this.value = value;
    }

    @Override
    public void doSomething() {
        System.out.println("Doing something with value: " + value);
    }
}
