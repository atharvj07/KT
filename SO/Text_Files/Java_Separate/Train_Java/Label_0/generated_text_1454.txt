import java.util.Scanner;

public class heliocentric {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int caseNumber = 1;

        while (scan.hasNextInt()) {  // Keep reading until EOF
            int inE = scan.nextInt();
            int inM = scan.nextInt();
            int days = 0;

            while (inE != 0 || inM != 0) {  // Run until (0,0) alignment
                inE = (inE + 1) % 365;
                inM = (inM + 1) % 687;
                days++;
            }
            
            System.out.println("Case " + caseNumber + ": " + days);
            caseNumber++;
        }

        scan.close();
    }
}
