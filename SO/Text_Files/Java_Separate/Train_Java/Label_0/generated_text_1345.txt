public class Print implements Runnable {
    private String string;

    Print(String string) {
        this.string = string;
    }

    @Override
    public void run() {
        for (int i = string.length() - 1; i >= 0; i--) {
            // Synchronize on the Print object to avoid interleaving of prints
            synchronized (this) {
                // Print the current character followed by a hyphen
                System.out.print(string.charAt(i) + "-");
                // Yield to allow the other thread to run
                Thread.yield();
            }
        }
    }
}

public class Task1 {
    public static void main(String[] args) {
        Print r = new Print("NAME");
        Thread t1 = new Thread(r);
        Thread t2 = new Thread(r);
        
        t1.start();
        t2.start(); 
    }
}
