public int drawNodeRecursive(AnchorPane mainGraphTreeLayout, int x, int y, TreeItemNode parentNode) {
    // Create node (circle) for the current tree item
    Circle circle = new Circle(x, y, 15, Paint.valueOf("blue"));
    mainGraphTreeLayout.getChildren().add(circle);

    // Add text label inside the node
    Text txt = new Text(x - 5, y + 5, parentNode.getValue());
    mainGraphTreeLayout.getChildren().add(txt);

    List<TreeItemNode> nodeChildrenList = parentNode.getChildren();
    if (nodeChildrenList.isEmpty()) {
        return x; // No children, return current x position
    }

    int subtreeWidth = calculateSubtreeWidth(parentNode); // Calculate total subtree width
    int childX = x - (subtreeWidth / 2); // Center children under parent

    for (TreeItemNode child : nodeChildrenList) {
        child.setX(childX);
        child.setY(y + 60); // Move child downward

        // Draw line connecting parent to child
        Line line = new Line(x, y + 15, childX, y + 60 - 15);
        mainGraphTreeLayout.getChildren().add(line);

        // Recurse for child nodes
        childX = drawNodeRecursive(mainGraphTreeLayout, childX, child.getY(), child) + 50;
    }

    return x + subtreeWidth / 2;
}

private int calculateSubtreeWidth(TreeItemNode node) {
    if (node.getChildren().isEmpty()) {
        return 50; // Minimum spacing for leaf nodes
    }

    int width = 0;
    for (TreeItemNode child : node.getChildren()) {
        width += calculateSubtreeWidth(child); // Sum width of all subtrees
    }
    return Math.max(width, 100); // Ensure minimum spacing
}
