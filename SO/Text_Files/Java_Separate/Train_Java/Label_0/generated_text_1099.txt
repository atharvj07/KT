import java.lang.reflect.Field;

public abstract class A {
    public String p1;
    public String p2;

    // Includes getters/setters
}

public class B extends A {
    public String p3;
    public String p4;

    // Some implementation to populate base class attributes
    // Includes getters/setters
    public B(String p1, String p2, String p3, String p4) {
        this.p1 = p1;
        this.p2 = p2;
        this.p3 = p3;
        this.p4 = p4;
    }

    public static void main(String[] args) {
        // Initializing and populating parameters for B
        B b = new B("Value1", "Value2", "Value3", "Value4");

        // Print all attributes from the base class (A)
        printBaseClassFields(b);
    }

    public static void printBaseClassFields(Object obj) {
        Class<?> clazz = obj.getClass();

        // Iterate through fields of the class and its ancestors
        while (clazz != null) {
            Field[] fields = clazz.getDeclaredFields();
            for (Field field : fields) {
                try {
                    field.setAccessible(true); // Allows access to private/protected fields
                    System.out.println(field.getName() + ": " + field.get(obj));
                } catch (IllegalAccessException e) {
                    e.printStackTrace();
                }
            }
            // Move to the superclass
            clazz = clazz.getSuperclass();
        }
    }
}
