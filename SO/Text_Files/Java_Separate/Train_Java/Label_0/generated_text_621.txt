public static boolean isMonotonicallyDecreasing(int num) {
    int n = num; // Copy of num to be modified
    int b = (n % 10); // Step for a number if it is monotone
    n /= 10;
    if (num < 10) return true; // all single-digit numbers are monotonic
    while (n > 0 && n < b) {
        if ((n % 10) != b) {
            return false;
        }
        b = n % 10;
        n /= 10;
    }
    return true;
}
