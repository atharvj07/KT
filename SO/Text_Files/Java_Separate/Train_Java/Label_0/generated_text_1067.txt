import java.text.SimpleDateFormat
import java.util.*

fun main() {
    val df1: SimpleDateFormat = SimpleDateFormat("yyyy-MM-dd'T'HH:mm:ss.SSS'Z'")
    val string1 = "2022-12-29T19:59:20.783357Z"
    
    // Truncate microseconds to milliseconds
    val truncatedString = string1.substring(0, 23) + "Z"

    try {
        val result1: Date = df1.parse(truncatedString)
        println(result1)
    } catch (e: Exception) {
        e.printStackTrace()
    }
}
