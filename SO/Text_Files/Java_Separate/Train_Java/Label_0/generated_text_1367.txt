public Player hasWinner() {
    // Lines (rows)
    for (int y = 0; y < board.length; y++) {
        Player last = null;
        int nb = 0;
        Player[] line = board[y];
        for (int x = 0; x < line.length; x++) {
            Player played = line[x];
            if ((last == null || played == last) && played != null) {
                nb++;
                if (nb == 4) return played;
            } else {
                nb = (played == null ? 0 : 1);
            }
            last = played;
        }
    }

    // Columns
    for (int x = 0; x < board[0].length; x++) {
        Player last = null;
        int nb = 0;
        for (int y = 0; y < board.length; y++) {
            Player played = board[y][x];
            if ((last == null || played == last) && played != null) {
                nb++;
                if (nb == 4) return played;
            } else {
                nb = (played == null ? 0 : 1);
            }
            last = played;
        }
    }

    // Diagonals (Top-left to Bottom-right)
    for (int y = 0; y < board.length - 3; y++) { // iterate over rows
        for (int x = 0; x < board[0].length - 3; x++) { // iterate over columns
            Player last = board[y][x];
            if (last == null) continue;
            int nb = 1;
            for (int i = 1; i < 4; i++) {
                if (y + i < board.length && x + i < board[0].length && board[y + i][x + i] == last) {
                    nb++;
                    if (nb == 4) return last;
                } else {
                    break;
                }
            }
        }
    }

    // Diagonals (Bottom-left to Top-right)
    for (int y = 3; y < board.length; y++) { // start from the 4th row (index 3) for bottom-left to top-right
        for (int x = 0; x < board[0].length - 3; x++) { // iterate over columns
            Player last = board[y][x];
            if (last == null) continue;
            int nb = 1;
            for (int i = 1; i < 4; i++) {
                if (y - i >= 0 && x + i < board[0].length && board[y - i][x + i] == last) {
                    nb++;
                    if (nb == 4) return last;
                } else {
                    break;
                }
            }
        }
    }

    return null; // No winner
}
