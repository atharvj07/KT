import java.io.*;
import java.util.*;

public class CSVReaderExample {
    public static void main(String[] args) {
        System.out.println("Data from CSV file to be analysed:\n");
        
        String file = "jrc-covid-19-all-days-of-world_ASSIGNMENT-FIXED.csv";
        ArrayList<String[]> lines = new ArrayList<>();
        String line;

        try (BufferedReader bufferedReader = new BufferedReader(new FileReader(file))) {
            int i = 0;
            while ((line = bufferedReader.readLine()) != null && i < 27) {
                String[] values = line.split(",");
                lines.add(values);  // Store each row as a String array
                i++;
                System.out.println(Arrays.toString(values));  // Print the row
            }
        } catch (IOException e) {
            e.printStackTrace();
        }

        // Accessing the 4th row (Index 3)
        if (lines.size() > 3) {  // Ensure the row exists
            String[] thirdCountry = lines.get(3);
            
            // Accessing the 7th column (Index 6)
            if (thirdCountry.length > 6) {
                int cp3 = Integer.parseInt(thirdCountry[6].trim()); // Convert String to int
                System.out.println(cp3);  // Should return 107122
            } else {
                System.out.println("Column index 6 is out of bounds.");
            }
        } else {
            System.out.println("Row index 3 is out of bounds.");
        }
    }
}
