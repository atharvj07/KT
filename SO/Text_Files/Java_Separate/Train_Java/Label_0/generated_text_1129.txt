import com.fasterxml.jackson.core.JsonParser;
import com.fasterxml.jackson.databind.DeserializationContext;
import com.fasterxml.jackson.databind.JsonDeserializer;
import com.fasterxml.jackson.databind.annotation.JsonDeserialize;
import com.fasterxml.jackson.databind.node.NullNode;

import java.io.IOException;

public class Person {

    @JsonProperty(value = "name", defaultValue = "null")
    @JsonDeserialize(using = NullToNullStringDeserializer.class)
    private String name = "null";

    @JsonProperty(value = "surName", defaultValue = "null")
    @JsonDeserialize(using = NullToNullStringDeserializer.class)
    private String surName = "null";

    // Getters and setters

}

class NullToNullStringDeserializer extends JsonDeserializer<String> {
    @Override
    public String deserialize(JsonParser p, DeserializationContext ctxt) throws IOException {
        if (p.getCurrentToken() == NullNode.instance) {
            return "null";  // return "null" when the value is explicitly null
        }
        return p.getValueAsString();  // otherwise, return the actual value
    }
}

import com.fasterxml.jackson.annotation.JsonProperty;
import com.fasterxml.jackson.annotation.JsonSetter;

public class Person {

    @JsonProperty(value = "name", defaultValue = "null")
    @JsonSetter(nulls = JsonSetter.Nulls.SKIP)  // Skip when null in input
    private String name = "null";

    @JsonProperty(value = "surName", defaultValue = "null")
    @JsonSetter(nulls = JsonSetter.Nulls.SKIP)  // Skip when null in input
    private String surName = "null";

    // Getters and setters
}

import com.fasterxml.jackson.databind.ObjectMapper;

public class Main {
    public static void main(String[] args) throws Exception {
        String json = "{\"name\":null, \"surName\":null}";

        ObjectMapper objectMapper = new ObjectMapper();
        Person person = objectMapper.readValue(json, Person.class);
        System.out.println(person.getName()); // should print "null"
        System.out.println(person.getSurName()); // should print "null"
    }
}
