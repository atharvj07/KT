public class Candlestick {
    private int time;
    private double high;
    private double low;

    // Constructor
    public Candlestick(int time, double high, double low) {
        this.time = time;
        this.high = high;
        this.low = low;
    }

    // Getters
    public int getTime() {
        return time;
    }

    public double getHigh() {
        return high;
    }

    public double getLow() {
        return low;
    }

    // Compound Interest Calculation
    public static double calculateCompoundRate(double low, double high) {
        return high / low;
    }

    public static double findMaxCompoundRate(List<Candlestick> ohlcSeries) {
        double maxRate = 0;
        double maxHighSoFar = 0; // Track the highest 'high' value seen so far
        
        // Iterate over the series from the end to the beginning
        for (int i = ohlcSeries.size() - 1; i >= 0; i--) {
            Candlestick current = ohlcSeries.get(i);
            
            // If we can find a 'high' after the current 'low', calculate the compound rate
            if (maxHighSoFar > current.getLow()) {
                double rate = calculateCompoundRate(current.getLow(), maxHighSoFar);
                maxRate = Math.max(maxRate, rate);
            }
            
            // Update maxHighSoFar if the current 'high' is greater
            maxHighSoFar = Math.max(maxHighSoFar, current.getHigh());
        }
        
        return maxRate;
    }

    public static void main(String[] args) {
        // Example candlestick data (time, high, low)
        List<Candlestick> ohlcSeries = Arrays.asList(
            new Candlestick(3, 2, 0),
            new Candlestick(2, 0, 1.8),
            new Candlestick(1, 1.1, 0),
            new Candlestick(0, 0, 1)
        );

        double maxCompoundRate = findMaxCompoundRate(ohlcSeries);
        System.out.println("Maximum Compound Interest Rate: " + maxCompoundRate);
    }
}
