import org.springframework.http.HttpStatus;
import org.springframework.http.client.ClientHttpResponse;
import org.springframework.web.client.RestClient;
import org.springframework.web.client.ResponseErrorHandler;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.io.IOException;

class CmsClient {

    private final RestClient restClient;
    private final ObjectMapper objectMapper = new ObjectMapper(); // Jackson ObjectMapper

    public CmsClient(RestClient restClient) {
        this.restClient = restClient;
    }

    public Policy getPolicy() {
        return restClient.get()
            .uri("some/url")
            .accept(MediaType.APPLICATION_JSON)
            .retrieve()
            .onStatus(HttpStatusCode::isError, (request, response) -> {
                if (response.getStatusCode() == HttpStatus.NOT_FOUND) {
                    // Do nothing, return null
                } else {
                    throw new RuntimeException("Unexpected error: " + response.getStatusCode());
                }
            })
            .body(response -> {
                try {
                    String body = response.body(String.class);
                    if (body == null || body.isBlank()) {
                        return null; // Return null if the response body is empty
                    }
                    return objectMapper.readValue(body, Policy.class);
                } catch (IOException e) {
                    throw new RuntimeException("Error parsing JSON", e);
                }
            });
    }
}
