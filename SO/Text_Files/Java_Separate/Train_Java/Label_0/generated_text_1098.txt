import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.AfterEach;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.InjectMocks;
import org.mockito.MockedStatic;
import org.mockito.junit.jupiter.MockitoExtension;

@ExtendWith(MockitoExtension.class)
public class UsaTesteEstaticoTest {

    @InjectMocks
    UsaTesteEstatico usaTesteEstatico;

    private MockedStatic<TesteEstatico> mockedStatic;

    @BeforeEach
    void setUp() {
        // Start mocking static methods
        mockedStatic = Mockito.mockStatic(TesteEstatico.class);
    }

    @AfterEach
    void tearDown() {
        // Close the static mock to ensure proper cleanup
        mockedStatic.close();
    }

    @Test
    void teste1(){
        mockedStatic.when(TesteEstatico::teste)
                    .thenReturn("BANANA");

        String res = usaTesteEstatico.metodoParaTeste1();
        System.out.println(res); // This will print "BANANA BAR"
    }

    @Test
    void teste2(){
        mockedStatic.when(TesteEstatico::teste)
                    .thenReturn("LARANJA");

        String res = usaTesteEstatico.metodoParaTeste2();
        System.out.println(res); // This will print "FOO LARANJA BAR"
    }
}
