import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.*;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class FileSorter {
    public static void main(String[] args) {
        List<String> names = Arrays.asList(
                "f1660.msg.220523.xml.gpg",
                "f1660.msg.dscl.220523.xml.gpg",
                "f1660.msg.att.220523.tar.gz.gpg",
                "f1660.cit.msg.220520.xml1.gpg",
                "f1660.cit.msg.220520.xml2.gpg",
                "f1660.cit.msg.att.220520.tar.gz.gpg",
                "f1660.cit.msg.dscl.220520.xml.gpg",
                "8701D6.msg.220524.xml.gpg",
                "8701D6.dscl.220524.xml.gpg",
                "8701D6.msg.att.220524.tar.gz.gpg"
        );

        names.sort(new FileComparator());

        // Print the sorted list
        System.out.println("Sorted List: ");
        for (String name : names) {
            System.out.println(name);
        }
    }

    static class FileComparator implements Comparator<String> {
        @Override
        public int compare(String file1, String file2) {
            LocalDate date1 = extractDate(file1);
            LocalDate date2 = extractDate(file2);
            int dateComparison = date1.compareTo(date2);
            if (dateComparison != 0) {
                return dateComparison;
            }

            // Compare based on priority order
            int priority1 = getPriority(file1);
            int priority2 = getPriority(file2);

            return Integer.compare(priority1, priority2);
        }

        private LocalDate extractDate(String fileName) {
            Matcher matcher = Pattern.compile("(\\d{6})").matcher(fileName);
            if (matcher.find()) {
                DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyMMdd");
                return LocalDate.parse(matcher.group(1), formatter);
            }
            throw new IllegalArgumentException("No valid date found in filename: " + fileName);
        }

        private int getPriority(String fileName) {
            if (fileName.contains(".att.")) return 1;
            if (fileName.contains(".dscl.")) return 2;
            return 3; // All other XML files
        }
    }
}
