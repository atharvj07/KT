import software.amazon.awssdk.services.s3.model.GetObjectRequest;
import software.amazon.awssdk.services.s3.model.GetObjectResponse;
import software.amazon.awssdk.services.s3.S3Client;
import software.amazon.awssdk.services.s3.model.S3Exception;

public class S3Example {

    // Get object from S3
    public static void getObject(S3Client s3Client, String bucketName, String key) {
        System.out.println("Checking if bucket exists. Bucket Name: " + bucketName);
        System.out.printf("%n");
        
        try {
            // Create GetObjectRequest with response headers
            GetObjectRequest getObjectRequest = GetObjectRequest.builder()
                    .bucket(bucketName)
                    .key(key)
                    .cacheControl("No-cache") // Set cache control
                    .contentDisposition("attachment; filename=key.txt") // Set content disposition
                    .build();

            // Get the object from S3
            GetObjectResponse response = s3Client.getObject(getObjectRequest, resp -> resp);
            
            // Display the content of the object
            displayTextInputStream(response);
            
        } catch (S3Exception e) {
            System.err.println(e.awsErrorDetails().errorMessage());
        }        
    }

    // Helper method to display the object's content (you can replace this with your actual implementation)
    private static void displayTextInputStream(GetObjectResponse response) {
        // Logic to display the input stream or save to a file
        System.out.println("Object content: " + response);
    }

    public static void main(String[] args) {
        // Your S3Client and parameters here
        S3Client s3Client = S3Client.create();
        String bucketName = "your-bucket-name";
        String key = "your-object-key";

        // Call getObject to retrieve the S3 object
        getObject(s3Client, bucketName, key);
    }
}
