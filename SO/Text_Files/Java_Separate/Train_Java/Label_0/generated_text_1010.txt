private final int initialLife; // Store the original life value

public Char(String name, int skill, int defense, int life) {
    this.name = name;
    this.skill = skill;
    this.defense = defense;
    this.life = life;
    this.initialLife = life; // Store initial value
}

public boolean isWeak() {
    return life < initialLife * 0.25; // Compare with initialLife instead of current life
}

Char hero = new Char("Hero", 10, 5, 100);
System.out.println(hero.isWeak()); // false
hero.life = 20;
System.out.println(hero.isWeak()); // true (because 20 < 100 * 0.25)
