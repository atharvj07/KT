public class DiagonalMatrix {
    public static void main(String[] args) {
        int[][] m = new int[10][10];  // Declare 10x10 matrix
        int n = 1;

        for (int i = 0; i < 10; i++) {  // Iterate over rows
            for (int j = 9 - i, count = 0; j < 10; j++, count++) { 
                m[i + count][j] = n++;  // Move diagonally downward
            }
        }

        // Print the matrix for verification
        for (int i = 0; i < 10; i++) {
            for (int j = 0; j < 10; j++) {
                System.out.printf("%3d ", m[i][j]);
            }
            System.out.println();
        }
    }
}
