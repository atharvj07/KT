import java.lang.foreign.*;
import java.lang.invoke.MethodHandle;
import java.lang.invoke.MethodType;

public class Main {
    public static void main(String[] args) throws Throwable {
        System.loadLibrary("test_matrix"); // Ensure your native library is loaded

        Linker linker = Linker.nativeLinker(); // Get Foreign Linker API instance

        MethodHandle handle = linker.downcallHandle(
            SymbolLookup.loaderLookup().lookup("printMatrix").orElseThrow(),
            MethodType.methodType(void.class, MemoryAddress.class, int.class, int.class),
            FunctionDescriptor.ofVoid(ValueLayout.ADDRESS, ValueLayout.JAVA_INT, ValueLayout.JAVA_INT) // Corrected!
        );

        int[][] arr = {{1, 2}, {3, 4}, {5, 6}};

        try (Arena arena = Arena.ofConfined()) {  // Using Arena (Java 21+) instead of ResourceScope
            // Step 1: Allocate memory for each row and store their addresses
            MemorySegment[] rowSegments = new MemorySegment[arr.length];
            MemorySegment rowPointerSegment = arena.allocateArray(ValueLayout.ADDRESS, arr.length); // Store pointers

            for (int i = 0; i < arr.length; i++) {
                rowSegments[i] = allocateArray(arr[i], arena); // Allocate memory for row
                rowPointerSegment.setAtIndex(ValueLayout.ADDRESS, i, rowSegments[i].address()); // Store address
            }

            // Step 2: Invoke the C function
            handle.invokeExact(rowPointerSegment.address(), arr.length, arr[0].length);
        }
    }

    private static MemorySegment allocateArray(int[] array, Arena arena) {
        MemorySegment segment = arena.allocateArray(ValueLayout.JAVA_INT, array.length);
        segment.copyFrom(MemorySegment.ofArray(array)); // Copy Java array to native memory
        return segment;
    }
}
