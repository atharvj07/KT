import javax.net.ssl.HttpsURLConnection;
import java.io.IOException;
import java.net.URL;
import java.util.List;
import java.util.Map;

public class HttpHeadRequest {
    public static void main(String[] args) {
        String url2 = "https://pagalworld.com.se/siteuploads/files/sfd14/6934/Har Har Shambhu Ringtone_320(PagalWorld.com.se).mp3";
        String req_method = "HEAD";
        HttpsURLConnection conn = null;

        try {
            URL UrlConnection = new URL(url2);
            conn = (HttpsURLConnection) UrlConnection.openConnection();

            conn.setRequestMethod(req_method);
            conn.setInstanceFollowRedirects(true); // Follow redirects
            conn.setRequestProperty("Host", UrlConnection.getHost()); // Correct Host
            conn.setRequestProperty("Accept", "*/*");
            conn.setRequestProperty("Connection", "keep-alive");
            conn.setRequestProperty("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64)");

            System.out.println("Host: " + UrlConnection.getHost());
            System.out.println("Connection code: " + conn.getResponseCode());
            System.out.println("Connection Msg: " + conn.getResponseMessage());

            // Print request headers
            System.out.println("\nRequest Headers:");
            conn.getRequestProperties().forEach((key, value) -> System.out.println(key + ": " + value));

        } catch (IOException e) {
            e.printStackTrace();
        } finally {
            if (conn != null) {
                conn.disconnect();
            }
        }
    }
}
