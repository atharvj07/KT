import android.content.SharedPreferences;
import android.os.Bundle;
import android.widget.ArrayAdapter;
import android.widget.ListView;

import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {
    private ListView listView;
    private SharedPreferences sharedPreferences;
    private static final String PREFS_NAME = "UserPrefs";
    private static final String KEY_LIST = "UserList";

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        listView = findViewById(R.id.listView);
        sharedPreferences = getSharedPreferences(PREFS_NAME, MODE_PRIVATE);

        // Load the list from SharedPreferences or fallback to the default
        String[] userList = loadList();

        // Set the adapter for the list view
        ArrayAdapter<String> adapter = new ArrayAdapter<>(this, android.R.layout.simple_list_item_1, userList);
        listView.setAdapter(adapter);
    }

    // Load the list from SharedPreferences or fallback to the default list
    private String[] loadList() {
        // Check if the user has saved a custom list
        String savedList = sharedPreferences.getString(KEY_LIST, null);

        if (savedList != null) {
            // Parse the saved list and return it (e.g., splitting by commas)
            return savedList.split(",");
        } else {
            // Fallback to the default list in strings.xml
            return getResources().getStringArray(R.array.default_list);
        }
    }

    // Save the list to SharedPreferences (e.g., when the user modifies it)
    private void saveList(String[] newList) {
        SharedPreferences.Editor editor = sharedPreferences.edit();
        editor.putString(KEY_LIST, String.join(",", newList));
        editor.apply();
    }
}

Button addButton = findViewById(R.id.addButton);
addButton.setOnClickListener(v -> {
    // Show a dialog to let the user add an item
    showAddItemDialog();
});

private void showAddItemDialog() {
    AlertDialog.Builder builder = new AlertDialog.Builder(this);
    builder.setTitle("Add Item");

    final EditText input = new EditText(this);
    builder.setView(input);

    builder.setPositiveButton("Add", (dialog, which) -> {
        String newItem = input.getText().toString();
        addItemToList(newItem);
    });

    builder.setNegativeButton("Cancel", (dialog, which) -> dialog.cancel());

    builder.show();
}

private void addItemToList(String newItem) {
    // Add the new item to the list
    String[] currentList = loadList();
    List<String> list = new ArrayList<>(Arrays.asList(currentList));
    list.add(newItem);

    // Save the updated list
    saveList(list.toArray(new String[0]));

    // Update the list view
    ArrayAdapter<String> adapter = new ArrayAdapter<>(this, android.R.layout.simple_list_item_1, list);
    listView.setAdapter(adapter);
}

@Entity(tableName = "user_list")
public class ListItem {
    @PrimaryKey(autoGenerate = true)
    public int id;
    public String itemName;

    public ListItem(String itemName) {
        this.itemName = itemName;
    }
}

@Dao
public interface ListItemDao {
    @Insert
    void insert(ListItem listItem);

    @Query("SELECT * FROM user_list")
    List<ListItem> getAllItems();

    @Delete
    void delete(ListItem listItem);
}

@Database(entities = {ListItem.class}, version = 1)
public abstract class AppDatabase extends RoomDatabase {
    public abstract ListItemDao listItemDao();
}

AppDatabase db = Room.databaseBuilder(getApplicationContext(), AppDatabase.class, "database-name").build();
