static String[] Student = new String[6];

public static void main(String[] args) {
    // Step 1: Adding elements to the array
    Student[0] = "Alice";
    Student[1] = "Bob";
    Student[2] = "Charlie";
    Student[3] = "David";
    Student[4] = "Eve";
    Student[5] = "Frank";

    System.out.println("Initial array: " + Arrays.toString(Student));

    // Step 2: Remove "Charlie"
    removeElement("Charlie");

    System.out.println("After removal: " + Arrays.toString(Student));

    // Step 3: Add "Grace" at an empty spot
    addElement("Grace");

    System.out.println("After adding Grace: " + Arrays.toString(Student));
}

// Remove an element by shifting remaining elements left
static void removeElement(String name) {
    for (int i = 0; i < Student.length; i++) {
        if (Student[i] != null && Student[i].equals(name)) {
            // Shift elements left
            for (int j = i; j < Student.length - 1; j++) {
                Student[j] = Student[j + 1];
            }
            // Clear last element
            Student[Student.length - 1] = null;
            break; // Stop after removing one occurrence
        }
    }
}

// Add an element at the first empty position
static void addElement(String name) {
    for (int i = 0; i < Student.length; i++) {
        if (Student[i] == null) {
            Student[i] = name;
            break;
        }
    }
}
