package application;

import javafx.application.Application;
import javafx.stage.Stage;
import javafx.scene.Camera;
import javafx.scene.Group;
import javafx.scene.PerspectiveCamera;
import javafx.scene.Scene;
import javafx.scene.input.KeyEvent;
import javafx.scene.input.ScrollEvent;
import javafx.scene.layout.BorderPane;
import javafx.scene.paint.Color;
import javafx.scene.shape.Sphere;

public class Main extends Application {

    private static final int WIDTH = 900;
    private static final int HEIGHT = 600;

    // Tracks drag starting point for x and y
    private double anchorX, anchorY;
    private double cameraTranslateX, cameraTranslateY;

    @Override
    public void start(Stage primaryStage) {
        Sphere sphere = new Sphere(50);

        Group group = new Group();
        group.getChildren().add(sphere);

        Camera camera = new PerspectiveCamera();
        Scene scene = new Scene(group, WIDTH, HEIGHT);
        scene.setFill(Color.SILVER);
        scene.setCamera(camera);

        sphere.setTranslateX(WIDTH / 2);
        sphere.setTranslateY(HEIGHT / 2);

        initMouseControl(scene, camera, primaryStage, sphere);

        primaryStage.setTitle("Solar System Simulator");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    private void initMouseControl(Scene scene, Camera camera, Stage stage, Sphere sphere) {

        // When mouse is pressed, store the starting position and the current camera translation
        scene.setOnMousePressed(event -> {
            anchorX = event.getSceneX();
            anchorY = event.getSceneY();

            // Save the current camera translation when the drag starts
            cameraTranslateX = camera.getTranslateX();
            cameraTranslateY = camera.getTranslateY();
        });

        // When mouse is dragged, update the camera's position based on the difference
        scene.setOnMouseDragged(event -> {
            double deltaX = event.getSceneX() - anchorX;
            double deltaY = event.getSceneY() - anchorY;

            // Update camera's position relative to the starting translation
            camera.setTranslateX(cameraTranslateX + deltaX);
            camera.setTranslateY(cameraTranslateY + deltaY);
        });

        // Handle scroll to zoom
        stage.addEventHandler(ScrollEvent.SCROLL, event -> {
            sphere.setTranslateZ(sphere.getTranslateZ() + event.getDeltaY());
        });
    }

    public static void main(String[] args) {
        launch(args);
    }
}
