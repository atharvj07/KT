import java.math.BigInteger;
import java.util.random.RandomGenerator;

public class RandomBigIntegerExample {
    public static void main(String[] args) {
        int numBits = 128;  // Change this value to generate different bit-length numbers
        RandomGenerator rng = RandomGenerator.getDefault();  // Use a modern RandomGenerator

        BigInteger randomBigInt = new BigInteger(numBits, rng);
        
        System.out.println("Random BigInteger: " + randomBigInt);
    }
}

BigInteger randomBigInt = new BigInteger(numBits, rng.nextBytes(new byte[(numBits + 7) / 8]));
