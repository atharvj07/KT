import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ExecutionException;

public class AsyncExample {

    public static void main(String[] args) throws ExecutionException, InterruptedException {

        // Call async methods with different input ids
        CompletableFuture<Boolean> completableFuture1 = CompletableFuture.supplyAsync(() -> myMethod(1));
        CompletableFuture<Boolean> completableFuture2 = CompletableFuture.supplyAsync(() -> myMethod(2));
        CompletableFuture<Boolean> completableFuture3 = CompletableFuture.supplyAsync(() -> myMethod(3));
        CompletableFuture<Boolean> completableFuture4 = CompletableFuture.supplyAsync(() -> myMethod(4));

        // Combine all futures into an array
        CompletableFuture<?>[] futures = new CompletableFuture[]{
                completableFuture1, completableFuture2, completableFuture3, completableFuture4
        };

        // Continuously wait for a future that returns true
        while (true) {
            CompletableFuture<Object> anyOfFuture = CompletableFuture.anyOf(futures);

            // Process the first completed future
            Object result = anyOfFuture.get();  // Blocks until one completes
            if (result instanceof Boolean && (Boolean) result) {
                // When the result is true, we can break out and process
                System.out.println("Found true result!");
                break;
            } else {
                // If result is false, continue and wait for the next future
                System.out.println("Received false result, retrying...");
                // Remove the failed future and continue
                futures = removeCompletedFuture(futures, anyOfFuture);
            }
        }
    }

    // Simulate the method that returns boolean based on input
    public static boolean myMethod(int id) {
        // Simulate logic where sometimes it returns true or false
        System.out.println("Processing id: " + id);
        return id == 3;  // Assume 3 returns true, others return false
    }

    // Helper method to remove the completed future from the list
    public static CompletableFuture<?>[] removeCompletedFuture(CompletableFuture<?>[] futures, CompletableFuture<?> completedFuture) {
        CompletableFuture<?>[] newFutures = new CompletableFuture[futures.length - 1];
        int index = 0;
        for (CompletableFuture<?> future : futures) {
            if (future != completedFuture) {
                newFutures[index++] = future;
            }
        }
        return newFutures;
    }
}
