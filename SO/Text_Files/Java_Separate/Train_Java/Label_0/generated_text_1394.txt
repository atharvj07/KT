import org.jfree.chart.*;
import org.jfree.chart.plot.*;
import org.jfree.chart.annotations.*;
import org.jfree.data.xy.*;
import javax.swing.*;
import java.awt.*;
import java.awt.image.*;
import javax.imageio.*;

public class ChartWithFixedIcon {

    public static void main(String[] args) {
        // Sample data for the chart
        XYSeries series = new XYSeries("Data");
        series.add(1, 1);
        series.add(2, 3);
        series.add(3, 2);
        series.add(4, 4);

        XYSeriesCollection dataset = new XYSeriesCollection(series);
        JFreeChart chart = ChartFactory.createXYLineChart(
            "Chart with Fixed Icon", // chart title
            "X Axis",                // x axis label
            "Y Axis",                // y axis label
            dataset,                 // dataset
            PlotOrientation.VERTICAL,
            true,                    // include legend
            true,                    // tooltips
            false                    // urls
        );

        // Get the plot from the chart
        XYPlot plot = chart.getXYPlot();

        // Load the icon (triangle image)
        BufferedImage triangleIcon = null;
        try {
            triangleIcon = ImageIO.read(ChartWithFixedIcon.class.getClassLoader().getResource("Resources/Imagenes/triangulo.png"));
        } catch (IOException e) {
            e.printStackTrace();
        }

        // Position the icon at the fixed X value and the dynamic Y value
        XYImageAnnotation xyannotation = new XYImageAnnotation(0, 2, triangleIcon); // X=0 (left side), Y=2 (fixed value)

        // Add the annotation to the plot
        plot.addAnnotation(xyannotation);

        // Create a chart panel and display it
        ChartPanel chartPanel = new ChartPanel(chart);
        chartPanel.setPreferredSize(new Dimension(800, 600));
        JFrame frame = new JFrame("Chart with Fixed Icon");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.getContentPane().add(chartPanel);
        frame.pack();
        frame.setVisible(true);
    }
}
