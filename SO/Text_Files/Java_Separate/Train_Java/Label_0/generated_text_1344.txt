import java.util.Scanner;
import java.util.Objects;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String line = scanner.nextLine();
        
        // Split based on the delimiter ">>"
        String[] stringArr = line.split(">>");

        // Variable to hold the result
        int ask = 0;

        // Loop through each element in the string array
        for (int i = 0; i < stringArr.length; i++) {
            // Trim any leading or trailing spaces
            String currentWord = stringArr[i].trim();

            // Check the first word in each element
            if (currentWord.equals("radio")) {
                ask = 10;
                System.out.println(ask);  // Print ask immediately after setting it
            } else if (Objects.equals(currentWord, "tv")) {
                ask = 15;
                System.out.println(ask);  // Print ask immediately after setting it
            } else {
                System.out.println("Invalid media.");
            }
        }
    }
}
