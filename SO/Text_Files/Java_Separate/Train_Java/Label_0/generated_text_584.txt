// In InsertionSort.java
public class InsertionSort {
    // Other code...

    private int i = 1;
    private int j = 0;

    public void sortWithDelay(int arr[]) {
        int n = arr.length;

        if(i < n) {
            int key = arr[i];
            j = i - 1;

            // Move elements with delay
            if (j >= 0 && arr[j] > key) {
                arr[j + 1] = arr[j];
                arr[j] = key;
                j--;
            } else {
                i++;
            }
        }
    }

    // Other methods...
}
