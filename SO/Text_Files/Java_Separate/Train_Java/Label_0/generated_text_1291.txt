import jakarta.ws.rs.ext.ExceptionMapper;
import jakarta.ws.rs.core.Response;
import jakarta.validation.ValidationException;

@Provider
public class ValidationExceptionMapper implements ExceptionMapper<ValidationException> {
    @Override
    public Response toResponse(ValidationException exception) {
        return Response.status(Response.Status.BAD_REQUEST)
                .entity(new CCResponse("Validation error", exception.getMessage()))
                .build();
    }
}

import jakarta.ws.rs.ext.ExceptionMapper;
import jakarta.ws.rs.core.Response;
import javax.validation.ConstraintViolationException;

@Provider
public class GlobalExceptionMapper implements ExceptionMapper<Throwable> {
    @Override
    public Response toResponse(Throwable throwable) {
        if (throwable instanceof ConstraintViolationException) {
            ConstraintViolationException violationException = (ConstraintViolationException) throwable;
            // Customize error message and structure as needed
            return Response.status(Response.Status.BAD_REQUEST)
                    .entity(new CCResponse("Validation error", violationException.getMessage()))
                    .build();
        }

        // Default handling for other exceptions
        return Response.status(Response.Status.INTERNAL_SERVER_ERROR)
                .entity(new CCResponse("Something bad happened", throwable.getMessage()))
                .build();
    }
}
