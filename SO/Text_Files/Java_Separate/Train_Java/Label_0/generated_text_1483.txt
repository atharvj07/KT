import com.fasterxml.jackson.annotation.JsonCreator;
import com.fasterxml.jackson.annotation.JsonProperty;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.io.IOException;

record Person(Long id, String name, Integer age) {
    @JsonCreator
    public Person(@JsonProperty("id") Long id,
                  @JsonProperty("name") String name,
                  @JsonProperty("age") Integer age) {
        this.id = id;
        this.name = name;
        this.age = age;
    }
}

public class Main {
    public static void main(String[] args) throws IOException {
        String json = "{ \"person\": [1, \"Evgeni\", 35] }";

        ObjectMapper mapper = new ObjectMapper();
        PersonWrapper wrapper = mapper.readValue(json, PersonWrapper.class);

        System.out.println(wrapper.person());
    }
}

// Wrapper class to hold the array and extract its elements
class PersonWrapper {
    private final Person person;

    @JsonCreator
    public PersonWrapper(@JsonProperty("person") Object[] personArray) {
        this.person = new Person(
                (Long) personArray[0], 
                (String) personArray[1], 
                (Integer) personArray[2]
        );
    }

    public Person person() {
        return person;
    }
}

@JsonCreator
public Person(@JsonProperty("id") Long id,
              @JsonProperty("name") String name,
              @JsonProperty("age") Integer age) {
    this.id = id;
    this.name = name;
    this.age = age;
}
public static void main(String[] args) throws IOException {
    String json = "{ \"person\": [1, \"Evgeni\", 35] }";
    ObjectMapper mapper = new ObjectMapper();
    PersonWrapper wrapper = mapper.readValue(json, PersonWrapper.class);
    System.out.println(wrapper.person());
}
