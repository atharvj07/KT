   import io.micrometer.core.instrument.MeterRegistry;
   import io.micrometer.prometheus.PrometheusMeterRegistry;
   import org.springframework.boot.actuate.metrics.web.servlet.WebMvcMetricsFilter;
   import org.springframework.boot.actuate.endpoint.web.annotation.ServletEndpoint;
   import org.springframework.boot.actuate.web.exchanges.DefaultWebExchangeTagsProvider;
   import org.springframework.boot.context.event.ApplicationReadyEvent;
   import org.springframework.context.annotation.Bean;
   import org.springframework.context.annotation.Configuration;
   import org.springframework.context.event.EventListener;
   import org.springframework.web.bind.annotation.GetMapping;
   import org.springframework.web.bind.annotation.RestController;

   @Configuration
   public class PrometheusConfig {

       @Bean
       public PrometheusMeterRegistry prometheusRegistry() {
           return new PrometheusMeterRegistry(io.micrometer.prometheus.PrometheusConfig.DEFAULT);
       }

       // Custom endpoint for "thingworx"
       @Bean
       public PrometheusMeterRegistry thingworxRegistry() {
           return new PrometheusMeterRegistry(io.micrometer.prometheus.PrometheusConfig.DEFAULT);
       }

       // Custom actuator endpoint for Thingworx metrics
       @RestController
       public class ThingworxMetricsController {

           private final PrometheusMeterRegistry thingworxRegistry;

           public ThingworxMetricsController(PrometheusMeterRegistry thingworxRegistry) {
               this.thingworxRegistry = thingworxRegistry;
           }

           @GetMapping("/actuator/prometheus/thingworx")
           public String getThingworxMetrics() {
               return thingworxRegistry.scrape();
           }
       }

       // This method can be used to add custom Thingworx metrics at specific intervals
       @EventListener(ApplicationReadyEvent.class)
       public void addCustomMetrics() {
           // Example custom metrics to be added at intervals
           thingworxRegistry.gauge("thingworx.custom.metric", 1.0);
       }
   }
