import java.io.File;
import java.io.FilenameFilter;
import java.util.Arrays;
import java.util.Comparator;
import java.util.regex.Pattern;
import java.util.regex.Matcher;

public class FileFinder {

    public static File findLatestFile(String directoryPath, String prefix) {
        File dir = new File(directoryPath);
        if (!dir.exists() || !dir.isDirectory()) {
            System.out.println("Invalid directory: " + directoryPath);
            return null;
        }

        // Regex pattern for PREFIX_yyyy-mm-dd_hh-mm-ss
        String regex = prefix + "\\d{4}-\\d{2}-\\d{2}_\\d{2}-\\d{2}-\\d{2}"; 
        Pattern pattern = Pattern.compile(regex);

        // List and filter files
        File[] matchingFiles = dir.listFiles(new FilenameFilter() {
            @Override
            public boolean accept(File dir, String name) {
                Matcher matcher = pattern.matcher(name);
                return matcher.matches();
            }
        });

        // If no matching files found
        if (matchingFiles == null || matchingFiles.length == 0) {
            System.out.println("No matching files found.");
            return null;
        }

        // Sort files by last modified time (latest first)
        Arrays.sort(matchingFiles, Comparator.comparingLong(File::lastModified).reversed());

        // Return the latest file
        return matchingFiles[0];
    }

    public static void main(String[] args) {
        String directoryPath = "C:/path/to/your/files"; // Change to your actual directory
        String prefix = "PREFIX_";

        File latestFile = findLatestFile(directoryPath, prefix);
        if (latestFile != null) {
            System.out.println("Latest file found: " + latestFile.getName());
        } else {
            System.out.println("No file found matching the pattern.");
        }
    }
}
