import org.apache.poi.hssf.usermodel.*;
import org.apache.poi.ss.usermodel.*;

public class POITest {
    public static void main(String[] args) {
        try {
            // Create a new workbook
            HSSFWorkbook workBook = new HSSFWorkbook();
            HSSFSheet sheet = workBook.createSheet("Bank Statement");

            // Get the palette and set the custom color
            HSSFPalette palette = workBook.getCustomPalette();
            palette.setColorAtIndex((short) 1, (byte) 60, (byte) 120, (byte) 216);

            // Create the style for header cell
            CellStyle cellStyleHeader = workBook.createCellStyle();
            cellStyleHeader.setFillPattern(FillPatternType.SOLID_FOREGROUND);
            cellStyleHeader.setFillForegroundColor((short) 1); // Use the custom color index
            cellStyleHeader.setFont(createHeaderFont(workBook));

            // Create a row and cell
            HSSFRow row = sheet.createRow(0);
            HSSFCell cell = row.createCell(0);
            cell.setCellType(CellType.STRING);
            cell.setCellValue("Date");

            // Apply the style to the cell
            cell.setCellStyle(cellStyleHeader);

            // Write to file (optional)
            try (FileOutputStream fileOut = new FileOutputStream("workbook.xls")) {
                workBook.write(fileOut);
            }

            // Close the workbook
            workBook.close();

        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    // Helper function to create the font for header
    private static Font createHeaderFont(HSSFWorkbook workBook) {
        Font font = workBook.createFont();
        font.setBold(true);
        font.setColor(IndexedColors.WHITE.getIndex()); // White color for text
        return font;
    }
}
