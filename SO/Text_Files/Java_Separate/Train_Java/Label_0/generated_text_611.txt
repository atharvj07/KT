public static int pairsum(int arr[], int sum, int j) {
    int count = 0;
    int k = arr.length - 1;
    while (j < k) {
        if (arr[j] + arr[k] == sum) {
            if (arr[j] == arr[k]) {
                int n = k - j + 1;
                count = count + (n * (n - 1)) / 2;
                break;
            } else {
                count++;
                k--;
            }
        } else if (arr[j] + arr[k] > sum) {
            k--;
        } else {
            j++;
        }
    }
    return count;
}
