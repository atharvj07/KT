import ch.qos.logback.classic.Level;
import ch.qos.logback.classic.Logger;
import ch.qos.logback.classic.LoggerContext;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.slf4j.LoggerFactory;
import static org.junit.jupiter.api.Assertions.assertThrows;

@ExtendWith(LoggerContextExtension.class)
public class MyServiceTest {

    private static final Logger logger = (Logger) LoggerFactory.getLogger(MyService.class);

    @Test
    void isThisParameterGoodToUse() {
        // Backup old level
        Level oldLevel = logger.getLevel();

        try {
            // Temporarily disable logging (Set to OFF or WARN)
            logger.setLevel(Level.OFF);

            assertThrows(ServiceException.class, () -> {
                injectedService.isThisParameterGoodToUse("This is not a good parameter at all!");
            });

        } finally {
            // Restore old log level after test
            logger.setLevel(oldLevel);
        }
    }
}

    }
}
