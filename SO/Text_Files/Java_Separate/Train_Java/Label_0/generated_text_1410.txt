public void addNewDrug(Drug drug) {
    list.add(drug);
    
    if (list.size() > 0) {
        Log.d("actually works ffs", list.get(0).name);
        
        LinearLayout linearLayout = findViewById(R.id.rootLayout);
        
        // Check if LinearLayout is found
        if (linearLayout == null) {
            Log.e("MainActivity", "LinearLayout is null!");
            return;
        }

        // Clear the existing views to avoid adding new TextViews repeatedly
        linearLayout.removeAllViews();

        // Add TextViews based on the current list size
        for (int i = 0; i < list.size(); i++) {
            // Create TextView programmatically
            TextView textView = new TextView(this);
            textView.setLayoutParams(new LinearLayout.LayoutParams(ViewGroup.LayoutParams.WRAP_CONTENT, ViewGroup.LayoutParams.WRAP_CONTENT));
            textView.setGravity(Gravity.CENTER);
            textView.setText(list.get(i).name);  // Set the name of the drug

            // Add TextView to the LinearLayout
            linearLayout.addView(textView);
        }
        
        Log.d("Mainlist", list.get(0).name);
    }
}
