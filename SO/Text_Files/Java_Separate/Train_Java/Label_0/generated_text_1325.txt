import java.awt.*;
import java.awt.event.KeyEvent;
import java.util.Scanner;
import java.util.Timer;
import java.util.TimerTask;

public class Main {
    public static void main(String[] args) {
        try {
            // Create an instance of the Robot class to simulate key presses
            Robot robot = new Robot();

            // Create a TimerTask to simulate pressing "Enter" after 10 seconds
            TimerTask task = new TimerTask() {
                public void run() {
                    robot.keyPress(KeyEvent.VK_ENTER);
                    robot.keyRelease(KeyEvent.VK_ENTER);  // Don't forget to release the key
                    System.out.println("Time's up. Enter key pressed...");
                }
            };

            // Set up the Timer
            Timer timer = new Timer();
            timer.schedule(task, 10 * 1000); // Set the timer to trigger after 10 seconds

            // Use Scanner to take input from the user
            Scanner sc = new Scanner(System.in);
            System.out.println("Please enter some text within 10 seconds:");
            String in = sc.nextLine();  // Capture user input

            // Cancel the timer once the user finishes typing
            timer.cancel();

            // Print the captured input
            System.out.println("You entered: " + in);

        } catch (AWTException e) {
            e.printStackTrace();
        }
    }
}
