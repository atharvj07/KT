@SpringBootTest // Enables Spring Boot context
@TestPropertySource(properties = {
    "task.mode=KPI",
    "compaction.manually.triggered=false"
})
@ExtendWith(MockitoExtension.class) // Allows Mockito-based mocking
class MyRunnerTest {

    @MockBean // Ensures MyTask is mocked within the Spring context
    private MyTask myTask;

    @Autowired // Spring should inject the actual runner
    private MyRunner commandLineRunner;

    @Test
    void analyticsCleanupIsTriggeredOnlyManually() throws Exception {
        doNothing().when(myTask).executeTask(); // Prevent actual execution

        UnsupportedOperationException thrown = assertThrows(
            UnsupportedOperationException.class,
            () -> commandLineRunner.run()
        );

        assertTrue(thrown.getMessage().contains("Reporting compactor mode KPI is supported only using the parameter --compaction.manually.triggered=true."));
        verify(myTask, never()).executeTask(); // Ensure task execution was prevented
    }
}
