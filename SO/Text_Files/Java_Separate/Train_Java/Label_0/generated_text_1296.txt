public Node DFSTime(Node node) {
    if (node == null) {
        System.out.println("No path found");
        return null;
    }
    else if (node.element.equals(destinationAirport)) {
        return node;
    }
    else {
        stack.push(node);
        // Make the recursive call and return its result
        return DFSTime(node.nextNode);            
    }
}
