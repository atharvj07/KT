import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        
        System.out.println("1 - login");
        System.out.println("2 - register");

        // Declare the variable firstSelection only once
        int firstSelection = scanner.nextInt();

        // Validate the input inside the while loop
        while (firstSelection < 1 || firstSelection > 2) {
            System.out.println(firstSelection + " is not a valid entry. Please enter 1 or 2.");
            firstSelection = scanner.nextInt();  // Re-prompt the user for input
        }

        // Output the valid selection
        System.out.println("You picked " + firstSelection);
    }
}
