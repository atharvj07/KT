int month = dateDueInt / 1000000;      // Get the month (MM)
int day = (dateDueInt / 10000) % 100;  // Get the day (dd)
int year = dateDueInt % 10000;         // Get the year (yyyy)

// Create a Calendar instance and set the extracted values
Calendar calendar = Calendar.getInstance();
calendar.set(Calendar.MONTH, month - 1); // Calendar.MONTH is 0-based, so subtract 1
calendar.set(Calendar.DAY_OF_MONTH, day);
calendar.set(Calendar.YEAR, year);

// Format the date using SimpleDateFormat
String date = format.format(calendar.getTime());
