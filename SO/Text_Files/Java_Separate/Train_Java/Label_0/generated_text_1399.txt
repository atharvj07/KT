@SpringBootTest
@ContextConfiguration(classes = SecurityConfig.class)
@AutoConfigureMockMvc
@EnableAutoConfiguration
@Import(DataImporterControllerConfig.class)
public class SecurityConfigTest {

    @Autowired
    private MockMvc mockMvc;

    @MockBean
    private DataImporterService dataImporterService;  // Mock the service

    @Test
    void doImportExpect200() throws Exception {
        // Mock the service method behavior
        when(dataImporterService.doImport(anyString())).thenReturn("some value");  // Return a mocked value

        mockMvc.perform(put("/dataimporter/someurl")  // Make sure this matches the URI in your controller
                .with(jwt().authorities(new SimpleGrantedAuthority("SCOPE_data:write"))) // Mock JWT token with required scope
                .contentType(MediaType.APPLICATION_JSON)
                .accept(MediaType.APPLICATION_JSON)
                .content("{\"key\":\"value\"}"))  // Example body
                .andExpect(status().isOk());  // Assert 200 OK
    }

    @Test
    void doImportExpect401WhenNoToken() throws Exception {
        mockMvc.perform(put("/dataimporter/someurl")
                .contentType(MediaType.APPLICATION_JSON)
                .accept(MediaType.APPLICATION_JSON)
                .content("{\"key\":\"value\"}"))
                .andExpect(status().isUnauthorized());  // Expect 401 when no token is provided
    }

    @Test
    void doImportExpect403WhenWrongScope() throws Exception {
        mockMvc.perform(put("/dataimporter/someurl")
                .with(jwt().authorities(new SimpleGrantedAuthority("SCOPE_invalid")))  // Mock JWT token with invalid scope
                .contentType(MediaType.APPLICATION_JSON)
                .accept(MediaType.APPLICATION_JSON)
                .content("{\"key\":\"value\"}"))
                .andExpect(status().isForbidden());  // Expect 403 when token has wrong scope
    }
}
