import java.sql.*;

public class OracleFunctionCall {
    public static void main(String[] args) {
        Connection con = null;
        CallableStatement cs = null;

        String input = "{ \"p_principal\": " +
                "[ { \"p_emp\": 200, \"p_request\": 23, \"p_date\": \"23-10-2024\", \"p_info\": \"test\" } ], " +
                "\"p_detail\": [ { \"p_emp\": 200, \"p_date\": \"23-10-2024\", \"p_info\": \"test\" } ] }";

        try {
            // Establish connection
            con = DriverManager.getConnection("jdbc:oracle:thin:@10.25.25.25:1521/demoDB", "demo", "demo");

            // Prepare the call (function call)
            String query = "{ ? = call integ_pkg.insert_order(?) }";
            cs = con.prepareCall(query);

            // Register output parameter (CLOB)
            cs.registerOutParameter(1, Types.CLOB);

            // Set input parameter (CLOB)
            cs.setString(2, input);

            // Execute function
            cs.execute();

            // Retrieve and print the output
            Clob outputClob = cs.getClob(1);
            if (outputClob != null) {
                String result = outputClob.getSubString(1, (int) outputClob.length());
                System.out.println("Function Output: " + result);
            }

        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            // Close resources
            try {
                if (cs != null) cs.close();
                if (con != null) con.close();
            } catch (SQLException e) {
                e.printStackTrace();
            }
        }
    }
}
