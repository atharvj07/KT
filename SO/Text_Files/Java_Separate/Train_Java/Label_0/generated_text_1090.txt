private String filename = "myFile.txt";
private String filepath = "myFileDir"; // This should be a directory under getExternalFilesDir

public void onClick(final View view) {
    switch (view.getId()) {
        case R.id.camera_button:
            synchronized (mSync) {
                if (isTemp) {
                    tempTureing();
                    String fileContent = "Hello, I am a saved text inside a text file!";
                    if (!fileContent.isEmpty()) {
                        // Create the directory if it doesn't exist
                        File directory = new File(getExternalFilesDir(filepath));
                        if (!directory.exists()) {
                            directory.mkdirs(); // Create the directory if it doesn't exist
                        }

                        File myExternalFile = new File(directory, filename);
                        try (FileOutputStream fos = new FileOutputStream(myExternalFile, true)) { // true to append to the file
                            fos.write(fileContent.getBytes());
                            Log.e("TAG", "file written to: " + myExternalFile.getAbsolutePath());
                        } catch (Exception e) {
                            e.printStackTrace();
                        }
                    }
                    isTemp = false;
                } else {
                    stopTemp();
                    isTemp = true;
                }
            }
            break;
    }
}
