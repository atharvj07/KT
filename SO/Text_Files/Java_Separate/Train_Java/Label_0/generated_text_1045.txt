import java.sql.*;
import java.util.ArrayList;
import java.util.List;

public class LectureFetcher {
    private static final String URL = "jdbc:mysql://localhost/DB?serverTimezone=UTC";
    private static final String USER = "your_username";
    private static final String PASSWORD = "your_password";
    private static final String DRIVER_NAME = "com.mysql.cj.jdbc.Driver"; // Ensure you have the correct driver.

    public static List<String> getLectureData() {
        List<String> resultList = new ArrayList<>();
        String sql = "SELECT LectureDay, LectureStatus FROM lecturelist WHERE LectureName LIKE 'Java%'";

        try {
            Class.forName(DRIVER_NAME); // Load MySQL driver
            try (Connection conn = DriverManager.getConnection(URL, USER, PASSWORD);
                 PreparedStatement pstmt = conn.prepareStatement(sql);
                 ResultSet res = pstmt.executeQuery()) {

                while (res.next()) {
                    String lectureDay = res.getString("LectureDay");
                    String lectureStatus = res.getString("LectureStatus");
                    resultList.add(lectureDay + " - " + lectureStatus);
                }
            }
        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
        }
        return resultList;
    }

    public static void main(String[] args) {
        List<String> lectures = getLectureData();
        for (String lecture : lectures) {
            System.out.println(lecture);
        }
    }
}
