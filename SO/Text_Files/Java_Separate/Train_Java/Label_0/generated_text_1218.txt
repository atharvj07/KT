import java.util.Map;

public class Main {
    public static void main(String[] args) {
        // Sample data
        Map<String, Object> currencyRate = Map.of(
            "license", "https:/etc",
            "timestamp", 1654,
            "base", "USD",
            "rates", Map.of(
                "AED", 3.6,
                "AFN", 88.9,
                "ALL", 112
            )
        );

        // Extracting the "rates" map
        Map<String, Object> rates = (Map<String, Object>) currencyRate.get("rates");

        // Getting the value for "AED"
        if (rates != null && rates.containsKey("AED")) {
            double aedRate = (double) rates.get("AED");
            System.out.println("AED rate: " + aedRate);
        } else {
            System.out.println("AED rate not found.");
        }
    }
}
