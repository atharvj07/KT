import java.util.*;

public class Main {

    public static long solution(List<Integer> list, List<List<Integer>> ranges) {
        // Step 1: Create a list to store the "new list" by extracting ranges
        List<Integer> newList = new ArrayList<>();
        Set<Integer> contributingIndices = new HashSet<>();

        // Add elements from ranges to newList and track contributing indices
        for (List<Integer> range : ranges) {
            int start = range.get(0);
            int end = range.get(1);
            for (int i = start; i <= end; i++) {
                newList.add(list.get(i));
                contributingIndices.add(i);
            }
        }

        // Step 2: Create frequency map for the newList
        Map<Integer, Integer> freqMap = new HashMap<>();
        for (int num : newList) {
            freqMap.put(num, freqMap.getOrDefault(num, 0) + 1);
        }

        // Step 3: Create a sorted list of unique values in the new list
        List<Integer> sortedList = new ArrayList<>(freqMap.keySet());
        Collections.sort(sortedList);

        // Step 4: Build a prefix sum array to count how many numbers are smaller than a given number
        Map<Integer, Integer> prefixSum = new HashMap<>();
        int prefixCount = 0;
        for (int num : sortedList) {
            prefixSum.put(num, prefixCount);
            prefixCount += freqMap.get(num);
        }

        // Step 5: Calculate the result
        long result = 0;

        for (int i = 0; i < list.size(); i++) {
            if (!contributingIndices.contains(i)) {
                int value = list.get(i);
                // Count how many values in newList are less than the current value
                int smallerCount = prefixSum.getOrDefault(value - 1, 0);
                result += smallerCount;
            }
        }

        return result;
    }

    static void case1() {
        List<Integer> list = Arrays.asList(1, 2, 3, 4, 5);
        List<List<Integer>> ranges = Arrays.asList(
                Arrays.asList(0, 1),
                Arrays.asList(0, 2),
                Arrays.asList(1, 2)
        );
        System.out.println(solution(list, ranges)); // Output: 14
    }

    static void case2() {
        List<Integer> list = Arrays.asList(1, 2, 3, 4, 5);
        List<List<Integer>> ranges = Arrays.asList(
                Arrays.asList(1, 2),
                Arrays.asList(1, 1),
                Arrays.asList(2, 2),
                Arrays.asList(3, 3),
                Arrays.asList(4, 4)
        );
        System.out.println(solution(list, ranges)); // Output: 0
    }

    static void case3() {
        List<Integer> list = Arrays.asList(1, 2, 3, 2, 4, 5);
        List<List<Integer>> ranges = Arrays.asList(
                Arrays.asList(0,1),
                Arrays.asList(3,4),
                Arrays.asList(0,0),
                Arrays.asList(3,4)
        );
        System.out.println(solution(list, ranges)); // Output: 12
    }

    public static void main(String[] args) {
        case1();
        case2();
        case3();
    }
}
