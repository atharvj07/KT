       @Override
       public void postProcessBeanFactory(ConfigurableListableBeanFactory beanFactory) {

           ConfigurableEnvironment environment = context.getEnvironment();

           // Call the API to get properties dynamically
           Map<String, Object> dynamicProperties = fetchPropertiesFromAPI();

           // Add the dynamic properties to the environment as a new PropertySource
           PropertySource<?> dynamicPropertySource = new MapPropertySource("dynamicProperties", dynamicProperties);
           environment.getPropertySources().addFirst(dynamicPropertySource); // Add at the highest priority
       }

       }
