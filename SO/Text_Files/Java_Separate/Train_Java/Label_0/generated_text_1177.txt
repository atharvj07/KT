mCodeScanner.setDecodeCallback(new DecodeCallback() {
    @Override
    public void onDecoded(@NonNull final Result result) {
        runOnUiThread(new Runnable() {
            @Override
            public void run() {
                r = result.getText();  // This is the department scanned
                DatabaseReference ref = FirebaseDatabase.getInstance().getReference("students");

                // Query to find students with the given department
                Query query = ref.orderByChild("department").equalTo(r);

                // Listen for changes in the database
                query.addListenerForSingleValueEvent(new ValueEventListener() {
                    @Override
                    public void onDataChange(DataSnapshot dataSnapshot) {
                        // Check if the department exists in the database
                        if (dataSnapshot.exists()) {
                            // Extract matric numbers or other information from the snapshot
                            for (DataSnapshot snapshot : dataSnapshot.getChildren()) {
                                String matricNumber = snapshot.getKey(); // Get the key (matric number)
                                String block = snapshot.child("block").getValue(String.class);
                                String department = snapshot.child("department").getValue(String.class);

                                // Update UI or handle the result
                                name.setText("Matric Number: " + matricNumber + "\nBlock: " + block + "\nDepartment: " + department);
                            }
                        } else {
                            name.setText("No student found with department: " + r);
                        }
                    }

                    @Override
                    public void onCancelled(DatabaseError databaseError) {
                        name.setText("Error querying database: " + databaseError.getMessage());
                    }
                });
            }
        });
    }
});
