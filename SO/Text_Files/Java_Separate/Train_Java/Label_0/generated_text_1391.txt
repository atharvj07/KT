import com.fasterxml.jackson.databind.ObjectMapper;

import java.util.HashMap;
import java.util.Map;

public class Main {
    public static void main(String[] args) throws Exception {
        // Create a map with a string value and a map for the "c" key
        Map<String, Object> m = new HashMap<>();
        m.put("a", "b");

        // Create a map for the value of "c"
        Map<String, Object> nestedMap = new HashMap<>();
        nestedMap.put("a", 3);
        nestedMap.put("b", 5);

        // Put the nested map as the value of key "c"
        m.put("c", nestedMap);

        // Use ObjectMapper to serialize the map to a JSON string
        ObjectMapper objectMapper = new ObjectMapper();
        String jsonString = objectMapper.writeValueAsString(m);
        System.out.println(jsonString);
    }
}
