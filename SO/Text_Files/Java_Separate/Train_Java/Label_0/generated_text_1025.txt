import javax.swing.*;
import java.awt.*;

public class Main extends JPanel {

    // Override paintComponent to draw the chessboard
    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);

        int rectX = 0;
        int rectY = 0;

        // Loop through the 64 squares (8x8 board)
        for (int i = 1; i <= 64; i++) {

            // Draw alternating light and dark squares
            if ((i + rectY / 100) % 2 == 0) {
                g.fillRect(rectX, rectY, 100, 100);
            } else {
                g.drawRect(rectX, rectY, 100, 100);
            }

            rectX += 100;

            // Move to the next row after 8 squares
            if (i % 8 == 0) {
                rectX = 0;
                rectY += 100;
            }
        }
    }

    // Define preferred size
    @Override
    public Dimension getPreferredSize() {
        return new Dimension(800, 800);
    }

    // Create and show GUI
    private static void createAndShowGui() {
        JFrame frame = new JFrame("Java Chess V1.0");
        Main main = new Main();
        
        frame.getContentPane().add(main);
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.pack();
        frame.setLocationByPlatform(true);
        frame.setVisible(true);
    }

    public static void main(String[] args) {
        SwingUtilities.invokeLater(Main::createAndShowGui);
    }
}
