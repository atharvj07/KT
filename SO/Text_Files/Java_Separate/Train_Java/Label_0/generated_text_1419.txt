        System.loadLibrary("carlib");

#include <jni.h>
#include <string>
#include "Car.h"

extern "C" {

// JNI function to create a Car object
JNIEXPORT jlong JNICALL Java_com_example_CarJNI_createCar(JNIEnv *env, jobject obj) {
    // Create a Car instance and return a pointer to it (as jlong)
    Car* car = new Car();
    return reinterpret_cast<jlong>(car);
}

// JNI function to set the car model
JNIEXPORT void JNICALL Java_com_example_CarJNI_setCarModel(JNIEnv *env, jobject obj, jlong carPtr, jstring model) {
    // Convert the jstring to a C++ string
    const char *modelCStr = env->GetStringUTFChars(model, nullptr);
    std::string modelStr(modelCStr);

    // Get the Car pointer and set the model
    Car* car = reinterpret_cast<Car*>(carPtr);
    car->setModel(modelStr);

    // Release the jstring
    env->ReleaseStringUTFChars(model, modelCStr);
}

// JNI function to get the car model
JNIEXPORT jstring JNICALL Java_com_example_CarJNI_getCarModel(JNIEnv *env, jobject obj, jlong carPtr) {
    // Get the Car pointer and retrieve the model
    Car* car = reinterpret_cast<Car*>(carPtr);
    std::string model = car->getModel();

    // Return the model as a jstring
    return env->NewStringUTF(model.c_str());
}
