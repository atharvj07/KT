import java.util.*;
import java.util.function.Predicate;
import java.util.stream.Collectors;

public class FilterEmployee {

    public static void main(String[] args) {
        
        Employee emp1 = new Employee("John", 30, "Dallas", "HR");
        Employee emp2 = new Employee("Steve", 31, "Austin", "HR");
        Employee emp3 = new Employee("Andrew", 25, "Houston", "Finance");
        Employee emp4 = new Employee("Mike", 30, "Dallas", "HR");
        
        List<Employee> empList = new ArrayList<>();
        empList.add(emp1);
        empList.add(emp2);
        empList.add(emp3);
        empList.add(emp4);
        
        Person personObj = new Person(empList);
        
        Criteria c1 = new Criteria(CriteriaType.CITY, "Dallas");
        Criteria c2 = new Criteria(CriteriaType.NAME, "Mike");
        
        List<Criteria> criteriaList = new ArrayList<>();
        criteriaList.add(c1);
        criteriaList.add(c2);
        
        // Filter the employees based on criteria list
        List<Employee> filteredEmployees = filterEmployees(personObj, criteriaList);
        
        // Print the filtered employees
        filteredEmployees.forEach(emp -> System.out.println(emp.getName()));
    }

    public static List<Employee> filterEmployees(Person person, List<Criteria> criteriaList) {
        return person.getEmpList().stream()
                .filter(emp -> criteriaList.stream()
                        .allMatch(criteria -> matchesCriteria(emp, criteria)))
                .collect(Collectors.toList());
    }

    private static boolean matchesCriteria(Employee emp, Criteria criteria) {
        switch (criteria.getType()) {
            case NAME:
                return emp.getName().equalsIgnoreCase(criteria.getValue());
            case AGE:
                return Integer.toString(emp.getAge()).equals(criteria.getValue());
            case CITY:
                return emp.getCity().equalsIgnoreCase(criteria.getValue());
            case DEPT:
                return emp.getDept().equalsIgnoreCase(criteria.getValue());
            default:
                return false;
        }
    }
}
