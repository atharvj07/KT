import java.io.IOException;
import java.nio.file.Paths;

public class KeychainImporter {

    public static void importCertificateToMacOSKeychain(String certPath) throws IOException, InterruptedException {
        ProcessBuilder processBuilder = new ProcessBuilder(
                "security", "add-trusted-cert", "-d", "-r", "trustRoot",
                "-k", "/Library/Keychains/System.keychain", certPath
        );

        processBuilder.inheritIO();  // Redirects output to console
        Process process = processBuilder.start();
        int exitCode = process.waitFor();

        if (exitCode == 0) {
            System.out.println("Certificate imported successfully.");
        } else {
            System.err.println("Failed to import certificate. Exit code: " + exitCode);
        }
    }

    public static void main(String[] args) throws IOException, InterruptedException {
        String certPath = Paths.get(System.getProperty("user.home"), ".localhost-ssl", "localhost.crt").toString();
        importCertificateToMacOSKeychain(certPath);
    }
}
