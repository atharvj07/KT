import akka.actor.{Actor, ActorSystem, Props}

// Define the message type
case class ImageWork(imageId: Int, imageData: String)

// Producer actor (ImageWorkManager)
class ImageWorkManager(consumer: akka.actor.ActorRef) extends Actor {
  var imageIdCounter = 0

  def receive: Receive = {
    case "start" =>
      // Produce work for the consumer
      for (i <- 1 to 5) {
        val work = ImageWork(imageIdCounter, s"ImageData_$imageIdCounter")
        consumer ! work
        imageIdCounter += 1
        println(s"Produced work: ImageWork(${work.imageId}, ${work.imageData})")
      }
      context.stop(self) // Stop the producer after sending work
  }
}

// Consumer actor (ImageConverter)
class ImageConverter extends Actor {
  def receive: Receive = {
    case ImageWork(id, data) =>
      // Process the work
      println(s"Processing image with ID $id: $data")
      // Simulate processing time
      Thread.sleep(500)
      println(s"Finished processing image $id")
  }
}

// Main application
object MainApp extends App {
  // Create the ActorSystem
  val system = ActorSystem("ImageProcessingSystem")

  // Create the consumer actor
  val imageConverter = system.actorOf(Props[ImageConverter], "imageConverter")

  // Create the producer actor, passing the consumer actor reference
  val imageWorkManager = system.actorOf(Props(new ImageWorkManager(imageConverter)), "imageWorkManager")

  // Start the work flow
  imageWorkManager ! "start"

  // Shut down the system after all work is done
  system.terminate()
}
