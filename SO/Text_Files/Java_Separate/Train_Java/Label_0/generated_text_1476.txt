/**
 * A binary search tree (BST) with integer values.
 */
class Tree {

    /**
     * The value of this node in the tree.
     */
    private int node;

    /**
     * The left subtree.
     */
    private Tree lhs;

    /**
     * The right subtree.
     */
    private Tree rhs;

    /**
     * Creates a new tree with the value node.
     * @param node The value of this node in the tree.
     */
    public Tree(int node) {
        this.node = node;
        this.lhs = null;
        this.rhs = null;
    }

    /**
     * Method to add a node to the tree. Duplicates will be refused.
     *
     * @param insert the new number
     */
    public void add(int insert) {
        if (insert < node) {
            if (lhs == null) {
                lhs = new Tree(insert);
            } else {
                lhs.add(insert);
            }
        } else if (insert > node) { // Ignore duplicates
            if (rhs == null) {
                rhs = new Tree(insert);
            } else {
                rhs.add(insert);
            }
        }
    }

    /**
     * Method to calculate the depth of a tree.
     * The depth is the maximum number of edges to a leaf.
     *
     * @return the depth of the tree
     */
    public int depth() {
        // Base case: If no children, depth is 0
        if (lhs == null && rhs == null) {
            return 0;
        }

        int leftDepth = (lhs == null) ? 0 : lhs.depth();
        int rightDepth = (rhs == null) ? 0 : rhs.depth();

        return Math.max(leftDepth, rightDepth) + 1;
    }

    /**
     * Method to find a number in the tree.
     *
     * @param wanted the wanted number
     * @return true, if the wanted number exists in the tree, else false
     */
    public boolean exists(int wanted) {
        if (wanted == node) {
            return true;
        } else if (wanted < node && lhs != null) {
            return lhs.exists(wanted);
        } else if (wanted > node && rhs != null) {
            return rhs.exists(wanted);
        }
        return false;
    }

    /**
     * Method to find the smallest number in the tree.
     *
     * @return the smallest number in the tree
     */
    public int smallest() {
        if (lhs == null) {
            return node;
        }
        return lhs.smallest();
    }

    /**
     * Method to find the biggest number in the tree.
     *
     * @return the biggest number in the tree
     */
    public int biggest() {
        if (rhs == null) {
            return node;
        }
        return rhs.biggest();
    }

    /**
     * Method to check whether a tree is degenerate.
     * A tree is degenerate if every node has either no child or only one child.
     *
     * @return true if every node has either no or one child node, else false
     */
    public boolean isDegenerate() {
        if (lhs != null && rhs != null) {
            return false; // Has two children, so not degenerate
        }
        if (lhs != null) {
            return lhs.isDegenerate();
        }
        if (rhs != null) {
            return rhs.isDegenerate();
        }
        return true; // No children, so it's degenerate
    }

    /**
     * Method that formats the tree into a readable string.
     *
     * @return the formatted string
     */
    @Override
    public String toString() {
        String leftStr = (lhs == null) ? "" : lhs.toString() + " ";
        String rightStr = (rhs == null) ? "" : " " + rhs.toString();
        return leftStr + node + rightStr;
    }
}
