@Data
@AllArgsConstructor
@NoArgsConstructor
@Builder
public class DemoRequest {
    private String myId;
    private String myFirstName;
    private String myLastName;
    private ArrayList<DemoPackage> demoPackages;
}

@Data
@AllArgsConstructor
@NoArgsConstructor
@Builder
public class DemoPackage {
    private String myparts;
    private ArrayList<String> myPackages; // Ensuring it's an ArrayList of Strings
}

public Object createMyPayload(String myId, String myFirstName, String myLastName, List<String> myparts, List<String> myPackages) {
    // Build the DemoPackage object
    DemoPackage demoPackage = DemoPackage.builder()
                                         .myparts(myparts.get(0))  // Assuming one part
                                         .myPackages(new ArrayList<>(myPackages)) // Convert List to ArrayList
                                         .build();

    // Build the DemoRequest object
    return DemoRequest.builder()
                      .myId(myId)
                      .myFirstName(myFirstName)
                      .myLastName(myLastName)
                      .demoPackages(new ArrayList<>(List.of(demoPackage))) // Wrap demoPackage in ArrayList
                      .build();
}
