private static BufferedImage changeTo8BitDepth(BufferedImage bi) {
       ColorModel cm = bi.getColorModel();
       boolean hasAlpha = cm.hasAlpha();
       boolean isAlphaPre = cm.isAlphaPremultiplied();
       int transparency = cm.getTransparency();
       ColorSpace cs = cm.getColorSpace();
       
       // Convert the 16-bit to 8-bit by rescaling pixel values
       int width = bi.getWidth();
       int height = bi.getHeight();
       WritableRaster raster = bi.getRaster();
       
       // Create a new BufferedImage for 8-bit output
       BufferedImage newBi = new BufferedImage(bi.getWidth(), bi.getHeight(), BufferedImage.TYPE_BYTE_GRAY);
       WritableRaster newRaster = newBi.getRaster();
       
       for (int y = 0; y < height; y++) {
           for (int x = 0; x < width; x++) {
               int[] pixel = raster.getPixel(x, y, new int[bi.getRaster().getNumBands()]);
               
               // Assuming each channel is 16-bit, scale to 8-bit (16-bit values are 0-65535)
               for (int i = 0; i < pixel.length; i++) {
                   pixel[i] = Math.min(255, (pixel[i] >> 8)); // Rescale from 16-bit to 8-bit
               }
               newRaster.setPixel(x, y, pixel);
           }
       }
       
       return newBi;
   }
   
   private static BufferedImage changeTo8BitDepthWithProfile(BufferedImage bi) {
       ColorModel cm = bi.getColorModel();
       boolean hasAlpha = cm.hasAlpha();
       boolean isAlphaPre = cm.isAlphaPremultiplied();
       int transparency = cm.getTransparency();
       ColorSpace cs = cm.getColorSpace();
       
       // Assuming we want to preserve the color profile
       ColorProfile profile = bi.getColorModel().getColorProfile();
       
       // Perform the conversion while preserving the profile
       BufferedImage newBi = new BufferedImage(bi.getWidth(), bi.getHeight(), BufferedImage.TYPE_BYTE_GRAY);
       newBi.setColorModel(new ComponentColorModel(cs, hasAlpha, isAlphaPre, transparency, DataBuffer.TYPE_BYTE));
       newBi.setColorProfile(profile);
       
       return newBi;
   }
