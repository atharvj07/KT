public class MyService {

    private EntityManagerFactory entityManagerFactory;

    public MyService(EntityManagerFactory entityManagerFactory) {
        this.entityManagerFactory = entityManagerFactory;
    }

    public String getEntityName(int id) {
        EntityManager em = entityManagerFactory.createEntityManager();
        Entity entity = em.find(Entity.class, id);
        return entity.getName();
    }
}

import static org.mockito.Mockito.*;
import static org.junit.Assert.*;

import javax.persistence.EntityManager;
import javax.persistence.EntityManagerFactory;
import org.junit.Before;
import org.junit.Test;

public class MyServiceTest {

    private EntityManagerFactory mockEntityManagerFactory;
    private EntityManager mockEntityManager;
    private MyService myService;

    @Before
    public void setUp() {
        // Mocking EntityManagerFactory and EntityManager
        mockEntityManagerFactory = mock(EntityManagerFactory.class);
        mockEntityManager = mock(EntityManager.class);
        
        // Setup mock behavior
        when(mockEntityManagerFactory.createEntityManager()).thenReturn(mockEntityManager);

        // Initialize the service with mocked dependencies
        myService = new MyService(mockEntityManagerFactory);
    }

    @Test
    public void testGetEntityName() {
        // Mock the behavior of the EntityManager
        Entity mockEntity = mock(Entity.class);
        when(mockEntityManager.find(Entity.class, 1)).thenReturn(mockEntity);
        when(mockEntity.getName()).thenReturn("Test Entity");

        // Call the method under test
        String entityName = myService.getEntityName(1);

        // Assert the result
        assertEquals("Test Entity", entityName);
        
        // Verify interactions
        verify(mockEntityManagerFactory).createEntityManager();
        verify(mockEntityManager).find(Entity.class, 1);
    }
}
