public class DriverConfig {

    public static WebDriver driver;

    public static void setupBrowser() {
        if (driver == null) {
            System.setProperty("webdriver.chrome.driver", "D:\\eclipse-workspace\\SkyFramework\\src\\test\\java\\chromedriver.exe");
            driver = new ChromeDriver();  
        }
    }

    public static WebDriver getDriver() {
        return driver;
    }

    public static void quitDriver() {
        if (driver != null) {
            driver.quit();
        }
    }
}

public class Deals {

    static DriverConfig driverConfig = new DriverConfig();
    static HomePage home = new HomePage(DriverConfig.getDriver());
    static DealsPage deals = new DealsPage(DriverConfig.getDriver());
    static GenericFunctions genericFunctions = new GenericFunctions(DriverConfig.getDriver());

    @Given("I am on the home page")
    public void i_am_on_the_home_page() {
        driverConfig.setupBrowser();
        String strPageName = "home";
        String url = genericFunctions.getUrl(strPageName);
        genericFunctions.startBrowser(DriverConfig.getDriver(), url);
        genericFunctions.acceptCookies(DriverConfig.getDriver());
    }

    @When("^I navigate to ([^\"]*)$")
    public void i_navigate_to_page(String strPage) {
        genericFunctions.navigateTo(DriverConfig.getDriver(), strPage);
    }

    @Then("^the user should be on ([^\"]*)$")
    public void the_user_should_be_on_page(String strPage) {
        deals.verifyDealsPage(DriverConfig.getDriver(), strPage);
        DriverConfig.quitDriver();
    }
}

public class SignIn {

    static DriverConfig driverConfig = new DriverConfig();
    static GenericFunctions genericFunctions = new GenericFunctions(DriverConfig.getDriver());
    static HomePage home = new HomePage(DriverConfig.getDriver());
    
    By createPassText = By.xpath("//a[contains(.,'Create Password')]");

    @When("I try to sign in with invalid credentials")
    public void i_try_to_sign_in_with_invalid_credentials() {
        home.selectSignInButton(DriverConfig.getDriver());
    }

    @Then("I should see a box with the text Create your My Sky password")
    public void i_should_see_a_box_with_the_text_create_your_my_sky_password() {
        boolean passTextExists = DriverConfig.getDriver().findElement(createPassText).isDisplayed();

        if (passTextExists) {
            System.out.println("YES THE TEXT EXISTS");
            DriverConfig.getDriver().findElement(createPassText).click();
        }
    }
}
