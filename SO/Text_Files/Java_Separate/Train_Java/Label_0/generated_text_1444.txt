import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    public static ArrayList<String> getTrim() {
        Scanner t = new Scanner(System.in);
        ArrayList<String> trim = new ArrayList<>();

        System.out.println("Enter car trims (press Enter to finish):");

        while (true) {
            String input = t.nextLine().trim(); // Read input and remove extra spaces
            if (input.isEmpty()) { // Stop if the user just presses Enter
                break;
            }
            trim.add(input);
        }

        return trim;
    }

    public static void main(String[] args) {
        ArrayList<String> trims = getTrim();
        System.out.println("Trims entered: " + trims);
    }
}
