public void asyncHttp(HTTPMethod method,
                      String url,
                      String body,
                      Map<String, String> headers) {

    WebClient.UriSpec<WebClient.RequestBodySpec> uriSpec = webClient.method(method.getMethod());
    WebClient.RequestBodySpec bodySpec = uriSpec.uri(url);
    WebClient.RequestHeadersSpec<?> headersSpec = bodySpec.bodyValue(body);
    
    // Adding headers
    for (Map.Entry<String, String> entry : headers.entrySet()) {
        headersSpec.header(entry.getKey(), entry.getValue());
    }

    // Initiate the asynchronous request
    Mono<JsonObject> responseMono = headersSpec.retrieve().bodyToMono(JsonObject.class);
    
    // Subscribe to handle the response asynchronously
    responseMono.subscribe(response -> {
        // Process the response once it's received
        System.out.println("Response received: " + response);
        // Further processing...
    }, error -> {
        // Handle error
        System.err.println("Error: " + error.getMessage());
    });
}

import java.time.Duration;

public void asyncHttp(HTTPMethod method,
                      String url,
                      String body,
                      Map<String, String> headers) {

    WebClient.UriSpec<WebClient.RequestBodySpec> uriSpec = webClient.method(method.getMethod());
    WebClient.RequestBodySpec bodySpec = uriSpec.uri(url);
    WebClient.RequestHeadersSpec<?> headersSpec = bodySpec.bodyValue(body);
    
    // Adding headers
    for (Map.Entry<String, String> entry : headers.entrySet()) {
        headersSpec.header(entry.getKey(), entry.getValue());
    }

    // Initiate the asynchronous request with a timeout of 5 seconds
    Mono<JsonObject> responseMono = headersSpec.retrieve()
                                               .bodyToMono(JsonObject.class)
                                               .timeout(Duration.ofSeconds(5));  // Set timeout
    
    // Subscribe to handle the response asynchronously
    responseMono.subscribe(response -> {
        // Process the response once it's received
        System.out.println("Response received: " + response);
        // Further processing...
    }, error -> {
        // Handle error (e.g., timeout or network error)
        if (error instanceof TimeoutException) {
            System.err.println("Request timed out");
        } else {
            System.err.println("Error: " + error.getMessage());
        }
    });
}
