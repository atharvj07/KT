(?:<email>)?([\w!#$%&'*+/=?`{|}~^-]+(?:\.[\w!#$%&'*+/=?`{|}~^-]+)*@(?:[a-zA-Z0-9-]+\.)+[a-zA-Z]{1,6})(?:</email>)?

import java.util.regex.*;

public class EmailExtractor {
    public static void main(String[] args) {
        String text1 = "paris@france.c";
        String text2 = "<email>paris@france.c</email>";

        String regex = "(?:<email>)?([\\w!#$%&'*+/=?`{|}~^-]+(?:\\.[\\w!#$%&'*+/=?`{|}~^-]+)*@(?:[a-zA-Z0-9-]+\\.)+[a-zA-Z]{1,6})(?:</email>)?";

        Pattern pattern = Pattern.compile(regex);
        
        for (String text : new String[]{text1, text2}) {
            Matcher matcher = pattern.matcher(text);
            if (matcher.find()) {
                System.out.println("Extracted email: " + matcher.group(1));
            } else {
                System.out.println("No match found");
            }
        }
    }
}
