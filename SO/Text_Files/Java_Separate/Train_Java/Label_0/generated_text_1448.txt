import com.opensymphony.xwork2.TextProvider;
import com.opensymphony.xwork2.util.LocalizedTextProvider;
import org.apache.logging.log4j.LogManager;
import org.apache.logging.log4j.Logger;

import java.util.List;
import java.util.Locale;

public class CustomTextProvider implements TextProvider {
    private static final Logger LOGGER = LogManager.getLogger(CustomTextProvider.class);

    private final LocalizedTextProvider localizedTextProvider;

    public CustomTextProvider(LocalizedTextProvider localizedTextProvider) {
        this.localizedTextProvider = localizedTextProvider;
    }

    @Override
    public String getText(String key) {
        return localizedTextProvider.findDefaultText(key, Locale.getDefault());
    }

    @Override
    public String getText(String key, String defaultValue) {
        return localizedTextProvider.findDefaultText(key, Locale.getDefault(), defaultValue);
    }

    @Override
    public String getText(String key, List<?> args) {
        return localizedTextProvider.findDefaultText(key, Locale.getDefault(), args.toArray());
    }

    @Override
    public String getText(String key, String defaultValue, List<?> args) {
        return localizedTextProvider.findDefaultText(key, Locale.getDefault(), args.toArray(), defaultValue);
    }

    @Override
    public Locale getLocale() {
        return Locale.getDefault();
    }
}

import com.opensymphony.xwork2.inject.Inject;
import com.opensymphony.xwork2.util.LocalizedTextProvider;

public class CustomTextProviderFactory {
    private static LocalizedTextProvider localizedTextProvider;

    @Inject
    public static void setLocalizedTextProvider(LocalizedTextProvider provider) {
        localizedTextProvider = provider;
    }

    public static CustomTextProvider create() {
        return new CustomTextProvider(localizedTextProvider);
    }
}

public class TextProviderSupport implements TextProvider {
    private static final Logger LOGGER = LogManager.getLogger(TextProviderSupport.class);

    private final CustomTextProvider customTextProvider = CustomTextProviderFactory.create();

    @Override
    public String getText(String key, List<?> args) {
        return customTextProvider.getText(key, args);
    }

    @Override
    public String getText(String key, String defaultValue, List<?> args) {
        return customTextProvider.getText(key, defaultValue, args);
    }

    @Override
    public Locale getLocale() {
        return customTextProvider.getLocale();
    }
}

private static class I18nHandler implements PreResultListener {
    public void beforeResult(final ActionInvocation invocation, final String resultCode) {
        invocation.getStack().push(new TextProviderSupport());
    }
}
