public void draw(SpriteBatch batch) {
    // Get the current batch transformation matrix
    Matrix4 originalMatrix = batch.getTransformMatrix().cpy(); // Save original matrix

    // Compute the position of the NinePatch
    float x = body.getPosition().x - widthInWorld / 2;
    float y = body.getPosition().y - heightInWorld / 2;

    // Create a transformation matrix for rotating around the center
    Matrix4 tempMatrix = new Matrix4();
    tempMatrix.translate(widthInWorld / 2, heightInWorld / 2, 0) // Translate to center
            .rotate(0, 0, 1, body.getAngle() * MathUtils.radiansToDegrees) // Apply rotation
            .translate(-widthInWorld / 2, -heightInWorld / 2, 0); // Translate back to original position

    // Set the transformed matrix once for the entire batch
    batch.setTransformMatrix(tempMatrix);

    // Draw the NinePatch at the computed position
    ninePatchDrawable.draw(
        batch,
        x,
        y,
        0,
        0,
        widthInWorld,
        heightInWorld,
        1,
        1,
        0
    );

    // Restore the original batch transform matrix
    batch.setTransformMatrix(originalMatrix);
}
