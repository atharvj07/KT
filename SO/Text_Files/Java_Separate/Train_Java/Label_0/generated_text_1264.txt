import javax.xml.ws.BindingProvider;
import java.util.Map;

@RestController
@RequestMapping("/codes")
public class CodigosRestController {

    private static final String API_KEY = "your-api-key-here"; // Replace with your actual API key

    @GetMapping
    public String obtainCode() {
        CodesService_Service service = new CodesService_Service();
        CodesService port = service.getCodesServicePort();

        // Add API key to the request headers
        Map<String, Object> requestContext = ((BindingProvider) port).getRequestContext();
        requestContext.put(BindingProvider.ENDPOINT_ADDRESS_PROPERTY, "https://codes.thewsdlprovider.xyz/v2/Codes");
        requestContext.put(BindingProvider.USERNAME_PROPERTY, API_KEY); // If API key is passed as a username
        requestContext.put("javax.xml.ws.http.request.headers", Map.of("X-API-KEY", API_KEY)); // If API key is a header

        return port.methodTwo().toString();
    }
}

import javax.xml.namespace.QName;
import javax.xml.soap.SOAPElement;
import javax.xml.soap.SOAPException;
import javax.xml.soap.SOAPMessage;
import javax.xml.ws.handler.MessageContext;
import javax.xml.ws.handler.soap.SOAPHandler;
import javax.xml.ws.handler.soap.SOAPMessageContext;
import java.util.Collections;
import java.util.Set;

public class ApiKeySOAPHandler implements SOAPHandler<SOAPMessageContext> {
    private static final String API_KEY = "your-api-key-here"; // Replace with your actual API key

    @Override
    public boolean handleMessage(SOAPMessageContext context) {
        Boolean outboundProperty = (Boolean) context.get(MessageContext.MESSAGE_OUTBOUND_PROPERTY);
        if (outboundProperty) {
            try {
                SOAPMessage message = context.getMessage();
                SOAPElement securityHeader = message.getSOAPHeader()
                        .addChildElement(new QName("https://thewsdlprovider.xyz/", "Security"));
                SOAPElement apiKeyElement = securityHeader.addChildElement("APIKey");
                apiKeyElement.addTextNode(API_KEY);
                message.saveChanges();
            } catch (SOAPException e) {
                e.printStackTrace();
            }
        }
        return true;
    }

    @Override
    public boolean handleFault(SOAPMessageContext context) {
        return true;
    }

    @Override
    public void close(MessageContext context) {}

    @Override
    public Set<QName> getHeaders() {
        return Collections.emptySet();
    }
}

import javax.xml.ws.handler.HandlerResolver;
import javax.xml.ws.handler.PortInfo;
import java.util.List;

public class ApiKeyHandlerResolver implements HandlerResolver {
    @Override
    public List<javax.xml.ws.handler.Handler> getHandlerChain(PortInfo portInfo) {
        return List.of(new ApiKeySOAPHandler());
    }
}

@RestController
@RequestMapping("/codes")
public class CodigosRestController {

    @GetMapping
    public String obtainCode() {
        CodesService_Service service = new CodesService_Service();
        service.setHandlerResolver(new ApiKeyHandlerResolver()); // Attach the handler
        CodesService port = service.getCodesServicePort();

        return port.methodTwo().toString();
    }
}
