import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.IntStream;

public class ParallelProcessingExample {
    public static void main(String[] args) {
        // Creating a list of 1000 Estimates objects
        List<Estimates> estimateList = IntStream.range(0, 1000)
                .mapToObj(i -> {
                    Estimates estimate = new Estimates();
                    estimate.setMarket("Market_" + i);
                    return estimate;
                }).collect(Collectors.toList());

        // Parallel processing using parallelStream()
        estimateList.parallelStream().forEach(estimate -> {
            Account account = performCalculation(estimate);
            estimate.setAccount(account);
        });

        // Verify results (Printing first 5)
        estimateList.stream().limit(5).forEach(e ->
            System.out.println(e.getMarket() + " -> " + e.getAccount().getName())
        );
    }

    // Simulated computation that generates an Account object
    private static Account performCalculation(Estimates estimate) {
        Account account = new Account();
        account.setName("Calculated_Name_" + estimate.getMarket());
        account.setCompany("Company_" + estimate.getMarket());
        return account;
    }
}

import java.util.List;
import java.util.concurrent.*;

public class ParallelProcessingExecutor {
    public static void main(String[] args) throws InterruptedException {
        List<Estimates> estimateList = IntStream.range(0, 1000)
                .mapToObj(i -> {
                    Estimates estimate = new Estimates();
                    estimate.setMarket("Market_" + i);
                    return estimate;
                }).collect(Collectors.toList());

        ExecutorService executor = Executors.newFixedThreadPool(10); // 10 threads

        for (Estimates estimate : estimateList) {
            executor.submit(() -> {
                Account account = performCalculation(estimate);
                estimate.setAccount(account);
            });
        }

        executor.shutdown();
        executor.awaitTermination(1, TimeUnit.MINUTES);

        // Print first 5 results
        estimateList.stream().limit(5).forEach(e ->
            System.out.println(e.getMarket() + " -> " + e.getAccount().getName())
        );
    }

    private static Account performCalculation(Estimates estimate) {
        Account account = new Account();
        account.setName("Calculated_Name_" + estimate.getMarket());
        account.setCompany("Company_" + estimate.getMarket());
        return account;
    }
}
