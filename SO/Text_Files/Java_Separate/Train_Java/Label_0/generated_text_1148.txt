import com.opencsv.CSVReader;
import com.opencsv.exceptions.CsvValidationException;
import org.openqa.selenium.*;
import org.openqa.selenium.ie.InternetExplorerDriver;
import org.openqa.selenium.interactions.Actions;

import java.io.FileReader;
import java.io.IOException;

public class DataDrivenTestingUsingCSVInSelenium {
    
    // Provide test data CSV file path.
    String CSV_PATH = "C:\\Users\\SAAD\\Desktop\\Cam. Inventory.csv.csv";
    WebDriver driver;
    private CSVReader csvReader;
    
    public void setup() throws Exception {
        // Setup the WebDriver (Make sure the path to IEDriverServer is correct)
        System.setProperty("webdriver.ie.driver", "C:\\Users\\SAAD\\Downloads\\IEDriverServer_Win32_4.0.0\\IEDriverServer.exe");
        driver = new InternetExplorerDriver();  // Use the class-level driver variable
    }
    
    public void dataRead_CSV() throws IOException, CsvValidationException {
        // Create an object of CSVReader and open the CSV file
        csvReader = new CSVReader(new FileReader(CSV_PATH));
        
        // Read the CSV file line by line
        String[] csvCell;
        while ((csvCell = csvReader.readNext()) != null) {
            String ip = csvCell[0];  // IP address from CSV
            String pass = csvCell[1];  // Password from CSV

            // Navigate to the IP address page
            driver.get("http://" + ip + "/doc/page/config.asp");
            driver.manage().window().maximize();
            
            // Perform login action
            WebElement usernameField = driver.findElement(By.id("username"));
            usernameField.click();
            usernameField.sendKeys("admin");

            WebElement passwordField = driver.findElement(By.id("password"));
            passwordField.click();
            passwordField.sendKeys(pass);
            passwordField.sendKeys(Keys.RETURN);

            // Add any assertion or validation if needed here
            
            // Optional: pause or wait if needed before the next test iteration
            // Thread.sleep(2000);  // For example, wait for 2 seconds (not recommended in production code)
        }

        // Close the CSVReader
        csvReader.close();
    }
    
    public static void main(String[] args) throws Exception {
        DataDrivenTestingUsingCSVInSelenium test = new DataDrivenTestingUsingCSVInSelenium();
        test.setup();  // Setup WebDriver
        test.dataRead_CSV();  // Read and execute the test data from CSV
    }
}
