import java.text.DateFormat;
import java.text.SimpleDateFormat;
import java.util.Date;
import java.util.TimeZone;

public class Main {
    public static void main(String[] args) throws Exception {
        String value = "20220307150417";

        // Create formatter and set timezone to EET
        DateFormat formatter = new SimpleDateFormat("yyyyMMddHHmmss");
        formatter.setTimeZone(TimeZone.getTimeZone("EET"));

        // Parse date
        Date date = formatter.parse(value);

        // Print formatted date
        System.out.println(date.toString());
    }
}

SimpleDateFormat outputFormat = new SimpleDateFormat("EEE MMM dd HH:mm:ss z yyyy");
outputFormat.setTimeZone(TimeZone.getTimeZone("EET"));

System.out.println(outputFormat.format(date));  // Output: Mon Mar 07 15:04:17 EET 2022
