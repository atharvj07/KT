import javafx.scene.Scene;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class MainController {
    private Stage window;
    private MenuLeiste menuLeiste;
    private VBox rootLayout;

    public MainController(Stage primaryStage) {
        this.window = primaryStage;
        this.menuLeiste = new MenuLeiste();

        // Create a root layout that contains the menu
        this.rootLayout = new VBox();
        rootLayout.getChildren().add(menuLeiste.getMenuBar());
    }

    public void showMainMenu() {
        VBox mainContent = new VBox(); // Placeholder for main menu content
        mainContent.getChildren().add(new javafx.scene.control.Label("Main Menu Content"));

        updateScene(mainContent);
    }

    public void showCredits() {
        VBox creditsView = new Credits(this).sceneView3();
        updateScene(creditsView);
    }

    private void updateScene(VBox newContent) {
        rootLayout.getChildren().removeIf(node -> node != menuLeiste.getMenuBar()); // Keep only menu
        rootLayout.getChildren().add(newContent);

        Scene scene = new Scene(rootLayout, 800, 600);
        window.setScene(scene);
        window.show();
    }
}

import javafx.scene.control.Menu;
import javafx.scene.control.MenuBar;
import javafx.scene.control.MenuItem;

public class MenuLeiste {
    private MenuBar menuBar;
    private MainController controller;

    public MenuLeiste() {
        this.menuBar = new MenuBar();
    }

    public void setController(MainController controller) {
        this.controller = controller;
        initializeMenu();
    }

    private void initializeMenu() {
        Menu dataMenu = new Menu("Data");
        MenuItem exitItem = new MenuItem("Exit");
        exitItem.setOnAction(e -> System.exit(0));

        Menu extrasMenu = new Menu("Extras");
        MenuItem creditsItem = new MenuItem("Credits");
        creditsItem.setOnAction(e -> controller.showCredits()); // Call controller method

        extrasMenu.getItems().addAll(creditsItem);
        dataMenu.getItems().addAll(exitItem);
        menuBar.getMenus().addAll(dataMenu, extrasMenu);
    }

    public MenuBar getMenuBar() {
        return menuBar;
    }
}

import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.layout.VBox;

public class Credits {
    private MainController controller;

    public Credits(MainController controller) {
        this.controller = controller;
    }

    public VBox sceneView3() {
        Label label = new Label("Thanks");
        Button backButton = new Button("Back");

        backButton.setOnAction(e -> controller.showMainMenu()); // Go back to main menu

        VBox layout = new VBox(20);
        layout.getChildren().addAll(label, backButton);
        return layout;
    }
}
