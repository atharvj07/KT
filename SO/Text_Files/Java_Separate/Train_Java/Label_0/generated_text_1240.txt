class Foo<T> {
    public void basedOnGeneric(T bar) {
        if (bar instanceof String) {
            GenericOverwrite.foo((String) bar); // Explicit cast
        } else {
            GenericOverwrite.foo(bar);
        }
    }
}

abstract class GenericOverwrite<T> {
    abstract void foo(T bar);
}

class StringOverwrite extends GenericOverwrite<String> {
    @Override
    void foo(String bar) {
        System.out.println("string");
    }
}

class ObjectOverwrite extends GenericOverwrite<Object> {
    @Override
    void foo(Object bar) {
        System.out.println("object");
    }
}

// Modify Foo to use this
class Foo<T> {
    private final GenericOverwrite<T> overwrite;

    Foo(GenericOverwrite<T> overwrite) {
        this.overwrite = overwrite;
    }

    public void basedOnGeneric(T bar) {
        overwrite.foo(bar);
    }
}

class Test {
    void test() {
        var a = new Foo<>(new StringOverwrite());
        a.basedOnGeneric("I wish this could print string"); // ? Prints "string"

        var b = new Foo<>(new ObjectOverwrite());
        b.basedOnGeneric("This should print object"); // ? Prints "object"
    }
}

import java.util.function.Consumer;

class Foo<T> {
    private final Consumer<T> function;

    Foo(Consumer<T> function) {
        this.function = function;
    }

    public void basedOnGeneric(T bar) {
        function.accept(bar);
    }
}

        a.basedOnGeneric("This should print string");

        b.basedOnGeneric("This should print object");
    }
}
