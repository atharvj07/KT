public String auth(String username, String password) {
    HttpURLConnection urlRequest = null;
    String token = null;

    try {
        URL url = new URL(BASE_URL + OPERATION_AUTH + "?key=" + firebaseKey);
        urlRequest = (HttpURLConnection) url.openConnection();
        urlRequest.setDoOutput(true);
        urlRequest.setRequestProperty("Content-Type", "application/json; charset=UTF-8");
        
        // Prepare the request payload
        String requestBody = "{\"email\": \"" + username + "\", \"password\": \"" + password + "\", \"returnSecureToken\": true}";
        
        // Send the request
        OutputStream os = urlRequest.getOutputStream();
        OutputStreamWriter osw = new OutputStreamWriter(os, "UTF-8");
        osw.write(requestBody);
        osw.flush();
        osw.close();
        
        urlRequest.connect();

        int responseCode = urlRequest.getResponseCode();

        if (responseCode == HttpURLConnection.HTTP_OK) {
            // Parse the success response
            JsonParser jp = new JsonParser();
            JsonElement root = jp.parse(new InputStreamReader((InputStream) urlRequest.getInputStream()));
            JsonObject rootObj = root.getAsJsonObject();

            // Extract the token
            token = rootObj.get("idToken").getAsString();
            System.out.println("Login successful. ID Token: " + token);
        } else {
            // Handle error response
            JsonParser jp = new JsonParser();
            JsonElement root = jp.parse(new InputStreamReader((InputStream) urlRequest.getErrorStream()));
            JsonObject rootObj = root.getAsJsonObject();

            String errorMessage = rootObj.getAsJsonObject("error").get("message").getAsString();
            System.err.println("Error during login: " + errorMessage);

            // Optionally, throw an exception or handle different errors as needed
            return null;  // Or you could return an error code/message if needed
        }

    } catch (IOException e) {
        e.printStackTrace();
        return null;  // Handle IO exceptions
    } finally {
        if (urlRequest != null) {
            urlRequest.disconnect();
        }
    }
    return token;
}
