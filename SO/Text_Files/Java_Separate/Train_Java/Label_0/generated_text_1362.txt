package com.example.SpringRecap;

import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.boot.autoconfigure.security.servlet.SecurityAutoConfiguration;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@SpringBootApplication(exclude = SecurityAutoConfiguration.class)
@RequestMapping("/api/v1/customers")
public class SpringRecapApplication {

    // Dependency injection
    private final CustomerRepository customerRepository;

    // Constructor injection
    public SpringRecapApplication(CustomerRepository customerRepository) {
        this.customerRepository = customerRepository;
    }

    public static void main(String[] args) {
        SpringApplication.run(SpringRecapApplication.class, args);
    }

    // Get all customers
    @GetMapping
    public List<Customer> getCustomer() {
        return customerRepository.findAll();
    }

    // Add a new customer
    @PostMapping
    public void addCustomer(@RequestBody NewCustomerRequest newCustomerRequest) {
        Customer customer = new Customer();
        customer.setAge(newCustomerRequest.age());
        customer.setName(newCustomerRequest.name());
        customer.setEmail(newCustomerRequest.email());
        customerRepository.save(customer);
    }

    // NewCustomerRequest DTO (this is a simple record for data transfer)
    public static class NewCustomerRequest {
        private String name;
        private String email;
        private Integer age;

        // Constructor, Getters and Setters
        public NewCustomerRequest(String name, String email, Integer age) {
            this.name = name;
            this.email = email;
            this.age = age;
        }

        public String name() {
            return name;
        }

        public String email() {
            return email;
        }

        public Integer age() {
            return age;
        }

        public void setName(String name) {
            this.name = name;
        }

        public void setEmail(String email) {
            this.email = email;
        }

        public void setAge(Integer age) {
            this.age = age;
        }
    }
}
