import java.util.Scanner;

public class MatrixTraversal {

    public static void main(String[] args) {
        
        // Initialize Matrix randomly
        int R = 3;
        int C = 3;
        
        int[][] matrix = new int[R][C];
        
        // Fill the matrix with random values (0 or 1 for simplicity)
        for (int i = 0; i < matrix.length; i++) {
            for (int j = 0; j < matrix[i].length; j++) {
                matrix[i][j] = (int) (Math.random() * 10); // Random number between 0 and 9
            }
        }
        
        // Display the matrix for the user to see
        System.out.println("Matrix:");
        for (int i = 0; i < matrix.length; i++) {
            for (int j = 0; j < matrix[i].length; j++) {
                System.out.print(matrix[i][j] + " ");
            }
            System.out.println();
        }
        
        // Accept the starting position from the user
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter the starting row (0 to " + (R-1) + "): ");
        int startRow = sc.nextInt();
        System.out.println("Enter the starting column (0 to " + (C-1) + "): ");
        int startCol = sc.nextInt();
        
        // Call the function to start traversal from the given position
        System.out.println("Starting traversal from position (" + startRow + ", " + startCol + "):");
        boolean[][] visited = new boolean[R][C]; // To keep track of visited cells
        traverseMatrix(matrix, startRow, startCol, visited);
    }
    
    // DFS function to traverse the matrix
    public static void traverseMatrix(int[][] matrix, int row, int col, boolean[][] visited) {
        
        // Check if the current position is out of bounds or already visited
        if (row < 0 || col < 0 || row >= matrix.length || col >= matrix[0].length || visited[row][col]) {
            return;
        }
        
        // Mark the current cell as visited
        visited[row][col] = true;
        
        // Print the current cell's value
        System.out.print(matrix[row][col] + " ");
        
        // Traverse in all four directions (up, down, left, right)
        traverseMatrix(matrix, row - 1, col, visited); // Move up
        traverseMatrix(matrix, row + 1, col, visited); // Move down
        traverseMatrix(matrix, row, col - 1, visited); // Move left
        traverseMatrix(matrix, row, col + 1, visited); // Move right
    }
}
