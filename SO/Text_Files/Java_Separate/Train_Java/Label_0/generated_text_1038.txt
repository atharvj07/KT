private static void teePaarisPaaritudRek(int[] a, int i, int evenPos) {
    if (i == a.length) return;

    if (a[i] % 2 == 0) { // Found an even number
        int temp = a[i];

        // Shift all numbers between evenPos and i to the right
        for (int j = i; j > evenPos; j--) {
            a[j] = a[j - 1];
        }
        a[evenPos] = temp;

        // Move evenPos forward
        teePaarisPaaritudRek(a, i + 1, evenPos + 1);
    } else {
        teePaarisPaaritudRek(a, i + 1, evenPos);
    }
}
