import org.apache.poi.xwpf.usermodel.*;
import org.openxmlformats.schemas.wordprocessingml.x2006.main.*;

import java.io.FileOutputStream;

public class PageNumberFooter {
    public static void main(String[] args) throws Exception {
        XWPFDocument document = new XWPFDocument();
        CTSectPr sectPr = document.getDocument().getBody().addNewSectPr();
        XWPFHeaderFooterPolicy headerFooterPolicy = new XWPFHeaderFooterPolicy(document, sectPr);

        // Create Footer
        XWPFFooter footer = headerFooterPolicy.createFooter(XWPFHeaderFooterPolicy.DEFAULT);
        XWPFParagraph paragraph = footer.createParagraph();
        paragraph.setAlignment(ParagraphAlignment.CENTER); // Align center

        // Add "Page " text
        XWPFRun textRun = paragraph.createRun();
        textRun.setText("Page ");
        textRun.setFontSize(12);
        textRun.setColor("FF0000"); // Red color

        // Add Page Number Field
        XWPFRun fieldRun = paragraph.createRun();
        fieldRun.getCTR().addNewFldChar().setFldCharType(STFldCharType.BEGIN); // Field start
        XWPFRun instrText = paragraph.createRun();
        instrText.setText("PAGE");
        instrText.getCTR().addNewRPr().addNewRFonts().setAscii("Arial");
        instrText.setFontSize(12);
        instrText.setColor("FF0000"); // Apply color to page number
        instrText.getCTR().addNewInstrText().setStringValue(" PAGE "); // Field instruction
        paragraph.createRun().getCTR().addNewFldChar().setFldCharType(STFldCharType.END); // Field end

        // Save the document
        try (FileOutputStream out = new FileOutputStream("PageNumber.docx")) {
            document.write(out);
        }
        document.close();
        System.out.println("Document with styled page number created successfully!");
    }
}
