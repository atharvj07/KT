import org.apache.poi.xwpf.usermodel.*;

import org.apache.xmlbeans.XmlCursor;
import org.apache.xmlbeans.XmlCursor.TokenType;

import java.io.FileOutputStream;
import java.io.IOException;

public class WordWithHeaderImage {
    public static void main(String[] args) throws Exception {
        XWPFDocument document = new XWPFDocument();
        
        // Add header and footer policy
        CTSectPr sectPr = document.getDocument().getBody().addNewSectPr();
        XWPFHeaderFooterPolicy headerFooterPolicy = new XWPFHeaderFooterPolicy(document, sectPr);
        
        // First page header
        XWPFHeader header = document.createHeader(HeaderFooterType.FIRST);
        
        // Create a paragraph in the header for the image
        XWPFParagraph paragraph = header.createParagraph();
        paragraph.setSpacingBefore(0); // Remove space before the paragraph
        paragraph.setSpacingAfter(0); // Remove space after the paragraph
        paragraph.setAlignment(ParagraphAlignment.CENTER); // Center-align the image

        XWPFRun run = paragraph.createRun();

        // Add image to header (adjust the image size as needed)
        String imgFile = "path_to_your_image.png"; // replace with your image file
        run.addPicture(new FileInputStream(imgFile), XWPFDocument.PICTURE_TYPE_PNG, imgFile, Units.toEMU(600), Units.toEMU(100));  // Adjust width and height in EMU
        
        // Set margins of the header to zero for full use of space
        XmlCursor cursor = header.getCTR().newCursor();
        cursor.selectPath("./*");
        while (cursor.toNextSelection()) {
            if (cursor.getName().getLocalPart().equals("tblPr")) {
                cursor.removeXml();
            }
        }
        
        // Create the footer for all pages
        XWPFFooter footer = document.createFooter(HeaderFooterType.DEFAULT);
        
        // Write to file
        try (FileOutputStream out = new FileOutputStream("WordWithHeaderImage.docx")) {
            document.write(out);
        }
    }
}
