public interface ShapeVisitor {
    void visit(Square square);
    void visit(Triangle triangle);
    void visit(Circle circle);
}

public class Square implements Shape {
    private double side;

    public Square(double side) {
        this.side = side;
    }

    @Override
    public double calculateArea() {
        return side * side;
    }

    @Override
    public double calculatePerimeter() {
        return 4 * side;
    }

    public void squareSpecificMethod() {
        System.out.println("Square-specific method called.");
    }

    public void accept(ShapeVisitor visitor) {
        visitor.visit(this);
    }
}

public class Triangle implements Shape {
    private double side1, side2, side3;

    public Triangle(double side1, double side2, double side3) {
        this.side1 = side1;
        this.side2 = side2;
        this.side3 = side3;
    }

    @Override
    public double calculateArea() {
        double s = (side1 + side2 + side3) / 2;
        return Math.sqrt(s * (s - side1) * (s - side2) * (s - side3));
    }

    @Override
    public double calculatePerimeter() {
        return side1 + side2 + side3;
    }

    public void accept(ShapeVisitor visitor) {
        visitor.visit(this);
    }
}

public class Circle implements Shape {
    private double radius;

    public Circle(double radius) {
        this.radius = radius;
    }

    @Override
    public double calculateArea() {
        return Math.PI * radius * radius;
    }

    @Override
    public double calculatePerimeter() {
        return 2 * Math.PI * radius;
    }

    public void accept(ShapeVisitor visitor) {
        visitor.visit(this);
    }
}

public class ConcreteShapeVisitor implements ShapeVisitor {
    @Override
    public void visit(Square square) {
        square.squareSpecificMethod();
    }

    @Override
    public void visit(Triangle triangle) {
        System.out.println("Triangle visited, but no specific method.");
    }

    @Override
    public void visit(Circle circle) {
        System.out.println("Circle visited, but no specific method.");
    }
}

public class Main {
    public static void main(String[] args) {
        Shape square = new Square(5);
        Shape triangle = new Triangle(3, 4, 5);
        Shape circle = new Circle(3);

        ShapeVisitor visitor = new ConcreteShapeVisitor();

        if (square instanceof Square) {
            ((Square) square).accept(visitor);
        }

        if (triangle instanceof Triangle) {
            ((Triangle) triangle).accept(visitor);
        }

        if (circle instanceof Circle) {
            ((Circle) circle).accept(visitor);
        }
    }
}

import java.util.HashMap;
import java.util.Map;
import java.util.function.Consumer;

public class Main {
    public static void main(String[] args) {
        Shape square = new Square(5);
        Shape triangle = new Triangle(3, 4, 5);
        Shape circle = new Circle(3);

        Map<Class<? extends Shape>, Consumer<Shape>> shapeActions = new HashMap<>();
        shapeActions.put(Square.class, s -> ((Square) s).squareSpecificMethod());
        shapeActions.put(Triangle.class, s -> System.out.println("Triangle: No specific method"));
        shapeActions.put(Circle.class, s -> System.out.println("Circle: No specific method"));

        shapeActions.getOrDefault(square.getClass(), s -> System.out.println("Unknown shape")).accept(square);
        shapeActions.getOrDefault(triangle.getClass(), s -> System.out.println("Unknown shape")).accept(triangle);
        shapeActions.getOrDefault(circle.getClass(), s -> System.out.println("Unknown shape")).accept(circle);
    }
}

public abstract class BaseShape implements Shape {
    public void extendedBehavior() {
        System.out.println("Default behavior for all shapes.");
    }
}

public class Square extends BaseShape {
    private double side;

    public Square(double side) {
        this.side = side;
    }

    @Override
    public double calculateArea() {
        return side * side;
    }

    @Override
    public double calculatePerimeter() {
        return 4 * side;
    }

    public void squareSpecificMethod() {
        System.out.println("Square-specific method.");
    }

    @Override
    public void extendedBehavior() {
        squareSpecificMethod();
    }
}

public class Main {
    public static void main(String[] args) {
        BaseShape square = new Square(5);
        square.extendedBehavior(); // Calls squareSpecificMethod()
    }
}
