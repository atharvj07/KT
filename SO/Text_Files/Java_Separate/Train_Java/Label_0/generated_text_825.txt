import java.util.List;

public interface Projection {

    Team getTeam();
    List<DriverProjection> getDrivers();

    interface DriverProjection {
        Driver getDriver();
        List<CarProjection> getCars();
    }

    interface CarProjection {
        Car getCar();
    }

    interface Team {
        String getTeamId();
        String getName();
    }

    interface Driver {
        String getDriverId();
        String getName();
    }

    interface Car {
        String getCarId();
        String getName();
    }
}
