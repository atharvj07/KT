import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.util.stream.Collectors;

public class UnicodeProcessor {
    public static void main(String[] args) throws Exception {
        String rawString = "ÃÂ¡ÃÂ§ÃÂ±ÃÂ©ÃÂ¼ÃÂ¶";  // Example of incorrectly displayed characters

        // Fix the encoding (if necessary)
        String utf8String = new String(rawString.getBytes(StandardCharsets.ISO_8859_1), StandardCharsets.UTF_8);

        // Remove duplicates & sort Unicode characters properly
        String result = utf8String.codePoints()
                .distinct()
                .sorted()
                .mapToObj(Character::toString)
                .collect(Collectors.joining());

        // Print correctly in Eclipse Console
        System.out.println(result);

        // Save to a UTF-8 encoded file
        Files.write(Paths.get("output.txt"), result.getBytes(StandardCharsets.UTF_8));
    }
}
