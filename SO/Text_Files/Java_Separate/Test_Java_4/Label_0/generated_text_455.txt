import java.util.ArrayList;
public class Main {
public static void main(String[] args) {
ArrayList<String> collection=new ArrayList<>();
warmup();
long t1=System.nanoTime();

long t2=System.nanoTime();

long t3=System.nanoTime();

private static void warmup() {

}
}
