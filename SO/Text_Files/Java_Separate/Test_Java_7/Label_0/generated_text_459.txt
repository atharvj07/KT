import java.util.*;
import java.util.stream.*;

interface Ask<T> {
    Res<T> mk();
}

class Res<T> {
    T val;
}

class ARes extends Res<String> {
    ARes(String v) {
        this.val = v;
    }
}

class BRes extends Res<Integer> {
    BRes(Integer v) {
        this.val = v;
    }
}

class AAsk implements Ask<String> {
    public ARes mk() {
        return new ARes("a");
    }
}

class BAsk implements Ask<Integer> {
    public BRes mk() {
        return new BRes(9);
    }
}

public class M {
    public static <T, U extends Res<T>> List<U> run(List<? extends Ask<T>> lst, Class<U> cls) {
        return lst.stream().map(a -> cls.cast(a.mk())).collect(Collectors.toList());
    }

    public static void main(String[] x) {
        List<ARes> ar = run(Arrays.asList(new AAsk(), new AAsk()), ARes.class);
        List<BRes> br = run(Arrays.asList(new BAsk(), new BAsk()), BRes.class);
    }
}
