public MultipartFile toMulti(File f, String mt) {
    Path p = f.toPath();
    String n = f.getName();
    byte[] d;
    try {
        d = Files.readAllBytes(p);
    } catch (IOException e) {
        throw new RuntimeException(e);
    }
    return new ByteArrayMultipartFile(n, mt, d);
}

public class ByteArrayMultipartFile implements MultipartFile {
    private final String n;
    private final String mt;
    private final byte[] d;

    public ByteArrayMultipartFile(String n, String mt, byte[] d) {
        this.n = n;
        this.mt = mt;
        this.d = d;
    }

    public String getName() { return n; }

    public String getOriginalFilename() { return n; }

    public String getContentType() { return mt; }

    public boolean isEmpty() { return d.length == 0; }

    public long getSize() { return d.length; }

    public byte[] getBytes() { return d; }

    public InputStream getInputStream() { return new ByteArrayInputStream(d); }

    public void transferTo(File dest) throws IOException {
        Files.write(dest.toPath(), d);
    }
}
