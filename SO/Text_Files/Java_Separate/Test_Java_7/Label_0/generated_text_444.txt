import java.net.*;
import java.io.*;
import java.util.*;

public class Nse {
    public static void main(String[] args) throws Exception {
        while (true) {
            URL u1 = new URL("https://www.nseindia.com/");
            HttpURLConnection c1 = (HttpURLConnection) u1.openConnection();
            c1.setRequestProperty("User-Agent", "Mozilla/5.0");
            c1.setRequestProperty("Accept", "*/*");
            c1.setRequestProperty("Connection", "keep-alive");
            List<String> cks = c1.getHeaderFields().get("Set-Cookie");

            URL u2 = new URL("https://www.nseindia.com/api/option-chain-indices?symbol=MIDCPNIFTY");
            HttpURLConnection c2 = (HttpURLConnection) u2.openConnection();
            c2.setRequestMethod("GET");
            c2.setRequestProperty("User-Agent", "Mozilla/5.0");
            c2.setRequestProperty("Accept", "application/json");
            c2.setRequestProperty("Connection", "keep-alive");
            c2.setRequestProperty("Referer", "https://www.nseindia.com/");

            for (String ck : cks) {
                c2.addRequestProperty("Cookie", ck.split(";", 2)[0]);
            }

            InputStream in = c2.getInputStream();
            System.out.println("Got data.");
            in.close();
            Thread.sleep(2000);
        }
    }
}
