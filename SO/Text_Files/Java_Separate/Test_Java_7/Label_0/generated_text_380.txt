private static final ScheduledExecutorService SCH = Executors.newScheduledThreadPool(8);
private static final ExecutorService SER = Executors.newFixedThreadPool(8);

List<CompletableFuture<Boolean>> lst = new ArrayList<>();
for (int i = 0; i < 100; i++) {
    lst.add(CompletableFuture.supplyAsync(() -> run(new Cli()), SER));
}
SER.shutdown();

private static Boolean run(Cli cli) {
    var id = cli.gen();
    return chk(cli, id);
}

private static Boolean chk(Cli cli, String id) {
    CompletableFuture<Boolean> fut = new CompletableFuture<>();
    Runnable r = () -> {
        if (cli.sts(id).eq(Sts.PEN)) return;
        fut.complete(cli.rep(id));
    };
    SCH.scheduleAtFixedRate(r, 0, 10, TimeUnit.SECONDS);
    return fut.join();
}
