public void cpDb() {
    AssetManager am = Objects.requireNonNull(requireContext()).getAssets();
    Context ctx = requireContext();
    ctx.getExternalFilesDir(null);
    File dbFld = new File(ctx.getExternalFilesDir(null).getParent(), "com.mydb.orca/databases");
    dbFld.mkdirs();

    try {
        String[] fls = am.list("database/test");
        if (fls != null) {
            for (String fn : fls) {
                InputStream in = am.open("database/test/" + fn);
                File outF = new File(dbFld, fn);
                OutputStream out = new FileOutputStream(outF);
                cpFl(in, out);
                in.close();
                out.flush();
                out.close();
            }
        }
    } catch (IOException e) {
        Log.e("tag", "Failed to copy", e);
    }
}

private void cpFl(InputStream in, OutputStream out) throws IOException {
    byte[] buf = new byte[1024];
    int rd;
    while ((rd = in.read(buf)) != -1) {
        out.write(buf, 0, rd);
    }
}
