import javafx.application.Application;
import javafx.scene.Group;
import javafx.scene.Scene;
import javafx.scene.layout.FlowPane;
import javafx.scene.paint.Color;
import javafx.scene.shape.Arc;
import javafx.scene.shape.ArcType;
import javafx.stage.Stage;
import javafx.event.ActionEvent;
import javafx.geometry.Pos;
import javafx.scene.control.Button;

public class Rot extends Application {
    private Group grp;
    private int ang = 0;

    public void start(Stage stg) {
        Arc b = new Arc(100, 100, 80, 80, 0, 90);
        b.setType(ArcType.ROUND);
        b.setFill(Color.BLUE);
        
        Arc g = new Arc(100, 100, 80, 80, 90, 90);
        g.setType(ArcType.ROUND);
        g.setFill(Color.GREEN);
        
        Arc y = new Arc(100, 100, 80, 80, 180, 90);
        y.setType(ArcType.ROUND);
        y.setFill(Color.YELLOW);
        
        Arc r = new Arc(100, 100, 80, 80, 270, 90);
        r.setType(ArcType.ROUND);
        r.setFill(Color.RED);
        
        grp = new Group(b, g, y, r);
        
        Button rot = new Button("Rotate");
        rot.setOnAction(this::clk);
        
        FlowPane p = new FlowPane(grp, rot);
        p.setAlignment(Pos.CENTER);
        p.setHgap(20);
        
        stg.setScene(new Scene(p, 250, 250));
        stg.setTitle("Wheel");
        stg.show();
    }

    public void clk(ActionEvent e) {
        ang += 90;
        grp.setRotate(ang);
    }
