import javafx.application.Application;
import javafx.beans.value.*;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class SpnDis extends Application {
    public void start(Stage stg) {
        Spinner<Integer> spnMin = new Spinner<>(2, 6, 2);
        Spinner<Integer> spnMax = new Spinner<>(2, 6, 6);
        spnMin.valueProperty().addListener((o, old, val) -> {
            spnMax.setValueFactory(new SpinnerValueFactory.IntegerSpinnerValueFactory(val, 6, spnMax.getValue()));
            setBtn(spnMin, val == 6, true);
        });
        spnMax.valueProperty().addListener((o, old, val) -> {
            spnMin.setValueFactory(new SpinnerValueFactory.IntegerSpinnerValueFactory(2, val, spnMin.getValue()));
            setBtn(spnMax, val == 2, false);
        });
        VBox v = new VBox(10, spnMin, spnMax);
        stg.setScene(new Scene(v, 200, 100));
        stg.show();
    }

    private void setBtn(Spinner<Integer> spn, boolean d, boolean up) {
        spn.lookupAll(".increment-arrow").forEach(n -> n.setDisable(d && up));
        spn.lookupAll(".decrement-arrow").forEach(n -> n.setDisable(d && !up));
    }

    public static void main(String[] a) {
        launch(a);
    }
}
