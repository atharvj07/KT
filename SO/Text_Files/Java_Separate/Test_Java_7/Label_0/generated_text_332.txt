import java.util.*;

class SenPro {
    public boolean chk(String s) {
        return s.matches("[a-zA-Z ]+");
    }

    public String pro(String s) {
        if (!chk(s)) return s + " is an invalid sentence";

        String[] w = s.trim().split("\\s+");
        if (w.length <= 2) return "Invalid Length";

        String t = w[0];
        w[0] = w[w.length - 1];
        w[w.length - 1] = t;

        for (int i = 1; i < w.length - 1; i++) {
            w[i] = new StringBuilder(w[i]).reverse().toString();
        }
        return String.join(" ", w);
    }
}

public class UI {
    public static void main(String[] a) {
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter the sentence");
        String s = sc.nextLine();
        System.out.println(new SenPro().pro(s));
        sc.close();
    }
}
