import java.util.*;
import java.util.stream.*;

class Emp {
    String id, name, city;

    Emp(String id, String name, String city) {
        this.id = id;
        this.name = name;
        this.city = city;
    }

    @Override
    public String toString() {
        return "Emp(id=" + id + ", name=" + name + ", city=" + city + ")";
    }
}

class Prs {
    String id, city;

    Prs(String id, String city) {
        this.id = id;
        this.city = city;
    }
}

public class Main {
    public static void main(String[] args) {
        List<Emp> emps = Stream.of(
                new Emp("100", "Alex", ""),
                new Emp("200", "Rida", ""),
                new Emp("300", "Ganga", "")
        ).collect(Collectors.toList());

        List<Prs> prss = Stream.of(
                new Prs("100", "Atlanta"),
                new Prs("300", "Boston"),
                new Prs("400", "Pleasanton")
        ).collect(Collectors.toList());

        Map<String, Prs> prsMap = prss.stream().collect(Collectors.toMap(p -> p.id, p -> p));

        emps.forEach(e -> {
            Prs p = prsMap.get(e.id);
            if (p != null) e.city = p.city;
        });

        List<Emp> res = Stream.concat(
                emps.stream(),
                prss.stream().filter(p -> emps.stream().noneMatch(e -> e.id.equals(p.id)))
                        .map(p -> new Emp(p.id, null, p.city))
        ).collect(Collectors.toList());

        res.forEach(System.out::println);
    }
}
