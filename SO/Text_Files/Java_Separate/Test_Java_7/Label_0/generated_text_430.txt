import java.util.*;

public class SubSetChk {
    public static void main(String[] args) {
        Map<String, Set<Long>> mp = new HashMap<>();
        mp.put("a", Set.of(1L, 2L, 3L));
        mp.put("b", Set.of(4L, 5L));
        mp.put("c", Set.of(1L));
        mp.put("d", Set.of(2L, 3L));
        mp.put("e", Set.of(5L));
        mp.put("f", Set.of(6L));

        List<String> res = new ArrayList<>();
        List<String> ks = new ArrayList<>(mp.keySet());

        for (int i = 0; i < ks.size(); i++) {
            for (int j = i + 1; j < ks.size(); j++) {
                if (mp.get(ks.get(i)).containsAll(mp.get(ks.get(j))) ||
                    mp.get(ks.get(j)).containsAll(mp.get(ks.get(i)))) {
                    res.add(ks.get(i) + " and " + ks.get(j));
                }
            }
        }

        res.forEach(System.out::println);
    }
}
