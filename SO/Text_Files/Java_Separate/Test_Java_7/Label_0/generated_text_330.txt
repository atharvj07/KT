import java.util.*;
import javafx.application.Application;
import javafx.beans.property.IntegerProperty;
import javafx.css.*;
import javafx.css.converter.SizeConverter;
import javafx.scene.Scene;
import javafx.scene.control.Label;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class App extends Application {

    public static class FooLbl extends Label {
        private static final CssMetaData<FooLbl, Number> FOO = new CssMetaData<>("-fx-foo", SizeConverter.getInstance(), 10) {
            public boolean isSettable(FooLbl l) { return !l.foo.isBound(); }
            public StyleableProperty<Number> getStyleableProperty(FooLbl l) { return l.foo; }
        };

        private static final List<CssMetaData<? extends Styleable, ?>> CSS;
        static { List<CssMetaData<? extends Styleable, ?>> list = new ArrayList<>(Label.getClassCssMetaData()); list.add(FOO); CSS = Collections.unmodifiableList(list); }
        public static List<CssMetaData<? extends Styleable, ?>> getClassCssMetaData() { return CSS; }
        public List<CssMetaData<? extends Styleable, ?>> getCssMetaData() { return CSS; }

        private final StyleableIntegerProperty foo = new StyleableIntegerProperty(10) {
            public CssMetaData<FooLbl, Number> getCssMetaData() { return FOO; }
            public Object getBean() { return FooLbl.this; }
            public String getName() { return "foo"; }
        };

        public FooLbl() { foo.addListener((o, oldV, newV) -> System.out.println("NEW VALUE: " + newV)); }
        public IntegerProperty fooProperty() { return foo; }
        public void setFoo(int v) { foo.set(v); }
        public int getFoo() { return foo.get(); }
    }

    public void start(Stage s) {
        var lbl = new FooLbl();
        lbl.getStyleClass().add("test");
        lbl.setText("abc");
        var box = new VBox(lbl);
        box.getStylesheets().add(App.class.getResource("test.css").toExternalForm());
        s.setScene(new Scene(box, 100, 100));
        s.show();
    }

    public static void main(String[] a) { launch(a); }
}
