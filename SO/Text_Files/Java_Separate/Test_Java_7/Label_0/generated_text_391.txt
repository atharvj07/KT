package sec.cfg;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.core.annotation.Order;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.core.userdetails.*;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;

@Configuration
public class SecCfg {

    @Bean
    public PasswordEncoder enc() {
        return new BCryptPasswordEncoder();
    }

    @Bean
    public UserDetailsService memUsr() {
        UserDetails u = User.withUsername("spcUsr")
                .password(enc().encode("spcPwd"))
                .roles("SPC")
                .build();
        return new InMemoryUserDetailsManager(u);
    }

    @Bean
    public UserDetailsService dbUsr() {
        return new CstUsrSrv();
    }

    @Bean
    @Order(1)
    public SecurityFilterChain spcSec(HttpSecurity h) throws Exception {
        h.securityMatcher("/spc/**")
                .authorizeHttpRequests(a -> a.anyRequest().authenticated())
                .userDetailsService(memUsr())
                .httpBasic();
        return h.build();
    }

    @Bean
    @Order(2)
    public SecurityFilterChain genSec(HttpSecurity h) throws Exception {
        h.authorizeHttpRequests(a -> a.anyRequest().authenticated())
                .userDetailsService(dbUsr())
                .httpBasic();
        return h.build();
    }
}
