import com.fasterxml.jackson.annotation.JsonCreator;
import com.fasterxml.jackson.annotation.JsonProperty;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.annotation.JsonSerialize;
import com.fasterxml.jackson.databind.annotation.JsonDeserialize;

import java.util.UUID;

public record AggId(@JsonValue UUID id) {}

public class Agg {

    public final AggId aggId;

    @JsonCreator
    public Agg(@JsonProperty("aggregateId") AggId aggId) {
        this.aggId = aggId;
    }
}

class Main {
    public static void main(String[] args) throws Exception {
        ObjectMapper objMap = new ObjectMapper();
        String str = "{ \"aggregateId\": \"3f61aede-83dd-4049-a6ff-337887b6b807\" }";
        Agg agg = objMap.readValue(str, Agg.class);
        System.out.println(objMap.writeValueAsString(agg));
    }
}
