System.setProperty("https.proxyHost", "xxxx");
System.setProperty("https.proxyPort", "xxxx");

String url = "https://mvnrepository.com/artifact/log4j/apache-log4j-extras/1.2.17";
StringBuilder sb = new StringBuilder();

try {
    URL u = new URL(url);
    HttpsURLConnection c = (HttpsURLConnection) u.openConnection();
    c.setRequestMethod("GET");
    c.setRequestProperty("User-Agent", "Mozilla/5.0");

    BufferedReader br = new BufferedReader(new InputStreamReader(
        c.getResponseCode() < 300 ? c.getInputStream() : c.getErrorStream(), StandardCharsets.UTF_8));

    String ln;
    while ((ln = br.readLine()) != null) sb.append(ln);
    br.close();
} catch (Exception e) {
    e.printStackTrace();
}

String res = sb.toString();
