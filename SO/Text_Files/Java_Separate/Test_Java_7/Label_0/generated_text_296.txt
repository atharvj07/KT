import com.itextpdf.text.*;
import com.itextpdf.text.pdf.*;
import java.io.*;
import java.nio.file.*;

class PdfMark {
    private static final String DIR = "C:\\Users\\java\\Desktop\\TestWaterMark\\";
    private static final String SRC = DIR + "test.pdf";
    private static final String OUT = DIR + "output.pdf";

    public static void main(String[] args) throws IOException, DocumentException {
        ByteArrayOutputStream buf = new ByteArrayOutputStream();
        byte[] raw = Files.readAllBytes(Paths.get(SRC));
        String txt = "???";
        String fntPath = DIR + "myCustomFont.ttf";
        Font fnt = FontFactory.getFont(fntPath, BaseFont.IDENTITY_H, 16);
        BaseFont bf = fnt.getBaseFont();
        PdfReader rdr = new PdfReader(raw);
        PdfStamper stp = new PdfStamper(rdr, buf);
        int pg = rdr.getNumberOfPages();

        for (int i = 1; i <= pg; i++) {
            Rectangle sz = rdr.getPageSizeWithRotation(i);
            PdfContentByte lay = stp.getOverContent(i);
            PdfGState gs = new PdfGState();
            gs.setFillOpacity(0.3f);
            lay.saveState();
            lay.setGState(gs);
            lay.beginText();
            lay.setFontAndSize(bf, 16);
            for (float x = 70f; x < sz.getWidth(); x += 150) {
                for (float y = 90; y < sz.getHeight(); y += 150) {
                    lay.showTextAlignedKerned(Element.ALIGN_CENTER, txt, x, y, 45f);
                }
            }
            lay.endText();
            lay.restoreState();
        }

        stp.close();
        rdr.close();
        Files.write(Paths.get(OUT), buf.toByteArray());
    }
}
