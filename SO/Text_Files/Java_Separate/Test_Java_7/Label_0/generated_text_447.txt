import javafx.beans.property.*;
import javafx.scene.control.*;
import javafx.util.*;
import java.net.*;
import java.time.*;
import java.util.*;

public class Ctl implements Initializable {
    @FXML
    private TableView<Lib> tab;
    @FXML
    private TableColumn<Lib, String> cod, avl, tit, autDir;
    private Svc svc = new Svc();

    @Override
    public void initialize(URL loc, ResourceBundle res) {
        cod.setCellValueFactory(new PropertyValueFactory<>("itemCode"));
        avl.setCellValueFactory(new PropertyValueFactory<>("availability"));
        tit.setCellValueFactory(new PropertyValueFactory<>("title"));
        autDir.setCellValueFactory(c -> {
            if (c.getValue() instanceof Bok) 
                return new SimpleStringProperty(((Bok) c.getValue()).getAut());
            if (c.getValue() instanceof Mov) 
                return new SimpleStringProperty(((Mov) c.getValue()).getDir());
            return new SimpleStringProperty("");
        });

        tab.getItems().addAll(svc.getLibs());
    }
}

abstract class Lib {
    private int cod;
    private String tit;
    private boolean avl;
    private int mid;
    private LocalDate dt;

    protected Lib(int cod, String tit, boolean avl, int mid, LocalDate dt) {
        this.cod = cod;
        this.tit = tit;
        this.avl = avl;
        this.mid = mid;
        this.dt = dt;
    }

    public int getItemCode() { return cod; }
    public String getTitle() { return tit; }
    public boolean isAvailability() { return avl; }
}

class Bok extends Lib {
    private String aut;

    protected Bok(int cod, String tit, boolean avl, int mid, LocalDate dt, String aut) {
        super(cod, tit, avl, mid, dt);
        this.aut = aut;
    }

    public String getAut() { return aut; }
}

class Mov extends Lib {
    private String dir;

    protected Mov(int cod, String tit, boolean avl, int mid, LocalDate dt, String dir) {
        super(cod, tit, avl, mid, dt);
        this.dir = dir;
    }

    public String getDir() { return dir; }
}

class Svc {
    public List<Lib> getLibs() {
        return List.of(
            new Bok(1, "Bk1", true, 0, LocalDate.now(), "Aut1"),
            new Mov(2, "Mv1", false, 1, LocalDate.now(), "Dir1")
        );
    }
}
