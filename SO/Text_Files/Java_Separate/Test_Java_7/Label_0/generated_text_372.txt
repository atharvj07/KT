        Spinner<BigDecimal> spinner = new Spinner<>();
        SpinnerValueFactory<BigDecimal> factory = new SpinnerValueFactory<BigDecimal>() {
            @Override
            public void decrement(int steps) {
                setValue(getValue().subtract(BigDecimal.valueOf(steps * 0.25)));
            }

            @Override
            public void increment(int steps) {
                setValue(getValue().add(BigDecimal.valueOf(steps * 0.25)));
            }
        };

        factory.setValue(BigDecimal.ZERO);
        factory.setAmountToStepBy(BigDecimal.valueOf(0.25));
        factory.setConverter(new javafx.util.StringConverter<BigDecimal>() {
            @Override
            public String toString(BigDecimal object) {
                DecimalFormat df = new DecimalFormat("+#.00;-#.00");
                return df.format(object);
            }

            @Override
            public BigDecimal fromString(String string) {
                try {
                    return new BigDecimal(string.replace(",", "."));
                } catch (NumberFormatException e) {
                    return BigDecimal.ZERO;
                }
            }
        });

        spinner.setValueFactory(factory);

        TextField editor = spinner.getEditor();
        Pattern validText = Pattern.compile("[+-]?\\d{0,2}(\\,\\d{0,2})?");
        UnaryOperator<TextFormatter.Change> filter = change -> {
            if (validText.matcher(change.getControlNewText()).matches()) {
                return change;
            } else {
                return null;
            }
        };
