public String transform(List<CustomObject> list) {
    return list.stream()
        .filter(o -> o.objectType().equals("BUSINESS"))
        .filter(o -> o.objectStatus().equals("ACTIVE"))
        .filter(o -> o.locale().equals("en-US"))
        .findFirst()
        .orElseGet(() -> list.stream()
            .filter(o -> o.objectType().equals("BUSINESS"))
            .sorted(Comparator.comparing(CustomObject::url))
            .findFirst()
            .get())
        .getUrl();
}
