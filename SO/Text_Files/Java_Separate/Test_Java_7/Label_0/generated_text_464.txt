public interface Utl {
    void get();
}

@Service
public class UtlOne implements Utl {
    public void get() { }
}

@Service
public class UtlTwo implements Utl {
    public void get() { }
}

@Component
public class UtlMgr {
    private final UtlOne one;
    private final UtlTwo two;
    private Utl act;

    @Autowired
    public UtlMgr(UtlOne one, UtlTwo two) {
        this.one = one;
        this.two = two;
        this.act = one;
    }

    public void swp(String val) {
        act = "one".equals(val) ? one : two;
    }

    public Utl get() {
        return act;
    }
}

@Service
public class DmoSvc {
    private final UtlMgr mgr;

    @Autowired
    public DmoSvc(UtlMgr mgr) {
        this.mgr = mgr;
    }

    public void run() {
        mgr.get().get();
    }
}
