import org.bouncycastle.jce.provider.BouncyCastleProvider;
import java.security.*;
import java.security.spec.ECGenParameterSpec;
import java.util.Base64;

public class GenKey {
    private static final String CUR = "secp384r1";

    public static void main(String[] args) {
        try {
            Security.addProvider(new BouncyCastleProvider());
            ECGenParameterSpec par = new ECGenParameterSpec(CUR);
            KeyPairGenerator gen = KeyPairGenerator.getInstance("EC", "BC");
            gen.initialize(par, new SecureRandom());
            KeyPair kp = gen.generateKeyPair();
            String pri = toPem(kp.getPrivate());
            System.out.println(pri);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    private static String toPem(PrivateKey key) {
        String b64 = Base64.getEncoder().encodeToString(key.getEncoded());
        return "-----BEGIN PRIVATE KEY-----\n" + b64 + "\n-----END PRIVATE KEY-----";
    }
}
