public class Enc {
    public void run() {
        String nc = Base64.encodeToString("this_is_my_nonce".getBytes(), Base64.URL_SAFE | Base64.NO_WRAP | Base64.NO_PADDING);
        IntegrityManager im = IntegrityManagerFactory.create(getApplicationContext());
        Task<IntegrityTokenResponse> tk = im.requestIntegrityToken(IntegrityTokenRequest.builder().setNonce(nc).build());

        tk.addOnSuccessListener(new OnSuccessListener<IntegrityTokenResponse>() {
            @Override
            public void onSuccess(IntegrityTokenResponse tr) {
                String tkn = tr.token();
                Executor ex = Executors.newSingleThreadExecutor();
                ex.execute(() -> {
                    try {
                        String vrd = chk(tkn);
                        System.out.println(vrd);
                    } catch (Exception e) {
                        e.printStackTrace();
                    }
                });
            }
        });
    }

    private String chk(String tkn) throws IOException, GeneralSecurityException {
        GoogleCredentials gc = GoogleCredentials.fromStream(getClass().getResourceAsStream("/service-account.json"))
                .createScoped("https://www.googleapis.com/auth/playintegrity");
        gc.refreshIfExpired();
        IdToken idt = ((IdTokenProvider) gc).idTokenWithAudience("https://playintegrity.googleapis.com/v1", null);
        String jt = idt.getTokenValue();
        String url = "https://playintegrity.googleapis.com/v1/decodeIntegrityToken";
        String res = Http.post(url, jt, "{\"integrityToken\":\"" + tkn + "\"}");
        return parse(res);
    }

    private String parse(String js) throws IOException {
        JSONObject jo = new JSONObject(js);
        return jo.toString(2);
    }
}
