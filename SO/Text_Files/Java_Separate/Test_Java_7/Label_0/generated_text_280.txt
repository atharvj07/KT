import javafx.application.Application;
import javafx.geometry.Bounds;
import javafx.scene.Scene;
import javafx.scene.control.Label;
import javafx.scene.control.Tooltip;
import javafx.scene.layout.AnchorPane;
import javafx.stage.Stage;
import javafx.util.Duration;

public class TipFix extends Application {

    @Override
    public void start(Stage stg) {
        Label lbl = new Label("TEST\nTEST\nTEST");
        lbl.setStyle("-fx-background-color: green;");

        Tooltip tip = new Tooltip("TOOLTIP");
        tip.setShowDelay(Duration.seconds(0.5));
        lbl.setTooltip(tip);

        lbl.setOnMouseEntered(e -> {
            Bounds b = lbl.localToScreen(lbl.getBoundsInLocal());
            tip.show(lbl, b.getMaxX(), b.getMaxY());
        });

        lbl.setOnMouseExited(e -> tip.hide());

        stg.setScene(new Scene(new AnchorPane(lbl)));
        stg.show();
    }

    public static void main(String[] args) {
        launch();
    }
}
