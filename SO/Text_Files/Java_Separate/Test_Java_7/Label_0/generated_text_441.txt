import java.lang.module.*;
import java.util.*;

public class ModChk {
    public static void main(String[] args) {
        ModuleLayer l = ModuleLayer.boot();
        Set<String> cor = Set.of("java.base", "java.sql", "java.xml", "java.desktop",
                                 "java.logging", "java.management", "java.naming",
                                 "java.security.sasl", "java.instrument", "jdk.unsupported");

        Map<Boolean, List<String>> grp = new HashMap<>();
        grp.put(true, new ArrayList<>());
        grp.put(false, new ArrayList<>());

        for (Module m : l.modules()) {
            grp.get(cor.contains(m.getName())).add(m.getName());
        }

        System.out.println("Core: " + grp.get(true));
        System.out.println("Non-Core: " + grp.get(false));
    }
}
