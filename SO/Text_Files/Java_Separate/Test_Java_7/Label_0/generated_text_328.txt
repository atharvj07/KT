import java.lang.foreign.*;
import java.lang.invoke.VarHandle;

public class LArr {
    private static final long SIZ = Long.BYTES;
    private final MemorySegment seg;
    private final VarHandle vh;

    public LArr(long len) {
        seg = MemorySegment.allocateNative(len * SIZ, MemorySession.global());
        vh = MemorySegment.varHandle(long.class, MemoryLayout.PathElement.sequenceElement());
    }

    public void set(long idx, long val) {
        vh.set(seg, idx * SIZ, val);
    }

    public long get(long idx) {
        return (long) vh.get(seg, idx * SIZ);
    }

    public void free() {
        seg.session().close();
    }
}
