package io.armadillo.aftfileuploads;

import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.context.annotation.Bean;
import org.springframework.scheduling.concurrent.ThreadPoolTaskScheduler;

import java.util.concurrent.ThreadPoolExecutor;

@SpringBootApplication
public class AftFileUploadsApplication {

    public static void main(String[] args) {
        SpringApplication.run(AftFileUploadsApplication.class, args);
    }

    @Bean
    public ThreadPoolTaskScheduler taskSch() {
        ThreadPoolTaskScheduler sch = new ThreadPoolTaskScheduler();
        sch.setPoolSize(10);
        sch.setThreadNamePrefix("task-");
        sch.setRejectedExecutionHandler(new ThreadPoolExecutor.CallerRunsPolicy());
        return sch;
    }
}
