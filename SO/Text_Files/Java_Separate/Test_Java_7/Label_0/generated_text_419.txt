import javafx.application.Application;
import javafx.scene.Group;
import javafx.scene.Scene;
import javafx.scene.paint.Color;
import javafx.scene.shape.Circle;
import javafx.stage.Stage;

public class App extends Application {
    private double x, y;
    private Circle c;

    @Override
    public void start(Stage s) {
        Group g = new Group();
        Scene sc = new Scene(g, 960, 540);

        sc.setOnMousePressed(e -> {
            x = e.getX();
            y = e.getY();
            c = new Circle(x, y, 0, Color.BLUE);
            g.getChildren().add(c);
        });

        sc.setOnMouseDragged(e -> {
            double r = Math.sqrt(Math.pow(x - e.getX(), 2) + Math.pow(y - e.getY(), 2));
            c.setRadius(r);
        });

        sc.setOnMouseReleased(e -> {
            if (c.getRadius() == 0) g.getChildren().remove(c);
        });

        s.setTitle("App");
        s.setScene(sc);
        s.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
