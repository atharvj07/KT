(10)
public SecurityFilterChain sec1(HttpSecurity h) throws Exception {
    h.securityMatcher("/**")
     .authorizeHttpRequests(a -> a.requestMatchers(createAntRequestMatchers(whitelist)).permitAll()
     .anyRequest().authenticated())
     .oauth2ResourceServer(...);
    return h.build();
}

@Bean
@Order(100)
public SecurityFilterChain sec2(HttpSecurity h, ResourceFilter f) throws Exception {
    h.securityMatcher("/**").addFilterAfter(f, SessionManagementFilter.class);
    return h.build();
}
