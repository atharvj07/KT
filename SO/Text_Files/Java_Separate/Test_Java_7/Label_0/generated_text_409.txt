package com.remmymilkyway.customcomponent;

import javafx.fxml.FXML;
import javafx.fxml.FXMLLoader;
import javafx.scene.layout.Region;
import javafx.scene.layout.VBox;
import javafx.scene.web.WebEngine;
import javafx.scene.web.WebView;

import java.io.IOException;
import java.net.URL;

public class CusCmp extends Region {
    @FXML
    private VBox box;
    private final WebView web;
    private final WebEngine eng;

    public CusCmp() {
        FXMLLoader ldr = new FXMLLoader(getClass().getResource("/CusCmp.fxml"));
        ldr.setRoot(this);
        ldr.setController(this);
        try {
            ldr.load();
        } catch (IOException e) {
            throw new RuntimeException(e);
        }

        this.web = new WebView();
        this.eng = web.getEngine();
        URL url = getClass().getResource("/monaco_editor.html");
        if (url != null) eng.load(url.toExternalForm());
        box.getChildren().add(web);
    }

    public String getTxt() {
        return (String) eng.executeScript("getEditorValue()");
    }

    public void setTxt(String val) {
        String esc = val.replace("'", "\\'").replace("\n", "\\n");
        eng.executeScript("setEditorValue('" + esc + "');");
    }

    public void setFnt(String fnt) {
        eng.executeScript("setFontFamily('" + fnt + "');");
    }

    public void setSz(int sz) {
        eng.executeScript("setFontSize(" + sz + ");");
    }

    public void setLang(String lng) {
        eng.executeScript("setLanguage('" + lng + "');");
    }

    @Override
    protected void layoutChildren() {
        web.setPrefSize(getWidth(), getHeight());
        web.resize(getWidth(), getHeight());
    }
}
