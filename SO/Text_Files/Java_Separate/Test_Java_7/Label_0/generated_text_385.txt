public class Res {
    @JsonProperty("count")
    private int cnt;

    @JsonProperty("respCode")
    private int code;

    @JsonProperty("payLoad")
    @JsonSerialize(using = StrArrSer.class)
    @JsonDeserialize(using = StrArrDes.class)
    private String[] data;

    public String[] getData() { return data; }
    public void setData(String[] data) { this.data = data; }
}

public class StrArrSer extends JsonSerializer<String[]> {
    @Override
    public void serialize(String[] val, JsonGenerator gen, SerializerProvider sp) throws IOException {
        gen.writeStartArray();
        for (String s : val) gen.writeRawValue(s);
        gen.writeEndArray();
    }
}

public class StrArrDes extends JsonDeserializer<String[]> {
    @Override
    public String[] deserialize(JsonParser p, DeserializationContext ctx) throws IOException {
        List<String> lst = new ArrayList<>();
        while (p.nextToken() != JsonToken.END_ARRAY) lst.add(p.readValueAsTree().toString());
        return lst.toArray(new String[0]);
    }
}
