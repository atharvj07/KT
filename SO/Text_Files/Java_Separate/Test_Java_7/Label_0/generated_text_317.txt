import io.opentelemetry.api.trace.*;
import io.opentelemetry.sdk.trace.*;
import io.opentelemetry.sdk.trace.data.*;
import io.opentelemetry.sdk.trace.export.*;
import org.junit.jupiter.api.*;
import java.util.*;

import static org.junit.jupiter.api.Assertions.*;

public class TestSpan {
    private static InMemorySpanExporter ex;
    private static SdkTracerProvider tp;
    private static Tracer tr;

    @BeforeAll
    static void setUp() {
        ex = InMemorySpanExporter.create();
        tp = SdkTracerProvider.builder()
                .addSpanProcessor(SimpleSpanProcessor.create(ex))
                .build();
        tr = tp.get("test");
    }

    @AfterAll
    static void tearDown() {
        tp.close();
    }

    @Test
    void testHelloSpan() {
        tr.spanBuilder("hello").startSpan().end();
        List<SpanData> spans = ex.getFinishedSpanItems();
        assertEquals(1, spans.size());
        assertEquals("hello", spans.get(0).getName());
    }
}
