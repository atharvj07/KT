(localName = "Item")
@JsonPropertyOrder({"SNO", "Weight"})
public class Item {

    @JacksonXmlProperty(localName = "SNO")
    private String sNo;

    @JacksonXmlProperty(localName = "Weight")
    private Float weight;

    @JsonIgnore
    private List<Pair> lst;

    @JsonAnyGetter
    public Map<String, String> getXmlMap() {
        Map<String, String> map = new LinkedHashMap<>();
        lst.forEach(p -> {
            map.put("A", p.a);
            map.put("B", p.b);
        });
        return map;
    }
}

@AllArgsConstructor
class Pair {
    String a, b;
}
