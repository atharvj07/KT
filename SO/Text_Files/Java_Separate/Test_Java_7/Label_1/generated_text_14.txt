class Example {
  public static void main(String[] args) {
    Object o = new Object() {
      void test() {
        System.out.println("HELLO!");
      }
    };
    o.test(); // compiler error.

    var p = new Object() {
      void test() {
        System.out.println("HELLO!");
      }
    };

    p.test(); // works!!
  }

  void asAReminder() {
    new Object() {
      void test() {
        System.out.println("This has always worked"); 
      }
    }.test();
  }
}
