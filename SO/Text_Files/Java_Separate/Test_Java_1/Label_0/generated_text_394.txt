import javafx.application.Application;
import javafx.beans.value.ChangeListener;
import javafx.beans.value.ObservableValue;
import javafx.scene.Scene;
import javafx.scene.control.Spinner;
import javafx.scene.control.SpinnerValueFactory;
import javafx.stage.Stage;

public class SpinnerTest extends Application {

    @Override
    public void start(Stage primaryStage) {
        Spinner<Integer> spinnerMin = new Spinner<>();
        Spinner<Integer> spinnerMax = new Spinner<>();

        // Set value factory and range
        spinnerMin.setValueFactory(new SpinnerValueFactory.IntegerSpinnerValueFactory(2, 6, 2));
        spinnerMax.setValueFactory(new SpinnerValueFactory.IntegerSpinnerValueFactory(2, 6, 6));

        // Add listener for spinnerMin
        spinnerMin.valueProperty().addListener((changed, oldVal, newVal) -> {
            spinnerMax.setValueFactory(new SpinnerValueFactory.IntegerSpinnerValueFactory(newVal, 6, spinnerMax.getValue()));
            updateArrows(spinnerMin, spinnerMax);
        });

        // Add listener for spinnerMax
        spinnerMax.valueProperty().addListener((changed, oldVal, newVal) -> {
            spinnerMin.setValueFactory(new SpinnerValueFactory.IntegerSpinnerValueFactory(2, newVal, spinnerMin.getValue()));
            updateArrows(spinnerMin, spinnerMax);
        });

        updateArrows(spinnerMin, spinnerMax);

        // Layout and scene
        Scene scene = new Scene(spinnerMin);
        scene.setRoot(spinnerMin);
        primaryStage.setTitle("Spinner Test");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    // Method to update arrow buttons enabled/disabled based on values
    private void updateArrows(Spinner<Integer> spinnerMin, Spinner<Integer> spinnerMax) {
        spinnerMin.getEditor().setDisable(spinnerMin.getValue() == spinnerMax.getValue());
        spinnerMax.getEditor().setDisable(spinnerMin.getValue() == spinnerMax.getValue());
    }

    public static void main(String[] args) {
        launch(args);
    }
}
