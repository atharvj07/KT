public interface Util {
    void getClient();
}

@Service("utilOne")
public class UtilOne implements Util {
    public void getClient() {
        System.out.println("UtilOne client");
    }
}

@Service("utilTwo")
public class UtilTwo implements Util {
    public void getClient() {
        System.out.println("UtilTwo client");
    }
}

@Component
public class UtilProvider {

    @Autowired
    private ApplicationContext context;

    private Util currentUtil;

    @PostConstruct
    public void init() {
        switchUtil("utilOne"); // default implementation
    }

    public void switchUtil(String beanName) {
        this.currentUtil = context.getBean(beanName, Util.class);
    }

    public Util getUtil() {
        return currentUtil;
    }
}

@Component
public class DemoService {

    @Autowired
    private UtilProvider utilProvider;

    public void useClient() {
        utilProvider.getUtil().getClient();
    }
}

@RestController
@RequestMapping("/switch")
public class SwitchController {

    @Autowired
    private UtilProvider utilProvider;

    @PostMapping("/{beanName}")
    public void switchUtil(@PathVariable String beanName) {
        utilProvider.switchUtil(beanName);
    }
}
