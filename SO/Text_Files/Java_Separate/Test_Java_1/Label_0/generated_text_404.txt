private <T> void positionEntity(Class<T> clazz, Object... params) {
    int x, y;
    boolean entityIsCreated = false;
    while (!entityIsCreated) {
        x = random.nextInt(sizeX);
        y = random.nextInt(sizeY);
        if (field.getFieldable(x, y) instanceof Empty) {
            try {
                T entity = clazz.getConstructor(int.class, int.class, Object.class, Object.class, int.class, int.class)
                        .newInstance(x, y, params[0], params[1], sizeX, sizeY);
                field.setFieldable(x, y, entity);
                entityIsCreated = true;
            } catch (Exception e) {
                e.printStackTrace();
            }
        }
    }
}

// Usage
public void positionMagician() {
    positionEntity(Magician.class, player, field);
}

public void positionGoblin() {
    positionEntity(Goblin.class, player, field);
}
