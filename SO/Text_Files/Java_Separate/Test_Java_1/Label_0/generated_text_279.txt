public static boolean equalSplit(int[] arr) {
    if (arr.length % 2 != 0) return false;
    return equalSplit(arr, 0, 0, 0, 0);
}

private static boolean equalSplit(int[] arr, int index, int leftCount, int leftSum, int rightSum) {
    if (index == arr.length) {
        return leftCount == arr.length / 2 && leftSum == rightSum;
    }
    // Try including current element in left group
    if (leftCount < arr.length / 2 && equalSplit(arr, index + 1, leftCount + 1, leftSum + arr[index], rightSum)) {
        return true;
    }
    // Try including current element in right group
    if (equalSplit(arr, index + 1, leftCount, leftSum, rightSum + arr[index])) {
        return true;
    }
    return false;
}

public static void main(String[] args) {
    System.out.println(equalSplit(new int[]{1, 2, 3, 4})); // true (e.g., [1, 4] and [2, 3])
    System.out.println(equalSplit(new int[]{1, 1, 1, 1})); // true (e.g., [1, 1] and [1, 1])
    System.out.println(equalSplit(new int[]{1, 2, 3, 5})); // false
}
