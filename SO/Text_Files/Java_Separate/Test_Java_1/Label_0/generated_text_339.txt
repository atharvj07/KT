package application;

import javafx.animation.Interpolator;
import javafx.animation.Transition;
import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.StackPane;
import javafx.scene.paint.Color;
import javafx.scene.text.Font;
import javafx.scene.text.Text;
import javafx.stage.Stage;
import javafx.util.Duration;

public class Main extends Application {

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();

        StackPane stack = new StackPane();
        Text text = new Text("Hello there StackOverflow, how are you? (:");
        text.setFont(Font.font(16));
        stack.getChildren().add(text);

        Color color1 = Color.RED;
        Color color2 = Color.BLUE;

        Transition transition = new Transition() {
            {
                setCycleDuration(Duration.seconds(2));
                setInterpolator(Interpolator.EASE_BOTH);
                setCycleCount(INDEFINITE);
                setAutoReverse(true);
            }

            @Override
            protected void interpolate(double frac) {
                double r = color1.getRed() + (color2.getRed() - color1.getRed()) * frac;
                double g = color1.getGreen() + (color2.getGreen() - color1.getGreen()) * frac;
                double b = color1.getBlue() + (color2.getBlue() - color1.getBlue()) * frac;
                text.setFill(new Color(r, g, b, 1.0));
            }
        };

        transition.play();

        root.setCenter(stack);
        Scene scene = new Scene(root, 400, 400);
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
