import java.util.concurrent.Callable;

public class InferenceExample {
    public static void main(String[] args) {
        try {
            execute(() -> {
                throw new IllegalArgumentException("Unchecked exception");
            });

            // Uncommenting this will cause a compilation error:
            // execute(() -> {
            //     throw new Exception("Checked exception"); // Compilation error
            // });
        } catch (Throwable t) {
            System.out.println("Caught: " + t);
        }
    }

    static <T> T execute(Callable<T> callable) throws Exception {
        return callable.call();
    }
}
