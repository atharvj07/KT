import org.optaplanner.core.api.score.stream.Constraint;
import org.optaplanner.core.api.score.stream.ConstraintFactory;
import org.optaplanner.core.api.score.stream.ConstraintProvider;
import org.springframework.web.bind.annotation.*;

import java.util.HashSet;
import java.util.Set;

@RestController
@RequestMapping("/constraints")
class ConstraintController {
    private final Set<String> activeConstraints = new HashSet<>();

    @PostMapping
    public void setConstraints(@RequestBody Set<String> constraints) {
        activeConstraints.clear();
        activeConstraints.addAll(constraints);
    }

    @GetMapping
    public Set<String> getConstraints() {
        return activeConstraints;
    }

    public boolean isConstraintActive(String constraintName) {
        return activeConstraints.contains(constraintName);
    }
}

public class TimetableConstraintProvider implements ConstraintProvider {
    private final ConstraintController constraintController;

    public TimetableConstraintProvider(ConstraintController constraintController) {
        this.constraintController = constraintController;
    }

    @Override
    public Constraint[] defineConstraints(ConstraintFactory constraintFactory) {
        Set<Constraint> constraints = new HashSet<>();

        if (constraintController.isConstraintActive("roomConflict")) {
            constraints.add(roomConflict(constraintFactory));
        }
        if (constraintController.isConstraintActive("teacherConflict")) {
            constraints.add(teacherConflict(constraintFactory));
        }
        if (constraintController.isConstraintActive("studentGroupConflict")) {
            constraints.add(studentGroupConflict(constraintFactory));
        }
        if (constraintController.isConstraintActive("teacherRoomStability")) {
            constraints.add(teacherRoomStability(constraintFactory));
        }
        if (constraintController.isConstraintActive("teacherTimeEfficiency")) {
            constraints.add(teacherTimeEfficiency(constraintFactory));
        }
        if (constraintController.isConstraintActive("studentGroupSubjectVariety")) {
            constraints.add(studentGroupSubjectVariety(constraintFactory));
        }

        return constraints.toArray(new Constraint[0]);
    }

    private Constraint roomConflict(ConstraintFactory factory) { /* Implementation */ }
    private Constraint teacherConflict(ConstraintFactory factory) { /* Implementation */ }
    private Constraint studentGroupConflict(ConstraintFactory factory) { /* Implementation */ }
    private Constraint teacherRoomStability(ConstraintFactory factory) { /* Implementation */ }
    private Constraint teacherTimeEfficiency(ConstraintFactory factory) { /* Implementation */ }
    private Constraint studentGroupSubjectVariety(ConstraintFactory factory) { /* Implementation */ }
}
