import javax.crypto.Cipher;
import javax.crypto.SecretKey;
import javax.crypto.spec.IvParameterSpec;
import javax.crypto.spec.SecretKeySpec;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.Base64;
import org.apache.commons.codec.binary.Hex;

public class AESFileEncryption {

    public static void main(String[] args) throws Exception {
        Path keyPath = Paths.get("/home/debashishd/Downloads/s_key");
        byte[] keyBytes = Files.readAllBytes(keyPath);
        
        String testMessage = "Hello this is Roxane";
        
        // AES Encryption
        SecretKeySpec secretKey = new SecretKeySpec(keyBytes, "AES");
        Cipher cipher = Cipher.getInstance("AES/CBC/PKCS5Padding");
        cipher.init(Cipher.ENCRYPT_MODE, secretKey, new IvParameterSpec(new byte[16]));
        
        String encryptedValue = Base64.getEncoder().encodeToString(cipher.doFinal(testMessage.getBytes()));
        System.out.println("Encrypted value: " + encryptedValue);
        
        // AES Decryption
        Cipher decryptCipher = Cipher.getInstance("AES/CBC/PKCS5Padding");
        decryptCipher.init(Cipher.DECRYPT_MODE, secretKey, new IvParameterSpec(new byte[16]));
        
        String decryptedValue = new String(decryptCipher.doFinal(Base64.getDecoder().decode(encryptedValue)));
        System.out.println("Decrypted value: " + decryptedValue);
    }
}
