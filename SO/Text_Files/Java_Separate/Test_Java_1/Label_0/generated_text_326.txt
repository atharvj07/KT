import java.util.*;

class Edge {
    int to, weight;
    Edge(int to, int weight) {
        this.to = to;
        this.weight = weight;
    }
}

class Solver {
    static int minEdges;

    public static int solve(List<List<Edge>> T, long L) {
        minEdges = Integer.MAX_VALUE;
        for (int i = 0; i < T.size(); i++) {
            dfs(T, L, i, -1, 0, 0);
        }
        return minEdges == Integer.MAX_VALUE ? -1 : minEdges;
    }

    static void dfs(List<List<Edge>> T, long L, int node, int parent, long length, int edges) {
        if (length == L) {
            minEdges = Math.min(minEdges, edges);
            return;
        }
        if (length > L) return;

        for (Edge edge : T.get(node)) {
            if (edge.to != parent) {
                dfs(T, L, edge.to, node, length + edge.weight, edges + 1);
            }
        }
    }
}

public class Main {
    static void addEdge(List<List<Edge>> T, int u, int v, int w) {
        T.get(u).add(new Edge(v, w));
        T.get(v).add(new Edge(u, w));
    }

    public static void main(String[] args) {
        int N = 8;
        List<List<Edge>> T = new ArrayList<>();
        for (int i = 0; i < N; i++) T.add(new ArrayList<>());

        addEdge(T, 0, 1, 2);
        addEdge(T, 1, 2, 1);
        addEdge(T, 1, 3, 2);
        addEdge(T, 2, 6, 1);
        addEdge(T, 6, 7, 1);
        addEdge(T, 3, 4, 1);
        addEdge(T, 3, 5, 4);

        System.out.println(Solver.solve(T, 5L)); // Output: 2
    }
}
