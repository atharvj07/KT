import com.fasterxml.jackson.annotation.JsonAnySetter;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.io.IOException;
import java.util.Iterator;
import java.util.Map;

public class DynamicJson {
    private JsonNode rootNode;

    @JsonAnySetter
    public void handleUnknown(String key, JsonNode value) {
        if (rootNode == null) {
            rootNode = value;
        } else {
            ((ObjectNode) rootNode).set(key, value);
        }
    }

    public JsonNode getRootNode() {
        return rootNode;
    }

    public static void main(String[] args) throws IOException {
        String json = "{\n" +
                "    \"name\" : \"sahal\",\n" +
                "    \"address\" : [\n" +
                "        {\n" +
                "           \"state\" : \"FL\"\n" +
                "        },\n" +
                "        {\n" +
                "           \"country\" : \"FL\",\n" +
                "           \"city\" : {\n" +
                "               \"type\" : \"municipality\",\n" +
                "               \"value\" : \"California City\"\n" +
                "           }\n" +
                "        },\n" +
                "        {\n" +
                "           \"pin\" : \"87876\"\n" +
                "        }\n" +
                "    ]\n" +
                "}";

        ObjectMapper objectMapper = new ObjectMapper();
        DynamicJson dynamicJson = objectMapper.readValue(json, DynamicJson.class);

        JsonNode root = dynamicJson.getRootNode();
        Iterator<Map.Entry<String, JsonNode>> fields = root.fields();

        while (fields.hasNext()) {
            Map.Entry<String, JsonNode> field = fields.next();
            System.out.println(field.getKey() + ": " + field.getValue());
        }
    }
}
