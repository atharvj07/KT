import java.io.File;
import java.lang.management.ManagementFactory;
import java.lang.management.ThreadMXBean;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

public class FileDescriptorFinder {
    public static void main(String[] args) throws Exception {
        String targetFilePath = "C:\\path\\to\\your\\file.txt"; // Modify with your file path
        findProcessHoldingFile(targetFilePath);
    }

    public static void findProcessHoldingFile(String filePath) throws Exception {
        String pid = getProcessId();
        String command = String.format("handle.exe -a %s", filePath); // Ensure you have Sysinternals Handle tool

        Process process = Runtime.getRuntime().exec(command);
        process.waitFor();
        
        List<String> output = new java.io.BufferedReader(new java.io.InputStreamReader(process.getInputStream()))
            .lines().collect(Collectors.toList());

        output.forEach(line -> {
            if (line.contains(filePath)) {
                System.out.println("Process holding file: " + line);
            }
        });
    }

    public static String getProcessId() throws Exception {
        ThreadMXBean threadMXBean = ManagementFactory.getThreadMXBean();
        long[] threadIds = threadMXBean.getAllThreadIds();
        String pid = ManagementFactory.getRuntimeMXBean().getName().split("@")[0]; // Extracting PID
        return pid;
    }
}
