import javafx.application.Application;
import javafx.scene.Group;
import javafx.scene.Scene;
import javafx.scene.paint.Color;
import javafx.scene.shape.Circle;
import javafx.stage.Stage;

public class TestJavaFX extends Application {

    private double cX, cY;
    private boolean firstClick = true;
    private Circle tempCircle = null;

    @Override
    public void start(Stage primaryStage) {
        Group root = new Group();
        Scene scene = new Scene(root, 960, 540);

        scene.setOnMouseClicked(evt -> {
            if (firstClick) {
                cX = evt.getX();
                cY = evt.getY();
                firstClick = false;
            } else {
                double r = Math.sqrt(Math.pow(cX - evt.getX(), 2) + Math.pow(cY - evt.getY(), 2));
                Circle circle = new Circle(cX, cY, r, Color.BLUE);
                root.getChildren().add(circle);
                firstClick = true;
            }
        });

        scene.setOnMouseMoved(evt -> {
            if (!firstClick && tempCircle == null) {
                tempCircle = new Circle(cX, cY, 0, Color.BLUE);
                root.getChildren().add(tempCircle);
            }

            if (tempCircle != null) {
                double r = Math.sqrt(Math.pow(cX - evt.getX(), 2) + Math.pow(cY - evt.getY(), 2));
                tempCircle.setRadius(r);
            }
        });

        scene.setOnMouseExited(evt -> {
            if (tempCircle != null) {
                root.getChildren().remove(tempCircle);
                tempCircle = null;
            }
        });

        primaryStage.setTitle("TestJavaFX");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
