@Configuration
public class Jaxb2MarshallerConfig {

    @Bean
    public Jaxb2Marshaller myMarshaller() {
        Jaxb2Marshaller marshaller = new Jaxb2Marshaller();
        marshaller.setContextPath("com.company.project.xsd.some_portal.v4_0");
        marshaller.setMtomEnabled(true); 
        marshaller.setMarshallerProperties(Map.of(
            "com.sun.xml.bind.xmlHeaders", "<soapenv:Envelope xmlns:soapenv=\"http://schemas.xmlsoap.org/soap/envelope/\" xmlns:urn=\"http://my.company.com/xsd/portals/v4_0\"><soapenv:Header/><soapenv:Body>",
            "com.sun.xml.bind.v2.runtime.MarshallerImpl.PRETTY_PRINT", "true"
        ));
        return marshaller;
    }
}

java
private BinaryData buildBinaryData(byte[] documentData) {
    BinaryData binaryData = new BinaryData();
    DataHandler dataHandler = new DataHandler(documentData, "application/pdf");
    binaryData.setBlob(dataHandler);
    binaryData.setExtension("pdf");
    return binaryData;
}

java
@Component
@Log4j2
public class MySoapClient extends WebServiceGatewaySupport {
    private final WebServiceTemplate template;

    public MySoapClient(MyServiceProperties properties, Jaxb2Marshaller marshaller) {
        setMarshaller(marshaller);
        setUnmarshaller(marshaller);
        setDefaultUri(properties.getTargetUrl());
        template = getWebServiceTemplate();
    }

    @Override
    public void sendDocuments(MyRequest request) {
        try {
            template.marshalSendAndReceive(request);
        } catch (Exception e) {
            log.error(e, e.getCause());
            throw new RuntimeException(e);
        }
    }
}

public class MyRequestTest {

    @Test
    public void testSendDocuments() {
        byte[] documentData = "VeryLongDataBlobInHere".getBytes(StandardCharsets.UTF_8);
        BinaryData binaryData = buildBinaryData(documentData);

        MyRequest request = new MyRequest();
        request.setDocumentList(Collections.singletonList(new DocumentList(binaryData)));

        MyServiceProperties properties = new MyServiceProperties();
        properties.setTargetUrl("http://localhost:8080/soap");

        Jaxb2Marshaller marshaller = new Jaxb2Marshaller();
        marshaller.setContextPath("com.company.project.xsd.some_portal.v4_0");
        marshaller.setMtomEnabled(true);

        MySoapClient client = new MySoapClient(properties, marshaller);
        client.sendDocuments(request);
    }
}
