import java.awt.*;
import java.awt.geom.Line2D;
import javax.swing.*;

public class Main {
    public static void main(String[] args) {
        JFrame jf = new JFrame("Demo");
        Container cp = jf.getContentPane();
        MyCanvas tl = new MyCanvas();
        cp.add(tl);
        jf.setSize(500, 250);
        jf.setVisible(true);
        jf.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
    }
}

class MyCanvas extends JComponent {

    @Override
    public void paintComponent(Graphics g) {
        if (g instanceof Graphics2D g2) {
            g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            g2.setFont(Font.decode("Times New Roman-26"));
            g2.drawString("The poorly-scaled cake is a lie.", 70, 40);
            g2.setStroke(new BasicStroke(2.3f));
            g2.draw(new Line2D.Double(420, 10, 425, 70));

            Image I = createScalingAwareBuffer(this, 500, 150);
            g2.drawImage(I, 0, 130, null);
        }
    }

    private Image createScalingAwareBuffer(JComponent jp, int width, int height) {
        GraphicsConfiguration gc = jp.getGraphicsConfiguration();
        Graphics2D g2d = (Graphics2D) gc.createCompatibleImage(width, height, Transparency.TRANSLUCENT).getGraphics();
        double scaleFactor = Toolkit.getDefaultToolkit().getScreenResolution() / 96.0; // Assume 96 DPI is baseline
        AffineTransform transform = AffineTransform.getScaleInstance(scaleFactor, scaleFactor);
        g2d.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g2d.setRenderingHint(RenderingHints.KEY_RENDERING, RenderingHints.VALUE_RENDER_QUALITY);
        g2d.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BICUBIC);

        g2d.setTransform(transform);
        g2d.setFont(Font.decode("Times New Roman-26"));
        g2d.setColor(Color.BLACK);
        g2d.drawString("The poorly-scaled cake is a lie.", 70, 40);
        g2d.setStroke(new BasicStroke(2.3f));
        g2d.draw(new Line2D.Double(420, 10, 425, 70));
        g2d.dispose();
        
        return g2d.getDeviceConfiguration().createCompatibleImage(width, height, Transparency.TRANSLUCENT);
    }
}
