import javax.servlet.ServletConfig;
import javax.servlet.ServletException;
import javax.servlet.UnavailableException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;

@WebServlet("/main")
public class MainServlet extends HttpServlet {

    @Override
    public void init(ServletConfig config) throws ServletException {
        super.init(config);
        try {
            connectToBackendServer();
        } catch (Exception e) {
            getServletContext().log("Cannot connect to Backend Server", e);
            destroy(); // Force container to stop by throwing UnavailableException
            throw new UnavailableException("Cannot connect to Backend Server");
        }
    }

    private void connectToBackendServer() throws Exception {
        throw new NullPointerException("Simulated connection failure");
    }

    @Override
    public void destroy() {
        getServletContext().log("Shutting down due to backend connection failure");
        System.exit(1); // Force Tomcat to stop deployment
    }
}
