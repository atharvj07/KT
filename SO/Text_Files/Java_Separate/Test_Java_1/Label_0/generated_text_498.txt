import java.io.*;
import java.util.*;

public class CsvReader {

    public static void main(String[] args) throws IOException {
        FileReader fileReader = new FileReader(new File(System.getProperty("user.home") + "\\Desktop\\testData.txt"));
        BufferedReader br = new BufferedReader(fileReader);

        List<String> headers = new ArrayList<>();
        List<HashMap<String, String>> myTable = new ArrayList<>();

        String line;
        int ext = 0;

        while ((line = br.readLine()) != null) {
            if (ext == 0) {
                headers = Arrays.asList(line.split(","));
            } else {
                String[] values = line.split(",");
                HashMap<String, String> myRecord = new HashMap<>();
                for (int i = 0; i < headers.size(); i++) {
                    myRecord.put(headers.get(i), values[i]);
                }
                myTable.add(myRecord);
            }
            ext++;
        }

        br.close();

        // Print table for verification
        for (HashMap<String, String> record : myTable) {
            System.out.println(record);
        }
    }
}
