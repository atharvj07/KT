package com.question.controller;

import com.question.service.LanguageService;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Lazy;
import org.springframework.stereotype.Component;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

import javax.annotation.PostConstruct;

@RestController
public class LanguageController {

    private final LanguageService frenchLanguageService;
    private final LanguageService englishLanguageService;

    private LanguageService languageService;

    @Value("${configuration}")
    private String configuration;

    public LanguageController(@Qualifier("french") @Lazy LanguageService frenchLanguageService,
                              @Qualifier("english") @Lazy LanguageService englishLanguageService) {
        this.frenchLanguageService = frenchLanguageService;
        this.englishLanguageService = englishLanguageService;
    }

    @PostConstruct
    private void init() {
        if ("french".equalsIgnoreCase(configuration)) {
            this.languageService = frenchLanguageService;
        } else if ("english".equalsIgnoreCase(configuration)) {
            this.languageService = englishLanguageService;
        }
    }

    @GetMapping("/test")
    public String test(@RequestParam String name) {
        return languageService.process(name);
    }
}
