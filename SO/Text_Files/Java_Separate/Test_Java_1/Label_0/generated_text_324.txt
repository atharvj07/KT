import org.springframework.context.ApplicationContext;
import org.springframework.context.annotation.AnnotationConfigApplicationContext;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.ComponentScan;
import org.springframework.context.annotation.Configuration;
import org.springframework.stereotype.Component;

import java.util.List;
import java.util.Map;
import java.util.Scanner;

interface Command {
    void execute(List<Integer> params);
}

@Component
class A implements Command {
    @Override
    public void execute(List<Integer> params) {
        System.out.println("Class A created with params: " + params);
    }
}

@Component
class B implements Command {
    @Override
    public void execute(List<Integer> params) {
        System.out.println("Class B created with params: " + params);
    }
}

@Component
class C implements Command {
    @Override
    public void execute(List<Integer> params) {
        System.out.println("Class C created with params: " + params);
    }
}

@Component
class CommandFactory {
    private final Map<String, Command> commands;

    public CommandFactory(Map<String, Command> commands) {
        this.commands = commands;
    }

    public void executeCommand(String input) {
        String[] parts = input.split(" ");
        String commandKey = parts[0];
        List<Integer> params = List.of();
        if (parts.length > 1) {
            params = List.of(parts).subList(1, parts.length).stream()
                    .map(Integer::parseInt)
                    .toList();
        }
        Command command = commands.get(commandKey);
        if (command != null) {
            command.execute(params);
        } else {
            System.out.println("Invalid command: " + commandKey);
        }
    }
}

@Configuration
@ComponentScan(basePackages = "your.package.name") // Replace with your package name
class AppConfig {
}

public class Main {
    public static void main(String[] args) {
        ApplicationContext context = new AnnotationConfigApplicationContext(AppConfig.class);
        CommandFactory factory = context.getBean(CommandFactory.class);

        Scanner scanner = new Scanner(System.in);
        System.out.println("Enter command:");
        String input = scanner.nextLine();

        factory.executeCommand(input);
    }
}
