from itertools import combinations

def find_subset(nums, target, limit):
    nums.sort()  # Optional optimization for early pruning
    def backtrack(start, path, remaining):
        if len(path) == limit and remaining == 0:
            return path
        if len(path) >= limit or remaining < 0:
            return None
        for i in range(start, len(nums)):
            res = backtrack(i + 1, path + [nums[i]], remaining - nums[i])
            if res:
                return res
        return None

    return backtrack(0, [], target) or []

# Test Cases
print(find_subset([2,7,11,15], 20, 3))  # Output: [2, 7, 11]
print(find_subset([1,2,3,4,5,6], 10, 2))  # Output: [4, 6] or any valid pair
print(find_subset([5,5,5,5,5], 15, 3))  # Output: [5, 5, 5]
print(find_subset([1,3,5,7,9,11], 18, 3))  # Output: [3, 5, 10] or any valid triplet
print(find_subset([1,2,3,4,5], 50, 3))  # Output: []
