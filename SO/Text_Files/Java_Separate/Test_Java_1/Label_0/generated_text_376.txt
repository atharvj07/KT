@Test
void testCollectMap() {
    List<A> list = Arrays.asList(new A("name1"), new A("name2"));
    Map<String, String> result = Flux.fromIterable(list)
            .collectMap(a -> a.getName(),
                    b -> Mono.just(new C("url_" + b.getName()))
                            .map(C::url)
                            .block())
            .block();

    assertEquals(2, result.size());
    assertEquals("url_name1", result.get("name1"));
    assertEquals("url_name2", result.get("name2"));
}
