import java.util.ArrayList;
import java.util.List;

public class StringGenerator {

    public static void main(String[] args) {
        String[] first  = {"A", "B", "C"};
        String[] second = {"1", "2", "3"};
        String[] third  = {"X", "Y", "Z"};
        
        List<String[]> inputs = List.of(first, second, third);
        List<String> result = createStrings(inputs);
        result.forEach(System.out::println);
    }

    static List<String> createStrings(List<String[]> inputs) {
        List<String> result = new ArrayList<>();
        generateStrings(inputs, 0, "", result);
        return result;
    }

    static void generateStrings(List<String[]> inputs, int index, String current, List<String> result) {
        if (index == inputs.size()) {
            result.add(current);
            return;
        }
        for (String s : inputs.get(index)) {
            generateStrings(inputs, index + 1, current + s, result);
        }
    }
}
