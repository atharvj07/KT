import java.awt.*;
import java.awt.event.*;
import java.awt.image.BufferedImage;
import java.math.BigDecimal;
import javax.swing.*;

public class MandelbrotSet extends JFrame {

    static final int WIDTH = 800;
    static final int HEIGHT = 800;
    static final int MAX_ITER = 1000;

    BigDecimal topLeftX = new BigDecimal(-2);
    BigDecimal topLeftY = new BigDecimal(1.5);
    BigDecimal zoomFactor = new BigDecimal(3).divide(new BigDecimal(WIDTH), 20, BigDecimal.ROUND_HALF_UP);

    BufferedImage image;

    public MandelbrotSet() {
        setTitle("Mandelbrot Set");
        setSize(WIDTH, HEIGHT);
        setResizable(false);
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);

        image = new BufferedImage(WIDTH, HEIGHT, BufferedImage.TYPE_INT_RGB);
        addMouseListener(new MouseAdapter() {
            @Override
            public void mousePressed(MouseEvent e) {
                BigDecimal newX = topLeftX.add(zoomFactor.multiply(new BigDecimal(e.getX())));
                BigDecimal newY = topLeftY.subtract(zoomFactor.multiply(new BigDecimal(e.getY())));
                if (e.getButton() == MouseEvent.BUTTON1) {
                    zoomFactor = zoomFactor.multiply(new BigDecimal("0.5"));
                } else if (e.getButton() == MouseEvent.BUTTON3) {
                    zoomFactor = zoomFactor.multiply(new BigDecimal("2"));
                }
                topLeftX = newX.subtract(zoomFactor.multiply(new BigDecimal(WIDTH / 2)));
                topLeftY = newY.add(zoomFactor.multiply(new BigDecimal(HEIGHT / 2)));
                drawMandelbrotSet();
                repaint();
            }
        });

        drawMandelbrotSet();
        setVisible(true);
    }

    private void drawMandelbrotSet() {
        for (int x = 0; x < WIDTH; x++) {
            for (int y = 0; y < HEIGHT; y++) {
                BigDecimal zx = BigDecimal.ZERO;
                BigDecimal zy = BigDecimal.ZERO;
                BigDecimal cX = topLeftX.add(zoomFactor.multiply(new BigDecimal(x)));
                BigDecimal cY = topLeftY.subtract(zoomFactor.multiply(new BigDecimal(y)));
                int iter = 0;
                while (zx.multiply(zx).add(zy.multiply(zy)).compareTo(new BigDecimal("4")) <= 0 && iter < MAX_ITER) {
                    BigDecimal temp = zx.multiply(zx).subtract(zy.multiply(zy)).add(cX);
                    zy = zx.multiply(zy).multiply(new BigDecimal("2")).add(cY);
                    zx = temp;
                    iter++;
                }
                int color = iter | (iter << 8);
                image.setRGB(x, y, iter < MAX_ITER ? color : 0);
            }
        }
    }

    @Override
    public void paint(Graphics g) {
        g.drawImage(image, 0, 0, null);
    }

    public static void main(String[] args) {
        new MandelbrotSet();
    }
}
