@Configuration
public class StreamProcessorConfig {

    @Bean
    public List<Function<KStream<String, InputType>, KStream<String, OutputType>>> tenantProcessors(
            @Value("${tenants}") List<String> tenants) {
        return tenants.stream()
                .map(tenant -> (Function<KStream<String, InputType>, KStream<String, OutputType>>) inputTypeStream ->
                        inputTypeStream.map((k, v) -> KeyValue.pair(k, OutputType.createFrom(v))))
                .toList();
    }
}

@SpringBootTest
@Import(StreamProcessorConfig.class)
class StreamProcessorTest {

    @Autowired
    private List<Function<KStream<String, InputType>, KStream<String, OutputType>>> processors;

    @Test
    void testTenantProcessing() {
        assertNotNull(processors);
        assertEquals(3, processors.size());

        KStream<String, InputType> input = mock(KStream.class);
        when(input.map(any())).thenReturn(mock(KStream.class));

        processors.forEach(processor -> {
            KStream<String, OutputType> output = processor.apply(input);
            assertNotNull(output);
        });
    }
}
