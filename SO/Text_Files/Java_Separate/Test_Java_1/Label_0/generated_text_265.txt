package taskbar_test;

import com.sun.glass.ui.Window;
import javafx.application.Application;
import javafx.concurrent.Task;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;
import taskbar_test.gen.Xlib_h;

import java.lang.foreign.Arena;
import java.lang.foreign.MemorySegment;
import java.lang.foreign.ValueLayout;

public class AppLinuxXlib extends Application {

    private static final String NET_WM_NAME = "_NET_WM_NAME";
    private static final String NET_WM_STATE = "_NET_WM_STATE";
    private static final String NET_WM_XAPP_PROGRESS = "_NET_WM_XAPP_PROGRESS";

    @Override
    public void start(Stage primaryStage) {
        Button startButton = new Button("Start Long Running Task");

        startButton.setOnAction(event -> {
            final long rawHandle = Window.getWindows().getFirst().getRawHandle();
            Task<Void> longTask = new Task<>() {
                @Override
                protected Void call() throws Exception {
                    try (var arena = Arena.ofConfined()) {
                        MemorySegment x11Session = Xlib_h.XOpenDisplay(MemorySegment.NULL);
                        if (x11Session == null) throw new RuntimeException("Failed to open X11 display");

                        MemorySegment atomNetWmState = Xlib_h.XInternAtom(x11Session, arena.allocateFrom(NET_WM_STATE), 0);
                        MemorySegment atomNetWmName = Xlib_h.XInternAtom(x11Session, arena.allocateFrom(NET_WM_NAME), 0);
                        MemorySegment atomNetWmProgress = Xlib_h.XInternAtom(x11Session, arena.allocateFrom(NET_WM_XAPP_PROGRESS), 0);

                        for (int i = 0; i <= 100; i += 20) {
                            Thread.sleep(500);
                            MemorySegment progressData = arena.allocateFrom(ValueLayout.JAVA_INT, i);
                            Xlib_h.XChangeProperty(
                                x11Session,
                                MemorySegment.ofAddress(rawHandle).address(),
                                atomNetWmProgress.address(),
                                6,
                                32,
                                0,
                                progressData,
                                1
                            );
                            Xlib_h.XFlush(x11Session);
                        }

                        Xlib_h.XCloseDisplay(x11Session);
                    }
                    return null;
                }
            };

            new Thread(longTask).start();
        });

        VBox vbox = new VBox(10, startButton);
        Scene scene = new Scene(vbox, 300, 200);
        primaryStage.setScene(scene);
        primaryStage.setTitle("Taskbar Progress Example Linux");
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
