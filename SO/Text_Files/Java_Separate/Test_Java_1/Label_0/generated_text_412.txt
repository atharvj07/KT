import java.util.*;
import java.util.stream.*;

@RestController
public class CoordinateController {

    @PostMapping("/hole/coordinate")
    @ResponseBody
    public String saveCoordinate(@RequestBody Map<String, Object> params) {
        System.out.println("params = " + params);
        String coordinates = (String) params.get("coordinate");

        // Convert String to int[][]
        int[][] intArray = Arrays.stream(coordinates.substring(1, coordinates.length() - 1).split("\\],\\["))
                .map(str -> Arrays.stream(str.split(","))
                        .mapToInt(Integer::parseInt)
                        .toArray())
                .toArray(int[][]::new);

        // Output the resulting int[][] array
        for (int[] arr : intArray) {
            System.out.println(Arrays.toString(arr));
        }

        return "success";
    }
    
    public static void main(String[] args) {
        // Test Case
        Map<String, Object> testParams = new HashMap<>();
        testParams.put("coordinate", "[[445, 292], [585, 331], [612, 223], [205, 532]]");
        
        CoordinateController controller = new CoordinateController();
        controller.saveCoordinate(testParams);
    }
}
