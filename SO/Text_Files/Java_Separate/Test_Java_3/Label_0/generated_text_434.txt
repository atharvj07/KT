import java.util.*;

public class Solution {
    public List<Integer> findSubset(int[] nums, int target, int limit) {
        return findSubset(nums, target, limit, 0, new ArrayList<>());
    }

    private List<Integer> findSubset(int[] nums, int target, int limit, int start, List<Integer> current) {
        if (current.size() == limit) {
            if (target == 0) {
                return new ArrayList<>(current);
            }
            return null;
        }
        for (int i = start; i < nums.length; i++) {
            current.add(nums[i]);
            List<Integer> result = findSubset(nums, target - nums[i], limit, i + 1, current);
            if (result != null) {
                return result;
            }
            current.remove(current.size() - 1);
        }
        return null;
    }

    public static void main(String[] args) {
        Solution sol = new Solution();
        int[] nums = {2, 7, 11, 15};
        int target = 20;
        int limit = 3;
        System.out.println(sol.findSubset(nums, target, limit));  // Output: [2, 7, 11]
    }
}
