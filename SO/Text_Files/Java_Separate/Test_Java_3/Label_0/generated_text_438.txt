@RestController
public class TestCaseController {

    private final ExecutorService executorService = Executors.newFixedThreadPool(1);

    @PostMapping("/start-tests")
    public ResponseEntity<String> startTests() {
        CompletableFuture.runAsync(() -> runTests(), executorService);
        return ResponseEntity.ok("Test started");
    }

    private void runTests() {
        for (int i = 1; i <= 30; i++) {
            int testCaseId = i;
            executorService.submit(() -> {
                String status = runTestCase(testCaseId);
                updateStatus(testCaseId, status);
            });
        }
    }

    private String runTestCase(int testCaseId) {
        // Logic to run each test case (scheduler, producer, consumer)
        try {
            Thread.sleep(300000); // Simulating 5 min test case run time
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
        return "Test Case " + testCaseId + " completed";
    }

    private void updateStatus(int testCaseId, String status) {
        // Logic to update status in database or broadcast via WebSocket to UI
        System.out.println("Test Case " + testCaseId + ": " + status);
    }
}
