public Float someFunction(final AggregationType e) {
    return (float) provides.stream()
        .mapToDouble(this::someFunc)
        .collect(
            switch (e) {
                case MIN -> Collectors.minBy(Double::compare);
                case MAX -> Collectors.maxBy(Double::compare);
                case AVERAGE -> Collectors.averagingDouble(d -> d);
            })
        .orElse(-1);
}
