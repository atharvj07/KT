@Bean
public OpenAPI customOpenAPI() {
    return new OpenAPI()
            .info(new Info().title("API Documentation").version("1.0"))
            .components(new Components()
                .addResponses("ResponseEnvelope", new ApiResponse().description("Wrapper response")
                    .content(new Content().addMediaType("application/json", new MediaType().schema(new ObjectSchema()
                        .addProperty("status", new StringSchema().example("OK"))
                        .addProperty("result", new ObjectSchema().additionalProperties(new StringSchema()))))));
}

@Schema(name = "ResponseEnvelope", description = "Wrapper response")
public class ResponseEnvelope<T> {
    @Schema(description = "Response status", example = "OK")
    private String status;
    
    @Schema(description = "The result object")
    private T result;
    
    // getters and setters
}
