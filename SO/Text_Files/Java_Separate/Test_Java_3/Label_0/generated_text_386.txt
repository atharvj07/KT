import com.fasterxml.jackson.core.JsonParser;
import com.fasterxml.jackson.databind.DeserializationContext;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.jupiter.api.Test;
import org.mockito.Mockito;

import java.io.IOException;
import java.text.ParseException;
import java.util.Date;

import static org.junit.jupiter.api.Assertions.*;

public class StringToDateDeserializerTest {

    @Test
    public void testDeserialize() throws IOException, ParseException {
        StringToDateDeserializer deserializer = new StringToDateDeserializer();
        JsonParser mockParser = Mockito.mock(JsonParser.class);
        DeserializationContext mockContext = Mockito.mock(DeserializationContext.class);

        String testDate = "2025-03-17T00:00:00Z";
        Mockito.when(mockParser.getText()).thenReturn(testDate);

        Date expectedDate = new StdDateFormat().parse(testDate);

        Date result = deserializer.deserialize(mockParser, mockContext);

        assertEquals(expectedDate, result);
    }

    @Test
    public void testDeserializeInvalidDate() throws IOException {
        StringToDateDeserializer deserializer = new StringToDateDeserializer();
        JsonParser mockParser = Mockito.mock(JsonParser.class);
        DeserializationContext mockContext = Mockito.mock(DeserializationContext.class);

        String invalidDate = "invalid-date";
        Mockito.when(mockParser.getText()).thenReturn(invalidDate);

        MyCustomException exception = assertThrows(MyCustomException.class, () -> {
            deserializer.deserialize(mockParser, mockContext);
        });

        assertEquals("Invalid date passed, ISO 8601 is expected", exception.getMessage());
    }
}
