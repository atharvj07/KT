public List<Employee> mergeLists(List<Employee> employeeList, List<Person> personList) {
    Map<String, String> personMap = personList.stream()
            .collect(Collectors.toMap(Person::getId, Person::getCity));

    return employeeList.stream()
            .map(employee -> new Employee(employee.getId(), employee.getName(), personMap.get(employee.getId())))
            .collect(Collectors.toList());
}
