@Bean
public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
    http
        .oauth2ResourceServer(oauth2 -> oauth2
            .jwt()
            .and()
        )
        .exceptionHandling()
        .authenticationEntryPoint((request, response, exception) -> {
            // Custom logic for authentication failure when no token is provided
        })
        .and()
        .addFilterBefore(customBearerTokenAuthenticationFilter(), BearerTokenAuthenticationFilter.class);

    return http.build();
}

@Bean
public BearerTokenAuthenticationFilter customBearerTokenAuthenticationFilter() {
    BearerTokenAuthenticationFilter filter = new BearerTokenAuthenticationFilter(authenticationManager());
    filter.setAuthenticationFailureHandler(new JwtAuthenticationFailureHandler());
    return filter;
}

class JwtAuthenticationFailureHandler implements AuthenticationFailureHandler {
    @Override
    public void onAuthenticationFailure(HttpServletRequest request, HttpServletResponse response,
                                        AuthenticationException exception) throws IOException, ServletException {
        // Custom logging or response logic on failure
    }
}
