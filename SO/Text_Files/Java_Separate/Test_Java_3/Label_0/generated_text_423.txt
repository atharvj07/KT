@FXML
private LineChart<String, Number> linechart;
@FXML
private BarChart<String, Number> barchart;

@Override
public void initialize(URL url, ResourceBundle rb) {
    XYChart.Series<String, Number> series = new XYChart.Series<>();
    series.getData().add(new XYChart.Data<>("Jan", 12));
    series.getData().add(new XYChart.Data<>("Feb", 20));
    series.getData().add(new XYChart.Data<>("March", 10));
    series.getData().add(new XYChart.Data<>("April", 14));

    linechart.getData().add(series);
    barchart.getData().add(series);

    // Set colors for the BarChart
    for (XYChart.Data<String, Number> data : series.getData()) {
        data.getNode().setStyle("-fx-bar-fill: steelblue;");
    }

    // Set colors for the LineChart
    for (XYChart.Data<String, Number> data : series.getData()) {
        data.getNode().setStyle("-fx-stroke: steelblue;");
    }
}
