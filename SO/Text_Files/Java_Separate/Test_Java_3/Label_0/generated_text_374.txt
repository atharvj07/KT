import javafx.beans.property.SimpleObjectProperty;
import javafx.stage.Stage;

public class UserDataListener {
    public static void main(String[] args) {
        Stage stage = new Stage();

        SimpleObjectProperty<Object> userDataProperty = new SimpleObjectProperty<>(stage.getUserData());
        userDataProperty.addListener((observable, oldValue, newValue) -> {
            System.out.println("new userdata: " + newValue);
        });

        // Change userData to test listener
        stage.setUserData(2);
        userDataProperty.set(stage.getUserData()); // Manually update the property
        stage.setUserData(3);
        userDataProperty.set(stage.getUserData()); // Manually update the property
    }
}
