package com.question.controller;

import com.question.service.LanguageService;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.web.bind.annotation.*;

@RestController
public class LanguageController {

    private final LanguageService languageService;

    public LanguageController(@Value("${configuration}") String configuration, ApplicationContext context) {
        this.languageService = (LanguageService) context.getBean(configuration);
    }

    @GetMapping("/test")
    public String test(@RequestParam String name) {
        return languageService.process(name);
    }
}
