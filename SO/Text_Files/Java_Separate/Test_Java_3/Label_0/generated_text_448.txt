import static org.mockito.Mockito.*;
import javax.sql.DataSource;
import java.sql.Connection;
import java.sql.SQLException;

public class DatabaseSimulation {

    public DataSource createFaultyDataSource() throws SQLException {
        DataSource dataSource = mock(DataSource.class);
        Connection connection = mock(Connection.class);
        
        when(dataSource.getConnection()).thenThrow(new SQLException("HikariPool-1 - Connection is not available, request timed out after 30001ms"));
        
        return dataSource;
    }

    @Test
    public void testDatabaseConnectionTimeout() throws SQLException {
        DataSource faultyDataSource = createFaultyDataSource();
        
        // Simulate database operation with the faulty data source
        assertThrows(SQLException.class, () -> {
            faultyDataSource.getConnection();
        });
    }
}
