import org.web3j.crypto.ECKeyPair;
import org.web3j.crypto.Sign;
import org.web3j.crypto.Keys;
import org.web3j.utils.Numeric;

import java.math.BigInteger;

public boolean verifySignature(String signedMessage, String nonce, String publicWalletId) {
    try {
        // Remove the "0x" prefix from the signed message if present
        signedMessage = signedMessage.startsWith("0x") ? signedMessage.substring(2) : signedMessage;

        // Convert the signed message to byte array
        byte[] signedMessageBytes = Numeric.hexStringToByteArray(signedMessage);
        
        // Get the public key from the wallet ID (address)
        BigInteger publicKey = Keys.getAddress(publicWalletId);

        // Recover the signature
        Sign.SignatureData signatureData = Sign.SignatureData.fromByteArray(signedMessageBytes);
        ECKeyPair keyPair = Sign.signedMessageToKey(nonce.getBytes(), signatureData);

        // Compare the recovered public key with the expected public key
        return keyPair.getPublicKey().equals(publicKey);
    } catch (Exception e) {
        // Log exception if necessary
        return false;
    }
}
