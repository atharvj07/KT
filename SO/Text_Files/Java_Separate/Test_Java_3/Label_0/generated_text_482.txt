public void setAlignedText(TextView textView, String leftText, String rightText) {
    SpannableString finalString = new SpannableString(leftText + "\t\t\t\t" + rightText);
    finalString.setSpan(new TabStopSpan.Standard(1000), leftText.length(), leftText.length() + 1, Spannable.SPAN_EXCLUSIVE_EXCLUSIVE);
    textView.setText(finalString);
}
