// Step 1: Define an endpoint to receive selected constraints
@PostMapping("/selectedConstraints")
public ResponseEntity<Void> setSelectedConstraints(@RequestBody List<String> selectedConstraints) {
    constraintService.setSelectedConstraints(selectedConstraints);
    return ResponseEntity.ok().build();
}

// Step 2: Service to store selected constraints
@Service
public class ConstraintService {
    private List<String> selectedConstraints = new ArrayList<>();

    public void setSelectedConstraints(List<String> selectedConstraints) {
        this.selectedConstraints = selectedConstraints;
    }

    public List<String> getSelectedConstraints() {
        return selectedConstraints;
    }
}

// Step 3: Modify the TimetableConstraintProvider to apply selected constraints
public class TimetableConstraintProvider implements ConstraintProvider {
    
    @Autowired
    private ConstraintService constraintService;

    @Override
    public Constraint[] defineConstraints(ConstraintFactory constraintFactory) {
        List<String> selectedConstraints = constraintService.getSelectedConstraints();
        
        List<Constraint> constraints = new ArrayList<>();
        
        if (selectedConstraints.contains("roomConflict")) {
            constraints.add(roomConflict(constraintFactory));
        }
        if (selectedConstraints.contains("teacherConflict")) {
            constraints.add(teacherConflict(constraintFactory));
        }
        if (selectedConstraints.contains("studentGroupConflict")) {
            constraints.add(studentGroupConflict(constraintFactory));
        }
        if (selectedConstraints.contains("teacherRoomStability")) {
            constraints.add(teacherRoomStability(constraintFactory));
        }
        if (selectedConstraints.contains("teacherTimeEfficiency")) {
            constraints.add(teacherTimeEfficiency(constraintFactory));
        }
        if (selectedConstraints.contains("studentGroupSubjectVariety")) {
            constraints.add(studentGroupSubjectVariety(constraintFactory));
        }

        return constraints.toArray(new Constraint[0]);
    }
}
