@Inject
@JMSConnectionFactory("java:/jms/remoteCF")
private JMSContext context;

@Resource(lookup = "java:/queue/testQueue")
private Queue queue;

@Override
protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
    resp.setContentType("text/html");
    PrintWriter out = resp.getWriter();
    try {
        out.write("<p>Sending messages to <em>" + queue + "</em></p>");
        out.write("<p>Using context <em>" + context + "</em></p>");
        out.write("<h2>The following messages will be sent to the destination:</h2>");
        
        for (int i = 0; i < MSG_COUNT; i++) {
            String text = "This is message " + (i + 1);
            context.createProducer().send(queue, text);
            out.write("Message (" + i + "): " + text + "<br/>");
        }

        // Now handle message consumption separately
        JMSConsumer consumer = context.createConsumer(queue);
        for (int i = 0; i < MSG_COUNT; i++) {
            TextMessage message = (TextMessage) consumer.receive(5000); // Adjust timeout as needed
            if (message != null) {
                out.write("Message received (" + i + "): " + message.getText() + "<br/>");
            } else {
                out.write("Message receive timeout (" + i + ")<br/>");
            }
        }
        
    } catch (JMSException e) {
        e.printStackTrace();
    } finally {
        if (out != null) {
            out.close();
        }
    }
}
