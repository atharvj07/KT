@Component
public class UtilFactory {

    @Autowired
    private ApplicationContext applicationContext;

    public Util getUtil(String activeUtil) {
        if ("UtilTwo".equalsIgnoreCase(activeUtil)) {
            return applicationContext.getBean(UtilTwo.class);
        }
        return applicationContext.getBean(UtilOne.class);
    }
}

@Service
public class DemoService {

    @Autowired
    private UtilFactory utilFactory;

    @Value("${active.util}")
    private String activeUtil;

    public void useClient() {
        Util util = utilFactory.getUtil(activeUtil);
        util.getClient();
    }
}
