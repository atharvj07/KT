private void parsePath(String path) {
    String regex = "abfss?://([^/]+)@([^/]+)(?::([^/]+))?(/[^/]+)?(/.*)?";
    Pattern azurePathPattern = Pattern.compile(regex);
    Matcher matcher = azurePathPattern.matcher(path);
    
    if (matcher.find()) {
        String fileSystem = matcher.group(1); 
        String accountName = matcher.group(2);
        String accountSuffix = matcher.group(3) != null ? matcher.group(3) : "";
        String relativePath = matcher.group(4) != null ? matcher.group(4) + matcher.group(5) : matcher.group(5);
        
        // Print or use the extracted values
        System.out.println("fileSystem: " + fileSystem);
        System.out.println("accountName: " + accountName);
        System.out.println("accountSuffix: " + accountSuffix);
        System.out.println("relativePath: " + relativePath);
    }
}
