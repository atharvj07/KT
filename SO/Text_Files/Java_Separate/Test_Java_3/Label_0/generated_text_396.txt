private void processTemplateWithUTF8(String templateName, Context context) {
    TemplateEngine templateEngine = new TemplateEngine();
    ClassLoaderTemplateResolver resolver = new ClassLoaderTemplateResolver();
    resolver.setCharacterEncoding("UTF-8");
    resolver.setTemplateMode(TemplateMode.XML);
    templateEngine.setTemplateResolver(resolver);
    
    try (BufferedWriter out = new BufferedWriter(new OutputStreamWriter(new FileOutputStream("test_output.xml"), StandardCharsets.UTF_8))) {
        templateEngine.process(templateName, context, out);
    } catch (Exception e) {
        e.printStackTrace();
    }
}

public static void main(String[] args) {
    TestThymeleafUTF8 app = new TestThymeleafUTF8();
    Context context = new Context();
    context.setVariable("TEST_AR", "ARABIC_TEXT");
    context.setVariable("TEST_HE", "HEBREW_TEXT");
    context.setVariable("TEST_CN", "CHINESE_TEXT");
    app.processTemplateWithUTF8("test_template.xml", context);
}
