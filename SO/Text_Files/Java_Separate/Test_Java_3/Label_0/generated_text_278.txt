// Sample code illustrating the concept

public class InferenceExample {

    public <T extends Exception> void handleExceptionThrows(T exception) throws T {
        // throws clause with inferred exception type
        throw exception;
    }

    public static void main(String[] args) {
        try {
            new InferenceExample().handleExceptionThrows(new Exception("Checked Exception"));
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
