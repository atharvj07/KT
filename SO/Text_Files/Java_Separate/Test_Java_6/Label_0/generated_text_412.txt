import org.springframework.web.bind.annotation.*;

import java.util.*;
import java.util.stream.Collectors;

@RestController
public class CoordinateController {

    @PostMapping("/hole/coordinate")
    @ResponseBody
    public String saveCoordinate(@RequestBody Map<String, Object> params) {
        Object coordinateObj = params.get("coordinate");
        List<List<Integer>> coordinateList = (List<List<Integer>>) coordinateObj;
        
        int[][] coordinates = new int[coordinateList.size()][2];
        for (int i = 0; i < coordinateList.size(); i++) {
            coordinates[i][0] = coordinateList.get(i).get(0);
            coordinates[i][1] = coordinateList.get(i).get(1);
        }

        for (int[] row : coordinates) {
            System.out.println(Arrays.toString(row));
        }

        return "success";
    }
}
