import java.util.DoubleSummaryStatistics;
import java.util.List;
import java.util.function.ToDoubleFunction;

public class AggregationExample {

    public enum AggregationType {
        MIN(DoubleSummaryStatistics::getMin),
        MAX(DoubleSummaryStatistics::getMax),
        AVERAGE(DoubleSummaryStatistics::getAverage);

        private final ToDoubleFunction<DoubleSummaryStatistics> function;

        AggregationType(ToDoubleFunction<DoubleSummaryStatistics> function) {
            this.function = function;
        }

        public double apply(DoubleSummaryStatistics stats) {
            return function.applyAsDouble(stats);
        }
    }

    private List<Double> provides;

    public Float someFunction(AggregationType type) {
        DoubleSummaryStatistics stats = provides.stream()
                .mapToDouble(this::someFunc)
                .summaryStatistics();
        return (float) type.apply(stats);
    }

    private double someFunc(double value) {
        return value;
    }
}
