import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.ImageView;
import android.widget.LinearLayout;
import androidx.annotation.Nullable;
import androidx.fragment.app.Fragment;

public class MyFragment extends Fragment {

    private LinearLayout parentLayout;
    private Button addProduct;

    @Override
    public void onViewCreated(View view, @Nullable Bundle savedInstanceState) {
        super.onViewCreated(view, savedInstanceState);

        parentLayout = view.findViewById(R.id.ll_out);
        addProduct = view.findViewById(R.id.btn_prd);

        addProduct.setOnClickListener(v -> {
            LinearLayout newProductLayout = new LinearLayout(getActivity());
            newProductLayout.setOrientation(LinearLayout.HORIZONTAL);

            EditText productName = new EditText(getActivity());
            productName.setHint("Enter Product");
            productName.setLayoutParams(new LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1));

            EditText productQty = new EditText(getActivity());
            productQty.setHint("Enter Quantity");
            productQty.setLayoutParams(new LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1));

            EditText productPrice = new EditText(getActivity());
            productPrice.setHint("Prod Price");
            productPrice.setLayoutParams(new LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1));

            ImageView deleteButton = new ImageView(getActivity());
            deleteButton.setImageResource(android.R.drawable.ic_delete);
            deleteButton.setLayoutParams(new LinearLayout.LayoutParams(100, 100));
            deleteButton.setOnClickListener(v1 -> parentLayout.removeView(newProductLayout));

            newProductLayout.addView(productName);
            newProductLayout.addView(productQty);
            newProductLayout.addView(productPrice);
            newProductLayout.addView(deleteButton);

            parentLayout.addView(newProductLayout);
        });
    }
}
