import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

import org.junit.jupiter.api.Test;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

@RestController
class HappyRestController {
    private final HappyService happyService;

    public HappyRestController(HappyService happyService) {
        this.happyService = happyService;
    }

    @GetMapping(value = "/happy/{happyId}", produces = "application/json")
    public Response<?> getHappy(@PathVariable Long happyId) {
        Response response = new Response();
        response.update(happyService.getById(happyId));
        return response;
    }
}

class HappyService {
    public String getById(Long id) {
        if (id < 0) throw new UnhappyException("Invalid ID");
        return "Happy Data";
    }
}

class UnhappyException extends RuntimeException {
    public UnhappyException(String message) {
        super(message);
    }
}

class HappyRestControllerTest {
    private final HappyService happyService = mock(HappyService.class);
    private final HappyRestController controller = new HappyRestController(happyService);

    @Test
    void testUnHappy() {
        when(happyService.getById(-1L)).thenThrow(new UnhappyException("Invalid ID"));
        assertThrows(UnhappyException.class, () -> controller.getHappy(-1L));
    }
}
