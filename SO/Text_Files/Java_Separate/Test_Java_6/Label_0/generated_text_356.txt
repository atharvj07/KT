import java.util.ArrayList;
import java.util.List;

public class PatternSearch {
    public static void main(String[] args) {
        int[][] T = {
            {2, 3, 4, 3, 6},
            {4, 1, 5, 7, 8},
            {9, 1, 2, 3, 1},
            {2, 4, 5, 1, 5},
            {3, 1, 9, 0, 2}
        };

        int[][] P = {
            {2, 3, 0},
            {0, 1, 5},
            {9, 0, 2}
        };

        List<int[]> result = searchPattern(T, P);
        for (int[] pos : result) {
            System.out.println("(" + pos[0] + "," + pos[1] + ")");
        }
    }

    static List<int[]> searchPattern(int[][] T, int[][] P) {
        List<int[]> matches = new ArrayList<>();
        int m = T.length, n = T[0].length;
        int u = P.length, v = P[0].length;

        for (int i = 0; i <= m - u + 1; i++) {  
            for (int j = 0; j <= n - v; j++) {
                if (matchesAt(T, P, i, j)) {
                    matches.add(new int[]{i, j});
                }
            }
        }
        return matches;
    }

    static boolean matchesAt(int[][] T, int[][] P, int x, int y) {
        for (int i = 0; i < P.length; i++) {
            for (int j = 0; j < P[0].length; j++) {
                if (P[i][j] != 0 && P[i][j] != T[x + i][y + j]) {
                    return false;
                }
            }
        }
        return true;
    }
}
