import java.util.regex.*;

public class AzurePathParser {
    
    private static void parsePath(String path) {
        Pattern azurePathPattern = Pattern.compile("abfss?://([^/]+)@([^:/]+)(?::(\\d+))?(\\.[^/]+)?/?([^/]*)/(.+)?");
        Matcher matcher = azurePathPattern.matcher(path);
        
        if (matcher.find()) {
            String fileSystem = matcher.group(1);
            String accountName = matcher.group(5).isEmpty() ? matcher.group(2) : matcher.group(5);
            String accountSuffix = matcher.group(4) != null ? matcher.group(4) : "";
            String relativePath = matcher.group(6) != null ? matcher.group(6) : "";

            System.out.println("fileSystem: " + fileSystem);
            System.out.println("accountName: " + accountName);
            System.out.println("accountSuffix: " + accountSuffix);
            System.out.println("relativePath: " + relativePath);
        }
    }
