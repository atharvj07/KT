import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.env.Environment;
import org.springframework.stereotype.Component;

import java.util.Map;
import java.util.Properties;
import java.util.stream.Collectors;

@Component
public class PropertyLoader {

    private final Environment env;

    public PropertyLoader(Environment env) {
        this.env = env;
    }

    public Map<String, String> getPropertiesByPrefix(String prefix) {
        Properties props = new Properties();
        env.getPropertySources().forEach(ps -> {
            if (ps.getSource() instanceof Properties) {
                props.putAll((Properties) ps.getSource());
            }
        });

        return props.entrySet().stream()
                .filter(e -> e.getKey().toString().startsWith(prefix))
                .collect(Collectors.toMap(
                        e -> e.getKey().toString().replace(prefix + ".", ""),
                        e -> e.getValue().toString()
                ));
    }
}
