import java.lang.module.ModuleFinder;
import java.lang.module.ModuleReference;
import java.util.HashSet;
import java.util.Set;

public class CoreModulesCheck {
    public static void main(String[] args) {
        Set<String> coreModules = getCoreModules();
        Set<String> allModules = new HashSet<>();

        ModuleFinder.ofSystem().findAll().stream()
                .map(ModuleReference::descriptor)
                .map(m -> m.name())
                .forEach(allModules::add);

        System.out.println("Core Modules:");
        coreModules.forEach(System.out::println);

        System.out.println("\nNon-Core Modules:");
        allModules.stream()
                .filter(m -> !coreModules.contains(m))
                .forEach(System.out::println);
    }

    private static Set<String> getCoreModules() {
        Set<String> core = new HashSet<>();
        core.add("java.base");
        core.add("java.compiler");
        core.add("java.datatransfer");
        core.add("java.desktop");
        core.add("java.instrument");
        core.add("java.logging");
        core.add("java.management");
        core.add("java.management.rmi");
        core.add("java.naming");
        core.add("java.net.http");
        core.add("java.prefs");
        core.add("java.rmi");
        core.add("java.scripting");
        core.add("java.se");
        core.add("java.security.jgss");
        core.add("java.security.sasl");
        core.add("java.smartcardio");
        core.add("java.sql");
        core.add("java.sql.rowset");
        core.add("java.transaction.xa");
        core.add("java.xml");
        core.add("java.xml.crypto");
        return core;
    }
}
