import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.layout.BorderPane;
import javafx.scene.paint.Color;
import javafx.scene.paint.PhongMaterial;
import javafx.scene.shape.Box;
import javafx.scene.shape.CullFace;
import javafx.scene.shape.DrawMode;
import javafx.scene.transform.Rotate;
import javafx.stage.Stage;
import javafx.scene.input.KeyEvent;

public class DynamicDrawMode extends Application {
    private Box box;
    private boolean wireframe = false;

    public void start(Stage stage) {
        box = new Box(200, 200, 200);
        box.setMaterial(new PhongMaterial(Color.BLUE));
        box.setCullFace(CullFace.BACK);
        box.setDrawMode(DrawMode.FILL);
        box.getTransforms().add(new Rotate(30, Rotate.X_AXIS));
        box.getTransforms().add(new Rotate(30, Rotate.Y_AXIS));

        BorderPane root = new BorderPane(box);
        Scene scene = new Scene(root, 600, 400);
        scene.setOnKeyPressed(this::toggleDrawMode);

        stage.setTitle("3D Draw Mode Toggle");
        stage.setScene(scene);
        stage.show();
    }

    private void toggleDrawMode(KeyEvent event) {
        if (event.getText().equalsIgnoreCase("w")) {
            wireframe = !wireframe;
            box.setDrawMode(wireframe ? DrawMode.LINE : DrawMode.FILL);
        }
    }

    public static void main(String[] args) {
        launch(args);
    }
}
