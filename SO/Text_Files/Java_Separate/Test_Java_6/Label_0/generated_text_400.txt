package com.example.usb;

import android.content.BroadcastReceiver;
import android.content.Context;
import android.content.Intent;
import android.content.IntentFilter;
import android.hardware.usb.UsbDevice;
import android.hardware.usb.UsbManager;
import android.util.Log;

public class NativeUsbService {

    private static final String TAG = "NativeUsbService";
    private static final String ACTION_USB_PERMISSION = "com.android.example.USB_PERMISSION";
    private UsbDevice usbDevice;
    private UsbManager usbManager;
    private UsbConnection connection;
    private UsbInterface usbDeviceInterface;
    private UsbEndpoint endpointIn;
    private UsbEndpoint endpointOut;

    public NativeUsbService(Context context) {
        usbManager = (UsbManager) context.getSystemService(Context.USB_SERVICE);

        IntentFilter filter = new IntentFilter();
        filter.addAction(UsbManager.ACTION_USB_DEVICE_ATTACHED);
        filter.addAction(UsbManager.ACTION_USB_DEVICE_DETACHED);
        filter.addAction(ACTION_USB_PERMISSION);

        context.registerReceiver(usbReceiver, filter);
    }

    private final BroadcastReceiver usbReceiver = new BroadcastReceiver() {

        public void onReceive(Context context, Intent intent) {
            String action = intent.getAction();
            if (action.equals(UsbManager.ACTION_USB_DEVICE_ATTACHED)) {
                synchronized (this) {
                    UsbDevice device = (UsbDevice) intent.getParcelableExtra(UsbManager.EXTRA_DEVICE);
                    if (device != null) {
                        usbDevice = device;
                        // I'm not sure if permission is automatically granted here, so let's check
                        if (intent.getBooleanExtra(UsbManager.EXTRA_PERMISSION_GRANTED, false)) {
                            Log.d(TAG, "onReceive: Permission granted for device");
                        } else {
                            // Permission not granted, might need to request it
                            Log.d(TAG, "onReceive: Permission not granted for device");
                            // Attempt to request permission (if needed)
                            Intent permissionIntent = new Intent(ACTION_USB_PERMISSION);
                            usbManager.requestPermission(device, permissionIntent);
                        }
                    } else {
                        Log.d(TAG, "onReceive: DEVICE WAS ATTACHED AND WAS NULL :(");
                    }
                }
            } else if (action.equals(UsbManager.ACTION_USB_DEVICE_DETACHED)) {
                Log.d(TAG, "onReceive: Device was detached!");
                if (connection != null) {
                    connection.releaseInterface(usbDeviceInterface);
                    connection.close();
                }
                connection = null;
                usbDevice = null;
                endpointIn = null;
                endpointOut = null;
            } else if (action.equals(ACTION_USB_PERMISSION)) {
                if (intent.getBooleanExtra(UsbManager.EXTRA_PERMISSION_GRANTED, false)) {
                    Log.d(TAG, "onReceive: USB permission granted for device");
                    // Now that permission is granted, establish a connection with the device
                    if (usbDevice != null) {
                        // Proceed to setup the connection or handle device communication
                    }
                } else {
                    Log.d(TAG, "onReceive: USB permission denied for device");
                }
            }
        }
    };

    // Ensure to unregister the receiver when it's no longer needed
    public void unregisterReceiver(Context context) {
        context.unregisterReceiver(usbReceiver);
    }
}
