import org.springframework.scheduling.annotation.Async;
import org.springframework.scheduling.annotation.EnableAsync;
import org.springframework.stereotype.Service;
import org.springframework.web.bind.annotation.*;

import java.util.*;
import java.util.concurrent.ConcurrentHashMap;

@RestController
@RequestMapping("/test")
public class TestController {

    private final TestService testService;
    private final Map<Integer, String> statusMap = new ConcurrentHashMap<>();

    public TestController(TestService testService) {
        this.testService = testService;
    }

    @PostMapping("/start")
    public String startTests() {
        statusMap.clear();
        testService.runTests(statusMap);
        return "Tests started!";
    }

    @GetMapping("/status")
    public Map<Integer, String> getStatus() {
        return statusMap;
    }
}

@EnableAsync
@Service
class TestService {

    @Async
    public void runTests(Map<Integer, String> statusMap) {
        for (int i = 1; i <= 30; i++) {
            statusMap.put(i, "Running");
            try {
                executeTestCase(i);
                statusMap.put(i, "Completed");
            } catch (Exception e) {
                statusMap.put(i, "Failed: " + e.getMessage());
            }
        }
    }

    private void executeTestCase(int testCase) throws InterruptedException {
        Thread.sleep(5000);
    }
}
