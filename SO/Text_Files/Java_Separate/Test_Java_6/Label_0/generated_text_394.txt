import javafx.application.Application;
import javafx.beans.value.ChangeListener;
import javafx.beans.value.ObservableValue;
import javafx.scene.Scene;
import javafx.scene.control.Spinner;
import javafx.scene.control.SpinnerValueFactory;
import javafx.scene.layout.StackPane;
import javafx.stage.Stage;

public class SpinnerExample extends Application {

    @Override
    public void start(Stage primaryStage) {
        Spinner<Integer> spinnerMin = new Spinner<>();
        Spinner<Integer> spinnerMax = new Spinner<>();
        
        spinnerMin.setValueFactory(new SpinnerValueFactory.IntegerSpinnerValueFactory(2, 6, 2));
        spinnerMax.setValueFactory(new SpinnerValueFactory.IntegerSpinnerValueFactory(2, 6, 6));

        spinnerMin.valueProperty().addListener((observable, oldValue, newValue) -> {
            int newMax = newValue.intValue();
            if (newMax > spinnerMax.getValue()) {
                spinnerMax.getValueFactory().setValue(newMax);
            }
            spinnerMax.setValueFactory(new SpinnerValueFactory.IntegerSpinnerValueFactory(newValue.intValue(), 6, spinnerMax.getValue()));
            checkButtons(spinnerMin, spinnerMax);
        });

        spinnerMax.valueProperty().addListener((observable, oldValue, newValue) -> {
            int newMin = newValue.intValue();
            if (newMin < spinnerMin.getValue()) {
                spinnerMin.getValueFactory().setValue(newMin);
            }
            spinnerMin.setValueFactory(new SpinnerValueFactory.IntegerSpinnerValueFactory(2, newValue.intValue(), spinnerMin.getValue()));
            checkButtons(spinnerMin, spinnerMax);
        });

        StackPane root = new StackPane();
        root.getChildren().addAll(spinnerMin, spinnerMax);

        Scene scene = new Scene(root, 300, 250);
        primaryStage.setTitle("Spinner Example");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    private void checkButtons(Spinner<Integer> spinnerMin, Spinner<Integer> spinnerMax) {
        if (spinnerMin.getValue() == spinnerMax.getValue()) {
            spinnerMin.getEditor().setStyle("-fx-background-color: lightgray;");
            spinnerMax.getEditor().setStyle("-fx-background-color: lightgray;");
        } else {
            spinnerMin.getEditor().setStyle("-fx-background-color: white;");
            spinnerMax.getEditor().setStyle("-fx-background-color: white;");
        }
    }

    public static void main(String[] args) {
        launch(args);
    }
}
