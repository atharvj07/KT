import java.util.*;

public class SumWithLimit {
    
    public static List<Integer> findNumbers(int[] nums, int target, int limit) {
        Arrays.sort(nums);
        List<Integer> result = new ArrayList<>();
        if (backtrack(nums, target, limit, 0, result)) {
            return result;
        }
        return Collections.emptyList();
    }

    private static boolean backtrack(int[] nums, int target, int limit, int index, List<Integer> result) {
        if (limit == 0 && target == 0) return true;
        if (limit == 0 || index >= nums.length) return false;

        for (int i = index; i < nums.length; i++) {
            if (nums[i] > target) break;
            result.add(nums[i]);
            if (backtrack(nums, target - nums[i], limit - 1, i + 1, result)) {
                return true;
            }
            result.remove(result.size() - 1);
        }
        return false;
    }

    public static void main(String[] args) {
        int[] nums = {2, 7, 11, 15};
        int target = 20;
        int limit = 3;
        System.out.println(findNumbers(nums, target, limit));
    }
}
