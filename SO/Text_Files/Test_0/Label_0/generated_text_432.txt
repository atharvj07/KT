from itertools import product

lasts = ['x', 'y', 'z']
firsts = ['a', 'b', 'c']

for last, first in product(lasts, firsts):
    print(last, first)

lasts = ['x', 'y', 'z']
firsts = ['a', 'b', 'c']

for last in lasts:
    for first in firsts:
        print(last, first)
