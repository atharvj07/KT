from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()

class ProcessIdRequest(BaseModel):
    process_id: int

@app.post("/do_something/")
async def do_something(request: ProcessIdRequest):
    return {"process_id": request.process_id}
