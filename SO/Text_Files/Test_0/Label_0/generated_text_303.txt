import trimesh

# Load the mesh
mesh = trimesh.load_mesh('mushroom-shelve-1-merged.obj')

# Convert the mesh to quads (this works if the mesh is simple enough)
mesh = mesh.quads.to_trimesh()

# Export to a new file
mesh.export('quad_mesh.obj')
