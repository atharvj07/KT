import polars as pl

# Create the DataFrame
df = pl.DataFrame({
    "a": [1, 2, 3, 4, 5],
    "b": [2, 3, 4, 5, 6],
    "x": [1, 3, 5, 7, 9]
})

# Apply binning to column 'x' and append the result as 'x_cut'
df = df.with_columns(
    pl.col("x").cut(bins=[2, 4, 6]).alias("x_cut")
)

print(df)
