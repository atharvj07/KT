import polars as pl

# Create a LazyFrame
lf = pl.LazyFrame({
    "col_a": [1, None, 3, None, 5],
    "col_b": [10, 20, 30, 40, 50]
})

# Fill null values in col_a with values from col_b
lf_filled = lf.with_columns(
    pl.col("col_a").fill_null(pl.col("col_b"))
)

# Collect to execute
df_result = lf_filled.collect()

print(df_result)
