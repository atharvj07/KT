interface Show1 {
    default void display() {
        System.out.println("hello");
    }
}

interface Show2 {
    void display();
}

public class Person implements Show1, Show2 {

    public static void main(String args[]) {
        Person p = new Person();
        p.display();
    }

    @Override
    public void display() {
        Show1.super.display();
    }
}
