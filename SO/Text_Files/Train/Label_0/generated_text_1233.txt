import java.util.*;

public class RandomPairing {
    public static void main(String[] args) {
        int[] input = {1, 3, 4, 5, 6, 7, 8}; // Example input with odd numbers
        List<Integer> numbers = new ArrayList<>();
        
        for (int num : input) {
            numbers.add(num);
        }
        
        Collections.shuffle(numbers); // Shuffle for randomness
        
        List<List<Integer>> pairs = new ArrayList<>();
        
        for (int i = 0; i < numbers.size() - 1; i += 2) {
            if (i + 2 == numbers.size()) { 
                // If we reach the last element, form a triplet
                pairs.add(Arrays.asList(numbers.get(i), numbers.get(i + 1), numbers.get(i + 2)));
                break;
            } else {
                pairs.add(Arrays.asList(numbers.get(i), numbers.get(i + 1)));
            }
        }
        
        // Print result
        for (List<Integer> pair : pairs) {
            System.out.println(pair);
        }
    }
}
