import java.io.*;
import java.lang.*;
import java.util.*;

class GFG {

    // Pair class to represent activity (start time, finish time)
    static class Pair {

        int first; // start time
        int second; // finish time

        Pair(int first, int second) {
            this.first = first;
            this.second = second;
        }
    }

    // Function to select activities
    static void SelectActivities(int s[], int f[]) {

        // Vector to store the selected activities
        ArrayList<Pair> ans = new ArrayList<>();

        // Create a list of activities with start and finish times
        List<Pair> activities = new ArrayList<>();
        for (int i = 0; i < s.length; i++) {
            activities.add(new Pair(s[i], f[i]));
        }

        // Sort activities by start time first
        activities.sort((a, b) -> Integer.compare(a.first, b.first));

        // Now, select the first activity with the earliest start time
        Pair firstActivity = activities.get(0);
        int start = firstActivity.first;
        int end = firstActivity.second;
        ans.add(new Pair(start, end));

        // Proceed with greedy approach: Select non-overlapping activities
        for (int i = 1; i < activities.size(); i++) {
            Pair currentActivity = activities.get(i);
            // If the start time of the current activity is greater than or equal to the end time of the last selected activity
            if (currentActivity.first >= end) {
                start = currentActivity.first;
                end = currentActivity.second;
                ans.add(new Pair(start, end));
            }
        }

        // Print the selected activities
        System.out.println("Following Activities should be selected: ");
        for (Pair itr : ans) {
            System.out.println("Activity started at: " + itr.first + " and ends at " + itr.second);
        }
    }

    // Driver code
    public static void main(String[] args) {

        int s[] = { 1, 3, 0, 5, 8, 5 };
        int f[] = { 2, 4, 6, 7, 9, 9 };

        // Function call
        SelectActivities(s, f);
    }
}
