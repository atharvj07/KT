Thread thread = new Thread(() -> {
       // Thread task
       System.out.println("Thread task here");
   });
   thread.setPriority(Thread.MAX_PRIORITY); // or Thread.MIN_PRIORITY, Thread.NORM_PRIORITY
   thread.start();
   
   private static final Object printLock = new Object();

   public void printMessage(String message) {
       synchronized (printLock) {
           System.out.println(message);
       }
   }

   // Usage in multiple threads
   Thread thread1 = new Thread(() -> printMessage("Thread 1 printing"));
   Thread thread2 = new Thread(() -> printMessage("Thread 2 printing"));

   thread1.start();
   thread2.start();
   
   ExecutorService executor = Executors.newFixedThreadPool(1); // Ensure only one thread at a time
   executor.submit(() -> {
       printMessage("Task 1 printing");
   });
   executor.submit(() -> {
       printMessage("Task 2 printing");
   });
   
   private static final Object printLock = new Object();
   private static boolean isPrinting = false;

   public void printMessage(String message) {
       synchronized (printLock) {
           while (isPrinting) {
               try {
                   printLock.wait(); // Wait until it's allowed to print
               } catch (InterruptedException e) {
                   Thread.currentThread().interrupt();
               }
           }
           isPrinting = true;
           System.out.println(message);
           isPrinting = false;
           printLock.notify(); // Notify the next waiting thread
       }
   }
   
   // Thread task
   System.out.println("Thread task here");
   synchronized (printLock) {
       System.out.println(message);
   }
   printMessage("Task 1 printing");
   printMessage("Task 2 printing");
   synchronized (printLock) {
       while (isPrinting) {
           try {
               printLock.wait(); // Wait until it's allowed to print
           } catch (InterruptedException e) {
               Thread.currentThread().interrupt();
           }
       }
       isPrinting = true;
       System.out.println(message);
       isPrinting = false;
       printLock.notify(); // Notify the next waiting thread
   }
