@Entity
@Table(name = "house")
public class House {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "id")
    private long id;

    @OneToMany(mappedBy = "house")
    private Set<HouseOwner> houseOwners = new HashSet<>();
    
    // getters and setters
}

@Entity
@Table(name = "owner")
public class Owner {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "id")
    private long id;

    @OneToMany(mappedBy = "owner")
    private Set<HouseOwner> houseOwners = new HashSet<>();
    
    // getters and setters
}

@Entity
@Table(name = "house_owner")
public class HouseOwner {
    @EmbeddedId
    private HouseOwnerId id;
    
    @ManyToOne
    @JoinColumn(name = "house_id", referencedColumnName = "id", insertable = false, updatable = false)
    private House house;

    @ManyToOne
    @JoinColumn(name = "owner_id", referencedColumnName = "id", insertable = false, updatable = false)
    private Owner owner;

    @CreatedTimestamp
    private LocalDateTime createdAt;

    // getters and setters
}

@Embeddable
public class HouseOwnerId implements Serializable {
    @Column(name = "house_id")
    private long houseId;

    @Column(name = "owner_id")
    private long ownerId;

    // constructors, equals, hashCode methods
}
