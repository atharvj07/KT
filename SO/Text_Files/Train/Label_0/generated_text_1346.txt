import com.microsoft.playwright.*;

public class PwDownload {
    public static void main(String[] args) {
        try (Playwright playwright = Playwright.create()) {
            // Initialize browser types
            final BrowserType chromium = playwright.chromium();
            final BrowserType webkit = playwright.webkit(); // Using WebKit for testing

            // Launch browser (headless mode is often useful for downloads)
            Browser browser = chromium.launch(new BrowserType.LaunchOptions().setHeadless(true));
            // For WebKit:
            // Browser browser = webkit.launch(new BrowserType.LaunchOptions().setHeadless(true));

            // Open a new page
            Page page = browser.newPage();

            // Listen for download event
            Download download = page.waitForDownload(() -> {
                // Navigate to the file URL (in this case, itÂs a direct file link)
                page.navigate("https://www.fnac.es/sitemap-top-post.xml.gz");
            });

            // After the download is triggered, you can get the download path
            System.out.println("Downloaded file path: " + download.path());

            // Close the browser after download
            browser.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
