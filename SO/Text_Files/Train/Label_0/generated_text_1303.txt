import org.apache.poi.ss.usermodel.Header;
import org.apache.poi.ss.usermodel.Sheet;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;
import org.apache.poi.xssf.usermodel.XSSFPictureData;
import org.apache.poi.xssf.usermodel.XSSFRichTextString;
import org.apache.poi.ss.util.ImageUtils;
import java.io.FileInputStream;
import java.io.IOException;

public class ExcelHeaderImageExample {

    public void createOverview(XSSFWorkbook workbook) throws IOException {
        // Create a sheet
        Sheet overview = workbook.createSheet("Overview");
        
        // Load the image file (e.g., a PNG file)
        FileInputStream imageInputStream = new FileInputStream("path/to/your/image.png");
        byte[] imageBytes = imageInputStream.readAllBytes();
        imageInputStream.close();

        // Add the image to the workbook
        int pictureIndex = workbook.addPicture(imageBytes, XSSFWorkbook.PICTURE_TYPE_PNG);

        // Access the header and set the image
        Header header = overview.getHeader();
        
        // Set the image in the center, right, or left (using the &[Picture] syntax)
        header.setCenter("&[Picture]");
        
        // Optional: Set the footer
        overview.getFooter().setCenter("Page &[Page] of &[Pages]");
    }

    public static void main(String[] args) throws IOException {
        XSSFWorkbook workbook = new XSSFWorkbook();
        ExcelHeaderImageExample example = new ExcelHeaderImageExample();
        example.createOverview(workbook);

        // Write workbook to file
        try (FileOutputStream fileOut = new FileOutputStream("workbook_with_image.xlsx")) {
            workbook.write(fileOut);
        }
    }
}
