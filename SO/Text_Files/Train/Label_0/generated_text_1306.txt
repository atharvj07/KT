class Itens {
    private ObjectProperty<BigDecimal> value = new SimpleObjectProperty<>(BigDecimal.ZERO);

    public BigDecimal getValue() {
        return value.get();
    }

    public void setValue(BigDecimal b) {
        value.set(b);
    }

    public ObjectProperty<BigDecimal> valueProperty() {
        return value;
    }
}

void createExempleMenu() {
    ObservableList<Itens> orc = FXCollections.observableArrayList(o -> new Observable[] { o.valueProperty() });

    Label valueFinalLabel = new Label();

    // Create the binding for the sum of values
    valueFinalLabel.textProperty().bind(
        Bindings.createObjectBinding(() -> {
            if (orc == null || orc.isEmpty()) {
                return BigDecimal.ZERO; // case orc is empty
            } else {
                // Sum all the values in the list
                return orc.stream()
                          .map(Itens::getValue)
                          .reduce(BigDecimal.ZERO, BigDecimal::add);
            }
        }, orc) // ObservableList triggers binding updates
        .asString()
    );
}
