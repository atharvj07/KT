import java.time.LocalTime;
import java.time.format.DateTimeFormatter;
import java.util.Locale;

public class TimePeriodFormatter {

    public static void main(String[] args) {
        DateTimeFormatter FORMATTER = DateTimeFormatter.ofPattern("HH:mm", Locale.ENGLISH);

        System.out.println(formatTimeWithPeriod(LocalTime.of(8, 20, 30)));  // Expected output: in the morning
        System.out.println(formatTimeWithPeriod(LocalTime.of(20, 20, 30))); // Expected output: in the evening
        System.out.println(formatTimeWithPeriod(LocalTime.of(13, 45, 0)));  // Expected output: in the afternoon
        System.out.println(formatTimeWithPeriod(LocalTime.of(23, 59, 59))); // Expected output: at night
    }

    public static String formatTimeWithPeriod(LocalTime time) {
        // Determine period of day
        String period;
        int hour = time.getHour();

        if (hour >= 5 && hour < 12) {
            period = "in the morning";
        } else if (hour >= 12 && hour < 17) {
            period = "in the afternoon";
        } else if (hour >= 17 && hour < 21) {
            period = "in the evening";
        } else {
            period = "at night";
        }

        // Format time and append period
        DateTimeFormatter timeFormatter = DateTimeFormatter.ofPattern("h:mm a", Locale.ENGLISH);
        return time.format(timeFormatter) + " " + period;
    }
}
