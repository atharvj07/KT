import com.fasterxml.jackson.annotation.JsonInclude;

@JsonInclude(JsonInclude.Include.NON_NULL)
public class Simple {
    private String name;
    private String email;

    // Getters and Setters
    public String getName() { return name; }
    public void setName(String name) { this.name = name; }

    public String getEmail() { return email; }
    public void setEmail(String email) { this.email = email; }
}

import com.fasterxml.jackson.databind.annotation.JsonDeserialize;
import java.util.List;

@JsonDeserialize(using = SimpleWrapperDeserializer.class)
public class SimpleWrapper {
    private List<Simple> simple;

    public List<Simple> getSimple() { return simple; }
    public void setSimple(List<Simple> simple) { this.simple = simple; }
}

import com.fasterxml.jackson.core.JsonParser;
import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.DeserializationContext;
import com.fasterxml.jackson.databind.JsonDeserializer;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.io.IOException;
import java.util.Collections;
import java.util.List;

public class SimpleWrapperDeserializer extends JsonDeserializer<SimpleWrapper> {
    private final ObjectMapper objectMapper = new ObjectMapper();

    @Override
    public SimpleWrapper deserialize(JsonParser p, DeserializationContext ctxt) 
            throws IOException, JsonProcessingException {
        
        JsonNode node = p.getCodec().readTree(p);
        SimpleWrapper wrapper = new SimpleWrapper();

        if (node.isObject()) {
            // Single object case: Convert it into a list with one element
            Simple simple = objectMapper.treeToValue(node, Simple.class);
            wrapper.setSimple(Collections.singletonList(simple));
        } else if (node.isArray()) {
            // List case: Convert JSON array into List<Simple>
            List<Simple> simpleList = objectMapper.readValue(node.toString(),
                    objectMapper.getTypeFactory().constructCollectionType(List.class, Simple.class));
            wrapper.setSimple(simpleList);
        }

        return wrapper;
    }
}

import com.fasterxml.jackson.databind.ObjectMapper;

public class Main {
    public static void main(String[] args) throws Exception {
        ObjectMapper objectMapper = new ObjectMapper();

        String json1 = "{\"simple\": {\"name\":\"c\", \"email\" : \"c@z.com\"} }";
        String json2 = "{\"simple\": [ {\"name\":\"a\", \"email\" : \"a@z.com\"}, {\"name\":\"b\", \"email\" : \"b@z.com\"} ]}";

        SimpleWrapper wrapper1 = objectMapper.readValue(json1, SimpleWrapper.class);
        SimpleWrapper wrapper2 = objectMapper.readValue(json2, SimpleWrapper.class);

        System.out.println("Case 1 (Single Object): " + wrapper1.getSimple());
        System.out.println("Case 2 (List): " + wrapper2.getSimple());
    }
}
