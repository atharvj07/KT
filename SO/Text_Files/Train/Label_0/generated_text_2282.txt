def parse_numeric_string(s: str) -> float:
    multipliers = {'k': 1_000, 'M': 1_000_000, 'B': 1_000_000_000}
    if s[-1] in multipliers:
        return float(s[:-1]) * multipliers[s[-1]]
    return float(s)
