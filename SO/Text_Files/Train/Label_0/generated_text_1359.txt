import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;
import com.twilio.twiml.voice.VoiceResponse;
import com.twilio.twiml.voice.Dial;
import com.twilio.twiml.voice.Number;
import com.twilio.twiml.voice.Redirect;
import org.springframework.http.ResponseEntity;

@SpringBootApplication
public class TwilioForwardingApp {

    public static void main(String[] args) {
        SpringApplication.run(TwilioForwardingApp.class, args);
    }
}

@RestController
class TwilioController {

    // This endpoint will handle incoming calls and forward them
    @GetMapping("/incomingCall")
    public ResponseEntity<String> forwardCall(@RequestParam String to) {
        try {
            // Set the personal number where the call should be forwarded
            String forwardToNumber = "YOUR_PERSONAL_NUMBER"; // Replace with your personal number

            // Create the TwiML to forward the call
            Dial dial = new Dial.Builder()
                    .number(new Number(forwardToNumber))
                    .build();

            // Create the response and return TwiML
            VoiceResponse response = new VoiceResponse.Builder()
                    .dial(dial)
                    .build();

            // Return the TwiML response as XML
            return ResponseEntity.ok(response.toXml());
        } catch (Exception e) {
            return ResponseEntity.status(500).body("<Response><Say>Error processing the call.</Say></Response>");
        }
    }
}
