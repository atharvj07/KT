import polars as pl

input_df = pl.DataFrame(
    {"idx": [1, 1, 2, 2, 3, 3],
     "values": [["A", "B"], ["B", "C"], ["A", "B"], ["B", "C"], ["A", "B"], ["B", "C"]]
     }
)

output_df = (
    input_df.explode("values")
    .groupby("idx")
    .agg_list("values")
    .select([
        pl.col("idx"),
        pl.col("values").apply(lambda s: list(set.intersection(*map(set, s))))
    ])
)

print(output_df)
