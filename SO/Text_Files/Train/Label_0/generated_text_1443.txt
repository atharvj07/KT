public interface AuthClient {
    boolean authorize(String uid, String pass);
}

@Component("gmailClient")
public class GmailAuthClient implements AuthClient {
    @Override
    public boolean authorize(String uid, String pass) {
        // Implement Gmail authentication logic
        return "gmail_user".equals(uid) && "password".equals(pass);
    }
}

@Component("yahooClient")
public class YahooAuthClient implements AuthClient {
    @Override
    public boolean authorize(String uid, String pass) {
        // Implement Yahoo authentication logic
        return "yahoo_user".equals(uid) && "password".equals(pass);
    }
}

@Component
public class AuthHelper {
    private final AuthClient gmailClient;
    private final AuthClient yahooClient;

    @Autowired
    public AuthHelper(@Qualifier("gmailClient") AuthClient gmailClient,
                      @Qualifier("yahooClient") AuthClient yahooClient) {
        this.gmailClient = gmailClient;
        this.yahooClient = yahooClient;
    }

    public AuthClient getClient(String option) {
        if ("gmail".equalsIgnoreCase(option)) return gmailClient;
        if ("yahoo".equalsIgnoreCase(option)) return yahooClient;
        throw new IllegalArgumentException("Invalid client option: " + option);
    }
}

@Service
public class AuthService {
    private final AuthHelper authHelper;

    @Autowired
    public AuthService(AuthHelper authHelper) {
        this.authHelper = authHelper;
    }

    public boolean authenticate(String option, String uid, String pass) {
        return authHelper.getClient(option).authorize(uid, pass);
    }
}
