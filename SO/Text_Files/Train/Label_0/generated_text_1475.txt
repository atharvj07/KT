public void deliveryBtnClicked() {
    FXMLLoader loader = new FXMLLoader(getClass().getResource("/fragments/customers.fxml"));
    homePane.getChildren().remove(homePane.getCenter());
    try {
        Pane pane = loader.load();
        
        // Get the controller and set reference to HomeController
        CustomerController customerController = loader.getController();
        customerController.setHomeController(this);
        
        homePane.setCenter(pane);
    } catch (IOException e) {
        e.printStackTrace();
    }
}

public class CustomerController {

    private HomeController homeController; // Reference to the main controller

    public void setHomeController(HomeController homeController) {
        this.homeController = homeController;
    }

    @FXML
    public void loadOrderView() throws IOException {
        Customer customer = customerTable.getSelectionModel().getSelectedItem();
        
        if (homeController == null) {
            System.out.println("HomeController is null! Cannot switch view.");
            return;
        }

        // Remove current center
        homeController.getHomePane().getChildren().remove(homeController.getHomePane().getCenter());

        // Load order view
        FXMLLoader loader = new FXMLLoader(getClass().getResource("/fragments/order.fxml"));
        Pane orderView = loader.load();

        // Set new center
        homeController.getHomePane().setCenter(orderView);
    }
}
