import software.amazon.awssdk.services.s3.S3AsyncClient;
import software.amazon.awssdk.services.s3.model.ListObjectsV2Request;
import software.amazon.awssdk.services.s3.model.S3Object;
import software.amazon.awssdk.services.s3.paginators.ListObjectsV2Publisher;
import reactor.core.publisher.Mono;

import java.util.concurrent.CountDownLatch;

public class S3ListObjectsAsync {

    public static void main(String[] args) throws InterruptedException {
        // Initialize the async S3 client
        S3AsyncClient s3AsyncClient = S3AsyncClient.create();

        // Create the request
        ListObjectsV2Request request = ListObjectsV2Request.builder().bucket("my-bucket").build();

        // Get the paginator
        ListObjectsV2Publisher response = s3AsyncClient.listObjectsV2Paginator(request);

        // Using CountDownLatch to wait for the async process to finish
        CountDownLatch latch = new CountDownLatch(1);

        // Subscribe to the stream and print the objects
        response.contents().doOnTerminate(latch::countDown)  // Ensure we count down when the stream finishes
                .subscribe(s3object -> System.out.println("Object: " + s3object.key()));

        // Wait until the latch is counted down (when async process is complete)
        latch.await();

        // Close the S3 client when done
        s3AsyncClient.close();
    }
}
