public class AccountCreateDto {

    private final String name;
    private final Long accountNo;
    private final String email;
    private final LocalDate dob;
    private final String address;
    private final Long phone;
    private final Integer amount;
    private final LocalDate accountOpeningDate;
    private final Integer installmentAmount;
    private final Integer totalAmount;
    private final LocalDate installmentPaidDate;

    // Existing constructor and fields omitted for brevity...

    // Static builder class
    public static class Builder {
        private String name;
        private Long accountNo;
        private String email;
        private LocalDate dob;
        private String address;
        private Long phone;
        private Integer amount;
        private LocalDate accountOpeningDate;
        private Integer installmentAmount;
        private Integer totalAmount;
        private LocalDate installmentPaidDate;

        public Builder setName(String name) {
            this.name = name;
            return this;
        }

        public Builder setAccountNo(Long accountNo) {
            this.accountNo = accountNo;
            return this;
        }

        public Builder setEmail(String email) {
            this.email = email;
            return this;
        }

        public Builder setDob(LocalDate dob) {
            this.dob = dob;
            return this;
        }

        public Builder setAddress(String address) {
            this.address = address;
            return this;
        }

        public Builder setPhone(Long phone) {
            this.phone = phone;
            return this;
        }

        public Builder setAmount(Integer amount) {
            this.amount = amount;
            return this;
        }

        public Builder setAccountOpeningDate(LocalDate accountOpeningDate) {
            this.accountOpeningDate = accountOpeningDate;
            return this;
        }

        public Builder setInstallmentAmount(Integer installmentAmount) {
            this.installmentAmount = installmentAmount;
            return this;
        }

        public Builder setTotalAmount(Integer totalAmount) {
            this.totalAmount = totalAmount;
            return this;
        }

        public Builder setInstallmentPaidDate(LocalDate installmentPaidDate) {
            this.installmentPaidDate = installmentPaidDate;
            return this;
        }

        // Build method to create AccountCreateDto instance
        public AccountCreateDto build() {
            return new AccountCreateDto(
                    name, accountNo, email, dob, address, phone, amount, 
                    accountOpeningDate, installmentAmount, totalAmount, 
                    installmentPaidDate
            );
        }
    }
}

public class AccountCreateDtoTest {

    @Test
    public void testAccountCreation() {
        // Create an AccountCreateDto with dynamic values for accountOpeningDate and installmentPaidDate
        AccountCreateDto dto = new AccountCreateDto.Builder()
                .setName("John Doe")
                .setAccountNo(123456L)
                .setEmail("john.doe@example.com")
                .setDob(LocalDate.of(1990, 1, 1))
                .setAddress("123 Main St")
                .setPhone(1234567890L)
                .setAmount(1000)
                .setAccountOpeningDate(LocalDate.now()) // dynamic value for accountOpeningDate
                .setInstallmentPaidDate(LocalDate.now()) // dynamic value for installmentPaidDate
                .build();

        // Perform assertions or testing with the DTO
        assertNotNull(dto);
        assertEquals(LocalDate.now(), dto.getAccountOpeningDate());
    }
}

}
