import java.util.ArrayList;

public class ShoppingCart {

    private String customerName = "none";
    private String currentDate = "January 1, 2016";

    // Parameterized constructor
    ShoppingCart(String Name, String Date) {
        this.customerName = Name;  // Assign to instance variable
        this.currentDate = Date;    // Assign to instance variable
    }

    // Default constructor
    ShoppingCart() {
        this.customerName = "none"; 
        this.currentDate = "January 1, 2016";
    }

    // Constructor with only customer name
    ShoppingCart(String Name) {
        this.customerName = Name;
    }

    // Getter methods
    public String getCustomerName() {
        return customerName;
    }

    public String getDate() {
        return currentDate;
    }
}

public class ShoppingCartManager {
    public static void main(String[] args) {
        ShoppingCart cart1 = new ShoppingCart("John Doe", "February 1, 2016");
        System.out.println(cart1.getCustomerName()); // Output: John Doe
    }
}
