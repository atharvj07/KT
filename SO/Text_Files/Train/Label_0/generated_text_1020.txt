    "localStorage.setItem('showNotification', 'true'); location.reload();"
);

@Route("")
@PreserveOnRefresh
public class MainView extends VerticalLayout implements AfterNavigationObserver {

    @Override
    public void afterNavigation(AfterNavigationEvent event) {
        UI.getCurrent().getPage().executeJs(
            "if(localStorage.getItem('showNotification') === 'true') { " +
            "  localStorage.removeItem('showNotification'); " +
            "  $0.$server.showNotification();" +
            "}"
        , this);
    }

    @ClientCallable
    public void showNotification() {
        Notification.show("Success", 5000, Notification.Position.MIDDLE);
    }
}

VaadinSession.getCurrent().setAttribute("showNotification", true);
UI.getCurrent().getPage().reload();

@Route("")
public class MainView extends VerticalLayout {

    public MainView() {
        Boolean showNotification = (Boolean) VaadinSession.getCurrent().getAttribute("showNotification");
        if (Boolean.TRUE.equals(showNotification)) {
            VaadinSession.getCurrent().setAttribute("showNotification", null); // Clear it
            Notification.show("Success", 5000, Notification.Position.MIDDLE);
        }
    }
}
