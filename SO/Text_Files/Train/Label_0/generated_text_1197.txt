import java.util.*;

class HelloWorld {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        boolean noException = true;
        
        try {
            int n = sc.nextInt();
        } catch (Exception e) {
            System.out.println(e);
            noException = false;
        }

        if (noException) {
            System.out.println("No exception is raised");
        }
    }
}

import java.util.*;

class HelloWorld {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        boolean noException = true;

        try {
            int n = sc.nextInt();
        } catch (Exception e) {
            System.out.println(e);
            noException = false;
        } finally {
            if (noException) {
                System.out.println("No exception is raised");
            }
        }
    }
}
