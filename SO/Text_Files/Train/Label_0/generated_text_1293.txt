for (int i = 0; i < board.length; i++) {
    for (int j = 0; j < board[i].length; j++) {
        String buttonText = board[i][j];  // Get the string for the button
        Button word = new Button(buttonText);  // Create a button with the text
        word.setOnMouseClicked(new MouseClickedHandler(word));  // Attach handler
        grid.add(word, i, j);  // Add the button to the grid
    }
}

private class MouseClickedHandler implements EventHandler<MouseEvent> {
    private Button button;

    public MouseClickedHandler(Button button) {
        this.button = button;  // Save the reference to the clicked button
    }

    @Override
    public void handle(MouseEvent event) {
        String text = button.getText();
        if (text.charAt(0) == 'A' || text.charAt(0) == 'a') {
            button.setText("X");
        }
    }
}
