import pandas as pd
import numpy as np

# Sample DataFrame
df = pd.DataFrame({'A': [np.nan, 0.5, 0.5, 0.5, 0.5],
                   'B': [np.nan, 3, 4, 1, 2],
                   'C': [10, np.nan, np.nan, np.nan, np.nan]})

# Calculate col 'D'
df['D'] = df['C'] * df['A'].fillna(1).cumprod() + df['B'].shift().fillna(0)

print(df)
