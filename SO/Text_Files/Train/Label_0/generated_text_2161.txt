import multiprocessing

class CustomLock:
    def __init__(self, lock):
        self.lock = lock

    def __enter__(self):
        if not self.lock.acquire(block=False):
            raise RuntimeError("Could not acquire lock")
        return self.lock

    def __exit__(self, exc_type, exc_val, exc_tb):
        self.lock.release()
