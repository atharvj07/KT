import javafx.scene.control.Button;
import javafx.event.ActionEvent;
import java.util.HashMap;
import java.util.Map;

public void addEventListeners() {
    Map<Button, EventHandler<ActionEvent>> buttonActions = new HashMap<>();

    buttonActions.put(cleanButton, e -> handleCleanButton(e));
    buttonActions.put(advSett, e -> handleAdvSett(e));
    buttonActions.put(imageLoaderItem, e -> handleImageLoaderItem(e));
    buttonActions.put(outputButton, e -> handleOutputButton(e));

    // Set the listeners in one loop
    buttonActions.forEach((button, handler) -> button.setOnAction(handler));
}
