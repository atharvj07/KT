import java.awt.AlphaComposite;
import java.awt.Color;
import java.awt.Graphics2D;
import java.awt.geom.Area;
import java.awt.geom.Ellipse2D;
import java.awt.geom.Rectangle2D;
import java.awt.image.BufferedImage;
import java.io.File;
import java.io.IOException;
import java.net.URL;

import javax.imageio.ImageIO;

public class ImageProcessing {

    public static void main(String[] args) {
        try {      
            // Load the original image
            BufferedImage original = ImageIO.read(new URL("https://eskipaper.com/images/abstract-art-1.jpg"));

            // Define the sub-image region (ensure it doesn't exceed bounds)
            int x = 500, y = 400, width = 1050, height = 300;
            
            // Create a new BufferedImage with transparency
            BufferedImage subImage = new BufferedImage(width, height, BufferedImage.TYPE_INT_ARGB);
            Graphics2D g2d = subImage.createGraphics();

            // Draw the sub-image onto the new transparent image
            g2d.drawImage(original.getSubimage(x, y, width, height), 0, 0, null);

            // Define the entire area of the sub-image
            Rectangle2D.Double rect = new Rectangle2D.Double(0, 0, width, height);
            Area fullArea = new Area(rect);

            // Define the region of interest (ROI)
            Ellipse2D.Double ellipse = new Ellipse2D.Double(100, 50, 850, 200); // Adjusted position
            Area roi = new Area(ellipse);

            // Subtract the ROI from the full area to get the region to be cleared
            fullArea.subtract(roi);

            // Set composite mode to make areas transparent
            g2d.setComposite(AlphaComposite.Clear);
            g2d.fill(fullArea); // Fill everything except ROI with transparency

            // Cleanup
            g2d.dispose();

            // Save output
            ImageIO.write(subImage, "png", new File("abstract-art-out.png"));

        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
