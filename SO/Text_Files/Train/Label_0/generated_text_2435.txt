from dataclasses import dataclass

@dataclass
class X:
    x: int = 1
    y: int = 2

@dataclass
class Y:
    c1: X
    c2: X

    def __init__(self):
        self.c1 = X(3, 4)  # Initialize c1 with a new instance of X
        self.c2 = X(5, 6)  # Initialize c2 with a new instance of X

n1 = Y()
n2 = Y()

print(id(n1.c1))
print(id(n2.c1))

n1.c1.x = 99999
print(n2)
