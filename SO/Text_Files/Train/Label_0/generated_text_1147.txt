import org.apache.commons.jexl3.*;
import java.util.*;
import java.util.regex.*;

public class JexlRegexMatchExample {
    public static void main(String[] args) {
        JexlEngine jexl = new JexlBuilder().create();

        // Example target string
        String targetString = "abcdef";

        // Example array of regular expressions
        String[] regexArray = {"abc.*", "a.d.*"};

        // Preprocess the array of regular expressions into Pattern objects
        List<Pattern> patterns = new ArrayList<>();
        for (String regex : regexArray) {
            patterns.add(Pattern.compile(regex));
        }

        // Create JEXL context and set the preprocessed patterns
        JexlContext context = new MapContext();
        context.set("target", targetString);
        context.set("patterns", patterns);

        // Define the JEXL expression to match the target against the patterns
        String expression = "patterns.stream().anyMatch(p -> p.matcher(target).matches())";

        // Evaluate the expression
        JexlExpression e = jexl.createExpression(expression);
        Object result = e.evaluate(context);

        System.out.println("Match result: " + result);
    }
}

import org.apache.commons.jexl3.*;

public class RegexMatchFunction {
    public static boolean matchAny(String target, String[] regexes) {
        for (String regex : regexes) {
            if (target.matches(regex)) {
                return true;
            }
        }
        return false;
    }
}

public class CustomJexlExample {
    public static void main(String[] args) {
        JexlEngine jexl = new JexlBuilder().create();

        // Register the custom function
        jexl.setFunctions("regexMatch", RegexMatchFunction.class);

        // Example target string
        String targetString = "abcdef";

        // Example array of regular expressions
        String[] regexArray = {"abc.*", "a.d.*"};

        // Create JEXL context and set variables
        JexlContext context = new MapContext();
        context.set("target", targetString);
        context.set("patterns", regexArray);

        // Define the JEXL expression
        String expression = "regexMatch.matchAny(target, patterns)";

        // Evaluate the expression
        JexlExpression e = jexl.createExpression(expression);
        Object result = e.evaluate(context);

        System.out.println("Match result: " + result);
    }
}
