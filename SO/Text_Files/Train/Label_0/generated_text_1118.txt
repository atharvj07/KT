import org.junit.jupiter.api.extension.AfterTestExecutionCallback;
import org.junit.jupiter.api.extension.BeforeTestExecutionCallback;
import org.junit.jupiter.api.extension.ExtensionContext;
import org.junit.jupiter.api.extension.TestExecutionExceptionHandler;

public class TimeoutLoggingExtension implements BeforeTestExecutionCallback, AfterTestExecutionCallback, TestExecutionExceptionHandler {

    @Override
    public void beforeTestExecution(ExtensionContext context) throws Exception {
        // Optional: You can add any pre-test execution logic if needed
    }

    @Override
    public void afterTestExecution(ExtensionContext context) throws Exception {
        // After the test has been executed, check for timeouts and log stack trace
        if (context.getExecutionException().isPresent()) {
            Throwable throwable = context.getExecutionException().get();
            if (throwable instanceof java.util.concurrent.TimeoutException) {
                printAllThreadStackTraces();
            }
        }
    }

    @Override
    public void handleTestExecutionException(ExtensionContext context, Throwable throwable) throws Throwable {
        // You can handle the exception in custom ways here, but let's rethrow it for the normal flow.
        if (throwable instanceof java.util.concurrent.TimeoutException) {
            printAllThreadStackTraces();
        }
        throw throwable;
    }

    private void printAllThreadStackTraces() {
        Thread.getAllStackTraces().forEach((thread, stackTrace) -> {
            System.out.println("Thread: " + thread.getName());
            for (StackTraceElement element : stackTrace) {
                System.out.println("\t" + element);
            }
        });
    }
}
