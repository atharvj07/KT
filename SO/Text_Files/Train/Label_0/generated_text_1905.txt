def yield_first(gen):
    first_yielded = False
    for item in gen:
        if not first_yielded:
            yield item
            first_yielded = True
        else:
            yield item

# Example usage
def gen():
    yield 1
    yield 2
    yield 3

generator = gen()
first_item = next(yield_first(generator))
print("First item:", first_item)

# Loop through the rest of the generator
for item in generator:
    print("Next item:", item)
