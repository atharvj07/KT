import polars as pl

# Sample data
data = {'column': [1, 2, 3, 4, 97, 98, 99]}
df = pl.DataFrame(data)

# Define the recoding rules using when().then().otherwise()
df = df.with_columns(
    pl.when(pl.col('column').is_in([1, 2]))
        .then(0)
        .when(pl.col('column').is_in([3, 4]))
        .then(1)
        .otherwise(2)
        .alias('recode_column')
)

print(df)
