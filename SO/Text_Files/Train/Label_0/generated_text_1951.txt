import polars as pl

df = pl.DataFrame(
    {
        "integer": [1, 2, 3],
        "date": [
            "2010-01-31T23:00:00+00:00",
            "2010-02-01T00:00:00+00:00",
            "2010-02-01T01:00:00+00:00",
        ],
    }
)
df = df.with_columns(
    [
        pl.col("date")
        .str.strptime(pl.Datetime, fmt="%Y-%m-%dT%H:%M:%S%z")
        .dt.with_time_zone("Europe/Amsterdam")
        .cast(pl.Date)
        .alias("valueDay")
    ]
)
print(df)
