import java.util.*;

class Solution {
    public List<String> findRepeatedDnaSequences(String s) {
        Set<String> seen = new HashSet<>();
        Set<String> repeated = new HashSet<>();
        int i = 0;
        int j = 0;
        int k = 10;
        while (j < s.length()) {
            if (j - i + 1 == k) {
                String substr = s.substring(i, j + 1);
                if (seen.contains(substr)) {
                    repeated.add(substr);
                } else {
                    seen.add(substr);
                }
                i++;
            }
            j++;
        }
        return new ArrayList<>(repeated);
    }
}
