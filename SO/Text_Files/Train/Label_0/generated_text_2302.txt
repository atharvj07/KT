from typing import TypeVar, Type, Optional

class A:
    pass

class B(A):
    pass

Instance = TypeVar("Instance", bound=A)

def get_instance(cls: Optional[Type[Instance]] = None) -> Instance:
    if cls is None:
        cls = A  # Set default value if cls is not provided
    return cls()
