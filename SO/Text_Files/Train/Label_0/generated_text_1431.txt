path_tbl.addMouseListener(new MouseAdapter() {
    @Override
    public void mouseReleased(MouseEvent e) {
        if (SwingUtilities.isRightMouseButton(e)) {
            Point point = e.getPoint();
            int selectedRow = path_tbl.rowAtPoint(point);
            
            if (selectedRow != -1) { // Ensure the click is within table bounds
                path_tbl.setRowSelectionInterval(selectedRow, selectedRow);
                
                // Show the popup menu at the clicked location
                JPopupMenu popupMenu = new JPopupMenu();
                JMenuItem menuItem = new JMenuItem("Your Option");
                popupMenu.add(menuItem);
                popupMenu.show(path_tbl, e.getX(), e.getY());
            }
        }
    }
});
