JTextArea textArea = new JTextArea();
   textArea.setText("The clock has been wound too tight.");
   
   textArea.addMouseListener(new MouseAdapter() {
       @Override
       public void mousePressed(MouseEvent e) {
           if (SwingUtilities.isLeftMouseButton(e)) {
               int pos = textArea.viewToModel(e.getPoint());
               int wordStart = getWordStart(pos);
               int wordEnd = getWordEnd(pos);
               
               if (e.getClickCount() == 2) {
                   // Double-clicked, select the word
                   textArea.setSelectionStart(wordStart);
                   textArea.setSelectionEnd(wordEnd);
               }
           }
       }
       
       @Override
       public void mouseReleased(MouseEvent e) {
           if (SwingUtilities.isLeftMouseButton(e) && e.isShiftDown()) {
               // Extend the selection by entire word when shift-clicking
               int pos = textArea.viewToModel(e.getPoint());
               int wordStart = getWordStart(pos);
               int wordEnd = getWordEnd(pos);
               textArea.setSelectionEnd(wordEnd);
           }
       }
   });

   // Helper methods to get word boundaries
   private int getWordStart(int pos) {
       try {
           return textArea.getUI().modelToView(textArea, pos).getStartOffset();
       } catch (BadLocationException ex) {
           ex.printStackTrace();
       }
       return pos;
   }

   private int getWordEnd(int pos) {
       try {
           return textArea.getUI().modelToView(textArea, pos).getEndOffset();
       } catch (BadLocationException ex) {
           ex.printStackTrace();
       }
       return pos;
   }
   
   textArea.addMouseMotionListener(new MouseAdapter() {
       @Override
       public void mouseDragged(MouseEvent e) {
           int pos = textArea.viewToModel(e.getPoint());
           int wordStart = getWordStart(pos);
           int wordEnd = getWordEnd(pos);
           textArea.setSelectionStart(wordStart);
           textArea.setSelectionEnd(wordEnd);
       }
   });
   
   textArea.addMouseListener(new MouseAdapter() {
       @Override
       public void mousePressed(MouseEvent e) {

               textArea.setSelectionEnd(paragraphEnd);
           }
       }
   });

   // Helper methods to get paragraph boundaries
   private int getParagraphStart(int pos) {
       return textArea.getDocument().getDefaultRootElement().getElementIndex(pos) * textArea.getDocument().getLength();
   }

   private int getParagraphEnd(int pos) {
       return textArea.getDocument().getDefaultRootElement().getElementIndex(pos) * textArea.getDocument().getLength() + textArea.getDocument().getLength();
   }
