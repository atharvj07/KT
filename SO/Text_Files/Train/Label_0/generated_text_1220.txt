import java.io.IOException;
import java.nio.file.*;

public class Test {
    public static void main(String[] args) {
        // Get the applications directory
        Path applicationsDir = Paths.get(System.getProperty("user.dir"), "applications");

        // Check if the applications directory exists
        if (Files.exists(applicationsDir) && Files.isDirectory(applicationsDir)) {
            try {
                // Walk through each subdirectory in applications/
                Files.list(applicationsDir)
                    .filter(Files::isDirectory) // Only consider directories
                    .forEach(subProject -> {
                        Path targetDir = subProject.resolve("target");
                        Path conduitDir = targetDir.resolve("conduit");

                        if (Files.exists(conduitDir) && Files.isDirectory(conduitDir)) {
                            Path baseDir = targetDir.resolve("base");

                            // Create 'base' folder if it does not exist
                            if (!Files.exists(baseDir)) {
                                try {
                                    Files.createDirectory(baseDir);
                                    System.out.println("Created: " + baseDir);
                                } catch (IOException e) {
                                    System.err.println("Error creating " + baseDir + ": " + e.getMessage());
                                }
                            } else {
                                System.out.println("Already exists: " + baseDir);
                            }
                        }
                    });

            } catch (IOException e) {
                System.err.println("Error reading applications directory: " + e.getMessage());
            }
        } else {
            System.err.println("Applications directory does not exist: " + applicationsDir);
        }
    }
}
