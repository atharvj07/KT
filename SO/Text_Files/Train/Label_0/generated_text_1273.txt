import java.io.*;
import java.nio.file.*;
import java.util.*;

public class GUIDDifferenceFinder {
    public static void main(String[] args) throws IOException {
        String pathFile1 = "input1.txt";
        String pathFile2 = "input2.txt";

        // Use HashSet for fast lookups
        Set<String> guidFile1 = readFileGuid(pathFile1);
        Set<String> guidFile2 = readFileGuid(pathFile2);

        // Compute differences
        Set<String> leftFromFile1 = new HashSet<>(guidFile1);
        leftFromFile1.removeAll(guidFile2); // GUIDs unique to file1

        Set<String> leftFromFile2 = new HashSet<>(guidFile2);
        leftFromFile2.removeAll(guidFile1); // GUIDs unique to file2

        // Output results
        System.out.println("Leftover from file1: " + leftFromFile1.size());
        System.out.println("Leftover from file2: " + leftFromFile2.size());

        // Write output to files
        writeFileGuid("output1.txt", leftFromFile1);
        writeFileGuid("output2.txt", leftFromFile2);
    }

    private static Set<String> readFileGuid(String filePath) throws IOException {
        return new HashSet<>(Files.readAllLines(Paths.get(filePath))); // Read lines into HashSet
    }

    private static void writeFileGuid(String filePath, Set<String> guids) throws IOException {
        Files.write(Paths.get(filePath), guids);
    }
}

public static void main(String[] args) throws IOException {
    String pathFile1 = "input1.txt";
    String pathFile2 = "input2.txt";
    // Use HashSet for fast lookups
    Set<String> guidFile1 = readFileGuid(pathFile1);
    Set<String> guidFile2 = readFileGuid(pathFile2);
    // Compute differences
    Set<String> leftFromFile1 = new HashSet<>(guidFile1);
    leftFromFile1.removeAll(guidFile2); // GUIDs unique to file1
    Set<String> leftFromFile2 = new HashSet<>(guidFile2);
    leftFromFile2.removeAll(guidFile1); // GUIDs unique to file2
    // Output results
    System.out.println("Leftover from file1: " + leftFromFile1.size());
    System.out.println("Leftover from file2: " + leftFromFile2.size());
    // Write output to files
    writeFileGuid("output1.txt", leftFromFile1);
    writeFileGuid("output2.txt", leftFromFile2);
}
private static Set<String> readFileGuid(String filePath) throws IOException {
    return new HashSet<>(Files.readAllLines(Paths.get(filePath))); // Read lines into HashSet
}
private static void writeFileGuid(String filePath, Set<String> guids) throws IOException {
    Files.write(Paths.get(filePath), guids);
}
