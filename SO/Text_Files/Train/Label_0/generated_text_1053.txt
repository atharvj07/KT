@Component
public class GameRunner {
    private final GamingSoftware game;

    @Autowired
    public GameRunner(@Qualifier("superMario") GamingSoftware game) { // Specify the bean name
        this.game = game;
    }

    public void run() {
        game.up();
        game.down();
        game.left();
        game.right();
    }
}

@Component("marioGame")
public class SuperMario implements GamingSoftware { ... }

@Component("contraGame")
public class SuperContra implements GamingSoftware { ... }

@Autowired
public GameRunner(@Qualifier("marioGame") GamingSoftware game) { 
    this.game = game;
}

@Component
@Primary // Makes this the default implementation
public class SuperMario implements GamingSoftware { ... }

public static void main(String[] args) {
    ConfigurableApplicationContext applicationContext = SpringApplication.run(GamingDevice.class, args);
    
    GamingSoftware game = applicationContext.getBean(SuperMario.class); // Or SuperContra.class
    GameRunner gameRunner = new GameRunner(game); // Manually pass the instance
    gameRunner.run();
}
