import javax.imageio.ImageIO;
import java.awt.image.BufferedImage;
import java.io.*;
import java.nio.ByteBuffer;
import java.nio.channels.FileChannel;

public class MemoryDumpToPng {
    private static final int CHUNK_SIZE = 134217728; // 128MB
    private static final int WIDTH = 1024; // Define a suitable width
    private static final String INPUT_FILE = "memory_dump.bin";
    private static final String OUTPUT_FILE = "output.png";

    public static void main(String[] args) {
        try {
            convertMemoryDumpToPng(INPUT_FILE, OUTPUT_FILE, WIDTH);
            System.out.println("PNG file successfully created: " + OUTPUT_FILE);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    public static void convertMemoryDumpToPng(String inputFile, String outputFile, int width) throws IOException {
        File file = new File(inputFile);
        long totalBytes = file.length();
        int height = (int) Math.ceil((double) totalBytes / (3 * width)); // Calculate height based on 3 bytes per pixel

        BufferedImage image = new BufferedImage(width, height, BufferedImage.TYPE_INT_RGB);

        try (FileInputStream fis = new FileInputStream(file);
             FileChannel channel = fis.getChannel()) {

            ByteBuffer buffer = ByteBuffer.allocateDirect(CHUNK_SIZE);
            int x = 0, y = 0;

            while (channel.read(buffer) > 0) {
                buffer.flip(); // Switch to read mode
                while (buffer.remaining() >= 3) { // Ensure 3 bytes are available
                    int r = buffer.get() & 0xFF;
                    int g = buffer.get() & 0xFF;
                    int b = buffer.get() & 0xFF;
                    int rgb = (r << 16) | (g << 8) | b;

                    image.setRGB(x, y, rgb);

                    if (++x >= width) {
                        x = 0;
                        y++;
                    }
                }
                buffer.clear(); // Prepare buffer for the next read
            }
        }

        ImageIO.write(image, "PNG", new File(outputFile));
    }
}
