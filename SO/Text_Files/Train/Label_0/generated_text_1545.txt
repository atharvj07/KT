import itertools

letter_freq = [
    {'a': 10, 'b': 7},
    {'d': 15, 'g': 8},
    {'a': 12, 'q': 2}
]

perms = {}

for combo in itertools.product(*letter_freq):
    combined_dict = {}
    total_freq = 0
    for d in combo:
        for letter, freq in d.items():
            combined_dict[letter] = combined_dict.get(letter, 0) + freq
            total_freq += freq
    key = ''.join(sorted(combined_dict.keys()))
    perms[key] = total_freq

print(perms)
