Imgproc.threshold(grayMat, grayMat, 180, 255, Imgproc.THRESH_BINARY_INV);

Mat rectKernel = Imgproc.getStructuringElement(Imgproc.MORPH_RECT, new Size(25, 25));
Imgproc.morphologyEx(grayMat, dilatedMat, Imgproc.MORPH_CLOSE, rectKernel);

contours.sort(Comparator.comparingDouble(Imgproc::contourArea).reversed());

for (MatOfPoint contour : contours) {
    if (Imgproc.contourArea(contour) > 5000) { // Filter small noise
        RotatedRect minRect = Imgproc.minAreaRect(new MatOfPoint2f(contour.toArray()));
        Point[] box = new Point[4];
        minRect.points(box);

        for (int j = 0; j < 4; j++) {
            Imgproc.line(srcMat, box[j], box[(j + 1) % 4], new Scalar(0, 255, 0), 2);
        }
        
        // Crop the rotated ROI
        Mat cropped = cropRotatedRectangle(srcMat, minRect);
        Imgcodecs.imwrite("receipt_" + count + ".jpg", cropped);
    }
}

public static Mat cropRotatedRectangle(Mat src, RotatedRect rect) {
    Mat M, rotated, cropped;

    // Get the rotation matrix
    double angle = rect.angle;
    Size rect_size = rect.size;
    if (rect.angle < -45.) {
        angle += 90.0;
        rect_size = new Size(rect.size.height, rect.size.width);
    }

    M = Imgproc.getRotationMatrix2D(rect.center, angle, 1.0);
    Imgproc.warpAffine(src, rotated, M, src.size(), Imgproc.INTER_CUBIC);

    // Crop
    Rect roi = new Rect((int) rect.center.x - (int) (rect_size.width / 2),
                        (int) rect.center.y - (int) (rect_size.height / 2),
                        (int) rect_size.width, (int) rect_size.height);

    cropped = new Mat(rotated, roi);
    return cropped;
}
