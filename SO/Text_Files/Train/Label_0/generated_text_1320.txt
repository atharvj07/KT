private Map<String, Integer> buttonClickCounters = new HashMap<>();

@SuppressLint("SetTextI18n")
private void latestQuote() {
    // Initialize the click counter for this button if it's the first time
    buttonClickCounters.putIfAbsent("latestQuote", 0);

    // Increment the click counter for this button
    int countClicks = buttonClickCounters.get("latestQuote") + 1;
    buttonClickCounters.put("latestQuote", countClicks); // Update the counter

    if (countClicks != triggerClicks) {
        position = quotes_list.size() - 1;
        quotesTxt.setText(quotes_list.get(position));
        countTxt.setText(position + "/" + quotes_list.size());
        Log.d(TAG, quotes_list.toString());
    }

    if (mInterstitialAd != null && countClicks >= triggerClicks) {
        mInterstitialAd.show(HomeActivity.this);
        mInterstitialAd.setFullScreenContentCallback(new FullScreenContentCallback() {
            @Override
            public void onAdDismissedFullScreenContent() {
                super.onAdDismissedFullScreenContent();
                mInterstitialAd = null;
                intertitalAd();
                position = quotes_list.size() - 1;
                quotesTxt.setText(quotes_list.get(position));
                countTxt.setText(position + "/" + quotes_list.size());
                Log.d(TAG, quotes_list.toString());
                buttonClickCounters.put("latestQuote", 0); // Reset the counter after the ad
            }
        });
    }
}

private void back() {
    buttonClickCounters.putIfAbsent("back", 0);
    int countClicksBack = buttonClickCounters.get("back") + 1;
    buttonClickCounters.put("back", countClicksBack);

    if (position > 0) {
        position = (position - 1) % quotes_list.size();
        quotesTxt.setText(quotes_list.get(position));
        countTxt.setText(position + "/" + quotes_list.size());
        Log.d(TAG, quotes_list.toString());
    }

    if (mInterstitialAd != null && countClicksBack >= triggerClicksNav) {
        mInterstitialAd.show(HomeActivity.this);
        mInterstitialAd.setFullScreenContentCallback(new FullScreenContentCallback() {
            @Override
            public void onAdDismissedFullScreenContent() {
                super.onAdDismissedFullScreenContent();
                mInterstitialAd = null;
                intertitalAd();
                buttonClickCounters.put("back", 0); // Reset counter
            }
        });
    }
}

private void handleButtonClick(String buttonKey, int triggerClicks) {
    buttonClickCounters.putIfAbsent(buttonKey, 0);
    int countClicks = buttonClickCounters.get(buttonKey) + 1;
    buttonClickCounters.put(buttonKey, countClicks);

    if (countClicks >= triggerClicks) {
        if (mInterstitialAd != null) {
            mInterstitialAd.show(HomeActivity.this);
            mInterstitialAd.setFullScreenContentCallback(new FullScreenContentCallback() {
                @Override
                public void onAdDismissedFullScreenContent() {
                    super.onAdDismissedFullScreenContent();
                    mInterstitialAd = null;
                    intertitalAd();
                    buttonClickCounters.put(buttonKey, 0); // Reset the counter after ad
                }
            });
        }
    }
}
