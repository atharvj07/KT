FirebaseFirestore db = FirebaseFirestore.getInstance();
db.collection("User")
    .whereEqualTo("username", user)
    .get()
    .addOnCompleteListener(new OnCompleteListener<QuerySnapshot>() {
        @Override
        public void onComplete(@NonNull Task<QuerySnapshot> task) {
            if (task.isSuccessful()) {
                // Check if the query returned any documents
                QuerySnapshot querySnapshot = task.getResult();
                if (querySnapshot != null && !querySnapshot.isEmpty()) {
                    // If documents are found, it means the username already exists
                    Toast.makeText(Signup.this, "Username already exists, please try again", Toast.LENGTH_SHORT).show();
                } else {
                    // If no documents are found, the username is available, so proceed with adding the new user
                    User newUser = new User(user, pass, mail);
                    CollectionReference dbUser = db.collection("User");
                    dbUser.add(newUser);
                    Intent intent = new Intent(Signup.this, StartScreen.class);
                    startActivity(intent);
                }
            } else {
                // Handle query failure (e.g., network issues)
                Toast.makeText(Signup.this, "Error checking username", Toast.LENGTH_SHORT).show();
            }
        }
    });
