import de.jensd.fx.glyphs.materialdesignicons.MaterialDesignIcon;
import de.jensd.fx.glyphs.materialdesignicons.MaterialDesignIconView;
import javafx.application.Application;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.RadioButton;
import javafx.scene.layout.HBox;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class Main extends Application {

    public static void main(String[] args) {
        launch(args);
    }

    @Override
    public void start(Stage primaryStage) {
        primaryStage.setTitle("Hello World!");

        // Create buttons with icons and text
        RadioButton memberButton = createButton("Members", MaterialDesignIcon.WORKER);
        RadioButton companyButton = createButton("Companies", MaterialDesignIcon.CITY);
        RadioButton settingsButton = createButton("Settings", MaterialDesignIcon.WRENCH);

        // Style and set buttons
        stylizeAndSetButtons(memberButton, companyButton, settingsButton);

        // VBox layout
        VBox root = new VBox(10);
        root.getChildren().addAll(memberButton, companyButton, settingsButton);

        primaryStage.setScene(new Scene(root, 300, 250));
        primaryStage.show();
    }

    private RadioButton createButton(String text, MaterialDesignIcon icon) {
        RadioButton button = new RadioButton();
        button.setText(text);

        // Create an HBox to hold the icon and text
        HBox hbox = new HBox(10);
        hbox.setAlignment(Pos.CENTER_LEFT);

        // Create icon and add it to the left of the HBox
        MaterialDesignIconView iconView = new MaterialDesignIconView(icon);
        hbox.getChildren().add(iconView);

        // Set the HBox as the graphic (left-align the icon and right-align the text)
        button.setGraphic(hbox);

        // Align the text to the right in the HBox
        button.setStyle("-fx-text-fill: black;");
        
        return button;
    }

    private static void stylizeAndSetButtons(RadioButton... buttons) {
        for (RadioButton button : buttons) {
            button.setPrefWidth(250);
            button.setMinHeight(50);
            button.getStyleClass().remove("radio-button");
            button.getStyleClass().add("toggle-button");
            button.getStyleClass().add("default");
        }
    }
}

public static void main(String[] args) {
    launch(args);
}
@Override
public void start(Stage primaryStage) {
    primaryStage.setTitle("Hello World!");
    // Create buttons with icons and text
    RadioButton memberButton = createButton("Members", MaterialDesignIcon.WORKER);

    root.getChildren().addAll(memberButton, companyButton, settingsButton);
    primaryStage.setScene(new Scene(root, 300, 250));
    primaryStage.show();
}

    button.setGraphic(hbox);
    // Align the text to the right in the HBox

    return button;

    for (RadioButton button : buttons) {
        button.setPrefWidth(250);

}
