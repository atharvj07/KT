static void homework() throws IOException {
    int count = 0;
    Transaction tranxLst = new Transaction();
    Block genesis = new Block("0");
    System.out.println(genesis);

    BufferedReader bf = new BufferedReader(new FileReader("dummytranx.txt"));
    String line = bf.readLine();
    while (line != null) {
        tranxLst.add(line);
        line = bf.readLine();
        count++;

        // When we have 10 transactions, create a new block and reset the transaction list
        if (count % 10 == 0) {
            Block newBlock = new Block(genesis.getHeader().getPrevHash());
            newBlock.setTranx(tranxLst);
            System.out.println(newBlock);
            tranxLst.getTranxLst().clear();  // Reset transaction list after creating a block
        }

        // Check if we are at the last transaction set
        if (line == null && tranxLst.getTranxLst().size() > 0) {
            Block lastBlock = new Block(genesis.getHeader().getPrevHash());
            lastBlock.setTranx(tranxLst);
            System.out.println(lastBlock);
        }
    }
    bf.close();
}
