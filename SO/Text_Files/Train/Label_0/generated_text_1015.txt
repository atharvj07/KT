def search_in_csv(file_path, column):
    try:
        with open(file_path, "r", encoding="utf-8") as file:
            while True:
                prefix = input("Enter prefix to search (or 'end' to stop): ").strip()
                if prefix.lower() == "end":
                    break

                # Move file pointer back to the beginning for a new search
                file.seek(0)

                # Iterate through file line by line (no full loading into memory)
                for line in file:
                    columns = line.strip().split(",")  # Assuming CSV is comma-separated
                    if len(columns) >= column and columns[column - 1].strip().startswith(prefix):
                        print(line.strip())

    except FileNotFoundError:
        print("File not found.")
    except Exception as e:
        print(f"Error: {e}")

# Example usage
file_path = "data.csv"
column = int(input("Enter the column number to search: "))
search_in_csv(file_path, column)
