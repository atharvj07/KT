import javax.swing.*;
import javax.swing.event.PopupMenuEvent;
import javax.swing.event.PopupMenuListener;
import java.awt.*;
import java.awt.event.MouseAdapter;
import java.awt.event.MouseEvent;

public class FileChooserPopupCustomizer {

    public static void main(String[] args) {
        SwingUtilities.invokeLater(FileChooserPopupCustomizer::showFileChooser);
    }

    private static void showFileChooser() {
        JFileChooser fileChooser = new JFileChooser();

        // Attach listener to modify the popup menu when it appears
        attachPopupCustomizer(fileChooser);

        fileChooser.showOpenDialog(null);
    }

    private static void attachPopupCustomizer(JFileChooser fileChooser) {
        // Retrieve file chooser's components and add a listener to detect right-click
        for (Component component : fileChooser.getComponents()) {
            component.addMouseListener(new MouseAdapter() {
                @Override
                public void mousePressed(MouseEvent e) {
                    if (SwingUtilities.isRightMouseButton(e)) {
                        SwingUtilities.invokeLater(() -> customizePopupMenu());
                    }
                }
            });
        }
    }

    private static void customizePopupMenu() {
        for (Window window : Window.getWindows()) {
            for (Component component : window.getComponents()) {
                if (component instanceof JPopupMenu popupMenu) {
                    System.out.println("Popup menu found! Customizing...");
                    customizePopup(popupMenu);
                }
            }
        }
    }

    private static void customizePopup(JPopupMenu popupMenu) {
        popupMenu.setBackground(Color.DARK_GRAY);
        popupMenu.setBorder(BorderFactory.createLineBorder(Color.RED, 2));

        for (Component item : popupMenu.getComponents()) {
            if (item instanceof JMenuItem menuItem) {
                menuItem.setForeground(Color.WHITE);
                menuItem.setBackground(Color.DARK_GRAY);
            }
        }
    }
}

public static void main(String[] args) {
    SwingUtilities.invokeLater(FileChooserPopupCustomizer::showFileChooser);
}
private static void showFileChooser() {
    JFileChooser fileChooser = new JFileChooser();
    // Attach listener to modify the popup menu when it appears
    attachPopupCustomizer(fileChooser);
    fileChooser.showOpenDialog(null);
}

}
private static void customizePopupMenu() {
    for (Window window : Window.getWindows()) {
        for (Component component : window.getComponents()) {

                System.out.println("Popup menu found! Customizing...");

            }
        }
    }
}
private static void customizePopup(JPopupMenu popupMenu) {
    popupMenu.setBackground(Color.DARK_GRAY);
    popupMenu.setBorder(BorderFactory.createLineBorder(Color.RED, 2));
    for (Component item : popupMenu.getComponents()) {

}
