import time
import sys

place = 10

running = True

def Write():
    j = 1
    for i in range(place - 1):
        print("-", end="")
        j += 1
    print("a", end="")
    for k in range(10 - j):
        print("-", end="")
    sys.stdout.flush()  # Flush the output to ensure it is displayed immediately

while running:
    Write()
    time.sleep(1)  # Reduced sleep time for faster demonstration
    if place > 1:
        place -= 1
        print()  # Move to the next line after each loop
