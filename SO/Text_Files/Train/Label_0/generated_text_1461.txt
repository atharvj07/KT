public static void WriteToFile() {
    try {
        FileWriter myWriter = new FileWriter(myFile, true); // Append mode
        System.out.println("This is the contents of the file:");
        
        // Iterate through the bookTitles list in chunks of 4 (title, author, ISBN, genre)
        for (int i = 0; i < bookTitles.size(); i += 4) {
            if (i + 3 < bookTitles.size()) { // Ensure there are at least 4 elements to write
                myWriter.write(bookTitles.get(i) + " ");
                myWriter.write(bookTitles.get(i + 1) + " ");
                myWriter.write(bookTitles.get(i + 2) + " ");
                myWriter.write(bookTitles.get(i + 3) + "\n\n");
            }
        }
        
        myWriter.close();
        System.out.println("Successfully wrote to the file.");
    } catch (IOException e) {
        System.out.println("An error occurred.");
        e.printStackTrace();
    }
}
