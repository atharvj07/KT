@Override
public void customize(JFreeChart chart, JRChart jasperChart) {
    PiePlot plot = (PiePlot) chart.getPlot();

    // The parameter seriesColor_P is passed from the main report to the subreport.
    // It contains an ArrayNode with the status-color mappings.
    ArrayNode seriesColor = (ArrayNode) getParameterValue("seriesColor_P");
    
    // Create a map to store status-color pairs
    HashMap<String, String> customMap = new HashMap<>();
    
    for (int z = 0; z < seriesColor.size(); z++) {
        String status = seriesColor.get(z).get("Status").textValue();
        String color = seriesColor.get(z).get("Color").textValue();
        customMap.put(status, color);
    }

    // Get the dataset from the PiePlot
    PieDataset dataset = plot.getDataset();
    
    // Check if the dataset is not null and modify the colors
    if (dataset != null) {
        // Loop through all slices in the dataset
        for (int i = 0; i < dataset.getItemCount(0); i++) {
            String status = dataset.getKey(0, i).toString();
            
            // Check if a color mapping exists for the current status
            if (customMap.containsKey(status)) {
                String colorHex = customMap.get(status);
                
                // Convert the color to a Color object and set it for the slice
                plot.setSectionPaint(status, java.awt.Color.decode(colorHex));
            }
        }
    }
}
