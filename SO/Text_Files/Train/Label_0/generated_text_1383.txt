import org.apache.poi.ss.usermodel.*;
import org.apache.poi.xssf.usermodel.*;
import org.apache.poi.xssf.usermodel.extensions.XSSFPivotTable;
import org.apache.xmlbeans.XmlCursor;

import java.io.FileOutputStream;
import java.io.IOException;

public class PivotTableExample {

    public static void main(String[] args) throws IOException {
        // Create a new workbook and sheet
        XSSFWorkbook workbook = new XSSFWorkbook();
        XSSFSheet sheet = workbook.createSheet("Data");

        // Create some data for the pivot table
        Object[][] data = {
            {"Category", "Product", "Sales"},
            {"Electronics", "Laptop", 200},
            {"Electronics", "Phone", 300},
            {"Furniture", "Chair", 150},
            {"Furniture", "Table", 200}
        };

        // Fill the sheet with data
        int rowNum = 0;
        for (Object[] rowData : data) {
            Row row = sheet.createRow(rowNum++);
            int colNum = 0;
            for (Object cellData : rowData) {
                Cell cell = row.createCell(colNum++);
                if (cellData instanceof String) {
                    cell.setCellValue((String) cellData);
                } else if (cellData instanceof Integer) {
                    cell.setCellValue((Integer) cellData);
                }
            }
        }

        // Create a Pivot Table in the workbook
        XSSFPivotTable pivotTable = ((XSSFSheet) sheet).createPivotTable(
            new AreaReference("A1:C5", workbook.getSpreadsheetVersion()),  // Data range
            new CellReference("E1"),  // Pivot Table location
            sheet
        );

        // Enable Classic Pivot Table Layout
        // Note: You have to manipulate the underlying XML to change the layout
        XmlCursor cursor = pivotTable.getCTPivotTableDefinition().newCursor();
        cursor.selectPath("./*");

        while (cursor.toNextSelection()) {
            if (cursor.getName().getLocalPart().equals("pivotTableStyleInfo")) {
                cursor.getObject().set("showColStripes", "0");
                cursor.getObject().set("showRowStripes", "0");
                cursor.getObject().set("showLastColumn", "0");
                cursor.getObject().set("showRowHeaders", "1");
                cursor.getObject().set("showColumnHeaders", "1");
                // You can add more style info here
            }
        }

        // Save the workbook to a file
        try (FileOutputStream fileOut = new FileOutputStream("pivotTableExample.xlsx")) {
            workbook.write(fileOut);
        }

        workbook.close();
    }
}
