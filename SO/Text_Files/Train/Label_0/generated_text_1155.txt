import javafx.animation.KeyFrame;
import javafx.animation.Timeline;
import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.layout.StackPane;
import javafx.scene.paint.Color;
import javafx.scene.transform.Scale;
import javafx.util.Duration;
import javafx.stage.Stage;

public class ScaleButtonExample extends Application {

    @Override
    public void start(Stage primaryStage) {
        Button button = new Button("Hover over me!");

        // Add hover event handler
        button.setOnMouseEntered(event -> {
            // Animate the scaling effect using Timeline
            Timeline timeline = new Timeline(
                new KeyFrame(Duration.seconds(0), new javafx.animation.KeyValue(button.scaleXProperty(), 1)),
                new KeyFrame(Duration.seconds(0.3), new javafx.animation.KeyValue(button.scaleXProperty(), 1.5)),
                new KeyFrame(Duration.seconds(0), new javafx.animation.KeyValue(button.scaleYProperty(), 1)),
                new KeyFrame(Duration.seconds(0.3), new javafx.animation.KeyValue(button.scaleYProperty(), 1.5))
            );
            timeline.play();
        });

        button.setOnMouseExited(event -> {
            // Animate back to the original scale when mouse leaves
            Timeline timeline = new Timeline(
                new KeyFrame(Duration.seconds(0), new javafx.animation.KeyValue(button.scaleXProperty(), 1.5)),
                new KeyFrame(Duration.seconds(0.3), new javafx.animation.KeyValue(button.scaleXProperty(), 1)),
                new KeyFrame(Duration.seconds(0), new javafx.animation.KeyValue(button.scaleYProperty(), 1.5)),
                new KeyFrame(Duration.seconds(0.3), new javafx.animation.KeyValue(button.scaleYProperty(), 1))
            );
            timeline.play();
        });

        StackPane root = new StackPane();
        root.getChildren().add(button);
        Scene scene = new Scene(root, 300, 250);

        primaryStage.setTitle("Scale Button with Transition");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}

.button {
    -fx-background-color: green;
    -fx-padding: 10px;
    -fx-font-size: 14px;
    -fx-text-fill: white;
    -fx-background-radius: 5px;
    -fx-cursor: hand;
}

.button:hover {
    -fx-transform: scale(1.5);
    -fx-transition: all 0.3s ease-in-out;
}
