package biojava.biojava_project;

import java.net.URL;
import java.util.HashMap;
import java.util.Map;
import org.biojava.nbio.core.sequence.ProteinSequence;
import org.biojava.nbio.core.sequence.io.FastaReaderHelper;

public class BioSeq {
    
    // Defining categories
    private static final Map<Character, String> aminoAcidCategories = new HashMap<>();
    
    static {
        // Polar amino acids
        aminoAcidCategories.put('S', "polar"); // Serine
        aminoAcidCategories.put('T', "polar"); // Threonine
        aminoAcidCategories.put('N', "polar"); // Asparagine
        aminoAcidCategories.put('Q', "polar"); // Glutamine
        aminoAcidCategories.put('C', "polar"); // Cysteine
        aminoAcidCategories.put('Y', "polar"); // Tyrosine
        
        // Non-polar amino acids
        aminoAcidCategories.put('A', "non-polar"); // Alanine
        aminoAcidCategories.put('V', "non-polar"); // Valine
        aminoAcidCategories.put('L', "non-polar"); // Leucine
        aminoAcidCategories.put('I', "non-polar"); // Isoleucine
        aminoAcidCategories.put('P', "non-polar"); // Proline
        aminoAcidCategories.put('M', "non-polar"); // Methionine
        aminoAcidCategories.put('F', "aromatic"); // Phenylalanine
        aminoAcidCategories.put('W', "aromatic"); // Tryptophan
        
        // Aromatic amino acids
        aminoAcidCategories.put('F', "aromatic"); // Phenylalanine
        aminoAcidCategories.put('W', "aromatic"); // Tryptophan
        aminoAcidCategories.put('Y', "aromatic"); // Tyrosine
        
        // Heterocyclic amino acids
        aminoAcidCategories.put('H', "heterocyclic"); // Histidine
    }

    // Inserting the sequence from UNIPROT
    public static ProteinSequence getSequenceForId(String uniProtId) throws Exception {
        URL uniprotFasta = new URL(String.format("https://rest.uniprot.org/uniprotkb/%s.fasta", uniProtId));
        ProteinSequence seq = FastaReaderHelper.readFastaProteinSequence(uniprotFasta.openStream()).get(uniProtId);
        System.out.printf("id: %s\n", uniProtId);
        return seq;
    }

    // Method to calculate categories
    public static void calculateAminoAcidCategories(ProteinSequence seq) {
        int polarCount = 0;
        int nonPolarCount = 0;
        int aromaticCount = 0;
        int heterocyclicCount = 0;
        
        // Loop through each amino acid in the sequence
        for (int i = 0; i < seq.length(); i++) {
            char aa = seq.getSequenceAsString().charAt(i);
            
            // Classify amino acids
            String category = aminoAcidCategories.getOrDefault(aa, "other");

            // Count based on category
            if (category.equals("polar")) {
                polarCount++;
            } else if (category.equals("non-polar")) {
                nonPolarCount++;
            } else if (category.equals("aromatic")) {
                aromaticCount++;
            } else if (category.equals("heterocyclic")) {
                heterocyclicCount++;
            }
        }

        // Total number of amino acids
        int totalAminoAcids = seq.length();

        // Calculating percentages
        System.out.printf("Total Amino Acids: %d\n", totalAminoAcids);
        System.out.printf("Polar Amino Acids: %d (%.2f%%)\n", polarCount, (polarCount * 100.0) / totalAminoAcids);
        System.out.printf("Non-polar Amino Acids: %d (%.2f%%)\n", nonPolarCount, (nonPolarCount * 100.0) / totalAminoAcids);
        System.out.printf("Aromatic Amino Acids: %d (%.2f%%)\n", aromaticCount, (aromaticCount * 100.0) / totalAminoAcids);
        System.out.printf("Heterocyclic Amino Acids: %d (%.2f%%)\n", heterocyclicCount, (heterocyclicCount * 100.0) / totalAminoAcids);
    }

    public static void main(String[] args) {
        try {
            ProteinSequence seq = getSequenceForId("P31574");
            calculateAminoAcidCategories(seq);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
