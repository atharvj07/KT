// Node class representing each element in the tree
class Node {
    int value;
    Node left, right, parent;

    public Node(int value) {
        this.value = value;
        this.left = null;
        this.right = null;
        this.parent = null;
    }
}

// BST class
public class BalancedBST {

    // Helper method to create a balanced BST from a sorted array
    public Node sortedArrayToBST(int[] sortedArray) {
        if (sortedArray == null || sortedArray.length == 0) {
            return null;
        }

        // Stack to manage the nodes and subarrays
        java.util.Stack<Node> stack = new java.util.Stack<>();
        Node root = null;
        int start = 0;
        int end = sortedArray.length - 1;

        // Create the root first
        root = createBST(sortedArray, start, end, stack);
        
        return root;
    }

    // Helper method to iteratively create the BST by managing stack
    private Node createBST(int[] sortedArray, int start, int end, java.util.Stack<Node> stack) {
        if (start > end) {
            return null;
        }

        // Stack will store the information of the tree nodes and their bounds
        java.util.Stack<int[]> boundsStack = new java.util.Stack<>();
        Node root = null;

        // Initialize stack with the first bounds
        boundsStack.push(new int[] {start, end});

        while (!boundsStack.isEmpty()) {
            int[] bounds = boundsStack.pop();
            start = bounds[0];
            end = bounds[1];

            // Find middle element
            int mid = (start + end) / 2;
            Node newNode = new Node(sortedArray[mid]);

            // If root is null, assign it as root
            if (root == null) {
                root = newNode;
            }

            // Stack can be used to manage and link nodes' left and right children
            if (!stack.isEmpty()) {
                Node parentNode = stack.peek();

                if (parentNode.left == null) {
                    parentNode.left = newNode;
                } else {
                    parentNode.right = newNode;
                }
                newNode.parent = parentNode;
            }

            // Push the left and right subarrays into the stack
            boundsStack.push(new int[] {start, mid - 1});
            boundsStack.push(new int[] {mid + 1, end});

            // Push the current node into the stack to handle its children later
            stack.push(newNode);
        }

        return root;
    }

    // Method to print the tree (for testing)
    public void inorderTraversal(Node root) {
        if (root != null) {
            inorderTraversal(root.left);
            System.out.print(root.value + " ");
            inorderTraversal(root.right);
        }
    }

    public static void main(String[] args) {
        BalancedBST bst = new BalancedBST();
        int[] sortedArray = {-10, -3, 0, 5, 9};
        
        Node root = bst.sortedArrayToBST(sortedArray);
        bst.inorderTraversal(root);  // Output should be a sorted array: -10 -3 0 5 9
    }
}
