@Repository
public interface MyGeoRepository extends JpaRepository<Mygeo, String> {

    @Modifying
    @Query("INSERT INTO Mygeo (id, geodata) VALUES (:id, ST_GeomFromText(:geom))")
    void saveGeometry(@Param("id") String id, @Param("geom") String geometryWkt);
}

@Service
public class GeoService {

    @Autowired
    private MyGeoRepository geoRepository;

    public void savePolygon(String id, String geometryWkt) {
        geoRepository.saveGeometry(id, geometryWkt);
    }
}

import javax.persistence.AttributeConverter;
import javax.persistence.Converter;
import org.locationtech.jts.geom.Geometry;
import org.locationtech.jts.io.WKTReader;

@Converter
public class GeometryConverter implements AttributeConverter<Geometry, String> {

    private static final WKTReader reader = new WKTReader();

    @Override
    public String convertToDatabaseColumn(Geometry geometry) {
        if (geometry == null) {
            return null;
        }
        return geometry.toText();  // Converts the geometry to WKT (Well-Known Text)
    }

    @Override
    public Geometry convertToEntityAttribute(String dbData) {
        if (dbData == null) {
            return null;
        }
        try {
            return reader.read(dbData);  // Converts WKT string back to Geometry
        } catch (Exception e) {
            throw new IllegalArgumentException("Failed to convert WKT to Geometry", e);
        }
    }
}

@Entity
public class Mygeo {

    @Id
    @Column(name = "id")
    private String id;

    @Convert(converter = GeometryConverter.class)
    @Column(name = "geodata", nullable = false)
    private Geometry geodata;

    // Getters and setters
}

@Service
public class GeoService {

    @Autowired
    private MyGeoRepository geoRepository;

    public void saveGeometry(String id, Geometry geometry) {
        Mygeo mygeo = new Mygeo();
        mygeo.setId(id);
        mygeo.setGeodata(geometry);
        geoRepository.save(mygeo);
    }
}
