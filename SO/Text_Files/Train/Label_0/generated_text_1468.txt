import software.amazon.awssdk.core.SdkBytes;
import software.amazon.awssdk.services.textract.TextractClient;
import software.amazon.awssdk.services.textract.model.*;

import java.io.File;
import java.io.FileInputStream;
import java.io.InputStream;
import java.util.Arrays;
import java.util.List;

public class TextractQueryExample {
    public static void analyzeDoc(TextractClient textractClient, String sourceDoc) {

        try {
            // Read document as bytes
            InputStream sourceStream = new FileInputStream(new File(sourceDoc));
            SdkBytes sourceBytes = SdkBytes.fromInputStream(sourceStream);

            // Create document object
            Document myDoc = Document.builder()
                    .bytes(sourceBytes)
                    .build();

            // Define Queries
            Query query = Query.builder()
                    .text("What is the customer's name?")  // Your question to Textract
                    .alias("customer_name")  // Optional: Assign an alias to reference the answer
                    .build();

            // Add queries to QueriesConfig
            QueriesConfig queriesConfig = QueriesConfig.builder()
                    .queries(Arrays.asList(query))
                    .build();

            // Define feature types
            List<FeatureType> featureTypes = Arrays.asList(FeatureType.QUERIES);

            // Create AnalyzeDocumentRequest with queries
            AnalyzeDocumentRequest analyzeDocumentRequest = AnalyzeDocumentRequest.builder()
                    .featureTypes(featureTypes)
                    .document(myDoc)
                    .queriesConfig(queriesConfig)  // Add queries here
                    .build();

            // Perform document analysis
            AnalyzeDocumentResponse analyzeDocument = textractClient.analyzeDocument(analyzeDocumentRequest);

            // Process response blocks
            for (Block block : analyzeDocument.blocks()) {
                System.out.println("Block Type: " + block.blockType());
                if (block.blockType() == BlockType.QUERY) {
                    System.out.println("Query: " + block.query().text());
                    System.out.println("Answer: " + block.text());
                }
            }

        } catch (TextractException | java.io.IOException e) {
            System.err.println("Error: " + e.getMessage());
            System.exit(1);
        }
    }
}
