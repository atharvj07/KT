import javax.swing.*;
import javax.swing.tree.DefaultMutableTreeNode;
import java.io.FileInputStream;
import java.io.IOException;
import java.util.*;

public class DynamicJTree {

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            JFrame frame = new JFrame("Dynamic JTree from Properties");
            frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
            frame.setSize(400, 400);

            JTree tree = createTreeFromProperties("tree.properties");
            JScrollPane scrollPane = new JScrollPane(tree);
            frame.add(scrollPane);
            frame.setVisible(true);
        });
    }

    public static JTree createTreeFromProperties(String filePath) {
        Properties props = new Properties();
        try (FileInputStream fis = new FileInputStream(filePath)) {
            props.load(fis);
        } catch (IOException e) {
            e.printStackTrace();
            return new JTree();
        }

        // Map to store nodes by their name
        Map<String, DefaultMutableTreeNode> nodeMap = new HashMap<>();

        // Root node
        String rootName = props.getProperty("root", "Root");
        DefaultMutableTreeNode rootNode = new DefaultMutableTreeNode(rootName);
        nodeMap.put(rootName, rootNode);

        for (String key : props.stringPropertyNames()) {
            if (key.equals("root")) continue; // Skip root definition

            String parentName = key; 
            String childName = props.getProperty(key);

            // Get or create parent node
            DefaultMutableTreeNode parentNode = nodeMap.computeIfAbsent(parentName, DefaultMutableTreeNode::new);

            // Create child node
            DefaultMutableTreeNode childNode = new DefaultMutableTreeNode(childName);
            parentNode.add(childNode);

            // Store child in map
            nodeMap.put(childName, childNode);
        }

        return new JTree(rootNode);
    }
}
