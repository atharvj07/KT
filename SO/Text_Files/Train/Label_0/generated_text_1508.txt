import com.fasterxml.jackson.databind.module.SimpleModule;
import com.fasterxml.jackson.databind.ObjectMapper;
import feign.codec.Encoder;
import feign.jackson.JacksonEncoder;

public static class FooModule extends SimpleModule {
    public FooModule() {
        addAbstractTypeMapping(Foo.class, FooImpl.class); // Map Foo to FooImpl
    }
}

import feign.Feign;
import feign.Response;
import feign.jackson.JacksonEncoder;
import feign.codec.Encoder;
import com.fasterxml.jackson.databind.ObjectMapper;

@SpringBootTest(webEnvironment = WebEnvironment.RANDOM_PORT)
class FooClientTest {

    @LocalServerPort
    int port;

    @Test
    public void clientTest() {
        // Create an ObjectMapper and register your custom FooModule
        ObjectMapper objectMapper = new ObjectMapper();
        objectMapper.registerModule(new FooModule());

        // Use the ObjectMapper with JacksonEncoder
        Encoder encoder = new JacksonEncoder(objectMapper);

        // Build the Feign client with the encoder
        final FooClient lClient = Feign.builder()
            .encoder(encoder)
            .target(FooClient.class, String.format("http://localhost:%s", port));

        // Test the client methods
        Response response = lClient.createFooImpl(new FooImpl(10));
        assertThat(response.status()).isEqualTo(404);

        response = lClient.createFoo(new FooImpl(10));
        assertThat(response.status()).isEqualTo(404); // This should no longer fail
    }

    public static class FooModule extends SimpleModule {
        public FooModule() {
            addAbstractTypeMapping(Foo.class, FooImpl.class); // Map Foo to FooImpl
        }
    }
}
