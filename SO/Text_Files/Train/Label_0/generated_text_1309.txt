import com.fasterxml.jackson.annotation.JsonCreator;
import com.fasterxml.jackson.annotation.JsonGetter;
import com.fasterxml.jackson.annotation.JsonProperty;
import com.fasterxml.jackson.databind.ObjectMapper;

public class Person {
    private final String name;
    private final int age;

    @JsonCreator
    public Person(@JsonProperty("name") String name, @JsonProperty("n_age") int age) {
        this.name = name;
        this.age = age;  // Use the deserialized age value
    }

    public String getName() {
        return name;
    }

    @JsonGetter("n_age") // Serialize the "age" field as "n_age" during serialization
    public int getAge() {
        return age;
    }

    public static void main(String... args) throws Exception {
        ObjectMapper mapper = new ObjectMapper();
        String json = "{\"name\":\"John\",\"n_age\":500}";

        Person person = mapper.readValue(json, Person.class);
        System.out.println(person.getAge()); // It should now print 500, as expected
    }
}
