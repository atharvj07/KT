class apples {
    public static void main(String[] args) {
        int bucky[] = {3, 4, 5, 6, 7};

        change(bucky);

        // Print updated array
        for (int value : bucky) {
            System.out.print(value + " ");  // Output: 8 9 10 11 12
        }
    }

    public static void change(int x[]) {
        for (int i = 0; i < x.length; i++) {
            x[i] += 5;  // Modifies the original array
        }
    }
}
