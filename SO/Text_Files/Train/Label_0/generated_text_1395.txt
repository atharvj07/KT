class Main {
    static class Node {
        int data;
        Node next;

        Node(int data) {
            this.data = data;
            this.next = null;
        }
    }

    Node head = null;

    public void addFirst(int data) {
        Node newNode = new Node(data);
        if (head == null) {
            head = newNode;
            return;
        }
        newNode.next = head;
        head = newNode;
    }

    // Print the linked list
    public void printLL() {
        Node n = head;
        if (n == null) {
            System.out.println("NULL");
            return;
        }
        while (n != null) {
            System.out.print(n.data + " --> ");
            n = n.next;
        }
        System.out.println("NULL");
    }

    // Remove nth node from the end of the list
    public Node removeNthNode(Node head, int nth) {
        if (head == null) {
            return null;  // If the list is empty, return null
        }

        int size = 0;
        Node curNode = head;
        // Calculate the size of the linked list
        while (curNode != null) {
            curNode = curNode.next;
            size++;
        }

        // If nth equals the size of the list, remove the head node
        if (nth == size) {
            return head.next;
        }

        Node prevNode = head;
        int i = 1;
        // Traverse to the (size - nth) node
        while (i < size - nth) {
            prevNode = prevNode.next;
            i++;
        }
        // Remove the nth node from the end
        if (prevNode != null && prevNode.next != null) {
            prevNode.next = prevNode.next.next;
        }
        return head;
    }

    public static void main(String[] args) {
        Main ll = new Main();
        ll.addFirst(90);
        ll.addFirst(40);
        ll.addFirst(45);

        System.out.println("Before removing nth node:");
        ll.printLL();

        // Remove 3rd node from the end
        ll.head = ll.removeNthNode(ll.head, 3);  // Update head after removal
        System.out.println("After removing nth node:");
        ll.printLL();
    }
}

Before removing nth node:
45 --> 40 --> 90 --> NULL
After removing nth node:
40 --> 90 --> NULL
