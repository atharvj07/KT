  public <T extends Filterable> List<T> filter(List<T> items) {
      return items.stream().filter(Filterable::meetsCriteria).collect(Collectors.toList());
  }
  
  public interface FilterStrategy<T> {
      boolean apply(T item);
  }

  public class TypeAFilterStrategy implements FilterStrategy<TypeA> {
      @Override
      public boolean apply(TypeA item) {
          // TypeA-specific filtering logic
          return item.someCondition();
      }
  }

  public class TypeBFilterStrategy implements FilterStrategy<TypeB> {
      @Override
      public boolean apply(TypeB item) {
          // TypeB-specific filtering logic
          return item.anotherCondition();
      }
  }

  public class FilterService {

      public <T> List<T> filter(List<T> items, FilterStrategy<T> strategy) {
          return items.stream().filter(strategy::apply).collect(Collectors.toList());
      }
  }
