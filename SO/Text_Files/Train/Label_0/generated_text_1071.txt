import spock.lang.*
import com.jcraft.jsch.*
import org.apache.commons.net.ftp.FTPClient

class SftpClientSpec extends Specification {

    // Shared setup for the server
    @Shared FakeSftpServer sftpServer

    def setupSpec() {
        // Initialize FakeSftpServer and configure it here
        sftpServer = new FakeSftpServer() 
        sftpServer.start()
        println("SFTP server started")
    }

    def cleanupSpec() {
        // Stop FakeSftpServer after all tests are done
        sftpServer.stop()
        println("SFTP server stopped")
    }

    def "test custom SFTP client upload"() {
        setup:
        // Setup custom client and upload a file, for example
        CustomSftpClient sftpClient = new CustomSftpClient()
        sftpClient.connect(sftpServer.host, sftpServer.port)

        when:
        sftpClient.upload("somefile.txt", "remote/path")

        then:
        // Check if the file is uploaded
        assert sftpServer.getFile("remote/path") == "somefile.txt"
    }

    def "test custom SFTP client download"() {
        setup:
        CustomSftpClient sftpClient = new CustomSftpClient()
        sftpClient.connect(sftpServer.host, sftpServer.port)

        when:
        sftpClient.download("remote/path", "local/path")

        then:
        // Assert that the file is downloaded correctly
        assert new File("local/path").exists()
    }
}

import org.spockframework.runtime.extension.Extension
import org.spockframework.runtime.extension.IMethodInvocation
import org.spockframework.runtime.extension.MethodInvocationContext

class SftpServerExtension extends Extension {

    def beforeMethod(MethodInvocationContext context) {
        // Set up the FakeSftpServer
        FakeSftpServer sftpServer = new FakeSftpServer()
        sftpServer.start()
        context.getSharedData().put("sftpServer", sftpServer)
    }

    def afterMethod(MethodInvocationContext context) {
        // Tear down the FakeSftpServer
        FakeSftpServer sftpServer = context.getSharedData().get("sftpServer")
        sftpServer.stop()
    }
}
