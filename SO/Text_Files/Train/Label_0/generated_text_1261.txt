@Override
public void start(Stage stage) {
    double padding = 10; // Adjust padding to prevent clipping

    Canvas c = new Canvas(715 + 2 * padding, 495 + 2 * padding); // Add padding
    GraphicsContext gc = c.getGraphicsContext2D();
    VBox v = new VBox();
    Pane p = new Pane();

    p.setPrefSize(715, 495); // Keep actual visible area fixed
    p.getChildren().add(c);
    c.relocate(-padding, -padding); // Shift canvas to keep drawing centered

    Button b1 = new Button("Draw Sinus at Canvas");
    b1.setOnAction(e -> drawSinus(c, gc, padding));

    v.getChildren().addAll(b1, p);
    Scene sc = new Scene(v);
    stage.setScene(sc);
    stage.setTitle("Drawing Lines - Dynamically at Runtime");
    stage.show();
}

private void drawSinus(Canvas c, GraphicsContext gc, double padding) {
    double height = c.getHeight() - 2 * padding;
    double width = c.getWidth() - 2 * padding;
    double multiplier = (2 * Math.PI) / width;
    double x1 = padding;
    double y1 = height / 2 + padding;
    double x2 = x1 + 1;
    double y2 = 0;
    int i = 0;

    gc.setStroke(Color.BLACK);
    gc.setLineWidth(10);

    while (i < width) {
        y2 = (height / 2 + padding) - ((height / 2) * Math.sin((x2 - padding) * multiplier));
        gc.strokeLine(x1, y1, x2, y2);
        x1 = x2;
        y1 = y2;
        x2++;
        i++;
    }
}
