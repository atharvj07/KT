import polars as pl

# Example DataFrame
df = pl.DataFrame({
    "a": [1, 2, 3, 4],
    "b": ["x", "y", "z", "w"]
})

# Identity expression that always evaluates to True
idexpr = pl.lit(True)

# Filter the DataFrame using the identity expression (equivalent to SELECT *)
filtered_df = df.filter(idexpr)

print(filtered_df)
