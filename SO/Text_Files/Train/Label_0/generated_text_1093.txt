DatabaseReference selectedActivityDatabase = FirebaseDatabase.getInstance().getReference().child("Activity").child("Do");
DatabaseReference usersDatabase = FirebaseDatabase.getInstance().getReference().child("Users");

selectedActivityDatabase.addChildEventListener(new ChildEventListener() {
    @Override
    public void onChildAdded(@NonNull DataSnapshot activitySnapshot, @Nullable String previousChildName) {
        if (activitySnapshot.exists()) {
            String currentActivityId = activitySnapshot.getKey();
            String activityName = activitySnapshot.child("name").getValue(String.class);

            // Create the card item for the activity
            cards item = new cards(currentActivityId, activityName);
            rowItems.add(item);
            activityTpeArrayAdapter.notifyDataSetChanged();

            // Now check if the activity ID exists in the 'Users/{userId}/preferences/Do' path
            // Here we are assuming you have the user ID available (e.g., `userId`)
            String userId = "wkFbhmSH6lfPkkyy6OGke0be1ys2"; // Replace with actual user ID

            DatabaseReference userPreferencesPath = usersDatabase.child(userId).child("preferences").child("Do").child(currentActivityId);

            // Check if the activity ID is already present in the user's preferences
            userPreferencesPath.addListenerForSingleValueEvent(new ValueEventListener() {
                @Override
                public void onDataChange(@NonNull DataSnapshot userActivitySnapshot) {
                    if (userActivitySnapshot.exists() && userActivitySnapshot.getValue(Boolean.class)) {
                        // The user has already answered, don't ask again
                        Log.d("ActivityCheck", "User has already answered for activity: " + currentActivityId);

                    }
                }

                @Override
                public void onCancelled(@NonNull DatabaseError databaseError) {
                    Log.e("ActivityCheck", "Error checking user preferences: " + databaseError.getMessage());
                }
            });
        }
    }
