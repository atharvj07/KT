public class Main {
    public static void main(String[] args) {
        System.out.println(clean("1(234)67"));      // Expected: "234"
        System.out.println(clean("123(45))"));     // Expected: "45)"
        System.out.println(clean("x)"));           // Expected: ""
        System.out.println(clean(")x("));          // Expected: ""
        System.out.println(clean("(abc))"));       // Expected: "abc)"
    }

    public static String clean(String seq) {
        if (seq.isEmpty()) {
            return "";
        }

        // Find the first opening parenthesis
        int startIdx = seq.indexOf('(');
        if (startIdx == -1) {
            return "";  // No opening parenthesis found
        }

        // Find the corresponding closing parenthesis
        int depth = 0;
        int endIdx = -1;
        for (int i = startIdx; i < seq.length(); i++) {
            char ch = seq.charAt(i);
            if (ch == '(') {
                depth++;
            } else if (ch == ')') {
                depth--;
                if (depth == 0) {
                    endIdx = i;
                    break;
                }
            }
        }

        if (endIdx == -1) {
            return "";  // No matching closing parenthesis
        }

        // Extend to include trailing ')' characters
        int lastClosingIdx = endIdx;
        while (lastClosingIdx + 1 < seq.length() && seq.charAt(lastClosingIdx + 1) == ')') {
            lastClosingIdx++;
        }

        return seq.substring(startIdx + 1, lastClosingIdx + 1);
    }
}

public static void main(String[] args) {
    System.out.println(clean("1(234)67"));      // Expected: "234"
    System.out.println(clean("123(45))"));     // Expected: "45)"

    System.out.println(clean(")x("));          // Expected: ""
    System.out.println(clean("(abc))"));       // Expected: "abc)"
}
