import android.content.Context;
import android.os.Handler;
import android.util.AttributeSet;
import android.widget.TextView;

public class TypeWriter extends androidx.appcompat.widget.AppCompatTextView {

    private int mIndex;
    private String mText;
    private Handler mHandler;
    private long mDelay = 100; // Delay between each character
    private Runnable mCharacterAdder;

    public TypeWriter(Context context) {
        super(context);
        init();
    }

    public TypeWriter(Context context, AttributeSet attrs) {
        super(context, attrs);
        init();
    }

    public TypeWriter(Context context, AttributeSet attrs, int defStyle) {
        super(context, attrs, defStyle);
        init();
    }

    private void init() {
        mHandler = new Handler();
        mIndex = 0;
    }

    public void setCharacterDelays(long delay) {
        mDelay = delay;
    }

    public void animateText(final String text) {
        mText = text;
        mIndex = 0;
        setText("");
        mCharacterAdder = new Runnable() {
            @Override
            public void run() {
                setText(mText.subSequence(0, mIndex++));
                if (mIndex <= mText.length()) {
                    mHandler.postDelayed(mCharacterAdder, mDelay);
                }
            }
        };
        mHandler.post(mCharacterAdder);
    }
}
