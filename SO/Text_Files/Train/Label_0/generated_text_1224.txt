import javax.swing.*;
import java.awt.*;
import java.awt.geom.GeneralPath;
import java.awt.geom.RoundRectangle2D;

public class RoundPanel extends JPanel {

    private static final long serialVersionUID = 1L;
    private Shape shape;
    private final int arcSize; // Defines the roundness of the left corners

    public RoundPanel(int arcSize) {
        this.arcSize = arcSize;
        setOpaque(false);
    }

    @Override
    protected void paintComponent(Graphics g) {
        Graphics2D g2 = (Graphics2D) g.create();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        // Create a GeneralPath for custom rounded shape (left corners rounded, right sharp)
        GeneralPath path = new GeneralPath();
        path.moveTo(arcSize, 0);
        path.lineTo(getWidth() - 1, 0); // Top-right corner (sharp)
        path.lineTo(getWidth() - 1, getHeight() - 1); // Bottom-right corner (sharp)
        path.lineTo(arcSize, getHeight() - 1); // Bottom-left corner (straight before curve)
        path.quadTo(0, getHeight() - 1, 0, getHeight() - arcSize); // Bottom-left curve
        path.lineTo(0, arcSize); // Left straight part
        path.quadTo(0, 0, arcSize, 0); // Top-left curve
        path.closePath();

        g2.setColor(getBackground());
        g2.fill(path);
        g2.dispose();

        super.paintComponent(g);
    }

    @Override
    protected void paintBorder(Graphics g) {
        Graphics2D g2 = (Graphics2D) g.create();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g2.setColor(getForeground());

        GeneralPath path = new GeneralPath();
        path.moveTo(arcSize, 0);
        path.lineTo(getWidth() - 1, 0);
        path.lineTo(getWidth() - 1, getHeight() - 1);
        path.lineTo(arcSize, getHeight() - 1);
        path.quadTo(0, getHeight() - 1, 0, getHeight() - arcSize);
        path.lineTo(0, arcSize);
        path.quadTo(0, 0, arcSize, 0);
        path.closePath();

        g2.draw(path);
        g2.dispose();
    }

    @Override
    public boolean contains(int x, int y) {
        if (shape == null || !shape.getBounds().equals(getBounds())) {
            GeneralPath path = new GeneralPath();
            path.moveTo(arcSize, 0);
            path.lineTo(getWidth() - 1, 0);
            path.lineTo(getWidth() - 1, getHeight() - 1);
            path.lineTo(arcSize, getHeight() - 1);
            path.quadTo(0, getHeight() - 1, 0, getHeight() - arcSize);
            path.lineTo(0, arcSize);
            path.quadTo(0, 0, arcSize, 0);
            path.closePath();
            shape = path;
        }
        return shape.contains(x, y);
    }
}

JFrame frame = new JFrame();
frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
frame.setSize(300, 200);

JPanel container = new JPanel(null); // Using null layout for absolute positioning
frame.setContentPane(container);

JPanel user_icon = new RoundPanel(20);
user_icon.setBackground(Color.decode("#363b41"));
user_icon.setBounds(50, 50, 100, 50);
container.add(user_icon);

frame.setVisible(true);
