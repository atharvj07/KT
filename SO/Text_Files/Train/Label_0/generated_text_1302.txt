import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.layout.GridPane;
import javafx.scene.layout.Priority;
import javafx.stage.Stage;

public class Tester extends Application {
    private static final int WIDTH = 300;
    private static final int HEIGHT = 400;

    @Override
    public void start(Stage stage) {
        GridPane pane = new GridPane();
        pane.setAlignment(Pos.TOP_LEFT);
        pane.setHgap(10);
        pane.setVgap(10);
        pane.setPadding(new Insets(10, 10, 10, 10));

        Button catButton = new Button("cat");
        Button monkeyButton = new Button("monkey");
        Button elephantButton = new Button("elephant and lion");

        // Set the Hgrow property to ensure the buttons grow horizontally
        GridPane.setHgrow(catButton, Priority.ALWAYS);
        GridPane.setHgrow(monkeyButton, Priority.ALWAYS);
        GridPane.setHgrow(elephantButton, Priority.ALWAYS);

        // Add the buttons to the GridPane
        pane.add(catButton, 0, 0);
        pane.add(monkeyButton, 0, 1);
        pane.add(elephantButton, 0, 2);

        Scene scene = new Scene(pane, WIDTH, HEIGHT);
        stage.setScene(scene);
        stage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
