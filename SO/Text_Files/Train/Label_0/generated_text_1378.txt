@Configuration
public class WebMvcConfig {
    private final long MAX_AGE_SECS = 3600;

    @Bean
    public WebMvcConfigurer addCorsMappingsConfigurer() {
        return new WebMvcConfigurer() {
            @Override
            public void addCorsMappings(CorsRegistry registry) {
                // Allow CORS for all paths, for the frontend running on localhost:4200
                registry.addMapping("/**")
                        .allowedOrigins("http://localhost:4200")  // Make sure your Angular frontend is allowed
                        .allowedMethods("GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS")  // Include DELETE and OPTIONS
                        .allowedHeaders("*")  // Allow all headers
                        .allowCredentials(true)  // If you want to allow credentials (cookies, etc.)
                        .maxAge(MAX_AGE_SECS);  // Cache preflight response for 1 hour
            }
        };
    }
}
