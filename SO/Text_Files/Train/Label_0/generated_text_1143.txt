import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.ComponentScan;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.FilterType;

@Configuration
@ComponentScan(
    basePackages = "com.example", // Change to your package
    includeFilters = @ComponentScan.Filter(
        type = FilterType.ASSIGNABLE_TYPE, 
        value = MyInterface.class
    )
)
public class MyConfiguration {

    // This will automatically detect beans implementing MyInterface
    @Autowired
    private List<MyInterface> myInterfaces;

    @Bean
    public void printBeans() {
        myInterfaces.forEach(bean -> System.out.println(bean.getClass().getName()));
    }
}

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.ApplicationContext;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.event.EventListener;
import org.springframework.context.annotation.ComponentScan;
import org.springframework.context.annotation.FilterType;
import org.springframework.context.ApplicationReadyEvent;

import javax.annotation.PostConstruct;
import java.util.List;

@Configuration
@ComponentScan(basePackages = "com.example")
public class MyConfiguration {

    @Autowired
    private ApplicationContext applicationContext;

    @PostConstruct
    public void init() {
        // Scan the context for all beans implementing MyInterface
        List<MyInterface> beans = applicationContext.getBeansOfType(MyInterface.class).values()
                .stream()
                .toList();
        
        // Handle the beans as needed
        beans.forEach(bean -> System.out.println(bean.getClass().getName()));
    }
}
