package sqlPackage;

import java.sql.*;
import java.util.ArrayList;
import java.util.List;

public class SqlExample {
    public List<String> playerCountry() throws SQLException {
        List<String> players = new ArrayList<>();

        String url = "jdbc:mysql://localhost:3306/mxplayerdb";
        String user = "root";
        String password = "root";

        try (Connection connection = DriverManager.getConnection(url, user, password);
             Statement statement = connection.createStatement();
             ResultSet rs = statement.executeQuery("SELECT userName FROM myplayers")) {

            while (rs.next()) {
                players.add(rs.getString("userName"));
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
        return players;
    }
}

@WebServlet("/players")
public class PlayerServlet extends HttpServlet {
    protected void doGet(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
        SqlExample sqlExample = new SqlExample();
        List<String> players;

        try {
            players = sqlExample.playerCountry();
            response.getWriter().println(players);
        } catch (SQLException e) {
            e.printStackTrace();
            response.getWriter().println("Error fetching players.");
        }
    }
}
