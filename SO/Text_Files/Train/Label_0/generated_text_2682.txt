import duckdb

duckdb.sql("""
    SELECT *,
        CASE 
            WHEN count(sales) OVER (
                PARTITION BY id 
                ORDER BY d 
                ROWS BETWEEN 1 PRECEDING AND CURRENT ROW
            ) < 2 THEN NULL 
            ELSE avg(sales) OVER (
                PARTITION BY id 
                ORDER BY d 
                ROWS BETWEEN 1 PRECEDING AND CURRENT ROW
            )
        END AS sales_rolling
    FROM df
    ORDER BY id, d
""")
