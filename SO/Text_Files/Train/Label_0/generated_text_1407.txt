import java.util.*;
import java.util.regex.*;

public class JiraTableParser {

    private static final String JIRA_TABLE_REGEX = "(\\|\\|.*?\\|\\|\\s*(\\n|\\r\\n|\\r)*)+(\\|.*?\\|\\s*(\\n|\\r\\n|\\r)*)+";

    public static void main(String[] args) {
        String input = "||Heading 1 Table 2||Heading 2 Table 2||\n"
                     + "|Col A1|Col A2|\n"
                     + "|SECOND ROW|Second row too|\n"
                     + "|Row with image|!image-2023-04-24-17-51-07-167.png|width=359,height=253!|\n"
                     + "|Row with image and text|smol !image-2023-05-02-12-42-16-942.png|width=347,height=231! kitten|\n"
                     + "|Row with text and new lines|Text\n Ã \n New Line\n Ã \n More text\n Much more text|";

        JiraTableParser parser = new JiraTableParser();
        parser.parseJiraTable(input);
    }

    public void parseJiraTable(String input) {
        Pattern tablePattern = Pattern.compile(JIRA_TABLE_REGEX);
        Matcher matcher = tablePattern.matcher(input);

        // Loop through the matches (tables)
        while (matcher.find()) {
            String jiraTable = matcher.group();

            // Split the input string into rows based on the newline characters
            String[] rowArray = jiraTable.split("(?<=\\|)\\s*(?=\\n|\\r)");

            int size = 0;
            List<String> headers = new ArrayList<>();
            List<String> cells = new ArrayList<>();

            for (String row : rowArray) {
                row = row.trim();

                // If the row starts with "||", it's a header row
                if (row.startsWith("||")) {
                    headers.addAll(Arrays.asList(row.substring(2, row.length() - 2).split("\\|\\|")));
                    size = headers.size();
                } else if (row.startsWith("|")) {
                    // Handle rows with regular cells
                    cells.addAll(Arrays.asList(row.substring(1, row.length() - 1).split("\\|")));
                    if (size == 0) {
                        size = cells.size();
                    }
                }
            }

            // Debug output
            System.out.println("Headers: " + headers);
            System.out.println("Cells: " + cells);
        }
    }
