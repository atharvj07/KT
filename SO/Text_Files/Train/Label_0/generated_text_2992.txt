from collections import Counter

def find_unique(lists):
    unique_sets = set()
    result = []
    
    for lst in lists:
        key = frozenset(Counter(lst).items())  # Create an order-independent hashable representation
        if key not in unique_sets:
            unique_sets.add(key)
            result.append(lst)
    
    return result
