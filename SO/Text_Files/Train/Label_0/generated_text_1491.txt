public void print(String name) {
    Node node = this.find(name);
    printRecursive(node, 0);
}

private void printRecursive(Node node, int depth) {
    if (node == null) return;

    // Print the node with indentation
    System.out.println(" ".repeat(depth * 2) + "--".repeat(Math.max(0, depth)) + node.name + " (" + node.value + ")");

    // Recursively print children
    for (int i = 0; i < node.childrensCount; i++) {
        printRecursive(node.childrens[i], depth + 1);
    }
}
