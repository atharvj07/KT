import javax.swing.*;
import javax.swing.plaf.metal.MetalMenuBarUI;
import java.awt.*;

public class WithUIManager {
    private static final JTextArea pad = new JTextArea();
    private static JLabel statusBar = new JLabel("||       Ln 1, Col 1  ", JLabel.RIGHT);
    private static JFrame f = new JFrame("Overriding UIManager");

    public static void main(String[] args){
        try {
            UIManager.setLookAndFeel(UIManager.getSystemLookAndFeelClassName());
        } catch (Exception e) {
            e.printStackTrace();
        }

        f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        f.setSize(600, 600);
        f.setBackground(Color.decode("#00FF00"));
        statusBar.setOpaque(true);
        statusBar.setBackground(Color.decode("#00FFFF"));
        statusBar.setForeground(Color.BLACK);
        f.add(new JScrollPane(pad), BorderLayout.CENTER);
        f.add(statusBar, BorderLayout.SOUTH);

        // Dummy labels for decoration
        JLabel dummy = new JLabel("  ");
        dummy.setOpaque(true);
        dummy.setBackground(Color.decode("#00FFFF"));
        JLabel dummy1 = new JLabel("  ");
        dummy1.setOpaque(true);
        dummy1.setBackground(Color.decode("#00FFFF"));
        f.add(dummy, BorderLayout.EAST);
        f.add(dummy1, BorderLayout.WEST);

        // Create MenuBar
        JMenuBar mb = new JMenuBar();
        mb.setOpaque(true); // Make sure MenuBar is opaque
        mb.setBackground(Color.decode("#00FFFF")); // Set the background color for MenuBar

        // Set custom UI for JMenuBar to allow background color customization
        mb.setUI(new MetalMenuBarUI() {
            @Override
            public void installUI(JComponent c) {
                super.installUI(c);
                c.setOpaque(true); // Ensures MenuBar is opaque
                c.setBackground(Color.decode("#00FFFF")); // Set custom background color
            }
        });

        // Create Menus
        JMenu fileMenu = new JMenu("file");
        fileMenu.setOpaque(true); // Ensure JMenu is opaque
        fileMenu.setBackground(Color.decode("#00FFFF")); // Set background for JMenu

        JMenu editMenu = new JMenu("edit");
        editMenu.setOpaque(true);
        editMenu.setBackground(Color.decode("#00FFFF"));

        JMenu formatMenu = new JMenu("format");
        formatMenu.setOpaque(true);
        formatMenu.setBackground(Color.decode("#00FFFF"));

        JMenu viewMenu = new JMenu("view");
        viewMenu.setOpaque(true);
        viewMenu.setBackground(Color.decode("#00FFFF"));

        JMenu helpMenu = new JMenu("help");
        helpMenu.setOpaque(true);
        helpMenu.setBackground(Color.decode("#00FFFF"));

        // Add Menus to MenuBar
        mb.add(fileMenu);
        mb.add(editMenu);
        mb.add(formatMenu);
        mb.add(viewMenu);
        mb.add(helpMenu);

        // Add MenuBar to Frame
        f.setJMenuBar(mb);

        f.setVisible(true);
    }
}
