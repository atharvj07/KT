float gridSize = 20; // Number of grid columns
float pixelSize = 40; // Size of each rectangle
float radius;

void setup() {
  size(1920, 1080, P3D);
  frameRate(30);
  rectMode(CENTER);
  noStroke();
}

void draw() {
  background(0);
  radius = map(mouseX, 0, width, 50, 300); // Adjust radius dynamically

  translate(width / 2, height / 2);
  rotateX(PI / 4); // Tilt for better perspective

  for (int x = 0; x < gridSize; x++) {
    float angle = map(x, 0, gridSize, 0, TWO_PI); // Angle around the cylinder
    float xPos = radius * cos(angle); // Circular X position
    float zPos = radius * sin(angle); // Circular Z position

    for (int y = 0; y < gridSize; y++) {
      pushMatrix();
      translate(xPos, y * pixelSize, zPos); // Move each rectangle to a cylindrical position
      fill(255);
      rect(0, 0, pixelSize / 2, pixelSize / 2);
      popMatrix();
    }
  }
}
