public class ConsecutiveDigitSum {
    public static void main(String[] args) {
        System.out.println(solution("19938832066")); // Expected: "2831632012"
    }

    public static String solution(String number) {
        while (true) {
            String next = processNumber(number);
            if (next.equals(number)) {
                return next;
            }
            number = next; // Continue processing until stabilization
        }
    }

    private static String processNumber(String number) {
        StringBuilder result = new StringBuilder();
        int n = number.length();

        for (int i = 0; i < n; i++) {
            int count = 1;
            while (i + 1 < n && number.charAt(i) == number.charAt(i + 1)) {
                count++;
                i++; // Move to the next duplicate
            }
            // Convert character to integer and multiply by count
            int sum = count * Character.getNumericValue(number.charAt(i));
            result.append(sum);
        }

        return result.toString();
    }
}
