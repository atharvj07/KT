spring:
  cloud:
    loadbalancer:
      enabled: true

@Bean(name = "webClientConsulAvailability")
public WebClient webClientConsulAvailability(
  WebClient.Builder webClientBuilder,
  ReactorLoadBalancerExchangeFilterFunction lbFunction,
  ExchangeFilterFunction logFilter
) {
  return webClientBuilder
    .filter(lbFunction) // This ensures load balancing
    .filter(logFilter)  // Optional logging filter
    .baseUrl("http://cachedavailability-integrations-service") // This ensures it uses the service name
    .build();
}

webClientConsulAvailability.get()
  .uri("http://cachedavailability-integrations-service/testing")  // This should resolve using Consul
  .retrieve()
  .bodyToFlux(MyDTO.class)
  .doOnError(e -> {
    if (isErrorLogLevel(e)) {
      log.error("Communication error with uri: " + uri, e);
    } else {
      log.warn("Communication error with uri: " + uri, e);
    }
  })
  .onErrorResume(e -> Flux.empty());

logging:
  level:
    org.springframework.cloud: DEBUG
    org.springframework.web.reactive.function.client: DEBUG
    org.springframework.cloud.client.discovery: DEBUG
