from PyQt5.QtCore import Qt
from PyQt5.QtGui import QImage, QPainter, QColor
from PyQt5.QtWidgets import QApplication, QWidget, QVBoxLayout, QLabel
import sys

class UVDisplayWidget(QWidget):
    def __init__(self, texture_path, uv_coords):
        super().__init__()
        
        # Load the texture image
        self.texture = QImage(texture_path)
        self.uv_coords = uv_coords  # List of UV coordinates [(u1, v1), (u2, v2), ...]

        # Set up the window size based on texture size
        self.setFixedSize(self.texture.width(), self.texture.height())

    def paintEvent(self, event):
        painter = QPainter(self)
        
        # Draw the texture image
        painter.drawImage(0, 0, self.texture)
        
        # Draw UV coordinates
        painter.setPen(QColor(255, 0, 0))  # Red color for UV coordinates
        for uv in self.uv_coords:
            x = uv[0] * self.texture.width()  # Map U to X position on texture
            y = uv[1] * self.texture.height()  # Map V to Y position on texture
            painter.drawEllipse(x - 5, y - 5, 10, 10)  # Draw a small circle at each UV point

        painter.end()

class MainWindow(QWidget):
    def __init__(self):
        super().__init__()
        self.setWindowTitle('UV Mapping Viewer')

        # UV Coordinates and texture path (example data)
        uv_coords = [(0.2, 0.3), (0.5, 0.5), (0.7, 0.8)]  # Example UV points
        texture_path = "path/to/your/texture.jpg"  # Your texture image path
        
        # Set up the widget
        uv_widget = UVDisplayWidget(texture_path, uv_coords)
        
        layout = QVBoxLayout()
        layout.addWidget(uv_widget)
        self.setLayout(layout)

app = QApplication(sys.argv)
window = MainWindow()
window.show()
sys.exit(app.exec_())

# Make sure you're in object mode and have an active object with a mesh

uv_layer = obj.data.uv_layers.active.data

# Loop through all faces and print their UV coordinates
for poly in obj.data.polygons:
    for loop_index in poly.loop_indices:
        uv_coord = uv_layer[loop_index].uv
        print(f"UV Coord: {uv_coord}")
