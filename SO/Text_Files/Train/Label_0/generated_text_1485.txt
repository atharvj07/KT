public void checkUserPass(String username, String password, final UserCheckCallback callback) {
    UserModel user = new UserModel();
    user.setUsername(username);
    user.setPassword(password);

    Assistant.assistant.client.userCheck(user).enqueue(new Callback<Boolean>() {
        @Override
        public void onResponse(@NonNull Call<Boolean> call, @NonNull Response<Boolean> response) {
            if (response.isSuccessful() && response.body() != null) {
                callback.onResult(response.body()); // Notify caller with result
                Log.i("MyTag", "Response: " + response.body());
            } else {
                callback.onResult(false); // Handle failure
                Log.i("MyTag", "Not Success: " + response);
            }
        }

        @Override
        public void onFailure(@NonNull Call<Boolean> call, @NonNull Throwable throwable) {
            callback.onResult(false); // Notify failure
            Log.e("MyTag", "Request Failed: " + throwable.getMessage());
        }
    });
}

public interface UserCheckCallback {
    void onResult(boolean userExists);
}
