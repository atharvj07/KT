import java.io.FileOutputStream;
import java.io.OutputStream;
import java.util.concurrent.Callable;
import picocli.CommandLine.Command;
import picocli.CommandLine.Option;

@Command(name = "fileCli", description = "Performs file manipulation operations", mixinStandardHelpOptions = true, version = "File Client 1.0")
public class TranslateTXML implements Callable<String> {

    @Option(names = "-o", description = "Output file path", required = true)
    private String output;

    @Override
    public String call() throws Exception {
        try (OutputStream fileOutputStream = new FileOutputStream(output)) {
            WriteSourceTranslatedToTXML.makeTranslated(fileOutputStream, output);
            System.out.println("Translation complete.");
            System.out.println("File \"" + output + "\" has been created.");
        } catch (Exception e) {
            System.err.println("Error writing file: " + e.getMessage());
        }
        return "Done";
    }
}
