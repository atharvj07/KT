import java.util.ArrayList;
import java.util.List;

public class Main {
    public interface Card {
        String name();
    }

    public static void main(String[] args) {
        List<Card> cards = new ArrayList<>();
        cards.add(() -> "2");
        cards.add(() -> "4");

        // Print the names of the cards
        for (Card card : cards) {
            System.out.println(card.name());
        }
    }
}
