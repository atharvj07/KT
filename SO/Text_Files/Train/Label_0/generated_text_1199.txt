public String handleRequest(KinesisEvent kinesisEvent, Context context) {
    List<myObj> allObj = new ArrayList<>();

    for (KinesisEvent.KinesisEventRecord record : kinesisEvent.getRecords()) {
        String dataInString = null; // Declare outside try to use in catch

        try {
            ByteBuffer byteBuffer = record.getKinesis().getData();
            if (!byteBuffer.hasArray()) continue;

            dataInString = new String(byteBuffer.array(), StandardCharsets.UTF_8);
            myObj parsedObj = jsonSerDe.fromJson(dataInString, myObj.class);
            allObj.add(parsedObj);
        } 
        catch (Exception ex) {
            logger.error("Parsing failed for input: {}", dataInString, ex);
        }
    }

    // Some code using allObj

    return "Processing complete";
}
