.button {
    -fx-padding: 0 0 0 5;
    -fx-background-color: green;
}

public class NewMain extends Application {

    @Override
    public void start(Stage primaryStage) {
        var button = new Button("Button");

        // Set dynamic padding values
        double paddingTop = 0;
        double paddingRight = 0;
        double paddingBottom = 0;
        double paddingLeft = 5;

        // Apply padding to the button
        button.setStyle("-fx-padding: " + paddingTop + " " + paddingRight + " " + paddingBottom + " " + paddingLeft + ";");

        VBox root = new VBox(button);
        root.getStylesheets().add(NewMain.class.getResource("test.css").toExternalForm());
        Scene scene = new Scene(root, 100, 100);
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
