import java.awt.*;
import java.awt.image.BufferedImage;
import java.io.File;
import java.io.IOException;
import javax.imageio.ImageIO;
import java.util.Random;

public class GrayscaleToColor {
    
    public static void main(String[] args) {
        // Path to your grayscale image
        String inputImagePath = "path/to/your/grayscale/image.jpg";
        String outputImagePath = "path/to/your/output/colored_image.jpg";

        try {
            // Read the grayscale image
            BufferedImage grayscaleImage = ImageIO.read(new File(inputImagePath));

            // Convert grayscale to color
            BufferedImage coloredImage = convertGrayscaleToColor(grayscaleImage);

            // Save the colored image
            ImageIO.write(coloredImage, "jpg", new File(outputImagePath));

            System.out.println("Image successfully saved at: " + outputImagePath);

        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    public static BufferedImage convertGrayscaleToColor(BufferedImage grayscaleImage) {
        // Create a new image to hold the colored version
        int width = grayscaleImage.getWidth();
        int height = grayscaleImage.getHeight();
        BufferedImage coloredImage = new BufferedImage(width, height, BufferedImage.TYPE_INT_RGB);

        Random random = new Random();

        for (int y = 0; y < height; y++) {
            for (int x = 0; x < width; x++) {
                // Get the pixel value (grayscale intensity)
                int grayPixel = grayscaleImage.getRGB(x, y);

                // Extract the grayscale intensity (assuming RGB are the same in grayscale images)
                int gray = (grayPixel >> 16) & 0xFF; // Assuming RGB are the same, use Red value

                // Generate a random color (you can adjust how much you want the color to vary based on the gray value)
                int red = (random.nextInt(256) + gray) / 2;
                int green = (random.nextInt(256) + gray) / 2;
                int blue = (random.nextInt(256) + gray) / 2;

                // Ensure the color values are within the 0-255 range
                red = Math.min(255, Math.max(0, red));
                green = Math.min(255, Math.max(0, green));
                blue = Math.min(255, Math.max(0, blue));

                // Set the new color in the colored image
                int newColor = (red << 16) | (green << 8) | blue;
                coloredImage.setRGB(x, y, newColor);
            }
        }

        return coloredImage;
    }
}

public static void main(String[] args) {
    // Path to your grayscale image
    String inputImagePath = "path/to/your/grayscale/image.jpg";
    String outputImagePath = "path/to/your/output/colored_image.jpg";
    try {
        // Read the grayscale image
        BufferedImage grayscaleImage = ImageIO.read(new File(inputImagePath));
        // Convert grayscale to color
        BufferedImage coloredImage = convertGrayscaleToColor(grayscaleImage);
        // Save the colored image
        ImageIO.write(coloredImage, "jpg", new File(outputImagePath));
        System.out.println("Image successfully saved at: " + outputImagePath);
    } catch (IOException e) {
        e.printStackTrace();
    }
}

    // Create a new image to hold the colored version
    int width = grayscaleImage.getWidth();
    int height = grayscaleImage.getHeight();
    BufferedImage coloredImage = new BufferedImage(width, height, BufferedImage.TYPE_INT_RGB);
    Random random = new Random();
    for (int y = 0; y < height; y++) {
        for (int x = 0; x < width; x++) {
            // Get the pixel value (grayscale intensity)
            int grayPixel = grayscaleImage.getRGB(x, y);
            // Extract the grayscale intensity (assuming RGB are the same in grayscale images)
            int gray = (grayPixel >> 16) & 0xFF; // Assuming RGB are the same, use Red value
            // Generate a random color (you can adjust how much you want the color to vary based on the gray value)
            int red = (random.nextInt(256) + gray) / 2;
            int green = (random.nextInt(256) + gray) / 2;
            int blue = (random.nextInt(256) + gray) / 2;
            // Ensure the color values are within the 0-255 range
            red = Math.min(255, Math.max(0, red));
            green = Math.min(255, Math.max(0, green));
            blue = Math.min(255, Math.max(0, blue));
            // Set the new color in the colored image
            int newColor = (red << 16) | (green << 8) | blue;
            coloredImage.setRGB(x, y, newColor);
        }
    }
    return coloredImage;
}
