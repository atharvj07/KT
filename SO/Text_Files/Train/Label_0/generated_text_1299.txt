static List<Map<String, Object>> driverList = new ArrayList<>();

import java.util.*;

public class Main {
    public static void main(String[] args) {
        // Initialize the list to store driver data
        List<Map<String, Object>> driverList = new ArrayList<>();
        
        // Adding data for a driver
        Map<String, Object> driver1 = new HashMap<>();
        driver1.put("name", "John");
        driver1.put("age", 30);
        driver1.put("height", 5.9);
        driverList.add(driver1);
        
        // Adding another driver
        Map<String, Object> driver2 = new HashMap<>();
        driver2.put("name", "Jane");
        driver2.put("age", 28);
        driver2.put("height", 5.6);
        driverList.add(driver2);
        
        // Display all drivers' data
        for (Map<String, Object> driver : driverList) {
            System.out.println("Name: " + driver.get("name"));
            System.out.println("Age: " + driver.get("age"));
            System.out.println("Height: " + driver.get("height"));
            System.out.println("-----");
        }
    }
}

   class Driver {
       String name;
       int age;
       double height;
       
       // Constructor
       public Driver(String name, int age, double height) {
           this.name = name;
           this.age = age;
           this.height = height;
       }
       
       // Getters and setters (optional, if needed)
   }

   public class Main {
       public static void main(String[] args) {
           List<Driver> driverList = new ArrayList<>();

           driverList.add(new Driver("Jane", 28, 5.6));
           
           // Accessing data
           for (Driver driver : driverList) {
               System.out.println("Name: " + driver.name);
               System.out.println("Age: " + driver.age);
               System.out.println("Height: " + driver.height);
               System.out.println("-----");
           }
       }
   }
