import java.io.*;
import java.nio.*;
import java.nio.charset.*;
import java.nio.channels.*;
import java.nio.file.*;
import java.util.*;

public class ReverseFileReader {

    private static final Charset CHARSET = StandardCharsets.UTF_8;
    private static final int BUFFER_SIZE = 1024;  // Size of the buffer

    public static void readReverseFile(Path filePath) throws IOException {
        // Open a file channel to read the file
        try (FileChannel fileChannel = FileChannel.open(filePath, StandardOpenOption.READ)) {
            long fileSize = fileChannel.size();
            ByteBuffer buffer = ByteBuffer.allocate(BUFFER_SIZE);
            CharsetDecoder decoder = CHARSET.newDecoder();
            
            long position = fileSize - 1;  // Start from the end of the file
            int bytesRead;

            // Loop through the file in reverse
            while (position >= 0) {
                // Calculate how much to read in this iteration
                int bytesToRead = (int) Math.min(BUFFER_SIZE, position + 1);
                buffer.clear();
                
                // Read the chunk of the file into the buffer
                fileChannel.position(position - bytesToRead + 1);
                bytesRead = fileChannel.read(buffer);
                
                // Adjust the position for the next iteration
                position -= bytesRead;

                if (bytesRead > 0) {
                    buffer.flip();  // Prepare the buffer for reading

                    // Decode and process characters from the buffer
                    CharBuffer charBuffer = decoder.decode(buffer);
                    processReverse(charBuffer);  // Custom processing method for each chunk
                }
            }
        }
    }

    private static void processReverse(CharBuffer charBuffer) {
        // Process the characters in reverse order
        List<Character> reverseCharacters = new ArrayList<>();
        while (charBuffer.hasRemaining()) {
            reverseCharacters.add(charBuffer.get());
        }
        
        // Printing characters in reverse order
        for (int i = reverseCharacters.size() - 1; i >= 0; i--) {
            System.out.print(reverseCharacters.get(i));
        }
    }

    public static void main(String[] args) throws IOException {
        // Replace this with the actual file path
        Path filePath = Paths.get("largefile.txt");
        
        // Read and process the file in reverse
        readReverseFile(filePath);
    }
}
