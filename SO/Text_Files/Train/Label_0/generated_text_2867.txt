import polars as pl

df_pol = pl.DataFrame({'test_names': [['Mallesham', '', 'Bhavik', 'Jagarini', 'Jose', 'Fernando'],
                                      ['', '', '', 'ABC', '', 'XYZ']]})

df_pol = df_pol.with_columns(
    pl.col('test_names')
    .apply(lambda x: [i for i in x if i != ''], return_dtype=pl.List(pl.Utf8))  # filter empty strings
    .list.lengths()
    .alias('tot_names')
)

print(df_pol)
