import org.springframework.context.ApplicationContext;
import org.springframework.context.annotation.Lazy;
import org.springframework.stereotype.Component;
import test.spring.springtest.service.JavaService;

@Component
public class JavaStarter {
    private final ApplicationContext applicationContext;

    public JavaStarter(ApplicationContext applicationContext) {
        this.applicationContext = applicationContext;
    }

    public void invokeTestMethod(String input) {
        JavaService javaService = applicationContext.getBean(JavaService.class); // Retrieve the proxy
        for (int i = 0; i < 3; i++) {
            System.out.println("Output: " + javaService.testMethod(input));
            System.out.println();
        }
    }
}

@Component
public class JavaStarter {
    private final JavaService javaService;

    public JavaStarter(@Lazy JavaService javaService) { // Ensure proxy is created
        this.javaService = javaService;
    }

    public void invokeTestMethod(String input) {
        for (int i = 0; i < 3; i++) {
            System.out.println("Output: " + javaService.testMethod(input));
            System.out.println();
        }
    }
}

import org.springframework.boot.CommandLineRunner;
import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;

@SpringBootApplication

    private final JavaStarter javaStarter;

}
