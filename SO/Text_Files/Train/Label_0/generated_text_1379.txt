import com.github.kwhat.jnativehook.GlobalScreen;
import com.github.kwhat.jnativehook.NativeHookException;
import com.github.kwhat.jnativehook.keyboard.NativeKeyEvent;
import com.github.kwhat.jnativehook.keyboard.NativeKeyListener;
import javafx.event.ActionEvent;
import javafx.fxml.FXML;
import javafx.scene.control.Button;
import javafx.scene.control.Label;

public class HotKeyWindowController implements NativeKeyListener {
    @FXML
    private Label stopHotKeyLabel;
    @FXML
    private Label startHotKeyLabel;
    @FXML
    private Button stopHotKeyButton;
    @FXML
    private Button startHotKeyButton;

    private static String startKey = "F1";
    private static String stopKey = "F3";
    private static String key;
    
    private boolean isWaitingForKey = false;

    // Constructor to initialize the listener
    public HotKeyWindowController() {
        try {
            GlobalScreen.registerNativeHook();
            GlobalScreen.addNativeKeyListener(this);
        } catch (NativeHookException e) {
            e.printStackTrace();
        }
    }

    // This method is called whenever a key is pressed
    @Override
    public void nativeKeyPressed(NativeKeyEvent e) {
        if (isWaitingForKey) {
            key = NativeKeyEvent.getKeyText(e.getKeyCode());
            // Update the label based on the captured key
            if (stopHotKeyButton.isFocused()) {
                stopKey = key;
                stopHotKeyLabel.setText("Stop HotKey: " + stopKey);
            } else if (startHotKeyButton.isFocused()) {
                startKey = key;
                startHotKeyLabel.setText("Start HotKey: " + startKey);
            }
            // Disable waiting for key
            isWaitingForKey = false;
            // Unregister key listener after the key is captured
            try {
                GlobalScreen.removeNativeKeyListener(this);
            } catch (NativeHookException ex) {
                ex.printStackTrace();
            }
        }
    }

    // This method is called whenever a key is released (not used here)
    @Override
    public void nativeKeyReleased(NativeKeyEvent e) {}

    // This method is called when the stopHotKeyButton is clicked
    public void stopHotKeyButtonClicked(ActionEvent actionEvent) {
        // Enable waiting for a key press to set the stop hotkey
        isWaitingForKey = true;
        stopHotKeyLabel.setText("Press a key to set Stop HotKey");
        try {
            GlobalScreen.addNativeKeyListener(this);
        } catch (NativeHookException e) {
            e.printStackTrace();
        }
    }

    // This method is called when the startHotKeyButton is clicked
    public void startHotKeyButtonClicked(ActionEvent actionEvent) {
        // Enable waiting for a key press to set the start hotkey
        isWaitingForKey = true;
        startHotKeyLabel.setText("Press a key to set Start HotKey");
        try {
            GlobalScreen.addNativeKeyListener(this);
        } catch (NativeHookException e) {
            e.printStackTrace();
        }
    }
}
