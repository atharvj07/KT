import polars as pl

df = pl.DataFrame(
    {
        "frequency": [0.5, None, None, None],
        "frequency_ref": ["a", "z", "a", "a"],
        "a": [1, 2, 3, 4],
        "z": [5, 6, 7, 8],
    }
)

df = df.with_columns(
    pl.when(pl.col("frequency").is_not_null())
    .then(pl.col("frequency"))
    .otherwise(
        df.gather("frequency_ref", value_name="res")["res"]
    )
    .alias("res")
)

print(df)
