import polars as pl

df = pl.DataFrame(
    {
        "Case": ["case1", "case1"],
        "List": [[1, 2, 3], [4, 5, 6]],
    }
)

# Perform the groupby operation and sum the lists
result = (
    df.groupby("Case")
    .agg(pl.col("List").arr.sum().alias("List"))  # Sum the lists within each group
)

print(result)
