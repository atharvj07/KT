import java.io.File;
import java.io.FileNotFoundException;
import java.util.Scanner;
import java.util.regex.*;
import java.util.*;

public class GraphParser {
    // Regex patterns
    private static final String SOMMETREGEX = "[a-zA-Z0-9\\-_/]+";  // Must be at least one character
    private static final String POIDSREGEX = "[\\-+]?\\d+(\\.\\d+)?";
    private static final String EDGE_REGEX = "\\((" + SOMMETREGEX + "),(" + POIDSREGEX + ")\\)";

    // Read file and remove whitespace
    private static String fileToString(File f) {
        StringBuilder str = new StringBuilder();
        try {
            Scanner s = new Scanner(f);
            while (s.hasNextLine()) {
                str.append(s.nextLine().trim()).append("\n"); // Keep newlines
            }
            s.close();
        } catch (FileNotFoundException e) {
            System.out.println("ERREUR => FileNotFoundException");
            e.printStackTrace();
        }
        return str.toString();
    }

    // Build the graph from a file
    private static StdGraph<String> buildGraphFromFile(File f) {
        StdGraph<String> graph = new StdGraph<String>();
        String fileContent = fileToString(f);

        // Line-by-line processing
        Scanner scanner = new Scanner(fileContent);
        while (scanner.hasNextLine()) {
            String line = scanner.nextLine().trim();
            if (line.isEmpty()) continue;

            // Match vertex and its edges
            Pattern pattern = Pattern.compile("^(" + SOMMETREGEX + "):\\s*(.*)$");
            Matcher matcher = pattern.matcher(line);
            if (!matcher.matches()) {
                scanner.close();
                return null; // Invalid format
            }

            String vertex = matcher.group(1);
            String edgesPart = matcher.group(2);
            graph.addSommet(vertex);

            // Process edges
            Matcher edgeMatcher = Pattern.compile(EDGE_REGEX).matcher(edgesPart);
            while (edgeMatcher.find()) {
                String neighbor = edgeMatcher.group(1);
                double weight = Double.parseDouble(edgeMatcher.group(2));

                // Ensure neighbor exists
                if (!graph.contains(neighbor)) {
                    graph.addSommet(neighbor);
                }
                graph.addArete(vertex, neighbor, weight);
            }
        }
        scanner.close();
        return graph;
    }

    // Dummy graph class for testing
    static class StdGraph<T> {
        private Map<T, Map<T, Double>> adjList = new HashMap<>();

        public void addSommet(T vertex) {
            adjList.putIfAbsent(vertex, new HashMap<>());
        }

        public void addArete(T from, T to, double weight) {
            adjList.get(from).put(to, weight);
        }

        public boolean contains(T vertex) {
            return adjList.containsKey(vertex);
        }

        public void printGraph() {
            for (T vertex : adjList.keySet()) {
                System.out.print(vertex + ": ");
                for (Map.Entry<T, Double> edge : adjList.get(vertex).entrySet()) {
                    System.out.print("(" + edge.getKey() + ", " + edge.getValue() + ") ");
                }
                System.out.println();
            }
        }
    }

    // Test case
    public static void main(String[] args) {
        File file = new File("graph.txt");
        StdGraph<String> graph = buildGraphFromFile(file);

        if (graph != null) {
            System.out.println("Graph loaded successfully:");
            graph.printGraph();
        } else {
            System.out.println("Invalid graph format.");
        }
    }
}
