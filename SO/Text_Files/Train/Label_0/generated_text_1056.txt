\bOx\s+(\d*\.*?\d+).*?\s+(\d*\.*?\d+)\b

import re

text = "Ox   2.66 abcda   34.3 abfdasd"
pattern = r"\bOx\s+(\d*\.*?\d+).*?\s+(\d*\.*?\d+)\b"

match = re.search(pattern, text)
if match:
    num1, num2 = match.groups()
    print("First number:", num1)
    print("Second number:", num2)
