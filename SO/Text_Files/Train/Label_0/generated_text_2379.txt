import numpy as np
import matplotlib.pyplot as plt

x = [0, 1, 2, 3, 4, 5]
y = np.sin(x)
y2 = np.tan(x)

fig = plt.figure()
ax1 = fig.add_subplot(2, 1, 1)
ax1.plot(x, y)

ax2 = fig.add_subplot(2, 1, 2)
ax2.plot(x, y2)

colors = ['b', 'g', 'r', 'c', 'm', 'y']
ax1.set_xticks(x)

# Loop over ticklines and set color
for tickline, color in zip(ax1.get_xticklines(), colors):
    tickline.set_color(color)

plt.show()
