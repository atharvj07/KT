import java.io.*;
import java.nio.file.*;
import org.apache.commons.compress.archivers.ArchiveEntry;
import org.apache.commons.compress.archivers.zip.ZipArchiveInputStream;
import org.apache.commons.io.IOUtils;

public class SecureZipExtractor {

    private static final long MAX_UNCOMPRESSED_SIZE = 100 * 1024 * 1024; // 100 MB limit
    private static final int MAX_ENTRIES = 1000; // Limit total entries to 1000

    public void unzip(String srcZipFile, String destFolder) throws IOException {
        Path filePath = Paths.get(srcZipFile);
        long totalExtractedSize = 0; // Track total decompressed size
        int entryCount = 0; // Track total entries

        try (InputStream inputStream = Files.newInputStream(filePath);
             ZipArchiveInputStream zipInputStream = new ZipArchiveInputStream(inputStream)) {

            System.out.println("Begin extracting...");

            ArchiveEntry entry;
            while ((entry = zipInputStream.getNextEntry()) != null) {
                if (!zipInputStream.canReadEntryData(entry)) {
                    System.out.println("Skipping unreadable entry: " + entry.getName());
                    continue;
                }

                entryCount++;
                if (entryCount > MAX_ENTRIES) {
                    throw new IOException("Zip file contains too many entries. Possible Zip Bomb attack!");
                }

                Path path = Paths.get(destFolder, entry.getName());
                File file = path.toFile();

                if (entry.isDirectory()) {
                    if (!file.isDirectory() && !file.mkdirs()) {
                        throw new IOException("Failed to create directory: " + file);
                    }
                } else {
                    File parent = file.getParentFile();
                    if (!parent.isDirectory() && !parent.mkdirs()) {
                        throw new IOException("Failed to create parent directory: " + parent);
                    }

                    try (OutputStream outputStream = Files.newOutputStream(file.toPath())) {
                        long bytesCopied = IOUtils.copyLarge(zipInputStream, outputStream);
                        totalExtractedSize += bytesCopied;

                        if (totalExtractedSize > MAX_UNCOMPRESSED_SIZE) {
                            throw new IOException("Decompressed data exceeds safe limit. Possible Zip Bomb attack!");
                        }
                    }
                }
            }
            System.out.println("Extraction completed successfully.");
        }
    }
}
