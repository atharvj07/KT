import pandas as pd

# Example DataFrame
df = pd.DataFrame({
    'col1': [1, 2, 3, 4],
    'col2': [2, 3, 3, 5],
    'col3': [3, 1, 5, 2],
    'col4': [4, 5, 6, 3]
})

# Add a new column with the length of the DataFrame
df['length'] = len(df)

# Drop rows based on conditions
df = df[~(((df['col1'] == df['col2']) & (df['col4'] == df['col3'])) | 
          ((df['col1'] == df['col3']) & (df['col4'] == df['col2'])))]

print(df)
