import java.time.*;
import java.time.temporal.ChronoUnit;
import java.util.Random;
import java.util.Set;

import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;

public class Reminder extends ListenerAdapter {
    @Override
    public void onReady(ReadyEvent event) {
        final ZoneId zone = ZoneId.systemDefault();
        final ZoneId realzone = ZoneId.of(zone.getId());
        ZonedDateTime zdt = ZonedDateTime.now(realzone);
        String day = String.valueOf(zdt.getDayOfWeek());
        String targetday = String.valueOf(DayOfWeek.WEDNESDAY);

        // Target time is the next Wednesday at 8 AM
        ZonedDateTime targettime = zdt.withHour(8).withMinute(0).withSecond(0).withNano(0);

        // Adjust if it's already past the target time this week
        if (zdt.isAfter(targettime)) {
            targettime = targettime.plusWeeks(1);  // Move to the next week's Wednesday at 8 AM
        }

        // Calculate initial delay (time between now and the target time)
        long initialDelay = zdt.until(targettime, ChronoUnit.MILLIS);
        
        // Set a fixed weekly interval (7 days) in milliseconds
        final long period = TimeUnit.DAYS.toMillis(7);

        // For JDA (Java Discord API), your bot channel ID and text channel
        JDA bot = event.getJDA();
        final long realChannelID = fakeID; // Replace with your actual channel ID
        TextChannel textChannel = event.getJDA().getTextChannelById(realChannelID);

        System.out.println("Initial delay: " + initialDelay);

        ScheduledExecutorService scheduler = Executors.newScheduledThreadPool(1);

        Runnable WeeklyMessageTask = () -> {
            System.out.println("Starting task");

            // Send message on Wednesday at 8 AM
            if (day.equals(targetday)) {
                System.out.println("Sending weekly message...");
                switch (randomint()) {
                    case 1 -> textChannel.sendMessage("Tiamat is Titillated... Who is on board for putting an end to that?").queue();
                    case 2 -> textChannel.sendMessage("It is Wednesday and the War rages... Who will join the fight?").queue();
                    case 3 -> textChannel.sendMessage("Head count for tomorrow everyone?").queue();
                }
            } else {
                System.out.println("NOT sending weekly message...");
            }

            System.out.println("Ending task");
        };

        // Schedule the task to run at the correct time every Wednesday at 8 AM
        scheduler.scheduleAtFixedRate(WeeklyMessageTask, initialDelay, period, TimeUnit.MILLISECONDS);
    }

    // Helper function to generate a random number between 1 and 3
    public int randomint() {
        Random rand = new Random();
        int max = 3, min = 1;
        return rand.nextInt(max - min + 1) + min;
    }
}
