import javax.xml.transform.Source;
import javax.xml.transform.TransformerFactory;
import javax.xml.transform.URIResolver;
import javax.xml.transform.stream.StreamSource;
import java.io.InputStream;

public class ClasspathURIResolver implements URIResolver {
    @Override
    public Source resolve(String href, String base) {
        if (href.startsWith("file:/xsl/")) {
            String classpathResource = href.replace("file:/xsl/", "/xsl/");
            InputStream is = getClass().getResourceAsStream(classpathResource);
            if (is != null) {
                return new StreamSource(is, href);
            }
        }
        return null; // Let default resolution occur
    }

    public static TransformerFactory createTransformerFactoryWithResolver() {
        TransformerFactory factory = TransformerFactory.newInstance();
        factory.setURIResolver(new ClasspathURIResolver());
        return factory;
    }
}

public class SaxonTransformer {
    public static void main(String[] args) throws SaxonApiException {
        Processor processor = new Processor(false);
        processor.getUnderlyingConfiguration().setURIResolver(new CatalogResolver());

        XsltCompiler compiler = processor.newXsltCompiler();
        InputStream is = SaxonTransformer.class.getResourceAsStream("/xsl/juno-driver.xsl");
        XsltExecutable stylesheet = compiler.compile(new StreamSource(is, "file:/xsl/juno-driver.xsl"));

        XsltTransformer transformer = stylesheet.load();
        transformer.transform(new StreamSource("input.xml"), new Serializer(System.out));
    }
}
