public List<Employee> findAllEmployeesWithFilter() {
    Session session = entityManager.unwrap(Session.class);
    session.enableFilter("testFilter")
           .setParameter("companyId", securityAspect.getCompanyId());
    
    List<Employee> employees = session.createQuery("FROM Employee", Employee.class).getResultList();
    
    session.disableFilter("testFilter"); // Disable after use
    return employees;
}

@Bean
public FilterRegistrationBean<OpenEntityManagerInViewFilter> openSessionInViewFilter() {
    FilterRegistrationBean<OpenEntityManagerInViewFilter> filterRegBean = new FilterRegistrationBean<>();
    filterRegBean.setFilter(new OpenEntityManagerInViewFilter());
    return filterRegBean;
}

    private final EmployeeRepository repository;

    @Autowired
    public EmployeeApiServiceImpl(EmployeeRepository repository) {
        this.repository = repository;
    }

    @Transactional(readOnly = true)
    public Response listProfiles(SecurityContext securityContext) {
        List<Employee> employees = repository.findAll();
        return Response.ok().entity(employees).build();
    }
}
