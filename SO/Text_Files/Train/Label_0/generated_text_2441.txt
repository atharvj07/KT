def compare_lists(listA, listB):
    last_index = -1
    for item in listA:
        try:
            last_index = listB.index(item, last_index + 1)
        except ValueError:
            return False, f"Item '{item}' not found or out of order in listB"
    return True, None

# Test cases
listA = ['A', 'B', 'C', 'D', 'E']
listB = [':', 'A', '*', 'B', 'C', 'D', 'E', '`']
result, message = compare_lists(listA, listB)
print(result, message)

listA = ['A', 'B', 'C', 'D', 'E']
listB = ['A', '*', 'C', 'B', 'C', 'D', 'E']
result, message = compare_lists(listA, listB)
print(result, message)

listA = ['A', 'B', 'C', 'D', 'E']
listB = ['A', 'B', 'C', 'D', 'F']
result, message = compare_lists(listA, listB)
print(result, message)

listB = ['A', 'B', 'C', 'D']
result, message = compare_lists(listA, listB)
print(result, message)
