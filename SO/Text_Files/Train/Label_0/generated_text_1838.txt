import polars as pl

# Create the dataframe
df = pl.DataFrame(
    {
        "prefix_a": [1, 2, 3],
        "prefix_b": [1, 2, 3],
        "some_column": [3, 2, 1]
    }
)

# Select columns that start with the prefix "prefix_"
prefix = "prefix_"
filtered_df = df.select(pl.col("^" + prefix))

# Display the filtered dataframe
print(filtered_df)
