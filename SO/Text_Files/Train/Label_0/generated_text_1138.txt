import java.lang.invoke.*;
import jdk.incubator.foreign.*;

public class VarHandleExample {
    public static void main(String[] args) {
        // Create a MemorySegment, allocating space off-heap
        MemorySegment segment = MemorySegment.allocateNative(100); // 100 bytes

        // Get a VarHandle for a specific offset in the segment (e.g., long at position 0)
        VarHandle vh = MethodHandles.lookup().findStatic(
            VarHandleExample.class, "getLong", MethodType.methodType(long.class, MemorySegment.class, long.class));

        // Perform CAS (compare-and-swap) on the memory segment
        long expected = 0L;
        long newValue = 42L;
        boolean success = (boolean) vh.compareAndSet(segment, 0L, expected, newValue);
        System.out.println("CAS success: " + success);
    }

    public static long getLong(MemorySegment segment, long offset) {
        return segment.getLong(offset);
    }

    public static void setLong(MemorySegment segment, long offset, long value) {
        segment.setLong(offset, value);
    }
}
