private class IteratorForLinkedList implements Iterator<T> {
    private Node nextNode = head;

    @Override
    public boolean hasNext() {
        return nextNode != null;
    }

    @Override
    public T next() {
        if (!hasNext()) {
            throw new NoSuchElementException();
        }
        T data = nextNode.data;
        nextNode = nextNode.next;
        return data;
    }
}
