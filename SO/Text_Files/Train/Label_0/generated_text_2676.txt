import polars as pl
import numpy as np

# Sample DataFrame
test_data = pl.DataFrame(
    {
        "group": ["A", "A", "B", "B"],
        "values": [1, np.nan, 2, 3]
    }
)

# Group by and calculate mean while ignoring NaNs
result = (
    test_data.group_by("group")
    .agg(pl.col("values").drop_nans().mean())
)

print(result)
