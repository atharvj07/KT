public class Customer {
    private String customerName;
    private String customerAddr;
    private String currency;

    // Default constructor (required for Jackson)
    public Customer() {}

    // Parameterized constructor
    public Customer(String customerName, String customerAddr, String currency) {
        this.customerName = customerName;
        this.customerAddr = customerAddr;
        this.currency = currency;
    }

    // Getters and Setters
    public String getCustomerName() { return customerName; }
    public void setCustomerName(String customerName) { this.customerName = customerName; }

    public String getCustomerAddr() { return customerAddr; }
    public void setCustomerAddr(String customerAddr) { this.customerAddr = customerAddr; }

    public String getCurrency() { return currency; }
    public void setCurrency(String currency) { this.currency = currency; }

    @Override
    public String toString() {
        return "Customer{" +
                "customerName='" + customerName + '\'' +
                ", customerAddr='" + customerAddr + '\'' +
                ", currency='" + currency + '\'' +
                '}';
    }
}

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.io.IOException;
import java.io.InputStream;

public class JsonDeserializer {

    public static Customer deserializeJson(String fileName, String environment) throws IOException {
        // Load the JSON file from resources
        InputStream is = JsonDeserializer.class.getClassLoader().getResourceAsStream(fileName);
        if (is == null) {
            throw new IOException("File not found: " + fileName);
        }

        // Create Jackson ObjectMapper
        ObjectMapper objectMapper = new ObjectMapper();

        // Read the whole JSON as a tree (JsonNode)
        JsonNode rootNode = objectMapper.readTree(is);

        // Get the specific environment node (QA, STAGE)
        JsonNode envNode = rootNode.get(environment.toUpperCase()); // Ensure case insensitivity

        if (envNode == null) {
            throw new IllegalArgumentException("Invalid environment: " + environment);
        }

        // Deserialize JSON into Customer object
        return objectMapper.treeToValue(envNode, Customer.class);
    }

    public static void main(String[] args) {
        try {
            Customer customer = deserializeJson("data.json", "qa");
            System.out.println(customer);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
