private Single<BigInteger> createNetworkCall() {
    return Single.fromCallable(() -> 
        service.getBalance("validaddress").execute()
    ).subscribeOn(Schedulers.io())
     .onErrorReturn(throwable -> { // Handle errors gracefully
         Log.e(TAG, "createNetworkCall failed", throwable);
         return BigInteger.ZERO; // Default value for failed calls
     });
}

private Single<BigInteger> createFailedNetworkCall() {
    return Single.fromCallable(() -> 
        service.getBalance("invalidaddress").execute()
    ).subscribeOn(Schedulers.io())
     .onErrorReturn(throwable -> { // Handle errors gracefully
         Log.e(TAG, "createFailedNetworkCall failed", throwable);
         return BigInteger.ZERO; // Default value for failed calls
     });
}

private void makeParallelCalls() {
    List<Single<BigInteger>> iterable = new ArrayList<>();
    iterable.add(createNetworkCall());
    iterable.add(createNetworkCall());
    iterable.add(createNetworkCall());
    iterable.add(createNetworkCall());
    iterable.add(createFailedNetworkCall()); // Will now return BigInteger.ZERO instead of failing

    Single.zip(iterable, results -> {
        Log.d(TAG, "makeParallelCalls: " + Arrays.toString(results));
        return results;
    }).observeOn(AndroidSchedulers.mainThread())
      .subscribe(results -> {
          Log.d(TAG, "onSuccess: makeParallelCalls: " + Arrays.toString(results));
      }, exception -> {
          Log.e(TAG, "onError: makeParallelCalls", exception);
      });
}

private Single<Optional<BigInteger>> createSafeNetworkCall(String address) {
    return Single.fromCallable(() -> 
        Optional.of(service.getBalance(address).execute())
    ).subscribeOn(Schedulers.io())
     .onErrorReturnItem(Optional.empty()); // Return empty Optional on failure
}

private void makeParallelCalls() {
    List<Single<Optional<BigInteger>>> iterable = new ArrayList<>();
    iterable.add(createSafeNetworkCall("validaddress1"));
    iterable.add(createSafeNetworkCall("validaddress2"));
    iterable.add(createSafeNetworkCall("validaddress3"));
    iterable.add(createSafeNetworkCall("validaddress4"));
    iterable.add(createSafeNetworkCall("invalidaddress")); // Will return Optional.empty()

    Single.zip(iterable, results -> {
        // Convert Optional<BigInteger>[] to List<BigInteger>, filtering out empty optionals
        return Arrays.stream(results)
                     .map(result -> (Optional<BigInteger>) result)
                     .filter(Optional::isPresent)
                     .map(Optional::get)
                     .collect(Collectors.toList());
    }).observeOn(AndroidSchedulers.mainThread())
      .subscribe(results -> {
          Log.d(TAG, "onSuccess: makeParallelCalls: " + results);
      }, exception -> {
          Log.e(TAG, "onError: makeParallelCalls", exception);
      });
}
