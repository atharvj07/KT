package org.example.dto;

import io.micronaut.data.model.Page;
import io.micronaut.serde.annotation.Serdeable;
import java.util.List;

@Serdeable
public record PagedResponse<T>(
    List<T> content,
    int pageNumber,
    int totalPages,
    long totalSize,
    int size,
    int numberOfElements
) {
    public static <T> PagedResponse<T> from(Page<T> page) {
        return new PagedResponse<>(
            page.getContent(),
            page.getPageNumber(),
            page.getTotalPages(),
            page.getTotalSize(),
            page.getSize(),
            page.getNumberOfElements()
        );
    }
}

package org.example.controller;

import io.micronaut.data.model.Pageable;
import io.micronaut.http.annotation.Controller;
import io.micronaut.http.annotation.Get;
import io.swagger.v3.oas.annotations.Parameter;
import lombok.RequiredArgsConstructor;
import org.example.dto.GroupDto;
import org.example.dto.PagedResponse;
import org.example.service.GroupService;

@Controller("/groups")
@RequiredArgsConstructor
public class GroupController {
    private final GroupService groupService;

    @Get
    public PagedResponse<GroupDto> findAllGroups(@Parameter(hidden = true) Pageable pageable) {
        return PagedResponse.from(groupService.findAllGroups(pageable));
    }
}

package org.example.serializer;

import io.micronaut.data.model.Page;
import io.micronaut.serde.Encoder;
import io.micronaut.serde.Serializer;
import jakarta.inject.Singleton;
import java.io.IOException;

@Singleton
public class PageSerializer<T> implements Serializer<Page<T>> {
    @Override
    public void serialize(Encoder encoder, Serializer.EncoderContext context, Page<T> page) throws IOException {
        Encoder objectEncoder = encoder.encodeObject();
        objectEncoder.encodeKey("content").encodeArray(page.getContent(), context);
        objectEncoder.encodeKey("pageNumber").encodeInt(page.getPageNumber());
        objectEncoder.encodeKey("totalPages").encodeInt(page.getTotalPages());
        objectEncoder.encodeKey("totalSize").encodeLong(page.getTotalSize());
        objectEncoder.encodeKey("size").encodeInt(page.getSize());
        objectEncoder.encodeKey("numberOfElements").encodeInt(page.getNumberOfElements());
        objectEncoder.finishStructure();
    }
}
