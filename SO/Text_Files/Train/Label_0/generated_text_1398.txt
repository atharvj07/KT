import java.net.URI;
import java.net.http.HttpHeaders;
import java.net.http.HttpRequest;
import java.util.Map;
import java.util.Optional;

public class CurlGenerator {

    public static String generateCurlCommand(HttpRequest request) {
        StringBuilder curlCommand = new StringBuilder("curl");

        // Get HTTP method (GET, POST, etc.)
        String method = request.method();
        curlCommand.append(" -X ").append(method);

        // Append headers
        HttpHeaders headers = request.headers();
        for (Map.Entry<String, java.util.List<String>> header : headers.map().entrySet()) {
            for (String value : header.getValue()) {
                curlCommand.append(" -H \"").append(header.getKey()).append(": ").append(value).append("\"");
            }
        }

        // Append URI
        URI uri = request.uri();
        curlCommand.append(" ").append(uri.toString());

        // Handle the body if it exists
        Optional<HttpRequest.BodyPublisher> bodyPublisher = request.bodyPublisher();
        if (bodyPublisher.isPresent()) {
            String bodyContent = extractBodyContent(bodyPublisher.get());
            if (bodyContent != null) {
                // For POST/PUT requests with a body
                curlCommand.append(" -d \"").append(bodyContent).append("\"");
            }
        }

        return curlCommand.toString();
    }

    private static String extractBodyContent(HttpRequest.BodyPublisher bodyPublisher) {
        // This method should attempt to extract the actual body content from BodyPublisher
        // For example, if it's a string body, you might want to read it from the bodyPublisher
        // This will depend on your specific use case.
        
        // Here we just return a simple representation, but you may want to implement specific cases
        if (bodyPublisher instanceof HttpRequest.BodyPublishers.OfString) {
            try {
                return ((HttpRequest.BodyPublishers.OfString) bodyPublisher).get().toString();
            } catch (Exception e) {
                e.printStackTrace();
            }
        }
        return null;
    }

    public static void main(String[] args) {
        // Example: Generate curl command from an HttpRequest object
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create("https://api.example.com/data"))
                .header("Authorization", "Bearer token")
                .header("Content-Type", "application/json")
                .POST(HttpRequest.BodyPublishers.ofString("{\"key\":\"value\"}"))
                .build();

        String curl = generateCurlCommand(request);
        System.out.println(curl);
    }
}
