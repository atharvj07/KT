import javax.swing.*;
import java.awt.*;

public class JButtonHTMLExample {
    public static void main(String[] args) {
        JFrame frame = new JFrame("Keyboard Key Simulation");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setSize(300, 200);
        
        JPanel panel = new JPanel();
        panel.setLayout(new FlowLayout());

        // Using HTML inside JButton
        String htmlContent = "<html><div style='position: relative; width: 50px; height: 50px;'>"
                + "<div style='position: absolute; right: 3px; top: 3px; font-size: 10px;'>10</div>"
                + "<div style='position: absolute; left: 50%; top: 50%; transform: translate(-50%, -50%); font-size: 16px;'>A</div>"
                + "</div></html>";

        JButton keyButton = new JButton(htmlContent);
        keyButton.setPreferredSize(new Dimension(50, 50));
        keyButton.setFocusPainted(false);
        keyButton.setBackground(Color.LIGHT_GRAY);
        panel.add(keyButton);

        frame.add(panel);
        frame.setVisible(true);
    }
}

import javax.swing.*;
import java.awt.*;

public class JLabelInsideJButtonExample {
    public static void main(String[] args) {
        JFrame frame = new JFrame("Keyboard Key Simulation");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setSize(300, 200);
        
        JPanel panel = new JPanel();
        panel.setLayout(new FlowLayout());

        JButton button = new JButton();
        button.setPreferredSize(new Dimension(50, 50));
        button.setLayout(null); // Absolute positioning inside the button

        JLabel topRightLabel = new JLabel("10");
        topRightLabel.setFont(new Font("Arial", Font.PLAIN, 10));
        topRightLabel.setBounds(35, 5, 15, 15);  // Top-right position

        JLabel centerLabel = new JLabel("A", SwingConstants.CENTER);
        centerLabel.setFont(new Font("Arial", Font.BOLD, 16));
        centerLabel.setBounds(10, 15, 30, 30);  // Centered

        button.add(topRightLabel);
        button.add(centerLabel);

        button.setBackground(Color.LIGHT_GRAY);
        button.setBorderPainted(true);
        button.setFocusPainted(false);

        panel.add(button);

        frame.add(panel);
        frame.setVisible(true);
    }
}

import javax.swing.*;
import java.awt.*;

public class JLayeredPaneExample {
    public static void main(String[] args) {
        JFrame frame = new JFrame("Keyboard Key Simulation");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setSize(300, 200);
        
        JPanel panel = new JPanel();
        panel.setLayout(new FlowLayout());

        JLayeredPane layeredPane = new JLayeredPane();
        layeredPane.setPreferredSize(new Dimension(50, 50));

        JButton button = new JButton();
        button.setBounds(0, 0, 50, 50);
        button.setBackground(Color.LIGHT_GRAY);
        button.setFocusPainted(false);
        button.setBorderPainted(true);

        JLabel numberLabel = new JLabel("10");
        numberLabel.setFont(new Font("Arial", Font.PLAIN, 10));
        numberLabel.setBounds(35, 5, 15, 15);

        JLabel letterLabel = new JLabel("A", SwingConstants.CENTER);
        letterLabel.setFont(new Font("Arial", Font.BOLD, 16));
        letterLabel.setBounds(10, 15, 30, 30);

        layeredPane.add(button, JLayeredPane.DEFAULT_LAYER);
        layeredPane.add(numberLabel, JLayeredPane.PALETTE_LAYER);
        layeredPane.add(letterLabel, JLayeredPane.MODAL_LAYER);

        panel.add(layeredPane);

        frame.add(panel);
        frame.setVisible(true);
    }
}
