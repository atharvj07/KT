String[] instructions = MainActivity.sqLiteDB.getPreparationInstructions(item);
int numberOfCurrentInstruction = 1;
ViewGroup main = getView().findViewById(R.id.main);
LayoutInflater inflater = getLayoutInflater();

for (int i = 0; i < instructions.length; i++) {
    if (instructions[i] == null) {
        continue;
    }

    // Inflate the layout for each instruction
    View customLayout = inflater.inflate(R.layout.custom, main, false);

    // Get the TextViews from the inflated layout
    TextView num = customLayout.findViewById(R.id.instruction_number);
    TextView text = customLayout.findViewById(R.id.instruction_text);

    // Set the number and text for this instruction
    num.setText(numberOfCurrentInstruction + ".");
    text.setText(instructions[i]);

    // Add the inflated layout to the parent view
    main.addView(customLayout);

    // Increment the instruction number
    numberOfCurrentInstruction++;
}
