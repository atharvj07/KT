@Configuration
public class DataSourceConfig {

    @Bean(name = "dataSource1")
    @ConfigurationProperties(prefix = "spring.datasource.first")
    public DataSource dataSource1() {
        return DataSourceBuilder.create().build();
    }

    @Bean(name = "dataSource2")
    @ConfigurationProperties(prefix = "spring.datasource.second")
    public DataSource dataSource2() {
        return DataSourceBuilder.create().build();
    }
}

@Configuration
public class JdbcMappingContextConfig {

    @Bean(name = "jdbcMappingContext1")
    public JdbcMappingContext jdbcMappingContext1() {
        return new JdbcMappingContext("first");
    }

    @Bean(name = "jdbcMappingContext2")
    public JdbcMappingContext jdbcMappingContext2() {
        return new JdbcMappingContext("second");
    }
}

@Configuration
public class JdbcTemplateConfig {

    @Bean(name = "jdbcTemplate1")
    public JdbcTemplate jdbcTemplate1(@Qualifier("dataSource1") DataSource dataSource1,
                                      @Qualifier("jdbcMappingContext1") JdbcMappingContext jdbcMappingContext1) {
        return new JdbcTemplate(dataSource1, jdbcMappingContext1);
    }

    @Bean(name = "jdbcTemplate2")
    public JdbcTemplate jdbcTemplate2(@Qualifier("dataSource2") DataSource dataSource2,
                                      @Qualifier("jdbcMappingContext2") JdbcMappingContext jdbcMappingContext2) {
        return new JdbcTemplate(dataSource2, jdbcMappingContext2);
    }
}

@Configuration
public class TransactionManagerConfig {

    @Bean(name = "transactionManager1")
    public PlatformTransactionManager transactionManager1(@Qualifier("dataSource1") DataSource dataSource1) {
        return new DataSourceTransactionManager(dataSource1);
    }

    @Bean(name = "transactionManager2")
    public PlatformTransactionManager transactionManager2(@Qualifier("dataSource2") DataSource dataSource2) {
        return new DataSourceTransactionManager(dataSource2);
    }
}

    @Autowired

    @Autowired
