import java.awt.*;
import java.awt.image.*;
import javax.swing.*;

public class ImageBlurExample {

    public static void main(String[] args) {
        // Load the image (replace with your image path)
        try {
            ImageIcon imageIcon = new ImageIcon("path/to/your/image.jpg");
            Image image = imageIcon.getImage();

            // Convert Image to BufferedImage
            BufferedImage bufferedImage = new BufferedImage(
                    image.getWidth(null), image.getHeight(null), BufferedImage.TYPE_INT_ARGB);
            Graphics g = bufferedImage.createGraphics();
            g.drawImage(image, 0, 0, null);
            g.dispose();

            // Apply the Gaussian blur
            BufferedImage blurredImage = applyGaussianBlur(bufferedImage);

            // Display the original and blurred images
            JFrame frame = new JFrame();
            frame.setLayout(new GridLayout(1, 2));

            // Original image
            JLabel originalLabel = new JLabel(new ImageIcon(bufferedImage));
            frame.add(originalLabel);

            // Blurred image
            JLabel blurredLabel = new JLabel(new ImageIcon(blurredImage));
            frame.add(blurredLabel);

            frame.setSize(800, 400);
            frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
            frame.setVisible(true);

        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public static BufferedImage applyGaussianBlur(BufferedImage image) {
        // 5x5 Gaussian Blur Kernel
        float[] kernel = {
            1f/16f, 2f/16f, 3f/16f, 2f/16f, 1f/16f,
            2f/16f, 4f/16f, 6f/16f, 4f/16f, 2f/16f,
            3f/16f, 6f/16f, 9f/16f, 6f/16f, 3f/16f,
            2f/16f, 4f/16f, 6f/16f, 4f/16f, 2f/16f,
            1f/16f, 2f/16f, 3f/16f, 2f/16f, 1f/16f
        };
        
        Kernel gaussianKernel = new Kernel(5, 5, kernel);
        ConvolveOp convolve = new ConvolveOp(gaussianKernel, ConvolveOp.EDGE_NO_OP, null);

        // Apply convolution
        return convolve.filter(image, null);
    }
}

    };
    
    Kernel gaussianKernel = new Kernel(5, 5, kernel);
