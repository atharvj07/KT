import java.util.concurrent.ConcurrentHashMap;

public class CarService {
    private final ConcurrentHashMap<String, Object> locks = new ConcurrentHashMap<>();

    private Object getLock(Car car) {
        return locks.computeIfAbsent(car.partition, key -> new Object());
    }

    private void doSomething(Car car) {
        Object obj = getLock(car);
        synchronized (obj) {
            try {
                // Critical section - Only cars with the same partition are synchronized
                System.out.println("Processing car with partition: " + car.partition);
                Thread.sleep(1000); // Simulate work
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            } finally {
                // Cleanup: Remove the lock if no other thread is using it
                locks.compute(car.partition, (key, lock) -> 
                    Thread.holdsLock(lock) ? lock : null);
            }
        }
    }

    public static void main(String[] args) {
        CarService service = new CarService();
        Car car1 = new Car("A");
        Car car2 = new Car("A");
        Car car3 = new Car("B");

        new Thread(() -> service.doSomething(car1)).start();
        new Thread(() -> service.doSomething(car2)).start();
        new Thread(() -> service.doSomething(car3)).start();
    }
}
