import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.web.client.RestTemplate;

public class GitHubRepoFetcher {
    public static void main(String[] args) {
        try {
            RestTemplate restTemplate = new RestTemplate();
            String resourceUrl = "https://api.github.com/orgs/engineyard/repos"; // GitHub API URL
            String response = restTemplate.getForObject(resourceUrl, String.class); // Get response as string

            ObjectMapper mapper = new ObjectMapper(); // Mapper to parse JSON
            JsonNode jsonTree = mapper.readTree(response); // Parse response into JsonNode

            // Loop through the array of repos and extract 'name' and 'forks'
            for (JsonNode repo : jsonTree) {
                String name = repo.get("name").asText(); // Get 'name' field as text
                int forks = repo.get("forks").asInt(); // Get 'forks' field as int
                System.out.println("Repository: " + name + ", Forks: " + forks); // Print name and forks
            }

        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
