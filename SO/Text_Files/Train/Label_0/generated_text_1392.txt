import graphql.GraphQL;
import graphql.schema.GraphQLSchema;
import graphql.GraphQLException;
import graphql.ExecutionInput;
import graphql.ExecutionResult;

import java.util.LinkedHashMap;
import java.util.Map;

public class GraphQLExample {
    public static void main(String[] args) throws GraphQLException {
        // Define your GraphQL schema and query
        GraphQLSchema graphQLSchema = getGraphQLSchema();
        String graphqlQuery = "{ ResponseDTO { id description } }";

        // Create the execution input
        ExecutionInput builder = ExecutionInput
                .newExecutionInput()
                .query(graphqlQuery)
                .build();

        // Execute the query
        ExecutionResult executionResult = GraphQL
                .newGraphQL(graphQLSchema)
                .build()
                .execute(builder);

        // Get the data from the execution result (this is the LinkedHashMap)
        LinkedHashMap<String, Object> graphQLData = (LinkedHashMap<String, Object>) executionResult.getData();

        // Convert LinkedHashMap to ResponseDTO
        ResponseDTO responseDTO = mapToDTO(graphQLData);

        // Print or use the ResponseDTO as needed
        System.out.println(responseDTO);
    }

    private static GraphQLSchema getGraphQLSchema() {
        // Placeholder: Create and return your actual GraphQL schema here
        return null;
    }

    private static ResponseDTO mapToDTO(LinkedHashMap<String, Object> graphQLData) {
        // Extract the map for ResponseDTO from the GraphQL data
        LinkedHashMap<String, Object> responseData = (LinkedHashMap<String, Object>) graphQLData.get("ResponseDTO");

        // Now manually extract the values for id and description from the response data
        Integer id = (Integer) responseData.get("id");
        String description = (String) responseData.get("description");

        // Create and return the ResponseDTO
        ResponseDTO responseDTO = new ResponseDTO();
        responseDTO.setId(id);
        responseDTO.setDescription(description);

        return responseDTO;
    }
}

public class ResponseDTO {
    private Integer id;
    private String description;

    // Getters and setters
    public Integer getId() {
        return id;
    }

    public void setId(Integer id) {
        this.id = id;
    }

    public String getDescription() {
        return description;
    }

    public void setDescription(String description) {
        this.description = description;
    }

    @Override
    public String toString() {
        return "ResponseDTO{" +
                "id=" + id +
                ", description='" + description + '\'' +
                '}';
    }
}

import com.fasterxml.jackson.databind.ObjectMapper;

public static ResponseDTO mapToDTOUsingJackson(LinkedHashMap<String, Object> graphQLData) {
    // Create an ObjectMapper
    ObjectMapper objectMapper = new ObjectMapper();

    // Convert LinkedHashMap to ResponseDTO using Jackson
    ResponseDTO responseDTO = objectMapper.convertValue(graphQLData.get("ResponseDTO"), ResponseDTO.class);

    return responseDTO;
