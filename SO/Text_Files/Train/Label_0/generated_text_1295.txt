import com.google.gson.*;
import java.lang.reflect.Field;
import java.nio.ByteBuffer;
import java.util.Base64;

public class ByteBufferTypeAdapter implements JsonDeserializer<ByteBuffer>, JsonSerializer<ByteBuffer> {

    @Override
    public JsonElement serialize(ByteBuffer src, Type typeOfSrc, JsonSerializationContext context) {
        try {
            // Access the private 'hb' field (byte array) of the ByteBuffer
            Field hbField = ByteBuffer.class.getDeclaredField("hb");
            hbField.setAccessible(true);
            byte[] byteArray = (byte[]) hbField.get(src);

            // Encode the byte array to a Base64 string
            return new JsonPrimitive(Base64.getEncoder().encodeToString(byteArray));
        } catch (NoSuchFieldException | IllegalAccessException e) {
            throw new JsonParseException("Error accessing ByteBuffer internals", e);
        }
    }

    @Override
    public ByteBuffer deserialize(JsonElement json, Type typeOfT, JsonDeserializationContext context) throws JsonParseException {
        try {
            // Decode the Base64 string into a byte array
            byte[] bytes = Base64.getDecoder().decode(json.getAsString());
            return ByteBuffer.wrap(bytes); // Wrap the byte array in a ByteBuffer
        } catch (Exception e) {
            throw new JsonParseException("Error decoding ByteBuffer", e);
        }
    }
}

import com.google.gson.Gson;
import com.google.gson.GsonBuilder;

public class Main {
    public static void main(String[] args) {
        Gson gson = new GsonBuilder()
                .registerTypeAdapter(ByteBuffer.class, new ByteBufferTypeAdapter())
                .create();

        // Create a ByteBuffer instance for testing
        ByteBuffer test = ByteBuffer.wrap("Hello".getBytes());

        // Serialize the ByteBuffer to JSON
        String json = gson.toJson(test);
        System.out.println(json);

        // Deserialize the JSON back into a ByteBuffer
        ByteBuffer deserialized = gson.fromJson(json, ByteBuffer.class);
        System.out.println(new String(deserialized.array()));
    }
}
