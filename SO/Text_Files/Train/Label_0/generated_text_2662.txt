import fitz  # PyMuPDF

def pdf_to_images(pdf_path):
    doc = fitz.open(pdf_path)
    images = []
    for page in doc:
        pix = page.get_pixmap(dpi=300)  # Higher DPI for better quality
        img = pix.pil_tobytes(format="jpeg")  # Convert to JPEG bytes
        images.append(img)
    return images
