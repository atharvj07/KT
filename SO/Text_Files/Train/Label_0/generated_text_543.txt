class Solution {
    public static int binaryToDecimal(String str) { 
        int res = 0;
        int power = 0;
        for (int i = str.length() - 1; i >= 0; i--) {
            if (str.charAt(i) == '1') {
                res += Math.pow(2, power);
            }
            power++;
        }
        return res;
    }
}
