public void count(BM... bm) {
    int countSM = 0;
    int countKM = 0;

    System.out.println("Total objects: " + bm.length);

    for (BM b : bm) {
        if (b instanceof SM) {
            System.out.println("Instance of SM");
            countSM++;
        } else if (b instanceof KM) {
            System.out.println("Instance of KM");
            countKM++;
        }
    }

    System.out.println("Total SM instances: " + countSM);
    System.out.println("Total KM instances: " + countKM);
}

BM obj1 = new SM();
BM obj2 = new KM();
BM obj3 = new SM();

count(obj1, obj2, obj3);
