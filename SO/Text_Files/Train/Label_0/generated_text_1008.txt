import java.io.FileOutputStream;
import org.apache.poi.xslf.usermodel.*;

public class NumberedListPPTX {
    public static void main(String[] args) throws Exception {
        XMLSlideShow ppt = new XMLSlideShow();
        XSLFSlide slide = ppt.createSlide();
        
        // Create a text box
        XSLFTextBox textBox = slide.createTextBox();
        textBox.setAnchor(new java.awt.Rectangle(100, 100, 400, 300));

        // Add numbered list items
        addNumberedParagraph(textBox, "First Item", 1);
        addNumberedParagraph(textBox, "Second Item", 2);
        addNumberedParagraph(textBox, "Third Item", 3);

        // Save the PowerPoint file
        FileOutputStream out = new FileOutputStream("NumberedList.pptx");
        ppt.write(out);
        out.close();
        ppt.close();

        System.out.println("PowerPoint with numbered list created successfully!");
    }

    private static void addNumberedParagraph(XSLFTextShape textShape, String text, int bulletNumber) {
        XSLFTextParagraph paragraph = textShape.addNewTextParagraph();
        paragraph.setTextAlign(TextAlign.LEFT);
        paragraph.setBullet(true);  // Enable bullet
        paragraph.setBulletStyle(bulletNumber);  // Set numbering (1, 2, 3,...)
        
        XSLFTextRun textRun = paragraph.addNewTextRun();
        textRun.setText(text);
        textRun.setFontSize(20.0);
    }
}

paragraph.setBulletStyle(XSLFTextParagraph.BulletStyle.NUMBER_PLAIN); // 1, 2, 3...
paragraph.setBulletStyle(XSLFTextParagraph.BulletStyle.LC_ROMAN); // i, ii, iii...
paragraph.setBulletStyle(XSLFTextParagraph.BulletStyle.UC_ROMAN); // I, II, III...
paragraph.setBulletStyle(XSLFTextParagraph.BulletStyle.LC_ALPHA); // a, b, c...
paragraph.setBulletStyle(XSLFTextParagraph.BulletStyle.UC_ALPHA); // A, B, C...
