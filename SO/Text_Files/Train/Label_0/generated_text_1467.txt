import java.util.Objects;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        boolean MO = true;  // Controls if we should restart the calculator
        Scanner scanner = new Scanner(System.in);

        while (MO) { // Keep running until the user exits
            MO = false; // Reset restart flag

            System.out.println("Please enter your operation");
            System.out.println("1 - Addition");
            System.out.println("2 - Subtraction");
            System.out.println("3 - Multiplication");
            System.out.println("4 - Division");

            int op = scanner.nextInt();
            scanner.nextLine();  // Consume the newline left in the buffer

            if (op == 1) {  // Addition Mode
                System.out.println("Enter a starting value!");
                int as = scanner.nextInt();
                scanner.nextLine();  // Consume newline

                boolean ab = true;
                while (ab) {
                    System.out.println("Please enter a number to add or type 'End' to end or 'Clear' to reset or 'Restart' to exit the mode");
                    String an = scanner.nextLine();

                    if (Objects.equals(an, "End")) {
                        System.out.println("Quitting Addition Mode.");
                        break;
                    } else if (Objects.equals(an, "Clear")) {
                        as = 0;
                        System.out.println("Cache cleared! Current sum: 0");
                    } else if (Objects.equals(an, "Restart")) {
                        System.out.println("Restarting the calculator!");
                        MO = true;  // Set flag to restart operation selection
                        break;  // Exit inner loop
                    } else {
                        try {
                            int ra = Integer.parseInt(an);
                            as += ra;
                            System.out.println("Current sum: " + as);
                        } catch (NumberFormatException e) {
                            System.out.println("Invalid input. Please enter a number or a valid command.");
                        }
                    }
                }
            }
        }
        scanner.close(); // Close scanner when done
    }
}
