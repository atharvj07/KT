import io.smallrye.mutiny.Uni
import org.jboss.resteasy.reactive.RestResponse
import org.jboss.resteasy.reactive.server.ServerResponse
import org.jboss.resteasy.reactive.server.spi.ResponseContext
import org.jboss.resteasy.reactive.server.spi.StreamingResponse
import org.jboss.resteasy.reactive.server.spi.StreamingContext

import javax.ws.rs.GET
import javax.ws.rs.Path
import javax.ws.rs.Produces
import javax.ws.rs.QueryParam
import javax.ws.rs.PathParam
import javax.ws.rs.HeaderParam
import javax.ws.rs.core.Response
import javax.ws.rs.core.MediaType
import java.io.InputStream

@Path("/download")
class FileDownloadResource {

    @GET
    @Path("/video/{id}/ctx/{ctx}")
    @Produces(MediaType.APPLICATION_OCTET_STREAM)
    fun downloadFile(
        @PathParam("id") fileId: String,
        @PathParam("ctx") ctx: String,
        @QueryParam("preview") preview: Boolean,
        @HeaderParam("Range") httpRangeList: String?
    ): Uni<Response> {
        return loadFileFromMinIO(fileId, ctx)
            .onItem().transform { inputStream ->
                // You can handle Range here, but for simplicity, it's not included
                Response.ok(inputStream)
                    .header("Content-Disposition", "inline; filename=\"video.mp4\"")
                    .header("Content-Type", "video/mp4")
                    .header("Accept-Ranges", "bytes")
                    .build()
            }
    }

    private fun loadFileFromMinIO(fileId: String, ctx: String): Uni<InputStream> {
        return Uni.createFrom().item {
            // Replace with your actual MinIO service call
            minioClient.getObject(GetObjectArgs.builder().bucket("files").`object`(fileId).build())
        }
    }
}

fun loadFileFromMinIOWithRange(fileId: String, ctx: String, rangeStart: Long, rangeEnd: Long): Uni<InputStream> {
    return Uni.createFrom().item {
        val getObjectArgs = GetObjectArgs.builder()
            .bucket("files")
            .`object`(fileId)
            .range(rangeStart, rangeEnd)
            .build()

        minioClient.getObject(getObjectArgs)
    }
}
