import java.util.ServiceLoader;

public interface MyService {
    void execute();
}

public class MySingletonService implements MyService {
    
    private static MySingletonService instance;

    // Private constructor to follow Singleton pattern
    private MySingletonService() {}

    // Public static method to access the Singleton instance
    public static MySingletonService getInstance() {
        if (instance == null) {
            synchronized (MySingletonService.class) {
                if (instance == null) {
                    instance = new MySingletonService();
                }
            }
        }
        return instance;
    }

    @Override
    public void execute() {
        System.out.println("Executing Singleton Service...");
    }
}

com.example.MySingletonService

import java.util.ServiceLoader;

public class MyServiceFactory {
    public static MyService getService() {
        ServiceLoader<MyService> loader = ServiceLoader.load(MyService.class);
        for (MyService service : loader) {
            return service; // Since we only have one implementation, return the first one
        }
        throw new RuntimeException("Service not found!");
    }
}

public class Main {
    public static void main(String[] args) {
        MyService service = MyServiceFactory.getService();
        service.execute();
    }
}
