import java.util.*;

public class OptimalPairsFinder {
    public static Set<Integer> findOptimalPairs(ArrayList<ArrayList<int[]>> conditions) {
        ArrayList<ArrayList<int[]>> listOfConditions = new ArrayList<>();
        generateCombinations(conditions, 0, new ArrayList<>(), listOfConditions);

        Set<Integer> best = new HashSet<>();
        for (ArrayList<int[]> pairs : listOfConditions) {
            Set<Integer> curr = new HashSet<>();
            for (int[] pair : pairs) {
                for (int num : pair) curr.add(num);
            }
            best = (best.size() > 0 && best.size() <= curr.size()) ? best : curr;
        }

        return best;
    }

    private static void generateCombinations(ArrayList<ArrayList<int[]>> conditions, int index, 
                                             ArrayList<int[]> option, 
                                             ArrayList<ArrayList<int[]>> listOfConditions) {
        if (index == conditions.size()) {
            listOfConditions.add(new ArrayList<>(option));
            return;
        }

        for (int[] pair : conditions.get(index)) {
            option.add(pair);
            generateCombinations(conditions, index + 1, option, listOfConditions);
            option.remove(option.size() - 1); // Backtrack
        }
    }

    public static void main(String[] args) {
        ArrayList<ArrayList<int[]>> conditions = new ArrayList<>();

        conditions.add(new ArrayList<>(Arrays.asList(new int[]{2, 6}, new int[]{3, 7})));
        conditions.add(new ArrayList<>(Arrays.asList(new int[]{1, 3}, new int[]{2, 4}, new int[]{2, 3})));
        conditions.add(new ArrayList<>(Arrays.asList(new int[]{1, 2})));

        Set<Integer> bestSet = findOptimalPairs(conditions);
        System.out.println("Optimal Set: " + bestSet);
    }
}

        generateCombinations(conditions, index + 1, option, listOfConditions);

    }
}
public static void main(String[] args) {
