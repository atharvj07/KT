import org.junit.jupiter.api.Test;
import java.lang.reflect.Constructor;
import static org.junit.jupiter.api.Assertions.assertThrows;

class LineBreakerTest {

    @Test
    void testLineBreakerException() throws Exception {
        // Use reflection to create an invalid enum instance
        Constructor<LineBreaker.Mode> constructor = LineBreaker.Mode.class.getDeclaredConstructor(String.class, int.class);
        constructor.setAccessible(true); // Allow access to private constructor
        LineBreaker.Mode invalidMode = constructor.newInstance("INVALID", 999);

        assertThrows(
            IllegalStateException.class,
            () -> LineBreaker.process("Dog", invalidMode, 10),
            "Expected " + IllegalStateException.class.getSimpleName()
        );
    }
}
