import pandas as pd
import numpy as np

data = {f'Column {i + 1}': np.random.randint(0, 10, size=(10, )) for i in range(5)}
df = pd.DataFrame(data)

# Create Styler object
styler = df.style

# Set index=False to remove the index column
lat_new = styler.to_latex(index=False)

print(lat_new)
