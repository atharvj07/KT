private static void maximize(Window window) {
    Rectangle screenBounds = GraphicsEnvironment.getLocalGraphicsEnvironment().getMaximumWindowBounds();
    window.setBounds(screenBounds); // Ensures it fits the usable screen
    window.setLocation(screenBounds.x, screenBounds.y); // Ensures correct positioning
}
