import java.awt.*;
import java.awt.datatransfer.*;

public class MultiFormatClipboard {
    public static void main(String[] args) {
        String plainText = "goto stack";
        String htmlText = "<a href=\"https://stackoverflow.com/\">goto stack</a>";

        Transferable multiFormatTransferable = new MultiFormatTransferable(plainText, htmlText);
        Clipboard clipboard = Toolkit.getDefaultToolkit().getSystemClipboard();
        clipboard.setContents(multiFormatTransferable, null);

        System.out.println("Clipboard set with both text/plain and text/html!");
    }

    static class MultiFormatTransferable implements Transferable {
        private final String plainText;
        private final String htmlText;

        private static final DataFlavor[] SUPPORTED_FLAVORS = {
                DataFlavor.stringFlavor,  // text/plain
                DataFlavor.getTextHTMLFlavor() // text/html
        };

        public MultiFormatTransferable(String plainText, String htmlText) {
            this.plainText = plainText;
            this.htmlText = htmlText;
        }

        @Override
        public DataFlavor[] getTransferDataFlavors() {
            return SUPPORTED_FLAVORS;
        }

        @Override
        public boolean isDataFlavorSupported(DataFlavor flavor) {
            for (DataFlavor supportedFlavor : SUPPORTED_FLAVORS) {
                if (supportedFlavor.equals(flavor)) {
                    return true;
                }
            }
            return false;
        }

        @Override
        public Object getTransferData(DataFlavor flavor) throws UnsupportedFlavorException {
            if (flavor.equals(DataFlavor.stringFlavor)) {
                return plainText;
            } else if (flavor.equals(DataFlavor.getTextHTMLFlavor())) {
                return htmlText;
            } else {
                throw new UnsupportedFlavorException(flavor);
            }
        }
    }
}

public static void main(String[] args) {
    String plainText = "goto stack";
    String htmlText = "<a href=\"https://stackoverflow.com/\">goto stack</a>";
    Transferable multiFormatTransferable = new MultiFormatTransferable(plainText, htmlText);
    Clipboard clipboard = Toolkit.getDefaultToolkit().getSystemClipboard();
    clipboard.setContents(multiFormatTransferable, null);
    System.out.println("Clipboard set with both text/plain and text/html!");
}
