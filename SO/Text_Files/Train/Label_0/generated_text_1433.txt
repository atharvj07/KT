public static void main(String[] args){
    BaseballGame game = new BaseballGame();
    game.setGameId(12345L);

    Session session = Hibernate.getSessionFactory().openSession();
    Transaction transaction = session.beginTransaction();

    // Fetch existing BaseballWeather if it exists
    BaseballWeather weather = session.get(BaseballWeather.class, game.getGameId());
    if (weather == null) {
        weather = new BaseballWeather();
        weather.setId(game.getGameId());  // Ensure ID consistency
    }

    weather.setTemperature(70.0);
    weather.setGame(game);
    game.setWeather(weather);

    session.merge(game); // This should now work fine
    transaction.commit();
    session.close();
}

BaseballWeather mergedWeather = session.merge(weather);
game.setWeather(mergedWeather);
session.merge(game);
