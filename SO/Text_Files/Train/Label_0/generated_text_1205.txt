private ArrayList<Car> carsInStore = new ArrayList<>();

public boolean addCarToStore(String type, String model, String color, int maxSpeed, int year, String plateNumber) {
    boolean exists = carsInStore.stream().anyMatch(car -> car.getPlateNumber().equals(plateNumber));
    
    if (exists) {
        return false; // Car with the same plate number already exists
    }

    carsInStore.add(new Car(type, model, color, maxSpeed, year, plateNumber));
    return true;
}
