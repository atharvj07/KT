import com.fasterxml.jackson.annotation.JsonAnySetter;

import java.util.HashMap;
import java.util.Map;

public class Entity {
    public String marketId;
    public String propertyId;
    public String name;

    // A map to store unknown fields
    private Map<String, Object> otherFields = new HashMap<>();

    // Getter for otherFields
    public Map<String, Object> getOtherFields() {
        return otherFields;
    }

    // @JsonAnySetter to capture any properties not explicitly mapped to fields
    @JsonAnySetter
    public void setOtherField(String name, Object value) {
        this.otherFields.put(name, value);
    }
}

public class Message {
    public String appId;
    public String action;
    public String entityType;
    public Entity entity;
}

import com.fasterxml.jackson.databind.ObjectMapper;

public class Main {
    public static void main(String[] args) throws Exception {
        String jsonString = "{\n" +
                "    \"appId\": \"MyApp\",\n" +
                "    \"action\": \"Upsert\",\n" +
                "    \"entityType\": \"Property\",\n" +
                "    \"entity\": {\n" +
                "        \"marketId\": \"abc123\",\n" +
                "        \"propertyId\": \"def456\",\n" +
                "        \"name\": \"Grand Central Station\",\n" +
                "        \"hasFoo\": true,\n" +
                "        \"hasBar\": false,\n" +
                "        \"areaInSf\": 30000\n" +
                "    }\n" +
                "}";

        ObjectMapper objectMapper = new ObjectMapper();
        Message message = objectMapper.readValue(jsonString, Message.class);

        // Accessing the known fields
        System.out.println("App ID: " + message.appId);
        System.out.println("Action: " + message.action);
        System.out.println("Entity Type: " + message.entityType);
        System.out.println("Market ID: " + message.entity.marketId);
        System.out.println("Property ID: " + message.entity.propertyId);
        System.out.println("Name: " + message.entity.name);

        // Accessing the unknown fields via the Map
        Map<String, Object> otherFields = message.entity.getOtherFields();
        for (Map.Entry<String, Object> entry : otherFields.entrySet()) {
            System.out.println(entry.getKey() + ": " + entry.getValue());
        }
    }
}
