public static String translate(String text) {
    char[] charArray = text.toCharArray();
    StringBuilder sb = new StringBuilder();
    
    boolean previousWasVowel = false;  // Track if last character was a vowel

    for (char c : charArray) {
        if (isVowel(c)) {
            if (!previousWasVowel) {  // Only insert "av" if the previous character was NOT a vowel
                sb.append("av");
            }
            previousWasVowel = true;  // Mark that we just processed a vowel
        } else {
            previousWasVowel = false;  // Reset flag if current character is not a vowel
        }
        sb.append(c);
    }
    
    return sb.toString();
}

    return "aeiou".indexOf(c) != -1;  // Check if character is in "aeiou"
