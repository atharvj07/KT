import java.util.function.BiFunction
import java.util.stream.Collectors

// Original Node class (Node_1)
data class Node_1(val value: String, val children: List<Node_1>)

// Target Node class (Node_2)
data class Node_2(val value: String, val children: List<Node_2>, val additionalValue: String)

fun cloneTree(
    nodeA: Node_1,
    createB: BiFunction<Node_1, List<Node_2>, Node_2>
): Node_2 {
    // Clone all children recursively
    val children: List<Node_2> = nodeA.children
        .stream()
        .map { child -> cloneTree(child, createB) }
        .collect(Collectors.toList())

    // Now create the corresponding Node_2
    return createB.apply(nodeA, children)
}

fun main() {
    // Example function to create a Node_2 instance
    val createNode2: BiFunction<Node_1, List<Node_2>, Node_2> = BiFunction { nodeA, children ->
        Node_2(nodeA.value, children, "Additional value for ${nodeA.value}")
    }

    // Example data
    val nodeA = Node_1("root", listOf(Node_1("child1", emptyList()), Node_1("child2", emptyList())))

    // Clone the tree
    val nodeB = cloneTree(nodeA, createNode2)

    // Print result
    println(nodeB)
