<dependency>
    <groupId>org.mockito</groupId>
    <artifactId>mockito-inline</artifactId>
    <version>4.3.1</version>
    <scope>test</scope>
</dependency>

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.mockito.MockedConstruction;
import org.mockito.Mockito;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.mockito.Mockito.*;

class ConverterTest {

    @BeforeEach
    void setup() {
        MockitoAnnotations.openMocks(this);
    }

    @Test
    void convertTest() {
        try (MockedConstruction<Generator> mockedGenerator = mockConstruction(Generator.class, (mock, context) -> {
            when(mock.generate()).thenReturn("123");
        })) {
            String result = Converter.convert();
            assertEquals("123", result);
        }
    }
}

class SubConverter {
    private final Generator generator;

    SubConverter(Generator generator) {
        this.generator = generator;
    }

    String subConvert() {
        return generator.generate();
    }
}

class Converter {
    private final SubConverter subConverter;

    Converter(SubConverter subConverter) {
        this.subConverter = subConverter;
    }

    String convert() {
        return subConverter.subConvert();
    }
}

@Test
void convertTest() {
    Generator generatorMock = mock(Generator.class);
    when(generatorMock.generate()).thenReturn("123");

    SubConverter subConverter = new SubConverter(generatorMock);
    Converter converter = new Converter(subConverter);

    assertEquals("123", converter.convert());
}
