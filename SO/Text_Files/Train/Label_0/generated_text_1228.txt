import com.fasterxml.jackson.databind.ObjectMapper;
import java.util.Map;

public Map<String, Object> matchNameSearch(String text) throws ElasticsearchException, IOException {
    SearchResponse<Product> response = conf.client.search(s -> s
            .index("test")
            .query(q -> q.match(t -> t.field("name").query(text))),
        Product.class);

    // Convert response to JSON
    ObjectMapper objectMapper = new ObjectMapper();
    return objectMapper.convertValue(response, Map.class);
}

@GetMapping("/match/{text}")
public Map<String, Object> matchText(@PathVariable("text") String text) throws ElasticsearchException, IOException {
    return searcher.matchNameSearch(text);
}

import java.util.List;

@Data
public class SearchResultDTO {
    private long totalHits;
    private List<Product> results;
}

public SearchResultDTO matchNameSearch(String text) throws ElasticsearchException, IOException {
    SearchResponse<Product> response = conf.client.search(s -> s
            .index("test")
            .query(q -> q.match(t -> t.field("name").query(text))),
        Product.class);

    List<Product> products = response.hits().hits().stream()
            .map(hit -> hit.source())
            .toList();

    SearchResultDTO result = new SearchResultDTO();
    result.setTotalHits(response.hits().total().value());
    result.setResults(products);
    
    return result;
}

@Configuration
public class ClientConf {
    
    @Bean
    public ElasticsearchClient client() {
        RestClient restClient = RestClient.builder(new HttpHost("localhost", 9200)).build();
        ElasticsearchTransport transport = new RestClientTransport(restClient, new JacksonJsonpMapper());
        return new ElasticsearchClient(transport);
    }
}
