public interface VehicleType {
    String getKey();
    String getCode();
}

public enum CarType implements VehicleType {
    SEAT("2000", "001"),
    FIAT("3000", "002");

    private final String key;
    private final String code;

    CarType(String key, String code) {
        this.key = key;
        this.code = code;
    }

    @Override
    public String getKey() {
        return key;
    }

    @Override
    public String getCode() {
        return code;
    }
}

public enum TruckType implements VehicleType {
    MERCEDES("4000", "001"),
    FORD("5000", "002");

    private final String key;
    private final String code;

    TruckType(String key, String code) {
        this.key = key;
        this.code = code;
    }

    @Override
    public String getKey() {
        return key;
    }

    @Override
    public String getCode() {
        return code;
    }
}

public enum VanType implements VehicleType {
    JEEP("6000", "001", "40"),
    KIA("7000", "002", "50");

    private final String key;
    private final String code;
    private final String tankSize;

    VanType(String key, String code, String tankSize) {
        this.key = key;
        this.code = code;
        this.tankSize = tankSize;
    }

    @Override
    public String getKey() {
        return key;
    }

    @Override
    public String getCode() {
        return code;
    }

    public String getTankSize() {
        return tankSize;
    }
}
