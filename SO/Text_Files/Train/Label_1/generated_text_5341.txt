                for row in data:

                    cursor.execute(PLOT_PROJECT_DEACTIVATE, ('deleted', row[0], plot_id, project_id))
            result = { &quot;Remain&quot;: removed }, 200   
            conn.commit()
        return result

            conn.rollback()
