from fastapi import FastAPI
import pandas as pd
import json

app = FastAPI()
df = pd.read_csv(&quot;file.csv&quot;)

def parse_csv(df):
    res = df.to_json(orient=&quot;records&quot;)
    parsed = json.loads(res)
    return parsed
    
@app.get(&quot;/questions&quot;)
def load_questions():
    return parse_csv(df)
