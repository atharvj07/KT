def x(a: int = ..., t: Literal[True] = True, b: int = ...) -&gt; int: ...

@overload
def x(a: int = ..., *, t: Literal[False], b: int = ...) -&gt; str: ...

@overload
def x(a: int, t: Literal[False], b: int = ...) -&gt; str: ...

    if t:
        return 5
    return &quot;asd&quot;
