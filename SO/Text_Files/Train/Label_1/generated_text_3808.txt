enum Color {
    RED(() -&gt; Shape.CIRCLE),
    GREEN(() -&gt; Shape.TRIANGLE),
    BLUE(() -&gt; Shape.SQUARE);

    private final Supplier&lt;Shape&gt; shape;

    Color(Supplier&lt;Shape&gt; shape) {
        this.shape = shape;
    }

    Shape getShape() {
        return shape.get();
    }
}

enum Shape {
    CIRCLE(Color.RED),
    TRIANGLE(Color.GREEN),
    SQUARE(Color.BLUE);

    private final Color color;

    Shape(Color color) {
        this.color = color;
    }

    Color getColor() {
        return color;
    }
}
