void drawRectWithRoundedCorners(float cx,float cy, float dx, float dy, float r,unsigned __int32 rgba)
    {
    int i;
    float x0,y0,x,y,a=0.0;
    const int n=9;
    const float da=1.5707963267948966192313216916398/float(n);
    dx-=r+r;
    dy-=r+r;
    glColor4ubv((unsigned __int8*)(&amp;rgba));
    glBegin(GL_TRIANGLE_FAN);
    glVertex2f(cx,cy);
    x0=cx+(0.5*dx);
    y0=cy+(0.5*dy);
    for (i=0;i&lt;n;i++,a+=da)
        {
        x=x0+(r*cos(a));
        y=y0+(r*sin(a));
        glVertex2f(x,y);
        }
    x0-=dx;
    for (i=0;i&lt;n;i++,a+=da)
        {
        x=x0+(r*cos(a));
        y=y0+(r*sin(a));
        glVertex2f(x,y);
        }
    y0-=dy;
    for (i=0;i&lt;n;i++,a+=da)
        {
        x=x0+(r*cos(a));
        y=y0+(r*sin(a));
        glVertex2f(x,y);
        }
    x0+=dx;
    for (i=0;i&lt;n;i++,a+=da)
        {
        x=x0+(r*cos(a));
        y=y0+(r*sin(a));
        glVertex2f(x,y);
        }
    glVertex2f(x,cy+(0.5*dy));
    glEnd();
    }
