            .load(url)
            .apply(new RequestOptions()
                    .format(DecodeFormat.PREFER_ARGB_8888)
                    .override(Target.SIZE_ORIGINAL)) // you can also give your size here
            .into(new Target&lt;File&gt;() {

                }

                    storeImage(resource);
                }

                @Override

                }

                @Override
                public void getSize(@NonNull SizeReadyCallback cb) {

                }

                }

                @Override
