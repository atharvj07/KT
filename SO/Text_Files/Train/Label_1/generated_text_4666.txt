import com.google.gson.*;

class Temp {
    String name = "";
    int age = 0;
    String city = "";
}

class Scratch {
    public static void main(String[] args) {
        String response = "{\"name\":\"John\", \"age\":30, \"city\":\"New York\"}";
    
        //convert response to Temp object
        Gson g = new Gson();

        Temp t = g.fromJson(response, Temp.class);
        System.out.println(t.age);
    }
}
