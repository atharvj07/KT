public class StringBM {
    public String toStringPlus(String a) {
        return &quot;{a:&quot; + a + &quot;, b:&quot; + &quot;, c: &quot; + &quot;}&quot;;
    }

    public String toStringBuilder(String a) {
        StringBuilder sb = new StringBuilder(100);
        return sb.append(&quot;{a:&quot;).append(a)
                .append(&quot;, b:&quot;)
                .append(&quot;, c:&quot;)
                .append(&quot;}&quot;)
                .toString();
    }
}
