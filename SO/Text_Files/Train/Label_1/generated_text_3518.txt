package my.test;

class MyTest {

  static int A; // =0
  static volatile int B; // =0

  public static void main(String[] args) throws InterruptedException {
    var t1 = new Thread(() -&gt; {
      B = 42;
      A = 1;
    });
    var t2 = new Thread(() -&gt; {
      if (A == 1) {          // reads A==1 
        System.out.print(B); // reads B==0
      }
    });

    t1.start();
    t2.start();

    t1.join();
    t2.join();
  }
}
