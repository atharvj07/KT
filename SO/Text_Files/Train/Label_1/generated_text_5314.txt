import backoff
import random


@backoff.on_exception(backoff.constant, ValueError, interval=1, max_tries=5)
def do_it():
    x = random.randint(0, 100)
    if x &lt; 80:
        print(&quot;Failed!&quot;)
        raise ValueError(&quot;Failed!&quot;)

    print(&quot;Success!&quot;)


do_it()
