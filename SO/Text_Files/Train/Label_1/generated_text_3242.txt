public Schema generateSchema(Class&lt;?&gt; clazz) { 
        ReflectData rd = ReflectData.AllowNull().get();
        rd.addLogicalTypeConversion(new BigDecimalConversion());
        return rd.getSchema(clazz);    
    }

    private static class BigDecimalConversion extends Conversion&lt;BigDecimal&gt; {

        public static final String BIG_DECIMAL = &quot;arbitrary-precision-decimal&quot;;

        @Override
        public Class&lt;BigDecimal&gt; getConvertedType() {
            return BigDecimal.class;
        }

        @Override
        public String getLogicalTypeName() {
            return BIG_DECIMAL;
        }

        @Override
        public BigDecimal fromCharSequence(CharSequence value, Schema schema, LogicalType type) {
            return new BigDecimal((String) value);
        }

        @Override
        public CharSequence toCharSequence(BigDecimal value, Schema schema, LogicalType type) {
            return value.toString();
        }

        @Override
        public Schema getRecommendedSchema() {
            BigDecimalLogicalType type = new BigDecimalLogicalType();
            return type.addToSchema(Schema.create(Schema.Type.STRING));
        }
    }


    private static class BigDecimalLogicalType extends LogicalType {

        public static final String BIG_DECIMAL = &quot;arbitrary-precision-decimal&quot;;

        private BigDecimalLogicalType() {
            super(BIG_DECIMAL);
        }

        @Override
        public void validate(Schema schema) {
            if (schema.getType() != Schema.Type.STRING) {
                throw new IllegalArgumentException(&quot;Arbitrary-precision decimal can only be used with an underlying string type&quot;);
            }
        }
    }
