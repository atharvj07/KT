const
    getSorted = references =&gt; {
        const fn = (array = []) =&gt; array.flatMap(o =&gt; [o, ...fn(references[o.taskId])]);
        return fn(references[null]);
    },
    data = [{ taskId: 'SR-1', dependsOn: 'SR-2'}, { taskId: 'SR-2', dependsOn: 'SR-5'}, { taskId: 'SR-3', dependsOn: null}, { taskId: 'SR-4', dependsOn: 'SR-3'}, { taskId: 'SR-5', dependsOn: null}, { taskId: 'SR-6', dependsOn: 'SR-2'}],
    references = data.reduce((r, o) =&gt; {
        (r[o.dependsOn] ??= []).push(o);
        return r;
    }, {}),
    result = getSorted(references);

console.log(result);
