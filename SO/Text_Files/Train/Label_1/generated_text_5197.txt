import pandas as pd
import numpy as np

np.random.seed(42)
df = pd.DataFrame(np.random.randint(0,100,size=(20, 10)), columns=[f'col{i+1}' for i in range(10)])

df.to_xml(&quot;example.xml&quot;, root_name='all', row_name='col', encoding='utf-8', xml_declaration=True, pretty_print=True, index=False, stylesheet='example.xslt')
