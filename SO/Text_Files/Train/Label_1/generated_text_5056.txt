import signal
import asyncio
from fastapi import FastAPI

app = FastAPI()
running = True

def stop_server(*args):
    global running
    running = False

@app.on_event(&quot;startup&quot;)
def startup_event():
    signal.signal(signal.SIGINT, stop_server)

@app.get(&quot;/&quot;)
async def index():
    while running:
        await asyncio.sleep(0.1)
