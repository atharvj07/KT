import copy


data = {'n11': {'n12a': {'n13a': 10, 'n13b': &quot;some text&quot;}, 'n12b': {'n13c': {'n14a': 40}}}, 'n21': {'n22a': 20}}


def myreplace(d, path=[]):
for k, v in d.items():
    if isinstance(v, dict):
        myreplace(v, path + [k])
    else:
        print('_'.join(path + [k]))
        d[k] = '_'.join(path + [k])
return d


copy_of_data = copy.deepcopy(data)
copy_of_data = myreplace(copy_of_data)
print(data)
print(copy_of_data)
