import numpy as np
import pandas as pd

names=[['Pat','Sam', np.nan, 'Tom', ''], [&quot;Angela&quot;, np.nan, &quot;James&quot;, &quot;.&quot;, &quot;Jackie&quot;]]
values=[[1, 9, 1, 2, 1], [1, 3, 1, 5, 10]]

new_names = []
new_values = []
for names_, values_ in zip(names, values):
    n = []
    v = []
    for name, value in zip(names_, values_):
        if not pd.isnull(name):
            n.append(name)
            v.append(value)
    new_names.append(n)
    new_values.append(v)
