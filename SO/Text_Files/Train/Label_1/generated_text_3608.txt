    private static void parse( String text ){
        ObjectMapper om = new ObjectMapper();
        try {
            Map map = om.readValue( text, Map.class );
            System.out.println( map );
        }
        catch (JsonProcessingException e) {
            throw new RuntimeException(e);
        }

    }
