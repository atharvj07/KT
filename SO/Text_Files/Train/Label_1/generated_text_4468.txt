public class Maze {
    public static void main(String[] args) {
        int width = 20;
        int height = 20;

        boolean[][] mazeArray = new boolean[width][height];

        int x = 0;
        int y = 0;

        while (y < mazeArray.length) {
            while (x < mazeArray[y].length) {
                System.out.println("Traverse at (" + x + ", " + y + ")");

                x += 1;
            }

            x = 0;
            y += 1;
        }
    }
}
