import sys
import ctypes



class Location(ctypes.Structure):
    _fields_ = [
        ("x", ctypes.c_size_t),
        ("y", ctypes.c_size_t),
        ("z", ctypes.c_size_t)
    ]



pid = sys.argv[1]

with open(f"/proc/{pid}/maps", "r") as f:
    lines = f.readlines()

for line in lines:
    if "[stack]" in line:
        start, end = line.split()[0].split("-")

start = int(start, 16)
end = int(end, 16)

loc_size = ctypes.sizeof(Location)

with open (f"/proc/{pid}/mem", "rb") as f:
    f.seek(start)
    data = f.read(end-start)

print (len(data))
for i in range(0, len(data), loc_size):
    chunk = data[i:i+loc_size]
    if len (chunk) < loc_size:
        continue
    location = Location.from_buffer_copy(chunk)
    if location.z == 1337:
        print (f"\tx: {location.x}")
        print (f"\ty: {location.y}")
        print (f"\tz: {location.z}")
