def mk_reusable(f):
    &quot;&quot;&quot;
    Makes a reusable iterable out of generator by remembering its arguments
    &quot;&quot;&quot;

    class MyIterable:
        def __init__(self, *args, **kwargs):
            self._args = args
            self._kwargs = kwargs

        def __iter__(self):
            yield from f(*self._args, **self._kwargs)

    return MyIterable


# TEST
@mk_reusable
def myrange(n):
    for i in range(n):
        yield i

x = myrange(3)
assert list(x) == [0, 1, 2]
assert list(x) == [0, 1, 2]
