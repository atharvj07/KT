from tkinter import *
from PIL import ImageTk, Image

root = Tk()

f = open(&quot;data/itemIDsList.txt&quot;)
ids = []
for line in f:
    line = line.rstrip(&quot;\n&quot;)
    ids.append(line)
f.close()

imgs = []
for i in range(10):
    imgs.append(ImageTk.PhotoImage(Image.open(f&quot;website/images/{ids[i]}.png&quot;)))
    Label(root, image=imgs[-1], width=60, height=80).grid()

root.mainloop()
