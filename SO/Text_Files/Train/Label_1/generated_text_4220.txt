public class Main {
    public static void moveEvenToFront(int[] a) {
        moveEvenToFrontRec(a, 0);
    }

    private static void moveEvenToFrontRec(int[] a, int i) {
        if (i == a.length - 1) {
            return;
        }
        else if (a[i] % 2 != 0 && a[i+1] % 2 == 0) {
            int temp = a[i];
            a[i] = a[i+1];
            a[i+1] = temp;
            moveEvenToFrontRec(a, 0);
            return;
        }

        moveEvenToFrontRec(a, i + 1);
    }

    public static void main(String[] args) {
        int[] a = {-1, 0, -7, 3, 10, 4, 0, 2, -1, -5, 6};
        moveEvenToFront(a);

        for (int n : a) {
            System.out.print(n + " ");
        }
    }
}

class Main {
    public static void moveEvenToFront(int[] a) {
        int[] cpy = a.clone();
        int i = 0;

        for (int n : cpy) {
            if (n % 2 == 0) {
                a[i++] = n;
            }
        }

        for (int n : cpy) {
            if (n % 2 != 0) {
                a[i++] = n;
            }
        }
    }

    public static void main(String[] args) {
        int[] a = {-1, -2, 1, 0, 2, 3, 4, 5};
        moveEvenToFront(a);

        for (int n : a) {
            System.out.print(n + " "); // => -2 0 2 4 -1 1 3 5
        }
    }
}
