import asyncio


async def coro(i):
    await asyncio.sleep(i//2)


async def main():
    tasks = []
    for i in range(5):
        tasks.append(coro(i))
    await asyncio.gather(*tasks)


if __name__ == &quot;__main__&quot;:
    loop = asyncio.get_event_loop()
    loop.run_until_complete(main())
    loop.close()
