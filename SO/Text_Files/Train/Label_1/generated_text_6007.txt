    (gby:=asfreq.groupby(asfreq[C].notna().cumsum()))[C]
    .transform("first")
    .sub(gby.cumcount() * N)
    .clip(upper=N)
    .loc[lambda s: s.gt(0)]
    .reset_index(name=C)
    .convert_dtypes()
)
