import java.io.IOException;
public class Main {

    public static void main(String[] args) throws IOException {
        
        int[] count = new int[5];  // a e i o u
        int temp = System.in.read();
        while(temp != '\n') {
            switch(temp) {
                case 'a': // 97
                    ++count[0];
                    break;
                case 'e':
                    ++count[1];
                    break;
                case 'i':
                    ++count[2];
                    break;
                case 'o':
                    ++count[3];
                    break;
                case 'u':
                    ++count[4];
                    break;
            }
            temp = System.in.read();
        }
        // System.in.close();
        for(int item: count) {
            System.out.print(item);
            System.out.print(' ');
        }
    }
}
