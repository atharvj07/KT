import com.amazonaws.AmazonClientException;
import com.amazonaws.regions.Regions;
import com.amazonaws.services.s3.AmazonS3;
import com.amazonaws.services.s3.AmazonS3ClientBuilder;
import com.amazonaws.services.s3.model.BucketLifecycleConfiguration;
import com.amazonaws.services.s3.model.lifecycle.LifecycleFilter;

public class App {

     public static void setLifecycle(AmazonS3 s3, String bucketName) {

        try {
            // Create a rule to delete all objects after 30 days.
            BucketLifecycleConfiguration.Rule rule = new BucketLifecycleConfiguration.Rule()
                    .withId(&quot;Delete objects in 30 days&quot;)
                    .withFilter(new LifecycleFilter())
                    .withExpirationInDays(30)
                    .withStatus(BucketLifecycleConfiguration.ENABLED);

            BucketLifecycleConfiguration configuration = new BucketLifecycleConfiguration()
                    .withRules(rule);
            s3.setBucketLifecycleConfiguration(bucketName, configuration);
        } catch (AmazonClientException e) {
            System.err.println(e.getMessage());
            System.exit(1);
        }
    }

    public static void main(String[] args) {
        AmazonS3 s3 = AmazonS3ClientBuilder.standard().withRegion(Regions.US_EAST_1).build();
        setLifecycle(s3, &quot;mybucket&quot;);
    }
}
