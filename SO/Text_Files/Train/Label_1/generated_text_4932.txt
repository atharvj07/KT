def gen_ints_from_primes(p_list, upto):
    if p_list and upto &gt;= p_list[0]:
        p, *p_list = p_list
        pk = 1
        p_tup = tuple()
        while pk &lt;= upto:
            for q, p_distinct in gen_ints_from_primes(p_list, upto=upto // pk):
                yield pk * q, p_tup + p_distinct
            pk *= p
            p_tup = (p, )
    else:
        yield 1, tuple()
