import os
import time

place = 10

running = True

def Write():
  j = 1
  for i in range(place - 1):
    print(&quot;-&quot;, end = &quot;&quot;)
    j += 1
  print(&quot;a&quot;, end = &quot;&quot;, flush=True)
  for k in range(10 - j):
    print(&quot;-&quot;, end = &quot;&quot;, flush=True)

while running:
  Write()
  time.sleep(1)
  if place &gt; 1:
    place -= 1
  os.system(&quot;clear&quot;)
