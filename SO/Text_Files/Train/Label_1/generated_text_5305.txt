import PIL.Image
import glob
import os

if not &quot;converted&quot; in os.listdir():
    os.mkdir(&quot;converted&quot;)

lst_imgs = [i for i in glob.glob(&quot;*.jpeg&quot;)]
print(lst_imgs)
for i in lst_imgs:
    img = PIL.Image.open(i)
    img = img.convert(&quot;RGB&quot;)
    img.save(&quot;converted\\&quot;+i, &quot;JPEG&quot;)


print(&quot;Done.&quot;)
os.startfile(&quot;converted&quot;)
