from itertools import product, combinations

def pairs(list1, list2):
    a = {*list1}
    b = {*list2}
    ab = a & b
    return [
        *product(a, b-a),
        *product(a-b, ab),
        *combinations(ab, 2)
    ]

from itertools import product, combinations, chain

def pairs(list1, list2):
    a = {*list1}
    b = {*list2}
    ab = a & b
    return chain(
        product(a, b-a),
        product(a-b, ab),
        combinations(ab, 2)
    )
