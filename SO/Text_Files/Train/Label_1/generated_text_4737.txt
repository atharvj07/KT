import numpy as np
import random

np.random.seed(10)
random.seed(10)

data = list(np.random.binomial(size=215, n=1, p=0.3))

sample_mean = []

for i in range(1000):
    sample = random.choices(data, k=215)
    mean = np.mean(sample)
    sample_mean.append(mean)

print(np.mean(sample_mean))
