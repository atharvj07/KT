AWSCredentials credentials = new BasicAWSCredentials(
  &quot;&lt;AWS accesskey&gt;&quot;, 
  &quot;&lt;AWS secretkey&gt;&quot;
);

  .standard()
  .withCredentials(new AWSStaticCredentialsProvider(credentials))
  .withRegion(Regions.US_EAST_2)
  .build();
