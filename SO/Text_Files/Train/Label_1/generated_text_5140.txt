def count(t):
    c = 0
    maxx = max(t)
    right = [0]*len(t)
    left = [0]*len(t)
    maxx = t[0]
   
    for i in range(0, len(t)):
    
        if maxx &gt;= t[i]:
            left[i] = maxx
            
        if maxx &lt; t[i]:
            maxx = t[i]
            left[i] = maxx
             
    minn = t[-1] 
    for i in range(len(t)-1,-1,-1):
        if minn &lt;= t[i]:
            right[i] = minn
            
        if minn &gt; t[i]:
            minn = t[i]
            right[i] = minn
           
    for i in range(0, len(t)-1):
        if left[i] &lt; right[i+1] :
            c += 1
 
    return c
