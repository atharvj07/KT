public abstract class BaseItem {

    int id;
    String content;
    long date;

    public BaseItem() {
        content = &quot;&quot;;
    }

    public BaseItem(int id, String content, long date) {
        this.id = id;
        this.content = content;
        this.date = date;
    }

    public abstract int getViewType();

    public int getId() {
        return id;
    }

    public String getContent() {
        return content;
    }
}
