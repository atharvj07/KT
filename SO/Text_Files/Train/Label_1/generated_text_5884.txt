df = pl.DataFrame({
    "a": [1, 15, None, 20, None]
})
df.with_columns(
    # current row value for "a" minus the last non-null value
    # as the first row has no previous non-null value,
    # fill it with the first value (1)
    (
        pl.col("a") - 
        pl.col("a").shift(fill_value=pl.col("a").first()).forward_fill()
    ).alias("b")
)
# shape: (5, 2)
