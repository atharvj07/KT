        class bbb():

        def f1(typ):
            if typ is aaa:
                print(&quot;aaa&quot;)
            elif typ is bbb:
                print(&quot;bbb&quot;)
            else:
                print(&quot;???&quot;)

        def f2(typ):
            match typ():
                case aaa():
                    print(&quot;aaa&quot;)
                case bbb():
                    print(&quot;bbb&quot;)
                case _:
                    print(&quot;???&quot;)

        f1(aaa)

        f2(aaa)
