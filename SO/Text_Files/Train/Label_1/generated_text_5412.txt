from collections.abc import Hashable
from typing import TypeVar

K = TypeVar(&quot;K&quot;, bound=Hashable)
V = TypeVar(&quot;V&quot;)


class MyDict(dict[K, V]):
    ...


class MyStrKeyDict(dict[str, V]):
    ...


class MyIntValDict(dict[K, int]):
    ...


x: MyDict[str, int]
y: MyStrKeyDict[int]
z: MyIntValDict[str]
