import numpy as np
import scipy

a = np.asarray([[11, 21, 31], [21, 22, 32], [31, 32, 33]])
print(a)
# [[11 21 31]
#  [21 22 32]
#  [31 32 33]]

blocked = scipy.linalg.block_diag(*a)
print(blocked)
# [[11 21 31  0  0  0  0  0  0]
#  [ 0  0  0 21 22 32  0  0  0]
#  [ 0  0  0  0  0  0 31 32 33]]

result = np.repeat(blocked, a.shape[1], axis=0)
print(result)
# [[11 21 31  0  0  0  0  0  0]
#  [11 21 31  0  0  0  0  0  0]
#  [11 21 31  0  0  0  0  0  0]
#  [ 0  0  0 21 22 32  0  0  0]
#  [ 0  0  0 21 22 32  0  0  0]
#  [ 0  0  0 21 22 32  0  0  0]
#  [ 0  0  0  0  0  0 31 32 33]
#  [ 0  0  0  0  0  0 31 32 33]
#  [ 0  0  0  0  0  0 31 32 33]]
