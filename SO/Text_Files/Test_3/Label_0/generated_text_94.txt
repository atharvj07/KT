import java.lang.annotation.*;
import java.lang.reflect.Field;

@Target(ElementType.FIELD)
@Retention(RetentionPolicy.RUNTIME)
@Documented
public @interface Annotation {
    Class<?> className() default Object.class;
}

public class Main {

    @Annotation
    private AnotherClass annotatedAttribute;

    public static void main(String[] args) throws NoSuchFieldException {
        Field field = Main.class.getDeclaredField("annotatedAttribute");
        Annotation annotation = field.getAnnotation(Annotation.class);
        Class<?> className = annotation.className() == Object.class ? field.getType() : annotation.className();
        System.out.println(className); // Will print AnotherClass if not specified otherwise
    }
}

class AnotherClass {}
