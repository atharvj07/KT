public class EscapeXML {

    private static final String[] ESCAPES;

    static {
        int size = 62; // Size is based on the largest escape value '&gt;' (greater than 60)
        ESCAPES = new String[size];
        ESCAPES['<'] = "&lt;";
        ESCAPES['>'] = "&gt;";
        ESCAPES['&'] = "&amp;";
        ESCAPES['\''] = "&#039;";
        ESCAPES['\"'] = "&#034;";
    }

    // Using Map<Character, String> alternative
    private static final Map<Character, String> ESCAPE_MAP;

    static {
        ESCAPE_MAP = new HashMap<>();
        ESCAPE_MAP.put('<', "&lt;");
        ESCAPE_MAP.put('>', "&gt;");
        ESCAPE_MAP.put('&', "&amp;");
        ESCAPE_MAP.put('\'', "&#039;");
        ESCAPE_MAP.put('\"', "&#034;");
    }
}
