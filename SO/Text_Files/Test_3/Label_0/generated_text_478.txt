import polars as pl
import numpy as np

def simulate_events(df):
    return df.with_columns(
        pl.col("daily_n").zip(pl.col("prob"))
        .apply(lambda x: np.random.binomial(n=x[0], p=x[1], size=1), return_dtype=pl.Int32)
        .alias('events')
    )

df_sim = pl.DataFrame({
    "daily_n": [1000, 2000, 3000, 4000],
    "prob": [.5, .5, .5, .6],
    "size": 1
})

df_sim = simulate_events(df_sim)
