import br.inf.portalfiscal.mdfe.wsdl.mdferecepcaosinc.MDFeRecepcaoSinc;
import br.inf.portalfiscal.mdfe.wsdl.mdferecepcaosinc.MDFeRecepcaoSincSoap12;
import br.inf.portalfiscal.mdfe.wsdl.mdferecepcaosinc.MdfeRecepcaoResult;
import br.inf.portalfiscal.mdfe.wsdl.mdferecepcaosinc.ObjectFactory;

import javax.xml.bind.JAXBElement;

public class SoapClient {

    public static void main(String[] args) throws Exception {
        // Setup the ObjectFactory and request body
        ObjectFactory of = new ObjectFactory();
        JAXBElement<String> jaxb = of.createMdfeDadosMsg(
                "<soap:Envelope xmlns:soap=\"http://www.w3.org/2003/05/soap-envelope\" xmlns:mdf=\"http://www.portalfiscal.inf.br/mdfe/wsdl/MDFeRecepcaoSinc\">" +
                "<soap:Header/><soap:Body><mdf:mdfeDadosMsg>?<mdf:mdfeDadosMsg></soap:Body></soap:Envelope>"
        );

        // Create the service client
        MDFeRecepcaoSinc recepcao = new MDFeRecepcaoSinc();
        MDFeRecepcaoSincSoap12 soap = recepcao.getMDFeRecepcaoSincSoap12();

        // Call the service
        MdfeRecepcaoResult result = soap.mdfeRecepcao(jaxb.getValue());

        // Output the result
        if (result != null && result.getMdfeRecepcaoResult() != null) {
            System.out.println(result.getMdfeRecepcaoResult().getContent());
        }
    }
}
