import numpy as np

def ijIndicesToSourceArray(i, j, rows, cols):
    return (i + j - (cols - 1) * (i + j >= cols), j - i + (rows - 1) * (i > j))

def reverse_antidiagonals(arr):
    rows, cols = arr.shape
    result = np.empty_like(arr)
    
    for i in range(rows):
        for j in range(cols):
            ni, nj = ijIndicesToSourceArray(i, j, rows, cols)
            result[i, j] = arr[ni, nj]

    return result
