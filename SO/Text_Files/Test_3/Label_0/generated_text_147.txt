import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.io.IOException;
import java.util.Map;

public class DynamicJsonParser {

    public static Map<String, Object> parseJson(String jsonString) throws IOException {
        ObjectMapper objectMapper = new ObjectMapper();
        JsonNode rootNode = objectMapper.readTree(jsonString);
        return objectMapper.convertValue(rootNode, Map.class);
    }

    public static void main(String[] args) {
        String json = "{\"name\": \"sahal\", \"address\": [{\"state\": \"FL\"}, {\"country\": \"FL\", \"city\": {\"type\": \"municipality\", \"value\": \"California City\"}}, {\"pin\": \"87876\"}]}";
        try {
            Map<String, Object> result = parseJson(json);
            System.out.println(result);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
