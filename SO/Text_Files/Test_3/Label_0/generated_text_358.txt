import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from pmdarima.arima import ADFTest, auto_arima
from sklearn.metrics import r2_score

def run_auto_arima(data):
    data['date_index'] = pd.to_datetime(data['date'])
    data.set_index('date_index', inplace=True)
    arima_data = data[['value']]
    
    d = 1 if ADFTest(alpha=0.05).should_diff(arima_data)[1] else 0

    train_data, test_data = arima_data[:-24], arima_data[-24:]
    
    model = auto_arima(
        train_data, start_p=0, d=d, start_q=0,
        max_p=5, max_d=5, max_q=5,
        start_P=0, D=1, start_Q=0, max_P=5, max_D=5,
        max_Q=5, m=12, seasonal=True,
        stationary=False, error_action='ignore',
        suppress_warnings=True, stepwise=True,
        random_state=20, n_fits=50
    )

    prediction = pd.DataFrame(model.predict(n_periods=len(test_data)), index=test_data.index, columns=['predicted_value'])

    plt.figure(figsize=(7,4))
    plt.plot(train_data, label="Training")
    plt.plot(test_data, label="Test")
    plt.plot(prediction, label="Predicted")
    plt.legend(loc='upper right')
    plt.show()

    test_data['predicted_value'] = prediction
    return r2_score(test_data['value'], test_data['predicted_value'])
