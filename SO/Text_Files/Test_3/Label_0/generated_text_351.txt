from fastapi import FastAPI

def sub_lifespan(_):
    print("sub startup")
    yield
    print("sub shutdown")

sub_app = FastAPI(lifespan=sub_lifespan)

@sub_app.get("/")
async def sub_root():
    return {"message": "Hello Sub World"}

def lifespan(_):
    print("startup")
    yield
    print("shutdown")

app = FastAPI(lifespan=lifespan)
app.mount("/sub", sub_app)
