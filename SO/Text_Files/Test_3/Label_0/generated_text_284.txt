from dataclasses import dataclass
from typing import Dict

def mydecorator(cls):
    setattr(cls, 'save', lambda self: print("Saved"))
    setattr(cls, 'my_fields_dict', {'a': str})
    return cls

@mydecorator
@dataclass
class A:
    a: str = ""

# Usage
a = A()
a.save()
print(a.my_fields_dict)
