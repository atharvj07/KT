import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.Menu;
import javafx.scene.control.MenuBar;
import javafx.scene.control.MenuItem;
import javafx.scene.layout.BorderPane;
import javafx.stage.Stage;

public class BottomMenuApp extends Application {
    @Override
    public void start(Stage stage) {
        MenuBar menuBar = new MenuBar();
        
        Menu menu = new Menu("File");
        MenuItem menuItem = new MenuItem("Open");
        menu.getItems().add(menuItem);
        menuBar.getMenus().add(menu);

        BorderPane root = new BorderPane();
        root.setBottom(menuBar); // Position the menu bar at the bottom
        
        Scene scene = new Scene(root, 300, 200);
        scene.setOnMouseClicked(event -> {
            if (event.getY() < scene.getHeight() - menuBar.getHeight()) {
                menuBar.setVisible(false);  // Hide the menu when clicked outside
            }
        });

        stage.setScene(scene);
        stage.setTitle("Bottom Menu Example");
        stage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
