private void findOverlappingSets(Map<String, Set<Long>> myMap) {
    myMap.entrySet().stream()
        .flatMap(entry1 -> myMap.entrySet().stream()
            .filter(entry2 -> !entry1.getKey().equals(entry2.getKey()) && !Collections.disjoint(entry1.getValue(), entry2.getValue()))
            .map(entry2 -> new AbstractMap.SimpleEntry<>(entry1.getKey(), entry2.getKey())))
        .forEach(pair -> System.out.println(pair.getKey() + " and " + pair.getValue()));
}
