import numpy as np

def masked_mean(v, m):
    masked_v = np.ma.masked_array(v, ~m)
    return masked_v.mean(axis=1)

v = np.array([0, 1, 4, 0, 5])
m = np.array([
    [True, True, False, False, False],
    [True, False, True, False, True],
    [True, True, True, True, True],
])

results = masked_mean(v, m)
print(results)  # [0.5, 3.0, 2.0]
