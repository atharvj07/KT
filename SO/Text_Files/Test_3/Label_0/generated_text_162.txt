import java.util.*;

public class CoordinateConverter {

    public static int[][] convertStringToIntArray(String str) {
        String[] rows = str.replaceAll("[\\[\\]]", "").split("],");
        int[][] result = new int[rows.length][2];
        
        for (int i = 0; i < rows.length; i++) {
            String[] values = rows[i].replaceAll("[\\[\\]]", "").split(",");
            result[i][0] = Integer.parseInt(values[0].trim());
            result[i][1] = Integer.parseInt(values[1].trim());
        }
        
        return result;
    }

    public static void main(String[] args) {
        String str = "[[445, 292], [585, 331], [612, 223], [205, 532]]";
        int[][] coordinates = convertStringToIntArray(str);
        
        for (int[] coord : coordinates) {
            System.out.println(Arrays.toString(coord));
        }
    }
}
