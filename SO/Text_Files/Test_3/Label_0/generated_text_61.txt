java
@Configuration
public class StreamProcessorConfig {

    @Bean
    public Function<KStream<String, InputType>, KStream<String, OutputType>> myfunctiondefinition() {
        return inputTypeStream -> inputTypeStream.map((String k, InputType v) -> 
            KeyValue.pair(k, OutputType.createFrom(v))
        );
    }

    @Bean
    public KStream<String, InputType> tenantAStream(StreamsBuilder builder) {
        return builder.stream("${tenant}-input");
    }

    @Bean
    public KStream<String, OutputType> tenantAOutput(KStream<String, InputType> inputStream) {
        return inputStream.map((String k, InputType v) -> 
            KeyValue.pair(k, OutputType.createFrom(v))
        );
    }
}
