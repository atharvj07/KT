import java.util.*;

class Edge {
    int toVertex, weight;
    Edge(int v, int w) {
        toVertex = v; weight = w;
    }
}

class Solver {
    public static int solve(List<List<Edge>> T, long L) {
        int N = T.size();
        int minEdges = Integer.MAX_VALUE;
        
        for (int i = 0; i < N; i++) {
            boolean[] visited = new boolean[N];
            minEdges = Math.min(minEdges, dfs(T, L, i, -1, 0, 0, visited));
        }
        
        return minEdges == Integer.MAX_VALUE ? -1 : minEdges;
    }

    static int dfs(List<List<Edge>> T, long L, int vertex, int parent, long length, int edges, boolean[] visited) {
        if (length == L) {
            return edges;
        }
        if (length > L) {
            return Integer.MAX_VALUE;
        }
        visited[vertex] = true;
        int min = Integer.MAX_VALUE;
        
        for (Edge edge : T.get(vertex)) {
            if (!visited[edge.toVertex] && edge.toVertex != parent) {
                min = Math.min(min, dfs(T, L, edge.toVertex, vertex, length + edge.weight, edges + 1, visited));
            }
        }
        
        visited[vertex] = false;
        return min;
    }
}

public class Main {
    static void putEdge(List<List<Edge>> T, int vertex1, int vertex2, int weight) {
        T.get(vertex1).add(new Edge(vertex2, weight));
        T.get(vertex2).add(new Edge(vertex1, weight));
    }
    
    public static void main(String[] args) {
        List<List<Edge>> T = new ArrayList<>();
        int N = 8;
        for (int i = 0; i < N; i++) T.add(new ArrayList<>());
        
        putEdge(T, 0, 1, 2);
        putEdge(T, 1, 2, 1);
        putEdge(T, 1, 3, 2);
        putEdge(T, 2, 6, 1);
        putEdge(T, 6, 7, 1);
        putEdge(T, 3, 4, 1);
        putEdge(T, 3, 5, 4);
        
        System.out.println(Solver.solve(T, 5L)); 
    }
}
