public void decodeAndTransmitH264(byte[] bytes) {
    Map.Entry<Boolean, List<Integer>> b = hasMdat(bytes);
    if (b.getKey()) {
        byte[] h264Stream = Arrays.copyOfRange(bytes, b.getValue().get(3) + 1, bytes.length);
        // Transmit the h264Stream to frontend clients via SSE
        sseEmitter.send(h264Stream);
    }
}
