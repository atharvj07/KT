# properties.yml (default)
spring:
  profiles:
    active: dev

# properties-dev.yml
server:
  port: 8081

# properties-prod.yml
server:
  port: 8082

env
# .env (default)
DB_NAME=default_db
DB_USERNAME=default_user
DB_PASSWORD=default_pass

# .env-dev
DB_NAME=dev_db
DB_USERNAME=dev_user
DB_PASSWORD=dev_pass

# .env-prod
DB_NAME=prod_db
DB_USERNAME=prod_user
DB_PASSWORD=prod_pass

sh
# Run with dev profile using Maven
mvn spring-boot:run -Dspring-boot.run.profiles=dev --dotenv-file .env-dev

sh
# Run with prod profile using Maven
mvn spring-boot:run -Dspring-boot.run.profiles=prod --dotenv-file .env-prod
