import java.time.ZonedDateTime;
import java.time.format.DateTimeFormatter;
import java.time.ZoneOffset;

class Scratch {
    public static void main(String[] args) {
        System.out.println(parseDate("2022-03-12T03:59:59+0000Z"));
    }

    static long parseDate(String inputDate) {
        DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyy-MM-dd'T'HH:mm:ssZ'Z'");
        return ZonedDateTime.parse(inputDate, formatter).withZoneSameInstant(ZoneOffset.UTC).toEpochSecond();
    }
}
