private void addNewProductRow(LinearLayout parentLayout) {
    LinearLayout newProductLayout = new LinearLayout(getActivity());
    newProductLayout.setLayoutParams(new LinearLayout.LayoutParams(
            LinearLayout.LayoutParams.MATCH_PARENT, LinearLayout.LayoutParams.WRAP_CONTENT));
    newProductLayout.setOrientation(LinearLayout.HORIZONTAL);

    AutoCompleteTextView tvProduct = new AutoCompleteTextView(getActivity());
    tvProduct.setLayoutParams(new LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1));
    tvProduct.setHint("Enter Product");

    EditText prodQty = new EditText(getActivity());
    prodQty.setLayoutParams(new LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1));
    prodQty.setHint("Enter Quantity");

    EditText prodPrice = new EditText(getActivity());
    prodPrice.setLayoutParams(new LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1));
    prodPrice.setHint("Prod Price");
    prodPrice.setFocusable(false);

    EditText prodSpecs = new EditText(getActivity());
    prodSpecs.setLayoutParams(new LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1));
    prodSpecs.setHint("Prod Specs");
    prodSpecs.setFocusable(false);

    Button removeBtn = new Button(getActivity());
    removeBtn.setLayoutParams(new LinearLayout.LayoutParams(
            LinearLayout.LayoutParams.WRAP_CONTENT, LinearLayout.LayoutParams.WRAP_CONTENT));
    removeBtn.setText("X");
    removeBtn.setOnClickListener(v -> parentLayout.removeView(newProductLayout));

    newProductLayout.addView(tvProduct);
    newProductLayout.addView(prodQty);
    newProductLayout.addView(prodPrice);
    newProductLayout.addView(prodSpecs);
    newProductLayout.addView(removeBtn);

    parentLayout.addView(newProductLayout);
}
