import polars as pl

# Create the DataFrame
df = pl.DataFrame(
    {
        "Group": ["Group1", "Group2", "Group3"],
        "Weight": [100.0, 200.0, 300.0],
        "Vals": [[0.5, 0.5, 0.8], [0.5, 0.5, 0.8], [0.7, 0.9]],
    }
)

# Calculate the weighted values
df = df.with_columns(
    pl.col("Vals").list.eval(pl.element() * pl.col("Weight"), return_dtype=pl.List(pl.Float64)).alias("Weighted")
)

# Show the result
print(df)
