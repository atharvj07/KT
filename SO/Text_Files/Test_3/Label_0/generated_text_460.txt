import numpy as np
from collections import Counter

def balanceness(array: np.ndarray):
    classes = [(c, cnt) for c, cnt in Counter(array).items()]
    m = len(classes)
    n = len(array)
    H = -sum([(cnt / n) * np.log((cnt / n)) for c, cnt in classes])
    return H / np.log(m)

def weight(days_ago: int, epsilon: float = 0.005) -> float:
    return np.exp(-1 * days_ago * epsilon)

def find_most_balanced_recent_opponent(matches: list):
    opponent_data = {}
    
    for player, opponent, days_ago in matches:
        if opponent not in opponent_data:
            opponent_data[opponent] = {"player_1": [], "player_2": []}
        
        if player == "A":
            opponent_data[opponent]["player_1"].append(days_ago)
        elif player == "B":
            opponent_data[opponent]["player_2"].append(days_ago)

    best_opponent = None
    best_score = float('-inf')
    
    for opponent, data in opponent_data.items():
        bal_player_1 = balanceness(np.array(data["player_1"]))
        bal_player_2 = balanceness(np.array(data["player_2"]))
        
        weighted_player_1 = sum(weight(day) for day in data["player_1"])
        weighted_player_2 = sum(weight(day) for day in data["player_2"])
        
        combined_balanceness = (bal_player_1 + bal_player_2) / 2
        combined_weighted_matches = (weighted_player_1 + weighted_player_2) / 2
        
        score = combined_balanceness * combined_weighted_matches
        
        if score > best_score:
            best_score = score
            best_opponent = opponent
            
    return best_opponent

# Example usage
matches = [
    ("A", "C", 1), ("A", "C", 2), ("A", "D", 10), ("A", "F", 100), 
    ("A", "F", 101), ("A", "F", 102), ("A", "G", 1), ("B", "C", 1), 
    ("B", "C", 2), ("B", "D", 10), ("B", "F", 100), ("B", "F", 101), 
    ("B", "F", 102), ("B", "G", 1), ("B", "G", 2), ("B", "G", 3), 
    ("B", "G", 4), ("B", "G", 5), ("B", "G", 6), ("B", "G", 7), 
    ("B", "G", 8)
]

print(find_most_balanced_recent_opponent(matches))
