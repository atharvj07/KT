import javax.crypto.Cipher;
import javax.crypto.KeyGenerator;
import javax.crypto.SecretKey;
import javax.crypto.spec.IvParameterSpec;
import javax.crypto.spec.SecretKeySpec;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.security.NoSuchAlgorithmException;
import java.util.Base64;

public class EncryptionDecryption {

    public static void main(String[] args) throws Exception {
        String keyPath = "/home/debashishd/Downloads/s_key";
        String inputMessage = "Hello this is Roxane";

        // Step 1: Read the key file
        byte[] keyValue = Files.readAllBytes(Paths.get(keyPath));

        // Step 2: AES Encryption
        SecretKeySpec secretKey = new SecretKeySpec(keyValue, "AES");
        Cipher cipher = Cipher.getInstance("AES/CBC/PKCS5Padding");
        IvParameterSpec ivSpec = new IvParameterSpec(new byte[16]); // Initialization vector

        cipher.init(Cipher.ENCRYPT_MODE, secretKey, ivSpec);
        byte[] encryptedValue = cipher.doFinal(inputMessage.getBytes("UTF-8"));
        String encryptedBase64 = Base64.getEncoder().encodeToString(encryptedValue);
        System.out.println("Encrypted value: " + encryptedBase64);

        // Step 3: AES Decryption
        cipher.init(Cipher.DECRYPT_MODE, secretKey, ivSpec);
        byte[] decryptedValue = cipher.doFinal(Base64.getDecoder().decode(encryptedBase64));
        String decryptedMessage = new String(decryptedValue, "UTF-8");
        System.out.println("Decrypted value: " + decryptedMessage);
    }
}
