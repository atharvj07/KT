import polars as pl

def aggregate_fqid(df):
    return df.groupby("session_id").agg(
        pl.col("fqid").value_counts().alias("fqid_counts")
    ).with_columns(
        pl.col("fqid_counts").apply(lambda x: pl.DataFrame(x).rename({"fqid": "fqid_" + str(cat) for cat in x["fqid"].unique()}), return_dtype=pl.Object)
    ).explode("fqid_counts")

df = pl.DataFrame({
    "session_id": [20090312431273200, 20090312431273200, 20090312431273200, 20090312431273200],
    "fqid": ["intro", "gramps", "gramps", "teddy"],
    "elapsed_time": [0, 1323, 831, 5197]
})

result = aggregate_fqid(df)
result.show()
