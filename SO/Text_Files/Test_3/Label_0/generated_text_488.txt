from unittest import mock
import psycopg2
from db import PostgresqlApi

def mock_db_connection():
    mock_db = mock.Mock()
    mock_cursor = mock.Mock()
    mock_db.cursor.return_value = mock_cursor
    mock_db.commit = mock.MagicMock()
    return mock_db, mock_cursor

def test_postgresql_api():
    with mock.patch('psycopg2.connect', mock_db_connection):
        db = PostgresqlApi()
        db.add_event(event="Test Event")
        db.events()
        db.close()
