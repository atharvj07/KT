import java.util.*;
import java.util.concurrent.*;

public class FindMultiples {

    public static class DivisibleThread implements Callable<Set<Integer>> {
        private int n;
        private int divisor;

        public DivisibleThread(int n, int divisor) {
            this.n = n;
            this.divisor = divisor;
        }

        @Override
        public Set<Integer> call() {
            Set<Integer> multiples = new HashSet<>();
            for (int i = 1; i <= n; i++) {
                if (i % divisor == 0) {
                    multiples.add(i);
                }
            }
            return multiples;
        }
    }

    public int getSum(int n) throws InterruptedException, ExecutionException {
        ExecutorService executorService = Executors.newFixedThreadPool(3);

        Future<Set<Integer>> thread1 = executorService.submit(new DivisibleThread(n, 3));
        Future<Set<Integer>> thread2 = executorService.submit(new DivisibleThread(n, 5));
        Future<Set<Integer>> thread3 = executorService.submit(new DivisibleThread(n, 7));

        Set<Integer> resultSet = new HashSet<>();
        resultSet.addAll(thread1.get());
        resultSet.addAll(thread2.get());
        resultSet.addAll(thread3.get());

        executorService.shutdown();

        return resultSet.stream().mapToInt(Integer::intValue).sum();
    }

    public static void main(String[] args) throws InterruptedException, ExecutionException {
        FindMultiples finder = new FindMultiples();
        int sum = finder.getSum(1000);
        System.out.println("Sum: " + sum);
    }
}
