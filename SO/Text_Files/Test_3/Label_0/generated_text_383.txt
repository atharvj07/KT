import polars as pl

def truncate_explode(df):
    min_len = min(df["a"].arr.lengths().min(), df["b"].arr.lengths().min())
    df = df.with_columns([
        pl.col("a").arr.slice(0, min_len),
        pl.col("b").arr.slice(0, min_len)
    ])
    return df.explode(["a", "b"])

data = {
    "a": [[1], [2], [3, 4], [5, 6, 7]],
    "b": [[], [8], [9, 10], [11, 12]],
}
df = pl.DataFrame(data)

result = truncate_explode(df)
print(result)
