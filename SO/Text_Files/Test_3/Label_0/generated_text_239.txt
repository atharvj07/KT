public static void searchNumber(int[] array, int searching) {
    boolean found = false;
    for (int i = 0; i < array.length; i++) {
        if (searching == array[i]) {
            System.out.println(searching + " is at index " + i + ".");
            found = true;
            break;
        }
    }
    if (!found) {
        System.out.println(searching + " was not found.");
    }
}
