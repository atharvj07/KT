import polars as pl

def unpack_sector_exposure(df):
    return df.with_columns(
        pl.col("sector_exposure")
        .str.split(";")
        .arr.eval(
            pl.element().str.split("=").arr.get(1).cast(pl.Float64),
            return_dtype=pl.List(pl.Float64),
        )
        .alias("exposures")
    ).unnest("exposures")

df = pl.DataFrame(
    pl.Series("sector_exposure", [
        "Technology=0.207;Financials=0.090;Health Care=0.084;Consumer Discretionary=0.069", 
        "Financials=0.250;Health Care=0.200;Consumer Staples=0.150;Industrials=0.400"
    ])
)

unpacked_df = unpack_sector_exposure(df)
print(unpacked_df)
