import pandas as pd
from sklearn import datasets

def mutate_keep_unused(df, new_col_name, expr):
    used_cols = {name.strip() for name in expr.replace('+', ' ').replace('*', ' ').split()}
    df[new_col_name] = df.eval(expr)
    return df.drop(columns=used_cols, errors='ignore')

iris = pd.DataFrame(datasets.load_iris().data, columns=datasets.load_iris().feature_names)
result = mutate_keep_unused(iris, 'new_col', 'sepal length (cm) + petal length (cm) * petal width (cm)')
