from flask import Flask, Blueprint

app = Flask(__name__)

# Blueprint definition
http_blueprint = Blueprint('http', __name__)

@http_blueprint.route('/example')
def example():
    return 'Example route in blueprint'

# Register blueprint
app.register_blueprint(http_blueprint)

# You can have multiple blueprints
another_blueprint = Blueprint('another', __name__)

@another_blueprint.route('/another')
def another_example():
    return 'Another example route'

app.register_blueprint(another_blueprint)

if __name__ == '__main__':
    app.run()
