import numpy as np
from sklearn.preprocessing import PolynomialFeatures
from sklearn.linear_model import LinearRegression

def polynomial_gradient_hessian(X, y, degree=2):
    poly = PolynomialFeatures(degree=degree)
    Xp = poly.fit_transform(X)
    
    model = LinearRegression()
    model.fit(Xp, y)
    
    coef = model.coef_
    intercept = model.intercept_
    
    # Gradient: the partial derivatives of the polynomial coefficients
    gradient = coef[1:]  # Skip intercept term
    
    # Hessian: the second partial derivatives (for a linear model, it's 0)
    hessian = np.zeros((len(gradient), len(gradient)))
    
    return gradient, hessian
