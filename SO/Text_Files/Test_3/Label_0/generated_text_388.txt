import inspect

def fct1():
    caller_frame = inspect.stack()[1]
    caller_code = caller_frame[0].f_code
    caller_line = caller_frame[0].f_lineno
    with open(caller_code.co_filename, 'r') as file:
        lines = file.readlines()
        line_content = lines[caller_line - 1].strip()
    return line_content.split('=')[0].strip()

a = fct1()
