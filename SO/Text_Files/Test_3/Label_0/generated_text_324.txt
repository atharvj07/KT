import pandas as pd

def glimpse(df):
    print(f'Rows: {df.shape[0]}')
    print(f'Columns: {df.shape[1]}')
    for col in df.columns:
        dtype = df[col].dtype
        values = ', '.join(map(str, df[col].head(5).tolist()))
        print(f'$ {col} <{dtype}> {values}')

test = pd.DataFrame({'column_one': ['A', 'B', 'C', 'D'], 'column_two': [1, 2, 3, 4]})
glimpse(test)
