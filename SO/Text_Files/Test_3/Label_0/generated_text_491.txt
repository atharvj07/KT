import polars as pl
from polars import col as c

def convert_timezones(df):
    return df.with_columns(
        c.time.dt.convert_time_zone(c.tzone).dt.hour().alias("hour")
    )

df = pl.DataFrame({
    "time": [datetime(2023, 4, 3, 2), datetime(2023, 4, 4, 3), datetime(2023, 4, 5, 4)],
    "tzone": ["Asia/Tokyo", "America/Chicago", "Europe/Paris"]
})

df = convert_timezones(df)
df.show()
