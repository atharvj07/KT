import polars as pl
from datetime import datetime

def sum_values_in_range():
    data = {
        "ID" : [1, 2, 3, 4, 5],
        "Initial Date" : ["2022-01-01", "2022-01-02", "2022-01-03", "2022-01-04", "2022-01-05"],
        "Final Date" : ["2022-01-03", "2022-01-06", "2022-01-07", "2022-01-09", "2022-01-07"],
        "Value" : [10, 20, 30, 40, 50]
    }

    df = pl.DataFrame(data)

    dates = pl.datetime_range(
        start=datetime(2022,1,1),
        end=datetime(2022,1,7),
        interval="1d",
        eager=True,
        closed="both"
    ).to_frame("date")

    result = (
        dates.with_columns(
            pl.sum(
                pl.when(
                    (pl.col("date").is_between(df["Initial Date"], df["Final Date"])) 
                ).then(df["Value"]).otherwise(0)
            ).alias("Summed Value")
        )
    )
    return result

sum_values_in_range()
