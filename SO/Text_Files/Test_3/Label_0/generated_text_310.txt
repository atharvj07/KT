import pandas as pd

def transform_dataframe(df):
    df_result = pd.DataFrame({
        "ID": range(1, 4),
        "Price": df.filter(like="TP").values.flatten(),
        "Net": df.filter(like="VReal").values.flatten(),
        "Range": df.filter(like="TiV").values.flatten()
    })
    return df_result
