import hashlib

def get_hash(algorithm):
    filename = input("Enter the file name: ")
    hash_func = hashlib.new(algorithm)
    with open(filename, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            hash_func.update(byte_block)
    print(f"{algorithm} value has been calculated")
    return hash_func.hexdigest()

def verify_checksum(algorithm, expected_hash):
    calculated_hash = get_hash(algorithm)
    if calculated_hash == expected_hash:
        print(f"{algorithm} hash matches the expected value.")
    else:
        print(f"{algorithm} hash does not match the expected value.")
