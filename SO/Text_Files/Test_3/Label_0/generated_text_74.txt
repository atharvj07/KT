import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.context.ApplicationContext;
import org.springframework.stereotype.Component;

import java.util.List;

interface MyClass {
    void initialize(List<Integer> parameters);
}

@Component
class ClassA implements MyClass {
    @Override
    public void initialize(List<Integer> parameters) {
        System.out.println("ClassA initialized with: " + parameters);
    }
}

@Component
class ClassB implements MyClass {
    @Override
    public void initialize(List<Integer> parameters) {
        System.out.println("ClassB initialized with: " + parameters);
    }
}

@Component
class ClassC implements MyClass {
    @Override
    public void initialize(List<Integer> parameters) {
        System.out.println("ClassC initialized with: " + parameters);
    }
}

@Component
class MyClassFactory {
    @Autowired
    private ApplicationContext context;

    public MyClass createClass(String className, List<Integer> parameters) {
        MyClass myClass = (MyClass) context.getBean(className);
        myClass.initialize(parameters);
        return myClass;
    }
}

@Component
class InputParser {
    @Autowired
    private MyClassFactory classFactory;

    public void parseInput(String input) {
        String[] parts = input.split(" ");
        String className = parts[0];
        List<Integer> parameters = List.of(parts).subList(1, parts.length).stream()
                .map(Integer::parseInt)
                .toList();
        classFactory.createClass(className, parameters);
    }
}
