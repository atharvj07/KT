import org.yaml.snakeyaml.DumperOptions;
import org.yaml.snakeyaml.Yaml;
import java.util.Map;

public class YamlMapper {

    public static InterestingSetup mapYamlToClass(Map<String, Object> yamlConfig) {
        Yaml yaml = new Yaml();
        Map<String, Object> interestingMap = (Map<String, Object>) yamlConfig.get("interesting-setup");
        return yaml.loadAs(yaml.dump(interestingMap), InterestingSetup.class);
    }
}
