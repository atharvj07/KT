import java.util.ArrayList;
import java.util.List;

public class StringGenerator {
    public static void main(String[] args) {
        String[] first  = {"A", "B", "C"};
        String[] second = {"1", "2", "3"};
        String[] third  = {"X", "Y", "Z"};

        List<String> result = createStrings(first, second, third);

        result.forEach(System.out::println);
    }

    static List<String> createStrings(String[]... strs) {
        List<String> result = new ArrayList<>();
        generateCombinations(result, new StringBuilder(), strs, 0);
        return result;
    }

    static void generateCombinations(List<String> result, StringBuilder current, String[] strs[], int index) {
        if (index == strs.length) {
            result.add(current.toString());
            return;
        }

        for (String s : strs[index]) {
            current.append(s);
            generateCombinations(result, current, strs, index + 1);
            current.deleteCharAt(current.length() - 1);
        }
    }
}
