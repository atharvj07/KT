import numpy as np

def calculate_Fourier(points, num_coefficients):
    complex_points = np.array([p[0] + 1j * p[1] for p in points])
    t = np.linspace(0, 1, len(complex_points), endpoint=False)
    delta_t = 1 / len(complex_points)

    c_k = np.array([
        np.sum(complex_points * np.exp(-2j * np.pi * k * t)) * delta_t
        for k in range(-num_coefficients // 2, num_coefficients // 2)
    ])

    return c_k
