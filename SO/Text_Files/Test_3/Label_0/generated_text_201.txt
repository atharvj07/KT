@EnableMongoRepositories(basePackageClasses = PersistencyRepository.class, mongoTemplateRef = "mongoTemplate")
@EnableRedisRepositories(basePackageClasses = CacheRepository.class, redisTemplateRef = "redisTemplate")
@SpringBootApplication
public class StoreApplication {

    @Bean
    public MongoTemplate mongoTemplate(MongoDbFactory mongoDbFactory, MongoConverter mongoConverter) {
        return new MongoTemplate(mongoDbFactory, mongoConverter);
    }

    @Bean
    public RedisTemplate<String, Object> redisTemplate(RedisConnectionFactory factory) {
        RedisTemplate<String, Object> template = new RedisTemplate<>();
        template.setConnectionFactory(factory);
        return template;
    }

    public static void main(String[] args) {
        SpringApplication.run(StoreApplication.class, args);
    }
}
