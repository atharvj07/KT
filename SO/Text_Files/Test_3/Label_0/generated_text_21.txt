@SpringBootTest
@AutoConfigureMockMvc
public class SecurityConfigTest {

    @Autowired
    private MockMvc mockMvc;

    @Test
    public void testPermitAllEndpoint() throws Exception {
        mockMvc.perform(get("/data/actuator/health"))
            .andExpect(status().isOk());
    }

    @Test
    public void testHasScopeDataWrite() throws Exception {
        mockMvc.perform(put("/data/any")
                .header(HttpHeaders.AUTHORIZATION, "Bearer valid-token-with-scope-data:write"))
            .andExpect(status().isOk());
    }

    @Test
    public void testUnauthenticatedRequest() throws Exception {
        mockMvc.perform(get("/data/actuator/health"))
            .andExpect(status().isUnauthorized());
    }
}
