import org.junit.jupiter.api.extension.BeforeEachCallback;
import org.junit.jupiter.api.extension.ExtensionContext;
import org.mockito.MockedStatic;
import org.mockito.Mockito;

public class MockStaticExtension implements BeforeEachCallback {

    private MockedStatic<?> mockedStatic;

    @Override
    public void beforeEach(ExtensionContext context) throws Exception {
        mockedStatic = Mockito.mockStatic(SomeStaticClass.class);
        mockedStatic.when(SomeStaticClass::someStaticMethod).thenReturn("mockedValue");
    }

    public void close() {
        if (mockedStatic != null) {
            mockedStatic.close();
        }
    }
}

java
import org.junit.jupiter.api.extension.ExtendWith;
import org.junit.jupiter.api.Test;

@ExtendWith(MockStaticExtension.class)
public class SomeTest {

    @Test
    void testMethod() {
        // Test with mocked static method
    }
}
