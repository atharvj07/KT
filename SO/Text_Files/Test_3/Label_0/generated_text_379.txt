from typing import Any
import numpy as np

def inspect_ndarray_typing():
    return {
        "NDArray_generic": np.ndarray[Any, np.dtype[Any]],
        "NDArray_specific_dtype": np.ndarray[float],
        "NDArray_explicit_dtype": np.ndarray[Any, np.dtype[float]],
    }

result = inspect_ndarray_typing()
