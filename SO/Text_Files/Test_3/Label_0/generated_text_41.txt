import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.Label;
import javafx.scene.input.MouseEvent;
import javafx.scene.layout.Pane;
import javafx.scene.text.TextAlignment;
import javafx.stage.Stage;

public class MyApp extends Application {
    @Override
    public void start(Stage stage) {
        Pane root = new Pane();
        root.setStyle("-fx-background-color: black;");
        root.getChildren().add(createTestLabel());
        Scene scene = new Scene(root);

        scene.getStylesheets().add("/ExampleStylesheet.css");
        stage.setScene(scene);
        stage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }

    public Label createTestLabel() {
        Label testLabel = new Label("Placeholder\nText");
        testLabel.getStyleClass().add("test-label");
        testLabel.setTextAlignment(TextAlignment.CENTER);
        testLabel.setOnMouseEntered(this::doSomething);
        testLabel.setOnMouseExited(this::doSomethingElse);

        testLabel.setOnMousePressed(event -> handleMousePress(event));
        return testLabel;
    }

    public void doSomething(MouseEvent event) {
        // System.out.println("Inside label.");
    }

    public void doSomethingElse(MouseEvent event) {
        // System.out.println("Outside label.");
    }

    public void handleMousePress(MouseEvent event) {
        Label label = (Label) event.getSource();
        if (!label.getBoundsInParent().contains(event.getSceneX(), event.getSceneY())) {
            event.consume();
        }
    }
}
