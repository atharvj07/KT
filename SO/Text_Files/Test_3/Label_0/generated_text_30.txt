import javafx.application.Application;
import javafx.geometry.Point2D;
import javafx.scene.Scene;
import javafx.scene.control.Label;
import javafx.scene.control.Tooltip;
import javafx.scene.layout.AnchorPane;
import javafx.stage.Stage;
import javafx.util.Duration;

public class HelloApplication extends Application {

    @Override
    public void start(Stage stage) {
        Label label = new Label("TEST\nTEST\nTEST");
        label.setStyle("-fx-background-color: green;");

        Tooltip tooltip = new Tooltip("TOOLTIP");
        tooltip.setShowDelay(Duration.seconds(0.5));
        label.setTooltip(tooltip);

        Scene scene = new Scene(new AnchorPane(label));
        stage.setScene(scene);
        
        stage.show();

        label.setOnMouseEntered(event -> {
            Point2D p = label.localToScene(0.0, 0.0);
            double x = p.getX() + label.getScene().getX() + label.getScene().getWindow().getX();
            double y = p.getY() + label.getScene().getY() + label.getScene().getWindow().getY();
            tooltip.show(label, x + label.getWidth() - tooltip.getWidth(), y + label.getHeight());
        });
    }

    public static void main(String[] args) {
        launch();
    }
}
