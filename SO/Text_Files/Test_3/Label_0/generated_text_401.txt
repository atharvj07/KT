import torch
from transformers import LlamaTokenizer, LlamaForCausalLM, LlamaConfig
from sentencepiece import SentencePieceProcessor

def load_llama_model(model_path):
    tokenizer = SentencePieceProcessor(model_file=f'{model_path}/tokenizer.model')

    model_config = LlamaConfig(
        hidden_size=4096,
        num_hidden_layers=32,
        num_attention_heads=32,
        intermediate_size=int(4096 * 1.3),
        vocab_size=128256,
        use_scaled_rope=True
    )

    model = LlamaForCausalLM(config=model_config)
    model.load_state_dict(torch.load(f'{model_path}/consolidated.00.pth'))

    return model, tokenizer

def generate_text(model, tokenizer, input_text):
    inputs = tokenizer.encode(input_text, return_tensors='pt')
    outputs = model.generate(inputs)

    return tokenizer.decode(outputs[0], skip_special_tokens=True)

# Usage
model_path = 'Meta-Llama-3.1-8B-Instruct'
model, tokenizer = load_llama_model(model_path)
input_text = "Hello, how are you?"
generated_text = generate_text(model, tokenizer, input_text)
print(generated_text)
