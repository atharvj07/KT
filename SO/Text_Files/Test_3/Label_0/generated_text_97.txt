import com.sun.source.tree.*;
import com.sun.tools.javac.processing.JavacProcessingEnvironment;
import com.sun.tools.javac.tree.JCTree;
import com.sun.tools.javac.util.Context;

import javax.annotation.processing.Processor;
import javax.lang.model.element.Element;
import javax.lang.model.type.TypeMirror;

public class SwitchExpressionTypeFinder {

    public TypeMirror getSwitchExpressionType(SwitchTree switchTree) {
        ExpressionTree switchTreeExpression = switchTree.getExpression();
        Context context = ((JavacProcessingEnvironment) processingEnv).getContext();
        JCTree tree = (JCTree) switchTreeExpression;
        return tree.type;
    }
}
