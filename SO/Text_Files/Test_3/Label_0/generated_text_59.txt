import javafx.scene.image.Image;
import javafx.scene.material.PhongMaterial;
import javafx.scene.shape.Sphere;
import javafx.scene.paint.Color;
import javafx.scene.text.Text;

public class EarthSphere {
    public static Sphere createEarthSphere() {
        Sphere earthSphere = new Sphere(256);
        PhongMaterial material = new PhongMaterial();
        material.setDiffuseMap(new Image(EarthSphere.class.getResourceAsStream("/images/earth2.jpg")));
        earthSphere.setMaterial(material);
        return earthSphere;
    }
}
