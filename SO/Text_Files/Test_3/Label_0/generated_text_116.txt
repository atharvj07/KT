findAll()
    .flatMap(p -> Flux.fromStream(p.getRelations().stream())
        .filter(this::isPassing)
        .next() // Take the first matching element
        .flatMap(r -> findStorageId(r)))
    .handle((Long storageId, SynchronousSink<Long> sink) -> {
        if (storageId != null) {
            sink.next(storageId);
            sink.complete();
        }
    })
    .switchIfEmpty(Mono.error(new RuntimeException("Can't find storageId.")));
