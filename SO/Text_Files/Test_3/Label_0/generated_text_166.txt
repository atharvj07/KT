import java.awt.*;
import javax.swing.*;

public class Main {
    public static void main(String[] args) {
        JFrame jf = new JFrame("Demo");
        Container cp = jf.getContentPane();
        MyCanvas tl = new MyCanvas();
        cp.add(tl);
        jf.setSize(500, 250);
        jf.setVisible(true);
        jf.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
    }
}

class MyCanvas extends JComponent {

    @Override
    public void paintComponent(Graphics g) {
        if (g instanceof Graphics2D g2) {
            // Get scale factor for high DPI screens
            double scaleFactor = getGraphicsConfiguration().getDevice().getDefaultConfiguration().getDefaultTransform().getScaleX();

            g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            g2.setFont(Font.decode("Times New Roman-26"));
            g2.drawString("The poorly-scaled cake is a lie.", 70, 40);
            g2.setStroke(new BasicStroke(2.3f));
            g2.draw(new Line2D.Double(420, 10, 425, 70));

            Image I = createScalingAwareBuffer(this, 500, 150, scaleFactor);
            g2.drawImage(I, 0, 130, null);
        }
    }

    private Image createScalingAwareBuffer(JComponent component, int width, int height, double scaleFactor) {
        // Create image with appropriate scale factor
        Image img = component.createImage((int) (width * scaleFactor), (int) (height * scaleFactor));
        Graphics2D g2d = (Graphics2D) img.getGraphics();
        g2d.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g2d.setFont(Font.decode("Times New Roman-26"));
        g2d.drawString("The poorly-scaled cake is a lie.", (int) (70 * scaleFactor), (int) (40 * scaleFactor));
        g2d.setStroke(new BasicStroke(2.3f * (float) scaleFactor));
        g2d.draw(new Line2D.Double(420 * scaleFactor, 10 * scaleFactor, 425 * scaleFactor, 70 * scaleFactor));
        g2d.dispose();
        return img;
    }
}
