from pydantic import BaseModel, root_validator
from typing import List

def root_unique_validator(column: str):
    def _validate_unique(cls, values):
        root_values = values.get('__root__')
        value_set = set()
        for value in root_values:
            if value[column] in value_set:
                raise ValueError(f'Duplicate {column}')
            else:
                value_set.add(value[column])
        return values
    return classmethod(_validate_unique)

class CustomerRecord(BaseModel):
    id: int
    name: str
    address: str

class CustomerRecordDF(BaseModel):
    __root__: List[CustomerRecord]
