import pandas as pd
import requests
from bs4 import BeautifulSoup

def fetch_bmv_data():
    url = 'https://www.bmv.com.mx/es/mercados/capitales'
    page = requests.get(url)
    soup = BeautifulSoup(page.text, 'lxml')
    
    headers = ["EMISORA", "SERIE", "HORA", "ÃÂLTIMO", "PPP", "ANTERIOR", "MÃÂXIMO", "MÃÂNIMO", "VOLUMEN", "IMPORTE", "OPS.", "VAR PUNTOS", "VAR %"]
    df = pd.DataFrame(columns=headers)

    rows = soup.find_all('tr', {"role": "row"})
    for row in rows:
        cells = row.find_all('td')
        if len(cells) > 1:
            data = [cell.get_text(strip=True).replace('\n', '') for cell in cells]
            df.loc[len(df)] = data
    
    return df

df = fetch_bmv_data()
print(df)
