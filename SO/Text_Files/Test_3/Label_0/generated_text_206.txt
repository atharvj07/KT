public void paintComponent(Graphics g) {
    super.paintComponent(g);
    if (ApplicationImages.getImage() != null) {
        double scaleFactor = Math.min((double) getParent().getWidth() / 1280, (double) getParent().getHeight() / 859);
        int width = (int) (1280 * scaleFactor);
        int height = (int) (859 * scaleFactor);
        int x = (getParent().getWidth() - width) / 2;
        int y = (getParent().getHeight() - height) / 2;

        g.drawImage(ApplicationImages.getImage().getScaledInstance(width, height, BufferedImage.SCALE_SMOOTH), x, y, this);
    }
}
