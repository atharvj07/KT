public static void sortFeatureAnalyzeDTOResults(List<FeatureAnalyzeDTOResult> list) {
    YearMonth thisMonth = YearMonth.now();
    List<String> monthList = new ArrayList<>();
    for (int i = 0; i < 6; i++) {
        monthList.add(thisMonth.minusMonths(i).format(DateTimeFormatter.ofPattern("MMMM")).toUpperCase());
    }
    list.sort(Comparator.comparingInt(o -> monthList.indexOf(o.getMonth().toUpperCase())));
}
