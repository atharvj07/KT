import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.spatial import Voronoi, voronoi_plot_2d
from shapely.geometry import Polygon, MultiPolygon
from matplotlib.patches import Polygon as MplPolygon
from matplotlib.collections import PatchCollection

def plot_phase_diagram(df):
    points = df[['x', 'y']].values
    labels = df['label'].values
    colors = df['color'].values

    vor = Voronoi(points)
    fig, ax = plt.subplots(figsize=(8, 8))

    patches = []
    for region in vor.regions:
        if not -1 in region and region:
            poly = Polygon([vor.vertices[i] for i in region])
            center = poly.centroid
            label_idx = np.argmin(np.linalg.norm(points - np.array([center.x, center.y]), axis=1))
            color = colors[label_idx]
            patches.append(MplPolygon(list(poly.exterior.coords), closed=True, color=color))

    p = PatchCollection(patches, match_original=True, edgecolor='black', linewidth=0.5, alpha=0.7)
    ax.add_collection(p)
    ax.set_xlim([-40, 0])
    ax.set_ylim([-30, 10])
    ax.set_xlabel('Log pO2(g)')
    ax.set_ylabel('Log pSO2(g)')
    plt.show()

plot_phase_diagram(df)
