import polars as pl

def compute_sum_lower_max_other(df):
    sorted_df = df.sort("value")
    cumsum = sorted_df["value"].cumsum().shift_and_fill(fill_value=0)
    sum_lower = df.join(sorted_df.with_columns(cumsum.alias("sum_lower")), on="value", how="left")["sum_lower"]
    max_other = df["value"].max() - (df["value"] == df["value"].max()) * df["value"]
    return df.with_columns(sum_lower.alias("sum_lower"), max_other.alias("max_other"))
