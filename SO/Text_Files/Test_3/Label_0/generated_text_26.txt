public class DemoSecurityProblem {
    private String name;

    public DemoSecurityProblem(String name) {
        this.name = name;
    }

    @Override
    protected void finalize() throws Throwable {
        super.finalize();
        System.out.println("Finalizing DemoSecurityProblem");
    }
}

public class MaliciousSubClass extends DemoSecurityProblem {
    private static DemoSecurityProblem maliciousInstance;

    public MaliciousSubClass(String name) {
        super(name);
        System.out.println("MaliciousSubClass constructor");
    }

    @Override
    protected void finalize() throws Throwable {
        maliciousInstance = this; // Capture partially constructed object in static field
        super.finalize();
    }

    public static void attack() {
        try {
            MaliciousSubClass maliciousObject = new MaliciousSubClass("Malicious Object");
            maliciousObject = null;  // Allow object to be eligible for GC
            System.gc();  // Suggest garbage collection
            Thread.sleep(1000);  // Give GC time to run
            System.out.println("Malicious instance still referenced: " + (maliciousInstance != null));
        } catch (InterruptedException e) {
            e.printStackTrace();
        }
    }

    public static void main(String[] args) {
        attack();
    }
}
