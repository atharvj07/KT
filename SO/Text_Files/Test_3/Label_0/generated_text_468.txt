from http.server import SimpleHTTPRequestHandler, HTTPServer
import os

def run(server_class=HTTPServer, handler_class=SimpleHTTPRequestHandler, addr='', port=8000, directory=None):
    if directory:
        os.chdir(directory)
    server_address = (addr, port)
    httpd = server_class(server_address, handler_class)
    print(f"Serving on {addr}:{port}, directory: {directory}")
    httpd.serve_forever()

run(addr='127.0.0.1', port=8000, directory='path_to_directory')
