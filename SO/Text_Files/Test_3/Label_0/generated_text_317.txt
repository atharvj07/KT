import pandas as pd

def count_star_combinations(inp):
    return (inp.groupby(['month', 'item', 'star'])
              .size()
              .unstack(fill_value=0)
              .rename(columns=lambda x: f'star_{x}_cnt')
              .reset_index())

inp = pd.DataFrame({'month': [1, 1, 1, 1, 1, 2, 2, 2], 
                    'item': [10, 10, 10, 10, 20, 20, 20, 20], 
                    'star': [1, 2, 1, 3, 3, 2, 2, 3]})

output = count_star_combinations(inp)
