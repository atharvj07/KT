public class Record {
    private String sourceNode;
    private String destinationNode;
    private String weekIndex;
    private int quantity;

    // Getters and setters
}

public List<Record> calculatePercentage(List<Record> records) {
    return records.stream()
        .collect(Collectors.groupingBy(Record::getWeekIndex,
            Collectors.groupingBy(Record::getSourceNode,
                Collectors.collectingAndThen(
                    Collectors.toList(), sourceRecords -> {
                        int totalQuantity = sourceRecords.stream().mapToInt(Record::getQuantity).sum();
                        return sourceRecords.stream()
                            .map(record -> {
                                double percentage = (double) record.getQuantity() / totalQuantity * 100;
                                record.setQuantity((int) percentage); // Reusing the quantity field for percentage
                                return record;
                            })
                            .collect(Collectors.toList());
                    }
                )
            )
        ))
        .values().stream()
        .flatMap(sourceMap -> sourceMap.values().stream())
        .flatMap(List::stream)
        .collect(Collectors.toList());
}
