import io.opentelemetry.api.trace.Span;
import io.opentelemetry.api.trace.Tracer;
import io.opentelemetry.api.trace.TraceFlags;
import io.opentelemetry.api.trace.TraceState;
import io.opentelemetry.context.Context;
import io.opentelemetry.context.Scope;

@Inject
private Tracer tracer;

public void processMessage(String traceId) {
    // Extract traceId from SQS message
    Context parentContext = Context.root().with(Span.wrap(io.opentelemetry.api.trace.SpanContext.create(traceId, "0", TraceFlags.getDefault(), TraceState.getDefault())));

    // Create a new span and link it to the parent context
    Span consumeMessageSpan = tracer.spanBuilder("consumeMessage")
            .setParent(parentContext)
            .startSpan();

    try (Scope scope = consumeMessageSpan.makeCurrent()) {
        // Your processing logic here
    } finally {
        consumeMessageSpan.end();
    }
}
