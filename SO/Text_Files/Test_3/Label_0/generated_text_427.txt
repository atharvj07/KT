import polars as pl
import numpy as np
from numpy.linalg import norm

cosine_similarity = lambda a, b: (a @ b.T) / (norm(a) * norm(b))

def calculate_cosine_similarity(df):
    arr = df["col2"].to_list()
    similarities = {}
    for i, col1 in enumerate(df["col1"]):
        for j, col2 in enumerate(df["col1"]):
            sim = cosine_similarity(np.array(arr[i]), np.array(arr[j]))
            similarities[f"{col1}_{col2}"] = sim
    return similarities

# Example usage
data = {
    "col1": ["a", "b", "c", "d"],
    "col2": [[-0.06066, 0.072485, 0.548874, 0.158507],
             [-0.536674, 0.10478, 0.926022, -0.083722],
             [-0.21311, -0.030623, 0.300583, 0.261814],
             [-0.308025, 0.006694, 0.176335, 0.533835]],
}

df = pl.DataFrame(data)
cosine_sim = calculate_cosine_similarity(df)
print(cosine_sim)
