private <T> void positionEntity(Class<T> entityClass, Object... params) {
    int x;
    int y;
    boolean entityIsCreated = false;
    while (!entityIsCreated) {
        x = random.nextInt(sizeX);
        y = random.nextInt(sizeY);
        if (field.getFieldable(x, y) instanceof Empty) {
            try {
                T entity = entityClass.getConstructor(int.class, int.class, Object.class, Object.class, int.class, int.class)
                        .newInstance(x, y, params);
                field.setFieldable(x, y, entity);
                entityIsCreated = true;
            } catch (Exception e) {
                e.printStackTrace();
            }
        }
    }
}
