class DocProc extends SimpleDocTreeVisitor<Void, Trees> {
    private final Elements elm;
    private final Types typ;

    DocProc(ProcessingEnvironment e) {
        this.elm = e.getElementUtils();
        this.typ = e.getTypeUtils();
    }

    @Override
    public Void visitLink(LinkTree l, Trees t) {
        String sig = l.getReference().getSignature();
        for (Element e : elm.getAllModuleElements()) {
            for (Element p : e.getEnclosedElements()) {
                for (Element i : p.getEnclosedElements()) {
                    if (i.getSimpleName().toString().equals(sig)) {
                        System.out.println(elm.getBinaryName((TypeElement) i));
                        return null;
                    }
                }
            }
        }
        return null;
    }
}
