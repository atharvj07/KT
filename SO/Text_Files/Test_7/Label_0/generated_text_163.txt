import java.util.*;

public class Run {
    public static void main(String[] args) {
        Card[] d = {new Card("A"), new Card("B"), new Card("C"), new Card("D")};
        Random r = new Random();
        for (int i = d.length - 1; i > 0; i--) {
            int j = r.nextInt(i + 1);
            Card t = d[i];
            d[i] = d[j];
            d[j] = t;
        }
        for (Card c : d) System.out.println(c);
    }
}

class Card {
    String v;
    Card(String v) { this.v = v; }
    public String toString() { return v; }
}
