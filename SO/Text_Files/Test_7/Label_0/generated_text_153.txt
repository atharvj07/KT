package intcp;

import io.grpc.Metadata;
import io.grpc.ServerCall;
import io.grpc.ServerCall.Listener;
import io.grpc.ServerCallHandler;
import io.grpc.ServerInterceptor;

public class IntcpSrv implements ServerInterceptor {
    @Override
    public <R, S> Listener<R> interceptCall(ServerCall<R, S> cal, Metadata hdr, ServerCallHandler<R, S> nxt) {
        System.out.println("Hi");
        return nxt.startCall(cal, hdr);
    }
}
