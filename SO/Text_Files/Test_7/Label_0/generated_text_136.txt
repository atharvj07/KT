public class StrToDtDesTest {
    
    private StrToDtDes des;
    private JsonParser jp;
    private DeserializationContext ctx;

    @BeforeEach
    void setUp() {
        des = new StrToDtDes();
        jp = mock(JsonParser.class);
        ctx = mock(DeserializationContext.class);
    }

    @Test
    void testValidDate() throws Exception {
        when(jp.getText()).thenReturn("2025-03-17T10:15:30Z");
        Date dt = des.deserialize(jp, ctx);
        assertNotNull(dt);
    }

    @Test
    void testEmptyDate() throws Exception {
        when(jp.getText()).thenReturn("");
        Date dt = des.deserialize(jp, ctx);
        assertNull(dt);
    }

    @Test
    void testInvalidDate() throws Exception {
        when(jp.getText()).thenReturn("invalid-date");
        assertThrows(MyCustomException.class, () -> des.deserialize(jp, ctx));
    }
}
