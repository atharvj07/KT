import openai

client = openai.Client(api_key="your_api_key")

file = client.files.create(
  file=open("example.pdf", "rb"),
  purpose='answers'
)

response = client.completions.create(
  model="gpt-4",
  prompt="Answer the question based on the context from the PDF file.",
  file_ids=[file.id]
)
