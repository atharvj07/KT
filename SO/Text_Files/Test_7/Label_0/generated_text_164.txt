import org.junit.jupiter.api.extension.*;
import org.junit.platform.suite.api.*;

@Suite
@SelectClasses({FirstTest.class, SecondTest.class})
@ExtendWith(JUnit5Srv.class)
public class JUnit5S {}

class JUnit5Srv implements BeforeAllCallback, AfterAllCallback {
    @Override
    public void beforeAll(ExtensionContext c) throws Exception { start(); }
    @Override
    public void afterAll(ExtensionContext c) throws Exception { stop(); }
    void start() { System.out.println("Start"); }
    void stop() { System.out.println("Stop"); }
}
