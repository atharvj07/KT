from django.db import models
from django.conf import settings

class AbsMod(models.Model):
    crt_at = models.DateTimeField(auto_now_add=True, blank=True, null=True)

    def rel_nm(cls):
        return f"{cls.__name__.lower()}_crt_by"

    crt_by = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.SET_NULL,
        blank=True,
        null=True,
        related_name=rel_nm.__func__
    )

    class Meta:
        abstract = True
