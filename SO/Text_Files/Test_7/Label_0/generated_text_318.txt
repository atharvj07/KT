@client.event

    if message.author.bot:

        return await message.reply(f'Hey {message.author.name}!')

    if message.content.startswith(str(prefix_now)):
        try:
            await message.reply("Switching to slash commands.")
        except discord.Forbidden:
            pass
        return

    if client.user in message.mentions:
        response = "I'm the best giveaway bot!"
        try:
            await message.reply(f"{response} (My prefix: `/`)", mention_author=False)

            pass
        return

    print(f"5 if checks done - {time() - start}")

    count = 1
    mydb = await getConnection()

    in_database = await mycursor.fetchone()
    if in_database:
        if in_database[0] is not None:
            role_list = in_database[0].split(" ")
            for roleid in role_list:
                try:
                    int(roleid)
                except ValueError:
                    continue
