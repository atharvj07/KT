import pytest
import json
from starlette.types import Scope, Receive, Send
from starlette.requests import Request
from starlette.responses import JSONResponse
from fastapi import FastAPI
from starlette.testclient import TestClient

class Mid:
    def __init__(self, app):
        self.app = app

    async def __call__(self, sco: Scope, rec: Receive, snd: Send):
        if sco["type"] != "http":
            await self.app(sco, rec, snd)
            return
        async def mod():
            msg = await rec()
            if msg.get("type") != "http.request":
                return msg
            if not msg.get("body"):
                return msg
            bod = json.loads(msg["body"].decode("utf-8"))
            bod["key"] = "val"
            msg["body"] = json.dumps(bod).encode("utf-8")
            return msg
        await self.app(sco, mod, snd)

app = FastAPI()
app.add_middleware(Mid)

@app.post("/")
async def end(req: Request):
    return JSONResponse(await req.json())

cli = TestClient(app)

def test_mid():
    res = cli.post("/", json={})
    assert res.json().get("key") == "val"
