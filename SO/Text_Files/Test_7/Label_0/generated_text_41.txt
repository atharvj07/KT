import javafx.application.*;
import javafx.scene.*;
import javafx.scene.control.*;
import javafx.scene.input.*;
import javafx.scene.layout.*;
import javafx.stage.*;

public class MyApp extends Application {
    @Override
    public void start(Stage stg) {
        var root = new Pane();
        root.setStyle("-fx-background-color: black;");
        root.getChildren().add(mkLbl());
        var scn = new Scene(root);
        scn.getStylesheets().add("/ExampleStylesheet.css");
        stg.setScene(scn);
        stg.show();
    }

    public static void main(String[] args) {
        launch(args);
    }

    public Label mkLbl() {
        var lbl = new Label("Placeholder\nText");
        lbl.getStyleClass().add("test-label");
        lbl.setTextAlignment(javafx.scene.text.TextAlignment.CENTER);
        lbl.setOnMouseEntered(e -> inLbl(e, lbl));
        lbl.setOnMouseExited(e -> outLbl(e, lbl));
        return lbl;
    }

    public void inLbl(MouseEvent e, Label lbl) {
        var bnd = lbl.getBoundsInParent();
        if (bnd.contains(e.getX(), e.getY())) {
            System.out.println("Inside label.");
        }
    }

    public void outLbl(MouseEvent e, Label lbl) {
        var bnd = lbl.getBoundsInParent();
        if (!bnd.contains(e.getX(), e.getY())) {
            System.out.println("Outside label.");
        }
    }
}
