import polars as pl

df = pl.DataFrame(
    pl.Series("sector_exposure", [
        "Technology=0.207;Financials=0.090;Health Care=0.084;Consumer Discretionary=0.069", 
        "Financials=0.250;Health Care=0.200;Consumer Staples=0.150;Industrials=0.400"
    ])
)

out = df.select(
    pl.col("sector_exposure")
    .str.split(";")
    .arr.eval(pl.element().str.split("=").alias("split"))
    .arr.eval(pl.struct([pl.col("split").arr.get(0).alias("sector"), pl.col("split").arr.get(1).cast(pl.Float64).alias("value")]))
    .arr.to_struct()
)

out
