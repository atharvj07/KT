import pandas as pd

a = pd.DataFrame({
    "FIELD_A": [123123, 123124, 123144],
    "FIELD_B": [8, 7, 99],
    "FIELD_C": ["a", "c", "x"],
    "FIELD_D": [23423, 6464, 234]
})

b = a.melt(id_vars=["FIELD_A"], var_name="FIELD_NAME", value_name="VAL")
b["ID"] = range(1, len(b) + 1)
b["FIELD_NAME"] = b["FIELD_NAME"].str[-1]
b["FIELD_STR_VALUE"] = b["VAL"].where(b["VAL"].apply(lambda x: isinstance(x, str)))
b["FIELD_NUM_VALUE"] = b["VAL"].where(b["VAL"].apply(lambda x: isinstance(x, (int, float))))
b = b.drop(columns=["VAL"])

print(b)
