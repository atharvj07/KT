import java.lang.annotation.*;
import java.lang.reflect.Field;

@Target(ElementType.FIELD)
@Retention(RetentionPolicy.RUNTIME)
@Documented
public @interface MyAnnotation {
    Class<?> className() default Object.class;
}

public class AnotherClass {}

public class YetAnotherClass {}

public class Main {

    @MyAnnotation
    private AnotherClass annotatedAttribute;

    public static void main(String[] args) throws NoSuchFieldException {
        Main main = new Main();
        Field field = main.getClass().getDeclaredField("annotatedAttribute");
        MyAnnotation annotation = field.getAnnotation(MyAnnotation.class);
        Class<?> className = annotation.className();
        System.out.println(className);  // Prints: class AnotherClass
    }
}
