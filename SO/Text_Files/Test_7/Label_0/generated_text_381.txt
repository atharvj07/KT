import polars as pl

df = pl.DataFrame({
    "x": [[1], [1, 2], [1, 2, 3]],
    "y": [[True], None, [True, False, True]]
})

df = df.with_columns(
    pl.when(df["y"].is_null())
    .then(pl.lit(None).repeat_by(df["x"].list.len()))
    .otherwise(df["y"])
    .alias("y")
)

df = df.explode(["x", "y"])
print(df)
