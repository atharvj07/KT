public class FWr {
    public static void main(String[] args) throws Exception {
        try (RandomAccessFile f = new RandomAccessFile("test.txt", "rw")) {
            f.write("123456789".getBytes());
        }
        try (RandomAccessFile f = new RandomAccessFile("test.txt", "rw")) {
            f.seek(0);
            f.write("hello".getBytes());
        }
    }
}
