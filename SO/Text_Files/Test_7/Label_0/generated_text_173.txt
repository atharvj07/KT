 void initialize(URL u, ResourceBundle r) {
    XYChart.Series<String, Number> s1 = new XYChart.Series<>();
    s1.getData().add(new XYChart.Data<>("Jan", 12));
    s1.getData().add(new XYChart.Data<>("Feb", 20));
    s1.getData().add(new XYChart.Data<>("March", 10));
    s1.getData().add(new XYChart.Data<>("April", 14));

    XYChart.Series<String, Number> s2 = new XYChart.Series<>();
    for (XYChart.Data<String, Number> d : s1.getData()) {
        s2.getData().add(new XYChart.Data<>(d.getXValue(), d.getYValue()));
    }

    linechart.getData().add(s1);
    barchart.getData().add(s2);
}
