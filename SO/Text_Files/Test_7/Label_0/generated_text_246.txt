package com.question.controller;

import com.question.service.LangService;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.web.bind.annotation.*;

@RestController
public class LangCtrl {

    private final LangService lService;

    @Value("${configuration}")
    private String config;

    public LangCtrl(@Qualifier("french") LangService lService) {
        this.lService = lService;
    }

    @GetMapping("/test")
    public String tst(@RequestParam String name) {
        return lService.proc(name);
    }
}
