import numpy as np
from numba import njit

@njit
def calc_y(z, x_size, z_size):
    y = np.empty((x_size, z_size), dtype=np.uint16)
    for i in range(z_size):
        for j in range(x_size):
            y[j, i] = np.argmin(np.abs(z[j] - i))
    return y

x_size = 2000
y_size = 2500
z_size = 400
rng = np.random.default_rng(123)
z_coordinates = np.linspace(0, z_size, y_size) + rng.laplace(0, 1, (x_size, y_size))

y_coordinates = calc_y(z_coordinates, x_size, z_size)
