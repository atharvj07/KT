import org.mockito.*;
import javax.sql.*;
import java.sql.*;

public class DbSim {
    public static DataSource mockDS() throws SQLException {
        DataSource ds = Mockito.spy(DataSourceBuilder.create().build());
        Mockito.doAnswer(inv -> {
            if (Math.random() < 0.5) throw new SQLTransientConnectionException("Mock Timeout");
            return ds.getConnection();
        }).when(ds).getConnection();
        return ds;
    }
}
