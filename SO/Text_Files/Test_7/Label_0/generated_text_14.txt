public abstract class Par {
    protected abstract static class InPar {
        public InPar get() {
            return this;
        }
    }
}

public class Chi extends Par {
    public static class InChi extends InPar {}
}

public class Run {
    public static void main(String[] args) {
        var obj = new Chi.InChi().get();
        System.out.println(((Object) obj).getClass().getName());
    }
}
