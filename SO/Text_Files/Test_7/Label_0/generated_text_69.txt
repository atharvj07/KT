import java.util.*;
import java.util.stream.*;

class Rec {
    String wk, src, dst;
    int qty;

    Rec(String wk, String src, String dst, int qty) {
        this.wk = wk;
        this.src = src;
        this.dst = dst;
        this.qty = qty;
    }
}

public class Calc {
    public static void main(String[] args) {
        List<Rec> recs = Arrays.asList(
            new Rec("week1", "source1", "destination1", 100),
            new Rec("week1", "source1", "destination2", 200),
            new Rec("week1", "source2", "destination1", 200),
            new Rec("week1", "source2", "destination2", 100),
            new Rec("week2", "source1", "destination1", 200),
            new Rec("week2", "source1", "destination2", 200)
        );

        recs.stream()
            .collect(Collectors.groupingBy(r -> r.wk + r.src, 
                Collectors.collectingAndThen(
                    Collectors.toList(), lst -> {
                        int ttl = lst.stream().mapToInt(r -> r.qty).sum();
                        return lst.stream()
                            .map(r -> String.format("%s  %s  %s   %.2f%%", r.wk, r.src, r.dst, (r.qty * 100.0) / ttl))
                            .collect(Collectors.toList());
                    }
                )
            ))
            .values()
