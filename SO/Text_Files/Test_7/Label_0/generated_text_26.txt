import java.time.*;
import java.time.format.*;
import java.time.temporal.*;

public record Wk(int yr, int wk) implements Comparable<Wk> {

    private static final DateTimeFormatter fmt = DateTimeFormatter.ofPattern("YYYY-'W'ww");

    public static Wk of(String str) {
        TemporalAccessor t = fmt.parse(str);
        return new Wk(t.get(ChronoField.YEAR), t.get(ChronoField.ALIGNED_WEEK_OF_YEAR));
    }

    public boolean isBfr(Wk o) {
        return this.compareTo(o) < 0;
    }

    public int wksBtw(Wk o) {
        return Period.between(LocalDate.ofYearDay(yr, 1).with(WeekFields.ISO.weekOfYear(), wk),
                LocalDate.ofYearDay(o.yr, 1).with(WeekFields.ISO.weekOfYear(), o.wk)).getDays() / 7;
    }

    @Override
    public int compareTo(Wk o) {
        return yr != o.yr ? Integer.compare(yr, o.yr) : Integer.compare(wk, o.wk);
    }

    @Override
    public String toString() {
        return fmt.format(LocalDate.ofYearDay(yr, 1).with(WeekFields.ISO.weekOfYear(), wk));
    }
}
