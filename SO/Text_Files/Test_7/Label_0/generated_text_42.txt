import javafx.application.*;
import javafx.scene.*;
import javafx.scene.control.*;
import javafx.scene.layout.*;
import javafx.stage.*;

public class MyGridPanes extends Application {
    @Override
    public void start(Stage stg) {
        var col = new ColumnConstraints();
        col.setMinWidth(Region.USE_PREF_SIZE);
        col.setHgrow(Priority.ALWAYS);

        var g1 = mkGp("One two three", col);
        var g2 = mkGp("Four five six seven", col);

        var t1 = new TitledPane("Top", g1);
        var t2 = new TitledPane("Bottom", g2);
        t1.setCollapsible(false);
        t2.setCollapsible(false);

        var scn = new Scene(new VBox(t1, t2), 400, 200);
        stg.setScene(scn);
        stg.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
