import java.util.*;

class Edg {
    int v, w;
    Edg(int v, int w) {
        this.v = v;
        this.w = w;
    }
}

class Sol {
    static int sol(List<List<Edg>> T, long L) {
        int[] res = {1000000000};
        for (int i = 0; i < T.size(); i++) 
            dfs(T, L, i, -1, 0, 0, new HashMap<>(), res);
        return res[0] == 1000000000 ? -1 : res[0];
    }

    static void dfs(List<List<Edg>> T, long L, int v, int p, long len, int e, Map<Long, Integer> pre, int[] res) {
        if (pre.containsKey(len - L)) res[0] = Math.min(res[0], e - pre.get(len - L));
        pre.put(len, e);
        for (Edg ed : T.get(v)) 
            if (ed.v != p) dfs(T, L, ed.v, v, len + ed.w, e + 1, pre, res);
        pre.remove(len);
    }
}

public class Main {
    static void put(List<List<Edg>> T, int a, int b, int w) {
        T.get(a).add(new Edg(b, w));
        T.get(b).add(new Edg(a, w));
    }
    public static void main(String[] args) {
        List<List<Edg>> T = new ArrayList<>();
        int N = 8;
        for (int i = 0; i < N; i++) T.add(new ArrayList<>());
        put(T, 0, 1, 2);
        put(T, 1, 2, 1);
        put(T, 1, 3, 2);
        put(T, 2, 6, 1);
        put(T, 6, 7, 1);
        put(T, 3, 4, 1);
        put(T, 3, 5, 4);
        System.out.println(Sol.sol(T, 5L));
    }
}
