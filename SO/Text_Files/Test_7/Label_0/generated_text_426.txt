import polars as pl

df = pl.DataFrame({
    "a_0": [1, 4],
    "a_1": [2, 5],
    "a_2": [3, 6],
    "b_0": [7, 10],
    "b_1": [8, 11],
    "b_2": [9, 12]
})

df = df.melt(id_vars=[], variable_name="x").with_columns(
    pl.col("x").str.extract(r"(\D)").alias("g")
).pivot(values="value", index="g", columns="x").drop("g")

print(df)
