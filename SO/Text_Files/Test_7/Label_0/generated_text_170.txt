import java.lang.management.ManagementFactory;
import java.nio.file.*;

public class App {
    public static void main(String[] args) throws Exception {
        var mx = ManagementFactory.getRuntimeMXBean();
        var pid = mx.getName().split("@")[0];
        var path = Paths.get("path/to/file");
        var w = path.toFile().toPath();
        var ref = new Object() { String usr = ""; };

        Files.lines(Paths.get("/proc/" + pid + "/fd"))
            .filter(l -> l.contains(w.toString()))
            .forEach(l -> ref.usr = l);

        System.out.println(ref.usr);
    }
}
