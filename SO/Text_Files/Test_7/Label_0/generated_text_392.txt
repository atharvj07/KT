from typing import Callable, TypeVar, Dict
from functools import wraps

T = TypeVar("T", bound=Callable[[int], int])

def remember(f: T) -> T:
    mem: Dict[int, int] = {}

    @wraps(f)
    def wrap(n: int) -> int:
        if n not in mem:
            mem[n] = f(n)
        return mem[n]

    return wrap  # type: ignore

@remember
def fib(n: int) -> int:
    if n < 2:
        return 1
    return fib(n - 2) + fib(n - 1)
