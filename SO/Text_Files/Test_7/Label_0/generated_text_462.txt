import pydantic
from typing import Any
import random
import string

class Ex(pydantic.BaseModel):
    a: int
    b: str = pydantic.Field(min_length=10, max_length=10)
    
    @staticmethod
    def ex() -> dict[str, Any]:
        return {
            'a': random.randint(1, 100),
            'b': ''.join(random.choices(string.digits, k=10)),
        }

print(Ex.ex())
