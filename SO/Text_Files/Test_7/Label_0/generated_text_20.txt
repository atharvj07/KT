import java.util.*;

public class Esc {
    private static final Map<Character, String> map = new HashMap<>();

    static {
        map.put('<', "&lt;");
        map.put('>', "&gt;");
        map.put('&', "&amp;");
        map.put('\'', "&#039;");
        map.put('"', "&#034;");
    }

    public static String esc(String inp) {
        StringBuilder sb = new StringBuilder();
        for (char c : inp.toCharArray()) {
            sb.append(map.getOrDefault(c, String.valueOf(c)));
        }
        return sb.toString();
    }
}
