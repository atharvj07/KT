public class AlrmRcvr extends BroadcastReceiver {

    @Override
    public void onReceive(Context ctx, Intent it) {

        Vibrator vib = (Vibrator) ctx.getSystemService(Context.VIBRATOR_SERVICE);
        if (vib != null && vib.hasVibrator()) {
            vib.vibrate(VibrationEffect.createOneShot(500, VibrationEffect.DEFAULT_AMPLITUDE));
        }

        String ttl = it.getStringExtra("title");
        String cmt = it.getStringExtra("comment");
        Intent actIt = new Intent(ctx, MainActivity.class);
        actIt.setFlags(Intent.FLAG_ACTIVITY_NEW_TASK | Intent.FLAG_ACTIVITY_CLEAR_TASK);
        actIt.putExtra("stop", true);

        PendingIntent penIt = PendingIntent.getActivity(ctx, 0, actIt, PendingIntent.FLAG_UPDATE_CURRENT);

        String txt = "Reminder for: " + ttl + "\n" + "Comments: " + cmt;
        NotificationCompat.Builder bld = new NotificationCompat.Builder(ctx, "myandroid")
                .setSmallIcon(R.drawable.alarm)
                .setContentTitle(ttl)
                .setContentText(txt)
                .setAutoCancel(true)
                .setStyle(new NotificationCompat.BigTextStyle().bigText(txt))
                .setContentIntent(penIt)
                .setDeleteIntent(penIt)
                .setDefaults(NotificationCompat.DEFAULT_ALL)
                .setPriority(NotificationCompat.PRIORITY_HIGH);

        NotificationManagerCompat mgr = NotificationManagerCompat.from(ctx);
        mgr.notify(123, bld.build());

        Uri uri = RingtoneManager.getDefaultUri(RingtoneManager.TYPE_ALARM);
        Ringtone rng = RingtoneManager.getRingtone(ctx, uri);
        rng.play();

        AlrmSrv.rng = rng;
    }
}
