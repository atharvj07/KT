import java.time.*;
import java.time.format.*;

class App {
    public static void main(String[] a) {
        String d = "2022-03-12T03:59:59+0000Z";
        DateTimeFormatter f = DateTimeFormatter.ofPattern("yyyy-MM-dd'T'HH:mm:ssXX'Z'");
        ZonedDateTime z = ZonedDateTime.parse(d, f);
        System.out.println(z.toEpochSecond());
    }
}
