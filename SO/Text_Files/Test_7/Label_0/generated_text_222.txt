public class MyFrag extends Fragment {
    LinearLayout out;
    Button btn;
    int id = 0;

    public View onCreateView(LayoutInflater inf, ViewGroup grp, Bundle b) {
        View v = inf.inflate(R.layout.my_frag, grp, false);
        out = v.findViewById(R.id.ll_out);
        btn = v.findViewById(R.id.btn_prd);
        btn.setOnClickListener(x -> addRow());
        return v;
    }

    void addRow() {
        LinearLayout row = new LinearLayout(getActivity());
        row.setLayoutParams(new LinearLayout.LayoutParams(
                ViewGroup.LayoutParams.MATCH_PARENT,
                ViewGroup.LayoutParams.WRAP_CONTENT));
        row.setOrientation(LinearLayout.HORIZONTAL);
        row.setId(id++);

        EditText nm = new EditText(getActivity());
        nm.setLayoutParams(new LinearLayout.LayoutParams(0, ViewGroup.LayoutParams.WRAP_CONTENT, 1));
        nm.setHint("Name");

        EditText pr = new EditText(getActivity());
        pr.setLayoutParams(new LinearLayout.LayoutParams(0, ViewGroup.LayoutParams.WRAP_CONTENT, 1));
        pr.setHint("Price");

        EditText sp = new EditText(getActivity());
        sp.setLayoutParams(new LinearLayout.LayoutParams(0, ViewGroup.LayoutParams.WRAP_CONTENT, 1));
        sp.setHint("Specs");

        Button del = new Button(getActivity());
        del.setText("X");
        del.setOnClickListener(x -> out.removeView(row));

        row.addView(nm);
        row.addView(pr);
        row.addView(sp);
        row.addView(del);
        out.addView(row);
    }
}
