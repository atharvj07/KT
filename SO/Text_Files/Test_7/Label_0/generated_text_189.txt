public abstract class BPld {
}

public class APld extends BPld {
    private int id;
    private String nm;
    public APld(int id, String nm) {
        this.id = id;
        this.nm = nm;
    }
    public int getId() {
        return id;
    }
    public String getNm() {
        return nm;
    }
}

public class BCmd<T extends BPld> {
    String cmd;
    T pld;
    public BCmd(String cmd, T pld) {
        this.cmd = cmd;
        this.pld = pld;
    }
}

public class ACmd extends BCmd<APld> {
    public ACmd(APld pld) {
        super("Add", pld);
    }
}

public abstract class BEvt<C extends BCmd<P>, P extends BPld> {
    abstract protected BEH<? extends BEvt<C, P>, C, P> getHnd();
}

public abstract class BEH<E extends BEvt<C, P>, C extends BCmd<P>, P extends BPld> {
    abstract public void onTry(E e, ArrayList<BEvt<?, ?>> ev);
    abstract public void onCom(String uid, E e);
}

public class AEH extends BEH<AEvt, ACmd, APld> {
    @Override
    public void onTry(AEvt e, ArrayList<BEvt<?, ?>> ev) {
    }
    @Override
    public void onCom(String uid, AEvt e) {
    }
}

public class AEvt extends BEvt<ACmd, APld> {
    @Override
    protected BEH<AEvt, ACmd, APld> getHnd() {
        return new AEH();
    }
}
