package gd.rf.anuragsaini.stereotype;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

@Component
public class Co {

    private String n;
    private String m;

    public Co(@Value("Adventure of War") String n, @Value("A place for War") String m) {
        this.n = n;
        this.m = m;
    }

    @Override
    public String toString() {
        return "Co{" + "n='" + n + '\'' + ", m='" + m + '\'' + '}';
    }
}
