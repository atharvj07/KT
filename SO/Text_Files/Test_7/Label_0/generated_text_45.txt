import java.util.concurrent.*;

class Box {
    private volatile int val = 0;
    private final ConcurrentHashMap<String, Integer> map = new ConcurrentHashMap<>();

    int get(String key) {
        return map.computeIfAbsent(key, k -> upd());
    }

    private synchronized int upd() {
        return ++val;
    }

    public static void main(String[] args) {
        Box box = new Box();
        ExecutorService ex = Executors.newFixedThreadPool(2);

        Runnable r1 = () -> System.out.println(box.get("a"));
        Runnable r2 = () -> System.out.println(box.get("b"));

        ex.execute(r1);
        ex.execute(r2);
