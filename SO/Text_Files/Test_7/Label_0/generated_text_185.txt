import java.util.regex.*;

public class RegExParse {
    private void parse(String p) {
        Pattern rgx = Pattern.compile("abfss?://([^/@]+)@([^:/]+)(?::(\\d+))?/(?:([^/]+))?/(.*)");
        Matcher mtc = rgx.matcher(p);
        if (mtc.find()) {
            String fs = mtc.group(1);
            String acc = mtc.group(4) != null ? mtc.group(4) : mtc.group(2);
            String suf = mtc.group(3) != null ? ":" + mtc.group(3) : "";
            String rel = mtc.group(5);
            System.out.println(fs + " | " + acc + " | " + suf + " | " + rel);
        }
    }
