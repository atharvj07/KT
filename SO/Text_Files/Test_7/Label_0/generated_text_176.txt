interface S1 {
    default void dsp() {
        System.out.println("hello");
    }
}

interface S2 {
    void dsp();
}

public class Per implements S1, S2 {
    public void dsp() {
        S1.super.dsp();
    }

    public static void main(String[] a) {
        Per p = new Per();
        p.dsp();
    }
}
