import org.springframework.context.annotation.*;
import org.springframework.http.codec.*;

@Configuration
public class XmlCfg {

    @Bean
    public HttpMsgCod xmlCod() {
        return new Jaxb2XmlDecoder();
    }

    @Bean
    public HttpMsgCod xmlEnc() {
        return new Jaxb2XmlEncoder();
    }
}
