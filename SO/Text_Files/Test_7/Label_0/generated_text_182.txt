import org.springframework.stereotype.Service;
import org.springframework.transaction.reactive.*;
import reactor.core.publisher.*;

@Service
public class ChSvc {
    private final ChRep rep;
    private final RxtxMgr txMgr;

    public ChSvc(ChRep rep, RxtxMgr txMgr) {
        this.rep = rep;
        this.txMgr = txMgr;
    }

    public Mono<Ch> crt(Ch ch) {
        return txMgr.getRxtx()
            .execute(tx -> rep.sav(ch)
                .map(c -> {
                    if (true) throw new RuntimeException();
                    return c;
                })
            );
    }
}
