import geopandas as gpd
import requests
import zipfile
import io

def download_map(url, extract_to):
    r = requests.get(url)
    with zipfile.ZipFile(io.BytesIO(r.content)) as zip_ref:
        zip_ref.extractall(extract_to)

def load_world_map():
    url = "https://www.naturalearthdata.com/http//www.naturalearthdata.com/downloads/110m-cultural-vectors/countries-110m.zip"
    extract_to = "./data"
    download_map(url, extract_to)
    world = gpd.read_file(f"{extract_to}/ne_110m_admin_0_countries.shp")
    return world

world = load_world_map()
print(world.head())
