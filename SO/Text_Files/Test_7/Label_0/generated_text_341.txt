import polars as pl

df = pl.DataFrame(
    {
        "id": ["NY", "TK", "FD"], 
        "eat2000": [1, None, 3], 
        "eat2001": [-2, None, 4],
        "eat2002": [None, None, None],
        "eat2003": [-9, None, 8],
        "eat2004": [None, None, 8]
    }
)

col_list = [word for word in df.columns if word.startswith("eat")]
df = df.with_columns(
    [pl.col(col).filter(pl.col(col).is_not_null()) for col in col_list]
)

df
