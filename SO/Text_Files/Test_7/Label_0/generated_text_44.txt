import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.VBox;
import javafx.stage.Popup;
import javafx.stage.Stage;

public class UpMenu extends Application {
    @Override
    public void start(Stage stg) {
        MenuBar bar = new MenuBar();
        Menu itm = new Menu("Opt");
        MenuItem sub = new MenuItem("Sel");
        itm.getItems().add(sub);
        bar.getMenus().add(itm);

        Popup pop = new Popup();
        VBox box = new VBox(sub);
        box.setStyle("-fx-background-color: white; -fx-border-color: black;");
        pop.getContent().add(box);

        itm.setOnShowing(e -> {
            pop.show(bar, bar.getLayoutX(), bar.getLayoutY() - box.getHeight());
        });

        itm.setOnHidden(e -> pop.hide());

        VBox root = new VBox();
        root.getChildren().add(bar);
        root.setStyle("-fx-background-color: gray;");
        root.setTranslateY(300);

        Scene scn = new Scene(root, 400, 400);
        stg.setScene(scn);
        stg.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
