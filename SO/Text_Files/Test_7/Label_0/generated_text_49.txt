import javafx.scene.control.TextField;
import javafx.scene.layout.GridPane;

public class Gd extends GridPane {
    public Gd(int c, int r, int hg, int vg, int fh, int fv) {
        setHgap(hg);
        setVgap(vg);
        for (int y = 0; y < r; y++) {
            for (int x = 0; x < c; x++) {
                var f = new TextField();
                f.setMinSize(fh, fv);
                f.setMaxSize(fh, fv);
                add(f, x, y);
            }
        }
    }
}
