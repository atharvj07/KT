import javafx.application.*;
import javafx.scene.*;
import javafx.scene.paint.*;
import javafx.scene.shape.*;
import javafx.scene.control.*;
import javafx.stage.*;

public class Drw3D extends Application {
    private MeshView msh;
    private PhongMaterial mat1 = new PhongMaterial(Color.RED);
    private PhongMaterial mat2 = new PhongMaterial(Color.BLUE);

    public void start(Stage stg) {
        msh = new MeshView(new Sphere(50).getMesh());
        msh.setMaterial(mat1);

        Button btn = new Button("Tog");
        btn.setOnAction(e -> msh.setMaterial(msh.getMaterial() == mat1 ? mat2 : mat1));

        Group grp = new Group(msh);
        Scene scn = new Scene(new Group(grp, btn), 400, 400, true);
        scn.setFill(Color.GRAY);
        scn.setCamera(new PerspectiveCamera());

        stg.setScene(scn);
        stg.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
