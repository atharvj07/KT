abstract class Sup {
    private final Object fld;
    Sup(Object f) {
        this.fld = f;
    }
    public Object get() {
        return fld;
    }
}

public class Sub extends Sup {
    private final Object subFld;
    public Sub(Sub c) {
        super(chk(c).get());
        this.subFld = c.subFld;
    }
    private static Sub chk(Sub c) {
        if (c == null) throw new IllegalArgumentException();
        return c;
    }
}
