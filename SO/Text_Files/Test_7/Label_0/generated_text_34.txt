public class FR_Options extends Fragment implements View.OnClickListener {
    public static final String DE = "de";
    public static final String EN = "en";
    public static String curLng = EN;
    private FragmentOptionsBinding bind;

    public View onCreateView(LayoutInflater inf, ViewGroup con, Bundle s) {
        bind = FragmentOptionsBinding.inflate(inf, con, false);
        return bind.getRoot();
    }

    public void onViewCreated(@NonNull View v, @Nullable Bundle s) {
        super.onViewCreated(v, s);
        bind.imageButtonGermany.setOnClickListener(this);
        bind.imageButtonUK.setOnClickListener(this);
        if (curLng.equals(EN)) {
            bind.textViewCurrentLanguageValue.setText(EN);
            bind.imageButtonGermany.setAlpha(0.5f);
            bind.imageButtonUK.setAlpha(1.0f);
        } else {
            bind.textViewCurrentLanguageValue.setText(DE);
            bind.imageButtonGermany.setAlpha(1.0f);
            bind.imageButtonUK.setAlpha(0.5f);
        }
    }

    public void onClick(View v) {
        if (v.getId() == R.id.imageButtonGermany) curLng = DE;
        if (v.getId() == R.id.imageButtonUK) curLng = EN;
        Lng.setLng(requireContext(), curLng);
        requireActivity().recreate();
    }

    public void onDestroyView() {
        super.onDestroyView();
        bind = null;
    }
}
