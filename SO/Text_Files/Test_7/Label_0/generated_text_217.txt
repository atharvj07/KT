import javafx.application.*;
import javafx.geometry.*;
import javafx.scene.*;
import javafx.scene.image.*;
import javafx.scene.layout.*;
import javafx.stage.*;

public class XApp extends Application {
    private Stage stg;
    private ImageView img;

    private void cfg() {
        Screen scr = Screen.getPrimary();
        Rectangle2D bnd = scr.getBounds();
        stg.setWidth(bnd.getWidth());
        stg.setHeight(bnd.getHeight());
        Scene scn = new Scene(new StackPane(img));
        scn.setFill(null);
        stg.setScene(scn);
        stg.setAlwaysOnTop(true);
        stg.initStyle(StageStyle.TRANSPARENT);
        stg.setX(bnd.getMinX() + bnd.getWidth() / 2 - stg.getWidth() / 2);
        stg.setY(bnd.getMinY() + bnd.getHeight() / 2 - stg.getHeight() / 2);
    }

    @Override
    public void start(Stage ps) {
        stg = new Stage();
        img = new ImageView("crosshair.png");
        img.setPickOnBounds(false);
        img.setMouseTransparent(true);
        cfg();
        stg.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
