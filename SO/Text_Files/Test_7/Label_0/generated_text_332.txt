from datetime import date
from pydantic import BaseModel, field_validator

class CustomDate(date):
    POTENTIAL_FORMATS = []

    @classmethod
    def validate(cls, v, field):
        if isinstance(v, date):
            return v
        return to_date(field.name, v, cls.POTENTIAL_FORMATS, return_str=False)

    @classmethod
    def __get_pydantic_core_schema__(cls, source, handler):
        return handler.general_plain_validator_function(cls.validate)


def custom_date(formats):
    def validate_date(v, field):
        if isinstance(v, date):
            return v
        return to_date(field.name, v, formats, return_str=False)
    return field_validator(validate_date)

class ExampleModel(BaseModel):
    dt: CustomDate
    dt2: custom_date(['%Y-%m-%d', '%Y/%m/%d'])
