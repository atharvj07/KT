private <T> void pos(Class<T> cls) {
    int x, y;
    boolean crt = false;
    while (!crt) {
        x = rnd.nextInt(sx);
        y = rnd.nextInt(sy);
        if (fld.getFld(x, y) instanceof Emp) {
            try {
                T obj = cls.getConstructor(int.class, int.class, int.class, int.class, Fld.class, Ply.class, Obj.class)
                          .newInstance(x, y, sx, sy, fld, ply, this);
                fld.setFld(x, y, obj);
                crt = true;
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }
    }
}
