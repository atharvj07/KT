import pandas as pd
import numpy as np

df = pd.DataFrame(
    {
        'x': [1, np.nan, 3, np.nan, 5],
        'y': [np.nan, 7, 8, 9, np.nan],
        'x_a': [1, 2, 3, 4, 5],
        'y_a': [6, 7, 8, 9, 10]
    }
)

def fill_na(df):
    cols = [col for col in df.columns if not col.endswith('_a')]
    for col in cols:
        df[col] = df[col].fillna(df[f'{col}_a'])
    return df

df = fill_na(df)
print(df)
