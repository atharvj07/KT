 class JaxbCfg {
    @Bean
    public Jaxb2Marshaller mar() {
        Jaxb2Marshaller m = new Jaxb2Marshaller();
        m.setContextPath("com.company.project.xsd.some_portal.v4_0");
        m.setMtomEnabled(false);
        return m;
    }
}

@Component
public class MySoap extends WebServiceGatewaySupport {
    private final WebServiceTemplate tmp;

    public MySoap(Jaxb2Marshaller mar) {
        setMarshaller(mar);
        setUnmarshaller(mar);
        tmp = getWebServiceTemplate();
    }

    public void send(MyReq req) {
        try {
            tmp.marshalSendAndReceive(req);
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }
}

public class Bin {
    public static BinDat mkBin(byte[] dat) {
        BinDat b = new BinDat();
        b.setBlob(new String(Base64.getEncoder().encode(dat)));
        b.setExt("pdf");
        return b;
    }
}

@XmlAccessorType(XmlAccessType.FIELD)
@XmlType(name = "BinDat", propOrder = {"blob", "ext"})
public class BinDat {
    @XmlElement(required = true)
    protected String blob;
    @XmlElement(required = true)
    protected String ext;

    public void setBlob(String b) { this.blob = b; }
    public void setExt(String e) { this.ext = e; }
}
